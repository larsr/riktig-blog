Record BilinearMap (F U V W : Type) `{FF : Field.Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
    {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}
    {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
    {VW : VectorSpace F W} := {
  bm_fun :> U → V → W;
  #[global] bilinear_map_proper :: Proper (equ ==> equ ==> equ) bm_fun;
  bilinear_left_additive : ∀ (u u' : U) (v : V),
    bm_fun (u + u') v == bm_fun u v + bm_fun u' v;
  bilinear_left_scale : ∀ (a : F) (u : U) (v : V),
    bm_fun (a *: u) v == a *: bm_fun u v;
  bilinear_right_additive : ∀ (u : U) (v v' : V),
    bm_fun u (v + v') == bm_fun u v + bm_fun u v';
  bilinear_right_scale : ∀ (a : F) (u : U) (v : V),
    bm_fun u (a *: v) == a *: bm_fun u v
}.

Definition famEq {V : Type} {EV : EqRel V} {n : nat}
    (vs ws : Fin.t n → V) : Prop :=
  ∀ i : Fin.t n, vs i == ws i.

Definition upd {V : Type} {n : nat}
    (vs : Fin.t n → V) (i : Fin.t n) (x : V) : Fin.t n → V :=
  fun j ⇒ if dec (i = j) then x else vs j.


Record MultilinearForm (n : nat) (F V : Type) `{FF : Field.Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V} := {
  mf_fun :> (Fin.t n → V) → F;
  #[global] multilinear_form_proper :: Proper (@famEq V _ n ==> equ) mf_fun;
  multilinear_form_additive : ∀ (vs : Fin.t n → V) (i : Fin.t n) (x y : V),
    mf_fun (upd vs i (x + y)) == mf_fun (upd vs i x) + mf_fun (upd vs i y);
  multilinear_form_scale : ∀ (vs : Fin.t n → V) (i : Fin.t n) (a : F) (x : V),
    mf_fun (upd vs i (a *: x)) == a × mf_fun (upd vs i x)
}.

Arguments multilinear_form_proper {n F V Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.
Arguments multilinear_form_additive {n F V Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.
Arguments multilinear_form_scale {n F V Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.


Record Alternating {n : nat} {F V : Type} `{FF : Field.Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V} (m : MultilinearForm n F V) := {
  alternating_repeat_zero : ∀ (vs : Fin.t n → V) (i j : Fin.t n),
    i ≠ j → vs i == vs j → m vs == 0
}.

Arguments alternating_repeat_zero {n F V Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV m}.
