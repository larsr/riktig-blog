Module Bilinear.
  Class Bilinear (F U V W : Type)
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
      {VU : VectorSpace F U}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
      {VW : VectorSpace F W}
      (b : U → V → W) := {
    proper :: Proper (equ ==> equ ==> equ) b;

    linear_l :: ∀ v : V, Linear F U W (fun u ⇒ b u v);
    linear_r :: ∀ u : U, Linear F V W (b u);
  }.
  Hint Mode Bilinear - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - ! : typeclass_instances.
End Bilinear.

Record BilinearMap (F U V W : Type)
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
    {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}
    {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
    {VW : VectorSpace F W} := {
  bm_fun :> U → V → W;
  bm_bilinear :: Bilinear F U V W bm_fun;
}.

Arguments bm_fun {F U V W Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV eW zW aW nW sW VW}.
Arguments bm_bilinear {F U V W Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV eW zW aW nW sW VW}.

Section BilinearMapLaws.
  Context {F U V W : Type}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
    {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}
    {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
    {VW : VectorSpace F W}.

  Definition bilinear_map_proper (b : BilinearMap F U V W)
    : Proper (equ ==> equ ==> equ) (bm_fun b) := Bilinear.proper (b := bm_fun b).

  Definition bilinear_left_additive (b : BilinearMap F U V W)
    : ∀ (u u' : U) (v : V), b (u + u') v == b u v + b u' v :=
    fun u u' v ⇒ additive (f := fun x ⇒ bm_fun b x v) u u'.

  Definition bilinear_left_scale (b : BilinearMap F U V W)
    : ∀ (a : F) (u : U) (v : V), b (a *: u) v == a *: b u v :=
    fun a u v ⇒ homogeneous (f := fun x ⇒ bm_fun b x v) a u.

  Definition bilinear_right_additive (b : BilinearMap F U V W)
    : ∀ (u : U) (v v' : V), b u (v + v') == b u v + b u v' :=
    fun u v v' ⇒ additive (f := bm_fun b u) v v'.

  Definition bilinear_right_scale (b : BilinearMap F U V W)
    : ∀ (a : F) (u : U) (v : V), b u (a *: v) == a *: b u v :=
    fun a u v ⇒ homogeneous (f := bm_fun b u) a v.

  Definition bilinear_map_from_laws (b : U → V → W)
      (b_proper : Proper (equ ==> equ ==> equ) b)
      (b_left_additive : ∀ (u u' : U) (v : V), b (u + u') v == b u v + b u' v)
      (b_left_scale : ∀ (a : F) (u : U) (v : V), b (a *: u) v == a *: b u v)
      (b_right_additive : ∀ (u : U) (v v' : V), b u (v + v') == b u v + b u v')
      (b_right_scale : ∀ (a : F) (u : U) (v : V), b u (a *: v) == a *: b u v)
    : BilinearMap F U V W :=
    {| bm_fun := b;
       bm_bilinear :=
         {| Bilinear.proper := b_proper;
            Bilinear.linear_l := fun v ⇒
              {| Linear.proper := fun x x' Hx ⇒ b_proper x x' Hx v v (reflexivity v);
                 Linear.is_additive := fun x x' ⇒ b_left_additive x x' v;
                 Linear.is_homogeneous := fun a x ⇒ b_left_scale a x v |};
            Bilinear.linear_r := fun u ⇒
              {| Linear.proper := fun y y' Hy ⇒ b_proper u u (reflexivity u) y y' Hy;
                 Linear.is_additive := fun y y' ⇒ b_right_additive u y y';
                 Linear.is_homogeneous := fun a y ⇒ b_right_scale a u y |} |} |}.
End BilinearMapLaws.

Definition famEq {V : Type} {EV : EqRel V} {n : nat}
    (vs ws : Fin.t n → V) : Prop :=
  ∀ i : Fin.t n, vs i == ws i.

Definition upd {V : Type} {n : nat}
    (vs : Fin.t n → V) (i : Fin.t n) (x : V) : Fin.t n → V :=
  fun j ⇒ if dec (i = j) then x else vs j.

Module Multilinear.
  Class Multilinear (n : nat) (F V : Type)
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      (form : (Fin.t n → V) → F) := {
    proper :: Proper (@famEq V eV n ==> equ) form;

    is_additive :: ∀ (vs : Fin.t n → V) (i : Fin.t n),
      Additive V F add add (fun x ⇒ form (upd vs i x));
    is_homogeneous :: ∀ (vs : Fin.t n → V) (i : Fin.t n),
      Homogeneous F V F mult (fun a ⇒ a) (fun x ⇒ form (upd vs i x));
  }.
  Hint Mode Multilinear - - - - - - - - - - - - - - - - - ! : typeclass_instances.
End Multilinear.
Export Multilinear.

Record MultilinearForm (n : nat) (F V : Type)
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V} := {
  mf_fun :> (Fin.t n → V) → F;
  mf_multilinear :: Multilinear n F V mf_fun;
}.

Arguments mf_fun {n F V Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.
Arguments mf_multilinear {n F V Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.

Section MultilinearFormLaws.
  Context {n : nat} {F V : Type}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}.

  Definition multilinear_form_proper (m : MultilinearForm n F V)
    : Proper (@famEq V eV n ==> equ) (mf_fun m) :=
    Multilinear.proper (form := mf_fun m).

  Definition multilinear_form_additive (m : MultilinearForm n F V)
    : ∀ (vs : Fin.t n → V) (i : Fin.t n) (x y : V),
        m (upd vs i (x + y)) == m (upd vs i x) + m (upd vs i y) :=
    fun vs i ⇒ additive (f := fun x ⇒ mf_fun m (upd vs i x)).

  Definition multilinear_form_scale (m : MultilinearForm n F V)
    : ∀ (vs : Fin.t n → V) (i : Fin.t n) (a : F) (x : V),
        m (upd vs i (a *: x)) == a × m (upd vs i x) :=
    fun vs i ⇒ homogeneous (f := fun x ⇒ mf_fun m (upd vs i x)).

  Definition multilinear_form_from_laws (form : (Fin.t n → V) → F)
      (form_proper : Proper (@famEq V eV n ==> equ) form)
      (form_additive : ∀ (vs : Fin.t n → V) (i : Fin.t n) (x y : V),
         form (upd vs i (x + y)) == form (upd vs i x) + form (upd vs i y))
      (form_scale : ∀ (vs : Fin.t n → V) (i : Fin.t n) (a : F) (x : V),
         form (upd vs i (a *: x)) == a × form (upd vs i x))
    : MultilinearForm n F V :=
    {| mf_fun := form;
       mf_multilinear := {| Multilinear.proper := form_proper;
                            Multilinear.is_additive := form_additive;
                            Multilinear.is_homogeneous := form_scale |} |}.
End MultilinearFormLaws.

Module Alternating.
  Class Alternating {n : nat} {F V : Type}
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      (form : (Fin.t n → V) → F) :=
    repeat_zero : ∀ (vs : Fin.t n → V) (i j : Fin.t n),
      i ≠ j → vs i == vs j → form vs == 0.
  Hint Mode Alternating - - - - - - - - - - - - - - - - - ! : typeclass_instances.
End Alternating.
Notation Alternating := Alternating.Alternating.

Section AlternatingLaw.
  Context {n : nat} {F V : Type}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}.

  Definition alternating_repeat_zero {m : MultilinearForm n F V}
      (alt : Alternating m)
    : ∀ (vs : Fin.t n → V) (i j : Fin.t n),
        i ≠ j → vs i == vs j → m vs == 0 := alt.

  Definition alternating_from_law (form : (Fin.t n → V) → F)
      (form_repeat_zero : ∀ (vs : Fin.t n → V) (i j : Fin.t n),
         i ≠ j → vs i == vs j → form vs == 0)
    : Alternating form := form_repeat_zero.
End AlternatingLaw.

  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Example bilinear_map_check (b : BilinearMap F U V W)
    : ∀ (u u' : U) (v : V), b (u + u') v == b u v + b u' v :=
    fun u u' v ⇒ additive (f := fun u : U ⇒ b u v) u u'.
End BilinearMapCheck.
