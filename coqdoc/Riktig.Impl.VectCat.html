Record VectObj (F : Type) {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F}
    {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F} {Fi : HasInv F}
    {FF : Field F} := {
  vo_car : Type;
  vo_eq : EqRel vo_car;
  vo_zero : Zero vo_car;
  vo_add : HasAdd vo_car;
  vo_neg : HasNeg vo_car;
  vo_scale : HasScale F vo_car;
  vo_vs : VectorSpace F vo_car (Ve:=vo_eq) (Vz:=vo_zero) (Va:=vo_add) (Vn:=vo_neg) (Vs:=vo_scale)
}.

Arguments vo_car {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_eq {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_zero {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_add {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_neg {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_scale {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_vs {F Fe Fz Fa Fn Fo Fm Fi FF}.

Definition vectObj (F : Type) `{FF : Field F} (T : Type)
  {e : EqRel T} {z : Zero T} {a : HasAdd T} {n : HasNeg T} {sc : HasScale F T}
  {VS : VectorSpace F T} : VectObj F :=
  {| vo_car := T; vo_eq := e; vo_zero := z; vo_add := a; vo_neg := n;
     vo_scale := sc; vo_vs := VS |}.

Section VectCat.
  Context {F : Type} `{FF : Field F}.

  Definition VHom (X Y : VectObj F) : Type :=
    LinearMap F (vo_car X) (vo_car Y).

  Definition vhom_equ (X Y : VectObj F) : EqRel (VHom X Y) :=
    linear_map_equ.

  Lemma vhom_equ_refl (X Y : VectObj F) (f : VHom X Y) : vhom_equ X Y f f.
  Proof.
    intros u.
    reflexivity.
  Qed.

  Lemma vhom_equ_sym (X Y : VectObj F) (f g : VHom X Y) :
    vhom_equ X Y f g → vhom_equ X Y g f.
  Proof.
    intros Hfg u.
    symmetry.
    exact (Hfg u).
  Qed.

  Lemma vhom_equ_trans (X Y : VectObj F) (f g h : VHom X Y) :
    vhom_equ X Y f g → vhom_equ X Y g h → vhom_equ X Y f h.
  Proof.
    intros Hfg Hgh u.
    rewrite (Hfg u).
    exact (Hgh u).
  Qed.

  #[global] Instance vhom_equiv (X Y : VectObj F) :
    Equivalence (vhom_equ X Y).
  Proof.
    apply Build_Equivalence.
    - exact (vhom_equ_refl X Y).
    - exact (vhom_equ_sym X Y).
    - exact (vhom_equ_trans X Y).
  Qed.

  Definition vid (X : VectObj F) : VHom X X := linear_map_id.

  Definition vcomp (X Y Z : VectObj F)
    (g : VHom Y Z) (f : VHom X Y) : VHom X Z := linear_map_comp g f.

  Lemma vcomp_ext (X Y Z : VectObj F)
      (g g' : VHom Y Z) (f f' : VHom X Y) :
    vhom_equ Y Z g g' → vhom_equ X Y f f' →
    ∀ u : vo_car X, g (f u) == g' (f' u).
  Proof.
    intros Hg Hf u.
    rewrite (linear_map_proper g _ _ (Hf u)).
    exact (Hg (f' u)).
  Qed.

  #[global] Instance vcomp_proper (X Y Z : VectObj F) :
    Proper (vhom_equ Y Z ==> vhom_equ X Y ==> vhom_equ X Z)
           (vcomp X Y Z).
  Proof.
    intros g g' Hg f f' Hf u.
    exact (vcomp_ext X Y Z g g' f f' Hg Hf u).
  Qed.

  Lemma vcat_idl (X Y : VectObj F) (f : VHom X Y) :
    vhom_equ X Y (vcomp X Y Y (vid Y) f) f.
  Proof.
    intros u.
    reflexivity.
  Qed.

  Lemma vcat_idr (X Y : VectObj F) (f : VHom X Y) :
    vhom_equ X Y (vcomp X X Y f (vid X)) f.
  Proof.
    intros u.
    reflexivity.
  Qed.

  Lemma vcat_assoc (W X Y Z : VectObj F)
      (f : VHom Y Z) (g : VHom X Y) (h : VHom W X) :
    vhom_equ W Z (vcomp W Y Z f (vcomp W X Y g h))
                 (vcomp W X Z (vcomp X Y Z f g) h).
  Proof.
    intros u.
    reflexivity.
  Qed.

  #[global] Instance Vect_Category : Category (VectObj F) := {|
    Category.hom := VHom;
    Category.hom_equ := vhom_equ;
    Category.hom_equiv := vhom_equiv;
    Category.id := vid;
    Category.comp := vcomp;
    Category.comp_proper := vcomp_proper;
    Category.is_left_identity := vcat_idl;
    Category.is_right_identity := vcat_idr;
    Category.comp_associative := vcat_assoc
  |}.

End VectCat.
