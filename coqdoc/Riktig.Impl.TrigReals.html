  Notation natR := (from_nat (F := R)).


sin_ratio is the factor the terms of sin_term 1 shrink by: a quarter, which the two reciprocals of the two-step recursion undercut already at the first step.
  Definition sin_ratio : R := half (1 : R) × half (1 : R).

  Lemma sin_ratio_nonneg : 0 ≤ sin_ratio.
  Proof.
    exact (mul_nonneg_closed (half 1) (half 1)
             (lt_le 0 (half 1) (half_pos 1 lt_0_1))
             (lt_le 0 (half 1) (half_pos 1 lt_0_1))).
  Qed.

  Lemma sin_ratio_double_lt_one : sin_ratio + sin_ratio < 1.
  Proof.
    apply (le_lt_trans (sin_ratio + sin_ratio) (half (1 : R)) 1).
    - apply eq_le.
      calc (half (1 : R) × half (1 : R) + half (1 : R) × half (1 : R))
        == ((1 : R) × half (1 : R)) { (half_mul_sum 1 (half 1)) }
        == (half (1 : R)) { left_identity }.
    - exact (half_lt 1 lt_0_1).
  Qed.

  Lemma sin_ratio_lt_one : sin_ratio < 1.
  Proof.
    exact (le_lt_trans sin_ratio (sin_ratio + sin_ratio) 1
             (le_add_nonneg_r sin_ratio sin_ratio sin_ratio_nonneg)
             sin_ratio_double_lt_one).
  Qed.

  Lemma sin_term_one_succ : ∀ j : nat,
    sin_term (1 : R) (S j)
      == - (inv (natR (S (S (2 × j)))) × inv (natR (S (S (S (2 × j)))))
              × sin_term (1 : R) j).
  Proof.
    intros j.
    assert (Hdef : sin_term (1 : R) (S j)
                   == - (inv (natR (S (S (2 × j))))
                           × inv (natR (S (S (S (2 × j)))))
                           × ((1 : R) × (1 : R)) × sin_term (1 : R) j))
      by reflexivity.
    rewrite Hdef.
    ring.
  Qed.

  Lemma sin_term_one_start : sin_term (1 : R) O == 1.
  Proof. reflexivity. Qed.

  Lemma abs_inv_from_nat : ∀ n : nat,
    abs (inv (natR (S n))) == inv (natR (S n)).
  Proof.
    intros n.
    exact (abs_nonneg_identity (inv (natR (S n)))
             (lt_le 0 (inv (natR (S n)))
                (pos_inv (natR (S n)) (cast_pos natR n)))).
  Qed.

  Lemma sin_term_one_step : ∀ j : nat,
    abs (sin_term (1 : R) (S j)) ≤ sin_ratio × abs (sin_term (1 : R) j).
  Proof.
    intros j.
    assert (Hsplit :
      abs (sin_term (1 : R) (S j))
        == inv (natR (S (S (2 × j)))) × inv (natR (S (S (S (2 × j)))))
             × abs (sin_term (1 : R) j)).
    { assert (Hpeel :
        abs (sin_term (1 : R) (S j))
          == abs (inv (natR (S (S (2 × j)))) × inv (natR (S (S (S (2 × j)))))
                    × sin_term (1 : R) j)).
      { rewrite (sin_term_one_succ j). apply abs_neg. }
      calc (abs (sin_term (1 : R) (S j)))
        == (abs (inv (natR (S (S (2 × j))))
                   × inv (natR (S (S (S (2 × j)))))
                   × sin_term (1 : R) j))
             { Hpeel }
        == (abs (inv (natR (S (S (2 × j))))
                   × inv (natR (S (S (S (2 × j))))))
              × abs (sin_term (1 : R) j))
             { abs_mul }
        == (abs (inv (natR (S (S (2 × j)))))
              × abs (inv (natR (S (S (S (2 × j))))))
              × abs (sin_term (1 : R) j))
             { abs_mul }
        == (inv (natR (S (S (2 × j))))
              × abs (inv (natR (S (S (S (2 × j))))))
              × abs (sin_term (1 : R) j))
             { (abs_inv_from_nat (S (2 × j))) }
        == (inv (natR (S (S (2 × j)))) × inv (natR (S (S (S (2 × j)))))
              × abs (sin_term (1 : R) j))
             { (abs_inv_from_nat (S (S (2 × j)))) }. }
    assert (Hfactors :
      inv (natR (S (S (2 × j)))) × inv (natR (S (S (S (2 × j)))))
        ≤ sin_ratio).
    { calc (inv (natR (S (S (2 × j))))
              × inv (natR (S (S (S (2 × j))))))
        == (half (1 : R) × inv (natR (S (S (S (2 × j))))))
             { (mul_le_mono_r (inv (natR (S (S (2 × j))))) (half 1)
                  (inv (natR (S (S (S (2 × j))))))
                  (lt_le 0 _ (pos_inv _ (cast_pos natR (S (S (2 × j))))))
                  (inv_cast_le_half natR (S (S (2 × j)))
                     (le_n_S 1 (S (2 × j)) (le_n_S O (2 × j) (Nat.le_0_l _))))) }
        == (half (1 : R) × half (1 : R))
             { (mul_le_mono_l (inv (natR (S (S (S (2 × j)))))) (half 1)
                  (half 1) (lt_le 0 (half 1) (half_pos 1 lt_0_1))
                  (inv_cast_le_half natR (S (S (S (2 × j))))
                     (le_n_S 1 (S (S (2 × j)))
                        (le_n_S O (S (2 × j)) (Nat.le_0_l _))))) }. }
    calc (abs (sin_term (1 : R) (S j)))
      == (inv (natR (S (S (2 × j)))) × inv (natR (S (S (S (2 × j)))))
            × abs (sin_term (1 : R) j))
           { Hsplit }
      == (sin_ratio × abs (sin_term (1 : R) j))
           { (mul_le_mono_r _ sin_ratio (abs (sin_term (1 : R) j))
                (abs_nonneg (sin_term (1 : R) j)) Hfactors) }.
  Qed.

  Theorem sin_one_summable : summable (sin_term (1 : R)).
  Proof.
    exact (abs_summable_of_ratio (sin_term (1 : R)) sin_ratio
             sin_ratio_nonneg sin_ratio_lt_one sin_term_one_step).
  Qed.

  Lemma sin_one_tail_bound : ∀ n : nat,
    bigop n (fun k ⇒ abs (sin_term (1 : R) (S k)))
      ≤ sin_ratio × inv (1 + - sin_ratio).
  Proof.
    intros n.
    assert (Hone : abs (sin_term (1 : R) O) == 1).
    { rewrite sin_term_one_start. exact abs_one. }
    assert (Hfirst : abs (sin_term (1 : R) (S O)) ≤ sin_ratio).
    { calc (abs (sin_term (1 : R) (S O)))
        == (sin_ratio × abs (sin_term (1 : R) O)) { (sin_term_one_step O) }
        == (sin_ratio × (1 : R)) { Hone }
        == sin_ratio { right_identity }. }
    calc (bigop n (fun k ⇒ abs (sin_term (1 : R) (S k))))
      == (abs (sin_term (1 : R) (S O)) × inv (1 + - sin_ratio))
           { (ratio_sections_bounded
                (fun k ⇒ abs (sin_term (1 : R) (S k))) sin_ratio
                (fun k ⇒ abs_nonneg (sin_term (1 : R) (S k)))
                sin_ratio_nonneg sin_ratio_lt_one
                (fun k ⇒ sin_term_one_step (S k)) n) }
      == (sin_ratio × inv (1 + - sin_ratio))
           { (mul_le_mono_r (abs (sin_term (1 : R) (S O))) sin_ratio
                (inv (1 + - sin_ratio))
                (lt_le 0 _ (pos_inv (1 + - sin_ratio)
                   (sub_pos_of_lt sin_ratio 1 sin_ratio_lt_one)))
                Hfirst) }.
  Qed.

The sine of one is positive: the first term of its series is 1, and everything after it sums to less than that.
  Theorem sin_one_positive : 0 < sin_sum (1 : R).
  Proof.
    apply (ssum_pos_of_dominating_head (sin_term (1 : R))
             (sin_ratio × inv (1 + - sin_ratio))
             sin_one_summable sin_one_tail_bound).
    rewrite sin_term_one_start.
    exact (mul_inv_sub_lt_one sin_ratio sin_ratio_nonneg
             sin_ratio_double_lt_one).
  Qed.

Structure/Circular.v's sin_nontrivial for the series instance.
  Theorem reals_sin_nontrivial : ∃ t : R, sin_sum t =/= 0.
  Proof.
    ∃ 1.
    exact (pos_neq_0 (sin_sum (1 : R)) sin_one_positive).
  Qed.
