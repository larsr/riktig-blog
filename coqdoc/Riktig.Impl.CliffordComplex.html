  Lemma even_one : cl_even (1 : Cl F 2).
  Proof. exact (cl_even_one 2). Qed.

  Lemma even_mul_closed : ∀ x y : Cl F 2,
    cl_even x → cl_even y → cl_even (x × y).
  Proof. exact (cl_even_mul 2 sig2). Qed.

  Lemma even_mul_comm : ∀ x y : Cl F 2,
    cl_even x → cl_even y → x × y == y × x.
  Proof.
    intros x y Hx Hy.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    destruct (even_is_rotor y Hy) as [c [d Hcd]].
    rewrite Hab, Hcd, !rotor_mul.
    apply rotor_proper; ring.
  Qed.


  Definition even_norm2 (x : Cl F 2) : F := norm2 (even_to_complex x).

  Definition even_inv (x : Cl F 2) : Cl F 2 :=
    inv (even_norm2 x) *: cl_reverse x.

  #[export] Instance even_norm2_proper : Proper (equ ==> equ) even_norm2.
  Proof.
    intros x y Hxy.
    exact (norm2_proper (even_to_complex x) (even_to_complex y)
             (even_to_complex_proper x y Hxy)).
  Qed.

  #[export] Instance even_inv_proper : Proper (equ ==> equ) even_inv.
  Proof.
    intros x y Hxy.
    apply scale_proper;
      [ apply inv_proper; exact (even_norm2_proper x y Hxy)
      | exact (cl_reverse_proper 2 x y Hxy) ].
  Qed.

  Lemma even_inv_closed : ∀ x : Cl F 2,
    cl_even x → cl_even (even_inv x).
  Proof. intros x Hx. apply cl_even_scale, cl_even_reverse, Hx. Qed.

  Lemma even_norm2_rotor : ∀ a b : F,
    even_norm2 (rotor a b) == a × a + b × b.
  Proof. intros a b. reflexivity. Qed.

  Lemma even_inv_rotor : ∀ a b : F,
    even_inv (rotor a b)
    == rotor (inv (a×a + b×b) × a) (inv (a×a + b×b) × - b).
  Proof.
    intros a b.
    calc (even_inv (rotor a b))
      == ((inv (even_norm2 (rotor a b)) *: cl_reverse (rotor a b)) : Cl F 2)
         { reflexivity }
      == ((inv (a×a + b×b) *: rotor a (- b)) : Cl F 2)
         { rotor_reverse }
      == (rotor (inv (a×a + b×b) × a) (inv (a×a + b×b) × - b) : Cl F 2)
         { scale_rotor }.
  Qed.

  Definition e12 : Even2 F := exist cl_even (rotor 0 1) (rotor_is_even 0 1).

  Example e12_squared_in_even : e12 × e12 == - (1).
  Proof.
    assert (Hre : (0:F)*0 + - (1×1) == - (1) × 1) by ring.
    assert (Him : (0:F)*1 + 1×0 == - (1) × 0) by ring.
    apply even_eq_proj.
    calc ((rotor (0:F) 1 × rotor 0 1) : Cl F 2)
      == (rotor ((0:F)*0 + - (1×1)) ((0:F)*1 + 1×0) : Cl F 2)
         { rotor_mul }
      == (rotor (- (1:F) × 1) (- (1:F) × 0) : Cl F 2)
         { (rotor_proper _ _ Hre _ _ Him) }
      == ((- (1:F)) *: rotor (1:F) 0 : Cl F 2)
         { scale_rotor }
      == ((- (1:F)) *: (1 : Cl F 2) : Cl F 2)
         { rotor_one }.
  Qed.

Every even element is a combination of the unit and the bivector: that is `even_is_rotor` followed by `rotor_two_term`.
  Lemma even_spans : ∀ x : Even2 F, ∃ a b : F, x == a *: 1 + b *: e12.
  Proof.
    intros x.
    destruct (even_is_rotor (proj1_sig x) (proj2_sig x)) as [a [b Hab]].
    ∃ a, b.
    apply even_eq_proj.
    calc ((proj1_sig x) : Cl F 2)
      == ((rotor a b) : Cl F 2)
         { Hab }
      == ((a *: (1 : Cl F 2) + b *: rotor 0 1) : Cl F 2)
         { rotor_two_term }.
  Qed.

The unit and the bivector are different blades, which is Impl/PlaneRotor.v's `plane_independent_from_one` read here.
  Lemma even_independent : independent_from_one F e12.
  Proof. intros a b Hab. exact (plane_independent_from_one a b Hab). Qed.

The even part of the plane algebra IS the complex numbers, in the sense of Structure/ComplexPlane.v.

  Lemma even_to_complex_inv : ∀ x : Cl F 2,
    cl_even x → even_to_complex (even_inv x) == inv (even_to_complex x).
  Proof.
    intros x Hx.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    rewrite Hab, even_inv_rotor, !even_to_complex_rotor.
    split; simpl; unfold norm2; simpl; ring.
  Qed.

  Lemma even_to_C_inv : ∀ x : Even2 F,
    even_to_C (inv x) == inv (even_to_C x).
  Proof. intros x. exact (even_to_complex_inv (proj1_sig x) (proj2_sig x)). Qed.


  Lemma even_to_complex_reverse : ∀ x : Cl F 2,
    cl_even x →
    even_to_complex (cl_reverse x) == conjugate (even_to_complex x).
  Proof.
    intros x Hx.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    calc ((even_to_complex (cl_reverse x)) : C F)
      == ((even_to_complex (cl_reverse (rotor a b))) : C F)
         { Hab }
      == ((even_to_complex (rotor a (- b))) : C F)
         { rotor_reverse }
      == ((mkC a (- b)) : C F)
         { even_to_complex_rotor }
      == ((conjugate (mkC a b)) : C F)
         { reflexivity }
      == ((conjugate (even_to_complex (rotor a b))) : C F)
         { even_to_complex_rotor }
      == ((conjugate (even_to_complex x)) : C F)
         { Hab }.
  Qed.

Reversal in the plane algebra IS complex conjugation: the isomorphism of the even part with the coordinate pairs carries one to the other.
  Theorem even_to_C_reverse : ∀ x : Even2 F,
    even_to_C (even_reverse x) == conjugate (even_to_C x).
  Proof.
    intros x. exact (even_to_complex_reverse (proj1_sig x) (proj2_sig x)).
  Qed.

And it is the only such map: Theory/ComplexPlane.v's uniqueness theorem, at this plane.
  Corollary even_reverse_unique : ∀ f : Even2 F → Even2 F,
    Reversal (Even2 F) f →
    (∀ (a : F) (x : Even2 F), f (a *: x) == a *: f x) →
    f e12 == - e12 →
    ∀ x : Even2 F, f x == even_reverse x.
  Proof.
    assert (Hroot : even_reverse e12 == - e12).
    { apply even_eq_proj.
      calc ((cl_reverse (rotor (0 : F) 1)) : Cl F 2)
        == ((- rotor (0 : F) 1) : Cl F 2)
           { plane_reverse_bivector }
        == (((- (1 : F)) *: rotor (0 : F) 1) : Cl F 2)
           { neg1_scale }. }
    exact (plane_conjugate_unique e12 even_reverse even_reverse_scale Hroot).
  Qed.


  Corollary even_to_C_mul_swap : ∀ x y : Even2 F,
    even_to_C (y × x) == even_to_C (x × y).
  Proof. intros x y. apply even_to_C_proper. exact (commutative y x). Qed.

Rotating with u v and rotating with v u: the two products are conjugate complex numbers. This is the statement `even_to_C_mul_swap` cannot make, because a plane vector is odd and is not an element of the even part.
  Theorem vector_swap_conjugates : ∀ u1 u2 v1 v2 : F,
    even_to_complex (ga_vec (plane_vector v1 v2) × ga_vec (plane_vector u1 u2))
    == conjugate (even_to_complex
         (ga_vec (plane_vector u1 u2) × ga_vec (plane_vector v1 v2))).
  Proof.
    intros u1 u2 v1 v2.
    assert (Heven : cl_even (ga_vec (plane_vector u1 u2)
                               × ga_vec (plane_vector v1 v2)))
      by exact (odd_mul_odd (ga_vec (plane_vector u1 u2))
                  (ga_vec (plane_vector v1 v2))
                  (vec_odd (plane_vector u1 u2)) (vec_odd (plane_vector v1 v2))).
    calc ((even_to_complex (ga_vec (plane_vector v1 v2)
                              × ga_vec (plane_vector u1 u2))) : C F)
      == ((even_to_complex (cl_reverse (ga_vec (plane_vector u1 u2)
                                          × ga_vec (plane_vector v1 v2))))
          : C F)
         { (vector_product_reverse u1 u2 v1 v2) }
      == ((conjugate (even_to_complex (ga_vec (plane_vector u1 u2)
                                         × ga_vec (plane_vector v1 v2))))
          : C F)
         { (even_to_complex_reverse _ Heven) }.
  Qed.

Section EvenField.
  Context {F : Type} `{FR : MinusOneNotASquare F}.
  Add Ring even_field_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma rotor_norm2_nonzero : ∀ a b : F,
    rotor a b =/= (0 : Cl F 2) → a × a + b × b =/= 0.
  Proof.
    intros a b Hab Hn.
    apply Hab.
    apply cl2_ext; unfold rotor, mk2, pb; simpl;
      solve [ reflexivity
            | exact (two_squares_vanish a b Hn)
            | exact (two_squares_vanish_r a b Hn) ].
  Qed.

  Lemma even_inv_left : ∀ x : Cl F 2,
    cl_even x → x =/= 0 → even_inv x × x == 1.
  Proof.
    intros x Hx Hx0.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    rewrite Hab in Hx0 |- ×.
    assert (Hn : a × a + b × b =/= 0) by exact (rotor_norm2_nonzero a b Hx0).
    assert (Hre : (inv (a×a+b×b) × a) × a + - ((inv (a×a+b×b) × - b) × b)
                  == inv (a×a+b×b) × (a×a + b×b)) by ring.
    assert (Him : (inv (a×a+b×b) × a) × b + (inv (a×a+b×b) × - b) × a
                  == (0 : F)) by ring.
    calc ((even_inv (rotor a b) × rotor a b) : Cl F 2)
      == ((rotor (inv (a×a + b×b) × a) (inv (a×a + b×b) × - b) × rotor a b)
          : Cl F 2)
         { even_inv_rotor }
      == (rotor ((inv (a×a+b×b) × a) × a + - ((inv (a×a+b×b) × - b) × b))
                ((inv (a×a+b×b) × a) × b + (inv (a×a+b×b) × - b) × a)
          : Cl F 2)
         { rotor_mul }
      == (rotor (inv (a×a+b×b) × (a×a + b×b)) 0 : Cl F 2)
         { (rotor_proper _ _ Hre _ _ Him) }
      == (rotor (1 : F) 0 : Cl F 2)
         { (left_inverse_on (defined_on := nonzero F) (a×a+b×b) Hn) }
      == (1 : Cl F 2)
         { rotor_one }.
  Qed.

  #[export] Instance even_inv_proper_sub :
    Proper (equ ==> equ) (@inv (Even2 F) _).
  Proof.
    intros x y Hxy. exact (even_inv_proper (proj1_sig x) (proj1_sig y) Hxy).
  Qed.

  #[export] Instance even_mul_left_inverse :
    LeftInverseOn (Even2 F) (nonzero (Even2 F)) mult inv 1.
  Proof.
    intros x Hx. exact (even_inv_left (proj1_sig x) (proj2_sig x) Hx).
  Qed.
