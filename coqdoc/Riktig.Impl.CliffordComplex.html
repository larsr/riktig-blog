#[local] Instance Even_Cl_Ring {F : Type} `{FF : Field F} : Ring (Cl F 2) :=
  Cl_Ring sig2.

Notation Even2 F := (Sub (Cl F 2) cl_even).

Section EvenSubalgebra.
  Context {F : Type} `{FF : Field F}.
  Add Ring even_ring : (ring_theory_of_Ring (R := F) commutative).

  #[export] Instance Even_SubspacePred :
    SubspacePred F (Cl F 2) cl_even := {
      SubspacePred.zero := cl_even_zero 2;
      SubspacePred.add_closed := cl_even_add 2;
      SubspacePred.scale_closed := cl_even_scale 2;
    }.

  Lemma even_one : cl_even (1 : Cl F 2).
  Proof. rewrite <- rotor_one. apply rotor_even. Qed.

  Lemma even_mul_closed : ∀ x y : Cl F 2,
    cl_even x → cl_even y → cl_even (x × y).
  Proof.
    intros x y Hx Hy.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    destruct (even_is_rotor y Hy) as [c [d Hcd]].
    rewrite Hab, Hcd, rotor_mul.
    apply rotor_even.
  Qed.

  Lemma even_mul_comm : ∀ x y : Cl F 2,
    cl_even x → cl_even y → x × y == y × x.
  Proof.
    intros x y Hx Hy.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    destruct (even_is_rotor y Hy) as [c [d Hcd]].
    rewrite Hab, Hcd, !rotor_mul.
    apply rotor_proper; ring.
  Qed.


  Definition even_norm2 (x : Cl F 2) : F := norm2 (even_to_complex x).

  Definition even_inv (x : Cl F 2) : Cl F 2 :=
    inv (even_norm2 x) *: cl_reverse x.

  #[export] Instance even_norm2_proper : Proper (equ ==> equ) even_norm2.
  Proof.
    intros x y Hxy.
    exact (norm2_proper (even_to_complex x) (even_to_complex y)
             (even_to_complex_proper x y Hxy)).
  Qed.

  #[export] Instance even_inv_proper : Proper (equ ==> equ) even_inv.
  Proof.
    intros x y Hxy.
    apply scale_proper;
      [ apply inv_proper; exact (even_norm2_proper x y Hxy)
      | exact (cl_reverse_proper 2 x y Hxy) ].
  Qed.

  Lemma even_inv_closed : ∀ x : Cl F 2,
    cl_even x → cl_even (even_inv x).
  Proof. intros x Hx. apply cl_even_scale, cl_even_reverse, Hx. Qed.

  Lemma even_norm2_rotor : ∀ a b : F,
    even_norm2 (rotor a b) == a × a + b × b.
  Proof. intros a b. reflexivity. Qed.

  Lemma even_inv_rotor : ∀ a b : F,
    even_inv (rotor a b)
    == rotor (inv (a×a + b×b) × a) (inv (a×a + b×b) × - b).
  Proof.
    intros a b.
    calc (even_inv (rotor a b))
      == ((inv (even_norm2 (rotor a b)) *: cl_reverse (rotor a b)) : Cl F 2)
         { reflexivity }
      == ((inv (a×a + b×b) *: rotor a (- b)) : Cl F 2)
         { rotor_reverse }
      == (rotor (inv (a×a + b×b) × a) (inv (a×a + b×b) × - b) : Cl F 2)
         { scale_rotor }.
  Qed.


  #[export] Instance Even_One : One (Even2 F) :=
    exist cl_even 1 even_one.

  #[export] Instance Even_Mult : HasMult (Even2 F) :=
    fun x y ⇒
      exist cl_even (proj1_sig x × proj1_sig y)
        (even_mul_closed (proj1_sig x) (proj1_sig y)
           (proj2_sig x) (proj2_sig y)).

  #[export] Instance Even_Inv : HasInv (Even2 F) :=
    fun x ⇒
      exist cl_even (even_inv (proj1_sig x))
        (even_inv_closed (proj1_sig x) (proj2_sig x)).

  #[export] Instance even_mul_proper :
    Proper (equ ==> equ ==> equ) (@mult (Even2 F) _).
  Proof.
    intros x x' Hx y y' Hy.
    change (proj1_sig x == proj1_sig x') in Hx.
    change (proj1_sig y == proj1_sig y') in Hy.
    exact (op_proper (op := mult) (proj1_sig x) (proj1_sig x') Hx
             (proj1_sig y) (proj1_sig y') Hy).
  Qed.

  #[export] Instance even_mul_associative :
    Associative (Even2 F) (Even2 F) mult mult.
  Proof.
    intros x y z.
    exact (associative (proj1_sig x) (proj1_sig y) (proj1_sig z)).
  Qed.

  #[export] Instance even_mul_left_identity :
    LeftIdentity (Even2 F) (Even2 F) mult 1.
  Proof. intros x. exact (left_identity (B := Cl F 2) (proj1_sig x)). Qed.

  #[export] Instance even_mul_right_identity :
    RightIdentity (Even2 F) (Even2 F) mult 1.
  Proof. intros x. exact (right_identity (proj1_sig x)). Qed.

  #[export] Instance even_mul_commutative :
    Commutative (Even2 F) (Even2 F) mult.
  Proof.
    intros x y.
    exact (even_mul_comm (proj1_sig x) (proj1_sig y)
             (proj2_sig x) (proj2_sig y)).
  Qed.

  #[export] Instance even_left_distributive :
    LeftDistributive (Even2 F) (Even2 F) (Even2 F) mult.
  Proof.
    intros x y z.
    exact (left_distributive (C := Cl F 2)
             (proj1_sig x) (proj1_sig y) (proj1_sig z)).
  Qed.

  #[export] Instance even_right_distributive :
    RightDistributive (Even2 F) (Even2 F) (Even2 F) mult.
  Proof.
    intros x y z.
    exact (right_distributive (C := Cl F 2)
             (proj1_sig x) (proj1_sig y) (proj1_sig z)).
  Qed.

  #[export] Instance Even_MulMagma : Magma (Even2 F) mult := {}.
  #[export] Instance Even_MulSemigroup : Semigroup (Even2 F) mult := {}.
  #[export] Instance Even_MulMonoid :
    @Monoid (Even2 F) _ Even_One Even_Mult := {}.
  #[export] Instance Even_Ring : Ring (Even2 F) := {}.

  Lemma even_nontrivial : (1 : Even2 F) =/= 0.
  Proof. intros Heq. exact (nontrivial (Heq (pb false false))). Qed.


  Definition even_to_C (x : Even2 F) : C F := even_to_complex (proj1_sig x).

  Definition C_to_even (z : C F) : Even2 F :=
    exist cl_even (complex_to_even z) (complex_to_even_is_even z).

  #[export] Instance even_C_inverse : Inverse even_to_C C_to_even := {
    Inverse.from_to := fun x ⇒ even_round_trip (proj1_sig x) (proj2_sig x);
    Inverse.to_from := complex_round_trip;
  }.

  #[export] Instance even_to_C_proper : Proper (equ ==> equ) even_to_C.
  Proof. intros x y Hxy. exact (even_to_complex_proper _ _ Hxy). Qed.

  #[export] Instance C_to_even_proper : Proper (equ ==> equ) C_to_even.
  Proof. intros z w Hzw. exact (complex_to_even_proper z w Hzw). Qed.

  #[export] Instance even_to_C_additive :
    Additive (Even2 F) (C F) add add even_to_C.
  Proof.
    intros x y. exact (even_to_complex_add (proj1_sig x) (proj1_sig y)).
  Qed.

  #[export] Instance even_to_C_multiplicative :
    Additive (Even2 F) (C F) mult mult even_to_C.
  Proof.
    intros x y.
    exact (even_to_complex_mul (proj1_sig x) (proj1_sig y)
             (proj2_sig x) (proj2_sig y)).
  Qed.

  Lemma even_to_C_one : even_to_C 1 == 1.
  Proof. exact even_to_complex_one. Qed.

  Lemma even_to_complex_inv : ∀ x : Cl F 2,
    cl_even x → even_to_complex (even_inv x) == inv (even_to_complex x).
  Proof.
    intros x Hx.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    rewrite Hab, even_inv_rotor, !even_to_complex_rotor.
    split; simpl; unfold norm2; simpl; ring.
  Qed.

  Lemma even_to_C_inv : ∀ x : Even2 F,
    even_to_C (inv x) == inv (even_to_C x).
  Proof. intros x. exact (even_to_complex_inv (proj1_sig x) (proj2_sig x)). Qed.

End EvenSubalgebra.


Section EvenField.
  Context {F : Type} `{FR : MinusOneNotASquare F}.
  Add Ring even_field_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma rotor_norm2_nonzero : ∀ a b : F,
    rotor a b =/= (0 : Cl F 2) → a × a + b × b =/= 0.
  Proof.
    intros a b Hab Hn.
    apply Hab.
    apply cl2_ext; unfold rotor, mk2, pb; simpl;
      solve [ reflexivity
            | exact (two_squares_vanish a b Hn)
            | exact (two_squares_vanish_r a b Hn) ].
  Qed.

  Lemma even_inv_left : ∀ x : Cl F 2,
    cl_even x → x =/= 0 → even_inv x × x == 1.
  Proof.
    intros x Hx Hx0.
    destruct (even_is_rotor x Hx) as [a [b Hab]].
    rewrite Hab in Hx0 |- ×.
    assert (Hn : a × a + b × b =/= 0) by exact (rotor_norm2_nonzero a b Hx0).
    assert (Hre : (inv (a×a+b×b) × a) × a + - ((inv (a×a+b×b) × - b) × b)
                  == inv (a×a+b×b) × (a×a + b×b)) by ring.
    assert (Him : (inv (a×a+b×b) × a) × b + (inv (a×a+b×b) × - b) × a
                  == (0 : F)) by ring.
    calc ((even_inv (rotor a b) × rotor a b) : Cl F 2)
      == ((rotor (inv (a×a + b×b) × a) (inv (a×a + b×b) × - b) × rotor a b)
          : Cl F 2)
         { even_inv_rotor }
      == (rotor ((inv (a×a+b×b) × a) × a + - ((inv (a×a+b×b) × - b) × b))
                ((inv (a×a+b×b) × a) × b + (inv (a×a+b×b) × - b) × a)
          : Cl F 2)
         { rotor_mul }
      == (rotor (inv (a×a+b×b) × (a×a + b×b)) 0 : Cl F 2)
         { (rotor_proper _ _ Hre _ _ Him) }
      == (rotor (1 : F) 0 : Cl F 2)
         { (left_inverse_on (defined_on := nonzero F) (a×a+b×b) Hn) }
      == (1 : Cl F 2)
         { rotor_one }.
  Qed.

  #[export] Instance even_inv_proper_sub :
    Proper (equ ==> equ) (@inv (Even2 F) _).
  Proof.
    intros x y Hxy. exact (even_inv_proper (proj1_sig x) (proj1_sig y) Hxy).
  Qed.

  #[export] Instance even_mul_left_inverse :
    LeftInverseOn (Even2 F) (nonzero (Even2 F)) mult inv 1.
  Proof.
    intros x Hx. exact (even_inv_left (proj1_sig x) (proj2_sig x) Hx).
  Qed.

  #[export] Instance Even_Field : Field (Even2 F) :=
    { Field.nontrivial := even_nontrivial }.

End EvenField.


Section EvenChecks.
  Context {F : Type} `{FR : MinusOneNotASquare F}.
  Add Ring even_check_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition e12 : Even2 F := exist cl_even (rotor 0 1) (rotor_even 0 1).

  Example e12_squared_in_even : e12 × e12 == - (1).
  Proof.
    assert (Hre : (0:F)*0 + - (1×1) == - (1) × 1) by ring.
    assert (Him : (0:F)*1 + 1×0 == - (1) × 0) by ring.
    change ((rotor (0:F) 1 × rotor 0 1 : Cl F 2)
            == (- (1 : F)) *: (1 : Cl F 2)).
    calc ((rotor (0:F) 1 × rotor 0 1) : Cl F 2)
      == (rotor ((0:F)*0 + - (1×1)) ((0:F)*1 + 1×0) : Cl F 2)
         { rotor_mul }
      == (rotor (- (1:F) × 1) (- (1:F) × 0) : Cl F 2)
         { (rotor_proper _ _ Hre _ _ Him) }
      == ((- (1:F)) *: rotor (1:F) 0 : Cl F 2)
         { scale_rotor }
      == ((- (1:F)) *: (1 : Cl F 2) : Cl F 2)
         { rotor_one }.
  Qed.

  Example even_field_check : Field (Even2 F) := _.

  Example iso_round_trip_even : ∀ x : Even2 F,
    C_to_even (even_to_C x) == x.
  Proof. exact Inverse.from_to. Qed.

  Example iso_round_trip_complex : ∀ z : C F, even_to_C (C_to_even z) == z.
  Proof. exact Inverse.to_from. Qed.

  Example inv_both_ways : ∀ x : Even2 F,
    C_to_even (inv (even_to_C x)) == inv x.
  Proof.
    intros x.
    calc (C_to_even (inv (even_to_C x)))
      == (C_to_even (even_to_C (inv x)) : Even2 F)
         { even_to_C_inv }
      == (inv x : Even2 F)
         { Inverse.from_to }.
  Qed.

End EvenChecks.
