Coq's lambda terms over a type of variables are the lambda terms of Coq's types: build is Var, Abs and App, and cata is the fold.
Section L.
  Context (N : Type).

  #[export] Instance CoqLambda_HasBuild : HasBuild Type (LambdaTerm.shape (fun T0 : Type ⇒ T0) N) (lambda N) :=
    fun s ⇒ match s with
             | inl (inl n) ⇒ Var n
             | inl (inr b) ⇒ Abs b
             | inr (f, a) ⇒ App f a
             end.

  #[export] Instance CoqLambda_HasCata : HasCata Type (LambdaTerm.shape (fun T0 : Type ⇒ T0) N) (lambda N) :=
    fun B h ⇒ fix go (t : lambda N) : B :=
      match t with
      | Var n ⇒ h (inl (inl n))
      | Abs b ⇒ h (inl (inr (go b)))
      | App f a ⇒ h (inr (go f, go a))
      end.

  #[export] Instance CoqLambda_LambdaTerm
    : LambdaTerm Type (fun T0 : Type ⇒ T0) N (lambda N) CoqLambda_HasBuild CoqLambda_HasCata.
  Proof.
    split. split.
    - exact (layer_functor (fun T0 : Type ⇒ T0) (LambdaTerm.lamS (S := sig Type) N)).
    - intros B h h' Hh t.
      induction t as [n | b IHb | f IHf a IHa].
      + exact (Hh (inl (inl n))).
      + exact (eq_trans (f_equal (fun x ⇒ h (inl (inr x))) IHb) (Hh (inl (inr _)))).
      + exact (eq_trans (f_equal2 (fun x y ⇒ h (inr (x, y))) IHf IHa) (Hh (inr (_, _)))).
    - intros B h [[n | b] | [f a]]; reflexivity.
    - intros B h u Hu t.
      induction t as [n | b IHb | f IHf a IHa].
      + exact (Hu (inl (inl n))).
      + exact (eq_trans (Hu (inl (inr b))) (f_equal (fun x ⇒ h (inl (inr x))) IHb)).
      + exact (eq_trans (Hu (inr (f, a))) (f_equal2 (fun x y ⇒ h (inr (x, y))) IHf IHa)).
  Qed.
End L.
