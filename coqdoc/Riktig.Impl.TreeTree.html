Coq's binary tree is the tree of Coq's types: build is Tip and Fork, and cata is the fold.
Section T.
  Context (X : Type).

  #[export] Instance CoqTree_HasTip : HasTip X (tree X) := @Tip X.
  #[export] Instance CoqTree_HasFork : HasFork (tree X) := @Fork X.

  #[export] Instance CoqTree_HasBuild : HasBuild Type (Tree.shape (fun T0 : Type ⇒ T0) X) (tree X) :=
    fun s ⇒ match s with
             | inl a ⇒ Tip a
             | inr (l, r) ⇒ Fork l r
             end.

  #[export] Instance CoqTree_HasCata : HasCata Type (Tree.shape (fun T0 : Type ⇒ T0) X) (tree X) :=
    fun B h ⇒ fix go (t : tree X) : B :=
      match t with
      | Tip a ⇒ h (inl a)
      | Fork l r ⇒ h (inr (go l, go r))
      end.

  #[export] Instance CoqTree_Tree :
    Tree Type (fun T0 : Type ⇒ T0) X (tree X) CoqTree_HasBuild CoqTree_HasCata.
  Proof.
    split. split.
    - exact (layer_functor (fun T0 : Type ⇒ T0) (treeS (S := sig Type) X)).
    - intros B h h' Hh t. induction t as [a | l IHl r IHr].
      + exact (Hh (inl a)).
      + exact (eq_trans (f_equal2 (fun p q ⇒ h (inr (p, q))) IHl IHr) (Hh (inr (_, _)))).
    - intros B h [a | [l r]]; reflexivity.
    - intros B h u Hu t. induction t as [a | l IHl r IHr].
      + exact (Hu (inl a)).
      + exact (eq_trans (Hu (inr (l, r))) (f_equal2 (fun p q ⇒ h (inr (p, q))) IHl IHr)).
  Qed.
End T.
