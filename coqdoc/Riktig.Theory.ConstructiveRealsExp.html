The exponential series at the constructive reals: the terms x^n / n! by recursion, each the previous times x / n; past an Archimedean index the ratio is at most a half, so the terms are bounded by a geometric majorant and the partial sums are Cauchy, with a computed modulus.

  Context {R Q N : Type} `{ConstructiveReals R Q N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}
    {of_nat : N → Q} {AQ : Archimedean Q N of_nat} {LDQ : LtDec Q}.

  Local Notation "'exp_term' x n" :=
    (natrec R 1 (fun k acc ⇒ acc × (cast (inv (of_nat (succ k))) × x)) n)
    (at level 10, x at level 9, n at level 9).

  Local Notation "'psum' f n" := (natrec _ 0 (fun k acc ⇒ acc + f k) n) (at level 10, f at level 9, n at level 9).
  Local Notation "'is_cauchy' u" :=
    (∀ eps : Q, 0 < eps →
      {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps})
    (at level 10, u at level 9).
  Local Notation half := (inv ((1 : Q) + 1)).
  Local Notation "'pow' r n" := (natrec Q 1 (fun _ acc ⇒ acc × r) n) (at level 10, r at level 9, n at level 9).

  Lemma ratio_proper : ∀ x : R, Proper (equ ==> equ) (fun k : N ⇒ cast (inv (of_nat (succ k))) × x).
  Proof.
    intros x k k' Hk. apply op_proper; [ | reflexivity ].
    apply hom_proper. apply inv_proper. apply of_nat_equ. apply succ_proper. exact Hk.
  Qed.

  Theorem exp_term_proper : ∀ x : R, Proper (equ ==> equ) (fun n : N ⇒ exp_term x n).
  Proof.
    intros x. apply (natrec_proper R _ _ 1 _).
    intros k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | exact (ratio_proper x k k' Hk) ].
  Qed.

  Theorem exp_term_zero : ∀ x : R, exp_term x (0 : N) == 1.
  Proof. intros x. exact (natrec_zero R _ _ 1 _). Qed.

  Theorem exp_term_succ : ∀ (x : R) (k : N),
    exp_term x (succ k) == exp_term x k × (cast (inv (of_nat (succ k))) × x).
  Proof. intros x k. exact (natrec_succ R _ _ 1 _ k). Qed.

Past m, with |x| ≤ m / 2, each ratio is at most a half.
  Lemma ratio_le_half : ∀ (x : R) (m k : N),
    abs x ≼ cast (of_nat m × half) → m ≤ k →
    abs (cast (inv (of_nat (succ k))) × x) ≼ cast half.
  Proof.
    intros x m k Hx Hmk.
    assert (Hs : (0 : Q) < of_nat (succ k)) by exact (of_nat_succ_pos k).
    assert (Hq : (0 : Q) ≤ inv (of_nat (succ k))) by exact (Theory.Order.le_of_lt _ _ (Theory.Order.inv_pos _ Hs)).
    apply (le_equ_l (cast (inv (of_nat (succ k))) × abs x)).
    - calc (cast (inv (of_nat (succ k))) × abs x)
        == (cast (abs (inv (of_nat (succ k)))) × abs x) { (OrderedField.abs_of_nonneg (inv (of_nat (succ k))) Hq) }
        == (abs (cast (inv (of_nat (succ k)))) × abs x) { (abs_cast (inv (of_nat (succ k)))) }
        == (abs (cast (inv (of_nat (succ k))) × x)) { (ConstructiveReals.abs_mul (cast (inv (of_nat (succ k)))) x) }.
    - apply (Theory.ConstructiveReals.le_trans _ (cast (inv (of_nat (succ k))) × cast (of_nat m × half))).
      + exact (ConstructiveReals.mul_le_mono_l _ _ _ (cast_nonneg _ Hq) Hx).
      + apply (le_equ_l (cast (inv (of_nat (succ k)) × (of_nat m × half)))).
        × exact (additive (f := cast) (opA := mult) (inv (of_nat (succ k))) (of_nat m × half)).
        × apply cast_le.
          apply (transitivity (le_of_eq _ _ (symmetry (associative (op := mult) (action := mult) (inv (of_nat (succ k))) (of_nat m) half)))).
          apply (transitivity (y := 1 × half)).
          { apply (Theory.Order.mul_le_mono_r _ _ half (Theory.Order.le_of_lt _ _ (Theory.Order.inv_pos _ Theory.Order.two_pos))).
            apply (Theory.Order.inv_mul_le_one (of_nat m) (of_nat (succ k)) Hs).
            exact (of_nat_le m (succ k) (Theory.NatOrder.le_trans m k (succ k) Hmk (Theory.NatOrder.le_succ_diag k))). }
          exact (le_of_eq _ _ (left_identity (action := mult) half)).
  Qed.

Past m the terms are bounded by the m-th term times halves.
  Lemma abs_exp_term_tail : ∀ (x : R) (m : N), abs x ≼ cast (of_nat m × half) →
    ∀ j : N, abs (exp_term x (m + j)) ≼ abs (exp_term x m) × cast (pow half j).
  Proof.
    intros x m Hx.
    apply (nat_induction (fun j ⇒ abs (exp_term x (m + j)) ≼ abs (exp_term x m) × cast (pow half j))).
    - intros j j' Hj. split; intros Hle.
      + apply (le_equ_l (abs (exp_term x (m + j))));
          [ apply ConstructiveReals.abs_proper; exact (exp_term_proper x (m + j) (m + j') (op_proper m m (reflexivity m) j j' Hj)) | ].
        apply (le_equ_r _ (abs (exp_term x m) × cast (pow half j)));
          [ apply op_proper; [ reflexivity | apply hom_proper; exact (pow_proper half j j' Hj) ] | exact Hle ].
      + apply (le_equ_l (abs (exp_term x (m + j'))));
          [ apply ConstructiveReals.abs_proper; exact (exp_term_proper x (m + j') (m + j) (op_proper m m (reflexivity m) j' j (symmetry Hj))) | ].
        apply (le_equ_r _ (abs (exp_term x m) × cast (pow half j')));
          [ apply op_proper; [ reflexivity | apply hom_proper; exact (pow_proper half j' j (symmetry Hj)) ] | exact Hle ].
    - apply (le_equ_l (abs (exp_term x m)));
        [ apply ConstructiveReals.abs_proper; exact (exp_term_proper x m (m + 0) (symmetry (Theory.Naturals.add_zero m))) | ].
      apply le_of_equ.
      calc (abs (exp_term x m))
        == (abs (exp_term x m) × 1) { (right_identity (action := mult) (abs (exp_term x m))) }
        == (abs (exp_term x m) × cast 1) { hom_one }
        == (abs (exp_term x m) × cast (pow half (0 : N))) { (pow_zero half) }.
    - intros j IH.
      apply (le_equ_l (abs (exp_term x (m + j) × (cast (inv (of_nat (succ (m + j)))) × x)))).
      + apply ConstructiveReals.abs_proper.
        calc (exp_term x (m + j) × (cast (inv (of_nat (succ (m + j)))) × x))
          == (exp_term x (succ (m + j))) { (exp_term_succ x (m + j)) }
          == (exp_term x (m + succ j)) { (exp_term_proper x (succ (m + j)) (m + succ j) (symmetry (Theory.Naturals.add_succ m j))) }.
      + apply (le_equ_r _ ((abs (exp_term x m) × cast (pow half j)) × cast half)).
        × calc ((abs (exp_term x m) × cast (pow half j)) × cast half)
            == (abs (exp_term x m) × (cast (pow half j) × cast half)) { (associative (op := mult) (action := mult) (abs (exp_term x m)) (cast (pow half j)) (cast half)) }
            == (abs (exp_term x m) × cast (pow half j × half)) { (additive (f := cast) (opA := mult) (pow half j) half) }
            == (abs (exp_term x m) × cast (pow half (succ j))) { (pow_succ half j) }.
        × apply (le_equ_l (abs (exp_term x (m + j)) × abs (cast (inv (of_nat (succ (m + j)))) × x)));
            [ symmetry; exact (ConstructiveReals.abs_mul _ _) | ].
          apply Theory.ConstructiveReals.mul_le_mono.
          { exact (Theory.ConstructiveReals.abs_nonneg _). }
          { exact IH. }
          { exact (Theory.ConstructiveReals.abs_nonneg _). }
          { exact (ratio_le_half x m (m + j) Hx (Theory.NatOrder.le_add_r m j)). }
  Qed.

The index past which |x| ≤ m / 2.
  Lemma exp_index : ∀ x : R, {m : N & abs x ≼ cast (of_nat m × half)}.
  Proof.
    intros x.
    destruct (ConstructiveReals.archimedean (abs x)) as [q Hq].
    destruct (Archimedean.archimedean (N := N) (of_nat := of_nat) (q × (1 + 1))) as [m Hm].
    ∃ m.
    apply (Theory.ConstructiveReals.le_trans _ (cast q)); [ exact (Theory.ConstructiveReals.le_of_lt _ _ Hq) | ].
    apply cast_le.
    apply (transitivity (y := (q × (1 + 1)) × half)).
    - apply le_of_eq. symmetry.
      calc ((q × (1 + 1)) × half)
        == (q × ((1 + 1) × half)) { (associative (op := mult) (action := mult) q (1 + 1) half) }
        == (q × 1) { (mul_inv_cancel (1 + 1) (Theory.GeometricSeries.two_ne_zero (F := Q))) }
        == q { (right_identity (action := mult) q) }.
    - apply (Theory.Order.mul_le_mono_r _ _ half (Theory.GeometricSeries.half_nonneg (F := Q))).
      exact (Theory.Order.le_of_lt _ _ Hm).
  Defined.

A rational bound on the m-th term, and the majorant.
  Lemma abs_exp_term_le_majorant : ∀ (x : R) (m : N) (C : Q),
    abs x ≼ cast (of_nat m × half) → abs (exp_term x m) ≼ cast C →
    ∀ j : N, abs (exp_term x (m + j)) ≼ cast (C × pow half j).
  Proof.
    intros x m C Hx HC j.
    apply (Theory.ConstructiveReals.le_trans _ (abs (exp_term x m) × cast (pow half j))); [ exact (abs_exp_term_tail x m Hx j) | ].
    apply (le_equ_r _ (cast C × cast (pow half j)));
      [ symmetry; exact (additive (f := cast) (opA := mult) C (pow half j)) | ].
    apply Theory.ConstructiveReals.mul_le_mono_r; [ | exact HC ].
    exact (cast_nonneg _ (pow_nonneg half (Theory.GeometricSeries.half_nonneg (F := Q)) j)).
  Qed.

The tails of the majorant are small from an index on.
  Lemma majorant_tail_small : ∀ (C eps : Q), 0 ≤ C → 0 < eps →
    {n : N & ∀ i c : N, n ≤ i → psum (fun k ⇒ C × pow half (i + k)) c ≤ eps}.
  Proof.
    intros C eps HC Heps.
    assert (Hpos : (0 : Q) < C + 1)
      by exact (Theory.Order.lt_of_le_of_lt 0 C (C + 1) HC (Theory.Order.lt_add_one C)).
    assert (Hne : C + 1 =/= 0) by exact (fun H0 ⇒ proj2 Hpos (symmetry H0)).
    pose (eps' := eps × inv (C + 1)).
    assert (Heps' : 0 < eps')
      by exact (Theory.Order.mul_pos eps (inv (C + 1)) Heps (Theory.Order.inv_pos (C + 1) Hpos)).
    destruct (half_tail_small (F := Q) (N := N) eps' Heps') as [n Hn]. ∃ n.
    intros i c Hi.
    assert (Hsh : Proper (equ ==> equ) (fun k ⇒ pow half (i + k)))
      by (intros k k' Hk; exact (pow_proper half (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk))).
    apply (transitivity (le_of_eq _ _ (psum_mul_l C (fun k ⇒ pow half (i + k)) Hsh c))).
    apply (transitivity (y := (C + 1) × eps')).
    - apply Theory.Order.mul_le_mono.
      + exact HC.
      + exact (Theory.Order.le_of_lt _ _ (Theory.Order.lt_add_one C)).
      + apply (psum_nonneg _ Hsh). intros k. exact (pow_nonneg half (Theory.GeometricSeries.half_nonneg (F := Q)) (i + k)).
      + exact (Hn i c Hi).
    - exact (Theory.Order.le_of_eq _ _ (mul_mul_inv_cancel_left eps (C + 1) Hne)).
  Defined.

The partial sums of the exponential series are Cauchy: past the index, the terms are bounded by the majorant.
  Theorem exp_summable : ∀ x : R, is_cauchy (fun n ⇒ psum (fun j ⇒ exp_term x j) n).
  Proof.
    intros x.
    destruct (exp_index x) as [m Hm].
    destruct (ConstructiveReals.archimedean (abs (exp_term x m))) as [C HC].
    assert (HC0 : 0 ≤ C).
    { apply (Theory.Order.le_of_lt 0 C). apply (lt_cast 0 C).
      apply (lt_equ_l 0 (cast 0)); [ symmetry; exact (hom_zero cast) | ].
      exact (le_lt_trans 0 (abs (exp_term x m)) (cast C) (Theory.ConstructiveReals.abs_nonneg _) HC). }
    assert (Hshift : is_cauchy (fun c ⇒ psum (fun j ⇒ exp_term x (m + j)) c)).
    { apply (cauchy_psum_of_abs_le (fun j ⇒ exp_term x (m + j)) (fun j ⇒ C × pow half j)).
      - intros j j' Hj. exact (exp_term_proper x (m + j) (m + j') (op_proper m m (reflexivity m) j j' Hj)).
      - intros j j' Hj. apply op_proper; [ reflexivity | exact (pow_proper half j j' Hj) ].
      - intros j. exact (abs_exp_term_le_majorant x m C Hm (Theory.ConstructiveReals.le_of_lt _ _ HC) j).
      - intros eps Heps.
        destruct (majorant_tail_small C eps HC0 Heps) as [n Hn]. ∃ n.
        intros i c Hi.
        apply (transitivity (y := psum (fun k ⇒ C × pow half (i + k)) c)); [ | exact (Hn i c Hi) ].
        apply le_of_eq. apply psum_ext.
        + intros k k' Hk. apply op_proper; [ reflexivity | exact (pow_proper half (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk)) ].
        + intros k k' Hk. apply op_proper; [ reflexivity | exact (pow_proper half (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk)) ].
        + intros k. reflexivity. }
    apply (cauchy_of_shift (fun n ⇒ psum (fun j ⇒ exp_term x j) n) m (psum_proper _ (exp_term_proper x))).
    apply (Theory.ConstructiveRealsLimits.cauchy_proper (fun c ⇒ psum (fun j ⇒ exp_term x j) m + psum (fun j ⇒ exp_term x (m + j)) c)).
    - intros c. symmetry. exact (psum_shift (fun j ⇒ exp_term x j) (exp_term_proper x) m c).
    - exact (Theory.ConstructiveRealsLimits.cauchy_add _ _ (cauchy_of_converges _ _ (converges_const (psum (fun j ⇒ exp_term x j) m))) Hshift).
  Defined.
