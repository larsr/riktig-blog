Library Riktig.Theory.SpaceRotor

From Coq Require Import Morphisms Setoid Ring Bool.
From Coq.Vectors Require Fin.
From Riktig Require Export Impl.Clifford.
From Riktig Require Import Lib.Calc.

Section SpaceRotor.
  Context {F : Type} `{FF : Field F}.
  Add Ring space_rotor_ring : (ring_theory_of_Ring (R := F) commutative).

  #[local] Instance Space_One : One (Cl F 3) := cl_one.
  #[local] Instance Space_Mult : HasMult (Cl F 3) := cl_mul sig3.

  #[local] Instance Space_Ring : Ring (Cl F 3) := Cl_Ring sig3.

  Definition sb (b1 b2 b3 : bool) : Blade 3 :=
    bcons b1 (bcons b2 (bcons b3 bnil)).

  Lemma blade3_case : ∀ D : Blade 3,
    D = sb (bhead D) (bhead (btail D)) (bhead (btail (btail D))).
  Proof.
    intros D.
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    exact (blade_0 (btail (btail (btail D)))).
  Qed.

  Definition mk3 (c a1 a2 a3 p12 p13 p23 t : F) : Cl F 3 :=
    fun D ⇒
      if bhead D
      then if bhead (btail D)
           then if bhead (btail (btail D)) then t else p12
           else if bhead (btail (btail D)) then p13 else a1
      else if bhead (btail D)
           then if bhead (btail (btail D)) then p23 else a2
           else if bhead (btail (btail D)) then a3 else c.

  Definition spinor (r b12 b13 b23 : F) : Cl F 3 :=
    mk3 r 0 0 0 b12 b13 b23 0.

  Definition space_vector (a b c : F) : Fin.t 3 → F :=
    fun i ⇒ match i with
             | Fin.F1 ⇒ a
             | Fin.FS Fin.F1 ⇒ b
             | Fin.FS (Fin.FS _) ⇒ c
             end.

  Definition rotate (R x : Cl F 3) : Cl F 3 := R × x × cl_reverse R.

  Definition rotate_coordinates (R : Cl F 3) (v : Fin.t 3 → F)
      : Fin.t 3 → F :=
    fun i ⇒ match i with
             | Fin.F1 ⇒ rotate R (ga_vec v) (sb true false false)
             | Fin.FS Fin.F1 ⇒ rotate R (ga_vec v) (sb false true false)
             | Fin.FS (Fin.FS _) ⇒ rotate R (ga_vec v) (sb false false true)
             end.

  Definition cl_odd (x : Cl F 3) : Prop :=
    ∀ D : Blade 3, bparity D = false → x D == 0.

  Lemma cl3_ext : ∀ x y : Cl F 3,
    x (sb false false false) == y (sb false false false) →
    x (sb true false false) == y (sb true false false) →
    x (sb false true false) == y (sb false true false) →
    x (sb false false true) == y (sb false false true) →
    x (sb true true false) == y (sb true true false) →
    x (sb true false true) == y (sb true false true) →
    x (sb false true true) == y (sb false true true) →
    x (sb true true true) == y (sb true true true) →
    x == y.
  Proof.
    intros x y H000 H100 H010 H001 H110 H101 H011 H111 D.
    rewrite (blade3_case D).
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D)));
      assumption.
  Qed.

  Ltac blade_reduce :=
    unfold rotate_coordinates, rotate, spinor;
    repeat change (@mult (Cl F 3) Space_Mult) with Space_Mult;
    repeat change (@one (Cl F 3) Space_One) with Space_One;
    unfold Space_Mult, Space_One, cl_mul, cl_one, cl_reverse, bsign,
      parity_sign, mk3, sb, bxor, bdiff, band;
    simpl.
  Ltac by_blade := blade_reduce; ring.


  Lemma reverse_mk3 : ∀ c a1 a2 a3 p12 p13 p23 t : F,
    cl_reverse (mk3 c a1 a2 a3 p12 p13 p23 t)
    == mk3 c a1 a2 a3 (- p12) (- p13) (- p23) (- t).
  Proof. intros. apply cl3_ext; by_blade. Qed.

  Lemma reverse_anti3 : ∀ x y : Cl F 3,
    cl_reverse (x × y) == cl_reverse y × cl_reverse x.
  Proof. intros x y. apply cl3_ext; by_blade. Qed.


  Lemma spinor_even : ∀ r b12 b13 b23 : F,
    cl_even (spinor r b12 b13 b23).
  Proof.
    intros r b12 b13 b23 D HD.
    rewrite (blade3_case D) in ×.
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D)));
      solve [ discriminate HD | reflexivity ].
  Qed.

  Lemma vec_odd : ∀ v : Fin.t 3 → F, cl_odd (ga_vec v).
  Proof. intros v D HD. exact (ga_vec_parity 3 v D HD). Qed.

  Lemma reverse_even : ∀ x : Cl F 3, cl_even x → cl_even (cl_reverse x).
  Proof.
    intros x Hx D HD.
    rewrite (cl_reverse_at 3 x D), (Hx D HD). ring.
  Qed.

  Lemma even_mul_closed : ∀ x y : Cl F 3,
    cl_even x → cl_even y → cl_even (x × y).
  Proof.
    intros x y Hx Hy.
    pose proof (Hx (sb true false false) eq_refl) as Hx1.
    pose proof (Hx (sb false true false) eq_refl) as Hx2.
    pose proof (Hx (sb false false true) eq_refl) as Hx3.
    pose proof (Hx (sb true true true) eq_refl) as Hx123.
    pose proof (Hy (sb true false false) eq_refl) as Hy1.
    pose proof (Hy (sb false true false) eq_refl) as Hy2.
    pose proof (Hy (sb false false true) eq_refl) as Hy3.
    pose proof (Hy (sb true true true) eq_refl) as Hy123.
    unfold sb in Hx1, Hx2, Hx3, Hx123, Hy1, Hy2, Hy3, Hy123.
    intros D HD.
    rewrite (blade3_case D) in ×.
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D)));
      try discriminate HD;
      blade_reduce;
      rewrite ?Hx1, ?Hx2, ?Hx3, ?Hx123, ?Hy1, ?Hy2, ?Hy3, ?Hy123; ring.
  Qed.

  Lemma even_odd_mul_closed : ∀ x y : Cl F 3,
    cl_even x → cl_odd y → cl_odd (x × y).
  Proof.
    intros x y Hx Hy.
    pose proof (Hx (sb true false false) eq_refl) as Hx1.
    pose proof (Hx (sb false true false) eq_refl) as Hx2.
    pose proof (Hx (sb false false true) eq_refl) as Hx3.
    pose proof (Hx (sb true true true) eq_refl) as Hx123.
    pose proof (Hy (sb false false false) eq_refl) as Hy0.
    pose proof (Hy (sb true true false) eq_refl) as Hy12.
    pose proof (Hy (sb true false true) eq_refl) as Hy13.
    pose proof (Hy (sb false true true) eq_refl) as Hy23.
    unfold sb in Hx1, Hx2, Hx3, Hx123, Hy0, Hy12, Hy13, Hy23.
    intros D HD.
    rewrite (blade3_case D) in ×.
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D)));
      try discriminate HD;
      blade_reduce;
      rewrite ?Hx1, ?Hx2, ?Hx3, ?Hx123, ?Hy0, ?Hy12, ?Hy13, ?Hy23; ring.
  Qed.

  Lemma odd_even_mul_closed : ∀ x y : Cl F 3,
    cl_odd x → cl_even y → cl_odd (x × y).
  Proof.
    intros x y Hx Hy.
    pose proof (Hx (sb false false false) eq_refl) as Hx0.
    pose proof (Hx (sb true true false) eq_refl) as Hx12.
    pose proof (Hx (sb true false true) eq_refl) as Hx13.
    pose proof (Hx (sb false true true) eq_refl) as Hx23.
    pose proof (Hy (sb true false false) eq_refl) as Hy1.
    pose proof (Hy (sb false true false) eq_refl) as Hy2.
    pose proof (Hy (sb false false true) eq_refl) as Hy3.
    pose proof (Hy (sb true true true) eq_refl) as Hy123.
    unfold sb in Hx0, Hx12, Hx13, Hx23, Hy1, Hy2, Hy3, Hy123.
    intros D HD.
    rewrite (blade3_case D) in ×.
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D)));
      try discriminate HD;
      blade_reduce;
      rewrite ?Hx0, ?Hx12, ?Hx13, ?Hx23, ?Hy1, ?Hy2, ?Hy3, ?Hy123; ring.
  Qed.


  Lemma vec_space : ∀ v1 v2 v3 : F,
    ga_vec (space_vector v1 v2 v3) == mk3 0 v1 v2 v3 0 0 0 0.
  Proof. intros v1 v2 v3. apply cl3_ext; by_blade. Qed.

  Lemma rotate_compose : ∀ A B x : Cl F 3,
    rotate A (rotate B x) == rotate (A × B) x.
  Proof.
    intros A B x. unfold rotate.
    rewrite (reverse_anti3 A B).
    rewrite !associative.
    reflexivity.
  Qed.

  Lemma rotate_spinor_vec : ∀ r b12 b13 b23 v1 v2 v3 : F,
    rotate (spinor r b12 b13 b23) (ga_vec (space_vector v1 v2 v3))
    == ga_vec (space_vector
         ((r×r + - (b12×b12) + - (b13×b13) + b23×b23) × v1
          + (r×b12 + r×b12 + - (b13×b23 + b13×b23)) × v2
          + (r×b13 + r×b13 + b12×b23 + b12×b23) × v3)
         ((- (r×b12 + r×b12) + - (b13×b23 + b13×b23)) × v1
          + (r×r + - (b12×b12) + b13×b13 + - (b23×b23)) × v2
          + (r×b23 + r×b23 + - (b12×b13 + b12×b13)) × v3)
         ((b12×b23 + b12×b23 + - (r×b13 + r×b13)) × v1
          + (- (r×b23 + r×b23) + - (b12×b13 + b12×b13)) × v2
          + (r×r + b12×b12 + - (b13×b13) + - (b23×b23)) × v3)).
  Proof. intros. apply cl3_ext; by_blade. Qed.

  Lemma rotate_coordinates_correct : ∀ (R : Cl F 3) (v : Fin.t 3 → F),
    cl_even R → rotate R (ga_vec v) == ga_vec (rotate_coordinates R v).
  Proof.
    intros R v HR.
    pose proof (HR (sb true false false) eq_refl) as HR1.
    pose proof (HR (sb false true false) eq_refl) as HR2.
    pose proof (HR (sb false false true) eq_refl) as HR3.
    pose proof (HR (sb true true true) eq_refl) as HR123.
    unfold sb in HR1, HR2, HR3, HR123.
    apply cl3_ext; blade_reduce;
      rewrite ?HR1, ?HR2, ?HR3, ?HR123; ring.
  Qed.

  Lemma rotate_trivector_vanish : ∀ (R : Cl F 3) (v : Fin.t 3 → F),
    cl_even R → rotate R (ga_vec v) (sb true true true) == 0.
  Proof.
    intros R v HR.
    exact (rotate_coordinates_correct R v HR (sb true true true)).
  Qed.

End SpaceRotor.
