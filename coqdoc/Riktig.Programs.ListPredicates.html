From Riktig.Structure Require Export ListRec.

The predicates a list satisfies, each one recursion over the list's own operations: every element, some element, membership, element-by-element equality, and freedom from repeats.
Section Predicates.
  Context {X L : Type} {eqX : EqRel X} `{ListRec X L}.

  Definition lall (P : X → Prop) : L → Prop :=
    list_rec Prop True (fun x p ⇒ P x ∧ p).
  Definition lany (P : X → Prop) : L → Prop :=
    list_rec Prop False (fun x p ⇒ P x ∨ p).
  Definition lmem (x : X) : L → Prop := lany (fun y ⇒ x == y).

Taking a list apart one level: the head and the tail, or nothing, rebuilt beside the answer since recursion alone sees no tail.
  Definition luncons : L → option (X × L) :=
    fun l ⇒ snd (list_rec (L × option (X × L))%type
      (empty, None)
      (fun x r ⇒ (prepend x (fst r), Some (x, fst r))) l).

  Definition leq : L → L → Prop :=
    list_rec (L → Prop)
      (fun m ⇒ match luncons m with None ⇒ True | Some _ ⇒ False end)
      (fun x r m ⇒ match luncons m with
                    | None ⇒ False
                    | Some ym ⇒ x == fst ym ∧ r (snd ym)
                    end).

  Definition ldistinct : L → Prop :=
    fun l ⇒ fst (list_rec (Prop × (X → Prop))%type
      (True, fun _ ⇒ False)
      (fun x p ⇒ ((¬ snd p x) ∧ fst p, fun y ⇒ y == x ∨ snd p y)) l).
End Predicates.
