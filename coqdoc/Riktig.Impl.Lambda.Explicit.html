Require Export Riktig.Impl.Lambda.Beta.
From Riktig Require Export Theory.Lambda.
From Riktig Require Import Lib.Calc.

Local Notation "'0'" := O.
Local Notation "'1'" := (S O).


Inductive xtm : Type :=
  | XVar : nat → xtm
  | XLam : xtm → xtm
  | XApp : xtm → xtm → xtm
  | XSub : xtm → xsub → xtm
with xsub : Type :=
  | XId : xsub
  | XShift : xsub
  | XCons : xtm → xsub → xsub
  | XUp : xsub → xsub.

Fixpoint flat (t : xtm) : tm :=
  match t with
  | XVar n ⇒ Var n
  | XLam b ⇒ Abs (flat b)
  | XApp f a ⇒ App (flat f) (flat a)
  | XSub t' s ⇒ psub (flats s) (flat t')
  end
with flats (s : xsub) : nat → tm :=
  match s with
  | XId ⇒ Var
  | XShift ⇒ fun n : nat ⇒ Var (S n)
  | XCons t s' ⇒ scons (flat t) (flats s')
  | XUp s' ⇒ pup (flats s')
  end.


Instance Xtm_EqRel : EqRel xtm := fun t u : xtm ⇒ beta_conv (flat t) (flat u).

Instance Xtm_HasVar : HasVar xtm := XVar.
Instance Xtm_HasLam : HasLam xtm := XLam.
Instance Xtm_HasApp : HasApp xtm := XApp.
Instance Xtm_HasSub : HasSub xtm xsub := XSub.
Instance Xsub_HasSubId : HasSubId xsub := XId.
Instance Xsub_HasSubShift : HasSubShift xsub := XShift.
Instance Xsub_HasSubCons : HasSubCons xtm xsub := XCons.
Instance Xsub_HasSubUp : HasSubUp xsub := XUp.

Instance Xtm_HasVal : HasVal xtm := fun _ : xtm ⇒ True.


Lemma conv_of_flat : ∀ t u : xtm, beta_conv (flat t) (flat u) → t == u.
Proof.
  intros t u Hc. exact Hc.
Qed.

Lemma flat_eq_conv : ∀ t u : xtm, flat t = flat u → t == u.
Proof.
  intros t u Heq. apply conv_of_flat. rewrite Heq. apply beta_conv_refl.
Qed.

Instance Xtm_Equivalence : Equivalence (@equ xtm Xtm_EqRel).
Proof.
  split.
  - intros t. apply beta_conv_refl.
  - intros t u Hc. apply beta_conv_sym. exact Hc.
  - intros t u v H1 H2. exact (beta_conv_trans _ _ _ H1 H2).
Qed.

Instance XLam_Proper : Proper (equ ==> equ) XLam.
Proof.
  intros b b' Hb. apply conv_of_flat. apply beta_conv_abs. exact Hb.
Qed.

Instance XApp_Proper : Proper (equ ==> equ ==> equ) XApp.
Proof.
  intros f f' Hf a a' Ha. apply conv_of_flat. apply beta_conv_app.
  - exact Hf.
  - exact Ha.
Qed.


Instance Xtm_LambdaModel : LambdaModel xtm xsub.
Proof.
  constructor.
  - exact Xtm_Equivalence.
  - exact XLam_Proper.
  - exact XApp_Proper.
  -
    intros t t' Ht. apply conv_of_flat. simpl.
    rewrite !(ren_as_psub _ S). apply beta_conv_ren. exact Ht.
  -
    intros t. apply flat_eq_conv. simpl. apply psub_var.
  - intros u s. apply flat_eq_conv. reflexivity.
  - intros f a s. apply flat_eq_conv. reflexivity.
  - intros b s. apply flat_eq_conv. reflexivity.
  - intros s. apply flat_eq_conv. reflexivity.
  -
    intros n s. apply flat_eq_conv. simpl.
    rewrite (ren_as_psub _ S). reflexivity.
  - intros n. apply flat_eq_conv. reflexivity.
  -
    intros t u s. apply flat_eq_conv. simpl.
    rewrite (ren_as_psub _ S), psub_scons_ren_shift. reflexivity.
  -
    intros t s. apply flat_eq_conv. simpl.
    rewrite !(ren_as_psub _ S), psub_ren, ren_psub. reflexivity.
  -
    intros b u _. apply conv_of_flat. apply beta_conv_step.
    apply beta_step_beta.
  - intros b. exact I.
Qed.

Theorem xtm_beta : ∀ b u : xtm, app (lam b) u == b.[u .: sub_id].
Proof.
  intros b u. exact (beta b u I).
Qed.


Theorem xvar_zero_not_one : (var 0 : xtm) =/= var 1.
Proof.
  intros Hc.
  exact (var_not_beta_conv 0 1
           (fun E : 0 = 1 ⇒ Nat.neq_succ_diag_l 0 (eq_sym E)) Hc).
Qed.


Definition xself_application : xtm := XLam (XApp (XVar 0) (XVar 0)).
Definition xomega : xtm := XApp xself_application xself_application.

Example xomega_flattens_to_omega : flat xomega = omega.
Proof. reflexivity. Qed.

Example xomega_is_not_a_value : ¬ value (flat xomega).
Proof. intros Hv. inversion Hv. Qed.

Theorem xomega_not_var_zero : xomega =/= (var 0 : xtm).
Proof.
  exact (omega_not_beta_conv_var 0).
Qed.

Theorem beta_discards_omega : app (lam (var 1)) xomega == (var 0 : xtm).
Proof.
  calc (app (lam (var 1)) xomega : xtm)
    == ((var 1).[xomega .: sub_id] : xtm) { (xtm_beta (var 1) xomega) }
    == (var 0 : xtm) { (flat_eq_conv _ _ eq_refl) }.
Qed.


Example xtm_church_if_true : ∀ g h : xtm, church_if church_true g h == g.
Proof.
  intros g h. exact (church_if_true g h I I).
Qed.

Example xtm_church_fst_pair : ∀ v w : xtm,
  app (app (app church_pair v) w) church_true == v.
Proof.
  intros v w. exact (church_fst_pair v w I I).
Qed.

Example xtm_church_snd_pair : ∀ v w : xtm,
  app (app (app church_pair v) w) church_false == w.
Proof.
  intros v w. exact (church_snd_pair v w I I).
Qed.

Example church_if_selects_distinct_branches :
  church_if church_true (var 0) (var 1) == (var 0 : xtm)
  ∧ church_if church_false (var 0) (var 1) == (var 1 : xtm)
  ∧ (var 0 : xtm) =/= var 1.
Proof.
  split; [| split].
  - exact (church_if_true (var 0) (var 1) I I).
  - exact (church_if_false (var 0) (var 1) I I).
  - exact xvar_zero_not_one.
Qed.
