Instance PW_EqRel {X M : Type} `{EqRel M} : EqRel (X → M) :=
  fun f g ⇒ ∀ x, f x == g x.

Instance PW_Zero {X M : Type} `{Zero M} : Zero (X → M) :=
  fun _ ⇒ 0.

Instance PW_Add {X M : Type} `{HasAdd M} : HasAdd (X → M) :=
  fun f g x ⇒ f x + g x.

Instance PW_Neg {X M : Type} `{HasNeg M} : HasNeg (X → M) :=
  fun f x ⇒ - f x.

Instance PW_Scale {X F V : Type} `{HasScale F V} : HasScale F (X → V) :=
  fun a f x ⇒ a *: f x.

Lemma pw_add_at {X M : Type} {e : EqRel M} {E : Equivalence e}
    {aM : HasAdd M} :
  ∀ (f g : X → M) (x : X), (f + g) x == f x + g x.
Proof. reflexivity. Qed.

Lemma pw_scale_at {X F V : Type} {e : EqRel V} {E : Equivalence e}
    {sV : HasScale F V} :
  ∀ (a : F) (f : X → V) (x : X), (a *: f) x == a *: f x.
Proof. reflexivity. Qed.

Instance PW_Equivalence {X M : Type} {e : EqRel M} {E : Equivalence e} :
  Equivalence (@PW_EqRel X M e).
Proof.
  apply Build_Equivalence.
  - intros f x. reflexivity.
  - intros f g Hfg x. symmetry. apply Hfg.
  - intros f g h Hfg Hgh x. rewrite (Hfg x). apply Hgh.
Qed.

Section PointwiseMonoid.
  Context {X M : Type} `{Monoid M}.

  Lemma pw_assoc : ∀ f g h : X → M, (f + g) + h == f + (g + h).
  Proof. intros f g h x. exact (associative (f x) (g x) (h x)). Qed.

  Lemma pw_idl : ∀ f : X → M, 0 + f == f.
  Proof. intros f x. exact (left_identity (f x)). Qed.

  Lemma pw_idr : ∀ f : X → M, f + 0 == f.
  Proof. intros f x. exact (right_identity (f x)). Qed.

  #[export] Instance pw_add_proper : Proper (equ ==> equ ==> equ) (@add (X → M) _).
  Proof.
    intros f f' Hf g g' Hg x.
    exact (op_proper _ _ (Hf x) _ _ (Hg x)).
  Qed.

  #[export] Instance PW_Monoid : Monoid (X → M) :=
    {| Monoid.semigroup :=
         {| Semigroup.magma :=
              {| Magma.equivalence := PW_Equivalence;
                 Magma.op_proper := pw_add_proper |};
            Semigroup.is_associative := pw_assoc |};
       Monoid.is_left_identity := pw_idl;
       Monoid.is_right_identity := pw_idr |}.

End PointwiseMonoid.

Section PointwiseGroup.
  Context {X M : Type} `{Group M}.

  Lemma pw_invl : ∀ f : X → M, (- f) + f == 0.
  Proof. intros f x. exact (left_inverse (f x)). Qed.

  Lemma pw_invr : ∀ f : X → M, f + (- f) == 0.
  Proof. intros f x. exact (right_inverse (f x)). Qed.

  #[export] Instance pw_neg_proper : Proper (equ ==> equ) (@neg (X → M) _).
  Proof. intros f f' Hf x. exact (neg_proper _ _ (Hf x)). Qed.

  #[export] Instance PW_Group : Group (X → M) :=
    {| Group.monoid :=
         {| Monoid.semigroup :=
              {| Semigroup.magma :=
                   {| Magma.equivalence := PW_Equivalence;

End PointwiseGroup.

Section PointwiseVectorSpace.
  Context {X F V : Type} `{VectorSpace F V}.

  #[export] Instance pw_scale_proper : Proper (equ ==> equ ==> equ) (@scale F (X → V) _).
  Proof.
    intros a a' Ha f f' Hf x.
    exact (scale_proper a a' Ha (f x) (f' x) (Hf x)).
  Qed.

  Lemma pw_scale_add : ∀ (a : F) (u v : X → V),
    a *: (u + v) == a *: u + a *: v.
  Proof. intros a u v x. exact (left_distributive a (u x) (v x)). Qed.

  Lemma pw_add_scale : ∀ (a b : F) (u : X → V),
    (a + b) *: u == a *: u + b *: u.
  Proof. intros a b u x. exact (right_distributive a b (u x)). Qed.

  Lemma pw_scale_scale : ∀ (a b : F) (u : X → V),
    (a × b) *: u == a *: (b *: u).
  Proof. intros a b u x. exact (associative _ _ _). Qed.

  Lemma pw_scale_1 : ∀ u : X → V, 1 *: u == u.
  Proof. intros u x. exact (left_identity _). Qed.

  Instance PW_ScaleLaws : ScaleLaws F (X → V) :=
    {| ScaleLaws.scale_proper := pw_scale_proper;
       ScaleLaws.is_left_distributive := pw_scale_add;
       ScaleLaws.is_right_distributive := pw_add_scale;
       ScaleLaws.is_associative := pw_scale_scale;
       ScaleLaws.is_left_identity := pw_scale_1 |}.

  #[export] Instance PW_VectorSpace : VectorSpace F (X → V) := {}.

End PointwiseVectorSpace.
