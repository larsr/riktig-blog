The syntax of polynomial signatures over a set K of constant names: the inductive, its fold, and the fold's uniqueness -- enough to instantiate the Signature structure.
Inductive poly (K : Type) : Type :=
  PZero | POne | PSelf | PConst (k : K) | PPlus (a b : poly K) | PTimes (a b : poly K).
Arguments PZero {K}. Arguments POne {K}. Arguments PSelf {K}.
Arguments PConst {K} k. Arguments PPlus {K} a b. Arguments PTimes {K} a b.

Fixpoint poly_fold {K B : Type} (z o s : B) (c : K → B) (p t : B → B → B)
    (q : poly K) : B :=
  match q with
  | PZero ⇒ z | POne ⇒ o | PSelf ⇒ s | PConst k ⇒ c k
  | PPlus a b ⇒ p (poly_fold z o s c p t a) (poly_fold z o s c p t b)
  | PTimes a b ⇒ t (poly_fold z o s c p t a) (poly_fold z o s c p t b)
  end.

Lemma poly_fold_unique : ∀ (K B : Type) (z o s : B) (c : K → B)
    (p t : B → B → B) (u : poly K → B),
  u PZero = z → u POne = o → u PSelf = s → (∀ k, u (PConst k) = c k) →
  (∀ a b, u (PPlus a b) = p (u a) (u b)) →
  (∀ a b, u (PTimes a b) = t (u a) (u b)) →
  ∀ q, u q = poly_fold z o s c p t q.
Proof.
  intros K B z o s c p t u Hz Ho Hs Hc Hp Ht q.
  induction q as [ | | | k | a IHa b IHb | a IHa b IHb]; simpl.
  - exact Hz.
  - exact Ho.
  - exact Hs.
  - exact (Hc k).
  - exact (eq_trans (Hp a b) (f_equal2 p IHa IHb)).
  - exact (eq_trans (Ht a b) (f_equal2 t IHa IHb)).
Qed.
