The reversal of a doubled algebra: reversing a + b e gives reverse a + e (reverse b), and moving the generator e to the right turns reverse b into its grade involute.
Section ReversalDouble.
  Context {F V A N : Type} `{GradedGA N F V A} (s : F).

  #[export] Instance Double_HasReverse : HasReverse (Double s A) :=
    fun x ⇒ (reverse (fst x), involute (reverse (snd x))).

  Lemma involute_reverse_involute : ∀ c : A, involute (reverse (involute c)) == reverse c.
  Proof.
    intros c.
    calc (involute (reverse (involute c)))
      == (involute (involute (reverse c))) { (involute_proper _ _ (reverse_involute c)) }
      == (reverse c) { (involute_involute (reverse c)) }.
  Qed.

  #[export] Instance Double_Reversal : Reversal (Double s A) Double_HasReverse.
  Proof.
    split.
    - intros x x' [Hx1 Hx2]. split; simpl.
      + exact (reverse_proper _ _ Hx1).
      + apply involute_proper. exact (reverse_proper _ _ Hx2).
    - intros x y. split; simpl.
      + exact (additive (f := reverse) _ _).
      + calc (involute (reverse (snd x + snd y)))
          == (involute (reverse (snd x) + reverse (snd y))) { (involute_proper _ _ (additive (f := reverse) _ _)) }
          == (involute (reverse (snd x)) + involute (reverse (snd y))) { (additive (opA := add) (opB := add) (f := involute) _ _) }.
    - intros [a b] [c d]. split; simpl.
      + calc (reverse (a × c + s *: (b × involute d)))
          == (reverse (a × c) + reverse (s *: (b × involute d)))
               { (additive (f := reverse) _ _) }
          == (reverse c × reverse a + s *: reverse (b × involute d))
               { (op_proper _ _ (reverse_anti a c) _ _ (homogeneous (f := reverse) s _)) }
          == (reverse c × reverse a + s *: (reverse (involute d) × reverse b))
               { (op_proper _ _ (reflexivity _) _ _ (scale_proper _ _ (reflexivity _) _ _ (reverse_anti b (involute d)))) }
          == (reverse c × reverse a + s *: (involute (reverse d) × reverse b))
               { (op_proper _ _ (reflexivity _) _ _ (scale_proper _ _ (reflexivity _) _ _ (op_proper _ _ (reverse_involute d) _ _ (reflexivity _)))) }
          == (reverse c × reverse a + s *: (involute (reverse d) × involute (involute (reverse b))))
               { (op_proper _ _ (reflexivity _) _ _ (scale_proper _ _ (reflexivity _) _ _ (op_proper _ _ (reflexivity _) _ _ (symmetry (involute_involute (reverse b)))))) }.
      + calc (involute (reverse (a × d + b × involute c)))
          == (involute (reverse (a × d) + reverse (b × involute c)))
               { (involute_proper _ _ (additive (f := reverse) _ _)) }
          == (involute (reverse d × reverse a + reverse (involute c) × reverse b))
               { (involute_proper _ _ (op_proper _ _ (reverse_anti a d) _ _ (reverse_anti b (involute c)))) }
          == (involute (reverse d × reverse a) + involute (reverse (involute c) × reverse b))
               { (additive (opA := add) (opB := add) (f := involute) _ _) }
          == (involute (reverse d) × involute (reverse a) + involute (reverse (involute c)) × involute (reverse b))
               { (op_proper _ _ (additive (opA := mult) (opB := mult) (f := involute) _ _) _ _ (additive (opA := mult) (opB := mult) (f := involute) _ _)) }
          == (involute (reverse d) × involute (reverse a) + reverse c × involute (reverse b))
               { (op_proper _ _ (reflexivity _) _ _ (op_proper _ _ (involute_reverse_involute c) _ _ (reflexivity _))) }
          == (reverse c × involute (reverse b) + involute (reverse d) × involute (reverse a))
               { (commutative (op := add) _ _) }.
    - intros [a b]. split; simpl.
      + exact (reverse_involutive a).
      + calc (involute (reverse (involute (reverse b))))
          == (involute (involute (reverse (reverse b)))) { (involute_proper _ _ (reverse_involute (reverse b))) }
          == (reverse (reverse b)) { (involute_involute _) }
          == b { (reverse_involutive b) }.
  Qed.
End ReversalDouble.
