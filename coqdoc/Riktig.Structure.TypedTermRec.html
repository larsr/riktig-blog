From Riktig Require Export Props.

Recursion on typed lambda terms: from one answer per constructor, a function on terms, given by its three equations, which respects equality when the answers do. The type annotation rides along untouched.
Module TypedTermRec.
  Class TypedTermRec (N Ty T : Type) {eqN : EqRel N} {eqTy : EqRel Ty} {eqT : EqRel T}
      (varT : HasVar N T) (tylamT : HasTyLam Ty T) (appT : HasApp T)
      (rec : HasTypedTermRec N Ty T) := {
    stermrec_proper : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : Ty → B → B) (a : B → B → B),
      Proper (equ ==> equ) v → Proper (equ ==> equ ==> equ) l →
      Proper (equ ==> equ ==> equ) a →
      Proper (equ ==> equ) (stermrec B v l a : T → B);
    stermrec_var : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : Ty → B → B) (a : B → B → B) (n : N),
      stermrec B v l a (var n : T) == v n;
    stermrec_tylam : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : Ty → B → B) (a : B → B → B) (A : Ty) (b : T),
      stermrec B v l a (tylam A b) == l A (stermrec B v l a b);
    stermrec_app : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : Ty → B → B) (a : B → B → B) (f u : T),
      stermrec B v l a (app f u) == a (stermrec B v l a f) (stermrec B v l a u);
  }.
End TypedTermRec.
