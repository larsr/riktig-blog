Fixpoint bigop {M : Type} `{Monoid M} (n : nat) (f : nat → M) : M :=
  match n with
  | O ⇒ 0
  | S m ⇒ f O + bigop m (fun i ⇒ f (S i))
  end.

Section BigOpMonoid.
  Context {M : Type} `{Monoid M}.

  Theorem bigop_0 : ∀ f : nat → M, bigop O f == 0.
  Proof.
    intros f. reflexivity.
  Qed.

  Theorem bigop_S : ∀ n (f : nat → M),
    bigop (S n) f == f O + bigop n (fun i ⇒ f (S i)).
  Proof.
    intros n f. reflexivity.
  Qed.

  Theorem bigop_ext : ∀ n (f g : nat → M),
    (∀ i, f i == g i) → bigop n f == bigop n g.
  Proof.
    intros n; induction n as [| n IH]; intros f g Hfg.
    - reflexivity.
    - calc (bigop (S n) f)
        == (f O + bigop n (fun i ⇒ f (S i))) { bigop_S }
        == (g O + bigop n (fun i ⇒ f (S i))) { Hfg }
        == (g O + bigop n (fun i ⇒ g (S i)))
          { IH }
        == (bigop (S n) g) { bigop_S }.
  Qed.

  Theorem bigop_zero : ∀ n : nat,
    bigop n (fun _ : nat ⇒ (0 : M)) == 0.
  Proof.
    intros n; induction n as [| n IH].
    - reflexivity.
    - calc (bigop (S n) (fun _ : nat ⇒ (0 : M)))
        == ((0 : M) + bigop n (fun _ : nat ⇒ (0 : M)))
          { bigop_S }
        == ((0 : M) + (0 : M)) { IH }
        == (0 : M) { right_identity }.
  Qed.

  Theorem bigop_support_singleton : ∀ (n j : nat) (f : nat → M),
    (j < n)%nat → (∀ k : nat, k ≠ j → f k == 0) →
    bigop n f == f j.
  Proof.
    intros n; induction n as [| n IH]; intros j f Hj Hzero.
    - inversion Hj.
    - destruct j as [| j'].
      + calc (bigop (S n) f)
          == (f O + bigop n (fun i ⇒ f (S i)))
            { bigop_S }
          == (f O + bigop n (fun _ : nat ⇒ (0 : M)))
            { (bigop_ext n (fun i ⇒ f (S i)) (fun _ ⇒ 0)
                (fun i ⇒ Hzero (S i) (not_eq_sym (O_S i)))) }
          == (f O + (0 : M)) { bigop_zero }
          == (f O) { right_identity }.
      + calc (bigop (S n) f)
          == (f O + bigop n (fun i ⇒ f (S i)))
            { bigop_S }
          == ((0 : M) + bigop n (fun i ⇒ f (S i)))
            { (Hzero O (O_S j')) }
          == (bigop n (fun i ⇒ f (S i)))
            { left_identity }
          == (f (S j'))
            { (IH j' (fun i ⇒ f (S i)) (le_S_n (S j') n Hj)
                (fun k Hk ⇒ Hzero (S k) (fun Heq ⇒ Hk (eq_add_S k j' Heq)))) }.
  Qed.

End BigOpMonoid.

Section BigOpPointwise.
  Context {X M : Type} `{Monoid M}.

  Lemma bigop_app : ∀ (n : nat) (f : nat → X → M) (x : X),
    bigop n f x == bigop n (fun k ⇒ f k x).
  Proof.
    intros n; induction n as [| n IH]; intros f x.
    - reflexivity.
    - calc (bigop (S n) f x)
        == (f O x + bigop n (fun i ⇒ f (S i)) x)
          { (bigop_S n f x) }
        == (f O x + bigop n (fun i ⇒ f (S i) x))
          { (IH (fun i ⇒ f (S i)) x) }
        == (bigop (S n) (fun k ⇒ f k x))
          { (bigop_S n (fun k ⇒ f k x)) }.
  Qed.

End BigOpPointwise.

Instance bigop_proper {M : Type} `{Monoid M} (n : nat) :
  Proper (pointwise_relation nat equ ==> equ) (@bigop M _ _ _ _ n).
Proof.
  intros f g Hfg. apply bigop_ext. exact Hfg.
Qed.


Section AddSwapFront.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.

  Lemma add_swap_front : ∀ a b c : M, a + (b + c) == b + (a + c).
  Proof.
    intros a b c.
    calc (a + (b + c))
      == ((a + b) + c) { associative }
      == ((b + a) + c) { (commutative a) }
      == (b + (a + c)) { associative }.
  Qed.

End AddSwapFront.


Section BigOpList.
  Context {M : Type} `{MM : Monoid M}.

  Fixpoint bigop_list {J : Type} (f : J → M) (l : list J) : M :=
    match l with
    | nil ⇒ 0
    | j :: l' ⇒ f j + bigop_list f l'
    end.

  Lemma bigop_list_nil : ∀ {J : Type} (f : J → M),
    bigop_list f nil == 0.
  Proof.
    intros J f.
    reflexivity.
  Qed.

  Lemma bigop_list_cons : ∀ {J : Type} (f : J → M) (j : J) (l : list J),
    bigop_list f (j :: l) == f j + bigop_list f l.
  Proof.
    intros J f j l.
    reflexivity.
  Qed.

  Lemma bigop_list_ext : ∀ {J : Type} (f g : J → M) (l : list J),
    (∀ j : J, In j l → f j == g j) →
    bigop_list f l == bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH]; intros Hfg.
    - reflexivity.
    - rewrite (bigop_list_cons f a l), (bigop_list_cons g a l).
      apply op_proper.
      + exact (Hfg a (or_introl eq_refl)).
      + exact (IH (fun j Hj ⇒ Hfg j (or_intror Hj))).
  Qed.

  Lemma bigop_list_zero_on : ∀ {J : Type} (f : J → M) (l : list J),
    (∀ j : J, In j l → f j == 0) →
    bigop_list f l == 0.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf.
    - reflexivity.
    - rewrite (bigop_list_cons f a l),
        (Hf a (or_introl eq_refl)),
        (IH (fun j Hj ⇒ Hf j (or_intror Hj))).
      exact (left_identity 0).
  Qed.

  Lemma bigop_list_map : ∀ {J K : Type} (h : J → K) (f : K → M)
      (l : list J),
    bigop_list f (map h l) == bigop_list (fun j ⇒ f (h j)) l.
  Proof.
    intros J K h f l; induction l as [| a l IH].
    - reflexivity.
    - rewrite (map_cons h a l),
        (bigop_list_cons f (h a) (map h l)),
        (bigop_list_cons (fun j ⇒ f (h j)) a l).
      apply op_proper.
      + reflexivity.
      + exact IH.
  Qed.

  Lemma bigop_list_append : ∀ {J : Type} (f : J → M) (l1 l2 : list J),
    bigop_list f (l1 ++ l2) == bigop_list f l1 + bigop_list f l2.
  Proof.
    intros J f l1 l2; induction l1 as [| a l1 IH].
    - calc (bigop_list f (nil ++ l2))
        == (0 + bigop_list f l2) { left_identity }
        == (bigop_list f nil + bigop_list f l2) { (bigop_list_nil f) }.
    - calc (bigop_list f ((a :: l1) ++ l2))
        == (f a + bigop_list f (l1 ++ l2)) { (bigop_list_cons f a (l1 ++ l2)) }
        == (f a + (bigop_list f l1 + bigop_list f l2)) { IH }
        == ((f a + bigop_list f l1) + bigop_list f l2) { associative }
        == (bigop_list f (a :: l1) + bigop_list f l2) { (bigop_list_cons f a l1) }.
  Qed.

  Lemma bigop_list_flat_map : ∀ {J K : Type} (h : J → list K) (f : K → M)
      (l : list J),
    bigop_list f (flat_map h l) == bigop_list (fun j ⇒ bigop_list f (h j)) l.
  Proof.
    intros J K h f l; induction l as [| a l IH].
    - reflexivity.
    - calc (bigop_list f (flat_map h (a :: l)))
        == (bigop_list f (h a) + bigop_list f (flat_map h l))
             { (bigop_list_append f (h a) (flat_map h l)) }
        == (bigop_list f (h a) + bigop_list (fun j ⇒ bigop_list f (h j)) l)
             { IH }
        == (bigop_list (fun j ⇒ bigop_list f (h j)) (a :: l))
             { (bigop_list_cons (fun j ⇒ bigop_list f (h j)) a l) }.
  Qed.

End BigOpList.

Section BigOpListPointwise.
  Context {X M : Type} `{Monoid M}.

  Lemma bigop_list_app : ∀ {J : Type} (f : J → X → M) (l : list J)
      (x : X),
    bigop_list f l x == bigop_list (fun j ⇒ f j x) l.
  Proof.
    intros J f l x; induction l as [| a l IH].
    - reflexivity.
    - exact (op_proper
               (f a x) (f a x) (reflexivity (f a x))
               (bigop_list f l x)
               (bigop_list (fun j ⇒ f j x) l)
               IH).
  Qed.

End BigOpListPointwise.

Section BigOpListCommutative.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.

  Lemma bigop_list_permutation : ∀ {J : Type} (f : J → M) (l l' : list J),
    Permutation l l' → bigop_list f l == bigop_list f l'.
  Proof.
    intros J f l l' Hp;
      induction Hp as [| x l1 l2 Hp IH | x y l1 | l1 l2 l3 Hp1 IH1 Hp2 IH2].
    - reflexivity.
    - rewrite (bigop_list_cons f x l1), (bigop_list_cons f x l2).
      apply op_proper.
      + reflexivity.
      + exact IH.
    - rewrite (bigop_list_cons f y (x :: l1)), (bigop_list_cons f x l1),
        (bigop_list_cons f x (y :: l1)), (bigop_list_cons f y l1).
      exact (add_swap_front (f y) (f x) (bigop_list f l1)).
    - exact (transitivity IH1 IH2).
  Qed.

  Lemma bigop_list_extract : ∀ {J : Type} {eqdec : EqDec J}
      (p : J) (f : J → M) (l : list J),
    NoDup l → In p l →
    bigop_list f l == f p + bigop_list f (remove eq_dec p l).
  Proof.
    intros J eqdec p f l; induction l as [| a l IH]; intros Hnd Hin.
    - destruct Hin.
    - inversion Hnd as [| x xs Hnotin Hnd' Hxs]; subst.
      destruct Hin as [Heq | Hin].
      + rewrite <- Heq, (remove_cons eq_dec a l),
          (notin_remove eq_dec l a Hnotin).
        exact (bigop_list_cons f a l).
      + assert (Hne : p ≠ a).
        { intros Heq. rewrite Heq in Hin. exact (Hnotin Hin). }
        calc (bigop_list f (a :: l))
          == (f a + bigop_list f l)
               { bigop_list_cons }
          == (f a + (f p + bigop_list f (remove eq_dec p l)))
               { IH }
          == (f p + (f a + bigop_list f (remove eq_dec p l)))
               { add_swap_front }
          == (f p + bigop_list f (a :: remove eq_dec p l))
               { bigop_list_cons }
          == (f p + bigop_list f (remove eq_dec p (a :: l)))
               { remove_cons_other }.
  Qed.

End BigOpListCommutative.

Section BigOpListAbGroup.
  Context {G : Type} `{AbGroup G}.

  Lemma bigop_list_add : ∀ {J : Type} (f g : J → G) (l : list J),
    bigop_list (fun j ⇒ f j + g j) l == bigop_list f l + bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH].
    - symmetry. exact (left_identity 0).
    - calc (bigop_list (fun j ⇒ f j + g j) (a :: l))
        == ((f a + g a) + bigop_list (fun j ⇒ f j + g j) l)
          { bigop_list_cons }
        == ((f a + g a) + (bigop_list f l + bigop_list g l)) { IH }
        == ((f a + bigop_list f l) + (g a + bigop_list g l))
          { add_interchange }
        == (bigop_list f (a :: l) + (g a + bigop_list g l))
          { bigop_list_cons }
        == (bigop_list f (a :: l) + bigop_list g (a :: l))
          { bigop_list_cons }.
  Qed.

  Lemma bigop_list_neg : ∀ {J : Type} (f : J → G) (l : list J),
    bigop_list (fun j ⇒ - f j) l == - bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH].
    - symmetry. exact neg_zero.
    - calc (bigop_list (fun j ⇒ - f j) (a :: l))
        == (- f a + bigop_list (fun j ⇒ - f j) l)
          { bigop_list_cons }
        == (- f a + - bigop_list f l) { IH }
        == (- (f a + bigop_list f l)) { additive[neg] }
        == (- bigop_list f (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_swap : ∀ {J K : Type} (f : J → K → G)
      (l1 : list J) (l2 : list K),
    bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) l1
      == bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) l1) l2.
  Proof.
    intros J K f l1; induction l1 as [| a l1 IH]; intros l2.
    - symmetry.
      exact (bigop_list_zero_on
               (fun k ⇒ bigop_list (fun j : J ⇒ f j k) nil) l2
               (fun k _ ⇒ reflexivity 0)).
    - calc (bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) (a :: l1))
        == (bigop_list (fun k ⇒ f a k) l2
            + bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) l1)
          { bigop_list_cons }
        == (bigop_list (fun k ⇒ f a k) l2
            + bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) l1) l2)
          { IH }
        == (bigop_list (fun k ⇒ f a k + bigop_list (fun j ⇒ f j k) l1) l2)
          { bigop_list_add }
        == (bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) (a :: l1)) l2)
          { bigop_list_ext }.
  Qed.

End BigOpListAbGroup.

Section BigOpListRing.
  Context {R : Type} `{Ring R}.

  Lemma bigop_list_mul_l : ∀ {J : Type} (a : R) (f : J → R)
      (l : list J),
    bigop_list (fun j ⇒ a × f j) l == a × bigop_list f l.
  Proof.
    intros J a f l; induction l as [| b l IH].
    - symmetry. exact (mul_0_r a).
    - calc (bigop_list (fun j ⇒ a × f j) (b :: l))
        == (a × f b + bigop_list (fun j ⇒ a × f j) l)
          { bigop_list_cons }
        == (a × f b + a × bigop_list f l) { IH }
        == (a × (f b + bigop_list f l))
          { left_distributive }
        == (a × bigop_list f (b :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_mul_r : ∀ {J : Type} (a : R) (f : J → R)
      (l : list J),
    bigop_list (fun j ⇒ f j × a) l == bigop_list f l × a.
  Proof.
    intros J a f l; induction l as [| b l IH].
    - symmetry. exact (mul_0_l a).
    - calc (bigop_list (fun j ⇒ f j × a) (b :: l))
        == (f b × a + bigop_list (fun j ⇒ f j × a) l)
          { bigop_list_cons }
        == (f b × a + bigop_list f l × a) { IH }
        == ((f b + bigop_list f l) × a)
          { right_distributive }
        == (bigop_list f (b :: l) × a) { bigop_list_cons }.
  Qed.

End BigOpListRing.

Section BigOpListVectorSpace.
  Context {F V : Type} `{VectorSpace F V}.

  Lemma bigop_list_scale : ∀ {J : Type} (a : F) (f : J → V)
      (l : list J),
    bigop_list (fun j ⇒ a *: f j) l == a *: bigop_list f l.
  Proof.
    intros J a f l; induction l as [| b l IH].
    - symmetry. exact (scale_zero a).
    - calc (bigop_list (fun j ⇒ a *: f j) (b :: l))
        == (a *: f b + bigop_list (fun j ⇒ a *: f j) l)
          { bigop_list_cons }
        == (a *: f b + a *: bigop_list f l) { IH }
        == (a *: (f b + bigop_list f l))
          { left_distributive }
        == (a *: bigop_list f (b :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_scale_r : ∀ {J : Type} (f : J → F) (v : V)
      (l : list J),
    bigop_list (fun j ⇒ f j *: v) l == bigop_list f l *: v.
  Proof.
    intros J f v l; induction l as [| b l IH].
    - symmetry. exact (zero_scale v).
    - calc (bigop_list (fun j ⇒ f j *: v) (b :: l))
        == (f b *: v + bigop_list (fun j ⇒ f j *: v) l)
          { bigop_list_cons }
        == (f b *: v + bigop_list f l *: v) { IH }
        == ((f b + bigop_list f l) *: v)
          { right_distributive }
        == (bigop_list f (b :: l) *: v) { bigop_list_cons }.
  Qed.

End BigOpListVectorSpace.

Section BigOpListOrder.
  Context {F : Type} `{OrderedField F}.

  Lemma bigop_list_nonneg : ∀ {J : Type} (f : J → F) (l : list J),
    (∀ j : J, In j l → 0 ≤ f j) → 0 ≤ bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf.
    - apply eq_le. reflexivity.
    - calc (0 : F)
        == ((0 : F) + (0 : F)) { left_identity }
        == (f a + bigop_list f l)
          { (add_le_mono 0 (f a) 0 (bigop_list f l)
              (Hf a (or_introl eq_refl))
              (IH (fun j Hj ⇒ Hf j (or_intror Hj)))) }
        == (bigop_list f (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_le : ∀ {J : Type} (f g : J → F) (l : list J),
    (∀ j : J, In j l → f j ≤ g j) →
    bigop_list f l ≤ bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH]; intros Hfg.
    - apply eq_le. reflexivity.
    - calc (bigop_list f (a :: l))
        == (f a + bigop_list f l) { bigop_list_cons }
        == (g a + bigop_list g l)
          { (add_le_mono (f a) (g a) (bigop_list f l) (bigop_list g l)
              (Hfg a (or_introl eq_refl))
              (IH (fun j Hj ⇒ Hfg j (or_intror Hj)))) }
        == (bigop_list g (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_nonneg_zero : ∀ {J : Type} (f : J → F) (l : list J),
    (∀ j : J, In j l → 0 ≤ f j) → bigop_list f l == 0 →
    ∀ j : J, In j l → f j == 0.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf Hsum j Hj.
    - destruct Hj.
    - rewrite (bigop_list_cons f a l) in Hsum.
      destruct (nonneg_add_eq_zero (f a) (bigop_list f l)
        (Hf a (or_introl eq_refl))
        (bigop_list_nonneg f l (fun j' Hj' ⇒ Hf j' (or_intror Hj')))
        Hsum) as [H1 H2].
      destruct Hj as [Heq | Hin].
      + rewrite <- Heq. exact H1.
      + exact (IH (fun j' Hj' ⇒ Hf j' (or_intror Hj')) H2 j Hin).
  Qed.

  Lemma bigop_list_le_one_term : ∀ {J : Type} (f : J → F) (l : list J)
      (j : J),
    In j l → (∀ j' : J, In j' l → 0 ≤ f j') →
    f j ≤ bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH]; intros j Hj Hf.
    - destruct Hj.
    - rewrite (bigop_list_cons f a l).
      destruct Hj as [Heq | Hin].
      + rewrite <- Heq.
        apply le_add_nonneg_r.
        exact (bigop_list_nonneg f l (fun j' Hj' ⇒ Hf j' (or_intror Hj'))).
      + calc (f j)
          == (bigop_list f l)
            { (IH j Hin (fun j' Hj' ⇒ Hf j' (or_intror Hj'))) }
          == (f a + bigop_list f l)
            { (le_add_nonneg_l (f a) (bigop_list f l)
                (Hf a (or_introl eq_refl))) }.
  Qed.

  Lemma abs_bigop_list_le : ∀ {J : Type} (f : J → F) (l : list J),
    abs (bigop_list f l) ≤ bigop_list (fun j ⇒ abs (f j)) l.
  Proof.
    intros J f l; induction l as [| a l IH].
    - apply eq_le. exact abs_zero.
    - calc (abs (bigop_list f (a :: l)))
        == (abs (f a + bigop_list f l)) { bigop_list_cons }
        == (abs (f a) + abs (bigop_list f l))
          { abs_triangle }
        == (abs (f a) + bigop_list (fun j ⇒ abs (f j)) l)
          { add_le_mono }
        == (bigop_list (fun j ⇒ abs (f j)) (a :: l))
          { bigop_list_cons }.
  Qed.

End BigOpListOrder.

Section BigOpIndexedCommutative.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.
  Context {n : nat} {I : Type} {eqdec : EqDec I} {FI : FinIndex n I}.

  Lemma bigopI_extract : ∀ (p : I) (f : I → M),
    bigopI f == f p + bigop_list f (remove eq_dec p enum).
  Proof.
    intros p f.
    exact (bigop_list_extract p f enum enum_NoDup (enum_complete p)).
  Qed.

  Lemma bigopI_support_singleton : ∀ (p : I) (f : I → M),
    (∀ i : I, i ≠ p → f i == 0) →
    bigopI f == f p.
  Proof.
    intros p f Hsupp.
    calc (bigopI f)
      == (f p + bigop_list f (remove eq_dec p enum))
           { bigopI_extract }
      == (f p + (0 : M))
           { (bigop_list_zero_on f (remove eq_dec p enum)
               (fun i Hi ⇒ Hsupp i (proj2 (in_remove eq_dec enum i p Hi)))) }
      == (f p)
           { right_identity }.
  Qed.

  Lemma bigopI_support_pair : ∀ (p q : I) (f : I → M),
    p ≠ q →
    (∀ i : I, i ≠ p → i ≠ q → f i == 0) →
    bigopI f == f p + f q.
  Proof.
    intros p q f Hpq Hsupp.
    calc (bigopI f)
      == (f p + bigop_list f (remove eq_dec p enum))
           { bigopI_extract }
      == (f p + (f q + bigop_list f (remove eq_dec q (remove eq_dec p enum))))
           { (bigop_list_extract q f (remove eq_dec p enum)
               (remove_preserves_NoDup p enum enum_NoDup)
               (in_in_remove eq_dec enum
                 (fun Heq ⇒ Hpq (eq_sym Heq)) (enum_complete q))) }
      == (f p + (f q + (0 : M)))
           { (bigop_list_zero_on f (remove eq_dec q (remove eq_dec p enum))
               (fun i Hi ⇒
                 Hsupp i
                   (proj2 (in_remove eq_dec enum i p
                     (proj1 (in_remove eq_dec (remove eq_dec p enum) i q Hi))))
                   (proj2 (in_remove eq_dec (remove eq_dec p enum) i q Hi)))) }
      == (f p + f q)
           { right_identity }.
  Qed.

  Lemma bigopI_reindex : ∀ (f : I → M) (tau tau' : I → I),
    (∀ x : I, tau' (tau x) = x) →
    (∀ y : I, tau (tau' y) = y) →
    bigopI (fun i ⇒ f (tau i)) == bigopI f.
  Proof.
    intros f tau tau' H1 H2.
    assert (Hperm : Permutation (map tau enum) enum).
    { apply NoDup_Permutation.
      - apply (NoDup_map_injective tau enum).
        + intros x y Hxy.
          rewrite <- (H1 x), <- (H1 y), Hxy.
          reflexivity.
        + exact enum_NoDup.
      - exact enum_NoDup.
      - intros x.
        split.
        + intros _. exact (enum_complete x).
        + intros _.
          rewrite <- (H2 x).
          exact (in_map tau enum (tau' x) (enum_complete (tau' x))). }
    calc (bigopI (fun i ⇒ f (tau i)))
      == (bigop_list f (map tau enum))
           { bigop_list_map }
      == (bigopI f)
           { bigop_list_permutation }.
  Qed.

End BigOpIndexedCommutative.

Section BigOpDelta.
  Context {R : Type} `{RR : Ring R}.
  Context {n : nat} {I : Type} {eqdec : EqDec I} {FI : FinIndex n I}.

  Definition delta (i j : I) : R := if dec (i = j) then 1 else 0.

  Lemma delta_at_eq : ∀ i j : I, i = j → delta i j == 1.
  Proof.
    intros i j Hij.
    unfold delta; destruct (dec (i = j)) as [_ | Hne].
    - reflexivity.
    - destruct (Hne Hij).
  Qed.

  Lemma delta_at_neq : ∀ i j : I, i ≠ j → delta i j == 0.
  Proof.
    intros i j Hij.
    unfold delta; destruct (dec (i = j)) as [Heq | _].
    - destruct (Hij Heq).
    - reflexivity.
  Qed.

  Lemma mul_delta_eq : ∀ (a : R) (i j : I), i = j → a × delta i j == a.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_eq i j Hij).
    exact (right_identity a).
  Qed.

  Lemma mul_delta_neq : ∀ (a : R) (i j : I), i ≠ j → a × delta i j == 0.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_neq i j Hij).
    exact (mul_0_r a).
  Qed.

  Lemma bigopI_delta : ∀ (c : I → R) (j : I),
    bigopI (fun i ⇒ c i × delta i j) == c j.
  Proof.
    intros c j.
    calc (bigopI (fun i ⇒ c i × delta i j))
      == (c j × delta j j)
           { (bigopI_support_singleton j (fun i ⇒ c i × delta i j)
               (fun i Hne ⇒ mul_delta_neq (c i) i j Hne)) }
      == (c j)
           { mul_delta_eq }.
  Qed.

End BigOpDelta.

Section BigOpAbGroup.
  Context {G : Type} `{AbGroup G}.

  Theorem bigop_add : ∀ n (f g : nat → G),
    bigop n (fun i ⇒ f i + g i) == bigop n f + bigop n g.
  Proof.
    intros n; induction n as [| n IH]; intros f g.
    - calc (bigop O (fun i ⇒ f i + g i))
        == (0 : G) { bigop_0 }
        == ((0 : G) + (0 : G)) { left_identity }
        == (bigop O f + (0 : G)) { bigop_0 }
        == (bigop O f + bigop O g) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ f i + g i))
        == ((f O + g O) + bigop n (fun i ⇒ f (S i) + g (S i)))
          { bigop_S }
        == ((f O + g O) + (bigop n (fun i ⇒ f (S i)) + bigop n (fun i ⇒ g (S i))))
          { IH }
        == ((f O + bigop n (fun i ⇒ f (S i))) + (g O + bigop n (fun i ⇒ g (S i))))
          { add_interchange }
        == (bigop (S n) f + (g O + bigop n (fun i ⇒ g (S i)))) { bigop_S }
        == (bigop (S n) f + bigop (S n) g) { bigop_S }.
  Qed.

  Theorem bigop_swap : ∀ (m n : nat) (f : nat → nat → G),
    bigop m (fun i ⇒ bigop n (fun j ⇒ f i j))
      == bigop n (fun j ⇒ bigop m (fun i ⇒ f i j)).
  Proof.
    intros m; induction m as [| m IH]; intros n f.
    - symmetry.
      exact (bigop_zero n).
    - calc (bigop (S m) (fun i ⇒ bigop n (fun j ⇒ f i j)))
        == (bigop n (f O) + bigop m (fun i ⇒ bigop n (fun j ⇒ f (S i) j)))
          { bigop_S }
        == (bigop n (f O) + bigop n (fun j ⇒ bigop m (fun i ⇒ f (S i) j)))
          { IH }
        == (bigop n (fun j ⇒ bigop (S m) (fun i ⇒ f i j)))
          { bigop_add }.
  Qed.

End BigOpAbGroup.

  Context {R : Type} `{Ring R}.

  Theorem bigop_mul_l : ∀ n (a : R) (f : nat → R),
    bigop n (fun i ⇒ a × f i) == a × bigop n f.
  Proof.
    intros n; induction n as [| n IH]; intros a f.
    - calc (bigop O (fun i ⇒ a × f i))
        == (0 : R) { bigop_0 }
        == (a × (0 : R)) { mul_0_r }
        == (a × bigop O f) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ a × f i))
        == (a × f O + bigop n (fun i ⇒ a × f (S i))) { bigop_S }
        == (a × f O + a × bigop n (fun i ⇒ f (S i))) { IH }
        == (a × (f O + bigop n (fun i ⇒ f (S i))))
          { left_distributive }
        == (a × bigop (S n) f) { bigop_S }.
  Qed.

  Theorem bigop_mul_r : ∀ n (a : R) (f : nat → R),
    bigop n (fun i ⇒ f i × a) == bigop n f × a.
  Proof.
    intros n; induction n as [| n IH]; intros a f.
    - calc (bigop O (fun i ⇒ f i × a))
        == (0 : R) { bigop_0 }
        == ((0 : R) × a) { mul_0_l }
        == (bigop O f × a) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ f i × a))
        == (f O × a + bigop n (fun i ⇒ f (S i) × a)) { bigop_S }
        == (f O × a + bigop n (fun i ⇒ f (S i)) × a) { IH }
        == ((f O + bigop n (fun i ⇒ f (S i))) × a)
          { right_distributive }

  Theorem bigop_scale : ∀ n (a : F) (f : nat → V),
    bigop n (fun i ⇒ a *: f i) == a *: bigop n f.
  Proof.
    intros n; induction n as [| n IH]; intros a f.
    - calc (bigop O (fun i ⇒ a *: f i))
        == (0 : V) { bigop_0 }
        == (a *: (0 : V)) { scale_zero }
        == (a *: bigop O f) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ a *: f i))
        == (a *: f O + bigop n (fun i ⇒ a *: f (S i)))
          { bigop_S }
        == (a *: f O + a *: bigop n (fun i ⇒ f (S i))) { IH }
        == (a *: (f O + bigop n (fun i ⇒ f (S i))))
          { left_distributive }
        == (a *: bigop (S n) f) { bigop_S }.
  Qed.

  Theorem bigop_scale_r : ∀ (n : nat) (f : nat → F) (v : V),
    bigop n (fun i ⇒ f i *: v) == bigop n f *: v.

    - symmetry.
      exact (zero_scale v).

        == (f O *: v + bigop n (fun i ⇒ f (S i)) *: v) { IH }

Section BigOpVectorSpaceList.
  Context {F V : Type} `{VectorSpace F V}.

  Fixpoint list_lincomb (coeffs : list F) (vecs : list V) : V :=
    match coeffs with
    | nil ⇒ 0
    | c :: coeffs' ⇒
      match vecs with
      | nil ⇒ 0
      | v :: vecs' ⇒ c *: v + list_lincomb coeffs' vecs'
      end
    end.

  Lemma list_lincomb_nil_l : ∀ vecs : list V, list_lincomb nil vecs == 0.
  Proof.
    intros vecs.
    reflexivity.
  Qed.

  Lemma list_lincomb_cons : ∀ (c : F) (coeffs' : list F) (v : V)
      (vecs' : list V),
    list_lincomb (c :: coeffs') (v :: vecs') == c *: v + list_lincomb coeffs' vecs'.
  Proof.
    intros c coeffs' v vecs'.
    reflexivity.
  Qed.

  Lemma list_lincomb_map_ext : ∀ (l : list V) (g g' : V → F),
    (∀ d : V, In d l → g d == g' d) →
    list_lincomb (map g l) l == list_lincomb (map g' l) l.
  Proof.
    intros l; induction l as [| d l' IH]; intros g g' Hext.
    - reflexivity.
    - rewrite (list_lincomb_cons (g d) (map g l') d l'),
        (list_lincomb_cons (g' d) (map g' l') d l'),
        (Hext d (or_introl eq_refl)),
        (IH g g' (fun d' Hd' ⇒ Hext d' (or_intror Hd'))).
      reflexivity.
  Qed.

End BigOpVectorSpaceList.
