Any two sums over a finite index agree: a table is the entrywise sum of its indicator tables, one per index, and the laws fix the sum of each. The decomposition is by recursion on the count, restricting the table to the indexes numbered below n.
Section BigOpUnique.
  Context {I N A T : Type} `{Finite I N} `{!LtDec N} `{!EquDec I}
          `{Monoid A} `{!Commutative A A add}
          {eqT : EqRel T} {get : HasGet T I A} {tabulate : HasTabulate T I A}
          `{!Table I A T get tabulate}
          {S1 S2 : HasBigOp T A} {B1 : BigOp I A T S1} {B2 : BigOp I A T S2}.

  Local Notation restrict t n :=
    (tabulate (fun j ⇒ if lt_dec (index j) n then get t j else 0) : T).
  Local Notation delta i a :=
    (tabulate (fun j ⇒ if equ_dec j i then a else 0) : T).

  Theorem get_restrict : ∀ (t : T) (n : N) (j : I),
    get (restrict t n) j == (if lt_dec (index j) n then get t j else 0).
  Proof. intros t n j. exact (get_tabulate (T := T) _ j). Qed.

  Theorem restrict_zero : ∀ t : T, restrict t 0 == tabulate (fun _ ⇒ 0).
  Proof.
    intros t. apply table_ext. intros j.
    rewrite get_restrict, (get_tabulate (fun _ ⇒ (0 : A)) j).
    destruct (lt_dec (index j) 0) as [Hlt | _]; [ destruct (not_lt_zero _ Hlt) | reflexivity ].
  Qed.

  Theorem restrict_card : ∀ t : T, restrict t card == t.
  Proof.
    intros t. apply table_ext. intros j.
    rewrite get_restrict.
    destruct (lt_dec (index j) card) as [_ | Hn]; [ reflexivity | destruct (Hn (index_lt j)) ].
  Qed.

  Theorem restrict_proper : ∀ (t : T) (n n' : N), n == n' → restrict t n == restrict t n'.
  Proof.
    intros t n n' Hn. apply table_ext. intros j.
    rewrite !get_restrict.
    destruct (lt_dec (index j) n) as [Hlt | Hnlt]; destruct (lt_dec (index j) n') as [Hlt' | Hnlt'].
    - reflexivity.
    - destruct (Hnlt' (proj1 (lt_proper _ _ (reflexivity _) n n' Hn) Hlt)).
    - destruct (Hnlt (proj2 (lt_proper _ _ (reflexivity _) n n' Hn) Hlt')).
    - reflexivity.
  Qed.

  Theorem delta_single : ∀ (S : HasBigOp T A) (B : BigOp I A T S) (i : I) (a : A),
    S (delta i a) == a.
  Proof.
    intros S B i a.
    apply (bigop_single (delta i a) i a).
    - rewrite (get_tabulate (T := T) _ i). destruct (equ_dec i i) as [_ | Hne]; [ reflexivity | destruct (Hne (reflexivity i)) ].
    - intros j Hji. rewrite (get_tabulate (T := T) _ j).
      destruct (equ_dec j i) as [Heq | _]; [ destruct (Hji Heq) | reflexivity ].
  Qed.

Restricting one index further adds that index's indicator.
  Theorem restrict_succ : ∀ (t : T) (n : N), n < card →
    restrict t (succ n)
      == tabulate (fun j ⇒ get (restrict t n) j + get (delta (elem n) (get t (elem n))) j).
  Proof.
    intros t n Hn. apply table_ext. intros j.
    rewrite get_restrict, (get_tabulate (T := T) _ j), get_restrict, (get_tabulate (T := T) _ j).
    destruct (lt_dec (index j) (succ n)) as [Hlt | Hnlt];
      destruct (lt_dec (index j) n) as [Hlt' | Hnlt'];
      destruct (equ_dec j (elem n)) as [Heq | Hne].
    -
      exfalso. apply (lt_irrefl n).
      assert (Hjn : index j == n).
      { calc (index j)
          == (index (elem n : I)) { Heq }
          == n { (index_elem n Hn) }. }
      exact (proj1 (lt_proper _ _ Hjn n n (reflexivity n)) Hlt').
    - symmetry. exact (right_identity (get t j)).
    -
      rewrite Heq. symmetry. exact (left_identity (get t (elem n))).
    - exfalso. destruct (proj1 (lt_succ_iff _ _) Hlt) as [Hlt' | Heq']; [ exact (Hnlt' Hlt') | ].
      apply Hne.
      calc j
        == (elem (index j) : I) { elem_index }
        == (elem n : I) { Heq' }.
    - exfalso. apply Hnlt. apply lt_succ_iff. left. exact Hlt'.
    - exfalso. apply Hnlt. apply lt_succ_iff. left. exact Hlt'.
    - exfalso. apply Hnlt. apply lt_succ_iff. right.
      calc (index j)
        == (index (elem n : I)) { Heq }
        == n { (index_elem n Hn) }.
    - symmetry. exact (left_identity 0).
  Qed.

  Theorem bigop_unique_below : ∀ (t : T) (n : N), n ≤ card →
    S1 (restrict t n) == S2 (restrict t n).
  Proof.
    intros t.
    apply (nat_induction (fun n ⇒ n ≤ card → S1 (restrict t n) == S2 (restrict t n))).
    - intros n n' Hn. split; intros Hp Hle.
      + assert (Hle' : n ≤ card) by exact (proj2 (le_proper n n' Hn card card (reflexivity card)) Hle).
        rewrite <- (restrict_proper t n n' Hn). exact (Hp Hle').
      + assert (Hle' : n' ≤ card) by exact (proj1 (le_proper n n' Hn card card (reflexivity card)) Hle).
        rewrite (restrict_proper t n n' Hn). exact (Hp Hle').
    - intros _.
      calc (S1 (restrict t 0))
        == (S1 (tabulate (fun _ ⇒ 0))) { (restrict_zero t) }
        == (0 : A) { bigop_zero }
        == (S2 (tabulate (fun _ ⇒ 0))) { bigop_zero }
        == (S2 (restrict t 0)) { (restrict_zero t) }.
    - intros n IH Hsn.
      assert (Hn : n < card) by exact (lt_of_lt_of_le n (succ n) card (lt_succ_diag n) Hsn).
      calc (S1 (restrict t (succ n)))
        == (S1 (tabulate (fun j ⇒ get (restrict t n) j + get (delta (elem n) (get t (elem n))) j)))
             { (restrict_succ t n Hn) }
        == (S1 (restrict t n) + S1 (delta (elem n) (get t (elem n)))) { bigop_add }
        == (S1 (restrict t n) + get t (elem n)) { (delta_single S1 B1 (elem n) (get t (elem n))) }
        == (S2 (restrict t n) + get t (elem n)) { (IH (le_of_lt n card Hn)) }
        == (S2 (restrict t n) + S2 (delta (elem n) (get t (elem n)))) { (delta_single S2 B2 (elem n) (get t (elem n))) }
        == (S2 (tabulate (fun j ⇒ get (restrict t n) j + get (delta (elem n) (get t (elem n))) j)))
             { bigop_add }
        == (S2 (restrict t (succ n))) { (restrict_succ t n Hn) }.
  Qed.

  Theorem bigop_unique : ∀ t : T, S1 t == S2 t.
  Proof.
    intros t.
    calc (S1 t)
      == (S1 (restrict t card)) { (restrict_card t) }
      == (S2 (restrict t card)) { (bigop_unique_below t card (le_refl card)) }
      == (S2 t) { (restrict_card t) }.
  Qed.

End BigOpUnique.

A table read through a permutation sums to the same: reading through the permutation turns a sum into a sum, and sums are unique.
Section BigOpPermutation.
  Context {I N A T : Type} `{Finite I N} `{!LtDec N} `{!EquDec I}
          `{Monoid A} `{!Commutative A A add}
          {eqT : EqRel T} {get : HasGet T I A} {tabulate : HasTabulate T I A}
          `{!Table I A T get tabulate}
          {bigop : HasBigOp T A} {B : BigOp I A T bigop}
          {p q : I → I} `{!Permutation I p q}.

  Local Notation reindex t := (tabulate (fun i ⇒ get t (p i)) : T).

  Theorem reindex_bigop : BigOp I A T (fun t ⇒ bigop (reindex t)).
  Proof.
    split.
    - intros s t Hst. apply bigop_proper. apply table_ext. intros i.
      rewrite !(get_tabulate (T := T)). rewrite Hst. reflexivity.
    - assert (Hz : reindex (tabulate (fun _ ⇒ 0)) == tabulate (fun _ ⇒ 0)).
      { apply table_ext. intros i. rewrite !(get_tabulate (T := T)). reflexivity. }
      calc (bigop (reindex (tabulate (fun _ ⇒ 0))))
        == (bigop (tabulate (fun _ ⇒ (0 : A)) : T)) { (bigop_proper _ _ Hz) }
        == (0 : A) { bigop_zero }.
    - intros s t.
      assert (Ha : reindex (tabulate (fun i ⇒ get s i + get t i))
                   == tabulate (fun i ⇒ get (reindex s) i + get (reindex t) i)).
      { apply table_ext. intros i. rewrite !(get_tabulate (T := T)). reflexivity. }
      calc (bigop (reindex (tabulate (fun i ⇒ get s i + get t i))))
        == (bigop (tabulate (fun i ⇒ get (reindex s) i + get (reindex t) i) : T))
             { (bigop_proper _ _ Ha) }
        == (bigop (reindex s) + bigop (reindex t)) { bigop_add }.
    - intros t i a Hi Hj.
      apply (bigop_single (reindex t) (q i) a).
      + rewrite (get_tabulate (T := T) _ (q i)).
        calc (get t (p (q i)))
          == (get t i) { (to_from (to := p) i) }
          == a { Hi }.
      + intros j Hjq. rewrite (get_tabulate (T := T) _ j). apply Hj.
        intros Hpj. apply Hjq.
        calc j
          == (q (p j)) { (from_to (to := p) j) }
          == (q i) { Hpj }.
  Qed.

  Theorem bigop_permutation : ∀ t : T, bigop (reindex t) == bigop t.
  Proof.
    intros t.
    exact (bigop_unique (S1 := fun t ⇒ bigop (reindex t)) (S2 := bigop)
             (B1 := reindex_bigop) (B2 := B) t).
  Qed.

End BigOpPermutation.
