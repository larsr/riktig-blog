bigop n f is the bounded sum f 0 + (f 1 + (... + 0)).
Class HasBigop (M : Type) := bigop : nat → (nat → M) → M.
Hint Mode HasBigop ! : typeclass_instances.
Typeclasses Transparent HasBigop.

Module Bigop.
  Class Bigop (M : Type) {Me : EqRel M} {Mz : Zero M} {Ma : HasAdd M}
      {MM : Monoid M} {Mb : HasBigop M} := {
    is_bigop_0 : ∀ f : nat → M, bigop O f == 0;
    is_bigop_S : ∀ (n : nat) (f : nat → M),
      bigop (S n) f == f O + bigop n (fun i ⇒ f (S i));
  }.
  Hint Mode Bigop ! - - - - - : typeclass_instances.
End Bigop.
Export Bigop.

Lemma bigop_0 {M : Type} `{Monoid M} {Mb : HasBigop M} {MbL : Bigop M} :
  ∀ f : nat → M, bigop O f == 0.
Proof. exact is_bigop_0. Qed.

Lemma bigop_S {M : Type} `{Monoid M} {Mb : HasBigop M} {MbL : Bigop M} :
  ∀ (n : nat) (f : nat → M),
    bigop (S n) f == f O + bigop n (fun i ⇒ f (S i)).
Proof. exact is_bigop_S. Qed.

bigop_list f l is the sum of f over an explicit index list.
Class HasBigopList (M : Type) :=
  bigop_list : ∀ {J : Type}, (J → M) → list J → M.
Hint Mode HasBigopList ! : typeclass_instances.
Typeclasses Transparent HasBigopList.

Module BigopList.

      {MM : Monoid M} {Ml : HasBigopList M} := {
    is_bigop_list_nil : ∀ {J : Type} (f : J → M),
      bigop_list f nil == 0;
    is_bigop_list_cons : ∀ {J : Type} (f : J → M) (j : J) (l : list J),
      bigop_list f (j :: l) == f j + bigop_list f l;
  }.
  Hint Mode BigopList ! - - - - - : typeclass_instances.

Lemma bigop_list_nil {M : Type} `{Monoid M} {Ml : HasBigopList M}
    {MlL : BigopList M} :
  ∀ {J : Type} (f : J → M), bigop_list f nil == 0.

Lemma bigop_list_cons {M : Type} `{Monoid M} {Ml : HasBigopList M}
    {MlL : BigopList M} :

    bigop_list f (j :: l) == f j + bigop_list f l.

Section BigOpMonoid.
  Context {M : Type} `{Monoid M} {Mb : HasBigop M} {MbL : Bigop M}.

  Theorem bigop_ext : ∀ n (f g : nat → M),
    (∀ i, f i == g i) → bigop n f == bigop n g.
  Proof.
    intros n; induction n as [| n IH]; intros f g Hfg.
    - calc (bigop O f)
        == (0 : M) { bigop_0 }
        == (bigop O g) { bigop_0 }.
    - calc (bigop (S n) f)
        == (f O + bigop n (fun i ⇒ f (S i))) { bigop_S }
        == (g O + bigop n (fun i ⇒ f (S i))) { Hfg }
        == (g O + bigop n (fun i ⇒ g (S i)))
          { IH }
        == (bigop (S n) g) { bigop_S }.
  Qed.

  Theorem bigop_ext_lt : ∀ n (f g : nat → M),
    (∀ i, (i < n)%nat → f i == g i) → bigop n f == bigop n g.
  Proof.
    intros n; induction n as [| n IH]; intros f g Hfg.
    - calc (bigop O f)
        == (0 : M) { bigop_0 }
        == (bigop O g) { bigop_0 }.
    - calc (bigop (S n) f)
        == (f O + bigop n (fun i ⇒ f (S i))) { bigop_S }
        == (g O + bigop n (fun i ⇒ f (S i)))
          { (Hfg O (Nat.lt_0_succ n)) }
        == (g O + bigop n (fun i ⇒ g (S i)))
          { (IH (fun i ⇒ f (S i)) (fun i ⇒ g (S i))
              (fun i Hi ⇒ Hfg (S i) (proj1 (Nat.succ_lt_mono i n) Hi))) }
        == (bigop (S n) g) { bigop_S }.
  Qed.

  Theorem bigop_snoc : ∀ (n : nat) (f : nat → M),
    bigop (S n) f == bigop n f + f n.
  Proof.
    intros n; induction n as [| n IH]; intros f.
    - calc (bigop 1 f)
        == (f O + bigop O (fun i ⇒ f (S i))) { bigop_S }
        == (f O + (0 : M)) { bigop_0 }
        == (f O) { right_identity }
        == ((0 : M) + f O) { left_identity }
        == (bigop O f + f O) { bigop_0 }.
    - calc (bigop (S (S n)) f)
        == (f O + bigop (S n) (fun i ⇒ f (S i))) { bigop_S }
        == (f O + (bigop n (fun i ⇒ f (S i)) + f (S n)))
          { (IH (fun i ⇒ f (S i))) }
        == ((f O + bigop n (fun i ⇒ f (S i))) + f (S n)) { associative }
        == (bigop (S n) f + f (S n)) { bigop_S }.
  Qed.

  Theorem bigop_zero : ∀ n : nat,
    bigop n (fun _ : nat ⇒ (0 : M)) == 0.
  Proof.
    intros n; induction n as [| n IH].
    - exact (bigop_0 (fun _ : nat ⇒ (0 : M))).
    - calc (bigop (S n) (fun _ : nat ⇒ (0 : M)))
        == ((0 : M) + bigop n (fun _ : nat ⇒ (0 : M)))
          { bigop_S }
        == ((0 : M) + (0 : M)) { IH }
        == (0 : M) { right_identity }.
  Qed.

  Theorem bigop_support_singleton : ∀ (n j : nat) (f : nat → M),
    (j < n)%nat → (∀ k : nat, k ≠ j → f k == 0) →
    bigop n f == f j.
  Proof.
    intros n; induction n as [| n IH]; intros j f Hj Hzero.
    - inversion Hj.
    - destruct j as [| j'].
      + calc (bigop (S n) f)
          == (f O + bigop n (fun i ⇒ f (S i)))
            { bigop_S }
          == (f O + bigop n (fun _ : nat ⇒ (0 : M)))
            { (bigop_ext n (fun i ⇒ f (S i)) (fun _ ⇒ 0)
                (fun i ⇒ Hzero (S i) (not_eq_sym (O_S i)))) }
          == (f O + (0 : M)) { bigop_zero }
          == (f O) { right_identity }.
      + calc (bigop (S n) f)
          == (f O + bigop n (fun i ⇒ f (S i)))
            { bigop_S }
          == ((0 : M) + bigop n (fun i ⇒ f (S i)))
            { (Hzero O (O_S j')) }
          == (bigop n (fun i ⇒ f (S i)))
            { left_identity }
          == (f (S j'))
            { (IH j' (fun i ⇒ f (S i)) (le_S_n (S j') n Hj)
                (fun k Hk ⇒ Hzero (S k) (fun Heq ⇒ Hk (eq_add_S k j' Heq)))) }.
  Qed.

End BigOpMonoid.

Section BigOpPointwise.
  Context {X M : Type} `{Monoid M} {Mb : HasBigop M} {MbL : Bigop M}.
  Context {MbX : HasBigop (X → M)} {MbXL : Bigop (X → M)}.

  Lemma bigop_app : ∀ (n : nat) (f : nat → X → M) (x : X),
    bigop n f x == bigop n (fun k ⇒ f k x).
  Proof.
    intros n; induction n as [| n IH]; intros f x.
    - calc (bigop O f x)
        == (0 : M)
          { (bigop_0 f x) }
        == (bigop O (fun k ⇒ f k x))
          { (symmetry (bigop_0 (fun k ⇒ f k x))) }.
    - calc (bigop (S n) f x)
        == (f O x + bigop n (fun i ⇒ f (S i)) x)
          { (bigop_S n f x) }
        == (f O x + bigop n (fun i ⇒ f (S i) x))
          { (IH (fun i ⇒ f (S i)) x) }
        == (bigop (S n) (fun k ⇒ f k x))
          { (bigop_S n (fun k ⇒ f k x)) }.
  Qed.

End BigOpPointwise.

Instance bigop_proper {M : Type} `{Monoid M} {Mb : HasBigop M}
    {MbL : Bigop M} (n : nat) :
  Proper (pointwise_relation nat equ ==> equ) (@bigop M Mb n).
Proof.
  intros f g Hfg. apply bigop_ext. exact Hfg.
Qed.


Section AddSwapFront.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.

  Lemma add_swap_front : ∀ a b c : M, a + (b + c) == b + (a + c).
  Proof.
    intros a b c.
    calc (a + (b + c))
      == ((a + b) + c) { associative }
      == ((b + a) + c) { (commutative a) }
      == (b + (a + c)) { associative }.
  Qed.

End AddSwapFront.


Section BigOpList.
  Context {M : Type} `{MM : Monoid M} {Ml : HasBigopList M}
    {MlL : BigopList M}.

  Lemma bigop_list_ext : ∀ {J : Type} (f g : J → M) (l : list J),
    (∀ j : J, In j l → f j == g j) →
    bigop_list f l == bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH]; intros Hfg.
    - calc (bigop_list f nil)
        == (0 : M) { bigop_list_nil }
        == (bigop_list g nil) { bigop_list_nil }.
    - rewrite (bigop_list_cons f a l), (bigop_list_cons g a l).
      apply op_proper.
      + exact (Hfg a (or_introl eq_refl)).
      + exact (IH (fun j Hj ⇒ Hfg j (or_intror Hj))).
  Qed.

  Lemma bigop_list_zero_on : ∀ {J : Type} (f : J → M) (l : list J),
    (∀ j : J, In j l → f j == 0) →
    bigop_list f l == 0.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf.
    - exact (bigop_list_nil f).
    - rewrite (bigop_list_cons f a l),
        (Hf a (or_introl eq_refl)),
        (IH (fun j Hj ⇒ Hf j (or_intror Hj))).
      exact (left_identity 0).
  Qed.

  Lemma bigop_list_map : ∀ {J K : Type} (h : J → K) (f : K → M)
      (l : list J),
    bigop_list f (map h l) == bigop_list (fun j ⇒ f (h j)) l.
  Proof.
    intros J K h f l; induction l as [| a l IH].
    - calc (bigop_list f (map h nil))
        == (0 : M)
          { bigop_list_nil }
        == (bigop_list (fun j ⇒ f (h j)) nil)
          { bigop_list_nil }.
    - rewrite (map_cons h a l),
        (bigop_list_cons f (h a) (map h l)),
        (bigop_list_cons (fun j ⇒ f (h j)) a l).
      apply op_proper.
      + reflexivity.
      + exact IH.
  Qed.

  Lemma bigop_list_append : ∀ {J : Type} (f : J → M) (l1 l2 : list J),
    bigop_list f (l1 ++ l2) == bigop_list f l1 + bigop_list f l2.
  Proof.
    intros J f l1 l2; induction l1 as [| a l1 IH].
    - calc (bigop_list f (nil ++ l2))
        == (0 + bigop_list f l2) { left_identity }
        == (bigop_list f nil + bigop_list f l2) { (bigop_list_nil f) }.
    - calc (bigop_list f ((a :: l1) ++ l2))
        == (f a + bigop_list f (l1 ++ l2)) { (bigop_list_cons f a (l1 ++ l2)) }
        == (f a + (bigop_list f l1 + bigop_list f l2)) { IH }
        == ((f a + bigop_list f l1) + bigop_list f l2) { associative }
        == (bigop_list f (a :: l1) + bigop_list f l2) { (bigop_list_cons f a l1) }.
  Qed.

  Lemma bigop_list_flat_map : ∀ {J K : Type} (h : J → list K) (f : K → M)
      (l : list J),
    bigop_list f (flat_map h l) == bigop_list (fun j ⇒ bigop_list f (h j)) l.
  Proof.
    intros J K h f l; induction l as [| a l IH].
    - calc (bigop_list f (flat_map h nil))
        == (0 : M)
          { bigop_list_nil }
        == (bigop_list (fun j ⇒ bigop_list f (h j)) nil)
          { bigop_list_nil }.
    - calc (bigop_list f (flat_map h (a :: l)))
        == (bigop_list f (h a) + bigop_list f (flat_map h l))
             { (bigop_list_append f (h a) (flat_map h l)) }
        == (bigop_list f (h a) + bigop_list (fun j ⇒ bigop_list f (h j)) l)
             { IH }
        == (bigop_list (fun j ⇒ bigop_list f (h j)) (a :: l))
             { (bigop_list_cons (fun j ⇒ bigop_list f (h j)) a l) }.
  Qed.

End BigOpList.

Section BigOpListPointwise.
  Context {X M : Type} `{Monoid M} {Ml : HasBigopList M}
    {MlL : BigopList M}.
  Context {MlX : HasBigopList (X → M)} {MlXL : BigopList (X → M)}.

  Lemma bigop_list_app : ∀ {J : Type} (f : J → X → M) (l : list J)
      (x : X),
    bigop_list f l x == bigop_list (fun j ⇒ f j x) l.
  Proof.
    intros J f l x; induction l as [| a l IH].
    - calc (bigop_list f nil x)
        == (0 : M)
          { (bigop_list_nil f x) }
        == (bigop_list (fun j ⇒ f j x) nil)
          { (symmetry (bigop_list_nil (fun j ⇒ f j x))) }.
    - calc (bigop_list f (a :: l) x)
        == (f a x + bigop_list f l x)
          { (bigop_list_cons f a l x) }
        == (f a x + bigop_list (fun j ⇒ f j x) l)
          { IH }
        == (bigop_list (fun j ⇒ f j x) (a :: l))
          { bigop_list_cons }.
  Qed.

End BigOpListPointwise.

Section BigOpListCommutative.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.
  Context {Ml : HasBigopList M} {MlL : BigopList M}.

  Lemma bigop_list_permutation : ∀ {J : Type} {eJ : EqRel J}
      {JE : Equivalence (@equ J eJ)} (f : J → M),
    Proper (equ ==> equ) f →
    ∀ l l' : list J,
    PermutationA equ l l' → bigop_list f l == bigop_list f l'.
  Proof.
    intros J eJ JE f Pf l l' Hp;
      induction Hp as [| x y l1 l2 Hxy _ IH | x y l1 | l1 l2 l3 _ IH1 _ IH2].
    - reflexivity.
    - rewrite (bigop_list_cons f x l1), (bigop_list_cons f y l2).
      apply op_proper.
      + exact (Pf x y Hxy).
      + exact IH.
    - rewrite (bigop_list_cons f y (x :: l1)), (bigop_list_cons f x l1),
        (bigop_list_cons f x (y :: l1)), (bigop_list_cons f y l1).
      exact (add_swap_front (f y) (f x) (bigop_list f l1)).
    - exact (transitivity IH1 IH2).
  Qed.

  Lemma bigop_list_extract : ∀ {J : Type} {eJ : EqRel J}
      {JE : Equivalence (@equ J eJ)} {Jeq : EquDec J}
      (p : J) (f : J → M),
    Proper (equ ==> equ) f →
    ∀ l : list J,
    NoDupA equ l → InA equ p l →
    bigop_list f l == f p + bigop_list f (removeA equ_dec p l).
  Proof.
    intros J eJ JE Jeq p f Pf l; induction l as [| a l IH]; intros Hnd Hin.
    - inversion Hin.
    - inversion Hnd as [| a0 l0 Hnotin Hnd' Heql]; subst.
      simpl removeA.
      destruct (equ_dec p a) as [Heq | Hne].
      + assert (Hpl : ¬ InA equ p l).
        { intros Hpin. exact (Hnotin (InA_eqA JE Heq Hpin)). }
        rewrite (removeA_notin equ_dec p l Hpl).
        calc (bigop_list f (a :: l))
          == (f a + bigop_list f l) { bigop_list_cons }
          == (f p + bigop_list f l) { (Pf a p (symmetry Heq)) }.
      + inversion Hin as [x0 l0 Hpa | x0 l0 Hin']; subst.
        × destruct (Hne Hpa).
        × calc (bigop_list f (a :: l))
            == (f a + bigop_list f l)
                 { bigop_list_cons }
            == (f a + (f p + bigop_list f (removeA equ_dec p l)))
                 { (IH Hnd' Hin') }
            == (f p + (f a + bigop_list f (removeA equ_dec p l)))
                 { add_swap_front }
            == (f p + bigop_list f (a :: removeA equ_dec p l))
                 { bigop_list_cons }.
  Qed.

End BigOpListCommutative.

Section BigOpListAbGroup.
  Context {G : Type} `{AbGroup G} {Ml : HasBigopList G}
    {MlL : BigopList G}.

  Lemma bigop_list_add : ∀ {J : Type} (f g : J → G) (l : list J),
    bigop_list (fun j ⇒ f j + g j) l == bigop_list f l + bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH].
    - calc (bigop_list (fun j ⇒ f j + g j) nil)
        == (0 : G) { bigop_list_nil }
        == ((0 : G) + (0 : G)) { left_identity }
        == (bigop_list f nil + (0 : G)) { bigop_list_nil }
        == (bigop_list f nil + bigop_list g nil) { (bigop_list_nil g) }.
    - calc (bigop_list (fun j ⇒ f j + g j) (a :: l))
        == ((f a + g a) + bigop_list (fun j ⇒ f j + g j) l)
          { bigop_list_cons }
        == ((f a + g a) + (bigop_list f l + bigop_list g l)) { IH }
        == ((f a + bigop_list f l) + (g a + bigop_list g l))
          { add_interchange }
        == (bigop_list f (a :: l) + (g a + bigop_list g l))
          { bigop_list_cons }
        == (bigop_list f (a :: l) + bigop_list g (a :: l))
          { bigop_list_cons }.
  Qed.

  Lemma bigop_list_neg : ∀ {J : Type} (f : J → G) (l : list J),
    bigop_list (fun j ⇒ - f j) l == - bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH].
    - calc (bigop_list (fun j ⇒ - f j) nil)
        == (0 : G) { bigop_list_nil }
        == (- (0 : G)) { neg_zero }
        == (- bigop_list f nil) { bigop_list_nil }.
    - calc (bigop_list (fun j ⇒ - f j) (a :: l))
        == (- f a + bigop_list (fun j ⇒ - f j) l)
          { bigop_list_cons }
        == (- f a + - bigop_list f l) { IH }
        == (- (f a + bigop_list f l)) { additive[neg] }
        == (- bigop_list f (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_swap : ∀ {J K : Type} (f : J → K → G)
      (l1 : list J) (l2 : list K),
    bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) l1
      == bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) l1) l2.
  Proof.
    intros J K f l1; induction l1 as [| a l1 IH]; intros l2.
    - calc (bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) nil)
        == (0 : G)
          { bigop_list_nil }
        == (bigop_list (fun k ⇒ bigop_list (fun j : J ⇒ f j k) nil) l2)
          { (bigop_list_zero_on
              (fun k ⇒ bigop_list (fun j : J ⇒ f j k) nil) l2
              (fun k _ ⇒ bigop_list_nil (fun j : J ⇒ f j k))) }.
    - calc (bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) (a :: l1))
        == (bigop_list (fun k ⇒ f a k) l2
            + bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) l1)
          { bigop_list_cons }
        == (bigop_list (fun k ⇒ f a k) l2
            + bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) l1) l2)
          { IH }
        == (bigop_list (fun k ⇒ f a k + bigop_list (fun j ⇒ f j k) l1) l2)
          { bigop_list_add }
        == (bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) (a :: l1)) l2)
          { (bigop_list_ext
              (fun k ⇒ f a k + bigop_list (fun j ⇒ f j k) l1)
              (fun k ⇒ bigop_list (fun j ⇒ f j k) (a :: l1)) l2

Section BigOpListOrder.
  Context {F : Type} `{OrderedField F} {Ml : HasBigopList F}
    {MlL : BigopList F}.

  Lemma bigop_list_nonneg : ∀ {J : Type} (f : J → F) (l : list J),
    (∀ j : J, In j l → 0 ≤ f j) → 0 ≤ bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf.
    - apply eq_le. symmetry. exact (bigop_list_nil f).
    - calc (0 : F)
        == ((0 : F) + (0 : F)) { left_identity }
        == (f a + bigop_list f l)
          { (add_le_mono 0 (f a) 0 (bigop_list f l)
              (Hf a (or_introl eq_refl))
              (IH (fun j Hj ⇒ Hf j (or_intror Hj)))) }
        == (bigop_list f (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_le : ∀ {J : Type} (f g : J → F) (l : list J),
    (∀ j : J, In j l → f j ≤ g j) →
    bigop_list f l ≤ bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH]; intros Hfg.
    - apply eq_le.
      calc (bigop_list f nil)
        == (0 : F) { bigop_list_nil }
        == (bigop_list g nil) { bigop_list_nil }.
    - calc (bigop_list f (a :: l))
        == (f a + bigop_list f l) { bigop_list_cons }
        == (g a + bigop_list g l)
          { (add_le_mono (f a) (g a) (bigop_list f l) (bigop_list g l)
              (Hfg a (or_introl eq_refl))
              (IH (fun j Hj ⇒ Hfg j (or_intror Hj)))) }
        == (bigop_list g (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_nonneg_zero : ∀ {J : Type} (f : J → F) (l : list J),
    (∀ j : J, In j l → 0 ≤ f j) → bigop_list f l == 0 →
    ∀ j : J, In j l → f j == 0.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf Hsum j Hj.
    - destruct Hj.
    - rewrite (bigop_list_cons f a l) in Hsum.
      destruct (nonneg_add_eq_zero (f a) (bigop_list f l)
        (Hf a (or_introl eq_refl))
        (bigop_list_nonneg f l (fun j' Hj' ⇒ Hf j' (or_intror Hj')))
        Hsum) as [H1 H2].
      destruct Hj as [Heq | Hin].
      + rewrite <- Heq. exact H1.
      + exact (IH (fun j' Hj' ⇒ Hf j' (or_intror Hj')) H2 j Hin).
  Qed.

  Lemma bigop_list_le_one_term : ∀ {J : Type} (f : J → F) (l : list J)
      (j : J),
    In j l → (∀ j' : J, In j' l → 0 ≤ f j') →
    f j ≤ bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH]; intros j Hj Hf.
    - destruct Hj.
    - rewrite (bigop_list_cons f a l).
      destruct Hj as [Heq | Hin].
      + rewrite <- Heq.
        apply le_add_nonneg_r.
        exact (bigop_list_nonneg f l (fun j' Hj' ⇒ Hf j' (or_intror Hj'))).
      + calc (f j)
          == (bigop_list f l)
            { (IH j Hin (fun j' Hj' ⇒ Hf j' (or_intror Hj'))) }
          == (f a + bigop_list f l)
            { (le_add_nonneg_l (f a) (bigop_list f l)
                (Hf a (or_introl eq_refl))) }.
  Qed.

  Lemma abs_bigop_list_le : ∀ {J : Type} (f : J → F) (l : list J),
    abs (bigop_list f l) ≤ bigop_list (fun j ⇒ abs (f j)) l.
  Proof.
    intros J f l; induction l as [| a l IH].
    - apply eq_le.
      calc (abs (bigop_list f nil))
        == (abs (0 : F)) { bigop_list_nil }
        == (0 : F) { abs_zero }
        == (bigop_list (fun j ⇒ abs (f j)) nil) { bigop_list_nil }.
    - calc (abs (bigop_list f (a :: l)))
        == (abs (f a + bigop_list f l)) { bigop_list_cons }
        == (abs (f a) + abs (bigop_list f l))
          { abs_triangle }
        == (abs (f a) + bigop_list (fun j ⇒ abs (f j)) l)
          { add_le_mono }
        == (bigop_list (fun j ⇒ abs (f j)) (a :: l))
          { bigop_list_cons }.
  Qed.

End BigOpListOrder.

Section BigOpIndexedCommutative.

      == (bigop_list f enum) { bigopI_def }

    Proper (equ ==> equ) f →
    (∀ i : I, ¬ (i == p) → f i == 0) →
    bigopI f == f p.
  Proof.
    intros p f Pf Hsupp.
    calc (bigopI f)
      == (f p + bigop_list f (removeA equ_dec p enum))
           { (bigopI_extract p f Pf) }
      == (f p + (0 : M))
           { (bigop_list_zero_on f (removeA equ_dec p enum)
               (fun i Hi ⇒
                 Hsupp i
                   (fun Hip ⇒
                     proj2 (proj1 (removeA_InA enum_equivalence equ_dec
                                     enum p i)
                             (In_InA enum_equivalence Hi))
                       (symmetry Hip)))) }
      == (f p)
           { right_identity }.
  Qed.

  Lemma bigopI_support_pair : ∀ (p q : I) (f : I → M),
    Proper (equ ==> equ) f →
    ¬ (p == q) →
    (∀ i : I, ¬ (i == p) → ¬ (i == q) → f i == 0) →
    bigopI f == f p + f q.
  Proof.
    intros p q f Pf Hpq Hsupp.
    calc (bigopI f)
      == (f p + bigop_list f (removeA equ_dec p enum))
           { (bigopI_extract p f Pf) }
      == (f p + (f q + bigop_list f (removeA equ_dec q (removeA equ_dec p enum))))
           { (bigop_list_extract q f Pf (removeA equ_dec p enum)
               (removeA_NoDupA enum_equivalence equ_dec p enum_NoDup)
               (proj2 (removeA_InA enum_equivalence equ_dec enum p q)
                 (conj (enum_complete q) (fun Heq ⇒ Hpq Heq)))) }
      == (f p + (f q + (0 : M)))
           { (bigop_list_zero_on f
               (removeA equ_dec q (removeA equ_dec p enum))
               (fun i Hi ⇒
                 let HiA := In_InA enum_equivalence Hi in
                 let Hq := proj1 (removeA_InA enum_equivalence equ_dec
                                    (removeA equ_dec p enum) q i) HiA in
                 let Hp := proj1 (removeA_InA enum_equivalence equ_dec
                                    enum p i) (proj1 Hq) in
                 Hsupp i
                   (fun Hip ⇒ proj2 Hp (symmetry Hip))
                   (fun Hiq ⇒ proj2 Hq (symmetry Hiq)))) }
      == (f p + f q)
           { right_identity }.
  Qed.

  Lemma bigopI_reindex : ∀ (f : I → M) (tau tau' : I → I),
    Proper (equ ==> equ) f →
    Proper (equ ==> equ) tau → Proper (equ ==> equ) tau' →
    (∀ x : I, tau' (tau x) == x) →
    (∀ y : I, tau (tau' y) == y) →
    bigopI (fun i ⇒ f (tau i)) == bigopI f.
  Proof.
    intros f tau tau' Pf Ptau Ptau' H1 H2.
    assert (Hperm : PermutationA equ (map tau enum) enum).
    { apply NoDupA_equivlistA_PermutationA.
      - exact enum_equivalence.
      - apply (NoDupA_map_injectiveA tau enum).
        + intros x y Hxy.
          calc x
            == (tau' (tau x)) { (symmetry (H1 x)) }
            == (tau' (tau y)) { (Ptau' (tau x) (tau y) Hxy) }
            == y { (H1 y) }.
        + exact enum_NoDup.
      - exact enum_NoDup.
      - intros x.
        split; intros _.
        + exact (enum_complete x).
        + apply (InA_eqA enum_equivalence (x := tau (tau' x))).
          × exact (H2 x).
          × exact (InA_map tau Ptau enum (tau' x) (enum_complete (tau' x))). }
    calc (bigopI (fun i ⇒ f (tau i)))
      == (bigop_list (fun i ⇒ f (tau i)) enum)
           { bigopI_def }
      == (bigop_list f (map tau enum))
           { (symmetry (bigop_list_map tau f enum)) }
      == (bigop_list f enum)
           { (bigop_list_permutation f Pf (map tau enum) enum Hperm) }
      == (bigopI f)
           { bigopI_def }.
  Qed.

      == (bigop_list (fun i ⇒ f i *: v) enum) { bigopI_def }
      == (bigop_list f enum *: v) { bigop_list_scale_r }
      == (bigopI f *: v) { bigopI_def }.
  Qed.

End BigOpIndexedVectorSpace.

Section BigOpIndexedOrder.
  Context {F : Type} `{OrderedField F}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {FI : FinIndex n I}.
  Context {Ml : HasBigopList F} {MlL : BigopList F}.
  Context {Mi : HasBigopI I F} {MiL : BigopI n I F}.

  Lemma bigopI_nonneg : ∀ f : I → F,
    (∀ i, 0 ≤ f i) → 0 ≤ bigopI f.
  Proof.
    intros f Hf.
    rewrite (bigopI_def f).
    exact (bigop_list_nonneg f enum (fun i _ ⇒ Hf i)).
  Qed.

  Lemma bigopI_le : ∀ f g : I → F,
    (∀ i, f i ≤ g i) → bigopI f ≤ bigopI g.
  Proof.
    intros f g Hfg.
    rewrite (bigopI_def f), (bigopI_def g).
    exact (bigop_list_le f g enum (fun i _ ⇒ Hfg i)).
  Qed.

  Lemma bigopI_nonneg_zero : ∀ f : I → F,
    Proper (equ ==> equ) f →
    (∀ i, 0 ≤ f i) → bigopI f == 0 → ∀ i, f i == 0.
  Proof.
    intros f Pf Hf Hsum i.
    destruct (proj1 (InA_alt equ i enum) (enum_complete i))
      as [a [Hia Ha]].
    calc (f i)
      == (f a) { (Pf i a Hia) }
      == (0 : F) { (bigop_list_nonneg_zero f enum (fun i' _ ⇒ Hf i')
                     (transitivity (symmetry (bigopI_def f)) Hsum)
                     a Ha) }.
  Qed.

  Lemma bigopI_le_one_term : ∀ (f : I → F) (j : I),
    Proper (equ ==> equ) f →
    (∀ i, 0 ≤ f i) → f j ≤ bigopI f.
  Proof.
    intros f j Pf Hf.
    destruct (proj1 (InA_alt equ j enum) (enum_complete j))
      as [a [Hja Ha]].
    calc (f j)
      == (f a) { (eq_le (f j) (f a) (Pf j a Hja)) }
      == (bigop_list f enum) { (bigop_list_le_one_term f enum a Ha
                                 (fun i _ ⇒ Hf i)) }
      == (bigopI f) { bigopI_def }.
  Qed.

  Lemma abs_bigopI_le : ∀ f : I → F,
    abs (bigopI f) ≤ bigopI (fun i ⇒ abs (f i)).
  Proof.
    intros f.
    rewrite (bigopI_def f), (bigopI_def (fun i ⇒ abs (f i))).
    exact (abs_bigop_list_le f enum).
  Qed.

End BigOpIndexedOrder.

Section BigOpDelta.
  Context {R : Type} `{RR : Ring R}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {FI : FinIndex n I}.
  Context {Ml : HasBigopList R} {MlL : BigopList R}.
  Context {Mi : HasBigopI I R} {MiL : BigopI n I R}.

  Definition delta (i j : I) : R := if dec (i == j) then 1 else 0.

  Lemma delta_at_eq : ∀ i j : I, i == j → delta i j == 1.
  Proof.
    intros i j Hij.
    unfold delta; destruct (dec (i == j)) as [_ | Hne].
    - reflexivity.
    - destruct (Hne Hij).
  Qed.

  Lemma delta_at_neq : ∀ i j : I, ¬ (i == j) → delta i j == 0.
  Proof.
    intros i j Hij.
    unfold delta; destruct (dec (i == j)) as [Heq | _].
    - destruct (Hij Heq).
    - reflexivity.
  Qed.

  #[export] Instance delta_proper : Proper (equ ==> equ ==> equ) delta.
  Proof.
    intros i i' Hi j j' Hj.
    unfold delta.
    destruct (dec (i == j)) as [Hij | Hij];
      destruct (dec (i' == j')) as [Hij' | Hij'].
    - reflexivity.
    - destruct (Hij' (transitivity (symmetry Hi) (transitivity Hij Hj))).
    - destruct (Hij (transitivity Hi (transitivity Hij' (symmetry Hj)))).
    - reflexivity.
  Qed.

  Lemma mul_delta_eq : ∀ (a : R) (i j : I), i == j → a × delta i j == a.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_eq i j Hij).
    exact (right_identity a).
  Qed.

  Lemma mul_delta_neq : ∀ (a : R) (i j : I), ¬ (i == j) → a × delta i j == 0.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_neq i j Hij).
    exact (mul_0_r a).
  Qed.

  Lemma mul_delta_proper : ∀ (c : I → R) (j : I),
    Proper (equ ==> equ) c →
    Proper (equ ==> equ) (fun i ⇒ c i × delta i j).
  Proof.
    intros c j Pc i i' Hii'.
    calc (c i × delta i j)
      == (c i × delta i' j)
           { (mul_l_proper (c i) (delta i j) (delta i' j)
               (delta_proper i i' Hii' j j (reflexivity j))) }
      == (c i' × delta i' j)
           { (mul_r_proper (delta i' j) (c i) (c i') (Pc i i' Hii')) }.
  Qed.

  Lemma bigopI_delta : ∀ (c : I → R) (j : I),
    Proper (equ ==> equ) c →
    bigopI (fun i ⇒ c i × delta i j) == c j.
  Proof.
    intros c j Pc.
    calc (bigopI (fun i ⇒ c i × delta i j))
      == (c j × delta j j)
           { (bigopI_support_singleton j (fun i ⇒ c i × delta i j)
               (mul_delta_proper c j Pc)
               (fun i Hne ⇒ mul_delta_neq (c i) i j Hne)) }
      == (c j)
           { mul_delta_eq }.
  Qed.

End BigOpDelta.

Section BigOpAbGroup.
  Context {G : Type} `{AbGroup G} {Mb : HasBigop G} {MbL : Bigop G}.

  Theorem bigop_add : ∀ n (f g : nat → G),
    bigop n (fun i ⇒ f i + g i) == bigop n f + bigop n g.
  Proof.
    intros n; induction n as [| n IH]; intros f g.
    - calc (bigop O (fun i ⇒ f i + g i))
        == (0 : G) { bigop_0 }
        == ((0 : G) + (0 : G)) { left_identity }
        == (bigop O f + (0 : G)) { bigop_0 }
        == (bigop O f + bigop O g) { (bigop_0 g) }.
    - calc (bigop (S n) (fun i ⇒ f i + g i))
        == ((f O + g O) + bigop n (fun i ⇒ f (S i) + g (S i)))
          { bigop_S }
        == ((f O + g O) + (bigop n (fun i ⇒ f (S i)) + bigop n (fun i ⇒ g (S i))))
          { IH }
        == ((f O + bigop n (fun i ⇒ f (S i))) + (g O + bigop n (fun i ⇒ g (S i))))
          { add_interchange }
        == (bigop (S n) f + (g O + bigop n (fun i ⇒ g (S i)))) { bigop_S }
        == (bigop (S n) f + bigop (S n) g) { bigop_S }.
  Qed.

  Theorem bigop_neg : ∀ n (f : nat → G),
    bigop n (fun i ⇒ - f i) == - bigop n f.
  Proof.
    intros n; induction n as [| n IH]; intros f.
    - calc (bigop O (fun i ⇒ - f i))
        == (0 : G) { bigop_0 }
        == (- (0 : G)) { neg_zero }
        == (- bigop O f) { (bigop_0 f) }.
    - calc (bigop (S n) (fun i ⇒ - f i))
        == (- f O + bigop n (fun i ⇒ - f (S i))) { bigop_S }
        == (- f O + - bigop n (fun i ⇒ f (S i))) { IH }
        == (- (f O + bigop n (fun i ⇒ f (S i)))) { neg_add }
        == (- bigop (S n) f) { bigop_S }.
  Qed.

  Theorem bigop_swap : ∀ (m n : nat) (f : nat → nat → G),
    bigop m (fun i ⇒ bigop n (fun j ⇒ f i j))
      == bigop n (fun j ⇒ bigop m (fun i ⇒ f i j)).
  Proof.
    intros m; induction m as [| m IH]; intros n f.
    - calc (bigop O (fun i ⇒ bigop n (fun j ⇒ f i j)))
        == (0 : G)
          { bigop_0 }
        == (bigop n (fun _ : nat ⇒ (0 : G)))
          { bigop_zero }
        == (bigop n (fun j ⇒ bigop O (fun i ⇒ f i j)))
          { (bigop_ext n (fun _ ⇒ 0)
              (fun j ⇒ bigop O (fun i ⇒ f i j))
              (fun j ⇒ symmetry (bigop_0 (fun i ⇒ f i j)))) }.
    - calc (bigop (S m) (fun i ⇒ bigop n (fun j ⇒ f i j)))
        == (bigop n (f O) + bigop m (fun i ⇒ bigop n (fun j ⇒ f (S i) j)))
          { bigop_S }
        == (bigop n (f O) + bigop n (fun j ⇒ bigop m (fun i ⇒ f (S i) j)))
          { IH }
        == (bigop n (fun j ⇒ f O j + bigop m (fun i ⇒ f (S i) j)))
          { bigop_add }
        == (bigop n (fun j ⇒ bigop (S m) (fun i ⇒ f i j)))
          { (bigop_ext n (fun j ⇒ f O j + bigop m (fun i ⇒ f (S i) j))
              (fun j ⇒ bigop (S m) (fun i ⇒ f i j))
              (fun j ⇒ symmetry (bigop_S m (fun i ⇒ f i j)))) }.
  Qed.

End BigOpAbGroup.

Section BigOpRing.
  Context {R : Type} `{Ring R} {Mb : HasBigop R} {MbL : Bigop R}.

  Theorem bigop_mul_l : ∀ n (a : R) (f : nat → R),
    bigop n (fun i ⇒ a × f i) == a × bigop n f.
  Proof.
    intros n; induction n as [| n IH]; intros a f.
    - calc (bigop O (fun i ⇒ a × f i))
        == (0 : R) { bigop_0 }
        == (a × (0 : R)) { mul_0_r }
        == (a × bigop O f) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ a × f i))
        == (a × f O + bigop n (fun i ⇒ a × f (S i))) { bigop_S }
        == (a × f O + a × bigop n (fun i ⇒ f (S i))) { IH }
        == (a × (f O + bigop n (fun i ⇒ f (S i))))
          { left_distributive }
        == (a × bigop (S n) f) { bigop_S }.
  Qed.

  Theorem bigop_mul_r : ∀ n (a : R) (f : nat → R),
    bigop n (fun i ⇒ f i × a) == bigop n f × a.
  Proof.
    intros n; induction n as [| n IH]; intros a f.
    - calc (bigop O (fun i ⇒ f i × a))
        == (0 : R) { bigop_0 }
        == ((0 : R) × a) { mul_0_l }
        == (bigop O f × a) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ f i × a))
        == (f O × a + bigop n (fun i ⇒ f (S i) × a)) { bigop_S }
        == (f O × a + bigop n (fun i ⇒ f (S i)) × a) { IH }
        == ((f O + bigop n (fun i ⇒ f (S i))) × a)
          { right_distributive }
        == (bigop (S n) f × a) { bigop_S }.
  Qed.

End BigOpRing.

Section BigOpVectorSpace.
  Context {F V : Type} `{VectorSpace F V}.
  Context {MbF : HasBigop F} {MbFL : Bigop F}.
  Context {MbV : HasBigop V} {MbVL : Bigop V}.

  Theorem bigop_scale : ∀ n (a : F) (f : nat → V),
    bigop n (fun i ⇒ a *: f i) == a *: bigop n f.
  Proof.
    intros n; induction n as [| n IH]; intros a f.
    - calc (bigop O (fun i ⇒ a *: f i))
        == (0 : V) { bigop_0 }
        == (a *: (0 : V)) { scale_zero }
        == (a *: bigop O f) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ a *: f i))
        == (a *: f O + bigop n (fun i ⇒ a *: f (S i)))
          { bigop_S }
        == (a *: f O + a *: bigop n (fun i ⇒ f (S i))) { IH }
        == (a *: (f O + bigop n (fun i ⇒ f (S i))))
          { left_distributive }
        == (a *: bigop (S n) f) { bigop_S }.
  Qed.

  Theorem bigop_scale_r : ∀ (n : nat) (f : nat → F) (v : V),
    bigop n (fun i ⇒ f i *: v) == bigop n f *: v.
  Proof.
    intros n; induction n as [| n IH]; intros f v.
    - calc (bigop O (fun i ⇒ f i *: v))
        == (0 : V) { bigop_0 }
        == ((0 : F) *: v) { zero_scale }
        == (bigop O f *: v) { bigop_0 }.
    - calc (bigop (S n) (fun i ⇒ f i *: v))
        == (f O *: v + bigop n (fun i ⇒ f (S i) *: v))
          { bigop_S }
        == (f O *: v + bigop n (fun i ⇒ f (S i)) *: v) { IH }
        == ((f O + bigop n (fun i ⇒ f (S i))) *: v)
          { right_distributive }
        == (bigop (S n) f *: v) { bigop_S }.
  Qed.

End BigOpVectorSpace.


Section BigOpVectorSpaceList.
  Context {F V : Type} `{VectorSpace F V}.

  Fixpoint list_lincomb (coeffs : list F) (vecs : list V) : V :=
    match coeffs with
    | nil ⇒ 0
    | c :: coeffs' ⇒
      match vecs with
      | nil ⇒ 0
      | v :: vecs' ⇒ c *: v + list_lincomb coeffs' vecs'
      end
    end.

  Lemma list_lincomb_nil_l : ∀ vecs : list V, list_lincomb nil vecs == 0.
  Proof.
    intros vecs.
    reflexivity.
  Qed.

  Lemma list_lincomb_cons : ∀ (c : F) (coeffs' : list F) (v : V)
      (vecs' : list V),
    list_lincomb (c :: coeffs') (v :: vecs') == c *: v + list_lincomb coeffs' vecs'.
  Proof.
    intros c coeffs' v vecs'.
    reflexivity.
  Qed.

  Lemma list_lincomb_map_ext : ∀ (l : list V) (g g' : V → F),
    (∀ d : V, In d l → g d == g' d) →
    list_lincomb (map g l) l == list_lincomb (map g' l) l.
  Proof.
    intros l; induction l as [| d l' IH]; intros g g' Hext.
    - reflexivity.
    - rewrite (list_lincomb_cons (g d) (map g l') d l'),
        (list_lincomb_cons (g' d) (map g' l') d l'),
        (Hext d (or_introl eq_refl)),
        (IH g g' (fun d' Hd' ⇒ Hext d' (or_intror Hd'))).
      reflexivity.
  Qed.

End BigOpVectorSpaceList.
