bigop_list f l is the sum of f over an explicit index list, and bigop_list_nil and bigop_list_cons are the two equations it satisfies.

bigop n f is the bounded sum f 0 + (f 1 + (... + 0)).




Section BigOpList.
  Context {M : Type} `{MM : Monoid M}.
  Context {bigop_list : ∀ J : Type, (J → M) → list J → M}
          {BL : BigOp M bigop_list}.
  Arguments bigop_list {J}.

  Lemma bigop_list_ext : ∀ {J : Type} (f g : J → M) (l : list J),
    (∀ j : J, In j l → f j == g j) →
    bigop_list f l == bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH]; intros Hfg.
    - calc (bigop_list f nil)
        == (0 : M) { bigop_list_nil }
        == (bigop_list g nil) { bigop_list_nil }.
    - rewrite (bigop_list_cons f a l), (bigop_list_cons g a l).
      apply op_proper.
      + exact (Hfg a (or_introl eq_refl)).
      + exact (IH (fun j Hj ⇒ Hfg j (or_intror Hj))).
  Qed.

  Lemma bigop_list_zero_on : ∀ {J : Type} (f : J → M) (l : list J),
    (∀ j : J, In j l → f j == 0) →
    bigop_list f l == 0.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf.
    - exact (bigop_list_nil f).
    - rewrite (bigop_list_cons f a l),
        (Hf a (or_introl eq_refl)),
        (IH (fun j Hj ⇒ Hf j (or_intror Hj))).
      exact (left_identity 0).
  Qed.

  Lemma bigop_list_map : ∀ {J K : Type} (h : J → K) (f : K → M)
      (l : list J),
    bigop_list f (map h l) == bigop_list (fun j ⇒ f (h j)) l.
  Proof.
    intros J K h f l; induction l as [| a l IH].
    - calc (bigop_list f (map h nil))
        == (0 : M)
          { bigop_list_nil }
        == (bigop_list (fun j ⇒ f (h j)) nil)
          { bigop_list_nil }.
    - rewrite (map_cons h a l),
        (bigop_list_cons f (h a) (map h l)),
        (bigop_list_cons (fun j ⇒ f (h j)) a l).
      apply op_proper.
      + reflexivity.
      + exact IH.
  Qed.

  Lemma bigop_list_append : ∀ {J : Type} (f : J → M) (l1 l2 : list J),
    bigop_list f (l1 ++ l2) == bigop_list f l1 + bigop_list f l2.
  Proof.
    intros J f l1 l2; induction l1 as [| a l1 IH].
    - calc (bigop_list f (nil ++ l2))
        == (0 + bigop_list f l2) { left_identity }
        == (bigop_list f nil + bigop_list f l2) { (bigop_list_nil f) }.
    - calc (bigop_list f ((a :: l1) ++ l2))
        == (f a + bigop_list f (l1 ++ l2)) { (bigop_list_cons f a (l1 ++ l2)) }
        == (f a + (bigop_list f l1 + bigop_list f l2)) { IH }
        == ((f a + bigop_list f l1) + bigop_list f l2) { associative }
        == (bigop_list f (a :: l1) + bigop_list f l2) { (bigop_list_cons f a l1) }.
  Qed.

  Lemma bigop_list_single : ∀ {J : Type} (f : J → M) (j : J),
    bigop_list f (j :: nil) == f j.
  Proof.
    intros J f j.
    calc (bigop_list f (j :: nil))
      == (f j + bigop_list f nil) { (bigop_list_cons f j nil) }
      == (f j + (0 : M)) { (bigop_list_nil f) }
      == (f j) { right_identity }.
  Qed.

  Lemma bigop_list_support : ∀ {J : Type} (f : J → M)
      (l1 l2 : list J) (j : J),
    (∀ k : J, In k l1 → f k == 0) →
    (∀ k : J, In k l2 → f k == 0) →
    bigop_list f (l1 ++ j :: l2) == f j.
  Proof.
    intros J f l1 l2 j H1 H2.
    calc (bigop_list f (l1 ++ j :: l2))
      == (bigop_list f l1 + bigop_list f (j :: l2))
           { (bigop_list_append f l1 (j :: l2)) }
      == ((0 : M) + bigop_list f (j :: l2))
           { (bigop_list_zero_on f l1 H1) }
      == (bigop_list f (j :: l2))
           { left_identity }
      == (f j + bigop_list f l2)
           { (bigop_list_cons f j l2) }
      == (f j + (0 : M))
           { (bigop_list_zero_on f l2 H2) }
      == (f j)
           { right_identity }.
  Qed.

  Lemma bigop_list_flat_map : ∀ {J K : Type} (h : J → list K) (f : K → M)
      (l : list J),
    bigop_list f (flat_map h l) == bigop_list (fun j ⇒ bigop_list f (h j)) l.
  Proof.
    intros J K h f l; induction l as [| a l IH].
    - calc (bigop_list f (flat_map h nil))
        == (0 : M)
          { bigop_list_nil }
        == (bigop_list (fun j ⇒ bigop_list f (h j)) nil)
          { bigop_list_nil }.
    - calc (bigop_list f (flat_map h (a :: l)))
        == (bigop_list f (h a) + bigop_list f (flat_map h l))
             { (bigop_list_append f (h a) (flat_map h l)) }
        == (bigop_list f (h a) + bigop_list (fun j ⇒ bigop_list f (h j)) l)
             { IH }
        == (bigop_list (fun j ⇒ bigop_list f (h j)) (a :: l))
             { (bigop_list_cons (fun j ⇒ bigop_list f (h j)) a l) }.
  Qed.

End BigOpList.

  Lemma bigop_list_app : ∀ {J : Type} (f : J → X → M) (l : list J)
      (x : X),
    bigop_list_fun f l x == bigop_list (fun j ⇒ f j x) l.
  Proof.
    intros J f l x; induction l as [| a l IH].
    - calc (bigop_list_fun f nil x)
        == (0 : M)
          { (bigop_list_nil f x) }
        == (bigop_list (fun j ⇒ f j x) nil)
          { (symmetry (bigop_list_nil (fun j ⇒ f j x))) }.
    - calc (bigop_list_fun f (a :: l) x)
        == (f a x + bigop_list_fun f l x)
          { (bigop_list_cons f a l x) }
        == (f a x + bigop_list (fun j ⇒ f j x) l)
          { IH }
        == (bigop_list (fun j ⇒ f j x) (a :: l))
          { bigop_list_cons }.
  Qed.

Section BigOpListCommutative.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.
  Context {bigop_list : ∀ J : Type, (J → M) → list J → M}
          {BL : BigOp M bigop_list}.
  Arguments bigop_list {J}.

  Lemma bigop_list_permutation : ∀ {J : Type} {eJ : EqRel J}
      {JE : Equivalence (@equ J eJ)} (f : J → M),
    Proper (equ ==> equ) f →
    ∀ l l' : list J,
    PermutationA equ l l' → bigop_list f l == bigop_list f l'.
  Proof.
    intros J eJ JE f Pf l l' Hp;
      induction Hp as [| x y l1 l2 Hxy _ IH | x y l1 | l1 l2 l3 _ IH1 _ IH2].
    - reflexivity.
    - rewrite (bigop_list_cons f x l1), (bigop_list_cons f y l2).
      apply op_proper.
      + exact (Pf x y Hxy).
      + exact IH.
    - rewrite (bigop_list_cons f y (x :: l1)), (bigop_list_cons f x l1),
        (bigop_list_cons f x (y :: l1)), (bigop_list_cons f y l1).
      exact (add_swap_front (f y) (f x) (bigop_list f l1)).
    - exact (transitivity IH1 IH2).
  Qed.

  Lemma bigop_list_extract : ∀ {J : Type} {eJ : EqRel J}
      {JE : Equivalence (@equ J eJ)} {Jeq : EquDec J}
      (p : J) (f : J → M),
    Proper (equ ==> equ) f →
    ∀ l : list J,
    NoDupA equ l → InA equ p l →
    bigop_list f l == f p + bigop_list f (removeA equ_dec p l).
  Proof.
    intros J eJ JE Jeq p f Pf l; induction l as [| a l IH]; intros Hnd Hin.
    - inversion Hin.
    - inversion Hnd as [| a0 l0 Hnotin Hnd' Heql]; subst.
      simpl removeA.
      destruct (equ_dec p a) as [Heq | Hne].
      + assert (Hpl : ¬ InA equ p l).
        { intros Hpin. exact (Hnotin (InA_eqA JE Heq Hpin)). }
        rewrite (removeA_notin equ_dec p l Hpl).
        calc (bigop_list f (a :: l))
          == (f a + bigop_list f l) { bigop_list_cons }
          == (f p + bigop_list f l) { (Pf a p (symmetry Heq)) }.
      + inversion Hin as [x0 l0 Hpa | x0 l0 Hin']; subst.
        × destruct (Hne Hpa).
        × calc (bigop_list f (a :: l))
            == (f a + bigop_list f l)
                 { bigop_list_cons }
            == (f a + (f p + bigop_list f (removeA equ_dec p l)))
                 { (IH Hnd' Hin') }
            == (f p + (f a + bigop_list f (removeA equ_dec p l)))
                 { add_swap_front }
            == (f p + bigop_list f (a :: removeA equ_dec p l))
                 { bigop_list_cons }.
  Qed.

End BigOpListCommutative.

Section BigOpListAbGroup.
  Context {G : Type} `{AbGroup G}.
  Context {bigop_list : ∀ J : Type, (J → G) → list J → G}
          {BL : BigOp G bigop_list}.
  Arguments bigop_list {J}.

  Lemma bigop_list_add : ∀ {J : Type} (f g : J → G) (l : list J),
    bigop_list (fun j ⇒ f j + g j) l == bigop_list f l + bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH].
    - calc (bigop_list (fun j ⇒ f j + g j) nil)
        == (0 : G) { bigop_list_nil }
        == ((0 : G) + (0 : G)) { left_identity }
        == (bigop_list f nil + (0 : G)) { bigop_list_nil }
        == (bigop_list f nil + bigop_list g nil) { (bigop_list_nil g) }.
    - calc (bigop_list (fun j ⇒ f j + g j) (a :: l))
        == ((f a + g a) + bigop_list (fun j ⇒ f j + g j) l)
          { bigop_list_cons }
        == ((f a + g a) + (bigop_list f l + bigop_list g l)) { IH }
        == ((f a + bigop_list f l) + (g a + bigop_list g l))
          { add_interchange }
        == (bigop_list f (a :: l) + (g a + bigop_list g l))
          { bigop_list_cons }
        == (bigop_list f (a :: l) + bigop_list g (a :: l))
          { bigop_list_cons }.
  Qed.

  Lemma bigop_list_neg : ∀ {J : Type} (f : J → G) (l : list J),
    bigop_list (fun j ⇒ - f j) l == - bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH].
    - calc (bigop_list (fun j ⇒ - f j) nil)
        == (0 : G) { bigop_list_nil }
        == (- (0 : G)) { neg_zero }
        == (- bigop_list f nil) { bigop_list_nil }.
    - calc (bigop_list (fun j ⇒ - f j) (a :: l))
        == (- f a + bigop_list (fun j ⇒ - f j) l)
          { bigop_list_cons }
        == (- f a + - bigop_list f l) { IH }
        == (- (f a + bigop_list f l)) { additive[neg] }
        == (- bigop_list f (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_swap : ∀ {J K : Type} (f : J → K → G)
      (l1 : list J) (l2 : list K),
    bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) l1
      == bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) l1) l2.
  Proof.
    intros J K f l1; induction l1 as [| a l1 IH]; intros l2.
    - calc (bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) nil)
        == (0 : G)
          { bigop_list_nil }
        == (bigop_list (fun k ⇒ bigop_list (fun j : J ⇒ f j k) nil) l2)
          { (bigop_list_zero_on
              (fun k ⇒ bigop_list (fun j : J ⇒ f j k) nil) l2
              (fun k _ ⇒ bigop_list_nil (fun j : J ⇒ f j k))) }.
    - calc (bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) (a :: l1))
        == (bigop_list (fun k ⇒ f a k) l2
            + bigop_list (fun j ⇒ bigop_list (fun k ⇒ f j k) l2) l1)
          { bigop_list_cons }
        == (bigop_list (fun k ⇒ f a k) l2
            + bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) l1) l2)
          { IH }
        == (bigop_list (fun k ⇒ f a k + bigop_list (fun j ⇒ f j k) l1) l2)
          { bigop_list_add }
        == (bigop_list (fun k ⇒ bigop_list (fun j ⇒ f j k) (a :: l1)) l2)
          { (bigop_list_ext
              (fun k ⇒ f a k + bigop_list (fun j ⇒ f j k) l1)
              (fun k ⇒ bigop_list (fun j ⇒ f j k) (a :: l1)) l2
              (fun k _ ⇒ symmetry (bigop_list_cons (fun j ⇒ f j k) a l1))) }.
  Qed.

End BigOpListAbGroup.

Section BigOpListRing.
  Context {R : Type} `{Ring R}.
  Context {bigop_list : ∀ J : Type, (J → R) → list J → R}
          {BL : BigOp R bigop_list}.
  Arguments bigop_list {J}.

  Lemma bigop_list_mul_l : ∀ {J : Type} (a : R) (f : J → R)
      (l : list J),
    bigop_list (fun j ⇒ a × f j) l == a × bigop_list f l.
  Proof.
    intros J a f l; induction l as [| b l IH].
    - calc (bigop_list (fun j ⇒ a × f j) nil)
        == (0 : R) { bigop_list_nil }
        == (a × (0 : R)) { mul_0_r }
        == (a × bigop_list f nil) { bigop_list_nil }.
    - calc (bigop_list (fun j ⇒ a × f j) (b :: l))
        == (a × f b + bigop_list (fun j ⇒ a × f j) l)
          { bigop_list_cons }
        == (a × f b + a × bigop_list f l) { IH }
        == (a × (f b + bigop_list f l))
          { left_distributive }
        == (a × bigop_list f (b :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_mul_r : ∀ {J : Type} (a : R) (f : J → R)
      (l : list J),
    bigop_list (fun j ⇒ f j × a) l == bigop_list f l × a.
  Proof.
    intros J a f l; induction l as [| b l IH].
    - calc (bigop_list (fun j ⇒ f j × a) nil)
        == (0 : R) { bigop_list_nil }
        == ((0 : R) × a) { mul_0_l }
        == (bigop_list f nil × a) { bigop_list_nil }.
    - calc (bigop_list (fun j ⇒ f j × a) (b :: l))
        == (f b × a + bigop_list (fun j ⇒ f j × a) l)
          { bigop_list_cons }
        == (f b × a + bigop_list f l × a) { IH }
        == ((f b + bigop_list f l) × a)
          { right_distributive }
        == (bigop_list f (b :: l) × a) { bigop_list_cons }.
  Qed.

End BigOpListRing.

Section BigOpListVectorSpace.
  Context {F V : Type} `{VectorSpace F V}.
  Context {bigop_list : ∀ J : Type, (J → V) → list J → V}
          {BL : BigOp V bigop_list}.
  Arguments bigop_list {J}.
  Context {bigop_list_scalar : ∀ J : Type, (J → F) → list J → F}
          {BLS : BigOp F bigop_list_scalar}.
  Arguments bigop_list_scalar {J}.

  Lemma bigop_list_scale : ∀ {J : Type} (a : F) (f : J → V)
      (l : list J),
    bigop_list (fun j ⇒ a *: f j) l == a *: bigop_list f l.
  Proof.
    intros J a f l; induction l as [| b l IH].
    - calc (bigop_list (fun j ⇒ a *: f j) nil)
        == (0 : V) { bigop_list_nil }
        == (a *: (0 : V)) { scale_zero }
        == (a *: bigop_list f nil) { bigop_list_nil }.
    - calc (bigop_list (fun j ⇒ a *: f j) (b :: l))
        == (a *: f b + bigop_list (fun j ⇒ a *: f j) l)
          { bigop_list_cons }
        == (a *: f b + a *: bigop_list f l) { IH }
        == (a *: (f b + bigop_list f l))
          { left_distributive }
        == (a *: bigop_list f (b :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_scale_r : ∀ {J : Type} (f : J → F) (v : V)
      (l : list J),
    bigop_list (fun j ⇒ f j *: v) l == bigop_list_scalar f l *: v.
  Proof.
    intros J f v l; induction l as [| b l IH].
    - calc (bigop_list (fun j ⇒ f j *: v) nil)
        == (0 : V) { bigop_list_nil }
        == ((0 : F) *: v) { zero_scale }
        == (bigop_list_scalar f nil *: v) { bigop_list_nil }.
    - calc (bigop_list (fun j ⇒ f j *: v) (b :: l))
        == (f b *: v + bigop_list (fun j ⇒ f j *: v) l)
          { bigop_list_cons }
        == (f b *: v + bigop_list_scalar f l *: v) { IH }
        == ((f b + bigop_list_scalar f l) *: v)
          { right_distributive }
        == (bigop_list_scalar f (b :: l) *: v) { bigop_list_cons }.
  Qed.

End BigOpListVectorSpace.

Section BigOpListOrder.
  Context {F : Type} `{OrderedField F}.
  Context {bigop_list : ∀ J : Type, (J → F) → list J → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J}.

  Lemma bigop_list_nonneg : ∀ {J : Type} (f : J → F) (l : list J),
    (∀ j : J, In j l → 0 ≤ f j) → 0 ≤ bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf.
    - apply eq_le. symmetry. exact (bigop_list_nil f).
    - calc (0 : F)
        == ((0 : F) + (0 : F)) { left_identity }
        == (f a + bigop_list f l)
          { (add_le_mono 0 (f a) 0 (bigop_list f l)
              (Hf a (or_introl eq_refl))
              (IH (fun j Hj ⇒ Hf j (or_intror Hj)))) }
        == (bigop_list f (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_le : ∀ {J : Type} (f g : J → F) (l : list J),
    (∀ j : J, In j l → f j ≤ g j) →
    bigop_list f l ≤ bigop_list g l.
  Proof.
    intros J f g l; induction l as [| a l IH]; intros Hfg.
    - apply eq_le.
      calc (bigop_list f nil)
        == (0 : F) { bigop_list_nil }
        == (bigop_list g nil) { bigop_list_nil }.
    - calc (bigop_list f (a :: l))
        == (f a + bigop_list f l) { bigop_list_cons }
        == (g a + bigop_list g l)
          { (add_le_mono (f a) (g a) (bigop_list f l) (bigop_list g l)
              (Hfg a (or_introl eq_refl))
              (IH (fun j Hj ⇒ Hfg j (or_intror Hj)))) }
        == (bigop_list g (a :: l)) { bigop_list_cons }.
  Qed.

  Lemma bigop_list_nonneg_zero : ∀ {J : Type} (f : J → F) (l : list J),
    (∀ j : J, In j l → 0 ≤ f j) → bigop_list f l == 0 →
    ∀ j : J, In j l → f j == 0.
  Proof.
    intros J f l; induction l as [| a l IH]; intros Hf Hsum j Hj.
    - destruct Hj.
    - rewrite (bigop_list_cons f a l) in Hsum.
      destruct (nonneg_add_eq_zero (f a) (bigop_list f l)
        (Hf a (or_introl eq_refl))
        (bigop_list_nonneg f l (fun j' Hj' ⇒ Hf j' (or_intror Hj')))
        Hsum) as [H1 H2].
      destruct Hj as [Heq | Hin].
      + rewrite <- Heq. exact H1.
      + exact (IH (fun j' Hj' ⇒ Hf j' (or_intror Hj')) H2 j Hin).
  Qed.

  Lemma bigop_list_le_one_term : ∀ {J : Type} (f : J → F) (l : list J)
      (j : J),
    In j l → (∀ j' : J, In j' l → 0 ≤ f j') →
    f j ≤ bigop_list f l.
  Proof.
    intros J f l; induction l as [| a l IH]; intros j Hj Hf.
    - destruct Hj.
    - rewrite (bigop_list_cons f a l).
      destruct Hj as [Heq | Hin].
      + rewrite <- Heq.
        apply le_add_nonneg_r.
        exact (bigop_list_nonneg f l (fun j' Hj' ⇒ Hf j' (or_intror Hj'))).
      + calc (f j)
          == (bigop_list f l)
            { (IH j Hin (fun j' Hj' ⇒ Hf j' (or_intror Hj'))) }
          == (f a + bigop_list f l)
            { (le_add_nonneg_l (f a) (bigop_list f l)
                (Hf a (or_introl eq_refl))) }.
  Qed.

  Lemma abs_bigop_list_le : ∀ {J : Type} (f : J → F) (l : list J),
    abs (bigop_list f l) ≤ bigop_list (fun j ⇒ abs (f j)) l.
  Proof.
    intros J f l; induction l as [| a l IH].
    - apply eq_le.
      calc (abs (bigop_list f nil))
        == (abs (0 : F)) { bigop_list_nil }
        == (0 : F) { abs_zero }
        == (bigop_list (fun j ⇒ abs (f j)) nil) { bigop_list_nil }.
    - calc (abs (bigop_list f (a :: l)))
        == (abs (f a + bigop_list f l)) { bigop_list_cons }
        == (abs (f a) + abs (bigop_list f l))
          { abs_triangle }
        == (abs (f a) + bigop_list (fun j ⇒ abs (f j)) l)
          { add_le_mono }
        == (bigop_list (fun j ⇒ abs (f j)) (a :: l))
          { bigop_list_cons }.
  Qed.

  Proof. exact is_delta_at_eq. Qed.

  Proof. exact is_delta_at_neq. Qed.

    Proper (equ ==> equ ==> equ) delta.
  Proof.

    destruct (dec (i == j)) as [Hij | Hij].
    - calc (delta i j : R)
        == (1 : R) { (delta_at_eq i j Hij) }
        == (delta i' j' : R)
             { (symmetry (delta_at_eq i' j'
                 (transitivity (symmetry Hi) (transitivity Hij Hj)))) }.
    - calc (delta i j : R)
        == (0 : R) { (delta_at_neq i j Hij) }
        == (delta i' j' : R)
             { (symmetry (delta_at_neq i' j'
                 (fun Hc ⇒ Hij (transitivity Hi
                              (transitivity Hc (symmetry Hj)))))) }.
  Qed.

  Lemma mul_delta_eq : ∀ (a : R) (i j : I), i == j → a × delta i j == a.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_eq i j Hij).
    exact (right_identity a).
  Qed.

  Lemma mul_delta_neq : ∀ (a : R) (i j : I), ¬ (i == j) → a × delta i j == 0.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_neq i j Hij).
    exact (mul_0_r a).
  Qed.

  Lemma mul_delta_proper : ∀ (c : I → R) (j : I),
    Proper (equ ==> equ) c →
    Proper (equ ==> equ) (fun i ⇒ c i × delta i j).
  Proof.
    intros c j Pc i i' Hii'.
    calc (c i × delta i j)
      == (c i × delta i' j)
           { (mul_l_proper (c i) (delta i j) (delta i' j)
               (delta_proper i i' Hii' j j (reflexivity j))) }
      == (c i' × delta i' j)
           { (mul_r_proper (delta i' j) (c i) (c i') (Pc i i' Hii')) }.
  Qed.

  Lemma bigopI_delta : ∀ (c : I → R) (j : I),
    Proper (equ ==> equ) c →
    bigopI (fun i ⇒ c i × delta i j) == c j.
  Proof.
    intros c j Pc.
    calc (bigopI (fun i ⇒ c i × delta i j))
      == (c j × delta j j)
           { (bigopI_support_singleton j (fun i ⇒ c i × delta i j)
               (mul_delta_proper c j Pc)
               (fun i Hne ⇒ mul_delta_neq (c i) i j Hne)) }
      == (c j)
           { mul_delta_eq }.
  Qed.


  Lemma delta_mul_eq : ∀ (a : R) (i j : I), i == j → delta i j × a == a.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_eq i j Hij).
    exact (left_identity a).
  Qed.

  Lemma delta_mul_neq : ∀ (a : R) (i j : I), ¬ (i == j) → delta i j × a == 0.
  Proof.
    intros a i j Hij.
    rewrite (delta_at_neq i j Hij).
    exact (mul_0_l a).
  Qed.

  Lemma delta_mul_proper : ∀ (c : I → R) (i : I),
    Proper (equ ==> equ) c →
    Proper (equ ==> equ) (fun j ⇒ delta i j × c j).
  Proof.
    intros c i Pc j j' Hjj'.
    calc (delta i j × c j)
      == (delta i j' × c j)
           { (mul_r_proper (c j) (delta i j) (delta i j')
               (delta_proper i i (reflexivity i) j j' Hjj')) }
      == (delta i j' × c j')
           { (mul_l_proper (delta i j') (c j) (c j') (Pc j j' Hjj')) }.
  Qed.

  Lemma bigopI_delta_l : ∀ (c : I → R) (i : I),
    Proper (equ ==> equ) c →
    bigopI (fun j ⇒ delta i j × c j) == c i.
  Proof.
    intros c i Pc.
    calc (bigopI (fun j ⇒ delta i j × c j))
      == (delta i i × c i)
           { (bigopI_support_singleton i (fun j ⇒ delta i j × c j)
               (delta_mul_proper c i Pc)
               (fun j Hne ⇒ delta_mul_neq (c j) i j
                  (fun Hc ⇒ Hne (symmetry Hc)))) }
      == (c i)
           { (delta_mul_eq (c i) i i (reflexivity i)) }.
  Qed.

End BigOpDelta.

  Lemma delta_scale_proper : ∀ (b : I → V) (j : I),
    Proper (equ ==> equ) b →
    Proper (equ ==> equ) (fun i ⇒ delta j i *: b i).
  Proof.
    intros b j Pb i i' Hii'.
    exact (scale_proper (delta j i) (delta j i')
             (delta_proper j j (reflexivity j) i i' Hii')
             (b i) (b i') (Pb i i' Hii')).
  Qed.

    bigopI (fun i ⇒ delta j i *: b i) == b j.
  Proof.

    calc (bigopI (fun i ⇒ delta j i *: b i))
      == (delta j j *: b j)
           { (bigopI_support_singleton j (fun i ⇒ delta j i *: b i)
               (delta_scale_proper b j Pb)
               (fun i Hne ⇒
                 transitivity
                   (scale_proper (delta j i) 0

                     (b i) (b i) (reflexivity (b i)))
                   (zero_scale (b i)))) }

Section BigOpMonoid.
  Context {M : Type} `{Monoid M}.
  Context {bigop_list : ∀ J : Type, (J → M) → list J → M}
          {BL : BigOp M bigop_list}.
  Arguments bigop_list {J}.
  Notation bigop n f := (bigop_list f (seq 0 n)).
  Let bigop_seq (n : nat) (f : nat → M) :
    bigop n f == bigop_list f (seq 0 n) :=
    reflexivity (bigop_list f (seq 0 n)).

  Lemma bigop_0 : ∀ f : nat → M, bigop O f == 0.
  Proof. intros f. exact (bigop_list_nil f). Qed.

  Lemma bigop_S : ∀ (n : nat) (f : nat → M),
    bigop (S n) f == f O + bigop n (fun i ⇒ f (S i)).
  Proof.
    intros n f.
    calc (bigop (S n) f)
      == (f O + bigop_list f (seq 1 n))
           { (bigop_list_cons f O (seq 1 n)) }
      == (f O + bigop_list f (map S (seq 0 n)))
           { (seq_shift n 0) }
      == (f O + bigop n (fun i ⇒ f (S i)))
           { (bigop_list_map S f (seq 0 n)) }.
  Qed.

  Lemma bigop_list_seq_shift : ∀ (m k : nat) (f : nat → M),
    bigop_list f (seq m k) == bigop_list (fun i ⇒ f (m + i)%nat) (seq 0 k).
  Proof.
    intros m k f.
    calc (bigop_list f (seq m k))
      == (bigop_list f (map (Nat.add m) (seq 0 k)))
           { (map_add_seq k m 0) }
      == (bigop_list (fun i ⇒ f (m + i)%nat) (seq 0 k))
           { (bigop_list_map (Nat.add m) f (seq 0 k)) }.
  Qed.

  Theorem bigop_ext : ∀ n (f g : nat → M),
    (∀ i, f i == g i) → bigop n f == bigop n g.
  Proof.
    intros n f g Hfg.
    calc (bigop n f)
      == (bigop_list g (seq 0 n))
           { (bigop_list_ext f g (seq 0 n) (fun i _ ⇒ Hfg i)) }.
  Qed.

  Theorem bigop_ext_lt : ∀ n (f g : nat → M),
    (∀ i, (i < n)%nat → f i == g i) → bigop n f == bigop n g.
  Proof.
    intros n f g Hfg.
    calc (bigop n f)
      == (bigop_list g (seq 0 n))
           { (bigop_list_ext f g (seq 0 n)
            (fun i Hi ⇒ Hfg i (proj2 (proj1 (in_seq n 0 i) Hi)))) }.
  Qed.

  Theorem bigop_snoc : ∀ (n : nat) (f : nat → M),
    bigop (S n) f == bigop n f + f n.
  Proof.
    intros n f.
    calc (bigop (S n) f)
      == (bigop_list f (seq 0 n ++ (n :: nil)))
           { (seq_S n 0) }
      == (bigop_list f (seq 0 n) + bigop_list f (n :: nil))
           { (bigop_list_append f (seq 0 n) (n :: nil)) }
      == (bigop_list f (seq 0 n) + f n)
           { (bigop_list_single f n) }.
  Qed.

  Theorem bigop_zero : ∀ n : nat,
    bigop n (fun _ : nat ⇒ (0 : M)) == 0.
  Proof.
    intros n.
    calc (bigop n (fun _ : nat ⇒ (0 : M)))
      == ((0 : M))
           { (bigop_list_zero_on (fun _ : nat ⇒ (0 : M)) (seq 0 n)
            (fun i _ ⇒ reflexivity (0 : M))) }.
  Qed.

  Theorem bigop_support_singleton : ∀ (n j : nat) (f : nat → M),
    (j < n)%nat → (∀ k : nat, k ≠ j → f k == 0) →
    bigop n f == f j.
  Proof.
    intros n j f Hj Hzero.
    calc (bigop n f)
      == (bigop_list f (seq 0 j ++ j :: seq (S j) (n - S j)))
           { (seq_split_at n j Hj) }
      == (f j)
           { (bigop_list_support f (seq 0 j) (seq (S j) (n - S j)) j
               (fun k Hk ⇒
                  Hzero k (Nat.lt_neq k j
                             (proj2 (proj1 (in_seq j 0 k) Hk))))
               (fun k Hk ⇒
                  Hzero k (not_eq_sym
                             (Nat.lt_neq j k
                                (proj1 (proj1 (in_seq (n - S j) (S j) k)
                                          Hk)))))) }.
  Qed.

  Theorem bigop_split : ∀ (m k : nat) (f : nat → M),
    bigop (m + k)%nat f == bigop m f + bigop k (fun i ⇒ f (m + i)%nat).
  Proof.
    intros m k f.
    calc (bigop (m + k)%nat f)
      == (bigop_list f (seq 0 m ++ seq m k))
           { (seq_app m k 0) }
      == (bigop_list f (seq 0 m) + bigop_list f (seq m k))
           { (bigop_list_append f (seq 0 m) (seq m k)) }
      == (bigop_list f (seq 0 m) + bigop_list (fun i ⇒ f (m + i)%nat) (seq 0 k))
           { (bigop_list_seq_shift m k f) }.
  Qed.

End BigOpMonoid.

Section BigOpAbGroup.
  Context {G : Type} `{AbGroup G}.
  Context {bigop_list : ∀ J : Type, (J → G) → list J → G}
          {BL : BigOp G bigop_list}.
  Arguments bigop_list {J}.
  Notation bigop n f := (bigop_list f (seq 0 n)).
  Let bigop_seq (n : nat) (f : nat → G) :
    bigop n f == bigop_list f (seq 0 n) :=
    reflexivity (bigop_list f (seq 0 n)).

  Theorem bigop_add : ∀ n (f g : nat → G),
    bigop n (fun i ⇒ f i + g i) == bigop n f + bigop n g.
  Proof.
    intros n f g.
    calc (bigop n (fun i ⇒ f i + g i))
      == (bigop_list f (seq 0 n) + bigop_list g (seq 0 n))
           { (bigop_list_add f g (seq 0 n)) }.
  Qed.

  Theorem bigop_neg : ∀ n (f : nat → G),
    bigop n (fun i ⇒ - f i) == - bigop n f.
  Proof.
    intros n f.
    calc (bigop n (fun i ⇒ - f i))
      == (- bigop_list f (seq 0 n))
           { (bigop_list_neg f (seq 0 n)) }.
  Qed.

  Theorem bigop_swap : ∀ (m n : nat) (f : nat → nat → G),
    bigop m (fun i ⇒ bigop n (fun j ⇒ f i j))
      == bigop n (fun j ⇒ bigop m (fun i ⇒ f i j)).
  Proof.
    intros m n f.
    calc (bigop m (fun i ⇒ bigop n (fun j ⇒ f i j)))
      == (bigop_list (fun i ⇒ bigop_list (fun j ⇒ f i j) (seq 0 n)) (seq 0 m))
           { (bigop_list_ext (fun i ⇒ bigop n (fun j ⇒ f i j))
               (fun i ⇒ bigop_list (fun j ⇒ f i j) (seq 0 n)) (seq 0 m)
               (fun i _ ⇒ bigop_seq n (fun j ⇒ f i j))) }
      == (bigop_list (fun j ⇒ bigop_list (fun i ⇒ f i j) (seq 0 m)) (seq 0 n))
           { (bigop_list_swap f (seq 0 m) (seq 0 n)) }
      == (bigop_list (fun j ⇒ bigop m (fun i ⇒ f i j)) (seq 0 n))
           { (bigop_list_ext (fun j ⇒ bigop_list (fun i ⇒ f i j) (seq 0 m))
               (fun j ⇒ bigop m (fun i ⇒ f i j)) (seq 0 n)
               (fun j _ ⇒ symmetry (bigop_seq m (fun i ⇒ f i j)))) }.
  Qed.

  Theorem bigop_parity : ∀ (n : nat) (f : nat → G),
    bigop (2 × n)%nat f
      == bigop n (fun k ⇒ f (2 × k)%nat)
           + bigop n (fun k ⇒ f (S (2 × k)%nat)).
  Proof.
    intros n; induction n as [| n IH]; intros f.
    - calc (bigop (2 × O)%nat f)
        == (0 : G) { bigop_0 }
        == ((0 : G) + (0 : G)) { left_identity }
        == (bigop O (fun k ⇒ f (2 × k)%nat)
              + bigop O (fun k ⇒ f (S (2 × k)%nat)))
             { (op_proper _ _ (symmetry (bigop_0 (fun k ⇒ f (2 × k)%nat)))
                          _ _ (symmetry (bigop_0 (fun k ⇒ f (S (2 × k)%nat))))) }.
    - rewrite (two_mul_succ n).
      calc (bigop (S (S (2 × n))) f)
        == (bigop (S (2 × n)) f + f (S (2 × n)%nat))
             { bigop_snoc }
        == ((bigop (2 × n)%nat f + f (2 × n)%nat) + f (S (2 × n)%nat))
             { bigop_snoc }
        == (((bigop n (fun k ⇒ f (2 × k)%nat)
                + bigop n (fun k ⇒ f (S (2 × k)%nat))) + f (2 × n)%nat)
              + f (S (2 × n)%nat))
             { (IH f) }
        == (((bigop n (fun k ⇒ f (2 × k)%nat) + f (2 × n)%nat)
               + bigop n (fun k ⇒ f (S (2 × k)%nat)))
              + f (S (2 × n)%nat))
             { (add_swap_r (bigop n (fun k ⇒ f (2 × k)%nat))) }
        == ((bigop n (fun k ⇒ f (2 × k)%nat) + f (2 × n)%nat)
              + (bigop n (fun k ⇒ f (S (2 × k)%nat)) + f (S (2 × n)%nat)))
             { associative }
        == (bigop (S n) (fun k ⇒ f (2 × k)%nat)
              + (bigop n (fun k ⇒ f (S (2 × k)%nat)) + f (S (2 × n)%nat)))
             { (bigop_snoc n (fun k ⇒ f (2 × k)%nat)) }
        == (bigop (S n) (fun k ⇒ f (2 × k)%nat)
              + bigop (S n) (fun k ⇒ f (S (2 × k)%nat)))
             { (bigop_snoc n (fun k ⇒ f (S (2 × k)%nat))) }.
  Qed.

  Lemma triangle_row_step : ∀ (N i : nat), (i ≤ N)%nat →
    ∀ u : nat → nat → G,
    bigop (S N - i)%nat (u i) == bigop (N - i)%nat (u i) + u i (N - i)%nat.
  Proof.
    intros N i Hi u.
    rewrite (Nat.sub_succ_l i N Hi).
    exact (bigop_snoc (N - i)%nat (u i)).
  Qed.

  Lemma triangle_last_row : ∀ (N : nat) (u : nat → nat → G),
    bigop (S N - N)%nat (u N) == u N (N - N)%nat.
  Proof.
    intros N u.
    rewrite (Nat.sub_succ_l N N (le_n N)), (Nat.sub_diag N).
    calc (bigop 1 (u N))
      == (u N O + bigop O (fun i ⇒ u N (S i))) { bigop_S }
      == (u N O + (0 : G)) { bigop_0 }
      == (u N O) { right_identity }.
  Qed.

  Theorem bigop_triangle : ∀ (N : nat) (u : nat → nat → G),
    bigop N (fun n ⇒ bigop (S n) (fun k ⇒ u k (n - k)%nat))
      == bigop N (fun i ⇒ bigop (N - i)%nat (u i)).
  Proof.
    intros N; induction N as [| N IH]; intros u.
    - calc (bigop O (fun n ⇒ bigop (S n) (fun k ⇒ u k (n - k)%nat)))
        == (0 : G) { bigop_0 }
        == (bigop O (fun i ⇒ bigop (O - i)%nat (u i))) { bigop_0 }.
    - assert (Hrows : bigop N (fun i ⇒ bigop (N - i)%nat (u i))
                        + bigop N (fun k ⇒ u k (N - k)%nat)
                      == bigop N (fun i ⇒ bigop (S N - i)%nat (u i))).
      { calc (bigop N (fun i ⇒ bigop (N - i)%nat (u i))
                + bigop N (fun k ⇒ u k (N - k)%nat))
          == (bigop N (fun i ⇒ bigop (N - i)%nat (u i) + u i (N - i)%nat))
               { (bigop_add N (fun i ⇒ bigop (N - i)%nat (u i))
                    (fun i ⇒ u i (N - i)%nat)) }
          == (bigop N (fun i ⇒ bigop (S N - i)%nat (u i)))
               { (bigop_ext_lt N (fun i ⇒ bigop (N - i)%nat (u i)
                                             + u i (N - i)%nat)
                    (fun i ⇒ bigop (S N - i)%nat (u i))
                    (fun i Hi ⇒ symmetry
                       (triangle_row_step N i (Nat.lt_le_incl i N Hi) u))) }. }
      calc (bigop (S N) (fun n ⇒ bigop (S n) (fun k ⇒ u k (n - k)%nat)))
        == (bigop N (fun n ⇒ bigop (S n) (fun k ⇒ u k (n - k)%nat))
              + bigop (S N) (fun k ⇒ u k (N - k)%nat))
             { bigop_snoc }
        == (bigop N (fun i ⇒ bigop (N - i)%nat (u i))
              + bigop (S N) (fun k ⇒ u k (N - k)%nat))
             { (IH u) }
        == (bigop N (fun i ⇒ bigop (N - i)%nat (u i))
              + (bigop N (fun k ⇒ u k (N - k)%nat) + u N (N - N)%nat))
             { (bigop_snoc N (fun k ⇒ u k (N - k)%nat)) }
        == (bigop N (fun i ⇒ bigop (N - i)%nat (u i))
              + (bigop N (fun k ⇒ u k (N - k)%nat)
                   + bigop (S N - N)%nat (u N)))
             { (triangle_last_row N u) }
        == ((bigop N (fun i ⇒ bigop (N - i)%nat (u i))
               + bigop N (fun k ⇒ u k (N - k)%nat))
              + bigop (S N - N)%nat (u N))
             { associative }
        == (bigop N (fun i ⇒ bigop (S N - i)%nat (u i))
              + bigop (S N - N)%nat (u N))
             { Hrows }
        == (bigop (S N) (fun i ⇒ bigop (S N - i)%nat (u i)))
             { (bigop_snoc N (fun i ⇒ bigop (S N - i)%nat (u i))) }.
  Qed.

End BigOpAbGroup.

  Context {bigop_list : ∀ J : Type, (J → F) → list J → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J}.
  Notation bigop n f := (bigop_list f (seq 0 n)).
  Let bigop_seq (n : nat) (f : nat → F) :
    bigop n f == bigop_list f (seq 0 n) :=
    reflexivity (bigop_list f (seq 0 n)).

  Lemma bigop_nonneg : ∀ (n : nat) (f : nat → F),
    (∀ i : nat, 0 ≤ f i) → 0 ≤ bigop n f.
  Proof.
    intros n f Hf.
    exact (bigop_list_nonneg f (seq 0 n) (fun i _ ⇒ Hf i)).
  Qed.

  Lemma abs_bigop_le : ∀ (n : nat) (f : nat → F),
    abs (bigop n f) ≤ bigop n (fun i ⇒ abs (f i)).
  Proof.
    intros n f.
    exact (abs_bigop_list_le f (seq 0 n)).
  Qed.

  Lemma bigop_le_lt : ∀ (n : nat) (f g : nat → F),
    (∀ i : nat, (i < n)%nat → f i ≤ g i) → bigop n f ≤ bigop n g.
  Proof.
    intros n f g Hfg.
    exact (bigop_list_le f g (seq 0 n)
             (fun i Hi ⇒ Hfg i (proj2 (proj1 (in_seq n 0 i) Hi)))).
  Qed.

  Lemma bigop_le : ∀ (n : nat) (f g : nat → F),
    (∀ i : nat, f i ≤ g i) → bigop n f ≤ bigop n g.
  Proof.
    intros n f g Hfg.
    exact (bigop_le_lt n f g (fun i _ ⇒ Hfg i)).
  Qed.

End BigOpOrder.

  Let bigop_scalar_seq (n : nat) (f : nat → F) :
    bigop_scalar n f == bigop_list_scalar f (seq 0 n) :=
    reflexivity (bigop_list_scalar f (seq 0 n)).

  Theorem bigop_scale : ∀ n (a : F) (f : nat → V),
    bigop n (fun i ⇒ a *: f i) == a *: bigop n f.
  Proof.
    intros n a f.
    calc (bigop n (fun i ⇒ a *: f i))
      == (a *: bigop_list f (seq 0 n))
           { (bigop_list_scale a f (seq 0 n)) }.
  Qed.

  Theorem bigop_scale_r : ∀ (n : nat) (f : nat → F) (v : V),
    bigop n (fun i ⇒ f i *: v) == bigop_scalar n f *: v.
  Proof.
    intros n f v.
    calc (bigop n (fun i ⇒ f i *: v))
      == (bigop_list_scalar f (seq 0 n) *: v)
           { (bigop_list_scale_r f v (seq 0 n)) }.
  Qed.

    list_lincomb nil vecs == (0 : V).
  Proof. exact is_list_lincomb_nil_l. Qed.

  Lemma list_lincomb_nil_r : ∀ (c : F) (coeffs' : list F),
    list_lincomb (c :: coeffs') nil == (0 : V).
  Proof. exact is_list_lincomb_nil_r. Qed.

  Lemma list_lincomb_cons : ∀ (c : F) (coeffs' : list F) (v : V)
      (vecs' : list V),
    list_lincomb (c :: coeffs') (v :: vecs') == c *: v + list_lincomb coeffs' vecs'.
  Proof. exact is_list_lincomb_cons. Qed.

  Lemma list_lincomb_map_ext : ∀ (l : list V) (g g' : V → F),
    (∀ d : V, In d l → g d == g' d) →
    list_lincomb (map g l) l == list_lincomb (map g' l) l.
  Proof.
    intros l; induction l as [| d l' IH]; intros g g' Hext.
    - reflexivity.
    - rewrite (list_lincomb_cons (g d) (map g l') d l'),
        (list_lincomb_cons (g' d) (map g' l') d l'),
        (Hext d (or_introl eq_refl)),
        (IH g g' (fun d' Hd' ⇒ Hext d' (or_intror Hd'))).
      reflexivity.
  Qed.

End BigOpVectorSpaceList.
