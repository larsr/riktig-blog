Library Riktig.Theory.RingHom

From Riktig Require Export Structure.RingHom Theory.Ring.

Section RingHomTheory.
  Context {A B : Type} `{Ring A} `{Ring B} (h : A → B) `{!RingHom A B h}.

  Theorem hom_zero : h 0 == 0.
  Proof. exact (additive_zero h). Qed.

  Theorem hom_neg : ∀ a : A, h (- a) == - h a.
  Proof. intros a. exact (additive_neg h a). Qed.

  Theorem hom_sub : ∀ a b : A, h (a + - b) == h a + - h b.
  Proof.
    intros a b.
    calc (h (a + - b))
      == (h a + h (- b)) { additive[h] }
      == (h a + - h b) { hom_neg }.
  Qed.

End RingHomTheory.
