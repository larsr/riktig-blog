The arrow out of a product that an arrow into an exponential stands for.
Notation "'uncurry' k" := (eval ∘ lift (k ∘ pi1) pi2) (at level 10, k at level 9).

Section ExponentialsTheory.
  Context {O : Type} `{Exponentials O}.

  Theorem curry_uncurry : ∀ (x a b : O) (k : x ~> a ⟹ b), curry (uncurry k) == k.
  Proof. intros x a b k. symmetry. apply curry_unique. reflexivity. Qed.

  Theorem uncurry_curry : ∀ (x a b : O) (h : x ⨯ a ~> b), uncurry (curry h) == h.
  Proof. intros x a b h. apply eval_curry. Qed.

  Theorem curry_comp : ∀ (x y a b : O) (h : y ⨯ a ~> b) (g : x ~> y),
    curry h ∘ g == curry (h ∘ (g ×× cat_id)).
  Proof.
    intros x y a b h g. apply curry_unique.
    calc (eval ∘ lift ((curry h ∘ g) ∘ pi1) pi2)
      == (eval ∘ lift (curry h ∘ (g ∘ pi1)) pi2) { comp_assoc }
      == (eval ∘ lift (curry h ∘ (g ∘ pi1)) (cat_id ∘ pi2 (a := x) (b := a))) { left_identity }
      == (eval ∘ (lift (curry h ∘ pi1) pi2 ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))) { _ }
      == ((eval ∘ lift (curry h ∘ pi1) pi2) ∘ (g ×× cat_id)) { comp_assoc }
      == (h ∘ (g ×× cat_id)) { eval_curry }.
    calc (eval ∘ lift (curry h ∘ (g ∘ pi1)) (cat_id ∘ pi2 (a := x) (b := a)))
      == (eval ∘ lift ((curry h ∘ pi1) ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))
                      (pi2 ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))) { _ }
      == (eval ∘ (lift (curry h ∘ pi1) pi2 ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))) { lift_comp }.
    f_equiv.
    calc (lift (curry h ∘ (g ∘ pi1)) (cat_id ∘ pi2 (a := x) (b := a)))
      == (lift (curry h ∘ (pi1 ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))) (cat_id ∘ pi2 (a := x) (b := a))) { pi1_lift }
      == (lift ((curry h ∘ pi1) ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a))) (cat_id ∘ pi2 (a := x) (b := a))) { comp_assoc }
      == (lift ((curry h ∘ pi1) ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))
              (pi2 ∘ lift (g ∘ pi1) (cat_id ∘ pi2 (a := x) (b := a)))) { pi2_lift }.
  Qed.
End ExponentialsTheory.
