The consequences of initiality: folding with build is the identity, a map of algebras carries one fold to another, and build is an isomorphism, its inverse the fold that rebuilds one level down -- so every element is built in exactly one way.
Section InitialAlgebraTheory.
  Context {O : Type} `{Category O} {F : O → O} {map : HasFmap O O F} {t : O}
    {build : HasBuild O F t} {fold : HasCata O F t} {IA : InitialAlgebra O F t build fold}.

  Local Notation "F· f" := (fmap (F := F) f) (at level 9, f at level 8).

  Theorem cata_id : fold t build == cat_id.
  Proof.
    symmetry. apply InitialAlgebra.cata_unique.
    calc (cat_id ∘ build)
      == build { left_identity }
      == (build ∘ cat_id) { right_identity }
      == (build ∘ F· cat_id) { Functor.fmap_id }.
  Qed.

An algebra endomorphism of the initial algebra is the identity.
  Theorem alg_endo_id : ∀ u : t ~> t,
    u ∘ build == build ∘ F· u → u == cat_id.
  Proof.
    intros u Hu.
    calc u
      == (fold t build) { (InitialAlgebra.cata_unique t build u Hu) }
      == (cat_id (x := t)) { cata_id }.
  Qed.

  Theorem cata_fusion : ∀ (x y : O) (f : F x ~> x) (g : F y ~> y) (h : x ~> y),
    h ∘ f == g ∘ F· h → h ∘ fold x f == fold y g.
  Proof.
    intros x y f g h Hh. apply InitialAlgebra.cata_unique.
    calc (h ∘ fold x f ∘ build)
      == (h ∘ (fold x f ∘ build)) { Category.comp_assoc }
      == (h ∘ (f ∘ F· (fold x f))) { InitialAlgebra.cata_build }
      == (h ∘ f ∘ F· (fold x f)) { Category.comp_assoc }
      == (g ∘ F· h ∘ F· (fold x f)) { Hh }
      == (g ∘ (F· h ∘ F· (fold x f))) { Category.comp_assoc }
      == (g ∘ F· (h ∘ fold x f)) { Functor.fmap_comp }.
  Qed.

  Theorem build_unbuild : build ∘ fold (F t) (F· build) == cat_id.
  Proof.
    calc (build ∘ fold (F t) (F· build))
      == (fold t build) { (cata_fusion (F t) t (F· build) build build (reflexivity _)) }
      == (cat_id (x := t)) { cata_id }.
  Qed.

  Theorem unbuild_build : fold (F t) (F· build) ∘ build == cat_id.
  Proof.
    calc (fold (F t) (F· build) ∘ build)
      == (F· build ∘ F· (fold (F t) (F· build))) { InitialAlgebra.cata_build }
      == (F· (build ∘ fold (F t) (F· build))) { Functor.fmap_comp }
      == (F· (cat_id (x := t))) { build_unbuild }
      == (cat_id (x := F t)) { Functor.fmap_id }.
  Qed.

Lambek's lemma: building one level is an isomorphism.
  #[export] Instance build_iso : Iso O (F t) t build (fold (F t) (F· build)) :=
    {| Iso.hom_inv_id := unbuild_build; Iso.inv_hom_id := build_unbuild |}.
End InitialAlgebraTheory.
