Section ArchimedeanTheory.
  Context {F N : Type} `{Archimedean F N}.

  Lemma of_nat_equ : ∀ n m : N, n == m → of_nat n == of_nat m.
  Proof. intros n m Hnm. apply of_nat_proper. exact Hnm. Qed.

  Theorem of_nat_nonneg : ∀ n : N, (0 : F) ≤ of_nat n.
  Proof.
    apply (nat_induction (fun n ⇒ (0 : F) ≤ of_nat n)).
    - intros n n' Hn. setoid_rewrite (of_nat_equ n n' Hn). reflexivity.
    - setoid_rewrite of_nat_zero. reflexivity.
    - intros n IH. setoid_rewrite (of_nat_succ n).
      exact (le_of_lt _ _ (lt_of_le_of_lt 0 (of_nat n) (of_nat n + 1) IH (lt_add_one (of_nat n)))).
  Qed.

  Theorem of_nat_succ_pos : ∀ n : N, (0 : F) < of_nat (succ n).
  Proof.
    intros n.
    apply (proj2 (lt_proper (0 : F) 0 (reflexivity _) (of_nat (succ n)) (of_nat n + 1) (of_nat_succ n))).
    exact (lt_of_le_of_lt 0 (of_nat n) (of_nat n + 1) (of_nat_nonneg n) (lt_add_one (of_nat n))).
  Qed.

  Theorem of_nat_add : ∀ n m : N, of_nat (n + m) == of_nat n + of_nat m.
  Proof.
    intros n.
    apply (nat_induction (fun m ⇒ of_nat (n + m) == of_nat n + of_nat m)).
    - intros m m' Hm.
      setoid_rewrite (of_nat_equ (n + m) (n + m') (op_proper n n (reflexivity n) m m' Hm)).
      setoid_rewrite (of_nat_equ m m' Hm). reflexivity.
    - calc (of_nat (n + 0))
        == (of_nat n) { (of_nat_equ (n + 0) n (right_identity (action := add) n)) }
        == (of_nat n + (0 : F)) { (right_identity (action := add) (of_nat n)) }
        == (of_nat n + of_nat 0) { of_nat_zero }.
    - intros m IH.
      calc (of_nat (n + succ m))
        == (of_nat (succ (n + m))) { (of_nat_equ (n + succ m) (succ (n + m)) (Theory.Naturals.add_succ n m)) }
        == (of_nat (n + m) + 1) { (of_nat_succ (n + m)) }
        == ((of_nat n + of_nat m) + 1) { IH }
        == (of_nat n + (of_nat m + 1)) { associative }
        == (of_nat n + of_nat (succ m)) { (of_nat_succ m) }.
  Qed.

  Theorem of_nat_le : ∀ n m : N, n ≤ m → of_nat n ≤ of_nat m.
  Proof.
    intros n m Hnm.
    destruct (proj1 (le_add n m) Hnm) as [c Hc].
    apply (proj2 (le_proper (of_nat n) (of_nat n) (reflexivity _) (of_nat m) (of_nat n + of_nat c)
                    (transitivity (of_nat_equ m (n + c) (symmetry Hc)) (of_nat_add n c)))).
    exact (transitivity (le_of_eq _ _ (symmetry (right_identity (action := add) (of_nat n))))
             (add_le_mono_l (0 : F) (of_nat c) (of_nat n) (of_nat_nonneg c))).
  Qed.

Below every positive element there is the inverse of a natural.
  Theorem archimedean_inv : ∀ eps : F, (0 : F) < eps → {n : N & inv (of_nat (succ n)) < eps}.
  Proof.
    intros eps Heps.
    destruct (archimedean (N := N) (of_nat := of_nat) (inv eps)) as [n Hn]. ∃ n.
    assert (Hlt : inv eps < of_nat (succ n))
      by exact (lt_of_lt_of_le _ _ _ Hn (of_nat_le n (succ n) (Theory.NatOrder.le_succ_diag n))).
    apply (proj1 (lt_proper (inv (of_nat (succ n))) (inv (of_nat (succ n))) (reflexivity _)
                    (inv (inv eps)) eps (inv_inv eps (fun H0 ⇒ proj2 Heps (symmetry H0))))).
    exact (inv_lt_inv (inv eps) (of_nat (succ n)) (inv_pos eps Heps) Hlt).
  Defined.
End ArchimedeanTheory.
