Library Riktig.Structure.CPSLambda

From Riktig.Structure Require Export Lambda.

Continuation-passing style over a lambda calculus: triv are the terms with no work in them, tail the terms whose every call is in head position -- a call applies trivial terms only, and is the whole term. The translation cps makes every term trivial, and applying a translated trivial term to a continuation hands it the stripped value psi -- and so does the translation of any term that evaluates to one.
Module CPSLambda.
  Class CPSLambda (N T Sb : Type) `{LambdaCalculus N T Sb}
      (trivial : Pred T) (tail : Pred T) (cps : T → T) (psi : T → T)
      (evaluates : T → T → Prop) := {
    trivial_var : ∀ n : N, trivial (var n);
    trivial_lam : ∀ b : T, tail b → trivial (lam b);
    tail_of_trivial : ∀ v : T, trivial v → tail v;
    tail_call : ∀ f v : T, trivial f → trivial v → tail (app f v);
    tail_call2 : ∀ f v w : T, trivial f → trivial v → trivial w → tail (app (app f v) w);
    cps_trivial : ∀ t : T, trivial (cps t);
    cps_value : ∀ v k : T, trivial v → app (cps v) k == app k (psi v);
    
    cps_runs : ∀ l v k : T, evaluates l v → app (cps l) k == app k (psi v);
  }.
  Hint Mode CPSLambda - ! - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End CPSLambda.
Export CPSLambda.
