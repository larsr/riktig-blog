From Riktig.Structure Require Export Ring.

A Boolean ring: a ring in which every element is its own square. Read as bits, 0 is absent and 1 is present, addition is exclusive or and multiplication is conjunction.
Module Boolean.
  Class Boolean (K : Type) {eq : EqRel K} {zero : Zero K} {add : HasAdd K}
                {neg : HasNeg K} {one : One K} {mult : HasMult K} := {
    ring :: Ring K;
    is_idempotent :: Idempotent K mult;
  }.
  Hint Mode Boolean ! - - - - - - : typeclass_instances.
End Boolean.
Export Boolean.
