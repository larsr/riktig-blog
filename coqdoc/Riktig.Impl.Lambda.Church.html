#[local] Notation "'λ' a" := (Abs a) (at level 10).
#[local] Notation "# a" := (Var a) (at level 0).
#[local] Notation "⦅ a b .. c ⦆" := (App .. (App a b) .. c)
  (at level 0, a at level 0, b at level 0, c at level 0).


Definition TRUE : tm := λ λ #1.
Definition FALSE : tm := λ λ #0.

Definition IFB (b t e : tm) : tm := ⦅b t e⦆.

Definition PAIR : tm := λ λ λ ⦅#0 #2 #1⦆.
Definition FST (p : tm) : tm := ⦅p TRUE⦆.
Definition SND (p : tm) : tm := ⦅p FALSE⦆.

Instance IFB_Proper : Proper (equ ==> equ ==> equ ==> equ) IFB.
Proof.
  intros b b' Hb t t' Ht e e' He.
  unfold IFB. now rewrite Hb, Ht, He.
Qed.



Lemma TRUE_church : TRUE = church_true.
Proof. reflexivity. Qed.

Lemma FALSE_church : FALSE = church_false.
Proof. reflexivity. Qed.

Lemma PAIR_church : PAIR = church_pair.
Proof. reflexivity. Qed.

Lemma IFB_church : ∀ b g h : tm, IFB b g h = church_if b g h.
Proof. reflexivity. Qed.


Lemma IF_TRUE : ∀ g h, value g → value h → IFB TRUE g h == g.
Proof. exact church_if_true. Qed.

Lemma IF_FALSE : ∀ g h, value g → value h → IFB FALSE g h == h.
Proof. exact church_if_false. Qed.

Lemma FST_PAIR : ∀ v w, value v → value w →
  FST ⦅PAIR v w⦆ == v.
Proof. exact church_fst_pair. Qed.

Lemma SND_PAIR : ∀ v w, value v → value w →
  SND ⦅PAIR v w⦆ == w.
Proof. exact church_snd_pair. Qed.


Lemma if_of_true : ∀ b g h,
  value g → value h → b == TRUE → IFB b g h == g.
Proof.
  intros b g h Hg Hh Hb. rewrite Hb. apply IF_TRUE; assumption.
Qed.

Lemma if_of_false : ∀ b g h,
  value g → value h → b == FALSE → IFB b g h == h.
Proof.
  intros b g h Hg Hh Hb. rewrite Hb. apply IF_FALSE; assumption.
Qed.

Example fst_pair_closed : FST ⦅PAIR idt k⦆ == idt.
Proof. apply (conv_by_eval 20 10). reflexivity. Qed.


Fixpoint numeral_body (n : nat) : tm :=
  match n with
  | O ⇒ #0
  | S m ⇒ ⦅#1 (numeral_body m)⦆
  end.

Definition numeral (n : nat) : tm := λ λ (numeral_body n).

Definition PLUS : tm := λ λ λ λ ⦅#3 #1 ⦅#2 #1 #0⦆⦆.
Definition MULT : tm := λ λ λ ⦅#2 ⦅#1 #0⦆⦆.
