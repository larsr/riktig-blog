Section Level0Theorems.
  Context {F : Type} `{FF : Field F}.
  Context {Ob : Type} {car : Ob → VectObj F} {D : DerivSpec F Ob car}.

  Theorem deriv_of_identity : ∀ (X : Ob) (a : vo_car (car X)),
    differential X X (fun u : vo_car (car X) ⇒ u) a linear_map_id.
  Proof.
    intros X a.
    exact (deriv_of_linear X X linear_map_id a (admissible_identity X)).
  Qed.

  Theorem deriv_of_scaling : ∀ (X : Ob) (c : F) (a : vo_car (car X)),
    differential X X (fun u : vo_car (car X) ⇒ c *: u) a
      (c *: linear_map_id).
  Proof.
    intros X c a.
    exact (deriv_of_scaled X X c (fun u : vo_car (car X) ⇒ u) a
             linear_map_id (deriv_of_identity X a)).
  Qed.

  Theorem deriv_after_scaling : ∀ (X Y : Ob) (c : F)
      (g : vo_car (car X) → vo_car (car Y)) (a : vo_car (car X))
      (Lg : LinearMap F (vo_car (car X)) (vo_car (car Y))),
    Proper (equ ==> equ) g →
    differential X Y g (c *: a) Lg →
    differential X Y (fun x : vo_car (car X) ⇒ g (c *: x)) a
      (linear_map_comp Lg (c *: linear_map_id)).
  Proof.
    intros X Y c g a Lg Pg Hg.
    exact (deriv_of_composite X X Y (fun u : vo_car (car X) ⇒ c *: u) g a
             (c *: linear_map_id) Lg Pg (deriv_of_scaling X c a) Hg).
  Qed.

  Lemma composite_proper : ∀ (X Y Z : Ob)
      (f : vo_car (car X) → vo_car (car Y))
      (g : vo_car (car Y) → vo_car (car Z)),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    Proper (equ ==> equ) (fun u : vo_car (car X) ⇒ g (f u)).
  Proof.
    intros X Y Z f g Pf Pg u v Huv.
    exact (Pg _ _ (Pf _ _ Huv)).
  Qed.

  Theorem chain_reverse_mode : ∀ (X Y Z W : Ob)
      (f1 : vo_car (car X) → vo_car (car Y))
      (f2 : vo_car (car Y) → vo_car (car Z))
      (f3 : vo_car (car Z) → vo_car (car W)) (a : vo_car (car X))
      (L1 : LinearMap F (vo_car (car X)) (vo_car (car Y)))
      (L2 : LinearMap F (vo_car (car Y)) (vo_car (car Z)))
      (L3 : LinearMap F (vo_car (car Z)) (vo_car (car W))),
    Proper (equ ==> equ) f2 → Proper (equ ==> equ) f3 →
    differential X Y f1 a L1 → differential Y Z f2 (f1 a) L2 →
    differential Z W f3 (f2 (f1 a)) L3 →
    differential X W (fun u : vo_car (car X) ⇒ f3 (f2 (f1 u))) a
      (linear_map_comp (linear_map_comp L3 L2) L1).
  Proof.
    intros X Y Z W f1 f2 f3 a L1 L2 L3 P2 P3 H1 H2 H3.
    exact (deriv_of_composite X Y W f1 (fun v : vo_car (car Y) ⇒ f3 (f2 v))
             a L1 (linear_map_comp L3 L2) (composite_proper Y Z W f2 f3 P2 P3)
             H1 (deriv_of_composite Y Z W f2 f3 (f1 a) L2 L3 P3 H2 H3)).
  Qed.

  Theorem chain_forward_mode : ∀ (X Y Z W : Ob)
      (f1 : vo_car (car X) → vo_car (car Y))
      (f2 : vo_car (car Y) → vo_car (car Z))
      (f3 : vo_car (car Z) → vo_car (car W)) (a : vo_car (car X))
      (L1 : LinearMap F (vo_car (car X)) (vo_car (car Y)))
      (L2 : LinearMap F (vo_car (car Y)) (vo_car (car Z)))
      (L3 : LinearMap F (vo_car (car Z)) (vo_car (car W))),
    Proper (equ ==> equ) f2 → Proper (equ ==> equ) f3 →
    differential X Y f1 a L1 → differential Y Z f2 (f1 a) L2 →
    differential Z W f3 (f2 (f1 a)) L3 →
    differential X W (fun u : vo_car (car X) ⇒ f3 (f2 (f1 u))) a
      (linear_map_comp L3 (linear_map_comp L2 L1)).
  Proof.
    intros X Y Z W f1 f2 f3 a L1 L2 L3 P2 P3 H1 H2 H3.
    exact (deriv_of_composite X Z W (fun u : vo_car (car X) ⇒ f2 (f1 u)) f3
             a (linear_map_comp L2 L1) L3 P3
             (deriv_of_composite X Y Z f1 f2 a L1 L2 P2 H1 H2) H3).
  Qed.

Forward mode and reverse mode agree because a function has one derivative — not because matrix multiplication is associative. The two are the same fact seen twice: lm_comp_associative proves it for every pair of composable arrows, and uniqueness proves it only where the derivatives exist, so the categorical statement is the stronger one.
  Corollary forward_equals_reverse_mode : ∀ (X Y Z W : Ob)
      (f1 : vo_car (car X) → vo_car (car Y))
      (f2 : vo_car (car Y) → vo_car (car Z))
      (f3 : vo_car (car Z) → vo_car (car W)) (a : vo_car (car X))
      (L1 : LinearMap F (vo_car (car X)) (vo_car (car Y)))
      (L2 : LinearMap F (vo_car (car Y)) (vo_car (car Z)))
      (L3 : LinearMap F (vo_car (car Z)) (vo_car (car W))),
    Proper (equ ==> equ) f2 → Proper (equ ==> equ) f3 →
    differential X Y f1 a L1 → differential Y Z f2 (f1 a) L2 →
    differential Z W f3 (f2 (f1 a)) L3 →
    linear_map_comp (linear_map_comp L3 L2) L1
      == linear_map_comp L3 (linear_map_comp L2 L1).
  Proof.
    intros X Y Z W f1 f2 f3 a L1 L2 L3 P2 P3 H1 H2 H3.
    exact (differential_unique X W (fun u : vo_car (car X) ⇒ f3 (f2 (f1 u)))
             a (linear_map_comp (linear_map_comp L3 L2) L1)
             (linear_map_comp L3 (linear_map_comp L2 L1))
             (chain_reverse_mode X Y Z W f1 f2 f3 a L1 L2 L3 P2 P3 H1 H2 H3)
             (chain_forward_mode X Y Z W f1 f2 f3 a L1 L2 L3 P2 P3 H1 H2 H3)).
  Qed.

End Level0Theorems.
