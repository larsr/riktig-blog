From Riktig.Structure Require Export Order NatOrder.

An Archimedean ordered field: the naturals embed by zero and successor, and above every element there is a natural, given.
Module Archimedean.
  Class Archimedean (F N : Type) `{OrderedField F} `{NatOrder N} (of_nat : N → F) := {
    of_nat_proper :: Proper (equ ==> equ) of_nat;
    of_nat_zero : of_nat 0 == 0;
    of_nat_succ : ∀ n : N, of_nat (succ n) == of_nat n + 1;
    archimedean : ∀ q : F, {n : N & q < of_nat n};
  }.
  Hint Mode Archimedean ! ! - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Archimedean.
Export Archimedean.
