A field as a graded geometric algebra of top grade 0: everything has grade 0, the grade-0 part is the element itself, every other part is zero, and the reversal is the identity.
Section GradedScalar.
  Context {F N : Type} `{Field F} `{Naturals N} {EN : EquDec N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Lemma succ_ne_zero : ∀ k : N, succ k =/= 0.
  Proof. intros k. exact (succ_not_zero k). Qed.

  #[export] Instance Scalar_HasGrade : HasGrade N F := fun k a ⇒ if dec (k == 0) then a else 0.
  #[export] Instance Scalar_HasScalarPart : HasScalarPart F F := fun a ⇒ a.
  #[export] Instance Scalar_HasReverse : HasReverse F := fun a ⇒ a.

  Lemma scalar_grade_at_zero : ∀ (k : N) (a : F), k == 0 → ⟨a⟩_k == a.
  Proof. intros k a Hk. unfold grade_part, Scalar_HasGrade. destruct (dec (P := k == 0)) as [_ | Hnz]; [ reflexivity | destruct (Hnz Hk) ]. Qed.

  Lemma scalar_grade_off_zero : ∀ (k : N) (a : F), k =/= 0 → ⟨a⟩_k == 0.
  Proof. intros k a Hk. unfold grade_part, Scalar_HasGrade. destruct (dec (P := k == 0)) as [Hz | _]; [ destruct (Hk Hz) | reflexivity ]. Qed.

  #[export] Instance Scalar_grade_Linear : ∀ k : N, Linear F F F (grade_part k).
  Proof.
    intros k. unfold grade_part, Scalar_HasGrade. destruct (dec (P := k == 0)).
    - split.
      + intros a a' Ha. exact Ha.
      + intros a b. reflexivity.
      + intros a b. reflexivity.
    - split.
      + intros a a' _. reflexivity.
      + intros a b. symmetry. exact (left_identity (action := add) (0 : F)).
      + intros a b. symmetry. exact (Theory.Ring.mul_zero a).
  Qed.

  #[export] Instance Scalar_Graded : Graded N F F Scalar_HasGrade 0.
  Proof.
    split.
    - exact Scalar_grade_Linear.
    - intros k k' x Hk. unfold grade_part, Scalar_HasGrade.
      destruct (dec (P := k == 0)) as [Hz | Hz]; destruct (dec (P := k' == 0)) as [Hz' | Hz']; try reflexivity.
      + destruct Hz'. transitivity k; [ symmetry; exact Hk | exact Hz ].
      + destruct Hz. transitivity k'; [ exact Hk | exact Hz' ].
    - intros k x. unfold grade_part, Scalar_HasGrade. destruct (dec (P := k == 0)); reflexivity.
    - intros k l x Hkl. unfold grade_part, Scalar_HasGrade.
      destruct (dec (P := l == 0)) as [Hl | Hl]; destruct (dec (P := k == 0)) as [Hk | Hk]; try reflexivity.
      destruct Hkl. transitivity (0 : N); [ exact Hk | symmetry; exact Hl ].
    - intros x.
      calc (natrec F 0 (fun k acc ⇒ acc + ⟨x⟩_k) (succ 0))
        == (natrec F 0 (fun k acc ⇒ acc + ⟨x⟩_k) 0 + ⟨x⟩_0) { (natrec_succ F _ _ 0 _ 0) }
        == ((0 : F) + ⟨x⟩_0) { (natrec_zero F _ _ 0 _) }
        == (⟨x⟩_0) { (left_identity (action := add) (⟨x⟩_0)) }
        == x { (scalar_grade_at_zero 0 x (reflexivity 0)) }.
  Qed.

  #[export] Instance Scalar_Reversal : Reversal F Scalar_HasReverse.
  Proof.
    split.
    - intros a a' Ha. exact Ha.
    - intros a b. reflexivity.
    - intros a b. exact (commutative (op := mult) a b).
    - intros a. reflexivity.
  Qed.

  #[export] Instance Scalar_GradedGA : GradedGA N F unit F Scalar_HasGrade 0 Scalar_HasScalarPart Scalar_HasReverse.
  Proof.
    split.
    - exact Scalar_Graded.
    - exact Scalar_Reversal.
    - exact (scalar_grade_at_zero 0 1 (reflexivity 0)).
    - intros x. unfold scalar_part, Scalar_HasScalarPart. rewrite (scalar_grade_at_zero 0 x (reflexivity 0)). symmetry. exact (right_identity (action := mult) x).
    - intros u. change (vec u) with (0 : F). exact (scalar_grade_off_zero (succ 0) 0 (succ_ne_zero 0)).
    - intros u x. change (vec u) with (0 : F). rewrite (Theory.Ring.zero_mul (⟨x⟩_0)).
      symmetry. exact (scalar_grade_off_zero (succ 0) 0 (succ_ne_zero 0)).
    - intros u x k. change (vec u) with (0 : F). rewrite (Theory.Ring.zero_mul (⟨x⟩_(succ k))).
      rewrite (scalar_grade_off_zero (succ (succ k)) 0 (succ_ne_zero (succ k))), (right_identity (action := add) _).
      symmetry. exact (additive_zero (grade_part (A := F) k)).
    - intros k x. reflexivity.
    - intros k x. reflexivity.
    - intros a x. reflexivity.
    - intros u. reflexivity.
    - intros x. reflexivity.
  Qed.
End GradedScalar.
