From Coq Require Import Arith.
From Riktig Require Export Structure.Decidable Structure.Finite Theory.Finite Impl.NatOrderNat.

Equality on a finite index numbered by nat is decided through the numbering, since numbering is injective.
Section EquDecFinite.
  Context {I : Type} {card : HasCard I nat} {index : HasIndex I nat} {elem : HasElem I nat}
          {HF : Finite I nat card index elem (eqI := eq)}.

  #[export] Instance Finite_EquDec : EquDec I (eqA := eq) :=
    fun i j ⇒
      match Nat.eq_dec (index i) (index j) with
      | left H ⇒ left (index_injective i j H)
      | right H ⇒ right (fun Hij ⇒ H (f_equal index Hij))
      end.
End EquDecFinite.
