Library Riktig.Impl.VectorSpaceProd

From Riktig Require Export Structure.VectorSpace.

The product of two vector spaces over a field: pairs, with every operation taken in each component.
Section ProdVS.
  Context {F V W : Type} `{VectorSpace F V}
    {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
    {VSW : VectorSpace F W zW aW nW sW}.

  #[export] Instance Prod_EqRel : EqRel (V × W) := fun x y ⇒ fst x == fst y ∧ snd x == snd y.
  #[export] Instance Prod_Zero : Zero (V × W) := (0, 0).
  #[export] Instance Prod_HasAdd : HasAdd (V × W) := fun x y ⇒ (fst x + fst y, snd x + snd y).
  #[export] Instance Prod_HasNeg : HasNeg (V × W) := fun x ⇒ (- fst x, - snd x).
  #[export] Instance Prod_HasScale : HasScale F (V × W) := fun a x ⇒ (a *: fst x, a *: snd x).

  #[export] Instance Prod_Equivalence : Equivalence (@equ (V × W) Prod_EqRel).
  Proof.
    split.
    - intros x. split; reflexivity.
    - intros x y [H1 H2]. split; symmetry; assumption.
    - intros x y z [H1 H2] [H3 H4]. split; etransitivity; eassumption.
  Qed.

  #[export] Instance Prod_Magma : Magma (V × W) Prod_HasAdd.
  Proof.
    split.
    - exact Prod_Equivalence.
    - intros x x' [Hx1 Hx2] y y' [Hy1 Hy2]. split; simpl; apply op_proper; assumption.
  Qed.
  #[export] Instance Prod_Semigroup : Semigroup (V × W) Prod_HasAdd.
  Proof. split. exact Prod_Magma. intros x y z. split; simpl; apply associative. Qed.
  #[export] Instance Prod_Monoid : Monoid (V × W) Prod_Zero Prod_HasAdd.
  Proof.
    split. exact Prod_Semigroup.
    - intros x. split; simpl; apply left_identity.
    - intros x. split; simpl; apply right_identity.
  Qed.
  #[export] Instance Prod_Group : Group (V × W) Prod_Zero Prod_HasAdd Prod_HasNeg.
  Proof.
    split. exact Prod_Monoid.
    - intros x x' [H1 H2]. split; simpl; apply neg_proper; assumption.
    - intros x. split; simpl; apply left_inverse.
    - intros x. split; simpl; apply right_inverse.
  Qed.
  #[export] Instance Prod_AbGroup : AbGroup (V × W) Prod_Zero Prod_HasAdd Prod_HasNeg.
  Proof. split. exact Prod_Group. intros x y. split; simpl; apply commutative. Qed.
  #[export] Instance Prod_ScaleLaws : ScaleLaws F (V × W) Prod_HasAdd Prod_HasScale.
  Proof.
    split.
    - intros a a' Ha x x' [H1 H2]. split; simpl; apply scale_proper; assumption.
    - intros a x y. split; simpl; apply left_distributive.
    - intros a b x. split; simpl; apply right_distributive.
    - intros a b x. split; simpl; apply associative.
    - intros x. split; simpl; apply left_identity.
  Qed.
  #[export] Instance Prod_VectorSpace : VectorSpace F (V × W) Prod_Zero Prod_HasAdd Prod_HasNeg Prod_HasScale :=
    {| VectorSpace.vectors := Prod_AbGroup; VectorSpace.scaling := Prod_ScaleLaws |}.
End ProdVS.
