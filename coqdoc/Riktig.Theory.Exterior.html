Section ElementWedge.
  Context {A : Type} `{Ring A}.

  Definition wedge (x y : A) : A := x × y + - (y × x).

  #[export] Instance wedge_proper : Proper (equ ==> equ ==> equ) wedge.
  Proof.
    intros x x' Hx y y' Hy.
    unfold wedge.
    now rewrite Hx, Hy.
  Qed.

  Theorem wedge_self_zero : ∀ x : A, wedge x x == 0.
  Proof.
    intros x.
    unfold wedge.
    exact (right_inverse (x × x)).
  Qed.

  Theorem wedge_anti : ∀ x y : A, wedge x y == - wedge y x.
  Proof.
    intros x y.
    unfold wedge.
    rewrite (neg_add (y × x) (- (x × y))), (neg_involutive (x × y)).
    exact (commutative _ _).
  Qed.

  Theorem wedge_add_l : ∀ x x' y : A,
    wedge (x + x') y == wedge x y + wedge x' y.
  Proof.
    intros x x' y.
    unfold wedge.
    calc (((x + x') × y + - (y × (x + x'))) : A)
      == (((x × y + x' × y) + - (y × (x + x'))) : A)
        { right_distributive }
      == (((x × y + x' × y) + - (y × x + y × x')) : A)
        { left_distributive }
      == (((x × y + x' × y) + (- (y × x) + - (y × x'))) : A)
        { additive[neg] }
      == (((x × y + - (y × x)) + (x' × y + - (y × x'))) : A)
        { add_interchange }.
  Qed.

  Theorem wedge_add_r : ∀ x y y' : A,
    wedge x (y + y') == wedge x y + wedge x y'.
  Proof.
    intros x y y'.
    unfold wedge.
    calc ((x × (y + y') + - ((y + y') × x)) : A)
      == (((x × y + x × y') + - ((y + y') × x)) : A)
        { left_distributive }
      == (((x × y + x × y') + - (y × x + y' × x)) : A)
        { right_distributive }
      == (((x × y + x × y') + (- (y × x) + - (y' × x))) : A)
        { additive[neg] }
      == (((x × y + - (y × x)) + (x × y' + - (y' × x))) : A)
        { add_interchange }.
  Qed.

  #[export] Instance wedge_l_additive (y : A) :
    Additive A A add add (fun x ⇒ wedge x y) :=
    fun x x' ⇒ wedge_add_l x x' y.

  #[export] Instance wedge_r_additive (x : A) :
    Additive A A add add (wedge x) :=
    wedge_add_r x.

  Theorem mul_decompose : ∀ x y : A,
    x × y + x × y == (x × y + y × x) + wedge x y.
  Proof.
    intros x y.
    unfold wedge.
    calc ((x × y + x × y) : A)
      == (((x × y + x × y) + 0) : A)
        { right_identity }
      == (((x × y + x × y) + (y × x + - (y × x))) : A)
        { right_inverse }
      == (((x × y + y × x) + (x × y + - (y × x))) : A)
        { add_interchange }.
  Qed.

End ElementWedge.

Section ElementDecomposition.
  Context {F A : Type} `{Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A}
    {oA : One A} {mA : HasMult A} {sA : HasScale F A}.
  Context {RA : Ring A} {SL : ScaleLaws F A}.

  Theorem scalar_squares_decompose : ∀ (x y : A) (p q s : F),
    x × x == p *: 1 → y × y == q *: 1 → (x + y) × (x + y) == s *: 1 →
    x × y + x × y == ((s + - p) + - q) *: 1 + wedge x y.
  Proof.
    intros x y p q s Hx Hy Hs.
    calc ((x × y + x × y) : A)
      == (((x × y + y × x) + wedge x y) : A)
        { mul_decompose }
      == ((((s + - p) + - q) *: (1 : A) + wedge x y) : A)
        { scalar_squares_polarization }.
  Qed.

End ElementDecomposition.

Section ExteriorTheory.
  Context {F V A : Type} {Qform : V → F} `{GeometricAlgebra F V A (Q := Qform)}.

  Theorem ga_symmetric_part : ∀ u v : V,
    ga_vec u × ga_vec v + ga_vec v × ga_vec u == polar Qform u v *: (1 : A).
  Proof. exact polarization. Qed.

  Theorem ga_decompose : ∀ u v : V,
    ga_vec u × ga_vec v + ga_vec u × ga_vec v
    == polar Qform u v *: (1 : A) + wedge (ga_vec u) (ga_vec v).
  Proof.
    intros u v.
    exact (scalar_squares_decompose (ga_vec u) (ga_vec v)
             (Qform u) (Qform v) (Qform (u + v))
             (contraction u) (contraction v) (contraction_add u v)).
  Qed.

End ExteriorTheory.

Section VolumeElement.
  Context {F V : Type} `{VV : VectorSpace F V} {p : nat}.
  Context (B : Basis (S p) (Fin.t (S p)) F V).

  Lemma mat_of_fam_lm : ∀ (f : LinearMap F V V) (vs : Fin.t (S p) → V)
      (i j : Fin.t (S p)),
    mat_of_fam B (fun k ⇒ f (vs k)) i j
      == mat_mul (mat_of B B f) (mat_of_fam B vs) i j.
  Proof.
    intros f vs i j.
    exact (mat_of_coords B B f (vs j) i).
  Qed.

  Section VolumeElementConditional.
    Context (detM_mul : ∀ (m : nat) (A A' : Matrix m m F),
      detM (mat_mul A A') == detM A × detM A').

    Theorem omega_B_lm : ∀ (f : LinearMap F V V) (vs : Fin.t (S p) → V),
      omega_B B (fun k ⇒ f (vs k)) == det B f × omega_B B vs.
    Proof.
      intros f vs.
      calc (omega_B B (fun k ⇒ f (vs k)))
        == (detM (mat_of_fam B (fun k ⇒ f (vs k))))
        == (detM (mat_mul (mat_of B B f) (mat_of_fam B vs)))
        == (detM (mat_of B B f) × detM (mat_of_fam B vs))
        == (det B f × omega_B B vs).
      - reflexivity.
      - exact (detM_proper (S p) (mat_of_fam B (fun k ⇒ f (vs k)))
                 (mat_mul (mat_of B B f) (mat_of_fam B vs))
                 (fun i j ⇒ mat_of_fam_lm f vs i j)).
      - exact (detM_mul (S p) (mat_of B B f) (mat_of_fam B vs)).
      - reflexivity.
    Qed.

  End VolumeElementConditional.

  Theorem omega_B_lm_id : ∀ vs : Fin.t (S p) → V,
    omega_B B (fun k ⇒ (linear_map_id (F:=F) (U:=V)) (vs k)) == omega_B B vs.
  Proof.
    intros vs.
    apply (omega_B_proper B).
    intros k.
    reflexivity.
  Qed.

End VolumeElement.

Corollary omega_B_lm_Q : ∀ (V : Type)
    (eV : EqRel V) (zV : Zero V) (aV : HasAdd V) (nV : HasNeg V)
    (sV : HasScale Q V) (VV : VectorSpace Q V) (p : nat)
    (B : Basis (S p) (Fin.t (S p)) Q V) (f : LinearMap Q V V) (vs : Fin.t (S p) → V),
  omega_B B (fun k ⇒ f (vs k)) == det B f × omega_B B vs.
Proof.
  intros V eV zV aV nV sV VV p B f vs.
  exact (omega_B_lm B (fun m A A' ⇒ detM_mul_Q m A A') f vs).
Qed.
