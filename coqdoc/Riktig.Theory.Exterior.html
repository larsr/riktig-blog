Section ElementWedge.
  Context {A : Type} `{Ring A}.

  Definition wedge (x y : A) : A := x × y + - (y × x).

  #[export] Instance wedge_proper : Proper (equ ==> equ ==> equ) wedge.
  Proof.
    intros x x' Hx y y' Hy.
    unfold wedge.
    now rewrite Hx, Hy.
  Qed.

  Theorem wedge_self_zero : ∀ x : A, wedge x x == 0.
  Proof.
    intros x.
    unfold wedge.
    exact (right_inverse (x × x)).
  Qed.

  Theorem wedge_anti : ∀ x y : A, wedge x y == - wedge y x.
  Proof.
    intros x y.
    unfold wedge.
    rewrite (neg_add (y × x) (- (x × y))), (neg_involutive (x × y)).
    exact (commutative _ _).
  Qed.

  Theorem wedge_add_l : ∀ x x' y : A,
    wedge (x + x') y == wedge x y + wedge x' y.
  Proof.
    intros x x' y.
    unfold wedge.
    calc (((x + x') × y + - (y × (x + x'))) : A)
      == (((x × y + x' × y) + - (y × (x + x'))) : A)
        { right_distributive }
      == (((x × y + x' × y) + - (y × x + y × x')) : A)
        { left_distributive }
      == (((x × y + x' × y) + (- (y × x) + - (y × x'))) : A)
        { additive[neg] }
      == (((x × y + - (y × x)) + (x' × y + - (y × x'))) : A)
        { add_interchange }.
  Qed.

  Theorem wedge_add_r : ∀ x y y' : A,
    wedge x (y + y') == wedge x y + wedge x y'.
  Proof.
    intros x y y'.
    unfold wedge.
    calc ((x × (y + y') + - ((y + y') × x)) : A)
      == (((x × y + x × y') + - ((y + y') × x)) : A)
        { left_distributive }
      == (((x × y + x × y') + - (y × x + y' × x)) : A)
        { right_distributive }
      == (((x × y + x × y') + (- (y × x) + - (y' × x))) : A)
        { additive[neg] }
      == (((x × y + - (y × x)) + (x × y' + - (y' × x))) : A)
        { add_interchange }.
  Qed.

  #[export] Instance wedge_l_additive (y : A) :
    Additive A A add add (fun x ⇒ wedge x y) :=
    fun x x' ⇒ wedge_add_l x x' y.

  #[export] Instance wedge_r_additive (x : A) :
    Additive A A add add (wedge x) :=
    wedge_add_r x.

  Theorem mul_decompose : ∀ x y : A,
    x × y + x × y == (x × y + y × x) + wedge x y.
  Proof.
    intros x y.
    unfold wedge.
    exact (add_double_split (x × y) (y × x)).
  Qed.

End ElementWedge.

Section ElementDecomposition.
  Context {F A : Type} `{Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A}
    {oA : One A} {mA : HasMult A} {sA : HasScale F A}.
  Context {RA : Ring A} {SL : ScaleLaws F A}.

  Theorem scalar_squares_decompose : ∀ (x y : A) (p q s : F),
    x × x == p *: 1 → y × y == q *: 1 → (x + y) × (x + y) == s *: 1 →
    x × y + x × y == ((s + - p) + - q) *: 1 + wedge x y.
  Proof.
    intros x y p q s Hx Hy Hs.
    calc ((x × y + x × y) : A)
      == (((x × y + y × x) + wedge x y) : A)
        { mul_decompose }
      == ((((s + - p) + - q) *: (1 : A) + wedge x y) : A)
        { scalar_squares_polarization }.
  Qed.

End ElementDecomposition.

Section ExteriorTheory.

  Theorem ga_symmetric_part : ∀ u v : V,
    ga_vec u × ga_vec v + ga_vec v × ga_vec u == polar Qform u v *: (1 : A).
  Proof. exact polarization. Qed.

  Theorem ga_decompose : ∀ u v : V,
    ga_vec u × ga_vec v + ga_vec u × ga_vec v
    == polar Qform u v *: (1 : A) + wedge (ga_vec u) (ga_vec v).
  Proof.
    intros u v.
    exact (scalar_squares_decompose (ga_vec u) (ga_vec v)
             (Qform u) (Qform v) (Qform (u + v))
             (contraction u) (contraction v) (contraction_add u v)).
  Qed.

Section VolumeElement.
  Context {F V : Type} `{VV : VectorSpace F V} {p : nat}.
  Context (B : Basis (S p) (Fin.t (S p)) F V).

  Lemma mat_of_fam_lm : ∀ (f : LinearMap F V V) (vs : Fin.t (S p) → V)
      (i j : Fin.t (S p)),
    mat_of_fam B (fun k ⇒ f (vs k)) i j
      == mat_mul (mat_of B B f) (mat_of_fam B vs) i j.
  Proof.
    intros f vs i j.
    calc (mat_of_fam B (fun k ⇒ f (vs k)) i j)
      == (bigopF (fun l ⇒ mat_of B B f i l × mat_of_fam B vs l j))
           { (mat_of_coords B B f (vs j) i) }
      == (mat_mul (mat_of B B f) (mat_of_fam B vs) i j)
           { (bigopF_bigopI (S p)
               (fun l ⇒ mat_of B B f i l × mat_of_fam B vs l j)) }.
  Qed.

  Theorem omega_B_lm : ∀ (f : LinearMap F V V) (vs : Fin.t (S p) → V),
    omega_B B (fun k ⇒ f (vs k)) == det B f × omega_B B vs.
  Proof.
    intros f vs.
    calc (omega_B B (fun k ⇒ f (vs k)))
      == (detM (mat_of_fam B (fun k ⇒ f (vs k))))
      == (detM (mat_mul (mat_of B B f) (mat_of_fam B vs)))
      == (detM (mat_of B B f) × detM (mat_of_fam B vs))
      == (det B f × omega_B B vs).
    - reflexivity.
    - exact (detM_proper (S p) (mat_of_fam B (fun k ⇒ f (vs k)))
               (mat_mul (mat_of B B f) (mat_of_fam B vs))
               (fun i j ⇒ mat_of_fam_lm f vs i j)).
    - exact (detM_mul (S p) (mat_of B B f) (mat_of_fam B vs)).
    - reflexivity.
  Qed.

  Theorem omega_B_lm_id : ∀ vs : Fin.t (S p) → V,
    omega_B B (fun k ⇒ (linear_map_id (F:=F) (U:=V)) (vs k)) == omega_B B vs.
  Proof.
    intros vs.
    apply (omega_B_proper B).
    intros k.
    reflexivity.
  Qed.

End VolumeElement.


Section VectorProduct.
  Context {F V A : Type} {Qform : V → F}
    `{GA : GeometricAlgebra F V A (Q := Qform)}.
  Context (vec_product : list V → A)
    (vec_product_nil : vec_product [] == 1)
    (vec_product_cons : ∀ (v : V) (l : list V),
       vec_product (v :: l) == ga_vec v × vec_product l).

  Lemma vec_product_ext : ∀ l l' : list V,
    Forall2 equ l l' → vec_product l == vec_product l'.
  Proof.
    intros l l' Hl; induction Hl as [| v v' l l' Hv _ IH].
    - reflexivity.
    - calc (vec_product (v :: l))
        == (ga_vec v × vec_product l) { vec_product_cons }
        == (ga_vec v' × vec_product l) { Hv }
        == (ga_vec v' × vec_product l') { IH }
        == (vec_product (v' :: l')) { vec_product_cons }.
  Qed.

  Lemma vec_product_single : ∀ v : V, vec_product [v] == ga_vec v.
  Proof.
    intros v.
    calc (vec_product [v])
      == (ga_vec v × vec_product []) { vec_product_cons }
      == (ga_vec v × (1 : A)) { vec_product_nil }
      == ((ga_vec v) : A) { mul_right_identity }.
  Qed.

  Lemma vec_product_app : ∀ l l' : list V,
    vec_product (l ++ l') == vec_product l × vec_product l'.
  Proof.
    intros l l'; induction l as [| v l IH].
    - calc (vec_product l')
        == ((1 : A) × vec_product l') { left_identity }
        == (vec_product [] × vec_product l') { vec_product_nil }.
    - calc (vec_product (v :: (l ++ l')))
        == (ga_vec v × vec_product (l ++ l'))
             { vec_product_cons }
        == (ga_vec v × (vec_product l × vec_product l'))
             { IH }
        == ((ga_vec v × vec_product l) × vec_product l')
             { (associative (ga_vec v) (vec_product l) (vec_product l')) }
        == (vec_product (v :: l) × vec_product l')
             { vec_product_cons }.
  Qed.

x is short of grade k when every linear functional that kills the products of at most k vectors kills x as well.
  Definition short_product (k : nat) (x : A) : Prop :=
    ∀ t : A → F, Linear F A F t →
      (∀ l : list V, (length l ≤ k)%nat → t (vec_product l) == 0) →
      t x == 0.

  Lemma short_product_respects : ∀ (k : nat) (x y : A),
    x == y → short_product k x → short_product k y.
  Proof.
    intros k x y Hxy Hx t Ht Hkill.
    rewrite <- Hxy.
    exact (Hx t Ht Hkill).
  Qed.

  Lemma short_product_of_product : ∀ (k : nat) (l : list V),
    (length l ≤ k)%nat → short_product k (vec_product l).
  Proof.
    intros k l Hl t _ Hkill.
    exact (Hkill l Hl).
  Qed.

  Lemma short_product_add : ∀ (k : nat) (x y : A),
    short_product k x → short_product k y → short_product k (x + y).
  Proof.
    intros k x y Hx Hy t Ht Hkill.
    calc (t (x + y))
      == (t x + t y) { additive }
      == ((0 : F) + t y) { (Hx t Ht Hkill) }
      == ((0 : F) + 0) { (Hy t Ht Hkill) }
      == (0 : F) { left_identity }.
  Qed.

  Lemma short_product_neg : ∀ (k : nat) (x : A),
    short_product k x → short_product k (- x).
  Proof.
    intros k x Hx t Ht Hkill.
    calc (t (- x))
      == (- t x) { (additive_negation t x) }
      == (- (0 : F)) { (Hx t Ht Hkill) }
      == (0 : F) { neg_zero }.
  Qed.

  Lemma short_product_scale : ∀ (k : nat) (a : F) (x : A),
    short_product k x → short_product k (a *: x).
  Proof.
    intros k a x Hx t Ht Hkill.
    calc (t (a *: x))
      == (a × t x) { homogeneous }
      == (a × (0 : F)) { (Hx t Ht Hkill) }
      == (0 : F) { mul_0_r }.
  Qed.

  Lemma short_product_mul_l : ∀ (k : nat) (l : list V) (x : A),
    short_product k x → short_product (length l + k) (vec_product l × x).
  Proof.
    intros k l x Hx t Ht Hkill.
    apply (Hx (fun y ⇒ t (vec_product l × y))).
    - exact (linear_compose (fun y ⇒ vec_product l × y) t
               (mul_linear_r (vec_product l)) Ht).
    - intros l0 Hl0.
      assert (Hlen : (length (l ++ l0) ≤ length l + k)%nat)
        by (rewrite length_app; lia).
      calc (t (vec_product l × vec_product l0))
        == (t (vec_product (l ++ l0))) { (vec_product_app l l0) }
        == (0 : F) { (Hkill (l ++ l0) Hlen) }.
  Qed.

  Lemma short_product_mul_r : ∀ (k : nat) (x : A) (l : list V),
    short_product k x → short_product (k + length l) (x × vec_product l).
  Proof.
    intros k x l Hx t Ht Hkill.
    apply (Hx (fun y ⇒ t (y × vec_product l))).
    - exact (linear_compose (fun y ⇒ y × vec_product l) t
               (mul_linear_l (vec_product l)) Ht).
    - intros l0 Hl0.
      assert (Hlen : (length (l0 ++ l) ≤ k + length l)%nat)
        by (rewrite length_app; lia).
      calc (t (vec_product l0 × vec_product l))
        == (t (vec_product (l0 ++ l))) { (vec_product_app l0 l) }
        == (0 : F) { (Hkill (l0 ++ l) Hlen) }.
  Qed.

  Lemma ga_vec_swap : ∀ u w : V,
    ga_vec u × ga_vec w == polar Qform u w *: (1 : A) + - (ga_vec w × ga_vec u).
  Proof.
    intros u w.
    calc ((ga_vec u × ga_vec w) : A)
      == ((ga_vec u × ga_vec w + 0) : A)
           { (right_identity (ga_vec u × ga_vec w : A)) }
      == ((ga_vec u × ga_vec w
            + (ga_vec w × ga_vec u + - (ga_vec w × ga_vec u))) : A)
           { (right_inverse (ga_vec w × ga_vec u : A)) }
      == (((ga_vec u × ga_vec w + ga_vec w × ga_vec u)
            + - (ga_vec w × ga_vec u)) : A)
           { (associative (ga_vec u × ga_vec w : A) (ga_vec w × ga_vec u)
                (- (ga_vec w × ga_vec u))) }
      == ((polar Qform u w *: (1 : A) + - (ga_vec w × ga_vec u)) : A)
           { (polarization u w) }.
  Qed.

A product of vectors in which one vector appears at both ends is short by two: the contraction collapses the repeat.
  Theorem short_product_repeat : ∀ (u : V) (m : list V),
    short_product (length m) (ga_vec u × (vec_product m × ga_vec u)).
  Proof.
    intros u m; induction m as [| w m IH].
    - apply (short_product_respects 0 (Qform u *: vec_product [])).
      + calc ((Qform u *: vec_product []) : A)
          == ((Qform u *: (1 : A)) : A) { vec_product_nil }
          == ((ga_vec u × ga_vec u) : A) { (contraction u) }
          == ((ga_vec u × ((1 : A) × ga_vec u)) : A)
               { (left_identity (ga_vec u : A)) }
          == ((ga_vec u × (vec_product [] × ga_vec u)) : A)
               { vec_product_nil }.
      + exact (short_product_scale 0 (Qform u) (vec_product [])
                 (short_product_of_product 0 [] (le_n 0))).
    -
      pose (Y := vec_product m × ga_vec u).
      assert (Hgather : ga_vec u × (vec_product (w :: m) × ga_vec u)
                        == (ga_vec u × ga_vec w) × Y).

          == (ga_vec u × ((ga_vec w × vec_product m) × ga_vec u))
               { (vec_product_cons w m) }
          == (ga_vec u × (ga_vec w × (vec_product m × ga_vec u)))
               { (associative (ga_vec w) (vec_product m) (ga_vec u)) }
          == ((ga_vec u × ga_vec w) × Y)
               { (associative (ga_vec u) (ga_vec w) Y) }. }

          == (polar Qform u w *: Y + - ((ga_vec w × ga_vec u) × Y))
               { (mul_neg_l (ga_vec w × ga_vec u) Y) }
          == (polar Qform u w *: Y + - (ga_vec w × (ga_vec u × Y)))
               { (associative (ga_vec w) (ga_vec u) Y) }. }
      apply (short_product_respects (length (w :: m))
               (polar Qform u w *: Y + - (ga_vec w × (ga_vec u × Y)))).
      + rewrite Hgather, Hexpand. reflexivity.
      + apply short_product_add.
        × apply short_product_scale.
          apply (short_product_respects (length (w :: m))
                   (vec_product (m ++ [u]))).
          -- calc (vec_product (m ++ [u]))
               == (vec_product m × vec_product [u]) { (vec_product_app m [u]) }
               == Y { (vec_product_single u) }.
          -- apply short_product_of_product.
             rewrite length_app. simpl. lia.
        × apply short_product_neg.
          apply (short_product_respects (length (w :: m))
                   (vec_product [w] × (ga_vec u × Y))).

  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context (discrete : ∀ i j : I, i == j → i = j).
  Context {Vu : HasUpd I V} {VuL : Upd I V} {Vl : HasLincomb I F V}.
  Context (top : A → F) {top_linear : Linear F A F top}.
  Context (top_short : ∀ l : list V,
             (length l < n)%nat → top (vec_product l) == 0).

  Lemma enum_In : ∀ i : I, In i enum.
  Proof.
    intros i.
    exact (In_of_InA_discrete I equ i enum discrete (enum_complete i)).
  Qed.

  Lemma enum_NoDup_discrete : NoDup (@enum n I eI Ieq IFin).
  Proof.
    exact (NoDup_of_NoDupA I (@equ I eI) enum
             (@Equivalence_Reflexive I (@equ I eI) enum_equivalence) enum_NoDup).
  Qed.

  Lemma enum_split : ∀ i : I,
    ∃ l1 l2 : list I, enum = l1 ++ i :: l2 ∧ ¬ In i (l1 ++ l2).
  Proof.
    intros i.
    destruct (in_split i enum (enum_In i)) as [l1 [l2 Hsplit]].
    ∃ l1, l2. split; [ exact Hsplit |].
    apply NoDup_remove_2.
    rewrite <- Hsplit.
    exact enum_NoDup_discrete.
  Qed.

  Lemma map_famEq : ∀ (vs ws : I → V) (l : list I),
    famEq vs ws → Forall2 equ (map vs l) (map ws l).
  Proof.
    intros vs ws l Hvw; induction l as [| i l IH]; simpl.
    - constructor.
    - constructor; [ exact (Hvw i) | exact IH ].
  Qed.

  Lemma map_upd_off : ∀ (vs : I → V) (i : I) (x : V) (l : list I),
    ¬ In i l → Forall2 equ (map (upd vs i x) l) (map vs l).
  Proof.
    intros vs i x l; induction l as [| k l IH]; intros Hi; simpl.
    - constructor.
    - constructor.
      + apply (upd_off vs i k x).
        intros Hc. apply Hi. left. exact (eq_sym (discrete i k Hc)).
      + apply IH. intros Hc. apply Hi. right. exact Hc.
  Qed.

  Lemma vec_product_upd : ∀ (vs : I → V) (i : I) (x : V) (l1 l2 : list I),
    enum = l1 ++ i :: l2 → ¬ In i (l1 ++ l2) →
    vec_product (map (upd vs i x) enum)
      == vec_product (map vs l1) × (ga_vec x × vec_product (map vs l2)).
  Proof.
    intros vs i x l1 l2 Hsplit Hnotin.
    assert (Hl1 : ¬ In i l1)
      by (intros Hc; apply Hnotin, in_or_app; left; exact Hc).
    assert (Hl2 : ¬ In i l2)
      by (intros Hc; apply Hnotin, in_or_app; right; exact Hc).
    assert (Hfa : Forall2 equ (map (upd vs i x) enum)
                    (map vs l1 ++ x :: map vs l2)).
    { rewrite Hsplit, map_app. simpl.
      apply Forall2_app; [ exact (map_upd_off vs i x l1 Hl1) |].
      constructor;
        [ exact (upd_at vs i x i (reflexivity i))
        | exact (map_upd_off vs i x l2 Hl2) ]. }
    calc ((vec_product (map (upd vs i x) enum)) : A)
      == ((vec_product (map vs l1 ++ x :: map vs l2)) : A)
           { (vec_product_ext _ _ Hfa) }
      == ((vec_product (map vs l1) × vec_product (x :: map vs l2)) : A)
           { (vec_product_app (map vs l1) (x :: map vs l2)) }
      == ((vec_product (map vs l1) × (ga_vec x × vec_product (map vs l2))) : A)
           { (vec_product_cons x (map vs l2)) }.
  Qed.

The candidate determinant: multiply the family's vectors in the index's own order and read the product at the top.
  Definition ga_top_form (vs : I → V) : F := top (vec_product (map vs enum)).

  #[export] Instance ga_top_form_proper :
    Proper (@famEq V _ I ==> equ) ga_top_form.
  Proof.
    intros vs ws Hvw.
    unfold ga_top_form.
    exact (Linear.proper (f := top) _ _
             (vec_product_ext _ _ (map_famEq vs ws enum Hvw))).
  Qed.

  Lemma ga_top_form_add : ∀ (vs : I → V) (i : I) (x y : V),
    ga_top_form (upd vs i (x + y))
      == ga_top_form (upd vs i x) + ga_top_form (upd vs i y).
  Proof.
    intros vs i x y.
    destruct (enum_split i) as [l1 [l2 [Hsplit Hnotin]]].
    unfold ga_top_form.
    rewrite (vec_product_upd vs i (x + y) l1 l2 Hsplit Hnotin),
            (vec_product_upd vs i x l1 l2 Hsplit Hnotin),
            (vec_product_upd vs i y l1 l2 Hsplit Hnotin).
    set (P1 := vec_product (map vs l1)).
    set (P2 := vec_product (map vs l2)).
    calc ((top (P1 × (ga_vec (x + y) × P2))) : F)
      == ((top (P1 × ((ga_vec x + ga_vec y) × P2))) : F)
           { additive[ga_vec] }
      == ((top (P1 × (ga_vec x × P2 + ga_vec y × P2))) : F)
           { (right_distributive (ga_vec x) (ga_vec y) P2) }
      == ((top (P1 × (ga_vec x × P2) + P1 × (ga_vec y × P2))) : F)
           { (left_distributive P1 (ga_vec x × P2) (ga_vec y × P2)) }
      == ((top (P1 × (ga_vec x × P2)) + top (P1 × (ga_vec y × P2))) : F)
           { additive[top] }.
  Qed.

  Lemma ga_top_form_scale : ∀ (vs : I → V) (i : I) (a : F) (x : V),
    ga_top_form (upd vs i (a *: x)) == a × ga_top_form (upd vs i x).
  Proof.
    intros vs i a x.
    destruct (enum_split i) as [l1 [l2 [Hsplit Hnotin]]].
    unfold ga_top_form.
    rewrite (vec_product_upd vs i (a *: x) l1 l2 Hsplit Hnotin),
            (vec_product_upd vs i x l1 l2 Hsplit Hnotin).
    set (P1 := vec_product (map vs l1)).
    set (P2 := vec_product (map vs l2)).
    calc ((top (P1 × (ga_vec (a *: x) × P2))) : F)
      == ((top (P1 × ((a *: ga_vec x) × P2))) : F)
           { homogeneous[ga_vec] }
      == ((top (P1 × (a *: (ga_vec x × P2)))) : F)
           { (homogeneous (f := fun z : A ⇒ z × P2) a (ga_vec x)) }
      == ((top (a *: (P1 × (ga_vec x × P2)))) : F)
           { (homogeneous (f := fun z : A ⇒ P1 × z) a (ga_vec x × P2)) }
      == ((a × top (P1 × (ga_vec x × P2))) : F)
           { homogeneous[top] }.
  Qed.

  Definition ga_top_multilinear_form : MultilinearForm I F V :=
    multilinear_form_from_laws ga_top_form ga_top_form_proper
      ga_top_form_add ga_top_form_scale.

Two equal vectors in the family make the product short by two grades, and top does not reach that far down.
  Theorem ga_top_form_alternating : Alternating ga_top_form.
  Proof.
    apply alternating_from_law.
    intros vs i j Hij Heq.
    assert (Hne : i ≠ j) by (intros Hc; apply Hij; rewrite Hc; reflexivity).
    destruct (two_in_split enum i j Hne (enum_In i) (enum_In j))
      as [l1 [a [l2 [b [l3 [Hsplit Hab]]]]]].
    assert (Hvab : vs b == vs a).
    { destruct Hab as [[Ha Hb] | [Ha Hb]]; subst;
        [ symmetry; exact Heq | exact Heq ]. }
    assert (Hprod : vec_product (map vs enum)
              == vec_product (map vs l1)
                   × ((ga_vec (vs a) × (vec_product (map vs l2) × ga_vec (vs b)))
                      × vec_product (map vs l3))).
    { rewrite Hsplit, map_app. simpl. rewrite map_app. simpl.
      calc ((vec_product (map vs l1
              ++ vs a :: (map vs l2 ++ vs b :: map vs l3))) : A)
        == ((vec_product (map vs l1)
               × vec_product (vs a :: (map vs l2 ++ vs b :: map vs l3))) : A)
             { (vec_product_app (map vs l1)
                  (vs a :: (map vs l2 ++ vs b :: map vs l3))) }
        == ((vec_product (map vs l1)
               × (ga_vec (vs a)
                  × vec_product (map vs l2 ++ vs b :: map vs l3))) : A)
             { (vec_product_cons (vs a) (map vs l2 ++ vs b :: map vs l3)) }
        == ((vec_product (map vs l1)
               × (ga_vec (vs a)
                  × (vec_product (map vs l2)
                     × vec_product (vs b :: map vs l3)))) : A)
             { (vec_product_app (map vs l2) (vs b :: map vs l3)) }
        == ((vec_product (map vs l1)
               × (ga_vec (vs a)
                  × (vec_product (map vs l2)
                     × (ga_vec (vs b) × vec_product (map vs l3))))) : A)
             { (vec_product_cons (vs b) (map vs l3)) }
        == ((vec_product (map vs l1)
               × (ga_vec (vs a)
                  × ((vec_product (map vs l2) × ga_vec (vs b))
                     × vec_product (map vs l3)))) : A)
             { (associative (vec_product (map vs l2)) (ga_vec (vs b))
                  (vec_product (map vs l3))) }
        == ((vec_product (map vs l1)
               × ((ga_vec (vs a) × (vec_product (map vs l2) × ga_vec (vs b)))
                  × vec_product (map vs l3))) : A)
             { (associative (ga_vec (vs a))
                  (vec_product (map vs l2) × ga_vec (vs b))
                  (vec_product (map vs l3))) }. }
    assert (Hmid : short_product (length l2)
              (ga_vec (vs a) × (vec_product (map vs l2) × ga_vec (vs b)))).
    { apply (short_product_respects (length l2)
               (ga_vec (vs a) × (vec_product (map vs l2) × ga_vec (vs a)))).
      - rewrite Hvab. reflexivity.
      - rewrite <- (length_map vs l2).
        exact (short_product_repeat (vs a) (map vs l2)). }
    assert (Hshort : short_product
              (length l1 + (length l2 + length l3))
              (vec_product (map vs enum))).
    { apply (short_product_respects _ _ _ (symmetry Hprod)).
      rewrite <- (length_map vs l1), <- (length_map vs l3).
      exact (short_product_mul_l _ (map vs l1) _
               (short_product_mul_r (length l2) _ (map vs l3) Hmid)). }
    assert (Hn : (length l1 + S (length l2 + S (length l3)))%nat = n).
    { transitivity (length (enum (n := n) (I := I))).
      - rewrite Hsplit, length_app. simpl. rewrite length_app. reflexivity.
      - exact enum_length. }
    unfold ga_top_form.
    apply (Hshort top top_linear).
    intros l Hl.
    apply top_short.
    lia.
  Qed.

A geometric algebra's top-grade reading is a determinant: the form is multilinear and alternating over the index carrier, and the basis product is where top returns 1.
  Definition ga_top_determinant (B : Basis n I F V)
      (top_basis : top (vec_product (map (Basis.vec B) enum)) == 1)
    : Determinant n I F V B :=
    determinant_from_laws ga_top_multilinear_form
      ga_top_form_alternating top_basis.
