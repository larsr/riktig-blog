Trigonometry on a commutative ring is a sine and a cosine that respect equality, take 0 to 0 and to 1, obey the addition formulas, and are odd and even.
Module Trig.
  Class Trig (R : Type) `{Ring R} `{!Commutative R R mult} (sin : HasSin R) (cos : HasCos R) := {
    sin_proper :: Proper (equ ==> equ) sin;
    cos_proper :: Proper (equ ==> equ) cos;
    sin_zero : sin 0 == 0;
    cos_zero : cos 0 == 1;
    sin_add : ∀ a b : R, sin (a + b) == sin a × cos b + cos a × sin b;
    cos_add : ∀ a b : R, cos (a + b) == cos a × cos b + - (sin a × sin b);
    sin_neg : ∀ a : R, sin (- a) == - sin a;
    cos_neg : ∀ a : R, cos (- a) == cos a;
  }.
  Hint Mode Trig ! - - - - - - - - - - : typeclass_instances.
End Trig.
Export Trig.
