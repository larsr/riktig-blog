From Riktig.Structure Require Export VectorSpace.

An inner product space is a vector space with a symmetric form that is linear in its first argument.
Module InnerProductSpace.
  Class InnerProductSpace (F V : Type) `{VectorSpace F V} {Vi : HasInner F V} := {
    inner_proper :: Proper (equ ==> equ ==> equ) (@inner F V _);
    inner_additive :: ∀ v : V, Additive V F add add (fun u ⇒ ⟪ u , v ⟫);
    inner_homogeneous :: ∀ v : V, Homogeneous F V F mult (fun a ⇒ a) (fun u ⇒ ⟪ u , v ⟫);
    is_commutative :: Commutative V F inner;
  }.
  Hint Mode InnerProductSpace - ! - - - - - - - - - - - - - - - : typeclass_instances.
End InnerProductSpace.
Export InnerProductSpace.
