Library Riktig.Lib.Coq.Z

From Coq Require Import ZArith Lia.
Open Scope Z_scope.

Every integer is reached from zero by adding and subtracting one.
Lemma Z_induction : ∀ P : Z → Prop,
  P 0 →
  (∀ z : Z, P z → P (z + 1)) →
  (∀ z : Z, P z → P (z + - 1)) →
  ∀ z : Z, P z.
Proof.
  intros P H0 Hup Hdn z.
  destruct (Z_lt_le_dec z 0) as [Hneg | Hpos].
  - remember (Z.to_nat (- z)) as n eqn:Hn.
    revert z Hn Hneg.
    induction n as [| n IH]; intros z Hn Hneg; [ lia | ].
    destruct (Z.eq_dec z (-1)) as [-> | Hne].
    + exact (Hdn 0 H0).
    + assert (Hstep : P (z + 1)) by (apply (IH (z + 1)); lia).
      replace z with (z + 1 + - 1) by lia. exact (Hdn _ Hstep).
  - remember (Z.to_nat z) as n eqn:Hn.
    revert z Hn Hpos.
    induction n as [| n IH]; intros z Hn Hpos.
    + replace z with 0 by lia. exact H0.
    + assert (Hstep : P (z + - 1)) by (apply IH; lia).
      replace z with (z + - 1 + 1) by lia. exact (Hup _ Hstep).
Qed.
