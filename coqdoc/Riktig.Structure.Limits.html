Library Riktig.Structure.Limits

From Riktig.Structure Require Export VectorSpace NatOrder.

Limits of sequences in a vector space, indexed by the naturals: convergence to a point and the Cauchy property carry their moduli as evidence, respect equality, and behave algebraically -- a limit is unique, constants converge, sums, negations and scalings of convergent sequences converge to the sums, negations and scalings of the limits, and so does the sequence without its first term; Cauchy sequences are closed under the same operations and include the convergent ones.
Module Limits.
  Class Limits (F N V : Type) `{VectorSpace F V} `{NatOrder N}
      (tendsto : HasTendsto N V) (cauchy : HasCauchy N V) := {
    tendsto_proper : ∀ (u u' : N → V) (l l' : V),
      (∀ n : N, u n == u' n) → l == l' → tendsto u l → tendsto u' l';
    tendsto_unique : ∀ (u : N → V) (l l' : V), tendsto u l → tendsto u l' → l == l';
    tendsto_const : ∀ c : V, tendsto (fun _ ⇒ c) c;
    tendsto_add : ∀ (u v : N → V) (l m : V),
      tendsto u l → tendsto v m → tendsto (fun n ⇒ u n + v n) (l + m);
    tendsto_neg : ∀ (u : N → V) (l : V), tendsto u l → tendsto (fun n ⇒ - u n) (- l);
    tendsto_smul : ∀ (a : F) (u : N → V) (l : V),
      tendsto u l → tendsto (fun n ⇒ a *: u n) (a *: l);
    tendsto_tail : ∀ (u : N → V) (l : V), tendsto u l → tendsto (fun n ⇒ u (succ n)) l;

    cauchy_proper : ∀ u u' : N → V, (∀ n : N, u n == u' n) → cauchy u → cauchy u';
    cauchy_of_tendsto : ∀ (u : N → V) (l : V), tendsto u l → cauchy u;
    cauchy_add : ∀ u v : N → V, cauchy u → cauchy v → cauchy (fun n ⇒ u n + v n);
    cauchy_neg : ∀ u : N → V, cauchy u → cauchy (fun n ⇒ - u n);
    cauchy_smul : ∀ (a : F) (u : N → V), cauchy u → cauchy (fun n ⇒ a *: u n);
    cauchy_tail : ∀ u : N → V, cauchy u → cauchy (fun n ⇒ u (succ n));
  }.
  Hint Mode Limits - ! ! - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Limits.
