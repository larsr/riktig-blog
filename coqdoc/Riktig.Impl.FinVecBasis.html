From Coq Require Import RelationClasses.
From Coq.Vectors Require Fin.
From Riktig Require Export Theory.BigOp.
From Riktig Require Import Impl.BigopFold.

Section StandardBasisIsBasis.
  Context {F : Type} `{FF : Field F}.

  Local Notation e := (@e F _ _ _).


  Lemma e_at_eq : ∀ (n : nat) (i j : Fin.t n), i = j → e i j == (1 : F).
  Proof.
    intros n i j Hij.
    exact (delta_at_eq i j Hij).
  Qed.

  Lemma e_at_neq : ∀ (n : nat) (i j : Fin.t n), i ≠ j → e i j == (0 : F).
  Proof.
    intros n i j Hij.
    exact (delta_at_neq i j Hij).
  Qed.

  Lemma e_sym : ∀ (n : nat) (i j : Fin.t n), (e i j : F) == e j i.
  Proof.
    intros n i j.
    destruct (Fin.eq_dec i j) as [Heq | Hne].
    - rewrite (e_at_eq n i j Heq), (e_at_eq n j i (eq_sym Heq)).
      reflexivity.
    - rewrite (e_at_neq n i j Hne),
        (e_at_neq n j i (fun Heq ⇒ Hne (eq_sym Heq))).
      reflexivity.
  Qed.

  Lemma e_FS : ∀ (n : nat) (i j : Fin.t n),
    e (Fin.FS i) (Fin.FS j) == e i j.
  Proof.
    intros n i j.
    destruct (Fin.eq_dec i j) as [Heq | Hne].
    - rewrite (e_at_eq (S n) (Fin.FS i) (Fin.FS j) (f_equal Fin.FS Heq)),
        (e_at_eq n i j Heq).
      reflexivity.
    - rewrite (e_at_neq (S n) (Fin.FS i) (Fin.FS j)
                 (fun Hc ⇒ Hne (Fin.FS_inj i j Hc))),
        (e_at_neq n i j Hne).
      reflexivity.
  Qed.

  Lemma mul_e_eq : ∀ (n : nat) (a : F) (i j : Fin.t n),
    i = j → a × e i j == a.
  Proof.
    intros n a i j Hij.
    exact (mul_delta_eq a i j Hij).
  Qed.

  Lemma mul_e_neq : ∀ (n : nat) (a : F) (i j : Fin.t n),
    i ≠ j → a × e i j == (0 : F).
  Proof.
    intros n a i j Hij.
    exact (mul_delta_neq a i j Hij).
  Qed.

  Lemma mul_e_FS : ∀ (n : nat) (a : F) (i j : Fin.t n),
    a × e (Fin.FS i) (Fin.FS j) == a × e i j.
  Proof.
    intros n a i j.
    rewrite (e_FS n i j).
    reflexivity.
  Qed.

  Lemma lincomb_std_delta : ∀ (n : nat) (c : Fin.t n → F)
      (j : Fin.t n),
    bigopF (fun i ⇒ c i × e i j) == c j.
  Proof.
    intros n c j.
    calc (bigopF (fun i ⇒ c i × e i j))
      == (bigopI (fun i ⇒ c i × e i j))
        { (bigopF_bigopI n (fun i ⇒ c i × e i j)) }
      == (c j)
        { (bigopI_delta c j (leibniz_proper c)) }.
  Qed.

  Lemma lincomb_std : ∀ (n : nat) (c : Fin.t n → F),
    lincomb e c == c.
  Proof.
    intros n c j.
    calc (lincomb e c j)
      == (bigopI (fun i ⇒ (c i *: e i) j)) { (bigopI_app (fun i ⇒ c i *: e i) j) }
      == (c j) { (bigopI_delta c j (leibniz_proper c)) }.
  Qed.

  Definition FinVec_Basis (n : nat) : Basis n (Fin.t n) F (Fin.t n → F) :=
    {| Basis.vec := e;
       Basis.vec_proper := leibniz_proper e;
       Basis.coords := linear_map_id;
       Basis.coords_family_proper := fun v ⇒ leibniz_proper v;
       Basis.reconstruction := fun v ⇒ lincomb_std n v;
       Basis.coords_linear_combination := fun c _ ⇒ lincomb_std n c |}.

End StandardBasisIsBasis.

Example finvec_basis_check (F : Type) `{FF : Field F} (n : nat) :
  Basis n (Fin.t n) F (Fin.t n → F) := FinVec_Basis n.
