From Coq Require Export Morphisms.
From Riktig Require Export Ops Interface.Ring Interface.Field
  Interface.VectorSpace.


Section FieldAsVectorSpace.
  Context {F : Type} `{Field.Field F}.

  #[global] Instance Field_Scale : HasScale F F := mult.

  #[global] Instance field_sc_proper : Proper (equ ==> equ ==> equ) (@scale F F _).
  Proof.
    intros a a' Ha u u' Hu.
    exact (Ring.mul_proper a a' Ha u u' Hu).
  Qed.

  #[global] Instance Field_ScaleLaws : ScaleLaws F F :=
    {| scale_proper := field_sc_proper;
       scale_distributes_over_vector_add := Ring.left_distributive;
       scale_distributes_over_scalar_add := Ring.right_distributive;
       scale_associative := Ring.mul_associative;
       scale_identity := Ring.mul_left_identity |}.

  #[global] Instance Field_VectorSpace : VectorSpace F F :=
    VectorSpace_from_parts F F _ Field_ScaleLaws.

End FieldAsVectorSpace.
