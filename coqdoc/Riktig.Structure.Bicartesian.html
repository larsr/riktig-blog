From Riktig.Structure Require Export Terminal Initial Products Coproducts.

A bicartesian category has a terminal and an initial object, binary products and binary coproducts.
Module Bicartesian.
  Class Bicartesian (O : Type) `{Category O}
      (terminal : HasTerm O) (to_terminal : HasToTerm O)
      (initial : HasInit O) (from_initial : HasFromInit O)
      (product : HasProd O) (first : HasPi1 O) (second : HasPi2 O) (lifting : HasLift O)
      (coproduct : HasCoprod O) (left : HasInj1 O) (right : HasInj2 O) (descent : HasDesc O) := {
    has_terminal :: Terminal O terminal to_terminal;
    has_initial :: Initial O initial from_initial;
    has_products :: Products O product first second lifting;
    has_coproducts :: Coproducts O coproduct left right descent;
  }.
End Bicartesian.
