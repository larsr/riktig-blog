Library Riktig.Impl.Complex

From Coq Require Import Setoid Ring.
From Riktig Require Export Theory.Field.

Record C (F : Type) := mkC { re : F; im : F }.

Arguments mkC {F} re im.
Arguments re {F} c.
Arguments im {F} c.

Section ComplexOps.
  Context {F : Type} `{Field F}.

  #[export] Instance C_EqRel : EqRel (C F) :=
    fun z w ⇒ re z == re w ∧ im z == im w.

  #[export] Instance C_Zero : Zero (C F) := mkC 0 0.

  #[export] Instance C_One : One (C F) := mkC 1 0.

  #[export] Instance C_Add : HasAdd (C F) :=
    fun z w ⇒ mkC (re z + re w) (im z + im w).

  #[export] Instance C_Neg : HasNeg (C F) :=
    fun z ⇒ mkC (- re z) (- im z).

  #[export] Instance C_Mult : HasMult (C F) :=
    fun z w ⇒ mkC (re z × re w + - (im z × im w))
                   (re z × im w + im z × re w).

  Definition norm2 (z : C F) : F := re z × re z + im z × im z.

  #[export] Instance C_Inv : HasInv (C F) :=
    fun z ⇒ mkC (re z × inv (norm2 z)) (- im z × inv (norm2 z)).

  #[export] Instance C_Equivalence : Equivalence C_EqRel.
  Proof.
    split.
    - intros z. split; reflexivity.
    - intros z w [Hr Hi]. split; symmetry; assumption.
    - intros z w v [Hr Hi] [Hr' Hi']. split; etransitivity; eassumption.
  Qed.

  Lemma norm2_proper : ∀ z w : C F, z == w → norm2 z == norm2 w.
  Proof.
    intros z w [Hr Hi]. unfold norm2. rewrite Hr, Hi. reflexivity.
  Qed.
End ComplexOps.

Section ComplexRing.
  Context {F : Type} `{Field F}.
  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  #[export] Instance C_add_proper : Proper (equ ==> equ ==> equ) (@add (C F) _).
  Proof.
    intros z z' [Hr Hi] w w' [Hr' Hi'].
    split; simpl; [ rewrite Hr, Hr' | rewrite Hi, Hi' ]; reflexivity.
  Qed.

  #[export] Instance C_neg_proper : Proper (equ ==> equ) (@neg (C F) _).
  Proof.
    intros z w [Hr Hi].
    split; simpl; [ rewrite Hr | rewrite Hi ]; reflexivity.
  Qed.

  #[export] Instance C_mul_proper : Proper (equ ==> equ ==> equ) (@mult (C F) _).
  Proof.
    intros z z' [Hr Hi] w w' [Hr' Hi'].
    split; simpl; rewrite Hr, Hi, Hr', Hi'; reflexivity.
  Qed.

  #[export] Instance C_inv_proper : Proper (equ ==> equ) (@inv (C F) _).
  Proof.
    intros z w Hzw.
    assert (Hn : norm2 z == norm2 w) by exact (norm2_proper z w Hzw).
    destruct Hzw as [Hr Hi].
    split; simpl; [ rewrite Hr, Hn | rewrite Hi, Hn ]; reflexivity.
  Qed.

  #[export] Instance C_add_associative : Associative (C F) (C F) add add.
  Proof. intros z w v. split; simpl; ring. Qed.

  Lemma C_nontrivial : (1 : C F) =/= 0.
  Proof. intros [Hr Hi]. exact (nontrivial Hr). Qed.

Section ComplexField.
  Context {F : Type} `{FormallyReal F}.
  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma norm2_nonzero : ∀ z : C F, z =/= 0 → norm2 z =/= 0.
  Proof.
    intros z Hz Hn. apply Hz. split.
    - exact (sum_of_squares (re z) (im z) Hn).
    - exact (fr_sum_of_squares_r (re z) (im z) Hn).
  Qed.

  #[export] Instance C_mul_left_inverse
    : LeftInverseOn (C F) (nonzero (C F)) mult inv 1.
  Proof.
    intros z Hz.
    assert (Hn : norm2 z =/= 0) by exact (norm2_nonzero z Hz).
    split; simpl.
    - calc ((re z × inv (norm2 z)) × re z + - ((- im z × inv (norm2 z)) × im z))
        == [expand] (inv (norm2 z) × norm2 z)
        == (1 : F) { (left_inverse_on (defined_on := nonzero F) (norm2 z) Hn) }.
      [expand]: unfold norm2; ring.
    - ring.
  Qed.

  #[export] Instance C_Field : Field (C F) := { Field.nontrivial := C_nontrivial }.
End ComplexField.
