Record C (F : Type) := mkC { re : F; im : F }.

Arguments mkC {F} re im.
Arguments re {F} c.
Arguments im {F} c.


Section ComplexOps.
  Context {F : Type} `{Field.Field F}.

  #[global] Instance C_EqRel : EqRel (C F) :=
    fun z w ⇒ re z == re w ∧ im z == im w.

  #[global] Instance C_Zero : Zero (C F) := mkC 0 0.

  #[global] Instance C_One : One (C F) := mkC 1 0.

  #[global] Instance C_Add : HasAdd (C F) :=
    fun z w ⇒ mkC (re z + re w) (im z + im w).

  #[global] Instance C_Neg : HasNeg (C F) :=
    fun z ⇒ mkC (- re z) (- im z).

  #[global] Instance C_Mult : HasMult (C F) :=
    fun z w ⇒ mkC (re z × re w + - (im z × im w))
                   (re z × im w + im z × re w).

  Definition norm2 (z : C F) : F := re z × re z + im z × im z.

  #[global] Instance C_Inv : HasInv (C F) :=
    fun z ⇒ mkC (re z × inv (norm2 z)) (- im z × inv (norm2 z)).

  Lemma norm2_eq : ∀ z : C F, norm2 z = re z × re z + im z × im z.
  Proof.
    intros z.
    reflexivity.
  Qed.

  Lemma norm2_proper : ∀ z w : C F, z == w → norm2 z == norm2 w.
  Proof.
    intros z w [Hr Hi].
    rewrite norm2_eq, norm2_eq, Hr, Hi.
    reflexivity.
  Qed.

  #[global] Instance C_Equivalence : Equivalence C_EqRel.
  Proof.
    apply Build_Equivalence.
    - intros z. split.
      + reflexivity.
      + reflexivity.
    - intros z w [Hr Hi]. split.
      + symmetry. exact Hr.
      + symmetry. exact Hi.
    - intros z w v [Hr1 Hi1] [Hr2 Hi2]. split.
      + rewrite Hr1. exact Hr2.
      + rewrite Hi1. exact Hi2.
  Qed.

End ComplexOps.


Section ComplexRing.
  Context {F : Type} `{Field.Field F}.


  Lemma c_add_assoc : ∀ z w v : C F, (z + w) + v == z + (w + v).
  Proof.
    intros z w v. split.
    - exact (Monoid.associative (re z) (re w) (re v)).
    - exact (Monoid.associative (im z) (im w) (im v)).
  Qed.

  Lemma c_add_idl : ∀ z : C F, 0 + z == z.
  Proof.
    intros z. split.
    - exact (Monoid.left_identity (re z)).
    - exact (Monoid.left_identity (im z)).
  Qed.

  Lemma c_add_idr : ∀ z : C F, z + 0 == z.
  Proof.
    intros z. split.
    - exact (Monoid.right_identity (re z)).
    - exact (Monoid.right_identity (im z)).
  Qed.

  #[global] Instance c_add_proper : Proper (equ ==> equ ==> equ) (@add (C F) _).
  Proof.
    intros z z' [Hr Hi] w w' [Hr' Hi']. split.
    - exact (Monoid.add_proper (re z) (re z') Hr (re w) (re w') Hr').
    - exact (Monoid.add_proper (im z) (im z') Hi (im w) (im w') Hi').
  Qed.

  Lemma c_add_invl : ∀ z : C F, (- z) + z == 0.
  Proof.
    intros z. split.
    - exact (Group.left_inverse (re z)).
    - exact (Group.left_inverse (im z)).
  Qed.

  Lemma c_add_invr : ∀ z : C F, z + (- z) == 0.
  Proof.
    intros z. split.
    - exact (Group.right_inverse (re z)).
    - exact (Group.right_inverse (im z)).
  Qed.

  #[global] Instance c_neg_proper : Proper (equ ==> equ) (@neg (C F) _).
  Proof.
    intros z w [Hr Hi]. split.
    - exact (Group.neg_proper (re z) (re w) Hr).
    - exact (Group.neg_proper (im z) (im w) Hi).
  Qed.

  Lemma c_add_comm : ∀ z w : C F, z + w == w + z.
  Proof.
    intros z w. split.
    - exact (add_comm (re z) (re w)).
    - exact (add_comm (im z) (im w)).
  Qed.

  #[global] Instance C_Group : Group (C F) := {|
    Group.equivalence := C_Equivalence;
    Group.associative := c_add_assoc;
    Group.left_identity := c_add_idl;
    Group.right_identity := c_add_idr;
    Group.left_inverse := c_add_invl;
    Group.right_inverse := c_add_invr;
    Group.add_proper := c_add_proper;
    Group.neg_proper := c_neg_proper
  |}.

  #[global] Instance C_AbGroup : AbGroup (C F) := {|

    AbGroup.left_identity := c_add_idl;
    AbGroup.right_identity := c_add_idr;
    AbGroup.left_inverse := c_add_invl;
    AbGroup.right_inverse := c_add_invr;
    AbGroup.commutative := c_add_comm;
    AbGroup.add_proper := c_add_proper;
    AbGroup.neg_proper := c_neg_proper
  |}.


  Lemma cmul_1l_re : ∀ a b : F, 1 × a + - (0 × b) == a.
  Proof.
    intros a b.
    calc (1 × a + - (0 × b))
      == (a + - (0 × b)) { idl }
      == (a + - 0) { mul_0_l }
      == (a + 0) { neg_zero }
      == a { idr }.
  Qed.

  Lemma cmul_1l_im : ∀ a b : F, 1 × b + 0 × a == b.
  Proof.
    intros a b.
    calc (1 × b + 0 × a)
      == (b + 0 × a) { idl }
      == (b + 0) { mul_0_l }
      == b { idr }.
  Qed.

  Lemma cmul_1r_re : ∀ a b : F, a × 1 + - (b × 0) == a.
  Proof.
    intros a b.
    calc (a × 1 + - (b × 0))
      == (a + - (b × 0)) { idr }
      == (a + - 0) { mul_0_r }
      == (a + 0) { neg_zero }
      == a { idr }.
  Qed.

  Lemma cmul_1r_im : ∀ a b : F, a × 0 + b × 1 == b.
  Proof.
    intros a b.
    calc (a × 0 + b × 1)
      == (0 + b × 1) { mul_0_r }
      == (0 + b) { idr }
      == b { idl }.
  Qed.

  Lemma cmul_assoc_re : ∀ a b c d e f : F,
    (a × c + - (b × d)) × e + - ((a × d + b × c) × f)
    == a × (c × e + - (d × f)) + - (b × (c × f + d × e)).
  Proof.
    intros a b c d e f.
    rewrite distr, distr, distl, distl.
    rewrite mul_neg_l, mul_neg_r, neg_add, neg_add.
    rewrite Ring.mul_associative, Ring.mul_associative, Ring.mul_associative, Ring.mul_associative.
    rewrite add_interchange, (comm (- (b × (d × e))) (- (b × (c × f)))).
    reflexivity.
  Qed.

  Lemma cmul_assoc_im : ∀ a b c d e f : F,
    (a × c + - (b × d)) × f + (a × d + b × c) × e
    == a × (c × f + d × e) + b × (c × e + - (d × f)).
  Proof.
    intros a b c d e f.
    rewrite distr, distr, distl, distl.
    rewrite mul_neg_l, mul_neg_r.
    rewrite Ring.mul_associative, Ring.mul_associative, Ring.mul_associative, Ring.mul_associative.
    rewrite add_interchange, (comm (- (b × (d × f))) (b × (c × e))).
    reflexivity.
  Qed.

  Lemma cmul_distl_re : ∀ a b c d e f : F,
    a × (c + e) + - (b × (d + f)) == (a × c + - (b × d)) + (a × e + - (b × f)).
  Proof.
    intros a b c d e f.
    rewrite distl, distl, neg_add, add_interchange.
    reflexivity.
  Qed.

  Lemma cmul_distl_im : ∀ a b c d e f : F,
    a × (d + f) + b × (c + e) == (a × d + b × c) + (a × f + b × e).
  Proof.
    intros a b c d e f.
    rewrite distl, distl, add_interchange.
    reflexivity.
  Qed.

  Lemma cmul_distr_re : ∀ a b c d e f : F,
    (a + c) × e + - ((b + d) × f) == (a × e + - (b × f)) + (c × e + - (d × f)).
  Proof.
    intros a b c d e f.
    rewrite distr, distr, neg_add, add_interchange.
    reflexivity.
  Qed.

  Lemma cmul_distr_im : ∀ a b c d e f : F,
    (a + c) × f + (b + d) × e == (a × f + b × e) + (c × f + d × e).
  Proof.
    intros a b c d e f.
    rewrite distr, distr, add_interchange.
    reflexivity.
  Qed.

  Lemma cmul_re_proper : ∀ a a' b b' c c' d d' : F,
    a == a' → b == b' → c == c' → d == d' →
    a × c + - (b × d) == a' × c' + - (b' × d').
  Proof.
    intros a a' b b' c c' d d' Ha Hb Hc Hd.
    rewrite Ha, Hb, Hc, Hd.
    reflexivity.
  Qed.

  Lemma cmul_im_proper : ∀ a a' b b' c c' d d' : F,
    a == a' → b == b' → c == c' → d == d' →
    a × d + b × c == a' × d' + b' × c'.
  Proof.
    intros a a' b b' c c' d d' Ha Hb Hc Hd.
    rewrite Ha, Hb, Hc, Hd.
    reflexivity.
  Qed.


  Lemma c_mul_assoc : ∀ z w v : C F, (z × w) × v == z × (w × v).
  Proof.
    intros z w v. split.
    - exact (cmul_assoc_re (re z) (im z) (re w) (im w) (re v) (im v)).
    - exact (cmul_assoc_im (re z) (im z) (re w) (im w) (re v) (im v)).
  Qed.

  Lemma c_mul_1l : ∀ z : C F, 1 × z == z.
  Proof.
    intros z. split.
    - exact (cmul_1l_re (re z) (im z)).
    - exact (cmul_1l_im (re z) (im z)).
  Qed.

  Lemma c_mul_1r : ∀ z : C F, z × 1 == z.
  Proof.
    intros z. split.
    - exact (cmul_1r_re (re z) (im z)).
    - exact (cmul_1r_im (re z) (im z)).
  Qed.

  Lemma c_distl : ∀ z w v : C F, z × (w + v) == z × w + z × v.
  Proof.
    intros z w v. split.
    - exact (cmul_distl_re (re z) (im z) (re w) (im w) (re v) (im v)).
    - exact (cmul_distl_im (re z) (im z) (re w) (im w) (re v) (im v)).
  Qed.

  Lemma c_distr : ∀ z w v : C F, (z + w) × v == z × v + w × v.
  Proof.
    intros z w v. split.
    - exact (cmul_distr_re (re z) (im z) (re w) (im w) (re v) (im v)).
    - exact (cmul_distr_im (re z) (im z) (re w) (im w) (re v) (im v)).
  Qed.

  #[global] Instance c_mul_proper : Proper (equ ==> equ ==> equ) (@mult (C F) _).
  Proof.
    intros z z' [Hr Hi] w w' [Hr' Hi']. split.
    - exact (cmul_re_proper (re z) (re z') (im z) (im z')
               (re w) (re w') (im w) (im w') Hr Hi Hr' Hi').
    - exact (cmul_im_proper (re z) (re z') (im z) (im z')
               (re w) (re w') (im w) (im w') Hr Hi Hr' Hi').
  Qed.

  #[global] Instance C_Ring : Ring (C F) := {|
    Ring.equivalence := C_Equivalence;
    Ring.associative := c_add_assoc;
    Ring.left_identity := c_add_idl;
    Ring.right_identity := c_add_idr;
    Ring.left_inverse := c_add_invl;
    Ring.right_inverse := c_add_invr;
    Ring.commutative := c_add_comm;
    Ring.mul_associative := c_mul_assoc;
    Ring.mul_left_identity := c_mul_1l;
    Ring.mul_right_identity := c_mul_1r;
    Ring.left_distributive := c_distl;
    Ring.right_distributive := c_distr;
    Ring.add_proper := c_add_proper;
    Ring.neg_proper := c_neg_proper;
    Ring.mul_proper := c_mul_proper
  |}.

  Example c_mul_0_l : ∀ z : C F, 0 × z == 0.
  Proof.
    intros z.
    exact (mul_0_l z).
  Qed.


  Lemma cmul_comm_re : ∀ a b c d : F,
    a × c + - (b × d) == c × a + - (d × b).
  Proof.
    intros a b c d.
    calc (a × c + - (b × d))
      == (c × a + - (b × d)) { (mul_comm a c) }
      == (c × a + - (d × b)) { (mul_comm b d) }.
  Qed.

  Lemma cmul_comm_im : ∀ a b c d : F,
    a × d + b × c == c × b + d × a.
  Proof.
    intros a b c d.
    calc (a × d + b × c)
      == (d × a + b × c) { (mul_comm a d) }
      == (d × a + c × b) { (mul_comm b c) }
      == (c × b + d × a) { comm }.
  Qed.

  Lemma c_mul_comm : ∀ z w : C F, z × w == w × z.
  Proof.
    intros z w. split.
    - exact (cmul_comm_re (re z) (im z) (re w) (im w)).
    - exact (cmul_comm_im (re z) (im z) (re w) (im w)).
  Qed.

  Lemma mul_inv_shuffle : ∀ x n : F, (x × inv n) × x == inv n × (x × x).
  Proof.
    intros x n.
    calc ((x × inv n) × x)
      == ((inv n × x) × x) { (mul_comm x (inv n)) }
      == (inv n × (x × x)) { assoc }.
  Qed.

  Lemma mul_inv_swap : ∀ x y n : F, (x × inv n) × y == (y × inv n) × x.
  Proof.
    intros x y n.
    calc ((x × inv n) × y)
      == ((inv n × x) × y) { (mul_comm x (inv n)) }
      == (inv n × (x × y)) { assoc }
      == (inv n × (y × x)) { (mul_comm x y) }
      == ((inv n × y) × x) { assoc }
      == ((y × inv n) × x) { (mul_comm y (inv n)) }.
  Qed.

  Lemma mul_inv_proper2 : ∀ x y n m : F,
    x == y → n == m → x × inv n == y × inv m.
  Proof.
    intros x y n m Hxy Hnm.
    rewrite Hxy, Hnm.
    reflexivity.
  Qed.


  Lemma cinv_re : ∀ a b n : F,
    n =/= 0 → a × a + b × b == n →
    (a × inv n) × a + - ((- b × inv n) × b) == 1.
  Proof.
    intros a b n Hn Heq.
    rewrite (mul_neg_l b (inv n)), (mul_neg_l (b × inv n) b), neg_involutive.
    rewrite (mul_inv_shuffle a n), (mul_inv_shuffle b n), <- distl, Heq.
    exact (Field.mul_left_inverse n Hn).
  Qed.

  Lemma cinv_im : ∀ a b n : F,
    (a × inv n) × b + (- b × inv n) × a == 0.
  Proof.
    intros a b n.
    rewrite (mul_neg_l b (inv n)), (mul_neg_l (b × inv n) a).
    rewrite (mul_inv_swap a b n).
    apply invr.
  Qed.

  #[global] Instance c_inv_proper : Proper (equ ==> equ) (@inv (C F) _).
  Proof.
    intros z w Hzw.
    destruct Hzw as [Hr Hi]. split.
    - exact (mul_inv_proper2 (re z) (re w) (norm2 z) (norm2 w)
               Hr (norm2_proper z w (conj Hr Hi))).
    - exact (mul_inv_proper2 (- im z) (- im w) (norm2 z) (norm2 w)
               (Group.neg_proper (im z) (im w) Hi)
               (norm2_proper z w (conj Hr Hi))).
  Qed.

  Lemma c_nontrivial : (1 : C F) =/= 0.
  Proof.
    intros [Hr Hi].
    exact (Field.nontrivial Hr).
  Qed.

End ComplexRing.


Section ComplexField.
  Context {F : Type} `{FormallyReal.FormallyReal F}.

  Lemma norm2_nonzero : ∀ z : C F, z =/= 0 → norm2 z =/= 0.
  Proof.
    intros z Hz Hn.
    apply Hz. split.
    - exact (FormallyReal.sum_of_squares (re z) (im z) Hn).
    - exact (fr_sum_of_squares_r (re z) (im z) Hn).
  Qed.

  Lemma c_invl : ∀ z : C F, z =/= 0 → inv z × z == 1.
  Proof.
    intros z Hz. split.
    - apply (cinv_re (re z) (im z) (norm2 z)).
      + exact (norm2_nonzero z Hz).
      + reflexivity.
    - exact (cinv_im (re z) (im z) (norm2 z)).
  Qed.
