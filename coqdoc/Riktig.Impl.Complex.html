Record C (F : Type) := mkC { re : F; im : F }.

Arguments mkC {F} re im.
Arguments re {F} c.
Arguments im {F} c.

Section ComplexOps.
  Context {F : Type} `{Field F}.

  #[export] Instance C_EqRel : EqRel (C F) :=
    fun z w ⇒ re z == re w ∧ im z == im w.

  #[export] Instance C_Zero : Zero (C F) := mkC 0 0.

  #[export] Instance C_One : One (C F) := mkC 1 0.

  #[export] Instance C_Add : HasAdd (C F) :=
    fun z w ⇒ mkC (re z + re w) (im z + im w).

  #[export] Instance C_Neg : HasNeg (C F) :=
    fun z ⇒ mkC (- re z) (- im z).

  #[export] Instance C_Mult : HasMult (C F) :=
    fun z w ⇒ mkC (re z × re w + - (im z × im w))
                   (re z × im w + im z × re w).

  Definition norm2 (z : C F) : F := re z × re z + im z × im z.

  #[export] Instance C_Inv : HasInv (C F) :=
    fun z ⇒ mkC (re z × inv (norm2 z)) (- im z × inv (norm2 z)).

  #[export] Instance C_Equivalence : Equivalence C_EqRel.
  Proof.
    split.
    - intros z. split; reflexivity.
    - intros z w [Hr Hi]. split; symmetry; assumption.
    - intros z w v [Hr Hi] [Hr' Hi']. split; etransitivity; eassumption.
  Qed.

  Lemma norm2_proper : ∀ z w : C F, z == w → norm2 z == norm2 w.
  Proof.
    intros z w [Hr Hi]. unfold norm2. rewrite Hr, Hi. reflexivity.
  Qed.
End ComplexOps.

Section ComplexRing.
  Context {F : Type} `{Field F}.
  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  #[export] Instance C_add_proper : Proper (equ ==> equ ==> equ) (@add (C F) _).
  Proof.
    intros z z' [Hr Hi] w w' [Hr' Hi'].
    split; simpl; [ rewrite Hr, Hr' | rewrite Hi, Hi' ]; reflexivity.
  Qed.

  #[export] Instance C_neg_proper : Proper (equ ==> equ) (@neg (C F) _).
  Proof.
    intros z w [Hr Hi].
    split; simpl; [ rewrite Hr | rewrite Hi ]; reflexivity.
  Qed.

  #[export] Instance C_mul_proper : Proper (equ ==> equ ==> equ) (@mult (C F) _).
  Proof.
    intros z z' [Hr Hi] w w' [Hr' Hi'].
    split; simpl; rewrite Hr, Hi, Hr', Hi'; reflexivity.
  Qed.

  #[export] Instance C_inv_proper : Proper (equ ==> equ) (@inv (C F) _).
  Proof.
    intros z w Hzw.
    assert (Hn : norm2 z == norm2 w) by exact (norm2_proper z w Hzw).
    destruct Hzw as [Hr Hi].
    split; simpl; [ rewrite Hr, Hn | rewrite Hi, Hn ]; reflexivity.
  Qed.

  #[export] Instance C_add_associative : Associative (C F) (C F) add add.
  Proof. intros z w v. split; simpl; ring. Qed.

  #[export] Instance C_add_left_identity : LeftIdentity (C F) (C F) add 0.
  Proof. intros z. split; simpl; ring. Qed.

  #[export] Instance C_add_right_identity : RightIdentity (C F) (C F) add 0.
  Proof. intros z. split; simpl; ring. Qed.

  #[export] Instance C_add_left_inverse : LeftInverse (C F) add neg 0.
  Proof. intros z. split; simpl; ring. Qed.

  #[export] Instance C_add_right_inverse : RightInverse (C F) add neg 0.
  Proof. intros z. split; simpl; ring. Qed.

  #[export] Instance C_add_commutative : Commutative (C F) (C F) add.
  Proof. intros z w. split; simpl; ring. Qed.

  Proof. intros z. split; simpl; ring. Qed.

  Proof. intros z. split; simpl; ring. Qed.

Section ComplexAlgebra.
  Context {F : Type} `{Field F}.
  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  #[export] Instance C_Scale : HasScale F (C F) :=
    fun a z ⇒ mkC (a × re z) (a × im z).

  #[export] Instance C_scale_proper :
    Proper (equ ==> equ ==> equ) (@scale F (C F) _).
  Proof.
    intros a a' Ha z z' [Hr Hi].
    split; simpl; [ rewrite Ha, Hr | rewrite Ha, Hi ]; reflexivity.
  Qed.

  Lemma C_scale_add : ∀ (a : F) (z w : C F),
    a *: (z + w) == a *: z + a *: w.
  Proof. intros a z w. split; simpl; ring. Qed.

  Lemma C_add_scale : ∀ (a b : F) (z : C F),
    (a + b) *: z == a *: z + b *: z.
  Proof. intros a b z. split; simpl; ring. Qed.

  Lemma C_scale_scale : ∀ (a b : F) (z : C F),
    (a × b) *: z == a *: (b *: z).
  Proof. intros a b z. split; simpl; ring. Qed.

  Lemma C_scale_one : ∀ z : C F, 1 *: z == z.
  Proof. intros z. split; simpl; ring. Qed.

  #[export] Instance C_ScaleLaws : ScaleLaws F (C F) :=
    {| ScaleLaws.scale_proper := C_scale_proper;
       ScaleLaws.is_left_distributive := C_scale_add;
       ScaleLaws.is_right_distributive := C_add_scale;
       ScaleLaws.is_associative := C_scale_scale;
       ScaleLaws.is_left_identity := C_scale_one |}.

Scalars are central, which is what makes C F an algebra over F and not merely a ring carrying an action.
  Lemma C_scale_mul_l : ∀ (a : F) (z w : C F),
    (a *: z) × w == a *: (z × w).
  Proof. intros a z w. split; simpl; ring. Qed.

  Lemma C_scale_mul_r : ∀ (a : F) (z w : C F),
    z × (a *: w) == a *: (z × w).
  Proof. intros a z w. split; simpl; ring. Qed.

  #[export] Instance C_mul_linear_l (w : C F) :
    Linear F (C F) (C F) (fun z ⇒ z × w) :=
    {| Linear.proper := fun z z' Hz ⇒ C_mul_proper z z' Hz w w (reflexivity w);
       Linear.is_additive := fun z z' ⇒ right_distributive z z' w;
       Linear.is_homogeneous := fun a z ⇒ C_scale_mul_l a z w |}.

  #[export] Instance C_mul_linear_r (z : C F) :
    Linear F (C F) (C F) (fun w ⇒ z × w) :=
    {| Linear.proper := fun w w' Hw ⇒ C_mul_proper z z (reflexivity z) w w' Hw;
       Linear.is_additive := fun w w' ⇒ left_distributive z w w';
       Linear.is_homogeneous := fun a w ⇒ C_scale_mul_r a z w |}.

Conjugation.
  Definition conjugate (z : C F) : C F := mkC (re z) (- im z).

  #[export] Instance C_conjugate_proper : Proper (equ ==> equ) conjugate.
  Proof.
    intros z w [Hr Hi].
    split; simpl; [ exact Hr | rewrite Hi; reflexivity ].
  Qed.

  Lemma C_conjugate_add : ∀ z w : C F,
    conjugate (z + w) == conjugate z + conjugate w.
  Proof. intros z w. split; simpl; ring. Qed.

  Lemma C_conjugate_mul : ∀ z w : C F,
    conjugate (z × w) == conjugate w × conjugate z.
  Proof. intros z w. split; simpl; ring. Qed.

  Lemma C_conjugate_involutive : ∀ z : C F, conjugate (conjugate z) == z.
  Proof. intros z. split; simpl; ring. Qed.

  #[export] Instance C_Reversal : Reversal (C F) conjugate :=
    {| reversal_proper := C_conjugate_proper;
       reversal_additive := C_conjugate_add;
       reversal_anti := C_conjugate_mul;
       reversal_involutive := C_conjugate_involutive |}.

  Lemma C_conjugate_scale : ∀ (a : F) (z : C F),
    conjugate (a *: z) == a *: conjugate z.
  Proof. intros a z. split; simpl; ring. Qed.

The imaginary unit.
  Definition circle_i : C F := mkC 0 1.

  Lemma circle_i_square : circle_i × circle_i == - (1).
  Proof. split; simpl; ring. Qed.

  Lemma conjugate_circle_i : conjugate circle_i == - circle_i.
  Proof. split; simpl; ring. Qed.

A complex number IS its pair of coordinates on the plane the unit spans with i.
  Lemma scale_one_scale_i : ∀ a b : F,
    a *: (1 : C F) + b *: circle_i == mkC a b.
  Proof. intros a b. split; simpl; ring. Qed.

  Lemma C_spans : ∀ z : C F, ∃ a b : F, z == a *: 1 + b *: circle_i.
  Proof.
    intros z. ∃ (re z), (im z). symmetry.
    exact (scale_one_scale_i (re z) (im z)).
  Qed.

  Lemma circle_i_independent : independent_from_one F circle_i.
  Proof.
    intros a b Hab.
    rewrite (scale_one_scale_i a b) in Hab.
    destruct Hab as [Hre Him].
    split; [ exact Hre | exact Him ].
  Qed.

  #[export] Instance C_ComplexPlane : ComplexPlane F (C F) circle_i :=
    {| ComplexPlane.plane_root := circle_i_square;
       ComplexPlane.plane_spans := C_spans;
       ComplexPlane.plane_independent := circle_i_independent |}.

  Example c_conjugate_norm : ∀ z : C F, conjugate z × z == norm2 z *: 1.
  Proof. intros z. split; simpl; unfold norm2; ring. Qed.

End ComplexAlgebra.

Section ComplexField.
  Context {F : Type} `{MinusOneNotASquare F}.
  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma norm2_nonzero : ∀ z : C F, z =/= 0 → norm2 z =/= 0.
  Proof.
    intros z Hz Hn. apply Hz. split.
    - exact (two_squares_vanish (re z) (im z) Hn).
    - exact (two_squares_vanish_r (re z) (im z) Hn).
  Qed.

  #[export] Instance C_mul_left_inverse
    : LeftInverseOn (C F) (nonzero (C F)) mult inv 1.
  Proof.
    intros z Hz.
    assert (Hn : norm2 z =/= 0) by exact (norm2_nonzero z Hz).
    split; simpl.
    - calc ((re z × inv (norm2 z)) × re z + - ((- im z × inv (norm2 z)) × im z))
        == [expand] (inv (norm2 z) × norm2 z)
        == (1 : F) { (left_inverse_on (defined_on := nonzero F) (norm2 z) Hn) }.
      [expand]: unfold norm2; ring.
    - ring.
  Qed.

  #[export] Instance C_Field : Field (C F) := { Field.nontrivial := C_nontrivial }.
End ComplexField.
