From Riktig Require Export Structure.Category Lib.Calc.

Section CategoryTheory.
  Context {O : Type} `{Category O}.

  Theorem comp_cancel : ∀ (x y z : O) (a : y ~> z) (b : x ~> y) (b' : y ~> x) (a' : z ~> y),
    b ∘ b' == cat_id → a ∘ a' == cat_id → (a ∘ b) ∘ (b' ∘ a') == cat_id.
  Proof.
    intros x y z a b b' a' Hb Ha.
    calc ((a ∘ b) ∘ (b' ∘ a'))
      == (a ∘ (b ∘ (b' ∘ a'))) { comp_assoc }
      == (a ∘ ((b ∘ b') ∘ a')) { comp_assoc }
      == (a ∘ (cat_id ∘ a')) { Hb }
      == (a ∘ a') { left_identity }
      == (cat_id (x := z)) { Ha }.
  Qed.

  Theorem id_unique : ∀ (x : O) (e : x ~> x),
    (∀ (y : O) (f : x ~> y), f ∘ e == f) → e == cat_id.
  Proof.
    intros x e He.
    calc e
      == (cat_id ∘ e) { left_identity }
      == (cat_id (x := x)) { (He x cat_id) }.
  Qed.
End CategoryTheory.
