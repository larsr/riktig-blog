Module TwoSidedInverse.
  Class TwoSidedInverse {O : Type} `{Category O} {x y : O}
      (f : x ~> y) (g : y ~> x) := {
    from_to : g ∘ f == cat_id;
    to_from : f ∘ g == cat_id;
  }.
End TwoSidedInverse.
Notation TwoSidedInverse := TwoSidedInverse.TwoSidedInverse.

Hint Mode TwoSidedInverse.TwoSidedInverse - - - - ! ! : typeclass_instances.

Section CategoryTheory.
  Context {O : Type} `{Category O}.

  Record Iso (x y : O) := {
    iso_to : x ~> y;
    iso_from : y ~> x;
    iso_inverse :: TwoSidedInverse iso_to iso_from;
  }.

  Global Arguments iso_to {x y}.
  Global Arguments iso_from {x y}.
  Global Arguments iso_inverse {x y}.

  Definition iso_fromto {x y : O} (i : Iso x y) : iso_from i ∘ iso_to i == cat_id :=
    TwoSidedInverse.from_to (f := iso_to i) (g := iso_from i).

  Definition iso_tofrom {x y : O} (i : Iso x y) : iso_to i ∘ iso_from i == cat_id :=
    TwoSidedInverse.to_from (f := iso_to i) (g := iso_from i).

  Global Arguments iso_fromto {x y}.
  Global Arguments iso_tofrom {x y}.

  Definition iso_from_laws {x y : O} (to : x ~> y) (from : y ~> x)
      (fromto : from ∘ to == cat_id) (tofrom : to ∘ from == cat_id) : Iso x y :=
    {| iso_to := to;
       iso_from := from;
       iso_inverse :=
         {| TwoSidedInverse.from_to := fromto;
            TwoSidedInverse.to_from := tofrom |} |}.

  Definition iso_refl (x : O) : Iso x x :=
    iso_from_laws cat_id cat_id
      (Category.comp_left_identity x x cat_id)
      (Category.comp_left_identity x x cat_id).

  Definition iso_sym {x y : O} (f : Iso x y) : Iso y x :=
    iso_from_laws (iso_from f) (iso_to f) (iso_tofrom f) (iso_fromto f).

  Lemma comp_cancel :
    ∀ (x y z : O) (a : y ~> z) (b : x ~> y) (b' : y ~> x) (a' : z ~> y),
      b ∘ b' == cat_id → a ∘ a' == cat_id → (a ∘ b) ∘ (b' ∘ a') == cat_id.
  Proof.
    intros x y z a b b' a' Hb Ha.
    calc ((a ∘ b) ∘ (b' ∘ a'))
      == (a ∘ (b ∘ (b' ∘ a'))) { Category.comp_associative }
      == (a ∘ ((b ∘ b') ∘ a')) { (Category.comp_associative _ _ _ _ b b' a') }
      == (a ∘ (cat_id ∘ a')) { Hb }
      == (a ∘ a') { Category.comp_left_identity }
      == (cat_id : z ~> z) { Ha }.
  Qed.

  Definition iso_trans {x y z : O} (f : Iso x y) (g : Iso y z) : Iso x z :=
    iso_from_laws (iso_to g ∘ iso_to f) (iso_from f ∘ iso_from g)
      (comp_cancel _ _ _ _ _ _ _ (iso_fromto g) (iso_fromto f))
      (comp_cancel _ _ _ _ _ _ _ (iso_tofrom f) (iso_tofrom g)).

  Definition Isomorphic (x y : O) : Prop := inhabited (Iso x y).

  #[export] Instance Isomorphic_Equivalence : Equivalence Isomorphic.
  Proof.
    split.
    - intros x; exact (inhabits (iso_refl x)).
    - intros x y [f]; exact (inhabits (iso_sym f)).
    - intros x y z [f] [g]; exact (inhabits (iso_trans f g)).
  Qed.

  Theorem id_unique_l :
    ∀ (x : O) (e : x ~> x), (∀ y (f : x ~> y), f ∘ e == f) → e == cat_id.
  Proof.
    intros x e Hu.
    rewrite <- (Category.comp_left_identity x x e).
    exact (Hu x cat_id).
  Qed.

End CategoryTheory.
