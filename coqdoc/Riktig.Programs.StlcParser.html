Reading simply typed lambda terms, over structures alone: any characters a Lexicon describes, any parser structure over them, any list algebra for the names and the binders in scope, any types with a base and an arrow, any terms with the three typed constructors. Syntax: \x : A. body abstracts one name at its type, juxtaposition applies, parentheses group, the arrow associates to the right, and a name in a term is the innermost binder of that name. The fuel bounds every loop.

The type grammar: a name is a base type, the arrow associates to the right, parentheses group.
  Definition arrow_tok (fuel : N) : P unit :=
    token fuel (csym dash) >>= (fun _ ⇒ token fuel (csym gt) >>= (fun _ ⇒ ret tt)).

  Definition ty_step (fuel : N) (rec : P Ty × P Ty) : P Ty :=
    snd rec >>= (fun a ⇒
      (arrow_tok fuel >>= (fun _ ⇒ fst rec >>= (fun c ⇒ ret (a --> c))))
      <|> ret a).

  Definition ty_atom_step (fuel : N) (rec : P Ty × P Ty) : P Ty :=
    (ident fuel >>= (fun k ⇒ ret (ty_base k)))
    <|> (token fuel (csym lparen) >>= (fun _ ⇒
         fst rec >>= (fun t ⇒
         token fuel (csym rparen) >>= (fun _ ⇒ ret t)))).

  Definition ty_p (fuel : N) : P Ty :=
    fst (natrec (P Ty × P Ty)%type
           (fail, fail)
           (fun _ rec ⇒ (ty_step fuel rec, ty_atom_step fuel rec)) fuel).

The term grammar, carrying the binders in scope.
  Definition sterm_step (fuel : N) (rec : (E → P T) × (E → P T)) : E → P T :=
    fun env ⇒
      (token fuel (csym backslash) >>= (fun _ ⇒
       ident fuel >>= (fun k ⇒
       token fuel (csym colon) >>= (fun _ ⇒
       ty_p fuel >>= (fun A ⇒
       token fuel (csym dot_c) >>= (fun _ ⇒
       fst rec (prepend k env) >>= (fun b ⇒
       ret (tylam A b))))))))
      <|>
      (snd rec env >>= (fun a0 ⇒
       p_foldl fuel (snd rec env) a0 (fun f x ⇒ app f x))).

  Definition satom_step (fuel : N) (rec : (E → P T) × (E → P T)) : E → P T :=
    fun env ⇒
      (ident fuel >>= (fun x ⇒
         match debruijn x env with
         | Some n ⇒ ret (var n : T)
         | None ⇒ fail
         end))
      <|>
      (token fuel (csym lparen) >>= (fun _ ⇒
       fst rec env >>= (fun t ⇒
       token fuel (csym rparen) >>= (fun _ ⇒ ret t)))).

  Definition stlc_p (fuel : N) : E → P T :=
    fst (natrec ((E → P T) × (E → P T))%type
           ((fun _ ⇒ fail), (fun _ ⇒ fail))
           (fun _ rec ⇒ (sterm_step fuel rec, satom_step fuel rec)) fuel).
