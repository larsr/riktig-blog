Section ConstructiveRealsTheory.
  Context {R Q N : Type} `{ConstructiveReals R Q N}.

  Theorem lt_irrefl : ∀ x : R, x ≺ x → False.
  Proof. intros x Hx. exact (lt_asym x x Hx Hx). Qed.

  Theorem le_refl : ∀ x : R, x ≼ x.
  Proof. intros x. exact (lt_irrefl x). Qed.

  Theorem le_of_lt : ∀ x y : R, x ≺ y → x ≼ y.
  Proof. intros x y Hxy Hyx. exact (lt_asym x y Hxy Hyx). Qed.

  Theorem le_trans : ∀ x y z : R, x ≼ y → y ≼ z → x ≼ z.
  Proof.
    intros x y z Hxy Hyz Hzx.
    destruct (lt_cotrans z y x Hzx) as [Hzy | Hyx]; [ exact (Hyz Hzy) | exact (Hxy Hyx) ].
  Qed.

  Theorem le_of_equ : ∀ x y : R, x == y → x ≼ y.
  Proof. intros x y Hxy Hyx. exact (lt_irrefl x (lt_equ_l y x x (symmetry Hxy) Hyx)). Qed.

  Theorem apart_symm : ∀ x y : R, ((x ≺ y) + (y ≺ x))%type → ((y ≺ x) + (x ≺ y))%type.
  Proof. intros x y [Hx | Hx]; [ right; exact Hx | left; exact Hx ]. Defined.

What is apart from zero is not zero.
  Theorem ne_zero_of_apart : ∀ x : R, ((x ≺ 0) + (0 ≺ x))%type → x =/= 0.
  Proof.
    intros x [Hx | Hx] Heq.
    - exact (lt_irrefl 0 (lt_equ_l x 0 0 Heq Hx)).
    - exact (lt_irrefl 0 (lt_equ_r 0 x 0 Heq Hx)).
  Qed.

  Theorem lt_le_trans : ∀ x y z : R, x ≺ y → y ≼ z → x ≺ z.
  Proof.
    intros x y z Hxy Hyz.
    destruct (lt_cotrans x z y Hxy) as [Hxz | Hzy]; [ exact Hxz | destruct (Hyz Hzy) ].
  Defined.

  Theorem le_lt_trans : ∀ x y z : R, x ≼ y → y ≺ z → x ≺ z.
  Proof.
    intros x y z Hxy Hyz.
    destruct (lt_cotrans y x z Hyz) as [Hyx | Hxz]; [ destruct (Hxy Hyx) | exact Hxz ].
  Defined.

  Theorem cast_le : ∀ q r : Q, q ≤ r → cast q ≼ cast r.
  Proof.
    intros q r Hqr Hlt.
    destruct (lt_cast r q Hlt) as [Hrq Hne].
    apply Hne. apply antisymmetry; [ exact Hrq | exact Hqr ].
  Qed.

  Theorem cast_pos : ∀ eps : Q, 0 < eps → (0 : R) ≺ cast eps.
  Proof.
    intros eps Heps.
    exact (lt_equ_l (cast 0) 0 (cast eps) (hom_zero cast) (cast_lt 0 eps Heps)).
  Defined.

Comparison at a precision: two reals are ordered, or the second is below the first plus the precision. This is the decision that always computes.
  Theorem le_equ_l : ∀ x x' y : R, x == x' → x ≼ y → x' ≼ y.
  Proof. intros x x' y Hxx' Hxy Hyx'. exact (Hxy (lt_equ_r y x' x (symmetry Hxx') Hyx')). Qed.

  Theorem le_equ_r : ∀ x y y' : R, y == y' → x ≼ y → x ≼ y'.
  Proof. intros x y y' Hyy' Hxy Hy'x. exact (Hxy (lt_equ_l y' y x (symmetry Hyy') Hy'x)). Qed.

  Theorem add_le_mono_l : ∀ x y z : R, x ≼ y → z + x ≼ z + y.
  Proof. intros x y z Hxy Hlt. exact (Hxy (add_lt_reg_l z y x Hlt)). Qed.

  Theorem add_le_mono_r : ∀ x y z : R, x ≼ y → x + z ≼ y + z.
  Proof.
    intros x y z Hxy.
    apply (le_equ_l (z + x)); [ exact (commutative (op := add) z x) | ].
    apply (le_equ_r _ (z + y)); [ exact (commutative (op := add) z y) | ].
    exact (add_le_mono_l x y z Hxy).
  Qed.

  Theorem add_le_mono : ∀ x y z w : R, x ≼ y → z ≼ w → x + z ≼ y + w.
  Proof.
    intros x y z w Hxy Hzw.
    exact (le_trans _ _ _ (add_le_mono_r x y z Hxy) (add_le_mono_l z w y Hzw)).
  Qed.

  Theorem le_abs : ∀ x : R, x ≼ abs x.
  Proof. intros x. exact (proj1 (proj2 (abs_le_iff x (abs x)) (le_refl (abs x)))). Qed.

  Theorem neg_le_abs : ∀ x : R, - x ≼ abs x.
  Proof. intros x. exact (proj2 (proj2 (abs_le_iff x (abs x)) (le_refl (abs x)))). Qed.

  Theorem abs_nonneg : ∀ x : R, 0 ≼ abs x.
  Proof.
    intros x Hlt.
    destruct (lt_cotrans (abs x) x 0 Hlt) as [Hax | Hx0].
    - exact (le_abs x Hax).
    - apply (neg_le_abs x).
      apply (lt_trans (abs x) 0 (- x) Hlt).
      apply (lt_equ_l (- x + x) 0 (- x) (left_inverse (op := add) x)).
      apply (lt_equ_r (- x + x) (- x + 0) (- x) (right_identity (action := add) (- x))).
      exact (add_lt_mono_l (- x) x 0 Hx0).
  Qed.

  Theorem abs_triangle : ∀ x y : R, abs (x + y) ≼ abs x + abs y.
  Proof.
    intros x y. apply (proj1 (abs_le_iff (x + y) (abs x + abs y))). split.
    - exact (add_le_mono x (abs x) y (abs y) (le_abs x) (le_abs y)).
    - apply (le_equ_l (- x + - y)); [ symmetry; exact (neg_add x y) | ].
      exact (add_le_mono (- x) (abs x) (- y) (abs y) (neg_le_abs x) (neg_le_abs y)).
  Qed.

  Theorem abs_neg : ∀ x : R, abs (- x) == abs x.
  Proof.
    intros x. apply equ_of_le.
    - apply (proj1 (abs_le_iff (- x) (abs x))). split.
      + exact (neg_le_abs x).
      + apply (le_equ_l x); [ symmetry; exact (neg_neg x) | exact (le_abs x) ].
    - apply (proj1 (abs_le_iff x (abs (- x)))). split.
      + apply (le_equ_l (- - x)); [ exact (neg_neg x) | exact (neg_le_abs (- x)) ].
      + exact (le_abs (- x)).
  Qed.

  Theorem abs_zero : abs (0 : R) == 0.
  Proof.
    apply equ_of_le.
    - apply (proj1 (abs_le_iff 0 0)). split; [ exact (le_refl 0) | ].
      apply (le_equ_l 0); [ symmetry; exact neg_zero | exact (le_refl 0) ].
    - exact (abs_nonneg 0).
  Qed.

  Theorem abs_of_nonneg : ∀ x : R, 0 ≼ x → abs x == x.
  Proof.
    intros x Hx. apply equ_of_le.
    - apply (proj1 (abs_le_iff x x)). split; [ exact (le_refl x) | ].
      apply (le_trans (- x) 0 x); [ | exact Hx ].
      apply (le_equ_l (- x + 0)); [ exact (right_identity (action := add) (- x)) | ].
      apply (le_equ_r _ (- x + x)); [ exact (left_inverse (op := add) x) | ].
      exact (add_le_mono_l 0 x (- x) Hx).
    - exact (le_abs x).
  Qed.

  Theorem mul_le_mono_r : ∀ c x y : R, 0 ≼ c → x ≼ y → x × c ≼ y × c.
  Proof.
    intros c x y Hc Hxy.
    apply (le_equ_l (c × x)); [ exact (commutative (op := mult) c x) | ].
    apply (le_equ_r _ (c × y)); [ exact (commutative (op := mult) c y) | ].
    exact (mul_le_mono_l c x y Hc Hxy).
  Qed.

  Theorem cast_nonneg : ∀ q : Q, 0 ≤ q → 0 ≼ cast q.
  Proof.
    intros q Hq.
    apply (le_equ_l (cast 0)); [ exact (hom_zero cast) | exact (cast_le 0 q Hq) ].
  Qed.

The absolute value commutes with the cast of a rational.
  Theorem abs_cast : ∀ q : Q, abs (cast q) == cast (abs q).
  Proof.
    intros q.
    destruct (le_total 0 q) as [Hq | Hq].
    - calc (abs (cast q))
        == (cast q) { (abs_of_nonneg (cast q) (cast_nonneg q Hq)) }
        == (cast (abs q)) { (OrderedField.abs_of_nonneg q Hq) }.
    - assert (Hnq : 0 ≼ - cast q).
      { apply (le_equ_r _ (cast (- q))); [ exact (hom_neg cast q) | ].
        exact (cast_nonneg (- q) (Theory.Order.neg_nonneg_of_nonpos q Hq)). }
      calc (abs (cast q))
        == (abs (- cast q)) { (abs_neg (cast q)) }
        == (- cast q) { (abs_of_nonneg (- cast q) Hnq) }
        == (cast (- q)) { (hom_neg cast q) }
        == (cast (abs q)) { (Theory.Order.abs_of_nonpos q Hq) }.
  Qed.

What is below every positive rational in absolute value is zero.
  Theorem eq_zero_of_abs_le_all : ∀ x : R,
    (∀ eps : Q, 0 < eps → abs x ≼ cast eps) → x == 0.
  Proof.
    intros x Hx. apply equ_of_le.
    - intros Hlt.
      destruct (dense 0 x Hlt) as [q [H0q Hqx]].
      assert (Hq : 0 < q).
      { apply (lt_cast 0 q). apply (lt_equ_l 0 (cast 0)); [ symmetry; exact (hom_zero cast) | exact H0q ]. }
      exact (Hx q Hq (lt_le_trans (cast q) x (abs x) Hqx (le_abs x))).
    - intros Hlt.
      assert (Hnx : 0 ≺ - x).
      { apply (lt_equ_l (- x + x) 0 (- x) (left_inverse (op := add) x)).
        apply (lt_equ_r (- x + x) (- x + 0) (- x) (right_identity (action := add) (- x))).
        exact (add_lt_mono_l (- x) x 0 Hlt). }
      destruct (dense 0 (- x) Hnx) as [q [H0q Hqx]].
      assert (Hq : 0 < q).
      { apply (lt_cast 0 q). apply (lt_equ_l 0 (cast 0)); [ symmetry; exact (hom_zero cast) | exact H0q ]. }
      exact (Hx q Hq (lt_le_trans (cast q) (- x) (abs x) Hqx (neg_le_abs x))).
  Qed.

  Theorem cast_half : ∀ eps : Q,
    cast (eps × inv (1 + 1)) + cast (eps × inv (1 + 1)) == cast eps.
  Proof.
    intros eps.
    calc (cast (eps × inv (1 + 1)) + cast (eps × inv (1 + 1)))
      == (cast (eps × inv (1 + 1) + eps × inv (1 + 1))) { (additive (f := cast) (eps × inv (1 + 1)) (eps × inv (1 + 1))) }
      == (cast eps) { (Theory.Order.half_add_half eps) }.
  Qed.

  Theorem compare_at : ∀ (a b : R) (eps : Q), 0 < eps →
    sum (a ≺ b) (b ≺ a + cast eps).
  Proof.
    intros a b eps Heps.
    apply (lt_cotrans a b (a + cast eps)).
    exact (lt_equ_l (a + 0) a (a + cast eps) (right_identity a)
             (add_lt_mono_l a 0 (cast eps) (cast_pos eps Heps))).
  Defined.

Two reals further apart than a positive precision are ordered one way or the other; deciding which needs a decided order on the rationals.
  Theorem apart_of_abs_gt `{!LtDec Q} : ∀ (a b : R) (eps : Q), 0 < eps →
    cast eps ≺ abs (a + - b) → ((b ≺ a) + (a ≺ b))%type.
  Proof.
    intros a b eps Heps Habs.
    set (d := a + - b).
    assert (Hd : (0 : R) ≺ abs d) by exact (lt_trans _ _ _ (cast_pos eps Heps) Habs).
    assert (Hapart : ((d ≺ 0) + (0 ≺ d))%type).
    { destruct (lt_cotrans 0 d (abs d) Hd) as [Hpos | Hdabs]; [ right; exact Hpos | ].
      destruct (lt_cotrans d (- d) (abs d) Hdabs) as [Hneg | Hndabs].
      - left.
        assert (Hdd : d + d ≺ 0).
        { exact (lt_equ_r (d + d) (d + - d) 0 (right_inverse (op := add) d) (add_lt_mono_l d d (- d) Hneg)). }
        destruct (lt_cotrans (d + d) d 0 Hdd) as [Hddd | Hd0]; [ | exact Hd0 ].
        exact (add_lt_reg_l d d 0 (lt_equ_r (d + d) d (d + 0) (symmetry (right_identity d)) Hddd)).
      - exfalso.
        destruct (dense d (abs d) Hdabs) as [q [Hdq Hqabs]].
        destruct (dense (- d) (abs d) Hndabs) as [r [Hdr Hrabs]].
        destruct (lt_dec r q) as [Hrq | Hnrq].
        + exact (proj1 (abs_le_iff d (cast q))
                   (conj (le_of_lt d (cast q) Hdq)
                         (le_of_lt (- d) (cast q) (lt_trans _ _ _ Hdr (cast_lt r q Hrq))))
                   Hqabs).
        + exact (proj1 (abs_le_iff d (cast r))
                   (conj (le_of_lt d (cast r)
                            (lt_le_trans d (cast q) (cast r) Hdq (cast_le q r (Theory.Order.le_of_not_lt q r Hnrq))))
                         (le_of_lt (- d) (cast r) Hdr))
                   Hrabs). }
    destruct Hapart as [Hneg | Hpos].
    - right.
      assert (Hb : b + (a + - b) ≺ b + 0) by exact (add_lt_mono_l b (a + - b) 0 Hneg).
      exact (lt_equ_r a (b + 0) b (right_identity b)
               (lt_equ_l (b + (a + - b)) a (b + 0) (add_add_neg_cancel b a) Hb)).
    - left.
      assert (Hb : b + 0 ≺ b + (a + - b)) by exact (add_lt_mono_l b 0 (a + - b) Hpos).
      exact (lt_equ_l (b + 0) b a (right_identity b)
               (lt_equ_r (b + 0) (b + (a + - b)) a (add_add_neg_cancel b a) Hb)).
  Defined.
