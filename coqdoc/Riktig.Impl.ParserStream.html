Generalizable All Variables.

The list-of-successes parsers over any stream whose next decides the sequence's step implement the Parser structure: a parser reads a result when one of its readings is equal to it in the result's setoid, and two parsers are equal when they read the same. The stream is compared by Coq's equality.
Section ParserStream.
  Context {C S : Type} {eqC : EqRel C} {stepping : HasStep C S} {Seq : Sequence C S stepping (eqS := eq)}
    {NX : HasNext C S}.

  #[export] Instance Stream_EqRel (A : Type) (eqA : EqRel A) : EqRel (parser S A) :=
    fun p q ⇒ ∀ (s s' : S) (a : A), p_parses equ p s a s' ↔ p_parses equ q s a s'.
  #[export] Instance Stream_HasReturn : HasReturn (parser S) := @p_ret S.
  #[export] Instance Stream_HasBind : HasBind (parser S) := @p_bind S.
  #[export] Instance Stream_HasFail : HasFail (parser S) := @p_fail S.
  #[export] Instance Stream_HasAlt : HasAlt (parser S) := @p_alt S.
  #[export] Instance Stream_HasItem : HasItem C (parser S) := p_item.
  #[export] Instance Stream_HasParses : HasParses S (parser S) := fun A eqA ⇒ @p_parses S A equ.

next decides step, up to the element's equality.
  Context (next_step : ∀ (s : S) (c : C) (s' : S),
    step s c s' ↔ ∃ c0 : C, next s = Some (c0, s') ∧ c0 == c).

  Instance Stream_Parser : Parser C S (parser S)
      Stream_HasReturn Stream_HasBind Stream_HasFail Stream_HasAlt Stream_HasItem Stream_HasParses.
  Proof.
    split.
    - intros A eqA SA p q. split; intros Hpq; exact Hpq.
    - intros A eqA SA p s s' Hs a a' Ha t t' Ht. unfold equ in Hs, Ht. subst.
      unfold parses, Stream_HasParses, p_parses. split; intros [a0 [Hin H0]]; ∃ a0; split; try assumption.
      + transitivity a; assumption.
      + transitivity a'; [ exact H0 | symmetry; exact Ha ].
    - intros A eqA SA a a' s s'. exact (p_parses_ret equ a s a' s').
    - intros A eqA SA B eqB SB p f s s'' b Hf.
      apply (p_parses_bind equ equ p f). intros a a' Ha s0 b0 s0'. exact (Hf a a' Ha s0 s0' b0).
    - intros A eqA SA s s' a. exact (p_parses_fail equ s a s').
    - intros A eqA SA p q s s' a. exact (p_parses_alt equ p q s a s').
    - intros s s' c. unfold parses, Stream_HasParses, item, Stream_HasItem. rewrite next_step. split.
      + intros [c0 [Hin Hc]]. ∃ c0. split; [ | exact Hc ].
        apply (p_parses_item s c0 s'). ∃ c0. split; [ exact Hin | reflexivity ].
      + intros [c0 [Hn Hc]]. ∃ c0. split; [ | exact Hc ].
        destruct (proj2 (p_parses_item s c0 s') Hn) as [c1 [Hin Hc1]]. subst c1. exact Hin.
  Qed.
End ParserStream.
