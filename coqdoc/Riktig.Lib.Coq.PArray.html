A primitive array read as a table over an index I; the index is a phantom parameter, as for lists. An array holds at most max_length = 4194303 entries, so a table over an index of a larger count does not fit.
Definition ArrayTable (I A : Type) : Type := array A.

Definition max_nat : Z := 4194303.

Definition int_of_nat (k : nat) : int := of_Z (Z.of_nat k).

Lemma wB_value : wB = 9223372036854775808.
Proof. reflexivity. Qed.

Lemma to_Z_int_of_nat : ∀ k : nat, Z.of_nat k ≤ max_nat → to_Z (int_of_nat k) = Z.of_nat k.
Proof.
  intros k Hk. unfold int_of_nat. rewrite of_Z_spec. apply Z.mod_small.
  unfold max_nat in Hk. rewrite wB_value. lia.
Qed.

Lemma int_of_nat_inj : ∀ j k : nat, Z.of_nat j ≤ max_nat → Z.of_nat k ≤ max_nat →
  int_of_nat j = int_of_nat k → j = k.
Proof.
  intros j k Hj Hk H.
  apply Nat2Z.inj. rewrite <- (to_Z_int_of_nat j Hj), <- (to_Z_int_of_nat k Hk), H.
  reflexivity.
Qed.

Lemma ltb_int_of_nat : ∀ j n : nat, Z.of_nat j ≤ max_nat → Z.of_nat n ≤ max_nat →
  (int_of_nat j <? int_of_nat n)%uint63 = true ↔ (j < n)%nat.
Proof.
  intros j n Hj Hn. rewrite ltb_spec, (to_Z_int_of_nat j Hj), (to_Z_int_of_nat n Hn). lia.
Qed.

Lemma length_make_nat : ∀ (A : Type) (n : nat) (d : A), Z.of_nat n ≤ max_nat →
  PArray.length (make (int_of_nat n) d) = int_of_nat n.
Proof.
  intros A n d Hn. rewrite length_make.
  assert (Hle : (int_of_nat n <=? max_length)%uint63 = true).
  { rewrite leb_spec. rewrite (to_Z_int_of_nat n Hn). unfold max_nat in Hn.
    change (to_Z max_length) with 4194303. lia. }
  rewrite Hle. reflexivity.
Qed.

The array for f on 0 .. n-1: a fresh array of n defaults, its entries set in number order.
Fixpoint fill {A : Type} (f : nat → A) (m : nat) (a : array A) : array A :=
  match m with
  | O ⇒ a
  | S m' ⇒ (fill f m' a).[int_of_nat m' <- f m']
  end.

Definition tabulate_array {A : Type} (n : nat) (d : A) (f : nat → A) : array A :=
  fill f n (make (int_of_nat n) d).

Lemma fill_spec : ∀ (A : Type) (f : nat → A) (n m : nat) (a : array A),
  Z.of_nat n ≤ max_nat → (m ≤ n)%nat → PArray.length a = int_of_nat n →
  PArray.length (fill f m a) = int_of_nat n ∧
  ∀ k : nat, (k < m)%nat → (fill f m a).[int_of_nat k] = f k.
Proof.
  intros A f n m a Hn Hm Hlen.
  induction m as [| m IH].
  - simpl. split; [ exact Hlen | intros k Hk; lia ].
  - simpl. destruct (IH (ltac:(lia))) as [IHlen IHget].
    split.
    + rewrite length_set. exact IHlen.
    + intros k Hk.
      destruct (Nat.eq_dec k m) as [-> | Hne].
      × apply get_set_same. rewrite IHlen. apply ltb_int_of_nat; [ lia | exact Hn | lia ].
      × rewrite get_set_other.
        -- apply IHget. lia.
        -- intros Heq. apply Hne. symmetry. apply (int_of_nat_inj m k); [ lia | lia | exact Heq ].
Qed.

Lemma get_tabulate_array : ∀ (A : Type) (n : nat) (d : A) (f : nat → A) (k : nat),
  Z.of_nat n ≤ max_nat → (k < n)%nat → (tabulate_array n d f).[int_of_nat k] = f k.
Proof.
  intros A n d f k Hn Hk. unfold tabulate_array.
  destruct (fill_spec A f n n (make (int_of_nat n) d) Hn (le_n n) (length_make_nat A n d Hn))
    as [_ Hget].
  exact (Hget k Hk).
Qed.
