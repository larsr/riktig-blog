Library Riktig.Structure.MatrixDeterminant

From Riktig.Structure Require Export Field Decidable.

A determinant on square matrices over a field, indexed by a type with decided equality -- matrices being functions that respect equality: it respects equality, is linear in each column, vanishes when two distinct columns are equal, and is 1 on the identity.
Module MatrixDeterminant.
  Class MatrixDeterminant (F I : Type) `{Field F} {eqI : EqRel I} {EI : EquDec I} (determinant : HasMatDet F I) := {
    matdet_proper : ∀ M M' : I → I → F,
      Proper (equ ==> equ ==> equ) M → Proper (equ ==> equ ==> equ) M' →
      (∀ i j : I, M i j == M' i j) → matdet M == matdet M';
    matdet_col_add : ∀ (M : I → I → F) (j : I) (u v : I → F),
      Proper (equ ==> equ ==> equ) M → Proper (equ ==> equ) u → Proper (equ ==> equ) v →
      matdet (fun i k ⇒ if equ_dec k j then u i + v i else M i k)
        == matdet (fun i k ⇒ if equ_dec k j then u i else M i k)
           + matdet (fun i k ⇒ if equ_dec k j then v i else M i k);
    matdet_col_smul : ∀ (M : I → I → F) (j : I) (a : F) (u : I → F),
      Proper (equ ==> equ ==> equ) M → Proper (equ ==> equ) u →
      matdet (fun i k ⇒ if equ_dec k j then a × u i else M i k)
        == a × matdet (fun i k ⇒ if equ_dec k j then u i else M i k);
    matdet_alternating : ∀ (M : I → I → F) (j j' : I),
      Proper (equ ==> equ ==> equ) M →
      j =/= j' → (∀ i : I, M i j == M i j') → matdet M == 0;
    matdet_id : matdet (fun i j : I ⇒ if equ_dec i j then (1 : F) else 0) == 1;
  }.
  Hint Mode MatrixDeterminant - ! - - - - - - - - - - - : typeclass_instances.
End MatrixDeterminant.
