Library Riktig.Theory.Jacobian

From Coq Require Export Morphisms.
From Coq.Vectors Require Fin.
From Riktig Require Export Theory.Derivative Theory.MatRep.


Section Jacobian.
  Context {F U V : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context (n m : nat) (BU : Basis n (Fin.t n) F U) (BV : Basis m (Fin.t m) F V).

  Definition jacobian {f : U → V} {a : U} {L : LinearMap F U V}
      (Hd : is_deriv f a L) : Matrix m n F :=
    mat_of BU BV L.

  Theorem jacobian_col : ∀ {f : U → V} {a : U} {L : LinearMap F U V}
      (Hd : is_deriv f a L) (i : Fin.t m) (j : Fin.t n),
    jacobian Hd i j == Basis.coords BV (partial_at L BU j) i.
  Proof.
    intros f a L Hd i j.
    reflexivity.
  Qed.

  Example jacobian_of_linear : ∀ (g : LinearMap F U V)
      (Hb : linear_map_bounded g) (a : U),
    jacobian (deriv_lm g a Hb) == mat_of BU BV g.
  Proof.
    intros g Hb a.
    reflexivity.
  Qed.

End Jacobian.


Section JacobianComp.
  Context {F U V W : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VSW : VectorSpace F W} {nmW : HasNorm F W} {NSW : NormedSpace F W}.
  Context (n m p : nat) (BU : Basis n (Fin.t n) F U)
    (BV : Basis m (Fin.t m) F V) (BW : Basis p (Fin.t p) F W).

  Theorem jacobian_comp : ∀ (f : U → V) (g : V → W) (a : U)
      (Lf : LinearMap F U V) (Lg : LinearMap F V W),
    is_deriv f a Lf → is_deriv g (f a) Lg →
    is_deriv (fun u : U ⇒ g (f u)) a (linear_map_comp Lg Lf) →
    mat_of BU BW (linear_map_comp Lg Lf)
      == mat_mul (mat_of BV BW Lg) (mat_of BU BV Lf).
  Proof.
    intros f g a Lf Lg Hf Hg Hgf.
    apply mat_of_comp.
  Qed.

  Theorem jacobian_chain_rule : ∀ (f : U → V) (g : V → W) (a : U)
      (Lf : LinearMap F U V) (Lg : LinearMap F V W)
      (Pg : Proper (equ ==> equ) g)
      (Hf : is_deriv f a Lf) (Hg : is_deriv g (f a) Lg),
    jacobian n p BU BW (deriv_comp f g a Lf Lg Pg Hf Hg)
      == mat_mul (jacobian m p BV BW Hg) (jacobian n m BU BV Hf).
  Proof.
    intros f g a Lf Lg Pg Hf Hg.
    apply mat_of_comp.
  Qed.

End JacobianComp.
