The matrix of a linear map between spaces of tables: its entry at i, j is the i-th coordinate of the image of the j-th unit table. The matrix of a composite is the product of the matrices: the chain rule for Jacobians.

  Lemma delta_fun_proper : ∀ j : J, Proper (equ ==> equ) (fun i : J ⇒ if equ_dec i j then (1 : F) else 0).
  Proof.
    intros j i i' Hi.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i' j) as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity (symmetry Hi) Hij).
    - destruct Hij. exact (transitivity Hi Hij').
  Qed.

  Lemma delta_proper : Proper (equ ==> equ) (fun j : J ⇒ delta j).
  Proof.
    intros j j' Hj. apply tabulate_proper. intros i.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i j') as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity Hij Hj).
    - destruct Hij. exact (transitivity Hij' (symmetry Hj)).
  Qed.

  Lemma get_delta : ∀ i j : J, get (delta j) i == if equ_dec i j then (1 : F) else 0.
  Proof. intros i j. exact (get_tabulate _ i). Qed.

  Lemma comb_step_proper : ∀ (c : J → F) (u : J → TJ),
    Proper (equ ==> equ) c → Proper (equ ==> equ) u →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : TJ) ⇒ acc + c (elem k) *: u (elem k)).
  Proof.
    intros c u Hc Hu k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
    apply scale_proper; [ apply Hc | apply Hu ]; apply elem_proper; exact Hk.
  Qed.

Reading the coordinate i of a partial combination of units: zero before the index of i, the coefficient from there on.
  Lemma get_comb_delta : ∀ (t : TJ) (i : J) (n : N), n ≤ card →
    (n ≤ index i → get (comb (get t) (fun j ⇒ delta j) n) i == 0)
    ∧ (index i < n → get (comb (get t) (fun j ⇒ delta j) n) i == get t i).
  Proof.
    intros t i.
    assert (Hstep := comb_step_proper (get t) (fun j ⇒ delta j) (get_proper t t (reflexivity t)) delta_proper).
    apply (nat_induction (fun n ⇒ n ≤ card →
      (n ≤ index i → get (comb (get t) (fun j ⇒ delta j) n) i == 0)
      ∧ (index i < n → get (comb (get t) (fun j ⇒ delta j) n) i == get t i))).
    - intros n n' Hn.
      assert (He : get (comb (get t) (fun j ⇒ delta j) n) i == get (comb (get t) (fun j ⇒ delta j) n') i)
        by (apply get_proper; [ exact (natrec_proper TJ _ _ 0 _ Hstep n n' Hn) | reflexivity ]).
      split; intros Hall Hc.
      + assert (Hc' : n ≤ card) by (setoid_rewrite Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite <- He. apply HZ. setoid_rewrite Hn. exact Hle.
        × intros Hlt. setoid_rewrite <- He. apply HB.
          exact (proj2 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
      + assert (Hc' : n' ≤ card) by (setoid_rewrite <- Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite He. apply HZ. setoid_rewrite <- Hn. exact Hle.
        × intros Hlt. setoid_rewrite He. apply HB.
          exact (proj1 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
    - intros _. split.
      + intros _. apply (transitivity (get_proper _ _ (natrec_zero TJ _ _ 0 _) i i (reflexivity i))). exact (get_zero i).
      + intros Hlt. destruct (Theory.NatOrder.not_lt_zero (index i) Hlt).
    - intros n IH Hsn.
      assert (Hn : n ≤ card) by exact (Theory.NatOrder.le_trans n (succ n) card (Theory.NatOrder.le_succ_diag n) Hsn).
      assert (Hlt : n < card) by exact (Theory.NatOrder.lt_of_lt_of_le n (succ n) card (Theory.NatOrder.lt_succ_diag n) Hsn).
      destruct (IH Hn) as [HZ HB].
      assert (Hs : get (comb (get t) (fun j ⇒ delta j) (succ n)) i
                   == get (comb (get t) (fun j ⇒ delta j) n) i + get t (elem n) × get (delta (elem n)) i).
      { apply (transitivity (get_proper _ _ (natrec_succ TJ _ _ 0 _ n) i i (reflexivity i))).
        apply (transitivity (get_add _ _ i)). apply op_proper; [ reflexivity | exact (get_smul _ _ i) ]. }
      setoid_rewrite (get_delta i (elem n)) in Hs.
      destruct (equ_dec i (elem n)) as [Hie | Hie].
      + assert (Hni : n == index i).
        { calc n == (index (I := J) (N := N) (elem (I := J) (N := N) n)) { (index_elem (I := J) (N := N) n Hlt) }
            == (index i) { (index_proper (elem n) i (symmetry Hie)) }. }
        split.
        × intros Hle. exfalso. apply (Theory.NatOrder.lt_irrefl n).
          apply (Theory.NatOrder.lt_of_lt_of_le n (succ n) n (Theory.NatOrder.lt_succ_diag n)).
          assert (Hback : index i ≤ n) by (setoid_rewrite Hni; exact (Theory.NatOrder.le_refl (index i))).
          exact (Theory.NatOrder.le_trans (succ n) (index i) n Hle Hback).
        × intros _.
          assert (Hle0 : n ≤ index i) by (setoid_rewrite <- Hni; exact (Theory.NatOrder.le_refl n)).
          calc (get (comb (get t) (fun j ⇒ delta j) (succ n)) i)
            == (get (comb (get t) (fun j ⇒ delta j) n) i + get t (elem n) × 1) { Hs }
            == (0 + get t (elem n) × 1) { (HZ Hle0) }
            == (get t (elem n) × 1) { (left_identity (action := add) (get t (elem n) × 1)) }
            == (get t (elem n)) { (right_identity (action := mult) (get t (elem n))) }
            == (get t i) { (get_proper t t (reflexivity t) (elem n) i (symmetry Hie)) }.
      + assert (Heq : get (comb (get t) (fun j ⇒ delta j) (succ n)) i == get (comb (get t) (fun j ⇒ delta j) n) i).
        { calc (get (comb (get t) (fun j ⇒ delta j) (succ n)) i)
            == (get (comb (get t) (fun j ⇒ delta j) n) i + get t (elem n) × 0) { Hs }
            == (get (comb (get t) (fun j ⇒ delta j) n) i + 0) { (mul_zero (get t (elem n))) }
            == (get (comb (get t) (fun j ⇒ delta j) n) i)
                 { (right_identity (action := add) (get (comb (get t) (fun j ⇒ delta j) n) i)) }. }
        split.
        × intros Hle. setoid_rewrite Heq. apply HZ.
          exact (Theory.NatOrder.le_trans n (succ n) (index i) (Theory.NatOrder.le_succ_diag n) Hle).
        × intros Hlt'. setoid_rewrite Heq. apply HB.
          destruct (proj1 (Theory.NatOrder.lt_succ_iff (index i) n) Hlt') as [Hlt'' | Heq'']; [ exact Hlt'' | ].
          exfalso. apply Hie.
          calc i == (elem (I := J) (N := N) (index i)) { (elem_index i) }
            == (elem (I := J) (N := N) n) { (elem_proper (index i) n Heq'') }.
  Qed.

Every table is the combination of the units by its coordinates.
  Theorem decompose : ∀ t : TJ, t == comb (get t) (fun j ⇒ delta j) card.
  Proof.
    intros t.
    apply (transitivity (symmetry (tabulate_get t))).
    apply (transitivity (y := tabulate (get (comb (get t) (fun j ⇒ delta j) card)))); [ | exact (tabulate_get _) ].
    apply tabulate_proper. intros i. symmetry.
    exact (proj2 (get_comb_delta t i card (Theory.NatOrder.le_refl card)) (index_lt i)).
  Qed.

A linear map carries a combination to the combination of the images.
  Lemma linear_comb : ∀ (L : TJ → TI) (c : J → F) (u : J → TJ),
    Linear F TJ TI L → Proper (equ ==> equ) c → Proper (equ ==> equ) u →
    ∀ n : N, L (comb c u n) == natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: L (u (elem k))) n.
  Proof.
    intros L c u HL Hc Hu.
    assert (HLu : Proper (equ ==> equ) (fun j ⇒ L (u j)))
      by (intros j j' Hj; apply (proper (f := L)); apply Hu; exact Hj).
    assert (HsI : Proper (equ ==> equ ==> equ) (fun (k : N) (acc : TI) ⇒ acc + c (elem k) *: L (u (elem k))))
      by (intros k k' Hk acc acc' Hacc; apply op_proper; [ exact Hacc | apply scale_proper; [ apply Hc | apply HLu ]; apply elem_proper; exact Hk ]).
    apply (nat_induction (fun n ⇒ L (comb c u n) == natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: L (u (elem k))) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper TJ _ _ 0 _ (comb_step_proper c u Hc Hu) n n' Hn).
      setoid_rewrite (natrec_proper TI _ _ 0 _ HsI n n' Hn). reflexivity.
    - calc (L (comb c u 0))
        == (L (0 : TJ)) { (proper (f := L) _ _ (natrec_zero TJ _ _ 0 _)) }
        == (0 : TI) { (additive_zero L) }
        == (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: L (u (elem k))) 0) { (natrec_zero TI _ _ 0 _) }.
    - intros n IH.
      calc (L (comb c u (succ n)))
        == (L (comb c u n + c (elem n) *: u (elem n))) { (proper (f := L) _ _ (natrec_succ TJ _ _ 0 _ n)) }
        == (L (comb c u n) + L (c (elem n) *: u (elem n))) { (additive (f := L) _ _) }
        == (L (comb c u n) + c (elem n) *: L (u (elem n))) { (homogeneous (f := L) (c (elem n)) (u (elem n))) }
        == (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: L (u (elem k))) n + c (elem n) *: L (u (elem n))) { IH }
        == (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: L (u (elem k))) (succ n)) { (natrec_succ TI _ _ 0 _ n) }.
  Qed.

A coordinate of a combination of tables.
  Lemma get_comb : ∀ (c : J → F) (v : J → TI) (i : I),
    Proper (equ ==> equ) c → Proper (equ ==> equ) v →
    ∀ n : N, get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) n) i
                  == natrec F 0 (fun k acc ⇒ acc + c (elem k) × get (v (elem k)) i) n.
  Proof.
    intros c v i Hc Hv.
    assert (Hsv : Proper (equ ==> equ ==> equ) (fun (k : N) (acc : TI) ⇒ acc + c (elem k) *: v (elem k)))
      by (intros k k' Hk acc acc' Hacc; apply op_proper; [ exact Hacc | apply scale_proper; [ apply Hc | apply Hv ]; apply elem_proper; exact Hk ]).
    assert (Hsf : Proper (equ ==> equ ==> equ) (fun (k : N) (acc : F) ⇒ acc + c (elem k) × get (v (elem k)) i))
      by (intros k k' Hk acc acc' Hacc; apply op_proper; [ exact Hacc | ];
          apply op_proper; [ apply Hc; apply elem_proper; exact Hk | apply get_proper; [ apply Hv; apply elem_proper; exact Hk | reflexivity ] ]).
    apply (nat_induction (fun n ⇒ get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) n) i
                  == natrec F 0 (fun k acc ⇒ acc + c (elem k) × get (v (elem k)) i) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper TI _ _ 0 _ Hsv n n' Hn).
      setoid_rewrite (natrec_proper F _ _ 0 _ Hsf n n' Hn). reflexivity.
    - calc (get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) 0) i)
        == (get (0 : TI) i) { (get_proper _ _ (natrec_zero TI _ _ 0 _) i i (reflexivity i)) }
        == (0 : F) { (get_zero i) }
        == (natrec F 0 (fun k acc ⇒ acc + c (elem k) × get (v (elem k)) i) 0) { (natrec_zero F _ _ 0 _) }.
    - intros n IH.
      calc (get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) (succ n)) i)
        == (get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) n + c (elem n) *: v (elem n)) i)
             { (get_proper _ _ (natrec_succ TI _ _ 0 _ n) i i (reflexivity i)) }
        == (get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) n) i + get (c (elem n) *: v (elem n)) i)
             { (get_add _ _ i) }
        == (get (natrec TI 0 (fun k acc ⇒ acc + c (elem k) *: v (elem k)) n) i + c (elem n) × get (v (elem n)) i)
             { (get_smul (c (elem n)) (v (elem n)) i) }
        == (natrec F 0 (fun k acc ⇒ acc + c (elem k) × get (v (elem k)) i) n + c (elem n) × get (v (elem n)) i)
             { IH }
        == (natrec F 0 (fun k acc ⇒ acc + c (elem k) × get (v (elem k)) i) (succ n))
             { (natrec_succ F _ _ 0 _ n) }.
  Qed.

The chain rule for matrices: the matrix of a composite is the product of the matrices.
  Theorem jacobian_comp : ∀ (L : TJ → TI) (M : TJ → TJ) (i : I) (k : J),
    Linear F TJ TI L → Linear F TJ TJ M →
    jacobian (fun h ⇒ L (M h)) i k
      == natrec F 0 (fun l acc ⇒ acc + jacobian L i (elem l) × jacobian M (elem l) k) card.
  Proof.
    intros L M i k HL HM.
    assert (HLd : Proper (equ ==> equ) (fun j ⇒ L (delta j)))
      by (intros j j' Hj; apply (proper (f := L)); apply delta_proper; exact Hj).
    calc (get (L (M (delta k))) i)
      == (get (L (comb (get (M (delta k))) (fun j ⇒ delta j) card)) i)
           { (get_proper _ _ (proper (f := L) _ _ (decompose (M (delta k)))) i i (reflexivity i)) }
      == (get (natrec TI 0 (fun l acc ⇒ acc + get (M (delta k)) (elem l) *: L (delta (elem l))) card) i)
           { (get_proper _ _ (linear_comb L (get (M (delta k))) (fun j ⇒ delta j) HL (get_proper _ _ (reflexivity _)) delta_proper card) i i (reflexivity i)) }
      == (natrec F 0 (fun l acc ⇒ acc + get (M (delta k)) (elem l) × get (L (delta (elem l))) i) card)
           { (get_comb (get (M (delta k))) (fun j ⇒ L (delta j)) i (get_proper _ _ (reflexivity _)) HLd card) }
      == (natrec F 0 (fun l acc ⇒ acc + jacobian L i (elem l) × jacobian M (elem l) k) card)
           { _ }.
    apply (natrec_ext F _ _ 0 _ _).
    - intros l l' Hl acc acc' Hacc. apply op_proper; [ exact Hacc | ].
      apply op_proper; [ apply get_proper; [ reflexivity | apply elem_proper; exact Hl ] | apply get_proper; [ apply HLd; apply elem_proper; exact Hl | reflexivity ] ].
    - intros l l' Hl acc acc' Hacc. apply op_proper; [ exact Hacc | ].
      apply op_proper; [ apply get_proper; [ apply HLd; apply elem_proper; exact Hl | reflexivity ] | apply get_proper; [ reflexivity | apply elem_proper; exact Hl ] ].
    - intros l acc. apply op_proper; [ reflexivity | ]. exact (commutative (op := mult) _ _).
  Qed.
