Definition orthogonal {F V : Type} `{InnerProductSpace F V} (u v : V) : Prop :=
  ⟪ u , v ⟫ == 0.

proj v u is the component of u along v.

Section InnerKit.
  Context {F V : Type} `{InnerProductSpace F V}.
  Context {Vp : HasProj V} {PJ : Proj F V}.

  Lemma orthogonal_eq : ∀ u v : V, orthogonal u v → ⟪ u , v ⟫ == 0.
  Proof.
    intros u v Huv.
    exact Huv.
  Qed.

  Lemma orthogonal_def : ∀ u v : V, ⟪ u , v ⟫ == 0 → orthogonal u v.
  Proof.
    intros u v Huv.
    exact Huv.
  Qed.

  #[export] Instance orthogonal_proper :
    Proper (equ ==> equ ==> iff) (orthogonal (F := F) (V := V)).
  Proof.
    intros u u' Hu v v' Hv.
    split; intros Ho; apply orthogonal_def.
    - rewrite <- Hu, <- Hv. exact (orthogonal_eq u v Ho).
    - rewrite Hu, Hv. exact (orthogonal_eq u' v' Ho).
  Qed.

  Lemma proj_eq : ∀ u v : V,
    proj v u == (⟪ u , v ⟫ × inv (norm2 v)) *: v.
  Proof. exact is_proj_def. Qed.

  Theorem orthogonal_sym : ∀ u v : V, orthogonal u v → orthogonal v u.
  Proof.
    intros u v Huv.
    apply orthogonal_def.
    rewrite (commutative v u).
    exact (orthogonal_eq u v Huv).
  Qed.

  Theorem orthogonal_scale_r : ∀ (u v : V) (a : F),
    orthogonal u v → orthogonal u (a *: v).
  Proof.
    intros u v a Huv.
    apply orthogonal_def.
    calc (⟪ u , a *: v ⟫)
      == (a × ⟪ u , v ⟫) { inner_right_scale }
      == (a × 0) { orthogonal_eq }
      == (0 : F) { mul_0_r }.
  Qed.

  Theorem norm2_neg : ∀ u : V, norm2 (- u) == norm2 u.
  Proof.
    intros u.
    calc (norm2 (- u))
      == (⟪ - u , - u ⟫) { norm2_spec }
      == (- ⟪ u , - u ⟫) { inner_left_neg }
      == (- (- ⟪ u , u ⟫)) { inner_right_neg }
      == (⟪ u , u ⟫) { neg_involutive }
      == (norm2 u) { norm2_spec }.
  Qed.

  Theorem pythagoras : ∀ u v : V,
    orthogonal u v → norm2 (u + v) == norm2 u + norm2 v.
  Proof.
    intros u v Huv.
    calc (norm2 (u + v))
      == ((norm2 u + norm2 v) + (⟪ u , v ⟫ + ⟪ v , u ⟫))
        { norm2_add }
      == ((norm2 u + norm2 v) + (⟪ u , v ⟫ + ⟪ u , v ⟫))
        { (commutative v) }
      == ((norm2 u + norm2 v) + (0 + 0))
        { orthogonal_eq }
      == ((norm2 u + norm2 v) + 0)
        { left_identity }
      == (norm2 u + norm2 v)
        { right_identity }.
  Qed.

  Theorem norm2_scale_sub_expand : ∀ (a b : F) (u v : V),
    norm2 (a *: u + - (b *: v)) ==
      ((a × a) × norm2 u + (b × b) × norm2 v)
      + (- (a × (b × ⟪ u , v ⟫)) + - (a × (b × ⟪ u , v ⟫))).
  Proof.
    intros a b u v.
    transitivity
      (((a × a) × norm2 u + (b × b) × norm2 v)
        + (- (a × ⟪ u , b *: v ⟫) + - ⟪ b *: v , a *: u ⟫)).
    - calc (norm2 (a *: u + - (b *: v)))
        == ((norm2 (a *: u) + norm2 (- (b *: v)))
              + (⟪ a *: u , - (b *: v) ⟫ + ⟪ - (b *: v) , a *: u ⟫))
          { norm2_add }
        == (((a × a) × norm2 u + norm2 (- (b *: v)))
              + (⟪ a *: u , - (b *: v) ⟫ + ⟪ - (b *: v) , a *: u ⟫))
          { norm2_scale }
        == (((a × a) × norm2 u + norm2 (b *: v))
              + (⟪ a *: u , - (b *: v) ⟫ + ⟪ - (b *: v) , a *: u ⟫))
          { norm2_neg }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (⟪ a *: u , - (b *: v) ⟫ + ⟪ - (b *: v) , a *: u ⟫))
          { norm2_scale }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- ⟪ a *: u , b *: v ⟫ + ⟪ - (b *: v) , a *: u ⟫))
          { inner_right_neg }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- ⟪ a *: u , b *: v ⟫ + - ⟪ b *: v , a *: u ⟫))
          { inner_left_neg }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × ⟪ u , b *: v ⟫) + - ⟪ b *: v , a *: u ⟫))
          { inner_left_scale }.
    - calc (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × ⟪ u , b *: v ⟫) + - ⟪ b *: v , a *: u ⟫))
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × (b × ⟪ u , v ⟫)) + - ⟪ b *: v , a *: u ⟫))
          { inner_right_scale }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × (b × ⟪ u , v ⟫)) + - (b × ⟪ v , a *: u ⟫)))
          { inner_left_scale }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × (b × ⟪ u , v ⟫)) + - (b × (a × ⟪ v , u ⟫))))
          { inner_right_scale }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × (b × ⟪ u , v ⟫)) + - (b × (a × ⟪ u , v ⟫))))
          { (commutative v) }
        == (((a × a) × norm2 u + (b × b) × norm2 v)
              + (- (a × (b × ⟪ u , v ⟫)) + - (a × (b × ⟪ u , v ⟫))))
          { mul_left_commutative }.
  Qed.

  Lemma proj_sub_scale : ∀ (u v : V) (c : F),
    proj v u + - (c *: v) == (⟪ u , v ⟫ × inv (norm2 v) + - c) *: v.
  Proof.
    intros u v c.
    rewrite (sub_scale (⟪ u , v ⟫ × inv (norm2 v)) c v), (proj_eq u v).
    reflexivity.
  Qed.

  Theorem proj_orthogonal : ∀ u v : V,
    norm2 v =/= 0 → orthogonal (u + - proj v u) v.
  Proof.
    intros u v Hv.
    apply orthogonal_def.
    transitivity (⟪ u , v ⟫ + - ⟪ u , v ⟫).
    - calc (⟪ u + - proj v u , v ⟫)
        == (⟪ u , v ⟫ + ⟪ - proj v u , v ⟫)
          { inner_left_additive }
        == (⟪ u , v ⟫ + - ⟪ proj v u , v ⟫)
          { inner_left_neg }
        == (⟪ u , v ⟫ + - ⟪ (⟪ u , v ⟫ × inv (norm2 v)) *: v , v ⟫)
          { proj_eq }
        == (⟪ u , v ⟫ + - ((⟪ u , v ⟫ × inv (norm2 v)) × ⟪ v , v ⟫))
          { inner_left_scale }
        == (⟪ u , v ⟫ + - ((⟪ u , v ⟫ × inv (norm2 v)) × norm2 v))
          { norm2_spec }
        == (⟪ u , v ⟫ + - (norm2 v × (⟪ u , v ⟫ × inv (norm2 v))))
          { mul_commutative }
        == (⟪ u , v ⟫ + - ⟪ u , v ⟫)
          { field_mul_div_cancel }.
    - exact (right_inverse _).
  Qed.

  Lemma proj_orthogonal_to : ∀ (u v w : V),
    orthogonal u v → orthogonal (proj v w) u.
  Proof.
    intros u v w Huv.
    apply orthogonal_def.
    calc (⟪ proj v w , u ⟫)
      == (⟪ (⟪ w , v ⟫ × inv (norm2 v)) *: v , u ⟫)
        { proj_eq }
      == ((⟪ w , v ⟫ × inv (norm2 v)) × ⟪ v , u ⟫)
        { inner_left_scale }
      == ((⟪ w , v ⟫ × inv (norm2 v)) × 0)
        { (orthogonal_eq v u (orthogonal_sym u v Huv)) }
      == (0 : F)
        { mul_0_r }.
  Qed.

  Lemma proj_self_zero : ∀ v w : V, orthogonal w v → proj v w == 0.
  Proof.
    intros v w Hw.
    transitivity ((0 : F) *: v).
    - calc (proj v w)
        == ((⟪ w , v ⟫ × inv (norm2 v)) *: v) { proj_eq }
        == ((0 × inv (norm2 v)) *: v) { orthogonal_eq }
        == ((0 : F) *: v) { mul_0_l }.
    - apply zero_scale.
  Qed.

End InnerKit.

Section PosInnerGeometry.
  Context {F V : Type} `{PosInner F V}.
  Context {Vp : HasProj V} {PJ : Proj F V}.

  Theorem norm2_nonneg : ∀ u : V, 0 ≤ norm2 u.
  Proof.
    intros u.
    exact (nonneg u).
  Qed.

  Theorem norm2_neq_0_iff : ∀ v : V, norm2 v =/= 0 ↔ v =/= 0.
  Proof.
    intros v.
    split.
    - intros Hn Hv.
      apply Hn.
      rewrite (norm2_spec v), Hv.
      apply inner_left_zero.
    - intros Hv Hn.
      apply Hv.
      apply definite.
      exact Hn.
  Qed.

  Theorem cauchy_schwarz_padded : ∀ (u v : V) (e : F), 0 < e →
    ⟪ u , v ⟫ × ⟪ u , v ⟫ ≤ (norm2 v + e) × norm2 u.
  Proof.
    intros u v e He.
    apply (mul_le_cancel_l (⟪ u , v ⟫ × ⟪ u , v ⟫)
      ((norm2 v + e) × norm2 u) (norm2 v + e)
      (add_nonneg_pos (norm2 v) e (norm2_nonneg v) He)).
    apply (add_le_cancel_r
      ((norm2 v + e) × (⟪ u , v ⟫ × ⟪ u , v ⟫))
      ((norm2 v + e) × ((norm2 v + e) × norm2 u))
      ((norm2 v + e) × (⟪ u , v ⟫ × ⟪ u , v ⟫))).
    transitivity (((norm2 v + e) × (norm2 v + e)) × norm2 u
        + (⟪ u , v ⟫ × ⟪ u , v ⟫) × norm2 v).
    - apply le_of_sub_nonneg.
      rewrite (neg_add ((norm2 v + e) × (⟪ u , v ⟫ × ⟪ u , v ⟫))
                       ((norm2 v + e) × (⟪ u , v ⟫ × ⟪ u , v ⟫))),
        <- (norm2_scale_sub_expand (norm2 v + e) (⟪ u , v ⟫) u v).
      apply norm2_nonneg.
    - apply add_le_mono.
      + apply eq_le.
        apply (associative (norm2 v + e) (norm2 v + e) (norm2 u)).
      + rewrite (mul_commutative (norm2 v + e) (⟪ u , v ⟫ × ⟪ u , v ⟫)).
        apply mul_le_mono_l.
        × apply sq_nonneg.
        × apply le_add_nonneg_r. apply lt_le. exact He.
  Qed.

  Theorem cauchy_schwarz : ∀ u v : V,
    ⟪ u , v ⟫ × ⟪ u , v ⟫ ≤ norm2 u × norm2 v.
  Proof.
    intros u v.
    apply (le_of_le_add_all_eps_mul (⟪ u , v ⟫ × ⟪ u , v ⟫)
      (norm2 u × norm2 v) (norm2 u) (norm2_nonneg u)).
    intros e He.
    rewrite (mul_commutative (norm2 u) (norm2 v)),
      <- (right_distributive (norm2 v) e (norm2 u)).
    exact (cauchy_schwarz_padded u v e He).
  Qed.

  Theorem best_approx : ∀ (u v : V) (c : F), norm2 v =/= 0 →
    norm2 (u + - proj v u) ≤ norm2 (u + - (c *: v)).
  Proof.
    intros u v c Hv.
    assert (Hsplit : norm2 (u + - (c *: v)) ==
      norm2 (u + - proj v u)
        + norm2 ((⟪ u , v ⟫ × inv (norm2 v) + - c) *: v)).
    { calc (norm2 (u + - (c *: v)))
        == (⟪ u + - (c *: v) , u + - (c *: v) ⟫)
          { norm2_spec }
        == (⟪ (u + - proj v u) + (proj v u + - (c *: v)) ,
              (u + - proj v u) + (proj v u + - (c *: v)) ⟫)
          { add_neg_telescope }
        == (⟪ (u + - proj v u) + ((⟪ u , v ⟫ × inv (norm2 v) + - c) *: v) ,
              (u + - proj v u) + ((⟪ u , v ⟫ × inv (norm2 v) + - c) *: v) ⟫)
          { proj_sub_scale }
        == (norm2
              ((u + - proj v u) + ((⟪ u , v ⟫ × inv (norm2 v) + - c) *: v)))
          { norm2_spec }
        == (norm2 (u + - proj v u)
              + norm2 ((⟪ u , v ⟫ × inv (norm2 v) + - c) *: v))
          { (pythagoras (u + - proj v u)
              ((⟪ u , v ⟫ × inv (norm2 v) + - c) *: v)
              (orthogonal_scale_r (u + - proj v u) v
                (⟪ u , v ⟫ × inv (norm2 v) + - c) (proj_orthogonal u v Hv))) }. }
    rewrite Hsplit.
    apply le_add_nonneg_r.
    apply norm2_nonneg.
  Qed.

  Theorem inner_nondegenerate : ∀ x : V,
    (∀ w : V, ⟪ w , x ⟫ == 0) → x == 0.
  Proof.
    intros x Hx.
    apply definite.
    exact (Hx x).
  Qed.

  Theorem adjoint_unique : ∀ f g1 g2 : LinearMap F V V,
    (∀ u w : V, ⟪ LinearMap.map g1 u , w ⟫ == ⟪ u , LinearMap.map f w ⟫) →
    (∀ u w : V, ⟪ LinearMap.map g2 u , w ⟫ == ⟪ u , LinearMap.map f w ⟫) →
    ∀ u : V, LinearMap.map g1 u == LinearMap.map g2 u.
  Proof.
    intros f g1 g2 Hg1 Hg2 u.
    apply (sub_eq_zero (LinearMap.map g1 u) (LinearMap.map g2 u)).
    apply inner_nondegenerate.
    intros w.
    transitivity (⟪ u , LinearMap.map f w ⟫ + - ⟪ u , LinearMap.map f w ⟫).
    - calc (⟪ w , LinearMap.map g1 u + - LinearMap.map g2 u ⟫)
        == (⟪ w , LinearMap.map g1 u ⟫ + ⟪ w , - LinearMap.map g2 u ⟫)
          { inner_right_additive }
        == (⟪ w , LinearMap.map g1 u ⟫ + - ⟪ w , LinearMap.map g2 u ⟫)
          { inner_right_neg }
        == (⟪ LinearMap.map g1 u , w ⟫ + - ⟪ w , LinearMap.map g2 u ⟫)
          { (commutative w) }
        == (⟪ LinearMap.map g1 u , w ⟫ + - ⟪ LinearMap.map g2 u , w ⟫)
          { (commutative w) }
        == (⟪ u , LinearMap.map f w ⟫ + - ⟪ LinearMap.map g2 u , w ⟫)
          { Hg1 }
        == (⟪ u , LinearMap.map f w ⟫ + - ⟪ u , LinearMap.map f w ⟫)
          { Hg2 }.
    - exact (right_inverse _).
  Qed.
