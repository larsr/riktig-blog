Section GATheory.
  Context {F V A : Type} `{GeometricAlgebra F V A}.

  Theorem vec_zero : ga_vec 0 == (0 : A).
  Proof.
    exact (hom_zero ga_vec (fun x y ⇒ additive (opA := add) (opB := add) (f := ga_vec) x y)).
  Qed.

  Lemma contract_add : ∀ u v : V,
    (Q u *: (1 : A) + Q v *: 1)
      + (ga_vec u × ga_vec v + ga_vec v × ga_vec u)
    == Q (u + v) *: 1.
  Proof.
    intros u v.
    calc (((Q u *: (1 : A) + Q v *: (1 : A))
            + (ga_vec u × ga_vec v + ga_vec v × ga_vec u)) : A)
      == (((ga_vec u × ga_vec u + Q v *: (1 : A))
            + (ga_vec u × ga_vec v + ga_vec v × ga_vec u)) : A)
        { (contraction u) }
      == (((ga_vec u × ga_vec u + ga_vec v × ga_vec v)
            + (ga_vec u × ga_vec v + ga_vec v × ga_vec u)) : A)
        { (contraction v) }
      == (((ga_vec u × ga_vec u + ga_vec v × ga_vec u)
            + (ga_vec v × ga_vec v + ga_vec u × ga_vec v)) : A)
        { (add_rotate (ga_vec u × ga_vec u) (ga_vec v × ga_vec v)
             (ga_vec u × ga_vec v) (ga_vec v × ga_vec u)) }
      == (((ga_vec u × ga_vec u + ga_vec v × ga_vec u)
            + (ga_vec u × ga_vec v + ga_vec v × ga_vec v)) : A)
        { (commutative (ga_vec v × ga_vec v) (ga_vec u × ga_vec v)) }
      == (((ga_vec u + ga_vec v) × (ga_vec u + ga_vec v)) : A)
        { (mul_add_expand (ga_vec u) (ga_vec v) (ga_vec u) (ga_vec v)) }
      == ((ga_vec (u + v) × ga_vec (u + v)) : A)
        { ((fun x y ⇒ additive (opA := add) (opB := add) (f := ga_vec) x y) u v) }
      == ((Q (u + v) *: (1 : A)) : A)
        { (contraction (u + v)) }.
  Qed.

  Lemma polar_recovers : ∀ u v : V, (Q u + Q v) + polar Q u v == Q (u + v).
  Proof.
    intros u v. unfold polar.
    rewrite <- (neg_add_assoc (Q (u + v)) (Q u) (Q v)).
    exact (add_sub_cancel (Q u + Q v) (Q (u + v))).
  Qed.

  Theorem polarization : ∀ u v : V,
    ga_vec u × ga_vec v + ga_vec v × ga_vec u == polar Q u v *: (1 : A).
  Proof.
    intros u v.
    apply (cancel_l (Q u *: (1 : A) + Q v *: 1)).
    calc (((Q u *: (1 : A) + Q v *: 1)
            + (ga_vec u × ga_vec v + ga_vec v × ga_vec u)) : A)
      == ((Q (u + v) *: (1 : A)) : A)
        { (contract_add u v) }
      == ((((Q u + Q v) + polar Q u v) *: (1 : A)) : A)
        { (polar_recovers u v) }
      == (((Q u + Q v) *: (1 : A) + polar Q u v *: (1 : A)) : A)
        { (right_distributive (Q u + Q v) (polar Q u v) (1 : A)) }
      == (((Q u *: (1 : A) + Q v *: (1 : A)) + polar Q u v *: (1 : A)) : A)
        { (right_distributive (Q u) (Q v) (1 : A)) }.
  Qed.

End GATheory.
