Section LincombDelta.
  Context {F V : Type} `{FF : Field F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Lemma lincomb_delta_sum : ∀ (k : nat) (b : Fin.t k → V)
      (j : Fin.t k),
    bigopF (fun i ⇒ e j i *: b i) == b j.
  Proof.
    intros k b j.
    calc (bigopF (fun i ⇒ e j i *: b i))
      == (lincomb b (e j)) { (symmetry (lincomb_bigopF b (e j))) }
      == (b j) { (lincomb_delta b j (leibniz_proper b)) }.
  Qed.

  Lemma lincomb_double : ∀ (k l : nat) (b : Fin.t k → V)
      (M : Fin.t k → Fin.t l → F) (c : Fin.t l → F),
    bigopF (fun j ⇒ c j *: bigopF (fun i ⇒ M i j *: b i))
      == lincomb b (fun i ⇒ bigopF (fun j ⇒ M i j × c j)).
  Proof.
    intros k l b M c.
    calc (bigopF (fun j ⇒ c j *: bigopF (fun i ⇒ M i j *: b i)))
      == (bigopF (fun j ⇒ bigopF (fun i ⇒ c j *: (M i j *: b i))))
      == (bigopF (fun i ⇒ bigopF (fun j ⇒ M i j × c j) *: b i))
      == (lincomb b (fun i ⇒ bigopF (fun j ⇒ M i j × c j))).
    - apply bigopF_ext. intros j.
      symmetry.
      exact (bigopF_scale k (c j) (fun i ⇒ M i j *: b i)).
    - rewrite (bigopF_swap l k (fun j i ⇒ c j *: (M i j *: b i))).
      apply bigopF_ext. intros i.
      calc (bigopF (fun j ⇒ c j *: (M i j *: b i)))
        == (bigopF (fun j ⇒ (M i j × c j) *: b i))
        == (bigopF (fun j ⇒ M i j × c j) *: b i).
      + apply bigopF_ext. intros j.
        exact (scale_flip (c j) (M i j) (b i)).
      + exact (bigopF_scale_r l (fun j ⇒ M i j × c j) (b i)).
    - symmetry.
      exact (lincomb_bigopF b (fun i ⇒ bigopF (fun j ⇒ M i j × c j))).
  Qed.

End LincombDelta.

Section LinearMapSums.
  Context {F U V : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context (f : LinearMap F U V).

  Lemma lm_bigop_list : ∀ {J : Type} (g : J → U) (l : list J),
    f (bigop_list g l) == bigop_list (fun j ⇒ f (g j)) l.
  Proof.
    intros J g l; induction l as [| a l IH].
    - exact (linear_map_zero f).
    - calc (f (bigop_list g (a :: l)))
        == (f (g a) + f (bigop_list g l))
             { (linear_map_additive f (g a) (bigop_list g l)) }
        == (f (g a) + bigop_list (fun j ⇒ f (g j)) l)
             { IH }
        == (bigop_list (fun j ⇒ f (g j)) (a :: l))
             { bigop_list_cons }.
  Qed.

  Theorem lm_bigopI : ∀ {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
      {IFin : FinIndex n I} (g : I → U),
    f (bigopI g) == bigopI (fun i ⇒ f (g i)).
  Proof.
    intros n I eI Ieq IFin g.
    exact (lm_bigop_list g enum).
  Qed.

  Theorem lm_bigopF : ∀ (k : nat) (g : Fin.t k → U),
    f (bigopF g) == bigopF (fun i ⇒ f (g i)).
  Proof.
    intros k g.
    calc (f (bigopF g))
      == (f (bigopI g)) { (linear_map_proper f _ _ (bigopF_bigopI k g)) }
      == (bigopI (fun i ⇒ f (g i))) { lm_bigopI }
      == (bigopF (fun i ⇒ f (g i))) { bigopF_bigopI }.
  Qed.

  Theorem lm_lincomb : ∀ {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
      {IFin : FinIndex n I} (b : I → U) (c : I → F),
    f (lincomb b c) == lincomb (fun i ⇒ f (b i)) c.
  Proof.
    intros n I eI Ieq IFin b c.
    calc (f (lincomb b c))
      == (bigopI (fun i ⇒ f (c i *: b i)))
           { lm_bigopI }
      == (lincomb (fun i ⇒ f (b i)) c)
           { (bigopI_ext (fun i ⇒ f (c i *: b i))
                (fun i ⇒ c i *: f (b i))
                (fun i ⇒ linear_map_scale f (c i) (b i))) }.
  Qed.

End LinearMapSums.

Section LinearMapOnBasis.
  Context {F U V : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {k : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex k I}.
  Context {delta : I → I → F} {FdL : Delta I F delta}.
  Context (B : Basis k I F U) (f : LinearMap F U V).

A linear map vanishing on a basis vanishes on the whole space.
  Theorem linear_map_zero_on_basis :
    (∀ i : I, f (Basis.vec B i) == 0) → ∀ u : U, f u == 0.
  Proof.
    intros Hbasis u.
    calc (f u)
      == (f (lincomb (Basis.vec B) (Basis.coords B u)))
      == (lincomb (fun i ⇒ f (Basis.vec B i)) (Basis.coords B u))
      == (bigopI (fun i ⇒ Basis.coords B u i *: f (Basis.vec B i)))
      == (bigopI (fun _ : I ⇒ (0 : V)))
      == (0 : V).
    - exact (linear_map_proper f _ _ (symmetry (Basis.reconstruction B u))).
    - exact (lm_lincomb f (Basis.vec B) (Basis.coords B u)).
    - exact (lincomb_def (fun i ⇒ f (Basis.vec B i)) (Basis.coords B u)).
    - apply bigopI_ext. intros i.
      rewrite (Hbasis i).
      exact (scale_zero (Basis.coords B u i)).
    - exact bigopI_zero.
  Qed.

End LinearMapOnBasis.

Section CoordsPointwise.
  Context {F V : Type} `{VV : VectorSpace F V}.
  Context {k : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex k I}.
  Context (B : Basis k I F V).
  Context {delta : I → I → F} {FdL : Delta I F delta}.

  Lemma coords_proper : ∀ u v : V,
    u == v → ∀ i : I, Basis.coords B u i == Basis.coords B v i.
  Proof.
    intros u v Huv i.
    exact (linear_map_proper (Basis.coords B) u v Huv i).
  Qed.

  Lemma coords_add : ∀ (u v : V) (i : I),
    Basis.coords B (u + v) i == Basis.coords B u i + Basis.coords B v i.
  Proof.
    intros u v i.
    exact (linear_map_additive (Basis.coords B) u v i).
  Qed.

  Lemma coords_scale : ∀ (a : F) (u : V) (i : I),
    Basis.coords B (a *: u) i == a × Basis.coords B u i.
  Proof.
    intros a u i.
    exact (linear_map_scale (Basis.coords B) a u i).
  Qed.

  Lemma coords_vec : ∀ j : I,
    Basis.coords B (Basis.vec B j) == (delta j : I → F).
  Proof.
    intros j.
    calc (Basis.coords B (Basis.vec B j))
      == (Basis.coords B (lincomb (Basis.vec B) (delta j)))
      == (delta j : I → F).
    - apply (linear_map_proper (Basis.coords B)).
      symmetry.
      exact (lincomb_delta (Basis.vec B) j (Basis.vec_proper B)).
    - exact (Basis.coords_linear_combination B (delta j)
               (delta_proper j j (reflexivity j))).
  Qed.

End CoordsPointwise.

Section MatRep.
  Context {F U V : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {m n : nat} (BU : Basis n (Fin.t n) F U) (BV : Basis m (Fin.t m) F V).

  Definition mat_of (f : LinearMap F U V) : Matrix m n F :=
    fun i j ⇒ Basis.coords BV (f (Basis.vec BU j)) i.

  Definition hom_of_coeff (A : Matrix m n F) (u : U) : Fin.t m → F :=
    fun i ⇒ bigopF (fun j ⇒ A i j × Basis.coords BU u j).

  Definition hom_of_fun (A : Matrix m n F) (u : U) : V :=
    lincomb (Basis.vec BV) (hom_of_coeff A u).

  Lemma hom_of_coeff_proper : ∀ (A : Matrix m n F) (u u' : U),
    u == u' →
    ∀ i : Fin.t m, hom_of_coeff A u i == hom_of_coeff A u' i.
  Proof.
    intros A u u' Hu i.
    exact (bigopF_ext n (fun j ⇒ A i j × Basis.coords BU u j)
             (fun j ⇒ A i j × Basis.coords BU u' j)
             (fun j ⇒ op_proper (A i j) (A i j) (reflexivity (A i j))
                         (Basis.coords BU u j) (Basis.coords BU u' j)
                         (coords_proper BU u u' Hu j))).
  Qed.

  Lemma hom_of_coeff_add : ∀ (A : Matrix m n F) (u v : U)
      (i : Fin.t m),
    hom_of_coeff A (u + v) i == hom_of_coeff A u i + hom_of_coeff A v i.
  Proof.
    intros A u v i.
    calc (hom_of_coeff A (u + v) i)
      == (bigopF (fun j ⇒ A i j × Basis.coords BU u j
                         + A i j × Basis.coords BU v j))
      == (hom_of_coeff A u i + hom_of_coeff A v i).
    - apply bigopF_ext. intros j.
      rewrite (coords_add BU u v j).
      exact (left_distributive (A i j) (Basis.coords BU u j)
               (Basis.coords BU v j)).
    - exact (bigopF_add n (fun j ⇒ A i j × Basis.coords BU u j)
               (fun j ⇒ A i j × Basis.coords BU v j)).
  Qed.

  Lemma hom_of_coeff_scale : ∀ (A : Matrix m n F) (a : F) (u : U)
      (i : Fin.t m),
    hom_of_coeff A (a *: u) i == a × hom_of_coeff A u i.
  Proof.
    intros A a u i.
    calc (hom_of_coeff A (a *: u) i)
      == (bigopF (fun j ⇒ a × (A i j × Basis.coords BU u j)))
      == (a × hom_of_coeff A u i).
    - apply bigopF_ext. intros j.
      rewrite (coords_scale BU a u j).
      exact (mul_left_commutative (A i j) a (Basis.coords BU u j)).
    - exact (bigopF_mul_l n a (fun j ⇒ A i j × Basis.coords BU u j)).
  Qed.

  #[export] Instance hom_of_fun_proper : ∀ A : Matrix m n F,
    Proper (equ ==> equ) (hom_of_fun A).
  Proof.
    intros A u u' Hu.
    apply (lincomb_proper (Basis.vec BV)
             (hom_of_coeff A u) (hom_of_coeff A u')).
    intros i.
    exact (hom_of_coeff_proper A u u' Hu i).
  Qed.

  Lemma hom_of_fun_add : ∀ (A : Matrix m n F) (u v : U),
    hom_of_fun A (u + v) == hom_of_fun A u + hom_of_fun A v.
  Proof.
    intros A u v.
    calc (hom_of_fun A (u + v))
      == (lincomb (Basis.vec BV) (hom_of_coeff A u + hom_of_coeff A v))
      == (hom_of_fun A u + hom_of_fun A v).
    - apply (lincomb_proper (Basis.vec BV) (hom_of_coeff A (u + v))
               (hom_of_coeff A u + hom_of_coeff A v)).
      intros i.
      exact (hom_of_coeff_add A u v i).
    - exact (lincomb_add (Basis.vec BV) (hom_of_coeff A u)
               (hom_of_coeff A v)).
  Qed.

  Lemma hom_of_fun_scale : ∀ (A : Matrix m n F) (a : F) (u : U),
    hom_of_fun A (a *: u) == a *: hom_of_fun A u.
  Proof.
    intros A a u.
    calc (hom_of_fun A (a *: u))
      == (lincomb (Basis.vec BV) (a *: hom_of_coeff A u))
      == (a *: hom_of_fun A u).
    - apply (lincomb_proper (Basis.vec BV) (hom_of_coeff A (a *: u))
               (a *: hom_of_coeff A u)).
      intros i.
      exact (hom_of_coeff_scale A a u i).
    - exact (lincomb_scale a (Basis.vec BV) (hom_of_coeff A u)).
  Qed.

  Definition hom_of (A : Matrix m n F) : LinearMap F U V :=
    linear_map_from_laws
      (hom_of_fun A)
      (hom_of_fun_proper A)
      (hom_of_fun_add A)
      (hom_of_fun_scale A).

  Lemma hom_expand : ∀ (f : LinearMap F U V) (u : U),
    f u == lincomb (Basis.vec BV) (hom_of_coeff (mat_of f) u).
  Proof.
    intros f u.
    calc (f u)
      == (f (lincomb (Basis.vec BU) (Basis.coords BU u)))
      == (lincomb (fun j ⇒ f (Basis.vec BU j)) (Basis.coords BU u))
      == (bigopF (fun j ⇒ Basis.coords BU u j *: f (Basis.vec BU j)))
      == (bigopF (fun j ⇒ Basis.coords BU u j
            *: bigopF (fun i ⇒ mat_of f i j *: Basis.vec BV i)))
      == (lincomb (Basis.vec BV) (hom_of_coeff (mat_of f) u)).
    - apply (linear_map_proper f).
      symmetry.
      exact (Basis.reconstruction BU u).
    - exact (lm_lincomb f (Basis.vec BU) (Basis.coords BU u)).
    - exact (lincomb_bigopF (fun j ⇒ f (Basis.vec BU j)) (Basis.coords BU u)).
    - apply bigopF_ext. intros j.
      apply scale_proper.
      + reflexivity.
      + calc (f (Basis.vec BU j))
          == (lincomb (Basis.vec BV) (Basis.coords BV (f (Basis.vec BU j))))
               { (symmetry (Basis.reconstruction BV (f (Basis.vec BU j)))) }
          == (bigopF (fun i ⇒ mat_of f i j *: Basis.vec BV i))
               { (lincomb_bigopF (Basis.vec BV)
                   (Basis.coords BV (f (Basis.vec BU j)))) }.
    - exact (lincomb_double m n (Basis.vec BV) (mat_of f)
               (Basis.coords BU u)).
  Qed.

  Theorem mat_of_coords : ∀ (f : LinearMap F U V) (u : U)
      (i : Fin.t m),
    Basis.coords BV (f u) i == hom_of_coeff (mat_of f) u i.
  Proof.
    intros f u i.
    calc (Basis.coords BV (f u) i)
      == (Basis.coords BV
            (lincomb (Basis.vec BV) (hom_of_coeff (mat_of f) u)) i)
      == (hom_of_coeff (mat_of f) u i).
    - exact (linear_map_proper (Basis.coords BV) (f u)
               (lincomb (Basis.vec BV) (hom_of_coeff (mat_of f) u))
               (hom_expand f u) i).
    - exact (Basis.coords_linear_combination BV (hom_of_coeff (mat_of f) u)
               (leibniz_proper (hom_of_coeff (mat_of f) u)) i).
  Qed.

  Theorem hom_of_mat_of : ∀ f : LinearMap F U V,
    hom_of (mat_of f) == f.
  Proof.
    intros f u.
    calc (hom_of (mat_of f) u)
      == (lincomb (Basis.vec BV) (Basis.coords BV (f u)))
      == (f u).
    - apply (lincomb_proper (Basis.vec BV)
               (hom_of_coeff (mat_of f) u) (Basis.coords BV (f u))).
      intros i.
      symmetry.
      exact (mat_of_coords f u i).
    - exact (Basis.reconstruction BV (f u)).
  Qed.

  Lemma hom_of_coeff_vec : ∀ (A : Matrix m n F) (j : Fin.t n)
      (i : Fin.t m),
    hom_of_coeff A (Basis.vec BU j) i == A i j.
  Proof.
    intros A j i.
    calc (hom_of_coeff A (Basis.vec BU j) i)
      == (bigopF (fun j' ⇒ A i j' × e j' j))
      == (A i j).
    - apply bigopF_ext. intros j'.
      apply op_proper.
      + reflexivity.
      + exact (transitivity (coords_vec BU j j') (e_sym n j j')).
    - exact (lincomb_std_delta n (fun j' ⇒ A i j') j).
  Qed.

  Theorem mat_of_hom_of : ∀ A : Matrix m n F,
    mat_of (hom_of A) == A.
  Proof.
    intros A i j.
    calc (mat_of (hom_of A) i j)
      == (Basis.coords BV
            (lincomb (Basis.vec BV) (fun i' : Fin.t m ⇒ A i' j)) i)
      == (A i j).
    - exact (linear_map_proper (Basis.coords BV)
               (hom_of A (Basis.vec BU j))
               (lincomb (Basis.vec BV) (fun i' : Fin.t m ⇒ A i' j))
               (lincomb_proper (Basis.vec BV)
                  (hom_of_coeff A (Basis.vec BU j))
                  (fun i' : Fin.t m ⇒ A i' j)
                  (fun i' ⇒ hom_of_coeff_vec A j i')) i).
    - exact (Basis.coords_linear_combination BV (fun i' : Fin.t m ⇒ A i' j)
               (leibniz_proper (fun i' : Fin.t m ⇒ A i' j)) i).
  Qed.

  #[export] Instance mat_of_proper : Proper (equ ==> equ) mat_of.
  Proof.
    intros f g Hfg i j.
    exact (coords_proper BV (f (Basis.vec BU j)) (g (Basis.vec BU j))
             (Hfg (Basis.vec BU j)) i).
  Qed.

  Theorem mat_of_add : ∀ f g : LinearMap F U V,
    mat_of (f + g) == mat_of f + mat_of g.
  Proof.
    intros f g i j.
    exact (coords_add BV (f (Basis.vec BU j)) (g (Basis.vec BU j)) i).
  Qed.

  Theorem mat_of_scale : ∀ (a : F) (f : LinearMap F U V),
    mat_of (a *: f) == a *: mat_of f.
  Proof.
    intros a f i j.
    exact (coords_scale BV a (f (Basis.vec BU j)) i).
  Qed.

  Lemma hom_of_coeff_matadd : ∀ (A A' : Matrix m n F) (u : U)
      (i : Fin.t m),
    hom_of_coeff (A + A') u i == hom_of_coeff A u i + hom_of_coeff A' u i.
  Proof.
    intros A A' u i.
    calc (hom_of_coeff (A + A') u i)
      == (bigopF (fun j ⇒ A i j × Basis.coords BU u j
                         + A' i j × Basis.coords BU u j))
      == (hom_of_coeff A u i + hom_of_coeff A' u i).
    - apply bigopF_ext. intros j.
      exact (right_distributive (A i j) (A' i j) (Basis.coords BU u j)).
    - exact (bigopF_add n (fun j ⇒ A i j × Basis.coords BU u j)
               (fun j ⇒ A' i j × Basis.coords BU u j)).
  Qed.

  Lemma hom_of_coeff_matscale : ∀ (a : F) (A : Matrix m n F) (u : U)
      (i : Fin.t m),
    hom_of_coeff (a *: A) u i == a × hom_of_coeff A u i.
  Proof.
    intros a A u i.
    calc (hom_of_coeff (a *: A) u i)
      == (bigopF (fun j ⇒ a × (A i j × Basis.coords BU u j)))
      == (a × hom_of_coeff A u i).
    - apply bigopF_ext. intros j.
      exact (associative a (A i j) (Basis.coords BU u j)).
    - exact (bigopF_mul_l n a (fun j ⇒ A i j × Basis.coords BU u j)).
  Qed.

  #[export] Instance hom_of_matproper : Proper (equ ==> equ) hom_of.
  Proof.
    intros A A' HA u.
    apply (lincomb_proper (Basis.vec BV)
             (hom_of_coeff A u) (hom_of_coeff A' u)).
    intros i.
    apply bigopF_ext. intros j.
    exact (op_proper (A i j) (A' i j) (HA i j)
             (Basis.coords BU u j) (Basis.coords BU u j)
             (reflexivity (Basis.coords BU u j))).
  Qed.

  Theorem hom_of_matadd : ∀ A A' : Matrix m n F,
    hom_of (A + A') == hom_of A + hom_of A'.
  Proof.
    intros A A' u.
    calc (hom_of (A + A') u)
      == (lincomb (Basis.vec BV) (hom_of_coeff A u + hom_of_coeff A' u))
      == ((hom_of A + hom_of A') u).
    - apply (lincomb_proper (Basis.vec BV)
               (hom_of_coeff (A + A') u)
               (hom_of_coeff A u + hom_of_coeff A' u)).
      intros i.
      exact (hom_of_coeff_matadd A A' u i).
    - exact (lincomb_add (Basis.vec BV) (hom_of_coeff A u)
               (hom_of_coeff A' u)).
  Qed.

  Theorem hom_of_matscale : ∀ (a : F) (A : Matrix m n F),
    hom_of (a *: A) == a *: hom_of A.
  Proof.
    intros a A u.
    calc (hom_of (a *: A) u)
      == (lincomb (Basis.vec BV) (a *: hom_of_coeff A u))
      == ((a *: hom_of A) u).
    - apply (lincomb_proper (Basis.vec BV)
               (hom_of_coeff (a *: A) u) (a *: hom_of_coeff A u)).
      intros i.
      exact (hom_of_coeff_matscale a A u i).
    - exact (lincomb_scale a (Basis.vec BV) (hom_of_coeff A u)).
  Qed.

  Local Notation HomUV := (LinearMap F U V).

  Let VHom : VectorSpace F HomUV := Hom_VectorSpace.
  Let VMat : VectorSpace F (Matrix m n F) := PW_VectorSpace.

  Definition mat_of_lm : LinearMap F HomUV (Matrix m n F) :=
    linear_map_from_laws
      mat_of
      mat_of_proper
      mat_of_add
      mat_of_scale.

  Definition hom_of_lm : LinearMap F (Matrix m n F) HomUV :=
    linear_map_from_laws
      hom_of
      hom_of_matproper
      hom_of_matadd
      hom_of_matscale.

  Definition vsiso_matrix : VSIso F HomUV (Matrix m n F) :=
    vsiso_from_laws mat_of_lm hom_of_lm hom_of_mat_of mat_of_hom_of.

End MatRep.


Section MatRepComp.
  Context {F U V W : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context {n m p : nat} (BU : Basis n (Fin.t n) F U)
          (BV : Basis m (Fin.t m) F V) (BW : Basis p (Fin.t p) F W).

  Theorem mat_of_comp : ∀ (g : LinearMap F V W) (f : LinearMap F U V),
    mat_of BU BW (linear_map_comp g f)
      == mat_mul (mat_of BV BW g) (mat_of BU BV f).
  Proof.
    intros g f i j.
    calc (mat_of BU BW (linear_map_comp g f) i j)
      == (bigopF (fun l ⇒ mat_of BV BW g i l × mat_of BU BV f l j))
           { (mat_of_coords BV BW g (f (Basis.vec BU j)) i) }
      == (mat_mul (mat_of BV BW g) (mat_of BU BV f) i j)
           { (bigopF_bigopI m
               (fun l ⇒ mat_of BV BW g i l × mat_of BU BV f l j)) }.
  Qed.

End MatRepComp.
