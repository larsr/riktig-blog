  Definition pb (b1 b2 : bool) : Blade 2 := bcons b1 (bcons b2 bnil).

  Lemma blade2_case : ∀ D : Blade 2,
    D = pb (bhead D) (bhead (btail D)).
  Proof.
    intros D.
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    exact (blade_0 (btail (btail D))).
  Qed.

  Definition mk2 (c a b p : F) : Cl F 2 :=
    fun D ⇒
      if bhead D
      then if bhead (btail D) then p else a
      else if bhead (btail D) then b else c.

  Definition rotor (c s : F) : Cl F 2 := mk2 c 0 0 s.

  Definition plane_vector (a b : F) : Fin.t 2 → F :=
    fun j ⇒ match j with Fin.F1 ⇒ a | Fin.FS _ ⇒ b end.

  Definition sandwich (R x : Cl F 2) : Cl F 2 := R × x × cl_reverse R.

  Definition even_to_complex (x : Cl F 2) : C F :=
    mkC (x (pb false false)) (x (pb true true)).

  Definition complex_to_even (z : C F) : Cl F 2 := mk2 (re z) 0 0 (im z).

  Lemma cl2_ext : ∀ x y : Cl F 2,
    x (pb false false) == y (pb false false) →
    x (pb true false) == y (pb true false) →
    x (pb false true) == y (pb false true) →
    x (pb true true) == y (pb true true) →
    x == y.
  Proof.
    intros x y H00 H10 H01 H11 D.
    rewrite (blade2_case D).
    destruct (bhead D), (bhead (btail D)); assumption.
  Qed.

  Ltac blade_reduce :=
    unfold sandwich, rotor, complex_to_even, even_to_complex;
    repeat change (@mult (Cl F 2) Plane_Mult) with Plane_Mult;
    repeat change (@one (Cl F 2) Plane_One) with Plane_One;
    repeat change (@neg (Cl F 2) PW_Neg)
      with (fun (g : Cl F 2) (D : Blade 2) ⇒ - g D);
    unfold Plane_Mult, Plane_One, cl_mul, cl_one, cl_reverse, bsign,
      parity_sign, mk2, pb, bxor, bdiff, band;
    simpl.
  Ltac by_blade := blade_reduce; ring.


  Lemma reverse_mk2 : ∀ c a b p : F,
    cl_reverse (mk2 c a b p) == mk2 c a b (- p).
  Proof. intros c a b p. apply cl2_ext; by_blade. Qed.

  Lemma reverse_anti2 : ∀ x y : Cl F 2,
    cl_reverse (x × y) == cl_reverse y × cl_reverse x.
  Proof. exact reversal_anti. Qed.


  Lemma rotor_is_even : ∀ c s : F, cl_even (rotor c s).
  Proof.
    intros c s D HD.
    rewrite (blade2_case D) in ×.
    destruct (bhead D), (bhead (btail D));
      solve [ discriminate HD | reflexivity ].
  Qed.

  Lemma rotor_two_term : ∀ c s : F,
    rotor c s == c *: (1 : Cl F 2) + s *: rotor 0 1.
  Proof.
    intros c s D.
    rewrite (pw_add_at (c *: (1 : Cl F 2)) (s *: rotor 0 1) D),
            (cl_scale_at 2 c (1 : Cl F 2) D),
            (cl_scale_at 2 s (rotor 0 1) D).
    rewrite (blade2_case D).
    destruct (bhead D), (bhead (btail D)); by_blade.
  Qed.

The bivector squares to - (1), which is what makes the plane it spans with the unit multiply like the complex numbers.
  Theorem plane_bivector_square :
    (rotor (0 : F) 1) × rotor 0 1 == - (1 : Cl F 2).
  Proof. apply cl2_ext; by_blade. Qed.

  Lemma rotor_mul : ∀ c1 s1 c2 s2 : F,
    rotor c1 s1 × rotor c2 s2
    == rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2).
  Proof.
    intros c1 s1 c2 s2.
    calc ((rotor c1 s1 × rotor c2 s2) : Cl F 2)
      == (((c1 *: (1 : Cl F 2) + s1 *: rotor 0 1) × rotor c2 s2) : Cl F 2)
         { (rotor_two_term c1 s1) }
      == (((c1 *: (1 : Cl F 2) + s1 *: rotor 0 1)
             × (c2 *: (1 : Cl F 2) + s2 *: rotor 0 1)) : Cl F 2)
         { (rotor_two_term c2 s2) }
      == (((c1×c2 + - (s1×s2)) *: (1 : Cl F 2)
             + (c1×s2 + s1×c2) *: rotor 0 1) : Cl F 2)
         { (two_term_product c1 s1 c2 s2 (rotor 0 1) plane_bivector_square) }
      == ((rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)) : Cl F 2)
         { (rotor_two_term (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)) }.
  Qed.

  Lemma rotor_reverse : ∀ c s : F,
    cl_reverse (rotor c s) == rotor c (- s).
  Proof. intros c s. exact (reverse_mk2 c 0 0 s). Qed.

  #[export] Instance rotor_proper : Proper (equ ==> equ ==> equ) rotor.
  Proof.
    intros c c' Hc s s' Hs D.
    rewrite (blade2_case D).
    unfold rotor, mk2, pb.
    destruct (bhead D), (bhead (btail D));
      solve [ exact Hc | exact Hs | reflexivity ].
  Qed.

  Lemma rotor_reverse_mul : ∀ c s : F,
    rotor c s × cl_reverse (rotor c s) == rotor (c×c + s×s) 0.
  Proof.
    intros c s.
    assert (Hscalar : c×c + - (s × - s) == c×c + s×s) by ring.
    assert (Hbivector : c × - s + s×c == (0 : F)) by ring.
    calc ((rotor c s × cl_reverse (rotor c s)) : Cl F 2)
      == ((rotor c s × rotor c (- s)) : Cl F 2)
         { rotor_reverse }
      == ((rotor (c×c + - (s × - s)) (c × - s + s×c)) : Cl F 2)
         { rotor_mul }
      == ((rotor (c×c + s×s) 0) : Cl F 2)
         { (rotor_proper _ _ Hscalar _ _ Hbivector) }.
  Qed.

  Lemma rotor_reverse_mul_l : ∀ c s : F,
    cl_reverse (rotor c s) × rotor c s == rotor (c×c + s×s) 0.
  Proof.
    intros c s.
    assert (Hscalar : c×c + - (- s × s) == c×c + s×s) by ring.
    assert (Hbivector : c×s + - s × c == (0 : F)) by ring.
    calc ((cl_reverse (rotor c s) × rotor c s) : Cl F 2)
      == ((rotor c (- s) × rotor c s) : Cl F 2)
         { rotor_reverse }
      == ((rotor (c×c + - (- s × s)) (c×s + - s × c)) : Cl F 2)
         { rotor_mul }
      == ((rotor (c×c + s×s) 0) : Cl F 2)
         { (rotor_proper _ _ Hscalar _ _ Hbivector) }.
  Qed.

  Lemma rotor_one : rotor 1 0 == (1 : Cl F 2).
  Proof. apply cl2_ext; by_blade. Qed.

  Lemma scale_rotor : ∀ c a b : F,
    c *: rotor a b == rotor (c × a) (c × b).
  Proof.
    intros c a b.
    apply cl2_ext; rewrite (cl_scale_at 2 c (rotor a b)); by_blade.
  Qed.

  Lemma rotor_neg_one : rotor (- (1 : F)) 0 == - (1 : Cl F 2).
  Proof.
    assert (Hscalar : (- (1 : F)) × 1 == - (1)) by ring.
    assert (Hbivector : (- (1 : F)) × 0 == 0) by ring.
    calc ((rotor (- (1 : F)) 0) : Cl F 2)
      == ((rotor ((- (1 : F)) × 1) 0) : Cl F 2)
         { Hscalar }
      == ((rotor ((- (1 : F)) × 1) ((- (1 : F)) × 0)) : Cl F 2)
         { Hbivector }
      == (((- (1 : F)) *: rotor 1 0) : Cl F 2)
         { scale_rotor }
      == (((- (1 : F)) *: (1 : Cl F 2)) : Cl F 2)
         { rotor_one }
      == ((- (1 : Cl F 2)) : Cl F 2)
         { neg1_scale }.
  Qed.

  Lemma rotor_unit : ∀ c s : F,
    c×c + s×s == 1 → rotor c s × cl_reverse (rotor c s) == 1.
  Proof.
    intros c s Hcs.
    calc ((rotor c s × cl_reverse (rotor c s)) : Cl F 2)
      == (rotor (c×c + s×s) 0) { rotor_reverse_mul }
      == (rotor 1 0) { Hcs }
      == (1 : Cl F 2) { rotor_one }.
  Qed.

  Lemma rotor_unit_l : ∀ c s : F,
    c×c + s×s == 1 → cl_reverse (rotor c s) × rotor c s == 1.
  Proof.
    intros c s Hcs.
    calc ((cl_reverse (rotor c s) × rotor c s) : Cl F 2)
      == (rotor (c×c + s×s) 0) { rotor_reverse_mul_l }
      == (rotor 1 0) { Hcs }
      == (1 : Cl F 2) { rotor_one }.
  Qed.

  Lemma rotor_unit_necessary : ∀ c s : F,
    rotor c s × cl_reverse (rotor c s) == 1 → c×c + s×s == 1.
  Proof.
    intros c s Hru.
    calc (c×c + s×s)
      == ((rotor c s × cl_reverse (rotor c s)) (pb false false))
         { rotor_reverse_mul }
      == ((1 : Cl F 2) (pb false false))
         { Hru }
      == (1 : F)
         { reflexivity }.
  Qed.

  Lemma rotor_unit_iff : ∀ c s : F,
    rotor c s × cl_reverse (rotor c s) == 1 ↔ c×c + s×s == 1.
  Proof.
    intros c s.
    split; [ exact (rotor_unit_necessary c s) | exact (rotor_unit c s) ].
  Qed.

rotor c s meets Structure/Rotor.v's definition exactly when its two coefficients lie on the unit circle. This is the bridge the rest of the file crosses: below it, the coordinate formula; above it, the three equations, and every general rotor theorem.
  Definition rotor_is_rotor (c s : F) (Hcs : c×c + s×s == 1)
    : Rotor cl_reverse cl_even (rotor c s) :=
    
    {| Rotor.rotor_even := rotor_is_even c s;
       Rotor.rotor_unit_r := rotor_unit c s Hcs;
       Rotor.rotor_unit_l := rotor_unit_l c s Hcs; |}.


  Lemma vec_plane : ∀ v1 v2 : F,
    ga_vec (plane_vector v1 v2) == mk2 0 v1 v2 0.
  Proof. intros v1 v2. apply cl2_ext; by_blade. Qed.

  Lemma sandwich_rotor_vec : ∀ c s v1 v2 : F,
    sandwich (rotor c s) (ga_vec (plane_vector v1 v2))
    == ga_vec (plane_vector
         ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
         (- ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2)).
  Proof. intros. apply cl2_ext; by_blade. Qed.

  Lemma sandwich_compose : ∀ A B x : Cl F 2,
    sandwich A (sandwich B x) == sandwich (A × B) x.
  Proof.
    intros A B x. unfold sandwich.
    exact (rotor_sandwich_compose cl_reverse A B x).
  Qed.

  Lemma qform_plane : ∀ v1 v2 : F,
    qform sig2 (plane_vector v1 v2) == v1×v1 + v2×v2.
  Proof.
    intros v1 v2.
    rewrite (qform_S 1 sig2 (plane_vector v1 v2)),
            (qform_S 0 (Vector.tl sig2) (fun i ⇒ plane_vector v1 v2 (Fin.FS i))),
            (qform_nil (Vector.tl (Vector.tl sig2)) _).
    unfold sig2, plane_vector. simpl. ring.
  Qed.

  Lemma plane_one_faithful : one_faithful F (Cl F 2).
  Proof.
    intros a b Hab.
    assert (Hblade : a × 1 == b × 1) by exact (Hab (pb false false)).
    calc a
      == (a × (1 : F)) { mul_right_identity }
      == (b × (1 : F)) { Hblade }
      == (b : F) { mul_right_identity }.
  Qed.

  Lemma rotation_preserves_qform : ∀ c s v1 v2 : F,
    c×c + s×s == 1 →
    qform sig2 (plane_vector
        ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
        (- ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2))
    == qform sig2 (plane_vector v1 v2).
  Proof.
    intros c s v1 v2 Hcs.
    exact (rotor_preserves_qform cl_reverse cl_even (rotor c s)
             (HR := rotor_is_rotor c s Hcs)
             (plane_vector v1 v2)
             (plane_vector ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
                           (- ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2))
             plane_one_faithful
             (symmetry (sandwich_rotor_vec c s v1 v2))).
  Qed.


  Lemma even_round_trip : ∀ x : Cl F 2,
    cl_even x → complex_to_even (even_to_complex x) == x.

    intros x Hx.

  Lemma even_to_complex_mul : ∀ x y : Cl F 2,
    cl_even x → cl_even y →
    even_to_complex (x × y) == even_to_complex x × even_to_complex y.
  Proof.
    intros x y Hx Hy.
    pose proof (Hx (pb true false) eq_refl) as Hx1.
    pose proof (Hx (pb false true) eq_refl) as Hx2.
    pose proof (Hy (pb true false) eq_refl) as Hy1.
    pose proof (Hy (pb false true) eq_refl) as Hy2.
    unfold pb in Hx1, Hx2, Hy1, Hy2.
    split; blade_reduce; rewrite Hx1, Hx2, Hy1, Hy2; ring.
  Qed.

  Theorem rotor_composition_is_u1 : ∀ c1 s1 c2 s2 : F,
    even_to_complex (rotor c1 s1) × even_to_complex (rotor c2 s2)
    == even_to_complex (rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)).
  Proof.
    intros c1 s1 c2 s2.
    calc (even_to_complex (rotor c1 s1) × even_to_complex (rotor c2 s2))
      == (even_to_complex (rotor c1 s1 × rotor c2 s2))
         { (even_to_complex_mul (rotor c1 s1) (rotor c2 s2)
              (rotor_is_even c1 s1) (rotor_is_even c2 s2)) }
      == (even_to_complex (rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)))
         { rotor_mul }.
  Qed.


The unit and the bivector are different blades, so no combination of them vanishes unless both coefficients do.
  Theorem plane_independent_from_one :
    independent_from_one F (rotor (0 : F) 1).
  Proof.
    intros a b Hab.
    split.
    - calc a
        == [scalar_blade] (a × 1 + b × 0)
        == (0 : F) { (Hab (pb false false)) }.
      [scalar_blade]: ring.
    - calc b
        == [bivector_blade] (a × 0 + b × 1)
        == (0 : F) { (Hab (pb true true)) }.
      [bivector_blade]: ring.
  Qed.

Reversal negates the bivector — the second hypothesis the angle theorems ask of the plane.
  Theorem plane_reverse_bivector :
    cl_reverse (rotor (0 : F) 1) == - rotor 0 1.
  Proof. apply cl2_ext; by_blade. Qed.

Reversal commutes with scaling: it multiplies each blade coefficient by a sign, and the field is commutative.
  Theorem plane_reverse_scale : ∀ (a : F) (x : Cl F 2),
    cl_reverse (a *: x) == a *: cl_reverse x.
  Proof. exact (cl_reverse_scale 2). Qed.

Reversal fixes a vector: grade one is the one grade whose sign it leaves alone.
  Theorem plane_vector_reverse : ∀ v1 v2 : F,
    cl_reverse (ga_vec (plane_vector v1 v2)) == ga_vec (plane_vector v1 v2).
  Proof. intros v1 v2. apply cl2_ext; by_blade. Qed.

So the two orders of a product of two vectors are each other's reversal — reversal reverses the product and fixes both factors.
  Theorem vector_product_reverse : ∀ u1 u2 v1 v2 : F,
    ga_vec (plane_vector v1 v2) × ga_vec (plane_vector u1 u2)
    == cl_reverse (ga_vec (plane_vector u1 u2) × ga_vec (plane_vector v1 v2)).
  Proof.
    intros u1 u2 v1 v2.
    calc ((ga_vec (plane_vector v1 v2) × ga_vec (plane_vector u1 u2))
          : Cl F 2)
      == ((cl_reverse (ga_vec (plane_vector v1 v2))
             × ga_vec (plane_vector u1 u2)) : Cl F 2)
         { (plane_vector_reverse v1 v2) }
      == ((cl_reverse (ga_vec (plane_vector v1 v2))
             × cl_reverse (ga_vec (plane_vector u1 u2))) : Cl F 2)
         { (plane_vector_reverse u1 u2) }
      == ((cl_reverse (ga_vec (plane_vector u1 u2)
                         × ga_vec (plane_vector v1 v2))) : Cl F 2)
         { reverse_anti2 }.
  Qed.
