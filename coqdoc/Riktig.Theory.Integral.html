Section IntegralTheory.
  Context {F V : Type} `{Integral F V}.

  Theorem integral_empty : ∀ (f : F → V) (a : F), int f a a == 0.
  Proof.
    intros f a.
    apply (cancel_l (int f a a)).
    rewrite Integral.additivity, right_identity.
    reflexivity.
  Qed.

  Theorem integral_rev : ∀ (f : F → V) (a b : F),
    int f a b == - int f b a.
  Proof.
    intros f a b.
    apply neg_unique_l.
    rewrite Integral.additivity.
    apply integral_empty.
  Qed.

  Theorem integral_zero_fn : ∀ a b : F,
    int (fun _ : F ⇒ (0 : V)) a b == 0.
  Proof.
    intros a b.
    rewrite (Integral.constant 0 a b).
    apply scale_zero.
  Qed.

End IntegralTheory.
