Record Joint2 (F : Type) := mkJoint2 {
  joint2_c : F;
  joint2_s : F;
  joint2_length : F
}.

Arguments mkJoint2 {F} joint2_c joint2_s joint2_length.
Arguments joint2_c {F} j.
Arguments joint2_s {F} j.
Arguments joint2_length {F} j.

Definition Chain2 (F : Type) : Type := list (Joint2 F).

Section Chain2Ops.
  Context {F : Type} `{OF : OrderedField F}.

  Definition beam_rest_direction2 : Cl2 F := Cl2.vec (plane_vector 1 0).

  Definition chain_direction2 (R : Cl2 F) : Cl2 F :=
    cl2_sandwich R beam_rest_direction2.

  Lemma chain_direction2_rotor : ∀ c s : F,
    chain_direction2 (rotor c s)
    == Cl2.vec (plane_vector ((c×c + - (s×s)) × 1 + (c×s + c×s) × 0)
                    (- ((c×s + c×s) × 1) + (c×c + - (s×s)) × 0)).
  Proof.
    intros c s. unfold chain_direction2, beam_rest_direction2.
    exact (cl2_sandwich_rotor_vec c s 1 0).
  Qed.

  Definition chain_rotor2 (joints : Chain2 F) : Cl2 F :=
    fold_right (fun j acc ⇒ rotor (joint2_c j) (joint2_s j) × acc) 1 joints.

  Lemma chain_rotor2_nil : chain_rotor2 [] = (1 : Cl2 F).
  Proof. reflexivity. Qed.

  Lemma chain_rotor2_cons : ∀ (j : Joint2 F) (rest : Chain2 F),
    chain_rotor2 (j :: rest) = rotor (joint2_c j) (joint2_s j) × chain_rotor2 rest.
  Proof. reflexivity. Qed.

  Lemma chain_rotor2_is_rotor : ∀ joints : Chain2 F,
    ∃ c s : F, chain_rotor2 joints == rotor c s.
  Proof.
    induction joints as [| j rest [c [s IH]]].
    - ∃ 1, 0. rewrite chain_rotor2_nil. reflexivity.
    - ∃ (joint2_c j × c + - (joint2_s j × s)), (joint2_c j × s + joint2_s j × c).
      calc (chain_rotor2 (j :: rest))
        == (rotor (joint2_c j) (joint2_s j) × chain_rotor2 rest)
        == (rotor (joint2_c j) (joint2_s j) × rotor c s)
        == (rotor (joint2_c j × c + - (joint2_s j × s)) (joint2_c j × s + joint2_s j × c)).
      + rewrite chain_rotor2_cons. reflexivity.
      + rewrite IH. reflexivity.
      + exact (rotor_mul (joint2_c j) (joint2_s j) c s).
  Qed.

  Fixpoint chain_orientation_from2 (orientation : Cl2 F) (joints : Chain2 F) : Cl2 F :=
    match joints with
    | [] ⇒ orientation
    | j :: rest ⇒
        chain_orientation_from2 (orientation × rotor (joint2_c j) (joint2_s j)) rest
    end.

  Lemma chain_orientation_from2_nil : ∀ orientation : Cl2 F,
    chain_orientation_from2 orientation [] = orientation.
  Proof. reflexivity. Qed.

  Lemma chain_orientation_from2_cons : ∀ (orientation : Cl2 F)
      (j : Joint2 F) (rest : Chain2 F),
    chain_orientation_from2 orientation (j :: rest)
    = chain_orientation_from2 (orientation × rotor (joint2_c j) (joint2_s j)) rest.
  Proof. reflexivity. Qed.

  Theorem chain_rotor_is_product2 : ∀ (joints : Chain2 F) (orientation : Cl2 F),
    chain_orientation_from2 orientation joints == orientation × chain_rotor2 joints.
  Proof.
    induction joints as [| j rest IH]; intros orientation.
    - calc (chain_orientation_from2 orientation [])
        == orientation
        == (orientation × 1)
        == (orientation × chain_rotor2 []).
      + rewrite (chain_orientation_from2_nil orientation). reflexivity.
      + symmetry. exact (right_identity orientation).
      + rewrite chain_rotor2_nil. reflexivity.
    - calc (chain_orientation_from2 orientation (j :: rest))
        == (chain_orientation_from2 (orientation × rotor (joint2_c j) (joint2_s j)) rest)
        == ((orientation × rotor (joint2_c j) (joint2_s j)) × chain_rotor2 rest)
        == (orientation × (rotor (joint2_c j) (joint2_s j) × chain_rotor2 rest))
        == (orientation × chain_rotor2 (j :: rest)).
      + rewrite (chain_orientation_from2_cons orientation j rest). reflexivity.
      + exact (IH (orientation × rotor (joint2_c j) (joint2_s j))).
      + exact (associative orientation (rotor (joint2_c j) (joint2_s j))
                 (chain_rotor2 rest)).
      + rewrite chain_rotor2_cons. reflexivity.
  Qed.

  Corollary chain_rotor_is_product2_from_identity : ∀ joints : Chain2 F,
    chain_orientation_from2 1 joints == chain_rotor2 joints.
  Proof.
    intros joints. rewrite (chain_rotor_is_product2 joints 1).
    exact (left_identity (chain_rotor2 joints)).
  Qed.

  Fixpoint chain_positions2_from (orientation position : Cl2 F)
           (joints : Chain2 F) : list (Cl2 F) :=
    match joints with
    | [] ⇒ []
    | j :: rest ⇒
        let orientation' := orientation × rotor (joint2_c j) (joint2_s j) in
        let position' := position + joint2_length j *: chain_direction2 orientation' in
        position' :: chain_positions2_from orientation' position' rest
    end.

  Lemma chain_positions2_from_nil : ∀ orientation position : Cl2 F,
    chain_positions2_from orientation position [] = [].
  Proof. reflexivity. Qed.

  Lemma chain_positions2_from_cons : ∀ (orientation position : Cl2 F)
      (j : Joint2 F) (rest : Chain2 F),
    chain_positions2_from orientation position (j :: rest)
    = let orientation' := orientation × rotor (joint2_c j) (joint2_s j) in
      let position' := position + joint2_length j *: chain_direction2 orientation' in
      position' :: chain_positions2_from orientation' position' rest.
  Proof. reflexivity. Qed.

  Definition chain_positions2 (base : Cl2 F) (joints : Chain2 F) : list (Cl2 F) :=
    chain_positions2_from 1 base joints.

  Lemma cl2_sandwich_compose : ∀ A B x : Cl2 F,
    cl2_sandwich A (cl2_sandwich B x) == cl2_sandwich (A × B) x.
  Proof.
    intros A B x. unfold cl2_sandwich.
    calc (A × (B × x × cl2_reverse B) × cl2_reverse A)
      == (A × (B × x) × cl2_reverse B × cl2_reverse A)
        { (Cl2.mult_associative A (B × x) (cl2_reverse B)) }
      == (A × B × x × cl2_reverse B × cl2_reverse A)
        { (Cl2.mult_associative A B x) }
      == (A × B × x × (cl2_reverse B × cl2_reverse A))
        { (Cl2.mult_associative (A × B × x) (cl2_reverse B) (cl2_reverse A)) }
      == (A × B × x × cl2_reverse (A × B))
        { (cl2_reverse_anti A B) }.
  Qed.

End Chain2Ops.

Record Joint3 (F : Type) := mkJoint3 {
  joint3_c : F;
  joint3_b12 : F;
  joint3_b13 : F;
  joint3_b23 : F;
  joint3_length : F
}.

Arguments mkJoint3 {F} joint3_c joint3_b12 joint3_b13 joint3_b23 joint3_length.
Arguments joint3_c {F} j.
Arguments joint3_b12 {F} j.
Arguments joint3_b13 {F} j.
Arguments joint3_b23 {F} j.
Arguments joint3_length {F} j.

Definition Chain3 (F : Type) : Type := list (Joint3 F).

Section Chain3Ops.
  Context {F : Type} `{FF : Field F}.

  Definition space_vector {A : Type} (a b c : A) : Fin.t 3 → A :=
    fun i ⇒ match i with
             | Fin.F1 ⇒ a
             | Fin.FS Fin.F1 ⇒ b
             | Fin.FS (Fin.FS _) ⇒ c
             end.

  Definition rotor3 (c b12 b13 b23 : F) : C3 F := mkC3 c 0 0 0 b12 b13 b23 0.

  Lemma rotor3_even : ∀ c b12 b13 b23 : F, c3_even (rotor3 c b12 b13 b23).
  Proof. intros. repeat split; reflexivity. Qed.

  Definition joint3_rotor (j : Joint3 F) : C3 F :=
    rotor3 (joint3_c j) (joint3_b12 j) (joint3_b13 j) (joint3_b23 j).

  Definition c3_rotate_coordinates (r : C3 F) (v : Fin.t 3 → F) : Fin.t 3 → F :=
    fun i : Fin.t 3 ⇒
      match i with
      | Fin.F1 ⇒ c3_1 (c3_rotate r (c3_vec v))
      | Fin.FS Fin.F1 ⇒ c3_2 (c3_rotate r (c3_vec v))
      | Fin.FS (Fin.FS _) ⇒ c3_3 (c3_rotate r (c3_vec v))
      end.

  Lemma c3_rotate_coordinates_correct : ∀ (r : C3 F) (v : Fin.t 3 → F),
    c3_even r → c3_rotate r (c3_vec v) == c3_vec (c3_rotate_coordinates r v).
  Proof.
    intros r v Hr.
    assert (Hcodd : c3_codd (c3_rotate r (c3_vec v))).
    { unfold c3_rotate.
      apply c3_codd_even_mul_closed.
      - apply c3_even_codd_mul_closed.
        + exact Hr.
        + exact (c3_vec_codd v).
      - exact (c3_reverse_even r Hr). }
    destruct Hcodd as [Hs [H12 [H13 H23]]].
    unfold c3_rotate_coordinates.
    repeat split.
    - exact Hs.
    - reflexivity.
    - reflexivity.
    - reflexivity.
    - exact H12.
    - exact H13.
    - exact H23.
    - exact (c3_rotate_trivector_vanish r v Hr).
  Qed.

  Definition beam_rest_direction3 : Fin.t 3 → F := space_vector 1 0 0.

  Definition chain_direction3 (R : C3 F) : C3 F :=
    c3_vec (c3_rotate_coordinates R beam_rest_direction3).

  Definition chain_rotor3 (joints : Chain3 F) : C3 F :=
    fold_right (fun j acc ⇒ joint3_rotor j × acc) 1 joints.

  Lemma chain_rotor3_nil : chain_rotor3 [] = (1 : C3 F).
  Proof. reflexivity. Qed.

  Lemma chain_rotor3_cons : ∀ (j : Joint3 F) (rest : Chain3 F),
    chain_rotor3 (j :: rest) = joint3_rotor j × chain_rotor3 rest.
  Proof. reflexivity. Qed.

  Lemma chain_rotor3_even : ∀ joints : Chain3 F, c3_even (chain_rotor3 joints).
  Proof.
    induction joints as [| j rest IH].
    - rewrite chain_rotor3_nil. repeat split; reflexivity.
    - rewrite chain_rotor3_cons.
      apply c3_even_mul_closed.
      + exact (rotor3_even (joint3_c j) (joint3_b12 j) (joint3_b13 j) (joint3_b23 j)).
      + exact IH.
  Qed.

  Fixpoint chain_orientation_from3 (orientation : C3 F) (joints : Chain3 F) : C3 F :=
    match joints with
    | [] ⇒ orientation
    | j :: rest ⇒ chain_orientation_from3 (orientation × joint3_rotor j) rest
    end.

  Lemma chain_orientation_from3_nil : ∀ orientation : C3 F,
    chain_orientation_from3 orientation [] = orientation.
  Proof. reflexivity. Qed.

  Lemma chain_orientation_from3_cons : ∀ (orientation : C3 F)
      (j : Joint3 F) (rest : Chain3 F),
    chain_orientation_from3 orientation (j :: rest)
    = chain_orientation_from3 (orientation × joint3_rotor j) rest.
  Proof. reflexivity. Qed.

  Theorem chain_rotor_is_product3 : ∀ (joints : Chain3 F) (orientation : C3 F),
    chain_orientation_from3 orientation joints == orientation × chain_rotor3 joints.
  Proof.
    induction joints as [| j rest IH]; intros orientation.
    - calc (chain_orientation_from3 orientation [])
        == orientation
        == (orientation × 1)
        == (orientation × chain_rotor3 []).
      + rewrite (chain_orientation_from3_nil orientation). reflexivity.
      + symmetry. exact (right_identity orientation).
      + rewrite chain_rotor3_nil. reflexivity.
    - calc (chain_orientation_from3 orientation (j :: rest))
        == (chain_orientation_from3 (orientation × joint3_rotor j) rest)
        == ((orientation × joint3_rotor j) × chain_rotor3 rest)
        == (orientation × (joint3_rotor j × chain_rotor3 rest))
        == (orientation × chain_rotor3 (j :: rest)).
      + rewrite (chain_orientation_from3_cons orientation j rest). reflexivity.
      + exact (IH (orientation × joint3_rotor j)).
      + exact (associative orientation (joint3_rotor j) (chain_rotor3 rest)).
      + rewrite chain_rotor3_cons. reflexivity.
  Qed.

  Corollary chain_rotor_is_product3_from_identity : ∀ joints : Chain3 F,
    chain_orientation_from3 1 joints == chain_rotor3 joints.
  Proof.
    intros joints. rewrite (chain_rotor_is_product3 joints 1).
    exact (left_identity (chain_rotor3 joints)).
  Qed.

  Fixpoint chain_positions3_from (orientation position : C3 F)
           (joints : Chain3 F) : list (C3 F) :=
    match joints with
    | [] ⇒ []
    | j :: rest ⇒
        let orientation' := orientation × joint3_rotor j in
        let position' := position + joint3_length j *: chain_direction3 orientation' in
        position' :: chain_positions3_from orientation' position' rest
    end.

  Lemma chain_positions3_from_nil : ∀ orientation position : C3 F,
    chain_positions3_from orientation position [] = [].
  Proof. reflexivity. Qed.

  Lemma chain_positions3_from_cons : ∀ (orientation position : C3 F)
      (j : Joint3 F) (rest : Chain3 F),
    chain_positions3_from orientation position (j :: rest)
    = let orientation' := orientation × joint3_rotor j in
      let position' := position + joint3_length j *: chain_direction3 orientation' in
      position' :: chain_positions3_from orientation' position' rest.
  Proof. reflexivity. Qed.

  Definition chain_positions3 (base : C3 F) (joints : Chain3 F) : list (C3 F) :=
    chain_positions3_from 1 base joints.

  Lemma c3_rotate_compose : ∀ A B x : C3 F,
    c3_rotate A (c3_rotate B x) == c3_rotate (A × B) x.
  Proof.
    intros A B x. unfold c3_rotate.
    calc (A × (B × x × c3_reverse B) × c3_reverse A)
      == (A × (B × x) × c3_reverse B × c3_reverse A)
        { (c3_mul_assoc A (B × x) (c3_reverse B)) }
      == (A × B × x × c3_reverse B × c3_reverse A)
        { (c3_mul_assoc A B x) }
      == (A × B × x × (c3_reverse B × c3_reverse A))
        { (c3_mul_assoc (A × B × x) (c3_reverse B) (c3_reverse A)) }
      == (A × B × x × c3_reverse (A × B))
        { (c3_reverse_anti A B) }.
  Qed.

End Chain3Ops.


Definition planar_arm_example : Chain2 Q :=
  [ mkJoint2 (1#1)%Q (0#1)%Q (1#1)%Q;
    mkJoint2 (0#1)%Q (1#1)%Q (1#1)%Q ].

Example planar_arm_positions :
  chain_positions2 (0 : Cl2 Q) planar_arm_example
  = [ Cl2.vec (plane_vector (1#1)%Q (0#1)%Q); Cl2.vec (plane_vector (0#1)%Q (0#1)%Q) ].
Proof. vm_compute. reflexivity. Qed.

Example planar_arm_cumulative_rotor_is_product :
  chain_orientation_from2 (1 : Cl2 Q) planar_arm_example == chain_rotor2 planar_arm_example.
Proof. exact (chain_rotor_is_product2_from_identity planar_arm_example). Qed.

Definition spatial_arm_example : Chain3 Q :=
  [ mkJoint3 (1#1)%Q (0#1)%Q (0#1)%Q (0#1)%Q (1#1)%Q;
    mkJoint3 (0#1)%Q (1#1)%Q (0#1)%Q (0#1)%Q (1#1)%Q ].

Example spatial_arm_positions :
  chain_positions3 (0 : C3 Q) spatial_arm_example
  = [ c3_vec (space_vector (1#1)%Q (0#1)%Q (0#1)%Q);
      c3_vec (space_vector (0#1)%Q (0#1)%Q (0#1)%Q) ].
Proof. vm_compute. reflexivity. Qed.

Example spatial_arm_cumulative_rotor_is_product :
  chain_orientation_from3 (1 : C3 Q) spatial_arm_example == chain_rotor3 spatial_arm_example.
Proof. exact (chain_rotor_is_product3_from_identity spatial_arm_example). Qed.
