A two-cocycle on an abelian group B with values in the multiplicative monoid of F: a coefficient sigma a b for each ordered pair, whose two regroupings of a triple agree, and which is trivial against the unit.
Module Cocycle.
  Class Cocycle (F B : Type) {eF : EqRel F} {oF : One F} {mF : HasMult F}
      {eB : EqRel B} {zB : Zero B} {aB : HasAdd B}
      (sigma : B → B → F) := {
    cocycle_proper :: Proper (equ ==> equ ==> equ) sigma;
    cocycle_equation : ∀ a b c : B,
                             sigma a b × sigma (a + b) c
                             == sigma b c × sigma a (b + c);
    cocycle_left_normal : ∀ a : B, sigma 0 a == 1;
    cocycle_right_normal : ∀ a : B, sigma a 0 == 1;
  }.
  Hint Mode Cocycle ! ! - - - - - - - : typeclass_instances.
End Cocycle.
Export Cocycle.
