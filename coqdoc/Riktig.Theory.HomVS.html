Section HomVS.
  Context {F U V : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Local Notation Hom := (LinearMap F U V).

  #[export] Instance Hom_EqRel : EqRel Hom := linear_map_equ.

  #[export] Instance Hom_Equivalence : Equivalence (@equ Hom Hom_EqRel).
  Proof.
    apply Build_Equivalence.
    - intros f u. reflexivity.
    - intros f g Hfg u. symmetry. apply Hfg.
    - intros f g h Hfg Hgh u. rewrite (Hfg u). apply Hgh.
  Qed.

  #[export] Instance Hom_Zero : Zero Hom :=
    linear_map_const0.

  Section HomPlus.
    Context (f g : Hom).

    #[export] Instance hom_plus_proper : Proper (equ ==> equ) (fun u : U ⇒ f u + g u).
    Proof.
      intros u v Huv.
      rewrite (linear_map_proper f _ _ Huv), (linear_map_proper g _ _ Huv).
      reflexivity.
    Qed.

    Lemma hom_plus_add : ∀ u v : U,
      f (u + v) + g (u + v) == (f u + g u) + (f v + g v).
    Proof.
      intros u v.
      rewrite (linear_map_additive f u v), (linear_map_additive g u v).
      apply add_interchange.
    Qed.

    Lemma hom_plus_scale : ∀ (a : F) (u : U),
      f (a *: u) + g (a *: u) == a *: (f u + g u).
    Proof.
      intros a u.
      rewrite (linear_map_scale f a u), (linear_map_scale g a u).
      symmetry.
      exact (left_distributive _ _ _).
    Qed.

    Definition hom_plus : Hom :=
      linear_map_from_laws
      (fun u : U ⇒ f u + g u)
      hom_plus_proper
      hom_plus_add
      hom_plus_scale.

  End HomPlus.

  #[export] Instance Hom_Add : HasAdd Hom := hom_plus.

  Section HomOpp.
    Context (f : Hom).

    #[export] Instance hom_opp_proper : Proper (equ ==> equ) (fun u : U ⇒ - f u).
    Proof.
      intros u v Huv.
      rewrite (linear_map_proper f _ _ Huv).
      reflexivity.
    Qed.

    Lemma hom_opp_add : ∀ u v : U, - f (u + v) == - f u + - f v.
    Proof.
      intros u v.
      rewrite (linear_map_additive f u v).
      apply neg_add.
    Qed.

    Lemma hom_opp_scale : ∀ (a : F) (u : U), - f (a *: u) == a *: - f u.
    Proof.
      intros a u.
      rewrite (linear_map_scale f a u).
      symmetry.
      apply scale_neg.
    Qed.

    Definition hom_opp : Hom :=
      linear_map_from_laws
      (fun u : U ⇒ - f u)
      hom_opp_proper
      hom_opp_add
      hom_opp_scale.

  End HomOpp.

  #[export] Instance Hom_Neg : HasNeg Hom := hom_opp.

  Section HomSmul.
    Context (a : F) (f : Hom).

    #[export] Instance hom_smul_proper : Proper (equ ==> equ) (fun u : U ⇒ a *: f u).
    Proof.
      intros u v Huv.
      rewrite (linear_map_proper f _ _ Huv).
      reflexivity.
    Qed.

    Lemma hom_smul_add : ∀ u v : U,
      a *: f (u + v) == a *: f u + a *: f v.
    Proof.
      intros u v.
      rewrite (linear_map_additive f u v).
      exact (left_distributive _ _ _).
    Qed.

    Lemma hom_smul_scale : ∀ (b : F) (u : U),
      a *: f (b *: u) == b *: (a *: f u).
    Proof.
      intros b u.
      rewrite (linear_map_scale f b u).
      apply scale_swap.
    Qed.

    Definition hom_smul : Hom :=
      linear_map_from_laws
      (fun u : U ⇒ a *: f u)
      hom_smul_proper
      hom_smul_add
      hom_smul_scale.

  End HomSmul.

  #[export] Instance Hom_Scale : HasScale F Hom := hom_smul.

  Lemma hom_assoc : ∀ f g h : Hom, (f + g) + h == f + (g + h).
  Proof.
    intros f g h u.
    exact (associative _ _ _).
  Qed.

  Lemma hom_idl : ∀ f : Hom, 0 + f == f.
  Proof.
    intros f u.
    exact (left_identity _).
  Qed.

  Lemma hom_idr : ∀ f : Hom, f + 0 == f.
  Proof.
    intros f u.
    exact (right_identity _).
  Qed.

  #[export] Instance hom_add_proper : Proper (equ ==> equ ==> equ) (@add Hom Hom_Add).
  Proof.
    intros f f' Hf g g' Hg u.
    exact (Magma.op_proper _ _ (Hf u) _ _ (Hg u)).
  Qed.

  Lemma hom_invl : ∀ f : Hom, (- f) + f == 0.
  Proof.
    intros f u.
    exact (left_inverse (f u)).
  Qed.

  Lemma hom_invr : ∀ f : Hom, f + (- f) == 0.
  Proof.
    intros f u.
    exact (right_inverse (f u)).
  Qed.

  #[export] Instance hom_neg_proper : Proper (equ ==> equ) (@neg Hom Hom_Neg).
  Proof.
    intros f f' Hf u.
    exact (Group.neg_proper (f u) (f' u) (Hf u)).
  Qed.

  #[export] Instance Hom_Group : Group Hom :=
    {| Group.monoid :=
         {| Monoid.semigroup :=
              {| Semigroup.magma :=
                   {| Magma.equivalence := Hom_Equivalence;
                      Magma.op_proper := hom_add_proper |};
                 Semigroup.is_associative := hom_assoc |};
            Monoid.is_left_identity := hom_idl;
            Monoid.is_right_identity := hom_idr |};
       Group.neg_proper := hom_neg_proper;
       Group.is_left_inverse := hom_invl;
       Group.is_right_inverse := hom_invr |}.

  Lemma hom_comm : ∀ f g : Hom, f + g == g + f.
  Proof.
    intros f g u.
    exact (commutative _ _).
  Qed.

  #[export] Instance Hom_AbGroup : AbGroup Hom :=
    {| AbGroup.group :=
         {| Group.monoid :=
              {| Monoid.semigroup :=
                   {| Semigroup.magma :=
                        {| Magma.equivalence := Hom_Equivalence;
                           Magma.op_proper := hom_add_proper |};
                      Semigroup.is_associative := hom_assoc |};
                 Monoid.is_left_identity := hom_idl;
                 Monoid.is_right_identity := hom_idr |};
            Group.neg_proper := hom_neg_proper;
            Group.is_left_inverse := hom_invl;
            Group.is_right_inverse := hom_invr |};
       AbGroup.is_commutative := hom_comm |}.

  #[export] Instance hom_scale_proper : Proper (equ ==> equ ==> equ) (@scale F Hom Hom_Scale).
  Proof.
    intros a a' Ha f f' Hf u.
    exact (ScaleLaws.scale_proper a a' Ha (f u) (f' u) (Hf u)).
  Qed.

  Lemma hom_scale_add : ∀ (a : F) (f g : Hom),
    a *: (f + g) == a *: f + a *: g.
  Proof.
    intros a f g u.
    exact (left_distributive a (f u) (g u)).
  Qed.

  Lemma hom_add_scale : ∀ (a b : F) (f : Hom),
    (a + b) *: f == a *: f + b *: f.
  Proof.
    intros a b f u.
    exact (right_distributive a b (f u)).
  Qed.

  Lemma hom_scale_scale : ∀ (a b : F) (f : Hom),
    (a × b) *: f == a *: (b *: f).
  Proof.
    intros a b f u.
    exact (associative _ _ _).
  Qed.

  Lemma hom_scale_1 : ∀ f : Hom, 1 *: f == f.
  Proof.
    intros f u.
    exact (left_identity _).
  Qed.

  #[export] Instance Hom_ScaleLaws : ScaleLaws F Hom :=
    {| ScaleLaws.scale_proper := hom_scale_proper;
       ScaleLaws.is_left_distributive := hom_scale_add;
       ScaleLaws.is_right_distributive := hom_add_scale;
       ScaleLaws.is_associative := hom_scale_scale;
       ScaleLaws.is_left_identity := hom_scale_1 |}.

  #[export] Instance Hom_VectorSpace : VectorSpace F Hom :=
    {| VectorSpace.vectors := Hom_AbGroup; VectorSpace.scaling := Hom_ScaleLaws |}.

End HomVS.

Section LmCompAssoc.
  Context {F P Q R S : Type} `{FF : Field F}.
  Context {eP : EqRel P} {zP : Zero P} {aP : HasAdd P} {nP : HasNeg P} {sP : HasScale F P}.
  Context {VP : VectorSpace F P}.
  Context {eQ : EqRel Q} {zQ : Zero Q} {aQ : HasAdd Q} {nQ : HasNeg Q} {sQ : HasScale F Q}.
  Context {VQ : VectorSpace F Q}.
  Context {eR : EqRel R} {zR : Zero R} {aR : HasAdd R} {nR : HasNeg R} {sR : HasScale F R}.
  Context {VR : VectorSpace F R}.
  Context {eS : EqRel S} {zS : Zero S} {aS : HasAdd S} {nS : HasNeg S} {sS : HasScale F S}.
  Context {VSp : VectorSpace F S}.

  Theorem lm_comp_assoc : ∀ (g : LinearMap F R S) (h : LinearMap F Q R)
      (k : LinearMap F P Q),
    linear_map_comp (linear_map_comp g h) k == linear_map_comp g (linear_map_comp h k).
  Proof.
    intros g h k u.
    reflexivity.
  Qed.

End LmCompAssoc.

Section LmCompAssoc4.
  Context {F U V W X Y : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context {eX : EqRel X} {zX : Zero X} {aX : HasAdd X} {nX : HasNeg X} {sX : HasScale F X}.
  Context {VX : VectorSpace F X}.
  Context {eY : EqRel Y} {zY : Zero Y} {aY : HasAdd Y} {nY : HasNeg Y} {sY : HasScale F Y}.
  Context {VY : VectorSpace F Y}.

  Theorem lm_comp_assoc4 : ∀ (L1 : LinearMap F U V)
      (L2 : LinearMap F V W) (L3 : LinearMap F W X) (L4 : LinearMap F X Y),
    linear_map_comp (linear_map_comp (linear_map_comp L4 L3) L2) L1
      == linear_map_comp L4 (linear_map_comp L3 (linear_map_comp L2 L1)).
  Proof.
    intros L1 L2 L3 L4.
    calc (linear_map_comp (linear_map_comp (linear_map_comp L4 L3) L2) L1)
      == (linear_map_comp (linear_map_comp L4 L3) (linear_map_comp L2 L1))
      == (linear_map_comp L4 (linear_map_comp L3 (linear_map_comp L2 L1))).
    - exact (lm_comp_assoc (linear_map_comp L4 L3) L2 L1).
    - exact (lm_comp_assoc L4 L3 (linear_map_comp L2 L1)).
  Qed.

End LmCompAssoc4.

Section EndRing.
  Context {F V : Type} `{FF : Field F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Local Notation EndV := (LinearMap F V V).

  #[export] Instance End_One : One EndV := linear_map_id.

  #[export] Instance End_Mult : HasMult EndV :=
    fun f g : EndV ⇒ linear_map_comp f g.

  Lemma end_mul_assoc : ∀ f g h : EndV, (f × g) × h == f × (g × h).
  Proof.
    intros f g h u.
    reflexivity.
  Qed.

  Lemma end_mul_1l : ∀ f : EndV, 1 × f == f.
  Proof.
    intros f u.
    reflexivity.
  Qed.

  Lemma end_mul_1r : ∀ f : EndV, f × 1 == f.
  Proof.
    intros f u.
    reflexivity.
  Qed.

  Lemma end_distl : ∀ f g h : EndV, f × (g + h) == f × g + f × h.
  Proof.
    intros f g h u.
    apply (linear_map_additive f).
  Qed.

  Lemma end_distr : ∀ f g h : EndV, (f + g) × h == f × h + g × h.
  Proof.
    intros f g h u.
    reflexivity.
  Qed.

  #[export] Instance end_mul_proper : Proper (equ ==> equ ==> equ) (@mult EndV End_Mult).
  Proof.
    intros f f' Hf g g' Hg u.
    rewrite (linear_map_proper f _ _ (Hg u)).
    apply Hf.
  Qed.

  #[export] Instance End_Ring : Ring EndV :=
    {| Ring.abgroup :=
         {| AbGroup.group :=
              {| Group.monoid :=
                   {| Monoid.semigroup :=
                        {| Semigroup.magma :=
                             {| Magma.equivalence := Hom_Equivalence;
                                Magma.op_proper := hom_add_proper |};
                           Semigroup.is_associative := hom_assoc |};
                      Monoid.is_left_identity := hom_idl;
                      Monoid.is_right_identity := hom_idr |};
                 Group.neg_proper := hom_neg_proper;
                 Group.is_left_inverse := hom_invl;
                 Group.is_right_inverse := hom_invr |};
            AbGroup.is_commutative := hom_comm |};
       Ring.mul_monoid :=
         {| Monoid.semigroup :=
              {| Semigroup.magma :=
                   {| Magma.equivalence := Hom_Equivalence;
                      Magma.op_proper := end_mul_proper |};
                 Semigroup.is_associative := end_mul_assoc |};
            Monoid.is_left_identity := end_mul_1l;
            Monoid.is_right_identity := end_mul_1r |};
       Ring.is_left_distributive := end_distl;
       Ring.is_right_distributive := end_distr |}.

End EndRing.
