From Coq Require Import List.
From Riktig.Structure Require Export Compiler.
From Riktig.Lib Require Export Coq.StackCode.
Import ListNotations.

The expressions and the stack code are a compiler: the code is the list monoid, running it is the left fold, and the source is a binary tree. The operator is a parameter, and no law is asked of it. Every carrier here is compared by Coq's equality, so the laws that ask a map to respect the comparison are congruences.
Section CompilerStackCode.
  Context {V : Type} (binop : V → V → V).

  Local Ltac leibniz := unfold equ in *; subst.

  #[export] Instance Code_Zero : Zero (code V) := [].
  #[export] Instance Code_HasAdd : HasAdd (code V) := @List.app (instr V).

  #[export] Instance Code_Magma : Magma (code V) Code_HasAdd (eq := eq).
  Proof. split; [ exact eq_equivalence | intros x x' Hx y y' Hy; leibniz; reflexivity ]. Qed.

  #[export] Instance Code_Semigroup : Semigroup (code V) Code_HasAdd (eq := eq).
  Proof. split; [ exact Code_Magma | intros x y z; symmetry; apply app_assoc ]. Qed.

  #[export] Instance Code_Monoid : Monoid (code V) Code_Zero Code_HasAdd (eq := eq).
  Proof. split; [ exact Code_Semigroup | intros x; reflexivity | intros x; apply app_nil_r ]. Qed.

  #[export] Instance Expr_HasTip : HasTip V (expr V) := @Lit V.
  #[export] Instance Expr_HasFork : HasFork (expr V) := @Node V.

  #[export] Instance Expr_Tree : Tree V (expr V) (eqA := eq) (eqTr := eq) Expr_HasTip Expr_HasFork.
  Proof.
    split.
    - exact eq_equivalence.
    - intros a b Hab; leibniz; reflexivity.
    - intros t t' Ht u u' Hu; leibniz; reflexivity.
    - intros a b Hab; unfold equ in *; injection Hab; trivial.
    - intros t u t' u' Hf; unfold equ in *; injection Hf; split; assumption.
    - intros a t u Hf; unfold equ in *; discriminate Hf.
    - intros P HP Htip Hfork t.
      induction t as [v | t IHt u IHu]; [ apply Htip | apply Hfork; assumption ].
  Qed.

  #[export] Instance Code_StackMachine :
    StackMachine V (code V) (list V) (eqV := eq) (eqC := eq) (eqS := eq)
      binop (fun v ⇒ [PushI v]) [OpI] (code_exec binop) (@cons V).
  Proof.
    split.
    - exact Code_Monoid.
    - exact eq_equivalence.
    - exact eq_equivalence.
    - intros a a' Ha b b' Hb; leibniz; reflexivity.
    - intros p p' Hp s s' Hs; leibniz; reflexivity.
    - intros v v' Hv s s' Hs; leibniz; reflexivity.
    - exact (code_exec_nil binop).
    - intros p q s. exact (code_exec_app binop p q s).
    - exact (code_exec_push binop).
    - exact (code_exec_op binop).
  Qed.

  #[export] Instance Code_Compiler :
    Compiler V (expr V) (code V) (list V) (eqV := eq) (eqE := eq) (eqC := eq) (eqS := eq)
      binop (fun v ⇒ [PushI v]) [OpI] (code_exec binop) (@cons V)
      Expr_HasTip Expr_HasFork (expr_eval binop) expr_compile.
  Proof.
    split.
    - exact Code_StackMachine.
    - exact Expr_Tree.
    - intros e e' He; leibniz; reflexivity.
    - intros e e' He; leibniz; reflexivity.
    - intros v; reflexivity.
    - intros e f; reflexivity.
    - intros v; reflexivity.
    - intros e f; reflexivity.
  Qed.
End CompilerStackCode.
