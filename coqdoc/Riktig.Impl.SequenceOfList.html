Every list over Coq's types is a sequence: it yields its head and leaves its tail, read off by taking the list apart.
Section S.
  Context {X L : Type}
    {build : HasBuild Type (List.shape (fun T0 : Type ⇒ T0) X) L}
    {fold : HasCata Type (List.shape (fun T0 : Type ⇒ T0) X) L}
    {LL : List Type (fun T0 : Type ⇒ T0) X L build fold}.

  #[export] Instance List_HasStep : HasStep X L :=
    fun l x l' ⇒ uncons (O := Type) (den := fun T0 : Type ⇒ T0) (x := X) l = inr (x, l').

  #[export] Instance List_Sequence : Sequence X L List_HasStep (eqC := eq) (eqS := eq).
  Proof.
    split.
    - exact eq_equivalence.
    - exact eq_equivalence.
    - intros l l' Hl x x' Hx m m' Hm. cbv [equ] in ×. subst. reflexivity.
    - intros l x x' l' l'' H1 H2. cbv [step List_HasStep] in ×.
      rewrite H1 in H2. injection H2 as Hx Hl. split; assumption.
  Qed.
End S.
