Library Riktig.Structure.NatCast

From Riktig Require Export Ops Props Structure.Monoid.

Module NatCast.
  Class NatCast (R : Type) {eR : EqRel R} {zR : Zero R} {oR : One R}
      {aR : HasAdd R} (c : nat → R) := {
    cast_zero : c O == 0;
    cast_succ : ∀ n : nat, c (S n) == 1 + c n;
  }.
  Hint Mode NatCast ! - - - - ! : typeclass_instances.
End NatCast.
Export NatCast.
