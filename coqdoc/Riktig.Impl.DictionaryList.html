Library Riktig.Impl.DictionaryList

From Riktig.Structure Require Export Dictionary.
From Riktig.Lib Require Import Coq.Assoc.

A dictionary as a list of pairs, newest binding first: the slow one that always works.
Notation assoc K V := (list (K × V)).
Section L.
  Context {K V : Type} {eqK : EqRel K} {eqV : EqRel V} {dK : EquDec K}
    {EK : Equivalence (@equ K eqK)} {EV : Equivalence (@equ V eqV)}.

  #[export] Instance Assoc_EqRel : EqRel (list (K × V)) := assoc_eq equ equ EquDec.equ_dec.
  #[export] Instance Assoc_HasEmpty : HasEmpty (list (K × V)) := nil.
  #[export] Instance Assoc_HasInsert : HasInsert K V (list (K × V)) :=
    fun k v d ⇒ cons (k, v) d.
  #[export] Instance Assoc_HasFind : HasFind K V (list (K × V)) := assoc_find equ EquDec.equ_dec.

  #[export] Instance Assoc_Dictionary
    : Dictionary K V (list (K × V)) Assoc_HasEmpty Assoc_HasInsert Assoc_HasFind :=
    { Dictionary.entries := assoc_eq_equivalence equ equ EquDec.equ_dec;
      Dictionary.insert_proper := fun k k' Hk v v' Hv d e Hd ⇒
        assoc_cons_proper equ equ EquDec.equ_dec k k' v v' d e Hk Hv Hd;
      Dictionary.find_congruent := assoc_find_congruent equ EquDec.equ_dec;
      Dictionary.find_empty := fun k ⇒ eq_refl;
      Dictionary.find_insert_here := assoc_find_cons_here equ equ EquDec.equ_dec;
      Dictionary.find_insert_there := assoc_find_cons_there equ EquDec.equ_dec }.
End L.
