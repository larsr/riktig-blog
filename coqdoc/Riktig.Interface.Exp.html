Library Riktig.Interface.Exp

From Coq Require Import Morphisms.
From Riktig Require Import Ops Interface.Field Interface.Order.

Module Exp.
  Class Exp (F : Type) `{OrderedField F} := {
    fn : F → F;
    fn_proper :: Proper (equ ==> equ) fn;

    zero_to_one : fn 0 == 1;
    add_to_mul :: Additive F F add mult fn;

    positive : ∀ a : F, 0 ≤ fn a;

    monotone : ∀ a b : F, a < b → fn a < fn b;
  }.
  Hint Mode Exp ! - - - - - - - - - - - : typeclass_instances.
End Exp.
Export Exp.
Notation exp_fn := Exp.fn.
