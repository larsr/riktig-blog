From Riktig Require Export Structure.Lambda.
From Riktig Require Import Lib.Calc.


Notation "'0'" := O.
Notation "'1'" := (S O).


Section LambdaModelTheory.
  Context {T S : Type} `{LambdaModel T S}.


  Lemma substitute_cancel : ∀ t u : T, (t.[sub_shift]).[u .: sub_id] == t.
  Proof.
    intros t u.
    calc ((t.[sub_shift]).[u .: sub_id])
      == (t.[sub_id]) { shift_cons }
      == t { substitute_id }.
  Qed.

  Lemma var_one_up_cons : ∀ (v : T) (s : S),
    (var 1).[sub_up (v .: s)] == v.[sub_shift].
  Proof.
    intros v s.
    calc ((var 1).[sub_up (v .: s)] : T)
      == ((((var 0).[v .: s]).[sub_shift]) : T) { (var_succ_up 0) }
      == (v.[sub_shift]) { (shift_respects _ _ (var_zero_cons v s)) }.
  Qed.

  Lemma var_one_up_up : ∀ s : S, (var 1).[sub_up (sub_up s)] == (var 1 : T).
  Proof.
    intros s.
    calc ((var 1).[sub_up (sub_up s)] : T)
      == ((((var 0).[sub_up s]).[sub_shift]) : T) { (var_succ_up 0) }
      == (((var 0).[sub_shift]) : T) { (shift_respects _ _ (var_zero_up s)) }
      == (var 1 : T) { (var_shift 0) }.
  Qed.

  Lemma var_two_up_up_cons : ∀ (v : T) (s : S),
    (var 2).[sub_up (sub_up (v .: s))] == (v.[sub_shift]).[sub_shift].
  Proof.
    intros v s.
    calc ((var 2).[sub_up (sub_up (v .: s))] : T)
      == ((((var 1).[sub_up (v .: s)]).[sub_shift]) : T)
           { (var_succ_up 1) }
      == ((v.[sub_shift]).[sub_shift])
           { (shift_respects _ _ (var_one_up_cons v s)) }.
  Qed.

  Lemma shift_shift_up_cancel : ∀ (t u : T),
    ((t.[sub_shift]).[sub_shift]).[sub_up (u .: sub_id)] == t.[sub_shift].
  Proof.
    intros t u.
    calc (((t.[sub_shift]).[sub_shift]).[sub_up (u .: sub_id)])
      == (((t.[sub_shift]).[u .: sub_id]).[sub_shift]) { shift_up }
      == (t.[sub_shift]) { (shift_respects _ _ (substitute_cancel t u)) }.
  Qed.


  Lemma app_church_true : ∀ g : T, val g →
    app church_true g == lam (g.[sub_shift]).
  Proof.
    intros g Hg. unfold church_true.
    calc (app (lam (lam (var 1))) g)
      == ((lam (var 1)).[g .: sub_id] : T)
           { (beta _ _ Hg) }
      == (lam ((var 1).[sub_up (g .: sub_id)]) : T)
           { lam_substitute }
      == (lam (g.[sub_shift]))
           { (var_one_up_cons g sub_id) }.
  Qed.

  Theorem church_if_true : ∀ g h : T, val g → val h →
    church_if church_true g h == g.
  Proof.
    intros g h Hg Hh.
    calc (church_if church_true g h)
      == (app (lam (g.[sub_shift])) h) { (app_church_true g Hg) }
      == ((g.[sub_shift]).[h .: sub_id]) { (beta _ _ Hh) }
      == g { substitute_cancel }.
  Qed.

  Lemma app_church_false : ∀ g : T, val g →
    app church_false g == lam (var 0).
  Proof.
    intros g Hg. unfold church_false.
    calc (app (lam (lam (var 0))) g)
      == ((lam (var 0)).[g .: sub_id] : T) { (beta _ _ Hg) }
      == (lam ((var 0).[sub_up (g .: sub_id)]) : T) { lam_substitute }
      == (lam (var 0) : T) { var_zero_up }.
  Qed.

  Theorem church_if_false : ∀ g h : T, val g → val h →
    church_if church_false g h == h.
  Proof.
    intros g h Hg Hh.
    calc (church_if church_false g h)
      == (app (lam (var 0)) h) { (app_church_false g Hg) }
      == ((var 0).[h .: sub_id] : T) { (beta _ _ Hh) }
      == h { var_zero_cons }.
  Qed.


  Lemma pair_body_substituted : ∀ v : T,
    (app (app (var 0) (var 2)) (var 1)).[sub_up (sub_up (v .: sub_id))]
    == app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1).
  Proof.
    intros v.
    calc ((app (app (var 0) (var 2)) (var 1)).[sub_up (sub_up (v .: sub_id))] : T)
      == (app ((app (var 0) (var 2)).[sub_up (sub_up (v .: sub_id))])
              ((var 1).[sub_up (sub_up (v .: sub_id))]) : T)
           { app_substitute }
      == (app (app ((var 0).[sub_up (sub_up (v .: sub_id))])
                   ((var 2).[sub_up (sub_up (v .: sub_id))]))
              ((var 1).[sub_up (sub_up (v .: sub_id))]) : T)
           { app_substitute }
      == (app (app (var 0) ((var 2).[sub_up (sub_up (v .: sub_id))]))
              ((var 1).[sub_up (sub_up (v .: sub_id))]) : T)
           { var_zero_up }
      == (app (app (var 0) ((v.[sub_shift]).[sub_shift]))
              ((var 1).[sub_up (sub_up (v .: sub_id))]))
           { (var_two_up_up_cons v sub_id) }
      == (app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1))
           { var_one_up_up }.
  Qed.

  Lemma app_church_pair : ∀ v : T, val v →
    app church_pair v
    == lam (lam (app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1))).
  Proof.
    intros v Hv. unfold church_pair.
    calc (app (lam (lam (lam (app (app (var 0) (var 2)) (var 1))))) v)
      == ((lam (lam (app (app (var 0) (var 2)) (var 1)))).[v .: sub_id] : T)
           { (beta _ _ Hv) }
      == (lam ((lam (app (app (var 0) (var 2)) (var 1))).[sub_up (v .: sub_id)]) : T)
           { lam_substitute }
      == (lam (lam ((app (app (var 0) (var 2)) (var 1))
                      .[sub_up (sub_up (v .: sub_id))])) : T)
           { lam_substitute }
      == (lam (lam (app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1))))
           { pair_body_substituted }.
  Qed.

  Lemma pair_body_second_substituted : ∀ v w : T,
    (app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1))
      .[sub_up (w .: sub_id)]
    == app (app (var 0) (v.[sub_shift])) (w.[sub_shift]).
  Proof.
    intros v w.
    calc ((app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1))
            .[sub_up (w .: sub_id)])
      == (app ((app (var 0) ((v.[sub_shift]).[sub_shift]))
                 .[sub_up (w .: sub_id)])
              ((var 1).[sub_up (w .: sub_id)]))
           { app_substitute }
      == (app (app ((var 0).[sub_up (w .: sub_id)])
                   (((v.[sub_shift]).[sub_shift]).[sub_up (w .: sub_id)]))
              ((var 1).[sub_up (w .: sub_id)]))
           { app_substitute }
      == (app (app (var 0)
                   (((v.[sub_shift]).[sub_shift]).[sub_up (w .: sub_id)]))
              ((var 1).[sub_up (w .: sub_id)]))
           { var_zero_up }
      == (app (app (var 0) (v.[sub_shift])) ((var 1).[sub_up (w .: sub_id)]))
           { (shift_shift_up_cancel v w) }
      == (app (app (var 0) (v.[sub_shift])) (w.[sub_shift]))
           { (var_one_up_cons w sub_id) }.
  Qed.

  Lemma app_church_pair_twice : ∀ v w : T, val v → val w →
    app (app church_pair v) w
    == lam (app (app (var 0) (v.[sub_shift])) (w.[sub_shift])).
  Proof.
    intros v w Hv Hw.
    calc (app (app church_pair v) w)
      == (app (lam (lam (app (app (var 0) ((v.[sub_shift]).[sub_shift]))
                            (var 1)))) w)
           { (app_church_pair v Hv) }
      == ((lam (app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1)))
            .[w .: sub_id])
           { (beta _ _ Hw) }
      == (lam ((app (app (var 0) ((v.[sub_shift]).[sub_shift])) (var 1))
                 .[sub_up (w .: sub_id)]))
           { lam_substitute }
      == (lam (app (app (var 0) (v.[sub_shift])) (w.[sub_shift])))
           { pair_body_second_substituted }.
  Qed.

  Lemma applied_pair_body : ∀ v w f : T,
    (app (app (var 0) (v.[sub_shift])) (w.[sub_shift])).[f .: sub_id]
    == app (app f v) w.
  Proof.
    intros v w f.
    calc ((app (app (var 0) (v.[sub_shift])) (w.[sub_shift])).[f .: sub_id])
      == (app ((app (var 0) (v.[sub_shift])).[f .: sub_id])
              ((w.[sub_shift]).[f .: sub_id]))
           { app_substitute }
      == (app (app ((var 0).[f .: sub_id]) ((v.[sub_shift]).[f .: sub_id]))
              ((w.[sub_shift]).[f .: sub_id]))
           { app_substitute }
      == (app (app f ((v.[sub_shift]).[f .: sub_id]))
              ((w.[sub_shift]).[f .: sub_id]))
           { var_zero_cons }
      == (app (app f v) ((w.[sub_shift]).[f .: sub_id]))
           { substitute_cancel }
      == (app (app f v) w)
           { substitute_cancel }.
  Qed.

  Theorem church_fst_pair : ∀ v w : T, val v → val w →
    app (app (app church_pair v) w) church_true == v.
  Proof.
    intros v w Hv Hw.
    calc (app (app (app church_pair v) w) church_true)
      == (app (lam (app (app (var 0) (v.[sub_shift])) (w.[sub_shift])))
              church_true)
           { (app_church_pair_twice v w Hv Hw) }
      == ((app (app (var 0) (v.[sub_shift])) (w.[sub_shift]))
            .[church_true .: sub_id])
           { (beta _ _ (val_lam (lam (var 1)))) }
      == (app (app church_true v) w)
           { applied_pair_body }
      == v
           { (church_if_true v w Hv Hw) }.
  Qed.

  Theorem church_snd_pair : ∀ v w : T, val v → val w →
    app (app (app church_pair v) w) church_false == w.
  Proof.
    intros v w Hv Hw.
    calc (app (app (app church_pair v) w) church_false)
      == (app (lam (app (app (var 0) (v.[sub_shift])) (w.[sub_shift])))
              church_false)
           { (app_church_pair_twice v w Hv Hw) }
      == ((app (app (var 0) (v.[sub_shift])) (w.[sub_shift]))
            .[church_false .: sub_id])
           { (beta _ _ (val_lam (lam (var 0)))) }
      == (app (app church_false v) w)
           { applied_pair_body }
      == w
           { (church_if_false v w Hv Hw) }.
  Qed.
End LambdaModelTheory.
