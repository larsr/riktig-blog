What holds in any lambda calculus, from the laws alone. Conversion is an equivalence -- transitivity is exactly one use of confluence -- and the constructors respect it, so calculation with == is available; then Church's booleans behave as they should, with no induction over syntax anywhere.
Notation "'church_true'" := (lam (lam (var (succ 0)))).
Notation "'church_false'" := (lam (lam (var 0))).
Notation "'church_if' b g h" := (app (app b g) h) (at level 10, b at level 9, g at level 9, h at level 9).

Section LambdaTheory.
  Context {N T Sb : Type} `{LambdaCalculus N T Sb}.

  Lemma reduces_equ : ∀ t u : T, t -->* u → t == u.
  Proof. intros t u Hr. apply equ_join. ∃ u. split; [ exact Hr | apply reduces_refl ]. Qed.

  #[export] Instance conversion_equivalence : Equivalence (@equ T eqT).
  Proof.
    split.
    - intros t. apply reduces_equ. apply reduces_refl.
    - intros t u Ht. apply equ_join in Ht. destruct Ht as [w [H1 H2]].
      apply equ_join. ∃ w. split; assumption.
    - intros t u v Htu Huv. apply equ_join in Htu. apply equ_join in Huv.
      destruct Htu as [w1 [Ht1 Hu1]]. destruct Huv as [w2 [Hu2 Hv2]].
      destruct (confluence u w1 w2 Hu1 Hu2) as [w [Hw1 Hw2]].
      apply equ_join. ∃ w. split.
      + exact (reduces_trans t w1 w Ht1 Hw1).
      + exact (reduces_trans v w2 w Hv2 Hw2).
  Qed.

  #[export] Instance lam_proper : Proper (equ ==> equ) (lam : T → T).
  Proof.
    intros b b' Hb. apply equ_join in Hb. destruct Hb as [w [H1 H2]].
    apply equ_join. ∃ (lam w). split; apply reduces_lam; assumption.
  Qed.

  #[export] Instance app_proper : Proper (equ ==> equ ==> equ) (app : T → T → T).
  Proof.
    intros f f' Hf a a' Ha. apply equ_join in Hf. apply equ_join in Ha.
    destruct Hf as [w1 [Hf1 Hf2]]. destruct Ha as [w2 [Ha1 Ha2]].
    apply equ_join. ∃ (app w1 w2). split; apply reduces_app; assumption.
  Qed.

  Lemma subst_proper_l : ∀ (t t' : T) (s : Sb), t == t' → t.[s] == t'.[s].
  Proof.
    intros t t' s Ht. apply equ_join in Ht. destruct Ht as [w [H1 H2]].
    apply equ_join. ∃ (w.[s]). split; apply reduces_subst; assumption.
  Qed.

  Theorem beta_equ : ∀ b u : T, app (lam b) u == b.[u .: sub_id].
  Proof. intros b u. apply reduces_equ. exact (beta b u). Qed.

  Theorem shift_cancel : ∀ t u : T, (t.[sub_shift]).[u .: sub_id] == t.
  Proof.
    intros t u.
    calc ((t.[sub_shift]).[u .: sub_id]) == (t.[sub_id]) { (shift_cons t u sub_id) } == t { (subst_id t) }.
  Qed.

  Theorem beta_id : ∀ u : T, app (lam (var 0)) u == u.
  Proof.
    intros u.
    calc (app (lam (var 0)) u) == ((var 0 : T).[u .: sub_id]) { (beta_equ (var 0) u) } == u { (var_zero_cons u sub_id) }.
  Qed.

Under one binder, a term shifted past it reads back the outer argument.
  Lemma var_one_up_cons : ∀ (g : T) (s : Sb), (var (succ 0) : T).[sub_up (g .: s)] == g.[sub_shift].
  Proof.
    intros g s.
    calc ((var (succ 0) : T).[sub_up (g .: s)])
      == (((var 0 : T).[g .: s]).[sub_shift]) { (var_succ_up 0 (g .: s)) }
      == (g.[sub_shift]) { (subst_proper_l _ _ sub_shift (var_zero_cons g s)) }.
  Qed.

  Theorem church_if_true : ∀ g h : T, church_if church_true g h == g.
  Proof.
    intros g h.
    calc (app (app (lam (lam (var (succ 0)))) g) h)
      == (app ((lam (var (succ 0))).[g .: sub_id]) h) { (app_proper _ _ (beta_equ (lam (var (succ 0))) g) _ _ (reflexivity h)) }
      == (app (lam ((var (succ 0)).[sub_up (g .: sub_id)])) h) { (app_proper _ _ (lam_subst (var (succ 0)) (g .: sub_id)) _ _ (reflexivity h)) }
      == (app (lam (g.[sub_shift])) h) { (app_proper _ _ (lam_proper _ _ (var_one_up_cons g sub_id)) _ _ (reflexivity h)) }
      == ((g.[sub_shift]).[h .: sub_id]) { (beta_equ (g.[sub_shift]) h) }
      == g { (shift_cancel g h) }.
  Qed.

  Theorem church_if_false : ∀ g h : T, church_if church_false g h == h.
  Proof.
    intros g h.
    calc (app (app (lam (lam (var 0))) g) h)
      == (app ((lam (var 0)).[g .: sub_id]) h) { (app_proper _ _ (beta_equ (lam (var 0)) g) _ _ (reflexivity h)) }
      == (app (lam ((var 0).[sub_up (g .: sub_id)])) h) { (app_proper _ _ (lam_subst (var 0) (g .: sub_id)) _ _ (reflexivity h)) }
      == (app (lam (var 0)) h) { (app_proper _ _ (lam_proper _ _ (var_zero_up (g .: sub_id))) _ _ (reflexivity h)) }
      == h { (beta_id h) }.
  Qed.

The setoid is not degenerate: distinct variables stay distinct.
  Theorem var_discrete : ∀ n m : N, n =/= m → var n =/= (var m : T).
  Proof. intros n m Hnm Hv. exact (Hnm (var_injective n m Hv)). Qed.
