What QArith states only one way round, or not at all, in the shape the rationals' structures ask for.
Lemma Qplus_opp_l : ∀ q : Q, - q + q == 0.
Proof. intros q. rewrite Qplus_comm. apply Qplus_opp_r. Qed.

Lemma Qmult_inv_l : ∀ q : Q, ¬ q == 0 → / q × q == 1.
Proof. intros q Hq. rewrite Qmult_comm. exact (Qmult_inv_r q Hq). Qed.

Lemma inject_Z_wd : ∀ a b : Z, a = b → inject_Z a == inject_Z b.
Proof. intros a b Hab. rewrite Hab. reflexivity. Qed.

Lemma inject_Z_inj : ∀ a b : Z, inject_Z a == inject_Z b → a = b.
Proof. intros a b Hab. exact (proj1 (inject_Z_injective a b) Hab). Qed.

Lemma inject_Z_add : ∀ a b : Z, inject_Z (a + b) == inject_Z a + inject_Z b.
Proof. intros a b. rewrite inject_Z_plus. reflexivity. Qed.

Lemma inject_Z_mul : ∀ a b : Z, inject_Z (a × b) == inject_Z a × inject_Z b.
Proof. intros a b. rewrite inject_Z_mult. reflexivity. Qed.

Lemma inject_Z_one : inject_Z 1 == 1.
Proof. reflexivity. Qed.

Lemma Q_is_fraction : ∀ q : Q, ∃ a b : Z,
  ¬ inject_Z b == 0 ∧ q × inject_Z b == inject_Z a.
Proof.
  intros q. ∃ (Qnum q), (Z.pos (Qden q)). split.
  - intros H. unfold Qeq, inject_Z in H. simpl in H. lia.
  - unfold Qeq, inject_Z, Qmult. simpl.
    rewrite Z.mul_1_r, Pos.mul_1_r. reflexivity.
Qed.

The strict order as "at most and not equal", both ways.
Lemma Qlt_of_le_ne : ∀ q r : Q, q ≤ r ∧ ¬ q == r → q < r.
Proof.
  intros q r [Hle Hne].
  destruct (proj1 (Qle_lteq q r) Hle) as [H | H]; [ exact H | destruct (Hne H) ].
Qed.

Lemma Qle_ne_of_lt : ∀ q r : Q, q < r → q ≤ r ∧ ¬ q == r.
Proof. intros q r H. split; [ exact (Qlt_le_weak q r H) | exact (Qlt_not_eq q r H) ]. Qed.

Lemma Qle_total : ∀ a b : Q, a ≤ b ∨ b ≤ a.
Proof.
  intros a b.
  destruct (Qlt_le_dec a b) as [H | H]; [ left; exact (Qlt_le_weak a b H) | right; exact H ].
Qed.

Lemma Qle_stable : ∀ a b : Q, ¬ ¬ a ≤ b → a ≤ b.
Proof.
  intros a b Hnn.
  destruct (Qlt_le_dec b a) as [H | H]; [ | exact H ].
  destruct (Hnn (Qlt_not_le b a H)).
Qed.

Lemma Qplus_le_mono_l : ∀ a b c : Q, a ≤ b → c + a ≤ c + b.
Proof. intros a b c Hab. exact (proj2 (Qplus_le_r a b c) Hab). Qed.

The naturals in the rationals, and a natural above any rational: the floor plus one, or zero.
Definition Q_of_nat (n : nat) : Q := inject_Z (Z.of_nat n).

Lemma Q_of_nat_zero : Q_of_nat 0 == 0.
Proof. reflexivity. Qed.

Lemma Q_of_nat_succ : ∀ n : nat, Q_of_nat (S n) == Q_of_nat n + 1.
Proof.
  intros n. unfold Q_of_nat. rewrite Nat2Z.inj_succ. unfold Z.succ.
  rewrite inject_Z_plus. reflexivity.
Qed.

Lemma Q_of_nat_proper : ∀ n m : nat, n = m → Q_of_nat n == Q_of_nat m.
Proof. intros n m H. rewrite H. reflexivity. Qed.

Definition Q_archimedean (q : Q) : {n : nat & q ≤ Q_of_nat n ∧ ¬ q == Q_of_nat n}.
Proof.
  ∃ (Z.to_nat (Qfloor q + 1)).
  abstract (unfold Q_of_nat; pose proof (Qlt_floor q) as Hf;
    assert (Hz : (Qfloor q + 1 ≤ Z.of_nat (Z.to_nat (Qfloor q + 1)))%Z) by lia;
    assert (Hq : inject_Z (Qfloor q + 1) ≤ inject_Z (Z.of_nat (Z.to_nat (Qfloor q + 1))))
      by (rewrite <- Zle_Qle; exact Hz);
    apply Qle_ne_of_lt; exact (Qlt_le_trans _ _ _ Hf Hq)).
Defined.
