Module GradedGA.
  Class GradedGA (F V A : Type) `{QuadraticForm F V}
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
      {Ao : One A} {Am : HasMult A} {As : HasScale F A}
      {GA : GeometricAlgebra F V A (Q := Q)}
      {Ag : HasGrade A} := {
    
An embedded vector is homogeneous of grade one.
    vec_is_grade_1 : ∀ u : V, ⟨ (ga_vec u : A) ⟩_ 1 == ga_vec u;
    
The unit is homogeneous of grade zero: this is what lets the scalars be read off grade 0 at all.
    one_is_grade_0 : ⟨ (1 : A) ⟩_ 0 == 1;
  }.
  Hint Mode GradedGA - - ! - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End GradedGA.
