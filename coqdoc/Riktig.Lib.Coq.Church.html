Church numerals as programs: successor, addition and multiplication, what they reduce to, and the discreteness of the numerals up to conversion. ZERO is the numeral church 0 written out, so that the constants a numeral is built from are all lambda terms: two is SUCC (SUCC ZERO).
Nothing here proves an arithmetic law. What the three programs satisfy is only that each computes what Coq's own operation computes, and Impl.NaturalsChurch turns that denotation into the Naturals instance; every law then comes from the structure.
Definition ZERO : term := λ λ #0.
Definition SUCC : term := λ λ λ ⦅#1 ⦅#2 #1 #0⦆⦆.
Definition PLUS : term := λ λ λ λ ⦅#3 #1 ⦅#2 #1 #0⦆⦆.
Definition MULT : term := λ λ λ λ ⦅#3 ⦅#2 #1⦆ #0⦆.

The body of a numeral has 0 and 1 free, nothing else

Lemma ren_body_fix : ∀ (r : nat → nat) (n : nat),
  r 0 = 0 → r 1 = 1 → ren r (church_body n) = church_body n.
Proof.
  intros r n H0 H1. induction n as [| n IH]; simpl.
  - rewrite H0. reflexivity.
  - rewrite H1, IH. reflexivity.
Qed.

Lemma psub_body_fix : ∀ (s : sub) (n : nat),
  s 0 = Var 0 → s 1 = Var 1 → psub s (church_body n) = church_body n.
Proof.
  intros s n H0 H1. induction n as [| n IH]; simpl.
  - exact H0.
  - rewrite H1, IH. reflexivity.
Qed.

Lemma psub_church : ∀ (s : sub) (n : nat), psub s (church n) = church n.
Proof. intros s n. simpl. rewrite psub_body_fix; reflexivity. Qed.

Lemma ren_church : ∀ (r : nat → nat) (n : nat), ren r (church n) = church n.
Proof. intros r n. rewrite ren_as_psub. apply psub_church. Qed.

Applying a numeral iterates its first argument

Fixpoint iterf (n : nat) (f x : term) : term :=
  match n with 0 ⇒ x | S n ⇒ App f (iterf n f x) end.

Lemma iterf_body : ∀ n : nat, iterf n #1 #0 = church_body n.
Proof. intros n. induction n as [| n IH]; simpl; [ reflexivity | rewrite IH; reflexivity ]. Qed.

Lemma iterf_plus : ∀ (a b : nat), iterf a #1 (church_body b) = church_body (a + b).
Proof. intros a b. induction a as [| a IH]; simpl; [ reflexivity | rewrite IH; reflexivity ]. Qed.

Lemma church_body_subst : ∀ (n : nat) (f x : term),
  psub (scons x (scons f Var)) (church_body n) = iterf n f x.
Proof. intros n f x. induction n as [| n IH]; simpl; [ reflexivity | rewrite IH; reflexivity ]. Qed.

Lemma church_ap : ∀ (n : nat) (f x : term), bsteps ⦅(church n) f x⦆ (iterf n f x).
Proof.
  intros n f x.
  apply (star_step bstep _ (App (psub (scons f Var) (Lam (church_body n))) x)).
  { apply bstep_app_l. exact (bstep_beta (Lam (church_body n)) f). }
  apply (star_step bstep _ (psub (scons x Var) (psub (up (scons f Var)) (church_body n)))).
  { exact (bstep_beta (psub (up (scons f Var)) (church_body n)) x). }
  rewrite psub_psub.
  rewrite (psub_ext (church_body n) _ (scons x (scons f Var))).
  - rewrite church_body_subst. apply star_refl.
  - intros [| [| m]]; simpl.
    + reflexivity.
    + rewrite psub_ren. simpl. apply psub_var.
    + reflexivity.
Qed.

Lemma joinable_ap : ∀ (n : nat) (f x : term), joinable ⦅(church n) f x⦆ (iterf n f x).
Proof. intros n f x. apply joinable_bsteps. apply church_ap. Qed.

What the programs reduce to

Each proof contracts the head redexes -- the substituted body is convertible to a shape whose only stuck subterms are a renamed or substituted numeral, which the closedness lemmas erase -- and then lets the numeral iterate.
Lemma succ_join : ∀ n : nat, joinable ⦅SUCC (church n)⦆ (church (S n)).
Proof.
  intros n.
  apply (joinable_trans _ (psub (scons (church n) Var) (λ λ ⦅#1 ⦅#2 #1 #0⦆⦆))).
  { apply joinable_bsteps. apply star_one. exact (bstep_beta _ (church n)). }
  change (psub (scons (church n) Var) (λ λ ⦅#1 ⦅#2 #1 #0⦆⦆))
    with (λ λ ⦅#1 ⦅(ren S (ren S (church n))) #1 #0⦆⦆).
  rewrite !ren_church.
  apply joinable_lam. apply joinable_lam.
  apply (joinable_app #1 #1 ⦅(church n) #1 #0⦆ (church_body n) (joinable_refl _)).
  rewrite <- (iterf_body n). apply joinable_ap.
Qed.

Lemma plus_join : ∀ a b : nat, joinable ⦅PLUS (church a) (church b)⦆ (church (a + b)).
Proof.
  intros a b.
  apply (joinable_trans _ ⦅(psub (scons (church a) Var) (λ λ λ ⦅#3 #1 ⦅#2 #1 #0⦆⦆)) (church b)⦆).
  { apply (joinable_app _ _ (church b) (church b)); [ | apply joinable_refl ].
    apply joinable_bsteps. apply star_one. exact (bstep_beta _ (church a)). }
  change (psub (scons (church a) Var) (λ λ λ ⦅#3 #1 ⦅#2 #1 #0⦆⦆))
    with (λ λ λ ⦅(ren S (ren S (ren S (church a)))) #1 ⦅#2 #1 #0⦆⦆).
  rewrite !ren_church.
  apply (joinable_trans _ (psub (scons (church b) Var) (λ λ ⦅(church a) #1 ⦅#2 #1 #0⦆⦆))).
  { apply joinable_bsteps. apply star_one. exact (bstep_beta _ (church b)). }
  change (psub (scons (church b) Var) (λ λ ⦅(church a) #1 ⦅#2 #1 #0⦆⦆))
    with (λ λ ⦅(psub (up (up (scons (church b) Var))) (church a)) #1 ⦅(ren S (ren S (church b))) #1 #0⦆⦆).
  rewrite psub_church, !ren_church.
  apply joinable_lam. apply joinable_lam.
  apply (joinable_trans _ ⦅(church a) #1 (church_body b)⦆).
  - apply joinable_app; [ apply joinable_refl | ].
    rewrite <- (iterf_body b). apply joinable_ap.
  - rewrite <- (iterf_plus a b). apply joinable_ap.
Qed.

Lemma mult_join : ∀ a b : nat, joinable ⦅MULT (church a) (church b)⦆ (church (a × b)).
Proof.
  intros a b.
  apply (joinable_trans _ ⦅(psub (scons (church a) Var) (λ λ λ ⦅#3 ⦅#2 #1⦆ #0⦆)) (church b)⦆).
  { apply (joinable_app _ _ (church b) (church b)); [ | apply joinable_refl ].
    apply joinable_bsteps. apply star_one. exact (bstep_beta _ (church a)). }
  change (psub (scons (church a) Var) (λ λ λ ⦅#3 ⦅#2 #1⦆ #0⦆))
    with (λ λ λ ⦅(ren S (ren S (ren S (church a)))) ⦅#2 #1⦆ #0⦆).
  rewrite !ren_church.
  apply (joinable_trans _ (psub (scons (church b) Var) (λ λ ⦅(church a) ⦅#2 #1⦆ #0⦆))).
  { apply joinable_bsteps. apply star_one. exact (bstep_beta _ (church b)). }
  change (psub (scons (church b) Var) (λ λ ⦅(church a) ⦅#2 #1⦆ #0⦆))
    with (λ λ ⦅(psub (up (up (scons (church b) Var))) (church a)) ⦅(ren S (ren S (church b))) #1⦆ #0⦆).
  rewrite psub_church, !ren_church.
  apply joinable_lam. apply joinable_lam.
  apply (joinable_trans _ (iterf a ⦅(church b) #1⦆ #0)); [ apply joinable_ap | ].
  induction a as [| a IH]; simpl.
  - apply joinable_refl.
  - apply (joinable_trans _ (App ⦅(church b) #1⦆ (church_body (a × b)))).
    { apply joinable_app; [ apply joinable_refl | exact IH ]. }
    apply (joinable_trans _ (iterf b #1 (church_body (a × b)))); [ apply joinable_ap | ].
    rewrite iterf_plus. apply joinable_refl.
Qed.

Numerals are normal, hence discrete up to conversion

Definition normal (t : term) : Prop := ∀ u : term, ¬ bstep t u.

Lemma church_body_normal : ∀ n : nat, normal (church_body n).
Proof.
  intros n. induction n as [| n IH]; intros u H; simpl in H.
  - inversion H.
  - inversion H; subst.
    + inversion H3.
    + exact (IH _ H3).
Qed.

Lemma church_normal : ∀ n : nat, normal (church n).
Proof.
  intros n u H. unfold church in H.
  inversion H; subst. inversion H1; subst.
  exact (church_body_normal n _ H2).
Qed.

Lemma bsteps_normal : ∀ t u : term, normal t → bsteps t u → u = t.
Proof.
  intros t u Hn H. destruct H as [t | t x u Hx Hs]; [ reflexivity | destruct (Hn _ Hx) ].
Qed.

Lemma church_body_inj : ∀ n m : nat, church_body n = church_body m → n = m.
Proof.
  intros n. induction n as [| n IH]; intros [| m] H; simpl in H.
  - reflexivity.
  - discriminate H.
  - discriminate H.
  - injection H as H. rewrite (IH m H). reflexivity.
Qed.

Lemma church_injective : ∀ n m : nat, joinable (church n) (church m) → n = m.
Proof.
  intros n m [w [Hn Hm]].
  rewrite (bsteps_normal _ _ (church_normal n) Hn) in Hm.
  apply bsteps_normal in Hm; [ | exact (church_normal m) ].
  unfold church in Hm. injection Hm as Hm. exact (church_body_inj n m Hm).
Qed.

Lemma is_numeral : ∀ n : nat, ∃ k : nat, joinable (church n) (church k).
Proof. intros n. ∃ n. apply joinable_refl. Qed.

The terms that are numerals

The carrier of the Church naturals, and what its instances cite: the witness that the programs stay numerals, and the denotation facts, all spoken in joinable alone.
Definition ChurchNat : Type := { t : term | ∃ n : nat, joinable t (church n) }.

Definition denotes (c : ChurchNat) (n : nat) : Prop := joinable (proj1_sig c) (church n).

Lemma succ_wit : ∀ t : term, (∃ n : nat, joinable t (church n)) →
  ∃ n : nat, joinable (App SUCC t) (church n).
Proof.
  intros t [n Hn]. ∃ (S n).
  apply (joinable_trans _ (App SUCC (church n))).
  - apply joinable_app; [ apply joinable_refl | exact Hn ].
  - exact (succ_join n).
Qed.

Lemma add_wit : ∀ t u : term,
  (∃ n : nat, joinable t (church n)) → (∃ n : nat, joinable u (church n)) →
  ∃ n : nat, joinable (App (App PLUS t) u) (church n).
Proof.
  intros t u [a Ha] [b Hb]. ∃ ((a + b)%nat).
  apply (joinable_trans _ (App (App PLUS (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Ha ] | exact Hb ].
  - exact (plus_join a b).
Qed.

Lemma mul_wit : ∀ t u : term,
  (∃ n : nat, joinable t (church n)) → (∃ n : nat, joinable u (church n)) →
  ∃ n : nat, joinable (App (App MULT t) u) (church n).
Proof.
  intros t u [a Ha] [b Hb]. ∃ ((a × b)%nat).
  apply (joinable_trans _ (App (App MULT (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Ha ] | exact Hb ].
  - exact (mult_join a b).
Qed.

Lemma denotes_total : ∀ c : ChurchNat, ∃ n : nat, denotes c n.
Proof. intros c. exact (proj2_sig c). Qed.

Lemma denotes_sur : ∀ n : nat, ∃ c : ChurchNat, denotes c n.
Proof. intros n. ∃ (exist _ (church n) (is_numeral n)). apply joinable_refl. Qed.

Lemma denotes_unique : ∀ (c : ChurchNat) (n m : nat), denotes c n → denotes c m → n = m.
Proof. intros c n m Hn Hm. exact (church_injective n m (joinable_trans _ _ _ (joinable_sym _ _ Hn) Hm)). Qed.

Lemma denotes_faithful : ∀ (c d : ChurchNat) (n m : nat),
  denotes c n → denotes d m → n = m → joinable (proj1_sig c) (proj1_sig d).
Proof.
  intros c d n m Hc Hd Hnm. subst m.
  exact (joinable_trans _ _ _ Hc (joinable_sym _ _ Hd)).
Qed.

Lemma denotes_join : ∀ (c d : ChurchNat) (n : nat),
  joinable (proj1_sig c) (proj1_sig d) → denotes c n → denotes d n.
Proof. intros c d n Hcd Hc. exact (joinable_trans _ _ _ (joinable_sym _ _ Hcd) Hc). Qed.

Lemma denotes_succ : ∀ (t : term) (w : ∃ n : nat, joinable t (church n)) (n : nat),
  joinable t (church n) → joinable (App SUCC t) (church (S n)).
Proof.
  intros t w n Hc.
  apply (joinable_trans _ (App SUCC (church n))).
  - apply joinable_app; [ apply joinable_refl | exact Hc ].
  - exact (succ_join n).
Qed.

Lemma denotes_add : ∀ (t u : term) (a b : nat),
  joinable t (church a) → joinable u (church b) →
  joinable (App (App PLUS t) u) (church ((a + b)%nat)).
Proof.
  intros t u a b Hc Hd.
  apply (joinable_trans _ (App (App PLUS (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Hc ] | exact Hd ].
  - exact (plus_join a b).
Qed.

Lemma denotes_mul : ∀ (t u : term) (a b : nat),
  joinable t (church a) → joinable u (church b) →
  joinable (App (App MULT t) u) (church ((a × b)%nat)).
Proof.
  intros t u a b Hc Hd.
  apply (joinable_trans _ (App (App MULT (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Hc ] | exact Hd ].
  - exact (mult_join a b).
Qed.
