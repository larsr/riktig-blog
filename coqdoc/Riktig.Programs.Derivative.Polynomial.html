#[local] Notation "'X'" := PolyVar.
#[local] Infix "⊕" := PolySum (at level 50, left associativity).
#[local] Infix "⊗" := PolyProduct (at level 40, left associativity).
#[local] Infix "▷" := PolyCompose (at level 45, right associativity).


Definition cubic : Polynomial Q :=
  X ⊗ X ⊗ X ⊕ PolyConst (-2#1)%Q ⊗ X ⊕ PolyConst (7#1)%Q.

Example cubic_at_two : poly_eval cubic (2#1)%Q == (11#1)%Q.
Proof. vm_compute. reflexivity. Qed.

Three x squared minus two, at two: ten.
Example cubic_slope_at_two : poly_deriv cubic (2#1)%Q == (10#1)%Q.
Proof. vm_compute. reflexivity. Qed.

Example cubic_slope_at_zero : poly_deriv cubic (0#1)%Q == (-2#1)%Q.
Proof. vm_compute. reflexivity. Qed.

Ten is the derivative, not a number that agrees with it: the citation is poly_deriv_is_deriv, and lm_scale_map_proper carries the computed slope into the statement.
Theorem cubic_derivative_at_two :
  is_deriv (poly_eval cubic) (2#1)%Q (lm_scale_map (10#1)%Q).
Proof.
  apply (is_deriv_L_equ (poly_eval cubic) (2#1)%Q
           (lm_scale_map (poly_deriv cubic (2#1)%Q)) (lm_scale_map (10#1)%Q)).
  - exact (lm_scale_map_proper _ _ cubic_slope_at_two).
  - exact (poly_deriv_is_deriv cubic (2#1)%Q).
Qed.


Definition affine_chain : Polynomial Q :=
  PolyConst (3#1)%Q ⊗ X ▷ (PolyConst (2#1)%Q ⊗ X ⊕ PolyConst (5#1)%Q).

Example affine_chain_at_one : poly_eval affine_chain (1#1)%Q == (21#1)%Q.
Proof. vm_compute. reflexivity. Qed.

Example affine_chain_slope : poly_deriv affine_chain (7#1)%Q == (6#1)%Q.
Proof. vm_compute. reflexivity. Qed.

Theorem affine_chain_derivative_at_seven :
  is_deriv (poly_eval affine_chain) (7#1)%Q (lm_scale_map (6#1)%Q).
Proof.
  apply (is_deriv_L_equ (poly_eval affine_chain) (7#1)%Q
           (lm_scale_map (poly_deriv affine_chain (7#1)%Q))
           (lm_scale_map (6#1)%Q)).
  - exact (lm_scale_map_proper _ _ affine_chain_slope).
  - exact (poly_deriv_is_deriv affine_chain (7#1)%Q).
Qed.


Definition affine_chain_expr : Expr Q Q :=
  ExprComp (ExprLinear (lm_scale_map (3#1)%Q))
    (ExprSum (ExprLinear (lm_scale_map (2#1)%Q)) (ExprConst (5#1)%Q)).

Example affine_chain_expr_denotes : ∀ x : Q,
  denote affine_chain_expr x == ((3#1) × ((2#1) × x + (5#1)))%Q.
Proof. intros x. reflexivity. Qed.

Example affine_chain_expr_slope :
  symbolic_deriv affine_chain_expr (7#1)%Q (1#1)%Q == (6#1)%Q.
Proof. vm_compute. reflexivity. Qed.

Lemma affine_chain_expr_bounded : expr_bounded affine_chain_expr.
Proof.
  split; [| split ].
  - exact (lm_scale_map_bounded (3#1)%Q).
  - exact (lm_scale_map_bounded (2#1)%Q).
  - exact I.
Qed.

Theorem affine_chain_expr_derivative :
  is_deriv (denote affine_chain_expr) (7#1)%Q
    (symbolic_deriv affine_chain_expr (7#1)%Q).
Proof.
  exact (symbolic_deriv_is_deriv affine_chain_expr affine_chain_expr_bounded
           (7#1)%Q).
Qed.

The two recursions return the same arrow, and nobody compared them: they are derivatives of one function at one point, and Theory/Derivative.v's deriv_unique says a function has one derivative. The computation two lemmas up says what the number is; this says why the two ways of getting it cannot disagree.
Lemma the_two_chains_denote_one_function : ∀ x : Q,
  denote affine_chain_expr x == poly_eval affine_chain x.
Proof. intros x. reflexivity. Qed.

Corollary the_two_recursions_agree :
  symbolic_deriv affine_chain_expr (7#1)%Q == lm_scale_map (6#1)%Q.
Proof.
  exact (deriv_unique (denote affine_chain_expr) (7#1)%Q
           (symbolic_deriv affine_chain_expr (7#1)%Q) (lm_scale_map (6#1)%Q)
           affine_chain_expr_derivative
           (is_deriv_fun_equ (poly_eval affine_chain)
              (denote affine_chain_expr) (7#1)%Q (lm_scale_map (6#1)%Q)
              (fun x : Q ⇒ symmetry (the_two_chains_denote_one_function x))
              affine_chain_derivative_at_seven)).
Qed.
