Definition Mask (n : nat) : Type := N.

Section MaskBlade.
  Context {n : nat}.

  Definition mask_bit (i : Fin.t n) : N := N.of_nat (fin_to_nat i).

  #[export] Instance MaskBlade_EqRel : EqRel (Mask n) :=
    fun x y ⇒ ∀ i : Fin.t n,
      N.testbit x (mask_bit i) = N.testbit y (mask_bit i).
  #[export] Instance MaskBlade_Zero : Zero (Mask n) := 0%N.
  #[export] Instance MaskBlade_One : One (Mask n) := N.ones (N.of_nat n).
  #[export] Instance MaskBlade_Add : HasAdd (Mask n) := N.lxor.
  #[export] Instance MaskBlade_Mult : HasMult (Mask n) := N.land.
  #[export] Instance MaskBlade_Neg : HasNeg (Mask n) := fun x ⇒ x.

  #[export] Instance MaskBlade_Equivalence : Equivalence (@equ (Mask n) _).
  Proof.
    constructor.
    - intros x i. reflexivity.
    - intros x y Hxy i. symmetry. exact (Hxy i).
    - intros x y z Hxy Hyz i.
      transitivity (N.testbit y (mask_bit i)); [ exact (Hxy i) | exact (Hyz i) ].
  Qed.

  Definition mask_member (x : Mask n) (i : Fin.t n) : bool :=
    N.testbit x (mask_bit i).

  #[export] Instance mask_member_proper :
    Proper (equ ==> eq ==> equ) mask_member.
  Proof.
    intros x y Hxy i i' Hii'. rewrite <- Hii'. exact (Hxy i).
  Qed.

  Lemma mask_member_faithful : ∀ x y : Mask n,
    (∀ i : Fin.t n, mask_member x i == mask_member y i) → x == y.
  Proof. intros x y Hxy. exact Hxy. Qed.

  Lemma mask_member_empty : ∀ i : Fin.t n, mask_member 0 i == (0 : bool).
  Proof. intros i. exact (N.bits_0 (mask_bit i)). Qed.

  Lemma mask_bit_below_width : ∀ i : Fin.t n, (mask_bit i < N.of_nat n)%N.
  Proof.
    intros i. unfold mask_bit.
    assert (Hbound := fin_to_nat_bound n i). lia.
  Qed.

  Lemma mask_member_full : ∀ i : Fin.t n, mask_member 1 i == (1 : bool).
  Proof.
    intros i. exact (N.ones_spec_low (N.of_nat n) (mask_bit i)
                       (mask_bit_below_width i)).
  Qed.

  Lemma mask_member_add : ∀ (x y : Mask n) (i : Fin.t n),
    mask_member (x + y) i == mask_member x i + mask_member y i.
  Proof. intros x y i. exact (N.lxor_spec x y (mask_bit i)). Qed.

  Lemma mask_member_mult : ∀ (x y : Mask n) (i : Fin.t n),
    mask_member (x × y) i == mask_member x i × mask_member y i.
  Proof. intros x y i. exact (N.land_spec x y (mask_bit i)). Qed.

  Lemma mask_member_neg : ∀ (x : Mask n) (i : Fin.t n),
    mask_member (- x) i == - mask_member x i.
  Proof. intros x i. reflexivity. Qed.

  Lemma mask_member_truncate : ∀ (x : Mask n) (i : Fin.t n),
    mask_member (x × 1) i == mask_member x i.
  Proof.
    intros x i.
    rewrite (mask_member_mult x 1 i), (mask_member_full i).
    apply Bool.andb_true_r.
  Qed.

  Lemma mask_agree_after_truncation : ∀ x y : Mask n,
    x × 1 = y × 1 → x == y.
  Proof.
    intros x y Hxy. apply mask_member_faithful. intros i.
    rewrite <- (mask_member_truncate x i), <- (mask_member_truncate y i), Hxy.
    reflexivity.
  Qed.

  #[export] Instance MaskBlade_Blade : Blade (Fin.t n) bool (Mask n) :=
    {| Blade.setoid := MaskBlade_Equivalence;
       Blade.member := mask_member;
       Blade.member_proper := mask_member_proper;
       Blade.member_faithful := mask_member_faithful;
       Blade.member_empty := mask_member_empty;
       Blade.member_full := mask_member_full;
       Blade.member_symmetric_difference := mask_member_add;
       Blade.member_intersection := mask_member_mult;
       Blade.member_complement := mask_member_neg |}.

  Definition mask_generator (i : Fin.t n) : Mask n := N.setbit 0 (mask_bit i).

End MaskBlade.

Example word_families_are_not_all_proper :
  ¬ (∀ f : Mask 1 → bool, Proper (equ ==> equ) f).
Proof.
  intros Hproper.
  assert (Hsame : (0%N : Mask 1) == 2%N)
    by (apply mask_agree_after_truncation; vm_compute; reflexivity).
  pose proof (Hproper (fun x : Mask 1 ⇒ N.eqb x 0) _ _ Hsame) as Hbit.
  vm_compute in Hbit. discriminate Hbit.
Qed.

Example member_at_both_carriers (n : nat)
    (v : Vector.t bool n) (x : Mask n) (i : Fin.t n) : bool × bool :=
  (member v i, member x i).

Example shift_at_both_carriers (n : nat)
    (v w : Vector.t bool n) (x y : Mask n) :
  (v + (v + w) == w) ∧ (x + (x + y) == y) :=
  conj (blade_shift v w) (blade_shift x y).
