Library Riktig.Impl.BinaryNumberN

From Riktig.Structure Require Export BinRec.
From Riktig.Lib Require Export Coq.N.

Coq's binary naturals are binary numbers with recursion: the two appendings are N.double and N.succ_double, and the recursion is N.binary_rect. Every cited fact is Lib.Coq.N's.
Instance N_EqRel : EqRel N := eq.
Instance N_Zero : Zero N := 0%N.
Instance N_HasBit0 : HasBit0 N := N.double.
Instance N_HasBit1 : HasBit1 N := N.succ_double.
Instance N_HasBinRec : HasBinRec N := binrec_N.

Instance N_BinaryNumber : BinaryNumber N N_Zero N_HasBit0 N_HasBit1 := {|
  BinaryNumber.equivalence := eq_equivalence;
  BinaryNumber.bit0_proper := fun n m E ⇒ f_equal N.double E;
  BinaryNumber.bit1_proper := fun n m E ⇒ f_equal N.succ_double E;
  BinaryNumber.bit0_injective := N.double_inj;
  BinaryNumber.bit1_injective := N.succ_double_inj;
  BinaryNumber.bit0_zero := eq_refl;
  BinaryNumber.bit1_not_zero := succ_double_not_zero_N;
  BinaryNumber.bit0_ne_bit1 := double_ne_succ_double_N;
  BinaryNumber.bin_induction := fun P _ Hz H0 H1 ⇒ N.binary_ind P Hz H0 H1;
|}.

Instance N_BinRec : BinRec N N_HasBinRec := {|
  BinRec.binrec_proper := fun A eA EA a f0 f1 _ _ n n' E ⇒
    eq_ind n (fun m ⇒ binrec_N A a f0 f1 n == binrec_N A a f0 f1 m) (reflexivity _) n' E;
  BinRec.binrec_ext := fun A eA EA a a' f0 f0' f1 f1' Hf0 Hf1 Ha E0 E1 n ⇒
    binrec_N_ext A equ a a' f0 f0' f1 f1' _ _ (fun k x y ⇒ Hf0 k x y) (fun k x y ⇒ Hf1 k x y) Ha E0 E1 n;
  BinRec.binrec_zero := fun A eA EA a f0 f1 ⇒ reflexivity _;
  BinRec.binrec_bit0 := fun A eA EA a f0 f1 n Hn ⇒
    eq_ind _ (fun t ⇒ t == _) (reflexivity _) _ (eq_sym (binrec_N_double A a f0 f1 n Hn));
  BinRec.binrec_bit1 := fun A eA EA a f0 f1 n ⇒
    eq_ind _ (fun t ⇒ t == _) (reflexivity _) _ (eq_sym (binrec_N_succ_double A a f0 f1 n));
|}.
