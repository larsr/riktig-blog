Library Riktig.Theory.Fin

From Coq Require Import Permutation.
From Riktig Require Export Structure.Fin Lib.ListFacts.
From Riktig Require Import Lib.Calc.


Module Swap.
  Class Swap {I : Type} {eqdec : EqDec I} (swap : I → I → I → I) := {
    swap_at_i : ∀ i j : I, swap i j i = j;
    swap_at_j : ∀ i j : I, i ≠ j → swap i j j = i;
    swap_ne : ∀ i j r : I, r ≠ i → r ≠ j → swap i j r = r;
  }.
End Swap.
Export Swap.
Hint Mode Swap - - ! : typeclass_instances.
Arguments swap_at_i {I eqdec} swap {Swap} i j.
Arguments swap_at_j {I eqdec} swap {Swap} i j _.
Arguments swap_ne {I eqdec} swap {Swap} i j r _ _.

Section SwapTheory.
  Context {I : Type} {eqdec : EqDec I}.
  Context (swap : I → I → I → I) {Sw : Swap swap}.

  Lemma swap_invol : ∀ i j x : I, swap i j (swap i j x) = x.
  Proof.
    intros i j x.
    destruct (dec (x = i)) as [Hxi | Hxi].
    - rewrite Hxi, (swap_at_i swap i j).
      destruct (dec (i = j)) as [Hij | Hij].
      + rewrite <- Hij. exact (swap_at_i swap i i).
      + exact (swap_at_j swap i j Hij).
    - destruct (dec (x = j)) as [Hxj | Hxj].
      + assert (Hij : i ≠ j).
        { intros Heq. apply Hxi. rewrite Hxj. symmetry. exact Heq. }
        rewrite Hxj.
        calc (swap i j (swap i j j))
          == (swap i j i) { (swap_at_j swap i j Hij) }
          == j { (swap_at_i swap i j) }.
      + rewrite (swap_ne swap i j x Hxi Hxj), (swap_ne swap i j x Hxi Hxj).
        reflexivity.
  Qed.

  Lemma swap_inj : ∀ i j x y : I, swap i j x = swap i j y → x = y.
  Proof.
    intros i j x y Heq.
    calc x
      == (swap i j (swap i j x)) { (swap_invol i j x) }
      == (swap i j (swap i j y)) { Heq }
      == y { (swap_invol i j y) }.
  Qed.

  Lemma swap_symmetric : ∀ i j x : I, swap i j x = swap j i x.
  Proof.
    intros i j x.
    destruct (dec (x = i)) as [Hxi | Hxi].
    - rewrite Hxi, (swap_at_i swap i j).
      destruct (dec (i = j)) as [Hij | Hij].
      + rewrite Hij, (swap_at_i swap j j). reflexivity.
      + exact (eq_sym (swap_at_j swap j i (fun Heq ⇒ Hij (eq_sym Heq)))).
    - destruct (dec (x = j)) as [Hxj | Hxj].
      + rewrite Hxj.
        destruct (dec (i = j)) as [Hij | Hij].
        × rewrite Hij, (swap_at_i swap j j). reflexivity.
        × rewrite (swap_at_j swap i j Hij).
          exact (eq_sym (swap_at_i swap j i)).
      + rewrite (swap_ne swap i j x Hxi Hxj), (swap_ne swap j i x Hxj Hxi).
        reflexivity.
  Qed.

  Lemma swap_conj : ∀ i j k : I,
    i ≠ j → i ≠ k → j ≠ k →
    ∀ x : I,
    swap i j x = swap i k (swap k j (swap i k x)).
  Proof.
    intros i j k Hij Hik Hjk x.
    assert (Hji : j ≠ i) by (intros Heq; exact (Hij (eq_sym Heq))).
    assert (Hki : k ≠ i) by (intros Heq; exact (Hik (eq_sym Heq))).
    assert (Hkj : k ≠ j) by (intros Heq; exact (Hjk (eq_sym Heq))).
    destruct (dec (x = i)) as [Hxi | Hxi].
    - rewrite Hxi.
      calc (swap i j i)
        == j { (swap_at_i swap i j) }
        == (swap i k j) { (swap_ne swap i k j Hji Hjk) }
        == (swap i k (swap k j k)) { (swap_at_i swap k j) }
        == (swap i k (swap k j (swap i k i))) { (swap_at_i swap i k) }.
    - destruct (dec (x = j)) as [Hxj | Hxj].
      + rewrite Hxj.
        calc (swap i j j)
          == i { (swap_at_j swap i j Hij) }
          == (swap i k k) { (swap_at_j swap i k Hik) }
          == (swap i k (swap k j j)) { (swap_at_j swap k j Hkj) }
          == (swap i k (swap k j (swap i k j))) { (swap_ne swap i k j Hji Hjk) }.
      + destruct (dec (x = k)) as [Hxk | Hxk].
        × rewrite Hxk.
          calc (swap i j k)
            == k { (swap_ne swap i j k Hki Hkj) }
            == (swap i k i) { (swap_at_i swap i k) }
            == (swap i k (swap k j i)) { (swap_ne swap k j i Hik Hij) }
            == (swap i k (swap k j (swap i k k))) { (swap_at_j swap i k Hik) }.
        × calc (swap i j x)
            == x { (swap_ne swap i j x Hxi Hxj) }
            == (swap i k x) { (swap_ne swap i k x Hxi Hxk) }
            == (swap i k (swap k j x)) { (swap_ne swap k j x Hxk Hxj) }
            == (swap i k (swap k j (swap i k x))) { (swap_ne swap i k x Hxi Hxk) }.
  Qed.

End SwapTheory.

Lemma finindex_cardinality_unique : ∀ (n m : nat) (I : Type)
    {eqdec : EqDec I} (F1 : FinIndex n I) (F2 : FinIndex m I),
  n = m.
Proof.
  intros n m I eqdec F1 F2.
  rewrite <- (@enum_length n I eqdec F1), <- (@enum_length m I eqdec F2).
  apply Permutation_length.
  apply NoDup_Permutation.
  - exact (@enum_NoDup n I eqdec F1).
  - exact (@enum_NoDup m I eqdec F2).
  - intros x.
    split; intros _.
    + exact (@enum_complete m I eqdec F2 x).
    + exact (@enum_complete n I eqdec F1 x).
Qed.
