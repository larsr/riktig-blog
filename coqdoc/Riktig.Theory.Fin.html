Section SwapTheory.
  Context {I : Type} {eI : EqRel I} {IE : Equivalence (@equ I eI)}
    {Ieq : EquDec I}.
  Context (swap : I → I → I → I) {Sw : Swap swap}.

  #[export] Instance swap_proper : ∀ i j : I,
    Proper (equ ==> equ) (swap i j).
  Proof.
    intros i j x y Hxy.
    destruct (dec (x == i)) as [Hxi | Hxi].
    - calc (swap i j x)
        == j { (swap_at_i swap i j x Hxi) }
        == (swap i j y) { (symmetry (swap_at_i swap i j y
                            (transitivity (symmetry Hxy) Hxi))) }.
    - destruct (dec (x == j)) as [Hxj | Hxj].
      + assert (Hij : ¬ (i == j)).
        { intros Hij. exact (Hxi (transitivity Hxj (symmetry Hij))). }
        calc (swap i j x)
          == i { (swap_at_j swap i j x Hij Hxj) }
          == (swap i j y) { (symmetry (swap_at_j swap i j y Hij
                              (transitivity (symmetry Hxy) Hxj))) }.
      + calc (swap i j x)
          == x { (swap_ne swap i j x Hxi Hxj) }
          == y { Hxy }
          == (swap i j y) { (symmetry (swap_ne swap i j y
                              (fun Hc ⇒ Hxi (transitivity Hxy Hc))
                              (fun Hc ⇒ Hxj (transitivity Hxy Hc)))) }.
  Qed.

  Lemma swap_invol : ∀ i j x : I, swap i j (swap i j x) == x.
  Proof.
    intros i j x.
    destruct (dec (x == i)) as [Hxi | Hxi].
    - destruct (dec (i == j)) as [Hij | Hij].
      + calc (swap i j (swap i j x))
          == j { (swap_at_i swap i j (swap i j x)
                   (transitivity (swap_at_i swap i j x Hxi) (symmetry Hij))) }
          == i { (symmetry Hij) }
          == x { (symmetry Hxi) }.
      + calc (swap i j (swap i j x))
          == i { (swap_at_j swap i j (swap i j x) Hij
                   (swap_at_i swap i j x Hxi)) }
          == x { (symmetry Hxi) }.
    - destruct (dec (x == j)) as [Hxj | Hxj].
      + assert (Hij : ¬ (i == j)).
        { intros Hij. exact (Hxi (transitivity Hxj (symmetry Hij))). }
        calc (swap i j (swap i j x))
          == j { (swap_at_i swap i j (swap i j x)
                   (swap_at_j swap i j x Hij Hxj)) }
          == x { (symmetry Hxj) }.
      + pose proof (swap_ne swap i j x Hxi Hxj) as H1.
        calc (swap i j (swap i j x))
          == (swap i j x)
               { (swap_ne swap i j (swap i j x)
                   (fun Hc ⇒ Hxi (transitivity (symmetry H1) Hc))
                   (fun Hc ⇒ Hxj (transitivity (symmetry H1) Hc))) }
          == x { H1 }.
  Qed.

  Lemma swap_inj : ∀ i j x y : I, swap i j x == swap i j y → x == y.
  Proof.
    intros i j x y Heq.
    calc x
      == (swap i j (swap i j x)) { (symmetry (swap_invol i j x)) }
      == (swap i j (swap i j y)) { (swap_proper i j _ _ Heq) }
      == y { (swap_invol i j y) }.
  Qed.

  Lemma swap_symmetric : ∀ i j x : I, swap i j x == swap j i x.
  Proof.
    intros i j x.
    destruct (dec (x == i)) as [Hxi | Hxi].
    - destruct (dec (i == j)) as [Hij | Hij].
      + calc (swap i j x)
          == j { (swap_at_i swap i j x Hxi) }
          == (swap j i x) { (symmetry (swap_at_i swap j i x
                              (transitivity Hxi Hij))) }.
      + assert (Hji : ¬ (j == i)) by (intros Hc; exact (Hij (symmetry Hc))).
        calc (swap i j x)
          == j { (swap_at_i swap i j x Hxi) }
          == (swap j i x) { (symmetry (swap_at_j swap j i x Hji Hxi)) }.
    - destruct (dec (x == j)) as [Hxj | Hxj].
      + assert (Hij : ¬ (i == j)).
        { intros Hij. exact (Hxi (transitivity Hxj (symmetry Hij))). }
        calc (swap i j x)
          == i { (swap_at_j swap i j x Hij Hxj) }
          == (swap j i x) { (symmetry (swap_at_i swap j i x Hxj)) }.
      + calc (swap i j x)
          == x { (swap_ne swap i j x Hxi Hxj) }
          == (swap j i x) { (symmetry (swap_ne swap j i x Hxj Hxi)) }.
  Qed.

  Lemma swap_conj : ∀ i j k : I,
    ¬ (i == j) → ¬ (i == k) → ¬ (j == k) →
    ∀ x : I,
    swap i j x == swap i k (swap k j (swap i k x)).
  Proof.
    intros i j k Hij Hik Hjk x.
    assert (Hji : ¬ (j == i)) by (intros Heq; exact (Hij (symmetry Heq))).
    assert (Hki : ¬ (k == i)) by (intros Heq; exact (Hik (symmetry Heq))).
    assert (Hkj : ¬ (k == j)) by (intros Heq; exact (Hjk (symmetry Heq))).
    destruct (dec (x == i)) as [Hxi | Hxi].
    - rewrite Hxi.
      calc (swap i j i)
        == j { (swap_at_i swap i j i (reflexivity i)) }
        == (swap i k j) { (swap_ne swap i k j Hji Hjk) }
        == (swap i k (swap k j k)) { (swap_at_i swap k j k (reflexivity k)) }
        == (swap i k (swap k j (swap i k i))) { (swap_at_i swap i k i (reflexivity i)) }.
    - destruct (dec (x == j)) as [Hxj | Hxj].
      + rewrite Hxj.
        calc (swap i j j)
          == i { (swap_at_j swap i j j Hij (reflexivity j)) }
          == (swap i k k) { (swap_at_j swap i k k Hik (reflexivity k)) }
          == (swap i k (swap k j j)) { (swap_at_j swap k j j Hkj (reflexivity j)) }
          == (swap i k (swap k j (swap i k j))) { (swap_ne swap i k j Hji Hjk) }.
      + destruct (dec (x == k)) as [Hxk | Hxk].
        × rewrite Hxk.
          calc (swap i j k)
            == k { (swap_ne swap i j k Hki Hkj) }
            == (swap i k i) { (swap_at_i swap i k i (reflexivity i)) }
            == (swap i k (swap k j i)) { (swap_ne swap k j i Hik Hij) }
            == (swap i k (swap k j (swap i k k))) { (swap_at_j swap i k k Hik (reflexivity k)) }.
        × calc (swap i j x)
            == x { (swap_ne swap i j x Hxi Hxj) }
            == (swap i k x) { (swap_ne swap i k x Hxi Hxk) }
            == (swap i k (swap k j x)) { (swap_ne swap k j x Hxk Hxj) }
            == (swap i k (swap k j (swap i k x))) { (swap_ne swap i k x Hxi Hxk) }.
  Qed.

End SwapTheory.

Section Rearrangements.
  Context {I : Type} {eI : EqRel I} {IE : Equivalence (@equ I eI)}
    {Ieq : EquDec I}.

  Lemma swap_exists : ∃ swap : I → I → I → I, Swap swap.
  Proof.
    ∃ (fun i j x ⇒ if dec (x == i) then j else if dec (x == j) then i else x).
    constructor; intros i j x; cbn beta.
    - intros Hxi.
      destruct (dec (x == i)) as [_ | Hne]; [reflexivity | destruct (Hne Hxi)].
    - intros Hij Hxj.
      destruct (dec (x == i)) as [Hxi | _].
      + destruct (Hij (transitivity (symmetry Hxi) Hxj)).
      + destruct (dec (x == j)) as [_ | Hne]; [reflexivity | destruct (Hne Hxj)].
    - intros Hxi Hxj.
      destruct (dec (x == i)) as [Hc | _]; [destruct (Hxi Hc) |].
      destruct (dec (x == j)) as [Hc | _]; [destruct (Hxj Hc) | reflexivity].
  Qed.

  Lemma family_update_exists : ∀ {A : Type} {eA : EqRel A},
    Equivalence (@equ A eA) →
    ∀ (f : I → A) (i : I) (y : A), Proper (equ ==> equ) f →
    ∃ g : I → A,
      Proper (equ ==> equ) g ∧ g i == y ∧
      (∀ x : I, ¬ (x == i) → g x == f x).
  Proof.
    intros A eA AE f i y Pf.
    ∃ (fun x ⇒ if dec (x == i) then y else f x).
    split; [| split]; cbn beta.
    - intros x x' Hxx'.
      destruct (dec (x == i)) as [Hxi | Hxi];
        destruct (dec (x' == i)) as [Hx'i | Hx'i].
      + reflexivity.
      + destruct (Hx'i (transitivity (symmetry Hxx') Hxi)).
      + destruct (Hxi (transitivity Hxx' Hx'i)).
      + exact (Pf x x' Hxx').
    - destruct (dec (i == i)) as [_ | Hne];
        [reflexivity | destruct (Hne (reflexivity i))].
    - intros x Hxi.
      destruct (dec (x == i)) as [Hc | _]; [destruct (Hxi Hc) | reflexivity].
  Qed.

  Lemma identifies_or_separates_at : ∀ (f : I → I) (a : I) (l : list I),
    Proper (equ ==> equ) f →
    (∃ j : I, InA equ j l ∧ ¬ (a == j) ∧ f a == f j)
    ∨ (∀ j : I, InA equ j l → f a == f j → a == j).
  Proof.
    intros f a l Pf; induction l as [| b l IH].
    - right. intros j Hj. inversion Hj.
    - destruct IH as [[j [Hjl [Haj Hfaj]]] | Hsep].
      + left. ∃ j. split; [exact (InA_cons_tl b Hjl) | split; assumption].
      + destruct (dec (a == b)) as [Hab | Hab].
        × right. intros j Hj Hfj.
          destruct (proj1 (InA_cons equ j b l) Hj) as [Hjb | Hjl].
          -- exact (transitivity Hab (symmetry Hjb)).
          -- exact (Hsep j Hjl Hfj).
        × destruct (dec (f a == f b)) as [Hfab | Hfab].
          -- left. ∃ b.
             split; [exact (InA_cons_hd l (reflexivity b)) | split; assumption].
          -- right. intros j Hj Hfj.
             destruct (proj1 (InA_cons equ j b l) Hj) as [Hjb | Hjl].
             ++ destruct (Hfab (transitivity Hfj (Pf j b Hjb))).
             ++ exact (Hsep j Hjl Hfj).
  Qed.

  Lemma identifies_or_separates_on : ∀ (f : I → I) (l : list I),
    Proper (equ ==> equ) f →
    (∃ i j : I, ¬ (i == j) ∧ f i == f j)
    ∨ (∀ i j : I, InA equ i l → InA equ j l → f i == f j → i == j).
  Proof.
    intros f l Pf; induction l as [| b l IH].
    - right. intros i j Hi. inversion Hi.
    - destruct IH as [Hid | Hsep]; [left; exact Hid |].
      destruct (identifies_or_separates_at f b l Pf) as [[j [_ [Hbj Hfbj]]] | Hsepb].
      + left. ∃ b, j. split; assumption.
      + right. intros i j Hi Hj Hfij.
        destruct (proj1 (InA_cons equ i b l) Hi) as [Hib | Hil];
          destruct (proj1 (InA_cons equ j b l) Hj) as [Hjb | Hjl].
        × exact (transitivity Hib (symmetry Hjb)).
        × assert (Hbj : b == j)
            by exact (Hsepb j Hjl (transitivity (symmetry (Pf i b Hib)) Hfij)).
          exact (transitivity Hib Hbj).
        × assert (Hbi : b == i)
            by exact (Hsepb i Hil
                        (transitivity (symmetry (Pf j b Hjb)) (symmetry Hfij))).
          exact (transitivity (symmetry Hbi) (symmetry Hjb)).
        × exact (Hsep i j Hil Hjl Hfij).
  Qed.

  Lemma finindex_identifies_or_injective : ∀ {n : nat}
      {IFin : FinIndex n I} (f : I → I),
    Proper (equ ==> equ) f →
    (∃ i j : I, ¬ (i == j) ∧ f i == f j) ∨ Injective I I f.
  Proof.
    intros n IFin f Pf.
    destruct (identifies_or_separates_on f enum Pf) as [Hid | Hsep];
      [left; exact Hid |].
    right. intros a b Hab.
    exact (Hsep a b (enum_complete a) (enum_complete b) Hab).
  Qed.

End Rearrangements.

Lemma finindex_cardinality_unique : ∀ (n m : nat) (I : Type)
    {eI : EqRel I} {Ieq : EquDec I}
    (F1 : FinIndex n I) (F2 : FinIndex m I),
  n = m.
Proof.
  intros n m I eI Ieq F1 F2.
  rewrite <- (@enum_length n I eI Ieq F1), <- (@enum_length m I eI Ieq F2).
  apply (PermutationA_length (eqA := @equ I eI)).
  apply NoDupA_equivlistA_PermutationA.
  - exact (@enum_equivalence n I eI Ieq F1).
  - exact (@enum_NoDup n I eI Ieq F1).
  - exact (@enum_NoDup m I eI Ieq F2).
  - intros x.
    split; intros _.
    + exact (@enum_complete m I eI Ieq F2 x).
    + exact (@enum_complete n I eI Ieq F1 x).
Qed.
