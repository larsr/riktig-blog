Section IsoCompose.
  Context {A B C : Type} {eqA : EqRel A} {eqB : EqRel B} {eqC : EqRel C}
          `{!Equivalence eqA} `{!Equivalence eqB} `{!Equivalence eqC}
          {f : A → B} {g : B → A} {h : B → C} {k : C → B}
          `{!Iso A B f g} `{!Iso B C h k}.

  #[export] Instance Iso_compose : Iso A C (fun a ⇒ h (f a)) (fun c ⇒ g (k c)).
  Proof.
    split.
    - intros a a' Haa'. rewrite Haa'. reflexivity.
    - intros c c' Hcc'. rewrite Hcc'. reflexivity.
    - intros a. rewrite (from_to (to := h) (f a)). exact (from_to (to := f) a).
    - intros c. rewrite (to_from (to := f) (k c)). exact (to_from (to := h) c).
  Qed.
End IsoCompose.
