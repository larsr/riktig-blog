An isomorphism of setoids: two functions mutually inverse up to ==.
Module Inverse.
  Class Inverse {U V : Type} {eU : EqRel U} {eV : EqRel V}
      (f : U → V) (g : V → U) := {
    from_to : ∀ u : U, g (f u) == u;
    to_from : ∀ v : V, f (g v) == v;
  }.
End Inverse.
Notation Inverse := Inverse.Inverse.
