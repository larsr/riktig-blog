Library Riktig.Impl.Blade

From Coq Require Import Bool.
From Coq.Vectors Require Vector.


Definition Blade (n : nat) : Type := Vector.t bool n.

Definition bnil : Blade 0 := Vector.nil bool.
Definition bcons {n} (b : bool) (A : Blade n) : Blade (S n) :=
  Vector.cons bool b n A.
Definition bhead {n} (A : Blade (S n)) : bool := Vector.hd A.
Definition btail {n} (A : Blade (S n)) : Blade n := Vector.tl A.

Lemma blade_0 : ∀ A : Blade 0, A = bnil.
Proof. intros A. apply (Vector.case0 (fun v ⇒ v = bnil)). reflexivity. Qed.

Lemma blade_S : ∀ n (A : Blade (S n)), A = bcons (bhead A) (btail A).
Proof.
  intros n A.
  apply (Vector.caseS' A (fun v ⇒ v = bcons (bhead v) (btail v))).
  reflexivity.
Qed.

Fixpoint bzip (f : bool → bool → bool) {n} (A : Blade n)
    : Blade n → Blade n :=
  match A in Vector.t _ k return Blade k → Blade k with
  | Vector.nil _ ⇒ fun _ ⇒ bnil
  | Vector.cons _ a m A' ⇒
      fun B ⇒ bcons (f a (bhead B)) (bzip f A' (btail B))
  end.

Definition bxor {n} : Blade n → Blade n → Blade n := bzip xorb.
Definition band {n} : Blade n → Blade n → Blade n := bzip andb.
Definition bdiff {n} : Blade n → Blade n → Blade n :=
  bzip (fun a b ⇒ andb a (negb b)).

Lemma bzip_cons : ∀ f n a b (A B : Blade n),
  bzip f (bcons a A) (bcons b B) = bcons (f a b) (bzip f A B).
Proof. reflexivity. Qed.

Lemma bhead_bzip : ∀ f n (A B : Blade (S n)),
  bhead (bzip f A B) = f (bhead A) (bhead B).
Proof.
  intros f n A B.
  rewrite (blade_S n A), (blade_S n B), bzip_cons. reflexivity.
Qed.

Lemma btail_bzip : ∀ f n (A B : Blade (S n)),
  btail (bzip f A B) = bzip f (btail A) (btail B).
Proof.
  intros f n A B.
  rewrite (blade_S n A), (blade_S n B), bzip_cons. reflexivity.
Qed.

Lemma bxor_cons : ∀ n a b (A B : Blade n),
  bxor (bcons a A) (bcons b B) = bcons (xorb a b) (bxor A B).
Proof. reflexivity. Qed.

Lemma band_cons : ∀ n a b (A B : Blade n),
  band (bcons a A) (bcons b B) = bcons (andb a b) (band A B).
Proof. reflexivity. Qed.

Lemma bdiff_cons : ∀ n a b (A B : Blade n),
  bdiff (bcons a A) (bcons b B) = bcons (andb a (negb b)) (bdiff A B).
Proof. reflexivity. Qed.

Lemma bhead_bxor : ∀ n (A B : Blade (S n)),
  bhead (bxor A B) = xorb (bhead A) (bhead B).
Proof. intros n A B. exact (bhead_bzip xorb n A B). Qed.

Lemma btail_bxor : ∀ n (A B : Blade (S n)),
  btail (bxor A B) = bxor (btail A) (btail B).
Proof. intros n A B. exact (btail_bzip xorb n A B). Qed.

Lemma bhead_band : ∀ n (A B : Blade (S n)),
  bhead (band A B) = andb (bhead A) (bhead B).
Proof. intros n A B. exact (bhead_bzip andb n A B). Qed.

Lemma btail_band : ∀ n (A B : Blade (S n)),
  btail (band A B) = band (btail A) (btail B).
Proof. intros n A B. exact (btail_bzip andb n A B). Qed.

Lemma bhead_bdiff : ∀ n (A B : Blade (S n)),
  bhead (bdiff A B) = andb (bhead A) (negb (bhead B)).
Proof. intros n A B. exact (bhead_bzip _ n A B). Qed.

Lemma btail_bdiff : ∀ n (A B : Blade (S n)),
  btail (bdiff A B) = bdiff (btail A) (btail B).
Proof. intros n A B. exact (btail_bzip _ n A B). Qed.

Lemma blade_eq : ∀ n (A B : Blade (S n)),
  bhead A = bhead B → btail A = btail B → A = B.
Proof.
  intros n A B Hh Ht.
  rewrite (blade_S n A), (blade_S n B), Hh, Ht. reflexivity.
Qed.

Lemma bxor_involutive : ∀ n (A B : Blade n), bxor A (bxor A B) = B.
Proof.
  induction n as [| m IH]; intros A B.
  - rewrite (blade_0 (bxor A (bxor A B))), (blade_0 B). reflexivity.
  - apply blade_eq.
    + rewrite !bhead_bxor. destruct (bhead A), (bhead B); reflexivity.
    + rewrite !btail_bxor. exact (IH (btail A) (btail B)).
Qed.

Lemma bxor_comm : ∀ n (A B : Blade n), bxor A B = bxor B A.
Proof.
  induction n as [| m IH]; intros A B.
  - rewrite (blade_0 (bxor A B)), (blade_0 (bxor B A)). reflexivity.
  - apply blade_eq.
    + rewrite !bhead_bxor. apply xorb_comm.
    + rewrite !btail_bxor. exact (IH (btail A) (btail B)).
Qed.

Lemma bxor_assoc : ∀ n (A B C : Blade n),
  bxor (bxor A B) C = bxor A (bxor B C).
Proof.
  induction n as [| m IH]; intros A B C.
  - rewrite (blade_0 (bxor (bxor A B) C)), (blade_0 (bxor A (bxor B C))).
    reflexivity.
  - apply blade_eq.
    + rewrite !bhead_bxor. destruct (bhead A), (bhead B), (bhead C); reflexivity.
    + rewrite !btail_bxor. exact (IH (btail A) (btail B) (btail C)).
Qed.

Lemma bdiff_bxor_self : ∀ n (B C : Blade n), bdiff B (bxor B C) = band B C.
Proof.
  induction n as [| m IH]; intros B C.
  - rewrite (blade_0 (bdiff B (bxor B C))), (blade_0 (band B C)). reflexivity.
  - apply blade_eq.
    + rewrite bhead_bdiff, bhead_band, bhead_bxor.
      destruct (bhead B), (bhead C); reflexivity.
    + rewrite btail_bdiff, btail_band, btail_bxor. exact (IH (btail B) (btail C)).
Qed.

Lemma bxor_swap_mid : ∀ n (B C D : Blade n),
  bxor (bxor B C) D = bxor C (bxor B D).
Proof.
  intros n B C D.
  rewrite (bxor_comm n B C), bxor_assoc. reflexivity.
Qed.

Fixpoint bempty {n : nat} : Blade n :=
  match n with
  | 0 ⇒ bnil
  | S m ⇒ bcons false bempty
  end.

Lemma bhead_bempty : ∀ n, bhead (@bempty (S n)) = false.
Proof. reflexivity. Qed.

Lemma btail_bempty : ∀ n, btail (@bempty (S n)) = bempty.
Proof. reflexivity. Qed.

Definition bfull (n : nat) : Blade n := Vector.const true n.

Lemma bfull_cons : ∀ n : nat, bfull (S n) = bcons true (bfull n).
Proof. reflexivity. Qed.

Lemma bxor_bempty_l : ∀ n (A : Blade n), bxor bempty A = A.
Proof.
  induction n as [| m IH]; intros A.
  - rewrite (blade_0 (bxor bempty A)), (blade_0 A). reflexivity.
  - apply blade_eq.
    + rewrite bhead_bxor, bhead_bempty. reflexivity.
    + rewrite btail_bxor, btail_bempty. exact (IH (btail A)).
Qed.

Lemma bxor_bempty_r : ∀ n (A : Blade n), bxor A bempty = A.
Proof. intros n A. rewrite (bxor_comm n A bempty). apply bxor_bempty_l. Qed.

Lemma bxor_self : ∀ n (A : Blade n), bxor A A = bempty.
Proof.
  induction n as [| m IH]; intros A.
  - rewrite (blade_0 (bxor A A)). reflexivity.
  - apply blade_eq.
    + rewrite bhead_bxor, bhead_bempty. destruct (bhead A); reflexivity.
    + rewrite btail_bxor, btail_bempty. exact (IH (btail A)).
Qed.

Lemma bxor_shift : ∀ n (A D : Blade n), bxor (bxor A D) D = A.
Proof.
  intros n A D.
  rewrite (bxor_assoc n A D D), (bxor_self n D), (bxor_bempty_r n A).
  reflexivity.
Qed.

Lemma bdiff_bxor_shift : ∀ n (A D : Blade n),
  bdiff (bxor A D) D = bdiff A D.
Proof.
  induction n as [| m IH]; intros A D.
  - rewrite (blade_0 (bdiff (bxor A D) D)), (blade_0 (bdiff A D)). reflexivity.
  - apply blade_eq.
    + rewrite !bhead_bdiff, bhead_bxor.
      destruct (bhead A), (bhead D); reflexivity.
    + rewrite !btail_bdiff, btail_bxor. exact (IH (btail A) (btail D)).
Qed.

Lemma bdiff_bempty_l : ∀ n (A : Blade n), bdiff bempty A = bempty.
Proof.
  induction n as [| m IH]; intros A.
  - rewrite (blade_0 (bdiff bempty A)). reflexivity.
  - apply blade_eq.
    + rewrite bhead_bdiff, !bhead_bempty. reflexivity.
    + rewrite btail_bdiff, !btail_bempty. exact (IH (btail A)).
Qed.

Lemma bdiff_self : ∀ n (A : Blade n), bdiff A A = bempty.
Proof.
  induction n as [| m IH]; intros A.
  - rewrite (blade_0 (bdiff A A)). reflexivity.
  - apply blade_eq.
    + rewrite bhead_bdiff, bhead_bempty. destruct (bhead A); reflexivity.
    + rewrite btail_bdiff, btail_bempty. exact (IH (btail A)).
Qed.

Fixpoint bparity {n} : Blade n → bool :=
  match n with
  | 0 ⇒ fun _ ⇒ false
  | S m ⇒ fun A ⇒ xorb (bhead A) (bparity (btail A))
  end.

Fixpoint bpar {n} : Blade n → Blade n → bool :=
  match n with
  | 0 ⇒ fun _ _ ⇒ false
  | S m ⇒ fun A B ⇒
      xorb (andb (bhead B) (bparity (btail A))) (bpar (btail A) (btail B))
  end.

Lemma bparity_S : ∀ n (A : Blade (S n)),
  bparity A = xorb (bhead A) (bparity (btail A)).
Proof. reflexivity. Qed.

Lemma bpar_S : ∀ n (A B : Blade (S n)),
  bpar A B = xorb (andb (bhead B) (bparity (btail A))) (bpar (btail A) (btail B)).
Proof. reflexivity. Qed.

Lemma bpar_cons : ∀ n a b (A B : Blade n),
  bpar (bcons a A) (bcons b B) = xorb (andb b (bparity A)) (bpar A B).
Proof. reflexivity. Qed.

Lemma bparity_bempty : ∀ n, bparity (@bempty n) = false.
Proof.
  induction n as [| m IH].
  - reflexivity.
  - rewrite bparity_S, bhead_bempty, btail_bempty, IH. reflexivity.
Qed.

Lemma bpar_bempty_l : ∀ n (A : Blade n), bpar bempty A = false.
Proof.
  induction n as [| m IH]; intros A.
  - reflexivity.
  - rewrite bpar_S, btail_bempty, bparity_bempty, (IH (btail A)).
    destruct (bhead A); reflexivity.
Qed.

Lemma bpar_bempty_r : ∀ n (A : Blade n), bpar A bempty = false.
Proof.
  induction n as [| m IH]; intros A.
  - reflexivity.
  - rewrite bpar_S, bhead_bempty, btail_bempty, (IH (btail A)). reflexivity.
Qed.

Lemma bparity_bxor : ∀ n (A B : Blade n),
  bparity (bxor A B) = xorb (bparity A) (bparity B).
Proof.
  induction n as [| m IH]; intros A B.
  - reflexivity.
  - rewrite !bparity_S, bhead_bxor, btail_bxor, (IH (btail A) (btail B)).
    destruct (bhead A), (bhead B),
             (bparity (btail A)), (bparity (btail B));
      reflexivity.
Qed.

Lemma bpar_bxor_l : ∀ n (A A' B : Blade n),
  bpar (bxor A A') B = xorb (bpar A B) (bpar A' B).
Proof.
  induction n as [| m IH]; intros A A' B.
  - reflexivity.
  - rewrite !bpar_S, btail_bxor.
    rewrite (bparity_bxor m (btail A) (btail A')).
    rewrite (IH (btail A) (btail A') (btail B)).
    destruct (bhead B),
             (bparity (btail A)), (bparity (btail A')),
             (bpar (btail A) (btail B)), (bpar (btail A') (btail B));
      reflexivity.
Qed.

Lemma bpar_bxor_r : ∀ n (A B B' : Blade n),
  bpar A (bxor B B') = xorb (bpar A B) (bpar A B').
Proof.
  induction n as [| m IH]; intros A B B'.
  - reflexivity.
  - rewrite !bpar_S, bhead_bxor, btail_bxor.
    rewrite (IH (btail A) (btail B) (btail B')).
    destruct (bhead B), (bhead B'),
             (bparity (btail A)),
             (bpar (btail A) (btail B)), (bpar (btail A) (btail B'));
      reflexivity.
Qed.

Theorem bpar_cocycle : ∀ n (A B C : Blade n),
  xorb (bpar A B) (bpar (bxor A B) C)
  = xorb (bpar B C) (bpar A (bxor B C)).
Proof.
  intros n A B C.
  rewrite (bpar_bxor_l n A B C).
  rewrite (bpar_bxor_r n A B C).
  destruct (bpar A B), (bpar A C), (bpar B C); reflexivity.
Qed.

Definition blade3 (x1 x2 x3 : bool) : Blade 3 :=
  bcons x1 (bcons x2 (bcons x3 bnil)).

Example bpar_agrees_with_sign_table : ∀ x1 x2 x3 y1 y2 y3 : bool,
  bpar (blade3 x1 x2 x3) (blade3 y1 y2 y3)
  = xorb (andb x2 y1) (andb x3 (xorb y1 y2)).
Proof.
  intros x1 x2 x3 y1 y2 y3.
  destruct x1, x2, x3, y1, y2, y3; reflexivity.
Qed.
