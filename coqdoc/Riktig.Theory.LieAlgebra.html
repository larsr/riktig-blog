A bracket that is bilinear and alternating is antisymmetric, and one that also satisfies the Jacobi identity brackets like a derivation.

Section BilinearBracket.
  Context {F V : Type} `{Field F}.
  Context {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V}
          {Vs : HasScale F V} {VS : VectorSpace F V} {Vb : HasBracket V}.
  Context {BL : Bilinear F V V V bracket}.

  Definition bracket_left_additive : ∀ x x' y : V,
      [ x + x' , y ] == [ x , y ] + [ x' , y ] :=
    fun x x' y ⇒ additive (f := fun u : V ⇒ [ u , y ]) x x'.

  Definition bracket_right_additive : ∀ x y y' : V,
      [ x , y + y' ] == [ x , y ] + [ x , y' ] :=
    fun x y y' ⇒ additive (f := bracket x) y y'.

  Definition bracket_left_scale : ∀ (a : F) (x y : V),
      [ a *: x , y ] == a *: [ x , y ] :=
    fun a x y ⇒ homogeneous (f := fun u : V ⇒ [ u , y ]) a x.

  Definition bracket_right_scale : ∀ (a : F) (x y : V),
      [ x , a *: y ] == a *: [ x , y ] :=
    fun a x y ⇒ homogeneous (f := bracket x) a y.

  Theorem bracket_zero_l : ∀ y : V, [ (0 : V) , y ] == 0.
  Proof. intros y. exact (additive_zero (fun u : V ⇒ [ u , y ])). Qed.

  Theorem bracket_zero_r : ∀ x : V, [ x , (0 : V) ] == 0.
  Proof. intros x. exact (additive_zero (bracket x)). Qed.

  Theorem bracket_neg_l : ∀ x y : V, [ - x , y ] == - [ x , y ].
  Proof. intros x y. exact (additive_negation (fun u : V ⇒ [ u , y ]) x). Qed.

  Theorem bracket_neg_r : ∀ x y : V, [ x , - y ] == - [ x , y ].
  Proof. intros x y. exact (additive_negation (bracket x) y). Qed.

End BilinearBracket.

Section AlternatingBracket.
  Context {F V : Type} `{Field F}.
  Context {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V}
          {Vs : HasScale F V} {VS : VectorSpace F V} {Vb : HasBracket V}.
  Context {BL : Bilinear F V V V bracket} {AL : Alternates V bracket}.

  Definition bracket_alternates : ∀ x : V, [ x , x ] == 0 :=
    alternates (op := bracket).

  Lemma bracket_sum_expand : ∀ x y : V,
    [ x + y , x + y ] == ([ x , x ] + [ y , x ]) + ([ x , y ] + [ y , y ]).
  Proof.
    intros x y.
    calc ([ x + y , x + y ])
      == ([ x + y , x ] + [ x + y , y ])
         { (bracket_right_additive (x + y) x y) }
      == (([ x , x ] + [ y , x ]) + [ x + y , y ])
         { (bracket_left_additive x y x) }
      == (([ x , x ] + [ y , x ]) + ([ x , y ] + [ y , y ]))
         { (bracket_left_additive x y y) }.
  Qed.

  Theorem bracket_antisymmetric : ∀ x y : V, [ y , x ] == - [ x , y ].
  Proof.
    intros x y.
    apply neg_unique_l.
    calc ([ y , x ] + [ x , y ])
      == (([ x , x ] + [ y , x ]) + ([ x , y ] + [ y , y ]))
      == ([ x + y , x + y ])
      == (0 : V).
    - rewrite (bracket_alternates x), (bracket_alternates y).
      autorewrite with grp.
      reflexivity.
    - symmetry. exact (bracket_sum_expand x y).
    - exact (bracket_alternates (x + y)).
  Qed.

  Theorem bracket_swap_sum : ∀ x y : V, [ x , y ] + [ y , x ] == 0.
  Proof.
    intros x y.
    rewrite (bracket_antisymmetric x y).
    exact (add_right_inverse [ x , y ]).
  Qed.

  Theorem bracket_neg_inner : ∀ x y z : V,
    [ x , [ z , y ] ] == - [ x , [ y , z ] ].
  Proof.
    intros x y z.
    calc ([ x , [ z , y ] ])
      == ([ x , - [ y , z ] ]) { (bracket_antisymmetric y z) }
      == (- [ x , [ y , z ] ]) { (bracket_neg_r x [ y , z ]) }.
  Qed.

  Theorem jacobiator_swap : ∀ x y z : V,
    [ y , [ x , z ] ] + [ x , [ z , y ] ] + [ z , [ y , x ] ]
      == - ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ]).
  Proof.
    intros x y z.
    calc ([ y , [ x , z ] ] + [ x , [ z , y ] ] + [ z , [ y , x ] ])
      == (- [ y , [ z , x ] ] + [ x , [ z , y ] ] + [ z , [ y , x ] ])
         { (bracket_neg_inner y z x) }
      == (- [ y , [ z , x ] ] + - [ x , [ y , z ] ] + [ z , [ y , x ] ])
         { (bracket_neg_inner x y z) }
      == (- [ y , [ z , x ] ] + - [ x , [ y , z ] ] + - [ z , [ x , y ] ])
         { (bracket_neg_inner z x y) }
      == (- [ x , [ y , z ] ] + - [ y , [ z , x ] ] + - [ z , [ x , y ] ])
         { (add_commutative (- [ y , [ z , x ] ]) (- [ x , [ y , z ] ])) }
      == (- ([ x , [ y , z ] ] + [ y , [ z , x ] ]) + - [ z , [ x , y ] ])
         { (neg_add [ x , [ y , z ] ] [ y , [ z , x ] ]) }
      == (- ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ]))
         { (neg_add ([ x , [ y , z ] ] + [ y , [ z , x ] ]) [ z , [ x , y ] ]) }.
  Qed.

  Theorem jacobiator_repeat : ∀ x z : V,
    [ x , [ x , z ] ] + [ x , [ z , x ] ] + [ z , [ x , x ] ] == 0.
  Proof.
    intros x z.
    calc ([ x , [ x , z ] ] + [ x , [ z , x ] ] + [ z , [ x , x ] ])
      == ([ x , [ x , z ] ] + - [ x , [ x , z ] ] + [ z , [ x , x ] ])
         { (bracket_neg_inner x x z) }
      == ([ x , [ x , z ] ] + - [ x , [ x , z ] ] + [ z , (0 : V) ])
         { (bracket_alternates x) }
      == ([ x , [ x , z ] ] + - [ x , [ x , z ] ] + (0 : V))
         { (bracket_zero_r z) }
      == ((0 : V) + (0 : V))
         { (add_right_inverse [ x , [ x , z ] ]) }
      == (0 : V)
         { (add_right_identity (0 : V)) }.
  Qed.

  Theorem jacobi_from_derivation :
    (∀ x y z : V, [ x , [ y , z ] ] == [ [ x , y ] , z ] + [ y , [ x , z ] ]) →
    ∀ x y z : V,
      [ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ] == 0.
  Proof.
    intros derivation x y z.
    assert (Hinner : [ y , [ x , z ] ] + [ y , [ z , x ] ] == 0).
    { calc ([ y , [ x , z ] ] + [ y , [ z , x ] ])
        == ([ y , [ x , z ] + [ z , x ] ])
           { (bracket_right_additive y [ x , z ] [ z , x ]) }
        == ([ y , (0 : V) ])
           { (bracket_swap_sum x z) }
        == (0 : V)
           { (bracket_zero_r y) }. }
    calc ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ])
      == (([ [ x , y ] , z ] + [ y , [ x , z ] ]) + [ y , [ z , x ] ]
            + [ z , [ x , y ] ])
         { (derivation x y z) }
      == (([ [ x , y ] , z ] + [ y , [ x , z ] ])
            + ([ y , [ z , x ] ] + [ z , [ x , y ] ]))
         { associative }
      == (([ [ x , y ] , z ] + [ z , [ x , y ] ])
            + ([ y , [ x , z ] ] + [ y , [ z , x ] ]))
         { add_rotate }
      == ((0 : V) + ([ y , [ x , z ] ] + [ y , [ z , x ] ]))
         { (bracket_swap_sum [ x , y ] z) }
      == ((0 : V) + (0 : V))
         { Hinner }
      == (0 : V)
         { (add_right_identity (0 : V)) }.
  Qed.

End AlternatingBracket.

Section LieAlgebraTheory.
  Context {F V : Type} `{LieAlgebra F V}.

Bracketing with x on the left is a derivation of the bracket.
  Theorem bracket_derivation : ∀ x y z : V,
    [ x , [ y , z ] ] == [ [ x , y ] , z ] + [ y , [ x , z ] ].
  Proof.
    intros x y z.
    assert (Hpair : [ x , [ y , z ] ] + [ y , [ z , x ] ] == - [ z , [ x , y ] ])
      by exact (neg_unique_l _ _ (jacobi x y z)).
    calc ([ x , [ y , z ] ])
      == ([ x , [ y , z ] ] + (0 : V))
         { (add_right_identity [ x , [ y , z ] ]) }
      == ([ x , [ y , z ] ] + ([ y , [ z , x ] ] + - [ y , [ z , x ] ]))
         { (add_right_inverse [ y , [ z , x ] ]) }
      == (([ x , [ y , z ] ] + [ y , [ z , x ] ]) + - [ y , [ z , x ] ])
         { associative }
      == (- [ z , [ x , y ] ] + - [ y , [ z , x ] ])
         { Hpair }
      == ([ [ x , y ] , z ] + - [ y , [ z , x ] ])
         { (bracket_antisymmetric z [ x , y ]) }
      == ([ [ x , y ] , z ] + [ y , [ x , z ] ])
         { (bracket_neg_inner y z x) }.
  Qed.

End LieAlgebraTheory.

Section LieAlgebraFromBasis.
  Context {F V : Type} `{Field F}.
  Context {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V}
          {Vs : HasScale F V} {VS : VectorSpace F V} {Vb : HasBracket V}.
  Context {BL : Bilinear F V V V bracket} {AL : Alternates V bracket}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context (B : Basis n I F V).

  Lemma jacobi_map_proper : ∀ y z : V,
    Proper (equ ==> equ)
      (fun x : V ⇒ [ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ]).
  Proof.
    intros y z x x' Hx.
    rewrite Hx.
    reflexivity.
  Qed.

  Lemma jacobi_map_additive : ∀ (y z x x' : V),
    [ x + x' , [ y , z ] ] + [ y , [ z , x + x' ] ] + [ z , [ x + x' , y ] ]
      == ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ])
         + ([ x' , [ y , z ] ] + [ y , [ z , x' ] ] + [ z , [ x' , y ] ]).
  Proof.
    intros y z x x'.
    calc ([ x + x' , [ y , z ] ] + [ y , [ z , x + x' ] ] + [ z , [ x + x' , y ] ])
      == (([ x , [ y , z ] ] + [ x' , [ y , z ] ])
            + [ y , [ z , x + x' ] ] + [ z , [ x + x' , y ] ])
         { (bracket_left_additive x x' [ y , z ]) }
      == (([ x , [ y , z ] ] + [ x' , [ y , z ] ])
            + [ y , [ z , x ] + [ z , x' ] ] + [ z , [ x + x' , y ] ])
         { (bracket_right_additive z x x') }
      == (([ x , [ y , z ] ] + [ x' , [ y , z ] ])
            + ([ y , [ z , x ] ] + [ y , [ z , x' ] ]) + [ z , [ x + x' , y ] ])
         { (bracket_right_additive y [ z , x ] [ z , x' ]) }
      == (([ x , [ y , z ] ] + [ x' , [ y , z ] ])
            + ([ y , [ z , x ] ] + [ y , [ z , x' ] ]) + [ z , [ x , y ] + [ x' , y ] ])
         { (bracket_left_additive x x' y) }
      == (([ x , [ y , z ] ] + [ x' , [ y , z ] ])
            + ([ y , [ z , x ] ] + [ y , [ z , x' ] ])
            + ([ z , [ x , y ] ] + [ z , [ x' , y ] ]))
         { (bracket_right_additive z [ x , y ] [ x' , y ]) }
      == (([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ])
            + ([ x' , [ y , z ] ] + [ y , [ z , x' ] ] + [ z , [ x' , y ] ]))
         { add_interchange_triple }.
  Qed.

  Lemma jacobi_map_scale : ∀ (a : F) (y z x : V),
    [ a *: x , [ y , z ] ] + [ y , [ z , a *: x ] ] + [ z , [ a *: x , y ] ]
      == a *: ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ]).
  Proof.
    intros a y z x.
    calc ([ a *: x , [ y , z ] ] + [ y , [ z , a *: x ] ] + [ z , [ a *: x , y ] ])
      == (a *: [ x , [ y , z ] ] + [ y , [ z , a *: x ] ] + [ z , [ a *: x , y ] ])
         { (bracket_left_scale a x [ y , z ]) }
      == (a *: [ x , [ y , z ] ] + [ y , a *: [ z , x ] ] + [ z , [ a *: x , y ] ])
         { (bracket_right_scale a z x) }
      == (a *: [ x , [ y , z ] ] + a *: [ y , [ z , x ] ] + [ z , [ a *: x , y ] ])
         { (bracket_right_scale a y [ z , x ]) }
      == (a *: [ x , [ y , z ] ] + a *: [ y , [ z , x ] ] + [ z , a *: [ x , y ] ])
         { (bracket_left_scale a x y) }
      == (a *: [ x , [ y , z ] ] + a *: [ y , [ z , x ] ] + a *: [ z , [ x , y ] ])
         { (bracket_right_scale a z [ x , y ]) }
      == (a *: ([ x , [ y , z ] ] + [ y , [ z , x ] ]) + a *: [ z , [ x , y ] ])
         { (left_distributive a [ x , [ y , z ] ] [ y , [ z , x ] ]) }
      == (a *: ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ]))
         { (left_distributive a ([ x , [ y , z ] ] + [ y , [ z , x ] ])
              [ z , [ x , y ] ]) }.
  Qed.

  Definition jacobi_map (y z : V) : LinearMap F V V :=
    linear_map_from_laws
      (fun x : V ⇒ [ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ])
      (jacobi_map_proper y z)
      (jacobi_map_additive y z)
      (fun a x ⇒ jacobi_map_scale a y z x).

  Lemma jacobi_map_cycle : ∀ x y z : V,
    jacobi_map y z x == jacobi_map z x y.
  Proof.
    intros x y z.
    exact (add_cycle [ x , [ y , z ] ] [ y , [ z , x ] ] [ z , [ x , y ] ]).
  Qed.

  Context (jacobi_on_basis : ∀ i j k : I,
    jacobi_map (Basis.vec B j) (Basis.vec B k) (Basis.vec B i) == 0).

  Lemma jacobi_two_basis_arguments : ∀ (i j : I) (z : V),
    jacobi_map (Basis.vec B i) (Basis.vec B j) z == 0.
  Proof.
    intros i j.
    exact (linear_map_zero_on_basis B (jacobi_map (Basis.vec B i) (Basis.vec B j))
             (fun k ⇒ jacobi_on_basis k i j)).
  Qed.

  Lemma jacobi_one_basis_argument : ∀ (i : I) (z y : V),
    jacobi_map z (Basis.vec B i) y == 0.
  Proof.
    intros i z.
    apply (linear_map_zero_on_basis B (jacobi_map z (Basis.vec B i))).
    intros j.
    calc (jacobi_map z (Basis.vec B i) (Basis.vec B j))
      == (jacobi_map (Basis.vec B i) (Basis.vec B j) z)
         { jacobi_map_cycle }
      == (0 : V)
         { (jacobi_two_basis_arguments i j z) }.
  Qed.

  Theorem jacobi_from_basis : ∀ x y z : V,
    [ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ] == 0.
  Proof.
    intros x y z.
    apply (linear_map_zero_on_basis B (jacobi_map y z)).
    intros i.
    calc (jacobi_map y z (Basis.vec B i))
      == (jacobi_map z (Basis.vec B i) y)
         { jacobi_map_cycle }
      == (0 : V)
         { (jacobi_one_basis_argument i z y) }.
  Qed.

  Definition lie_algebra_from_basis : LieAlgebra F V :=
    {| LieAlgebra.is_bilinear := BL;
       LieAlgebra.is_alternating := AL;
       LieAlgebra.jacobi := jacobi_from_basis |}.

End LieAlgebraFromBasis.
