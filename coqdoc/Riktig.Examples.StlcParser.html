The typed reader at the concrete syntax trees: parse a term and look at what it built.

Definition read : string → option (stlc nat (sty name)) :=
  parse_first
    (stlc_p (P := parser string) (T := stlc nat (sty name)) (Ty := sty name) 32 (nil : list name)
       >>= fun t ⇒ ret (Some t))
    None.

Eval vm_compute in read "\x : a. x".
Eval vm_compute in read "\f : a -> a. \x : a. f (f x)".

Example read_id :
  read "\x : a. x"
  = Some (tylam (ty_base (list_ascii_of_string "a")) (var 0%nat) : stlc nat (sty name)).
Proof. vm_compute. reflexivity. Qed.

Example read_twice :
  read "\f : a -> a. \x : a. f (f x)"
  = Some (tylam (ty_base (list_ascii_of_string "a") --> ty_base (list_ascii_of_string "a"))
            (tylam (ty_base (list_ascii_of_string "a"))
               (app (var 1%nat) (app (var 1%nat) (var 0%nat)))) : stlc nat (sty name)).
Proof. vm_compute. reflexivity. Qed.
