Primitive arrays are tables over a finite index numbered by nat, when the count fits: an entry is read at the index's number, and a function is tabulated by filling a fresh array in number order. Two array tables are equal when they agree at every index.
Section TableArray.
  Context {I A : Type} {eqA : EqRel A} {EA : Equivalence eqA} {dA : HasDefault A}
          {card : HasCard I nat} {index : HasIndex I nat} {elem : HasElem I nat}
          {HF : Finite I nat card index elem (eqI := eq)}
          (fits : (Z.of_nat card ≤ max_nat)%Z).

  #[export] Instance ArrayTable_EqRel : EqRel (ArrayTable I A) :=
    fun a a' ⇒ ∀ i : I, a.[int_of_nat (index i)] == a'.[int_of_nat (index i)].
  #[export] Instance ArrayTable_HasGet : HasGet (ArrayTable I A) I A :=
    fun a i ⇒ a.[int_of_nat (index i)].
  #[export] Instance ArrayTable_HasTabulate : HasTabulate (ArrayTable I A) I A :=
    fun f ⇒ tabulate_array card default (fun k ⇒ f (elem k)).

  #[export] Instance ArrayTable_Table :
    Table I A (ArrayTable I A) ArrayTable_HasGet ArrayTable_HasTabulate (eqI := eq).
  Proof.
    split.
    - split.
      + intros a i. reflexivity.
      + intros a a' Haa' i. symmetry. exact (Haa' i).
      + intros a a' a'' H1 H2 i. transitivity (a'.[int_of_nat (index i)]); [ exact (H1 i) | exact (H2 i) ].
    - intros a a' Haa' i j Hij. rewrite Hij. exact (Haa' j).
    - intros f g Hfg i. unfold get, ArrayTable_HasGet, tabulate, ArrayTable_HasTabulate.
      rewrite !get_tabulate_array by (exact fits || exact (lt_of_le_neq _ _ (index_lt i))).
      exact (Hfg (elem (index i))).
    - intros f i. unfold get, ArrayTable_HasGet, tabulate, ArrayTable_HasTabulate.
      rewrite get_tabulate_array by (exact fits || exact (lt_of_le_neq _ _ (index_lt i))).
      rewrite (elem_index i). reflexivity.
    - intros a i. unfold get, ArrayTable_HasGet, tabulate, ArrayTable_HasTabulate.
      rewrite get_tabulate_array by (exact fits || exact (lt_of_le_neq _ _ (index_lt i))).
      rewrite (elem_index i). reflexivity.
  Qed.
End TableArray.
