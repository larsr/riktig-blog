Grade projections are linear and kill zero; they pass through a sum over grades; and nothing has a grade above the top grade.
Section GradedTheory.
  Context {N F A : Type} `{Graded N F A}.
  Local Instance grade_linear_here : ∀ k : N, Linear F A A (grade_part (A := A) k) := fun k ⇒ Graded.grade_linear k.

  Theorem grade_proper : ∀ k : N, Proper (equ ==> equ) (grade_part (A := A) k).
  Proof. intros k. exact (Linear.proper (f := grade_part (A := A) k)). Qed.

  Theorem grade_add : ∀ (k : N) (x y : A), ⟨x + y⟩_k == ⟨x⟩_k + ⟨y⟩_k.
  Proof. intros k x y. exact (additive (f := grade_part (A := A) k) x y). Qed.

  Theorem grade_smul : ∀ (k : N) (a : F) (x : A), ⟨a *: x⟩_k == a *: ⟨x⟩_k.
  Proof. intros k a x. exact (homogeneous (f := grade_part (A := A) k) a x). Qed.

  Theorem grade_zero_elem : ∀ k : N, ⟨(0 : A)⟩_k == 0.
  Proof. intros k. exact (additive_zero (grade_part (A := A) k)). Qed.

  Theorem grade_neg : ∀ (k : N) (x : A), ⟨- x⟩_k == - ⟨x⟩_k.
  Proof. intros k x. exact (additive_neg (grade_part (A := A) k) x). Qed.

Sums over grades

  Lemma sum_step_proper : ∀ f : N → A, Proper (equ ==> equ) f →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc + f k).
  Proof. intros f Hf k k' Hk acc acc' Hacc. apply Magma.op_proper; [ exact Hacc | apply Hf; exact Hk ]. Qed.

  Lemma grade_fun_proper : ∀ (x : A), Proper (equ ==> equ) (fun k : N ⇒ ⟨x⟩_k).
  Proof. intros x k k' Hk. exact (Graded.grade_index_proper k k' x Hk). Qed.

  Theorem grade_natrec : ∀ (f : N → A) (l : N), Proper (equ ==> equ) f → ∀ m : N,
    ⟨natrec A 0 (fun k acc ⇒ acc + f k) m⟩_l == natrec A 0 (fun k acc ⇒ acc + ⟨f k⟩_l) m.
  Proof.
    intros f l Hf.
    assert (Hfl : Proper (equ ==> equ) (fun k ⇒ ⟨f k⟩_l)) by (intros k k' Hk; apply grade_proper; apply Hf; exact Hk).
    apply (Peano.nat_induction (fun m ⇒ ⟨natrec A 0 (fun k acc ⇒ acc + f k) m⟩_l == natrec A 0 (fun k acc ⇒ acc + ⟨f k⟩_l) m)).
    - intros m m' Hm.
      setoid_rewrite (NatRec.natrec_proper A _ _ 0 _ (sum_step_proper f Hf) m m' Hm).
      setoid_rewrite (NatRec.natrec_proper A _ _ 0 _ (sum_step_proper _ Hfl) m m' Hm). reflexivity.
    - calc (⟨natrec A 0 (fun k acc ⇒ acc + f k) 0⟩_l)
        == (⟨(0 : A)⟩_l) { (grade_proper l _ _ (NatRec.natrec_zero A _ _ 0 _)) }
        == (0 : A) { (grade_zero_elem l) }
        == (natrec A 0 (fun k acc ⇒ acc + ⟨f k⟩_l) 0) { (symmetry (NatRec.natrec_zero A _ _ 0 _)) }.
    - intros m IH.
      calc (⟨natrec A 0 (fun k acc ⇒ acc + f k) (succ m)⟩_l)
        == (⟨natrec A 0 (fun k acc ⇒ acc + f k) m + f m⟩_l) { (grade_proper l _ _ (NatRec.natrec_succ A _ _ 0 _ m)) }
        == (⟨natrec A 0 (fun k acc ⇒ acc + f k) m⟩_l + ⟨f m⟩_l) { (grade_add l _ _) }
        == (natrec A 0 (fun k acc ⇒ acc + ⟨f k⟩_l) m + ⟨f m⟩_l) { IH }
        == (natrec A 0 (fun k acc ⇒ acc + ⟨f k⟩_l) (succ m)) { (symmetry (NatRec.natrec_succ A _ _ 0 _ m)) }.
  Qed.

The grade-m + j part of the sum of the parts below m is zero.
  Lemma grade_parts_vanish : ∀ (x : A) (m j : N),
    natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m + j)) m == 0.
  Proof.
    intros x.
    assert (Hstep : ∀ l : N, Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc + ⟨⟨x⟩_k⟩_l)).
    { intros l. apply sum_step_proper. intros k k' Hk. apply grade_proper. exact (Graded.grade_index_proper k k' x Hk). }
    
    assert (Hbound : ∀ l m m' : N, m == m' →
      natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_l) m == natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_l) m')
      by (intros l; exact (NatRec.natrec_proper A _ _ 0 _ (Hstep l))).
    assert (Hgrade : ∀ m l l' : N, l == l' →
      natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_l) m == natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_l') m).
    { intros m l l' Hl. apply (natrec_ext A _ _ 0 _ _ (Hstep l) (Hstep l')). intros k acc.
      apply Magma.op_proper; [ reflexivity | exact (Graded.grade_index_proper _ _ _ Hl) ]. }
    apply (Peano.nat_induction (fun m ⇒ ∀ j : N, natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m + j)) m == 0)).
    - intros m m' Hm.
      assert (He : ∀ j : N,
        natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m + j)) m == natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m' + j)) m').
      { intros j.
        calc (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m + j)) m)
          == (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m' + j)) m) { (Hgrade m (m + j) (m' + j) (Magma.op_proper _ _ Hm j j (reflexivity j))) }
          == (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m' + j)) m') { (Hbound (m' + j) m m' Hm) }. }
      split; intros Hall j; [ rewrite <- (He j) | rewrite (He j) ]; exact (Hall j).
    - intros j. exact (NatRec.natrec_zero A _ _ 0 _).
    - intros m IH j.
      assert (Hidx : succ m + j == m + succ j).
      { calc (succ m + j) == (succ (m + j)) { (add_succ_l m j) } == (m + succ j) { (symmetry (add_succ m j)) }. }
      assert (Hne : m =/= succ m + j).
      { intros Hm. apply (ne_succ_add m j). transitivity (succ m + j); [ exact Hm | exact (add_succ_l m j) ]. }
      calc (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(succ m + j)) (succ m))
        == (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(succ m + j)) m + ⟨⟨x⟩_m⟩_(succ m + j)) { (NatRec.natrec_succ A _ _ 0 _ m) }
        == (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(m + succ j)) m + ⟨⟨x⟩_m⟩_(succ m + j)) { (Hgrade m _ _ Hidx) }
        == ((0 : A) + ⟨⟨x⟩_m⟩_(succ m + j)) { (IH (succ j)) }
        == ((0 : A) + 0) { (Graded.grade_orthogonal _ _ x (fun Hk ⇒ Hne (symmetry Hk))) }
        == (0 : A) { (left_identity (action := add) (0 : A)) }.
  Qed.

  Lemma grade_grade_fun_proper : ∀ (x : A) (l : N), Proper (equ ==> equ) (fun k : N ⇒ ⟨⟨x⟩_k⟩_l).
  Proof. intros x l k k' Hk. exact (grade_proper _ _ _ (Graded.grade_index_proper k k' x Hk)). Qed.

  Theorem grade_above : ∀ x : A, ⟨x⟩_(succ top_grade) == 0.
  Proof.
    intros x.
    calc (⟨x⟩_(succ top_grade))
      == (⟨natrec A 0 (fun k acc ⇒ acc + ⟨x⟩_k) (succ top_grade)⟩_(succ top_grade)) { (grade_proper _ _ _ (symmetry (Graded.grade_complete x))) }
      == (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(succ top_grade)) (succ top_grade)) { (grade_natrec _ _ (grade_fun_proper x) _) }
      == (natrec A 0 (fun k acc ⇒ acc + ⟨⟨x⟩_k⟩_(succ top_grade + 0)) (succ top_grade))
           { (natrec_ext A _ _ 0 _ _ (sum_step_proper _ (grade_grade_fun_proper x _)) (sum_step_proper _ (grade_grade_fun_proper x _))
                (fun k acc ⇒ Magma.op_proper _ _ (reflexivity acc) _ _ (Graded.grade_index_proper _ _ _ (symmetry (add_zero _)))) _) }
      == (0 : A) { (grade_parts_vanish x (succ top_grade) 0) }.
  Qed.

The grades of the second half of a doubling

low k b is the grade-k-1 part of b, and zero at k = 0: in a doubled algebra it is the grade-k part contributed by the second component.
  Local Notation "'low' k b" := (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨b⟩_j) k) (at level 10, k at level 9, b at level 9).

  Lemma low_step_proper : ∀ b : A, Proper (equ ==> equ ==> equ) (fun (j : N) (_ : A) ⇒ ⟨b⟩_j).
  Proof. intros b j j' Hj _ _ _. exact (Graded.grade_index_proper j j' b Hj). Qed.

  Lemma low_zero : ∀ b : A, low 0 b == 0.
  Proof. intros b. exact (NatRec.natrec_zero A _ _ 0 _). Qed.

  Lemma low_succ : ∀ (m : N) (b : A), low (succ m) b == ⟨b⟩_m.
  Proof. intros m b. exact (NatRec.natrec_succ A _ _ 0 (fun (j : N) (_ : A) ⇒ ⟨b⟩_j) m). Qed.

  Lemma low_index_proper : ∀ (k k' : N) (b : A), k == k' → low k b == low k' b.
  Proof. intros k k' b Hk. exact (NatRec.natrec_proper A _ _ 0 _ (low_step_proper b) k k' Hk). Qed.

  Lemma low_proper : ∀ (k : N) (b b' : A), b == b' → low k b == low k b'.
  Proof.
    intros k b b' Hb. apply (natrec_ext A _ _ 0 _ _ (low_step_proper b) (low_step_proper b')).
    intros j _. exact (grade_proper j b b' Hb).
  Qed.

A statement about low k follows from its two cases.
  Lemma low_cases : ∀ (P : A → Prop) (k : N) (b : A),
    Proper (equ ==> iff) P → P 0 → (∀ m : N, P ⟨b⟩_m) → P (low k b).
  Proof.
    intros P k b HP Hz Hs. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - apply (proj2 (HP _ _ (transitivity (low_index_proper k 0 b Hk) (low_zero b)))). exact Hz.
    - apply (proj2 (HP _ _ (transitivity (low_index_proper k (succ m) b Hk) (low_succ m b)))). exact (Hs m).
  Qed.

  Lemma low_add : ∀ (k : N) (b b' : A), low k (b + b') == low k b + low k b'.
  Proof.
    intros k b b'. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - calc (low k (b + b'))
        == (low 0 (b + b')) { (low_index_proper k 0 _ Hk) }
        == (0 : A) { (low_zero _) }
        == ((0 : A) + 0) { (symmetry (left_identity (action := add) (0 : A))) }
        == (low 0 b + low 0 b') { (Magma.op_proper _ _ (symmetry (low_zero b)) _ _ (symmetry (low_zero b'))) }
        == (low k b + low k b') { (Magma.op_proper _ _ (low_index_proper 0 k b (symmetry Hk)) _ _ (low_index_proper 0 k b' (symmetry Hk))) }.
    - calc (low k (b + b'))
        == (low (succ m) (b + b')) { (low_index_proper k (succ m) _ Hk) }
        == (⟨b + b'⟩_m) { (low_succ m _) }
        == (⟨b⟩_m + ⟨b'⟩_m) { (grade_add m b b') }
        == (low (succ m) b + low (succ m) b') { (Magma.op_proper _ _ (symmetry (low_succ m b)) _ _ (symmetry (low_succ m b'))) }
        == (low k b + low k b') { (Magma.op_proper _ _ (low_index_proper _ _ b (symmetry Hk)) _ _ (low_index_proper _ _ b' (symmetry Hk))) }.
  Qed.

  Lemma low_smul : ∀ (k : N) (a : F) (b : A), low k (a *: b) == a *: low k b.
  Proof.
    intros k a b. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - calc (low k (a *: b))
        == (low 0 (a *: b)) { (low_index_proper k 0 _ Hk) }
        == (0 : A) { (low_zero _) }
        == (a *: (0 : A)) { (symmetry (smul_zero a)) }
        == (a *: low 0 b) { (ScaleLaws.scale_proper _ _ (reflexivity _) _ _ (symmetry (low_zero b))) }
        == (a *: low k b) { (ScaleLaws.scale_proper _ _ (reflexivity _) _ _ (low_index_proper 0 k b (symmetry Hk))) }.
    - calc (low k (a *: b))
        == (low (succ m) (a *: b)) { (low_index_proper k (succ m) _ Hk) }
        == (⟨a *: b⟩_m) { (low_succ m _) }
        == (a *: ⟨b⟩_m) { (grade_smul m a b) }
        == (a *: low (succ m) b) { (ScaleLaws.scale_proper _ _ (reflexivity _) _ _ (symmetry (low_succ m b))) }
        == (a *: low k b) { (ScaleLaws.scale_proper _ _ (reflexivity _) _ _ (low_index_proper _ _ b (symmetry Hk))) }.
  Qed.

  Lemma low_zero_elem : ∀ k : N, low k (0 : A) == 0.
  Proof.
    intros k. apply (low_cases (fun y ⇒ y == 0) k 0).
    - intros y y' Hy. split; intros Hz; [ transitivity y; [ symmetry; exact Hy | exact Hz ] | transitivity y'; [ exact Hy | exact Hz ] ].
    - reflexivity.
    - intros m. exact (grade_zero_elem m).
  Qed.

  Lemma low_low : ∀ (k : N) (b : A), low k (low k b) == low k b.
  Proof.
    intros k b. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - calc (low k (low k b))
        == (low 0 (low k b)) { (low_index_proper k 0 _ Hk) }
        == (0 : A) { (low_zero _) }
        == (low 0 b) { (symmetry (low_zero b)) }
        == (low k b) { (low_index_proper 0 k b (symmetry Hk)) }.
    - calc (low k (low k b))
        == (low (succ m) (low (succ m) b)) { (transitivity (low_index_proper k (succ m) _ Hk) (low_proper _ _ _ (low_index_proper k (succ m) b Hk))) }
        == (⟨low (succ m) b⟩_m) { (low_succ m _) }
        == (⟨⟨b⟩_m⟩_m) { (grade_proper m _ _ (low_succ m b)) }
        == (⟨b⟩_m) { (Graded.grade_idempotent m b) }
        == (low (succ m) b) { (symmetry (low_succ m b)) }
        == (low k b) { (low_index_proper _ _ b (symmetry Hk)) }.
  Qed.

  Lemma low_low_ne : ∀ (k l : N) (b : A), k =/= l → low k (low l b) == 0.
  Proof.
    intros k l b Hkl. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - calc (low k (low l b)) == (low 0 (low l b)) { (low_index_proper k 0 _ Hk) } == (0 : A) { (low_zero _) }.
    - destruct (zero_or_succ l) as [Hl | [m' Hl]].
      + calc (low k (low l b))
          == (low (succ m) (low 0 b)) { (transitivity (low_index_proper k (succ m) _ Hk) (low_proper _ _ _ (low_index_proper l 0 b Hl))) }
          == (⟨low 0 b⟩_m) { (low_succ m _) }
          == (⟨(0 : A)⟩_m) { (grade_proper m _ _ (low_zero b)) }
          == (0 : A) { (grade_zero_elem m) }.
      + assert (Hmm : m =/= m').
        { intros Hm. apply Hkl. transitivity (succ m); [ exact Hk | transitivity (succ m'); [ apply Peano.succ_proper; exact Hm | symmetry; exact Hl ] ]. }
        calc (low k (low l b))
          == (low (succ m) (low (succ m') b)) { (transitivity (low_index_proper k (succ m) _ Hk) (low_proper _ _ _ (low_index_proper l (succ m') b Hl))) }
          == (⟨low (succ m') b⟩_m) { (low_succ m _) }
          == (⟨⟨b⟩_m'⟩_m) { (grade_proper m _ _ (low_succ m' b)) }
          == (0 : A) { (Graded.grade_orthogonal m m' b Hmm) }.
  Qed.

  Lemma low_neg : ∀ (k : N) (y : A), low k (- y) == - low k y.
  Proof.
    intros k y. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - calc (low k (- y))
        == (low 0 (- y)) { (low_index_proper k 0 (- y) Hk) }
        == (0 : A) { (low_zero (- y)) }
        == (- (0 : A)) { (symmetry neg_zero) }
        == (- low 0 y) { (Group.neg_proper _ _ (symmetry (low_zero y))) }
        == (- low k y) { (Group.neg_proper _ _ (low_index_proper 0 k y (symmetry Hk))) }.
    - calc (low k (- y))
        == (low (succ m) (- y)) { (low_index_proper k (succ m) (- y) Hk) }
        == (⟨- y⟩_m) { (low_succ m (- y)) }
        == (- ⟨y⟩_m) { (grade_neg m y) }
        == (- low (succ m) y) { (Group.neg_proper _ _ (symmetry (low_succ m y))) }
        == (- low k y) { (Group.neg_proper _ _ (low_index_proper (succ m) k y (symmetry Hk))) }.
  Qed.

The parts low k b for k up to succ m add up to the parts of b up to m.
  Lemma low_fun_proper : ∀ b : A, Proper (equ ==> equ) (fun k : N ⇒ low k b).
  Proof. intros b k k' Hk. exact (low_index_proper k k' b Hk). Qed.

  Lemma low_shift_fun_proper : ∀ (b : A) (i : N), Proper (equ ==> equ) (fun k : N ⇒ low (i + k) b).
  Proof. intros b i k k' Hk. exact (low_index_proper (i + k) (i + k') b (Magma.op_proper i i (reflexivity i) k k' Hk)). Qed.

The one-step sum of the parts is zero, and the part after a shift by one is the grade.
  Lemma low_sum_one : ∀ b : A, natrec A 0 (fun k acc ⇒ acc + low k b) (succ 0) == 0.
  Proof.
    intros b.
    calc (natrec A 0 (fun k acc ⇒ acc + low k b) (succ 0))
      == (natrec A 0 (fun k acc ⇒ acc + low k b) 0 + low 0 b) { (NatRec.natrec_succ A _ _ 0 _ 0) }
      == ((0 : A) + low 0 b) { (NatRec.natrec_zero A _ _ 0 _) }
      == ((0 : A) + 0) { (low_zero b) }
      == (0 : A) { (left_identity (action := add) (0 : A)) }.
  Qed.

  Lemma low_shift_one : ∀ (b : A) (k : N), low (succ 0 + k) b == ⟨b⟩_k.
  Proof.
    intros b k.
    calc (low (succ 0 + k) b)
      == (low (succ (0 + k)) b) { (low_index_proper _ _ b (add_succ_l 0 k)) }
      == (⟨b⟩_(0 + k)) { (low_succ (0 + k) b) }
      == (⟨b⟩_k) { (Graded.grade_index_proper _ _ b (left_identity (action := add) k)) }.
  Qed.

  Lemma low_sum : ∀ (b : A) (m : N),
    natrec A 0 (fun k acc ⇒ acc + low k b) (succ m) == natrec A 0 (fun k acc ⇒ acc + ⟨b⟩_k) m.
  Proof.
    intros b m.
    calc (natrec A 0 (fun k acc ⇒ acc + low k b) (succ m))
      == (natrec A 0 (fun k acc ⇒ acc + low k b) (succ 0 + m))
           { (NatRec.natrec_proper A _ _ 0 _ (sum_step_proper _ (low_fun_proper b)) _ _
                (symmetry (transitivity (add_succ_l 0 m) (Peano.succ_proper _ _ (left_identity (action := add) m))))) }
      == (natrec A (natrec A 0 (fun k acc ⇒ acc + low k b) (succ 0)) (fun k acc ⇒ acc + low (succ 0 + k) b) m)
           { (natrec_shift A _ _ 0 _ (sum_step_proper _ (low_fun_proper b)) (succ 0) m) }
      == (natrec A 0 (fun k acc ⇒ acc + low (succ 0 + k) b) m)
           { (natrec_init A _ _ _ 0 _ (sum_step_proper _ (low_shift_fun_proper b (succ 0))) (low_sum_one b) m) }
      == (natrec A 0 (fun k acc ⇒ acc + ⟨b⟩_k) m)
           { (natrec_ext A _ _ 0 _ _ (sum_step_proper _ (low_shift_fun_proper b (succ 0))) (sum_step_proper _ (grade_fun_proper b))
                (fun k acc ⇒ Magma.op_proper _ _ (reflexivity acc) _ _ (low_shift_one b k)) m) }.
  Qed.
Purity and splits transport across equality of the projected element.
  Lemma pure_of_equ : ∀ (k : N) (x y : A),
    x == y → y == ⟨y⟩_k → x == ⟨x⟩_k.
  Proof.
    intros k x y Hxy Hy.
    calc x
      == y { Hxy }
      == (⟨y⟩_k) { Hy }
      == (⟨x⟩_k) { (grade_proper k _ _ (symmetry Hxy)) }.
  Qed.

  Lemma split_of_equ : ∀ (k l : N) (x y : A),
    x == y → y == ⟨y⟩_k + ⟨y⟩_l → x == ⟨x⟩_k + ⟨x⟩_l.
  Proof.
    intros k l x y Hxy Hy.
    calc x
      == y { Hxy }
      == (⟨y⟩_k + ⟨y⟩_l) { Hy }
      == (⟨x⟩_k + ⟨x⟩_l)
