Any binary arithmetic with recursion is a naturals: denotation into Coq's naturals by the value provides every law through denotation_naturals; every cited fact is Theory.BinaryNumber's.
Section NaturalsBinary.
  Context {N : Type} {eqN : EqRel N} {zeroN : Zero N} {bit0N : HasBit0 N} {bit1N : HasBit1 N}
    {BN : BinaryNumber N zeroN bit0N bit1N} {binrecN : HasBinRec N} {BR : BinRec N binrecN}
    {succN : HasSucc N} {oneN : One N} {addN : HasAdd N} {mulN : HasMult N}
    {BA : BinaryArith N zeroN bit0N bit1N succN oneN addN mulN}.

  #[export] Instance Binary_Naturals : Naturals N zeroN succN oneN addN mulN :=
    denotation_naturals (C := N) (N0 := nat) (fun (c : N) (n : nat) ⇒ bin_val (N := N) (N0 := nat) c = n)
      (fun c ⇒ ex_intro _ (bin_val (N := N) (N0 := nat) c) eq_refl)
      (bin_val_onto (N := N) (N0 := nat))
      (fun c n m Hn Hm ⇒ eq_trans (eq_sym Hn) Hm)
      (fun c d n m Hc Hd Hnm ⇒ bin_val_injective (N := N) (N0 := nat) c d (eq_trans Hc (eq_trans Hnm (eq_sym Hd))))
      (fun c d n Hcd Hc ⇒ eq_trans (eq_sym (bin_val_proper (N := N) (N0 := nat) c d Hcd)) Hc)
      (fun c n m Hnm Hc ⇒ eq_trans Hc Hnm)
      (bin_val_zero (N := N) (N0 := nat))
      (eq_trans (bin_val_proper (N := N) (N0 := nat) _ _ BinaryArith.one_bit1) (bin_val_one (N := N) (N0 := nat)))
      (fun c n Hc ⇒ eq_trans (bin_val_succ (N := N) (N0 := nat) c) (f_equal S Hc))
      (fun c d a b Hc Hd ⇒ eq_trans (bin_val_add (N := N) (N0 := nat) c d) (f_equal2 Nat.add Hc Hd))
      (fun c d a b Hc Hd ⇒ eq_trans (bin_val_mult (N := N) (N0 := nat) c d) (f_equal2 Nat.mul Hc Hd)).
End NaturalsBinary.
