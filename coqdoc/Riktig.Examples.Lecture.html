Blackboard line 1: a multivector is the sum of its grade parts.
  Theorem lecture_grade_decomposition : ∀ a : Cl F 3,
    a == ⟨ a ⟩_ 0 + ⟨ a ⟩_ 1 + ⟨ a ⟩_ 2 + ⟨ a ⟩_ 3.
  Proof.
    exact grade_decomposition_three.
  Qed.

Blackboard line 2, in the tree's normalized form: over an ordered field the geometric product of two vectors is the inner product on the unit plus half the commutator wedge, u v == ⟪u,v⟫ 1 + ½ (uv - vu).
  Theorem lecture_product_split : ∀ u v : Fin.t 3 → F,
    ga_vec u × ga_vec v
    == ⟪ u , v ⟫ *: 1 + half 1 *: wedge (ga_vec u) (ga_vec v).
  Proof.
    intros u v.
    exact (ga_product_decompose u v).
  Qed.

  Lemma vec_grade_one : ∀ u : Fin.t 3 → F,
    ⟨ ga_vec u ⟩_ 1 == ga_vec u.
  Proof.
    intros u. exact (cl_grade_one_vec 3 u).
  Qed.

  Lemma vec_mul_grade_one_zero : ∀ u v : Fin.t 3 → F,
    ⟨ ga_vec u × ga_vec v ⟩_ 1 == 0.
  Proof.
    intros u v. exact (cl_grade_odd_vec_mul 3 sig3 u v 1%nat eq_refl).
  Qed.

  Lemma vec_mul_grade_three_zero : ∀ u v : Fin.t 3 → F,
    ⟨ ga_vec u × ga_vec v ⟩_ 3 == 0.
  Proof.
    intros u v. exact (cl_grade_odd_vec_mul 3 sig3 u v 3%nat eq_refl).
  Qed.

Blackboard line 2, through the grading: for vectors the dot is the grade-0 part and the wedge the grade-2 part of the geometric product, so u v == u ⋅ v + u ∧ v.
  Theorem lecture_graded_split : ∀ u v : Fin.t 3 → F,
    ga_vec u × ga_vec v == ga_vec u ⋅ ga_vec v + ga_vec u ∧ ga_vec v.
  Proof.
    intros u v.
    assert (H13 : (1 ≤ 3)%nat) by (apply le_S; apply le_S; apply le_n).
    assert (Hdot : ga_vec u ⋅ ga_vec v == ⟨ ga_vec u × ga_vec v ⟩_ 0).
    { exact (dot_homogeneous 1%nat 1%nat (ga_vec u) (ga_vec v)
               H13 H13 (vec_grade_one u) (vec_grade_one v)). }
    assert (Hwedge : ga_vec u ∧ ga_vec v == ⟨ ga_vec u × ga_vec v ⟩_ 2).
    { exact (wedge_homogeneous 1%nat 1%nat (ga_vec u) (ga_vec v)
               H13 H13 (vec_grade_one u) (vec_grade_one v)). }
    calc (ga_vec u × ga_vec v)
      == (⟨ ga_vec u × ga_vec v ⟩_ 0 + ⟨ ga_vec u × ga_vec v ⟩_ 1
          + ⟨ ga_vec u × ga_vec v ⟩_ 2 + ⟨ ga_vec u × ga_vec v ⟩_ 3)
        { (lecture_grade_decomposition (ga_vec u × ga_vec v)) }
      == (⟨ ga_vec u × ga_vec v ⟩_ 0 + 0
          + ⟨ ga_vec u × ga_vec v ⟩_ 2 + ⟨ ga_vec u × ga_vec v ⟩_ 3)
        { (vec_mul_grade_one_zero u v) }
      == (⟨ ga_vec u × ga_vec v ⟩_ 0 + 0
          + ⟨ ga_vec u × ga_vec v ⟩_ 2 + 0)
        { (vec_mul_grade_three_zero u v) }
      == (⟨ ga_vec u × ga_vec v ⟩_ 0 + 0 + ⟨ ga_vec u × ga_vec v ⟩_ 2)
        { right_identity }
      == (⟨ ga_vec u × ga_vec v ⟩_ 0 + ⟨ ga_vec u × ga_vec v ⟩_ 2)
        { (right_identity (⟨ ga_vec u × ga_vec v ⟩_ 0)) }
      == (ga_vec u ⋅ ga_vec v + ⟨ ga_vec u × ga_vec v ⟩_ 2)
        { Hdot }
      == (ga_vec u ⋅ ga_vec v + ga_vec u ∧ ga_vec v)
        { Hwedge }.
  Qed.


  Import ClThreeNotation.

  Ltac lecture_reduce :=
    repeat change (@mult (Cl F 3) L_Mult) with L_Mult;
    repeat change (@one (Cl F 3) L_One) with L_One;
    
    repeat change (@neg (Cl F 3) PW_Neg)
      with (fun (g : Cl F 3) (D : Blade 3) ⇒ - g D);
    unfold L_Mult, L_One, cl_mul, cl_one, cl_blade, bsign, parity_sign,
      sig3, blade3, sb, bxor, bdiff, band;
    simpl.
  Ltac lecture_blade := lecture_reduce; ring.

  Example lecture_e1_embedded :
    (e1 : Cl F 3) == ga_vec (space_vector 1 0 0).
  Proof.
    apply cl3_ext; lecture_blade.
  Qed.
