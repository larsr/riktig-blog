From Coq Require Export Morphisms.
From Riktig.Structure Require Export Ring.

A reversal of a ring: an additive map that reverses products and is its own inverse -- an involutive anti-automorphism.
Module Reversal.
  Class Reversal (A : Type) `{Ring A} (reversal : HasReverse A) := {
    reverse_proper :: Proper (equ ==> equ) (@reverse A reversal);
    reverse_additive :: Additive A A add add reverse;
    reverse_anti : ∀ x y : A, reverse (x × y) == reverse y × reverse x;
    reverse_involutive : ∀ x : A, reverse (reverse x) == x;
  }.
  Hint Mode Reversal ! - - - - - - - - : typeclass_instances.
End Reversal.
Export Reversal.
