Record Joint (F A : Type) := mkJoint {
  joint_rotor : A;
  joint_length : F
}.

Arguments mkJoint {F A} joint_rotor joint_length.
Arguments joint_rotor {F A} j.
Arguments joint_length {F A} j.

Definition Chain (F A : Type) : Type := list (Joint F A).


Section ChainOrientation.
  Context {F A : Type}.
  Context {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
          {Ao : One A} {Am : HasMult A}.
  Context {AR : Ring A}.

  Definition chain_rotor (joints : Chain F A) : A :=
    fold_right (fun j acc ⇒ joint_rotor j × acc) 1 joints.

  Lemma chain_rotor_nil : chain_rotor [] = (1 : A).
  Proof. reflexivity. Qed.

  Lemma chain_rotor_cons : ∀ (j : Joint F A) (rest : Chain F A),
    chain_rotor (j :: rest) = joint_rotor j × chain_rotor rest.
  Proof. reflexivity. Qed.

  Fixpoint chain_orientation_from (orientation : A) (joints : Chain F A) : A :=
    match joints with
    | [] ⇒ orientation
    | j :: rest ⇒ chain_orientation_from (orientation × joint_rotor j) rest
    end.

  Lemma chain_orientation_from_nil : ∀ orientation : A,
    chain_orientation_from orientation [] = orientation.
  Proof. reflexivity. Qed.

  Lemma chain_orientation_from_cons : ∀ (orientation : A)
      (j : Joint F A) (rest : Chain F A),
    chain_orientation_from orientation (j :: rest)
    = chain_orientation_from (orientation × joint_rotor j) rest.
  Proof. reflexivity. Qed.

Walking the chain from an orientation gives that orientation times the chain's own product: the unit laws and associativity, nothing else.
  Theorem chain_rotor_is_product : ∀ (joints : Chain F A) (orientation : A),
    chain_orientation_from orientation joints == orientation × chain_rotor joints.
  Proof.
    induction joints as [| j rest IH]; intros orientation.
    - calc (chain_orientation_from orientation [])
        == (orientation : A)
           { chain_orientation_from_nil }
        == (orientation × 1 : A)
           { right_identity }
        == (orientation × chain_rotor [] : A)
           { chain_rotor_nil }.
    - calc (chain_orientation_from orientation (j :: rest))
        == (chain_orientation_from (orientation × joint_rotor j) rest : A)
           { chain_orientation_from_cons }
        == ((orientation × joint_rotor j) × chain_rotor rest : A)
           { IH }
        == (orientation × (joint_rotor j × chain_rotor rest) : A)
           { associative }
        == (orientation × chain_rotor (j :: rest) : A)
           { chain_rotor_cons }.
  Qed.

  Corollary chain_rotor_is_product_from_identity : ∀ joints : Chain F A,
    chain_orientation_from 1 joints == chain_rotor joints.
  Proof.
    intros joints. rewrite (chain_rotor_is_product joints 1).
    exact (left_identity (chain_rotor joints)).
  Qed.

End ChainOrientation.


Section ChainClosure.
  Context {F A : Type}.
  Context {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
          {Ao : One A} {Am : HasMult A}.
  Context {AR : Ring A}.

Even joints accumulate to an even orientation: the Z/2 grading law of Structure/Graded.v, folded along the chain.
  Context (even odd : A → Prop) {PG : ParityGraded A even odd}.

  Theorem chain_rotor_even : ∀ joints : Chain F A,
    Forall (fun j ⇒ even (joint_rotor j)) joints → even (chain_rotor joints).
  Proof.
    intros joints Hjoints.
    induction Hjoints as [| j rest Hj Hrest IH].
    - exact parity_even_one.
    - exact (even_mul_even (joint_rotor j) (chain_rotor rest) Hj IH).
  Qed.

  Context (reversal : A → A) {RV : Reversal A reversal}.

Rotor joints accumulate to a rotor orientation, so no chain, however long, drifts out of the group: `unit_is_rotor` starts it and `mul_is_rotor` extends it, both from Theory/Rotor.v.
  Theorem chain_is_rotor : ∀ joints : Chain F A,
    Forall (fun j ⇒ Rotor reversal even (joint_rotor j)) joints →
    Rotor reversal even (chain_rotor joints).
  Proof.
    intros joints Hjoints.
    induction Hjoints as [| j rest Hj Hrest IH].
    - exact (unit_is_rotor reversal even odd).
    - exact (mul_is_rotor reversal even odd
               (joint_rotor j) (chain_rotor rest)).
  Qed.

End ChainClosure.


Section PlaneOfRotors.
  Context {F A : Type} `{Field F}.
  Context {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
          {Ao : One A} {Am : HasMult A} {As : HasScale F A}.
  Context {AR : Ring A} {SL : ScaleLaws F A}.
  Context {ML : ∀ y : A, Linear F A A (fun x ⇒ x × y)}.
  Context {MR : ∀ x : A, Linear F A A (fun y ⇒ x × y)}.

  Context (e : A) (root : e × e == - (1)).

The steering of a joint that turns in the plane of e.
  Definition turn (c s : F) : A := c *: 1 + s *: e.

  #[export] Instance turn_proper : Proper (equ ==> equ ==> equ) turn.
  Proof.
    intros c c' Hc s s' Hs. unfold turn.
    rewrite Hc, Hs. reflexivity.
  Qed.

Two turns compose to a turn — Theory/Algebra.v's two-term product, which is the complex multiplication table over any square root of - (1).
  Lemma turn_mul : ∀ c1 s1 c2 s2 : F,
    turn c1 s1 × turn c2 s2 == turn (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2).
  Proof.
    intros c1 s1 c2 s2.
    exact (two_term_product c1 s1 c2 s2 e root).
  Qed.

  Lemma turn_one : turn 1 0 == 1.
  Proof. exact (scale_one_scale_zero e). Qed.

A chain steered in one plane accumulates in that plane.
  Theorem chain_rotor_in_plane : ∀ joints : Chain F A,
    Forall (fun j ⇒ ∃ c s : F, joint_rotor j == turn c s) joints →
    ∃ c s : F, chain_rotor joints == turn c s.
  Proof.
    intros joints Hjoints.
    induction Hjoints as [| j rest [c1 [s1 Hj]] Hrest [c [s IH]]].
    - ∃ 1, 0. symmetry. exact turn_one.
    - ∃ (c1×c + - (s1×s)), (c1×s + s1×c).
      calc ((chain_rotor (j :: rest)) : A)
        == ((joint_rotor j × chain_rotor rest) : A)
           { chain_rotor_cons }
        == ((turn c1 s1 × chain_rotor rest) : A)
           { Hj }
        == ((turn c1 s1 × turn c s) : A)
           { IH }
        == ((turn (c1×c + - (s1×s)) (c1×s + s1×c)) : A)
           { turn_mul }.
  Qed.

End PlaneOfRotors.


Section TurnAngle.
  Context {F A : Type} `{CA : Circular F A}.
  Context (e : A) (root : e × e == - (1)).

A joint steering by t in the plane of e has rotor cexp (t *: e): that is Euler's identity, read right to left.
  Theorem turn_is_cexp : ∀ t : F,
    turn e (cos t) (sin t) == cexp (t *: e).
  Proof.
    intros t. symmetry. exact (euler t e root).
  Qed.

End TurnAngle.


Section ArmPositions.
  Context {F A : Type} `{Field F}.
  Context {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
          {Ao : One A} {Am : HasMult A} {As : HasScale F A}.
  Context {AR : Ring A} {SL : ScaleLaws F A}.

  Context (reversal : A → A) (rest_direction : A).

  Definition arm_sandwich (R x : A) : A := R × x × reversal R.

Where a beam points once its own cumulative rotor has acted.
  Definition beam_direction (R : A) : A := arm_sandwich R rest_direction.

  Fixpoint chain_positions_from (orientation position : A)
           (joints : Chain F A) : list A :=
    match joints with
    | [] ⇒ []
    | j :: rest ⇒
        let orientation' := orientation × joint_rotor j in
        let position' := position + joint_length j *: beam_direction orientation' in
        position' :: chain_positions_from orientation' position' rest
    end.

  Lemma chain_positions_from_nil : ∀ orientation position : A,
    chain_positions_from orientation position [] = [].
  Proof. reflexivity. Qed.

  Lemma chain_positions_from_cons : ∀ (orientation position : A)
      (j : Joint F A) (rest : Chain F A),
    chain_positions_from orientation position (j :: rest)
    = let orientation' := orientation × joint_rotor j in
      let position' := position + joint_length j *: beam_direction orientation' in
      position' :: chain_positions_from orientation' position' rest.
  Proof. reflexivity. Qed.

  Definition chain_positions (base : A) (joints : Chain F A) : list A :=
    chain_positions_from 1 base joints.

  Context (reversal_proper : Proper (equ ==> equ) reversal)
          (reversal_one : reversal 1 == 1)
          (reversal_anti : ∀ x y : A,
             reversal (x × y) == reversal y × reversal x).

Sandwiching by S then by R is sandwiching once by R × S — Theory/GeometricAlgebra.v's `sandwich_compose_anti`, whose only hypothesis is that reversal reverses products. This is what makes accumulating the rotor and turning the beam at the end agree with turning it joint by joint.
  Theorem arm_sandwich_compose : ∀ R S x : A,
    arm_sandwich R (arm_sandwich S x) == arm_sandwich (R × S) x.
  Proof.
    intros R S x. unfold arm_sandwich.
    exact (sandwich_compose_anti reversal reversal_anti R S x).
  Qed.

  Lemma beam_direction_congruence : ∀ R S : A,
    R == S → beam_direction R == beam_direction S.
  Proof.
    intros R S HRS. unfold beam_direction, arm_sandwich.
    rewrite HRS. reflexivity.
  Qed.

  Lemma beam_direction_at_one : beam_direction 1 == rest_direction.
  Proof.
    unfold beam_direction, arm_sandwich.
    calc (((1 : A) × rest_direction × reversal 1) : A)
      == ((rest_direction × reversal 1) : A) { left_identity }
      == ((rest_direction × 1) : A) { reversal_one }
      == (rest_direction : A) { right_identity }.
  Qed.

  Lemma chain_positions_two : ∀ (base : A) (j1 j2 : Joint F A),
    chain_positions base [j1; j2]
    = [ base + joint_length j1 *: beam_direction (1 × joint_rotor j1);
        base + joint_length j1 *: beam_direction (1 × joint_rotor j1)
             + joint_length j2 *:
                 beam_direction (1 × joint_rotor j1 × joint_rotor j2) ].
  Proof. reflexivity. Qed.

An arm of two equal beams whose first joint does not steer and whose second reverses the beam direction folds back onto its own base. The half turn enters as one equation about the algebra — u × rest × reversal u == - rest — and the tip and the base leave as parameters, so a coordinate reading of either is the instantiation's business.
  Theorem arm_folds_back : ∀ (base tip back : A) (j1 j2 : Joint F A),
    joint_rotor j1 == 1 →
    arm_sandwich (joint_rotor j2) rest_direction == - rest_direction →
    joint_length j1 == joint_length j2 →
    base + joint_length j1 *: rest_direction == tip →
    base == back →
    Forall2 equ (chain_positions base [j1; j2]) [tip; back].
  Proof.
    intros base tip back j1 j2 Hstraight Hhalf Hlengths Htip Hback.
    assert (Hone : (1 : A) × joint_rotor j1 == 1).
    { calc (((1 : A) × joint_rotor j1) : A)
        == ((joint_rotor j1) : A) { left_identity }
        == ((1 : A)) { Hstraight }. }
    assert (Hfirst : beam_direction (1 × joint_rotor j1) == rest_direction).
    { calc ((beam_direction (1 × joint_rotor j1)) : A)
        == ((beam_direction 1) : A)
           { (beam_direction_congruence (1 × joint_rotor j1) 1 Hone) }
        == (rest_direction : A)
           { beam_direction_at_one }. }
    assert (Hsecond : beam_direction (1 × joint_rotor j1 × joint_rotor j2)
                      == - rest_direction).
    { assert (Hstep : (1 : A) × joint_rotor j1 × joint_rotor j2
                      == joint_rotor j2).
      { calc (((1 : A) × joint_rotor j1 × joint_rotor j2) : A)
          == (((1 : A) × joint_rotor j2) : A) { Hone }
          == ((joint_rotor j2) : A) { left_identity }. }
      calc ((beam_direction (1 × joint_rotor j1 × joint_rotor j2)) : A)
        == ((beam_direction (joint_rotor j2)) : A)
           { (beam_direction_congruence _ _ Hstep) }
        == ((- rest_direction) : A)
           { Hhalf }. }
    rewrite (chain_positions_two base j1 j2).
    apply Forall2_cons.
    - calc ((base + joint_length j1 *: beam_direction (1 × joint_rotor j1)) : A)
        == ((base + joint_length j1 *: rest_direction) : A) { Hfirst }
        == (tip : A) { Htip }.
    - apply Forall2_cons; [ | apply Forall2_nil ].
      calc ((base + joint_length j1 *: beam_direction (1 × joint_rotor j1)
               + joint_length j2 *:
                   beam_direction (1 × joint_rotor j1 × joint_rotor j2)) : A)
        == ((base + joint_length j1 *: rest_direction
               + joint_length j2 *:
                   beam_direction (1 × joint_rotor j1 × joint_rotor j2)) : A)
           { Hfirst }
        == ((base + joint_length j1 *: rest_direction
               + joint_length j2 *: (- rest_direction)) : A)
           { Hsecond }
        == ((base + joint_length j1 *: rest_direction
               + - (joint_length j2 *: rest_direction)) : A)
           { scale_neg }
        == ((base + joint_length j1 *: rest_direction
               + - (joint_length j1 *: rest_direction)) : A)
           { Hlengths }
        == (base : A)
           { add_sub_cancel_r }
        == (back : A)
           { Hback }.
  Qed.

End ArmPositions.


Section BeamLength.
  Context {F V A : Type} `{GeometricAlgebra F V A}.
  Context (reversal : A → A) {RV : Reversal A reversal}.
  Context (even odd : A → Prop) {PG : ParityGraded A even odd}.
  Context (rest : V).

However far a chain of rotor joints has turned, the beam it points has the length it started with: `chain_is_rotor` says the accumulated orientation is still a rotor, and Theory/Rotor.v's `rotor_preserves_qform` says a rotor's sandwich is an isometry. Which vector the sandwich lands on — w — is left to the caller, since reading coordinates off is what a dimension is for.
  Theorem beam_keeps_its_length : ∀ (joints : Chain F A) (w : V),
    one_faithful F A →
    Forall (fun j ⇒ Rotor reversal even (joint_rotor j)) joints →
    ga_vec w == beam_direction reversal (ga_vec rest) (chain_rotor joints) →
    Q w == Q rest.
  Proof.
    intros joints w Hfaithful Hjoints Hw.
    pose proof (chain_is_rotor even odd reversal joints Hjoints) as Hrotor.
    exact (rotor_preserves_qform reversal even (chain_rotor joints)
             rest w Hfaithful Hw).
  Qed.

End BeamLength.
