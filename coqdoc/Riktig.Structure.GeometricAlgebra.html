A geometric algebra over an inner product space is an algebra with a linear embedding of the vectors, in which every vector squares to its inner product with itself.
Module GeometricAlgebra.
  Class GeometricAlgebra (F V A : Type) `{InnerProductSpace F V}
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
      {Ao : One A} {Am : HasMult A} {As : HasScale F A}
      {AA : Algebra F A Az Aa An Ao Am As} (vec : HasVec V A) := {
    vec_linear :: Linear F V A vec;
    vec_sq : ∀ u : V, vec u × vec u == ⟪ u , u ⟫ *: 1;
  }.
  Hint Mode GeometricAlgebra - - ! - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End GeometricAlgebra.
Export GeometricAlgebra.
