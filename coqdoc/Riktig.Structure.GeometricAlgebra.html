From Riktig.Structure Require Export QuadraticForm.

A geometric algebra over a quadratic space (V, Q) is an associative unital F-algebra with a linear map from V in which every vector squares to its quadratic value: ga_vec u × ga_vec u == Q u *: 1.
Module GeometricAlgebra.
  Class GeometricAlgebra (F V A : Type) `{QuadraticForm F V}
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
      {Ao : One A} {Am : HasMult A} {As : HasScale F A} := {
    ring :: Ring A;
    scaling :: ScaleLaws F A;

    mul_linear_l :: ∀ y : A, Linear F A A (fun x ⇒ x × y);
    mul_linear_r :: ∀ x : A, Linear F A A (fun y ⇒ x × y);

    ga_vec : V → A;
    vec_linear :: Linear F V A ga_vec;

    contraction : ∀ u : V, ga_vec u × ga_vec u == Q u *: 1;
  }.
  Hint Mode GeometricAlgebra - - ! - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End GeometricAlgebra.
Export GeometricAlgebra.
