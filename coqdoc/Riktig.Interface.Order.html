Library Riktig.Interface.Order

From Coq Require Export Morphisms.
From Riktig Require Export Ops Interface.Ring Interface.Field.

Class HasLe T := le : Rel T.
Infix "≤" := le (at level 70, no associativity).

Class HasAbs T := abs : T → T.

#[global] Hint Mode HasLe ! : typeclass_instances.
#[global] Hint Mode HasAbs ! : typeclass_instances.

Definition lt {T : Type} `{EqRel T} `{HasLe T} (a b : T) : Prop :=
  a ≤ b ∧ a =/= b.
Infix "<" := lt (at level 70, no associativity).

Module OrderedField.
  Class OrderedField (F : Type) `{Field.Field F}
                     {Fl : HasLe F} {Fb : HasAbs F} := {
    le_refl : ∀ a : F, a ≤ a;
    le_trans : ∀ a b c : F, a ≤ b → b ≤ c → a ≤ c;
    le_antisymmetric : ∀ a b : F, a ≤ b → b ≤ a → a == b;
    le_total : ∀ a b : F, a ≤ b ∨ b ≤ a;
    
    le_stable : ∀ a b : F, ¬ ¬ a ≤ b → a ≤ b;
    le_proper : ∀ a b : F, a == b →
                           ∀ c d : F, c == d → (a ≤ c ↔ b ≤ d);
    add_le_monotone_left : ∀ a b c : F, a ≤ b → c + a ≤ c + b;
    mul_nonneg_closed : ∀ a b : F, 0 ≤ a → 0 ≤ b → 0 ≤ a × b;
    abs_proper : ∀ a b : F, a == b → abs a == abs b;
    abs_nonneg : ∀ a : F, 0 ≤ abs a;
    abs_zero_iff : ∀ a : F, abs a == 0 ↔ a == 0;
    abs_mul : ∀ a b : F, abs (a × b) == abs a × abs b;
    
    abs_neg : ∀ a : F, abs (- a) == abs a;
    abs_triangle : ∀ a b : F, abs (a + b) ≤ abs a + abs b;
    abs_nonneg_identity : ∀ a : F, 0 ≤ a → abs a == a
  }.
End OrderedField.
Notation OrderedField := OrderedField.OrderedField.

#[global] Instance OrderedField_le_Proper (F : Type) `{Field.Field F}
  {Fl : HasLe F} {Fb : HasAbs F} (H : OrderedField.OrderedField F)
  : Proper (equ ==> equ ==> iff) (@le F _) := OrderedField.le_proper.

#[global] Instance OrderedField_abs_Proper (F : Type) `{Field.Field F}
  {Fl : HasLe F} {Fb : HasAbs F} (H : OrderedField.OrderedField F)
  : Proper (equ ==> equ) (@abs F _) := OrderedField.abs_proper.

#[global] Hint Mode OrderedField.OrderedField ! - - - - - - - - - - : typeclass_instances.
