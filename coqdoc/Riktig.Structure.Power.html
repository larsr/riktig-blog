Library Riktig.Structure.Power

From Riktig Require Export Structure.Monoid.

Module Power.
  Class Power (M : Type) {eM : EqRel M} {oM : One M} {mM : HasMult M}
      {MM : @Monoid M eM oM mM} (pow : M → nat → M) := {
    pow_zero : ∀ x : M, pow x 0%nat == 1;
    pow_succ : ∀ (x : M) (n : nat), pow x (S n) == x × pow x n;
  }.
  Hint Mode Power ! - - - - ! : typeclass_instances.
End Power.
Export Power.
