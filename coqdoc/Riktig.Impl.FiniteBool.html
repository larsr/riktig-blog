Library Riktig.Impl.FiniteBool

From Riktig Require Export Structure.Finite Impl.NatOrderNat Lib.Coq.Bool.

Coq's bool is a finite index of count two, numbered by nat.
Instance Bool_EqRel : EqRel bool := eq.

Instance Bool_HasCard : HasCard bool nat := 2.
Instance Bool_HasIndex : HasIndex bool nat := bool_index.
Instance Bool_HasElem : HasElem bool nat := bool_elem.

Instance Bool_Finite : Finite bool nat Bool_HasCard Bool_HasIndex Bool_HasElem (eqI := eq) :=
  {| Finite.equivalence := eq_equivalence;
     Finite.index_proper := f_equal bool_index;
     Finite.elem_proper := f_equal bool_elem;
     Finite.index_lt := bool_index_lt;
     Finite.elem_index := bool_elem_index;
     Finite.index_elem := bool_index_elem |}.
