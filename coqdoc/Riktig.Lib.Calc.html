Lemma equiv_iff {A : Type} {R : A → A → Prop} {E : Equivalence R} {a a' b b' : A} :
  R a a' → R b b' → (R a b ↔ R a' b').
Proof.
  intros Ha Hb. split; intros Hab.
  - exact (transitivity (symmetry Ha) (transitivity Hab Hb)).
  - exact (transitivity Ha (transitivity Hab (symmetry Hb))).
Qed.

Section Witnesses.
  Context {A B C D : Type}.

  Lemma ex2_and_l (P P' Q : A → B → Prop) :
    (∀ a b, P a b ↔ P' a b) → ((∃ a b, P a b ∧ Q a b) ↔ (∃ a b, P' a b ∧ Q a b)).
  Proof.
    intros HP. split; intros [a [b [Hp Hq]]]; ∃ a, b; split; try exact Hq; apply HP; exact Hp.
  Qed.

  Lemma ex2_and_r (P Q Q' : A → B → Prop) :
    (∀ a b, Q a b ↔ Q' a b) → ((∃ a b, P a b ∧ Q a b) ↔ (∃ a b, P a b ∧ Q' a b)).
  Proof.
    intros HQ. split; intros [a [b [Hp Hq]]]; ∃ a, b; split; try exact Hp; apply HQ; exact Hq.
  Qed.

  Lemma ex2_assoc (P : A → B → Prop) (Q : A → B → C → D → Prop) (R : C → D → Prop) :
    (∃ c d, (∃ a b, P a b ∧ Q a b c d) ∧ R c d) ↔ (∃ a b, P a b ∧ (∃ c d, Q a b c d ∧ R c d)).
  Proof.
    split.
    - intros [c [d [[a [b [Hp Hq]]] Hr]]]. ∃ a, b. split; [ exact Hp | ∃ c, d; split; [ exact Hq | exact Hr ] ].
    - intros [a [b [Hp [c [d [Hq Hr]]]]]]. ∃ c, d. split; [ ∃ a, b; split; [ exact Hp | exact Hq ] | exact Hr ].
  Qed.

  Lemma ex2_false_l (Q : A → B → Prop) : (∃ a b, False ∧ Q a b) ↔ False.
  Proof. split; [ intros [a [b [[] _]]] | intros [] ]. Qed.

  Lemma ex2_false_r (P : A → B → Prop) : (∃ a b, P a b ∧ False) ↔ False.
  Proof. split; [ intros [a [b [_ []]]] | intros [] ]. Qed.

  Lemma ex2_or_l (P P' Q : A → B → Prop) :
    (∃ a b, (P a b ∨ P' a b) ∧ Q a b) ↔ (∃ a b, P a b ∧ Q a b) ∨ (∃ a b, P' a b ∧ Q a b).
  Proof.
    split.
    - intros [a [b [[Hp | Hp] Hq]]]; [ left | right ]; ∃ a, b; split; assumption.
    - intros [[a [b [Hp Hq]]] | [a [b [Hp Hq]]]]; ∃ a, b; split; try assumption; [ left | right ]; exact Hp.
  Qed.

  Lemma ex2_or_r (P Q Q' : A → B → Prop) :
    (∃ a b, P a b ∧ (Q a b ∨ Q' a b)) ↔ (∃ a b, P a b ∧ Q a b) ∨ (∃ a b, P a b ∧ Q' a b).
  Proof.
    split.
    - intros [a [b [Hp [Hq | Hq]]]]; [ left | right ]; ∃ a, b; split; assumption.
    - intros [[a [b [Hp Hq]]] | [a [b [Hp Hq]]]]; ∃ a, b; split; try assumption; [ left | right ]; exact Hq.
  Qed.

  Context {RA : A → A → Prop} {RB : B → B → Prop} {EA : Equivalence RA} {EB : Equivalence RB}.

  Lemma ex2_equ_l (a : A) (b : B) (Q : A → B → Prop) : Proper (RA ==> RB ==> iff) Q →
    ((∃ a' b', (RA a' a ∧ RB b' b) ∧ Q a' b') ↔ Q a b).
  Proof.
    intros HQ. split.
    - intros [a' [b' [[Ha Hb] Hq]]]. exact (proj1 (HQ a' a Ha b' b Hb) Hq).
    - intros Hq. ∃ a, b. split; [ split; reflexivity | exact Hq ].
  Qed.

  Lemma ex2_equ_r (a : A) (b : B) (Q : A → B → Prop) : Proper (RA ==> RB ==> iff) Q →
    ((∃ a' b', Q a' b' ∧ (RA a a' ∧ RB b b')) ↔ Q a b).
  Proof.
    intros HQ. split.
    - intros [a' [b' [Hq [Ha Hb]]]]. exact (proj2 (HQ a a' Ha b b' Hb) Hq).
    - intros Hq. ∃ a, b. split; [ exact Hq | split; reflexivity ].
  Qed.
End Witnesses.

Lemma or_same (A : Prop) : A ∨ A ↔ A.
Proof. split; [ intros [H | H]; exact H | intros H; left; exact H ]. Qed.

Lemma or_false_l (A : Prop) : False ∨ A ↔ A.

Ltac calc_ends x y :=
  lazymatch goal with
  | |- _ ?l ?r ⇒ unify l x; unify r y
  end.

Ltac calc_by r :=
  first [ now rewrite r
        | now rewrite <- r
        | now setoid_rewrite r
        | now setoid_rewrite <- r
        | now apply r
        | now (symmetry; apply r)
        | now exact r
        | now (symmetry; exact r) ].

Ltac calc_step r :=
  tryif assert_succeeds (let c := open_constr:(r) in is_evar c)
  then idtac else calc_by r.
