From Coq Require Export Field_theory.
From Coq Require Import Setoid Ring.
From Riktig Require Export Structure.MinusOneNotASquare Theory.Ring.

Section FieldTheory.
  Context {F : Type} `{Field F}.
  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  Theorem mul_commutative : ∀ a b : F, a × b == b × a.
  Proof. intros a b. ring. Qed.

  Theorem field_right_inverse : ∀ a : F, a =/= 0 → a × inv a == 1.
  Proof.
    intros a Ha.
    calc (a × inv a)
      == (inv a × a) { commutative }
      == (1 : F) { ((left_inverse_on (defined_on := nonzero _)) a Ha) }.
  Qed.

  Theorem mul_cancel_l : ∀ a b c : F,
    a =/= 0 → a × b == a × c → b == c.
  Proof.
    intros a b c Ha Heq.
    calc b
      == (1 × b) { left_identity }
      == ((inv a × a) × b) { ((left_inverse_on (defined_on := nonzero _)) a Ha) }
      == (inv a × (a × b)) { associative }
      == (inv a × (a × c)) { Heq }
      == ((inv a × a) × c) { associative }
      == (1 × c) { ((left_inverse_on (defined_on := nonzero _)) a Ha) }
      == c { left_identity }.
  Qed.

  Theorem field_mul_neq_0 : ∀ a b : F,
    a =/= 0 → b =/= 0 → a × b =/= 0.
  Proof.
    intros a b Ha Hb Heq.
    apply Hb.
    calc b
      == (1 × b) { left_identity }
      == ((inv a × a) × b) { ((left_inverse_on (defined_on := nonzero _)) a Ha) }
      == (inv a × (a × b)) { associative }
      == (inv a × 0) { Heq }
      == (0 : F) { mul_0_r }.
  Qed.

  Theorem field_mul_div_cancel : ∀ a b : F,
    b =/= 0 → b × (a × inv b) == a.
  Proof.
    intros a b Hb.
    calc (b × (a × inv b))
      == (b × (inv b × a)) { (commutative a) }
      == ((b × inv b) × a) { associative }
      == (1 × a) { field_right_inverse }
      == a { left_identity }.
  Qed.

  Theorem inv_mul_cancel : ∀ a b : F, a =/= 0 → inv a × (a × b) == b.
  Proof.
    intros a b Ha.
    calc (inv a × (a × b))
      == ((inv a × a) × b) { associative }
      == (1 × b) { ((left_inverse_on (defined_on := nonzero _)) a Ha) }
      == b { left_identity }.
  Qed.

  Theorem zero_neq_one : (0 : F) =/= 1.
  Proof.
    intros Heq. exact (nontrivial (symmetry Heq)).
  Qed.

  Theorem mul_left_commutative : ∀ a b c : F, a × (b × c) == b × (a × c).
  Proof. intros a b c. ring. Qed.

  Theorem mul_right_commutative : ∀ a b c : F, a × b × c == a × c × b.
  Proof. intros a b c. ring. Qed.

  Definition field_theory_of_Field :
    field_theory (0 : F) (1 : F) add mult (fun x y ⇒ x + - y) neg
                 (fun x y ⇒ x × inv y) inv equ.
  Proof.
    constructor.
    - exact (ring_theory_of_Ring (fun a b ⇒ commutative a b)).
    - exact nontrivial.
    - reflexivity.
    - exact (left_inverse_on (defined_on := nonzero _)).
  Qed.

End FieldTheory.

Section MinusOneNotASquareTheory.
  Context {F : Type} `{MinusOneNotASquare.MinusOneNotASquare F}.

  Lemma two_squares_vanish_r : ∀ a b : F, a × a + b × b == 0 → b == 0.
  Proof.
    intros a b Hab.
    apply (two_squares_vanish b a).
    calc (b × b + a × a)
      == (a × a + b × b) { commutative }
      == (0 : F) { Hab }.
  Qed.

  Lemma two_squares_nonzero : ∀ a b : F, a =/= 0 → a × a + b × b =/= 0.
  Proof.
    intros a b Ha Hn.
    apply Ha.
    exact (two_squares_vanish a b Hn).
  Qed.

End MinusOneNotASquareTheory.
