Section FieldTheory.
  Context {F : Type} `{Field F}.

  Theorem mul_left_comm : ∀ a b c : F, a × (b × c) == b × (a × c).
  Proof.
    intros a b c.
    calc (a × (b × c))
      == ((a × b) × c) { associative }
      == ((b × a) × c) { (commutative (op := mult) a b) }
      == (b × (a × c)) { associative }.
  Qed.

  Theorem mul_right_comm : ∀ a b c : F, (a × b) × c == (a × c) × b.
  Proof.
    intros a b c.
    calc ((a × b) × c)
      == (a × (b × c)) { associative }
      == (a × (c × b)) { (commutative (op := mult) b c) }
      == ((a × c) × b) { associative }.
  Qed.

  Theorem zero_ne_one : (0 : F) =/= 1.
  Proof. intros Heq. exact (nontrivial (symmetry Heq)). Qed.

  Theorem mul_inv_cancel : ∀ a : F, a =/= 0 → a × inv a == 1.
  Proof.
    intros a Ha.
    calc (a × inv a)
      == (inv a × a) { commutative }
      == (1 : F) { (left_inverse_on (defined_on := nonzero F) a Ha) }.
  Qed.

  Theorem inv_mul_cancel_left : ∀ a b : F, a =/= 0 → inv a × (a × b) == b.
  Proof.
    intros a b Ha.
    calc (inv a × (a × b))
      == ((inv a × a) × b) { associative }
      == (1 × b) { (left_inverse_on (defined_on := nonzero F) a Ha) }
      == b { left_identity }.
  Qed.

  Theorem mul_inv_cancel_left : ∀ a b : F, a =/= 0 → a × (inv a × b) == b.
  Proof.
    intros a b Ha.
    calc (a × (inv a × b))
      == ((a × inv a) × b) { associative }
      == (1 × b) { (mul_inv_cancel a Ha) }
      == b { left_identity }.
  Qed.

  Theorem mul_left_cancel : ∀ a b c : F, a =/= 0 → a × b == a × c → b == c.
  Proof.
    intros a b c Ha Heq.
    calc b
      == (inv a × (a × b)) { (inv_mul_cancel_left a b Ha) }
      == (inv a × (a × c)) { Heq }
      == c { (inv_mul_cancel_left a c Ha) }.
  Qed.

  Theorem mul_right_cancel : ∀ a b c : F, a =/= 0 → b × a == c × a → b == c.
  Proof.
    intros a b c Ha Heq.
    apply (mul_left_cancel a b c Ha).
    calc (a × b)
      == (b × a) { commutative }
      == (c × a) { Heq }
      == (a × c) { commutative }.
  Qed.

  Theorem mul_ne_zero : ∀ a b : F, a =/= 0 → b =/= 0 → a × b =/= 0.
  Proof.
    intros a b Ha Hb Heq.
    apply Hb.
    calc b
      == (inv a × (a × b)) { (inv_mul_cancel_left a b Ha) }
      == (inv a × 0) { Heq }
      == (0 : F) { mul_zero }.
  Qed.

  Theorem inv_ne_zero : ∀ a : F, a =/= 0 → inv a =/= 0.
  Proof.
    intros a Ha Hinv.
    apply (nontrivial (F := F)).
    calc (1 : F)
      == (inv a × a) { (left_inverse_on (defined_on := nonzero F) a Ha) }
      == (0 × a) { Hinv }
      == (0 : F) { zero_mul }.
  Qed.

  Theorem inv_one : inv (1 : F) == 1.
  Proof.
    calc (inv (1 : F))
      == (inv (1 : F) × 1) { right_identity }
      == (1 : F) { (left_inverse_on (defined_on := nonzero F) 1 nontrivial) }.
  Qed.

  Theorem inv_inv : ∀ a : F, a =/= 0 → inv (inv a) == a.
  Proof.
    intros a Ha.
    apply (mul_left_cancel (inv a) (inv (inv a)) a (inv_ne_zero a Ha)).
    calc (inv a × inv (inv a))
      == (1 : F) { (mul_inv_cancel (inv a) (inv_ne_zero a Ha)) }
      == (inv a × a) { (left_inverse_on (defined_on := nonzero F) a Ha) }.
  Qed.

  Theorem inv_mul : ∀ a b : F, a =/= 0 → b =/= 0 → inv (a × b) == inv a × inv b.
  Proof.
    intros a b Ha Hb.
    apply (mul_left_cancel (a × b) (inv (a × b)) (inv a × inv b) (mul_ne_zero a b Ha Hb)).
    calc ((a × b) × inv (a × b))
      == (1 : F) { (mul_inv_cancel (a × b) (mul_ne_zero a b Ha Hb)) }
      == (a × inv a) { (mul_inv_cancel a Ha) }
      == (a × (b × (inv b × inv a))) { (mul_inv_cancel_left b (inv a) Hb) }
      == (a × (b × (inv a × inv b))) { (commutative (op := mult) (inv b) (inv a)) }
      == ((a × b) × (inv a × inv b)) { associative }.
  Qed.

End FieldTheory.
