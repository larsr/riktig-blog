From Riktig.Structure Require Export Category.

An isomorphism between two objects: an arrow and its inverse, composing to the identity each way.
Module Iso.
  Class Iso (O : Type) `{Category O} (a b : O) (to : a ~> b) (from : b ~> a) := {
    hom_inv_id : from ∘ to == cat_id;
    inv_hom_id : to ∘ from == cat_id;
  }.
End Iso.
Export Iso(Iso).
