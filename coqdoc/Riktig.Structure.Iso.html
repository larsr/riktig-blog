Library Riktig.Structure.Iso

From Coq Require Import RelationClasses.
From Riktig Require Import Ops Lib.Calc.
From Riktig Require Export Structure.Category Structure.TwoSidedInverse.

Section IsoStructure.
  Context {O : Type} `{Category O}.

  Class Iso (x y : O) := {
    iso_to : x ~> y;
    iso_from : y ~> x;
    iso_inverse :: TwoSidedInverse iso_to iso_from;
  }.

  Global Arguments iso_to {x y}.
  Global Arguments iso_from {x y}.
  Global Arguments iso_inverse {x y}.

End IsoStructure.
