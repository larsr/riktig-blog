Library Riktig.Theory.Iso

From Riktig.Structure Require Export Iso.
From Riktig.Theory Require Export Category.

Section IsoTheory.
  Context {O : Type} `{Category O}.

An arrow has at most one inverse.
  Theorem inv_unique : ∀ (a b : O) (to : a ~> b) (from from' : b ~> a),
    from ∘ to == cat_id → to ∘ from' == cat_id → from == from'.
  Proof.
    intros a b to from from' Hl Hr.
    calc from
      == (from ∘ cat_id) { right_identity }
      == (from ∘ (to ∘ from')) { Hr }
      == (from ∘ to ∘ from') { Category.comp_assoc }
      == (cat_id ∘ from') { Hl }
      == from' { left_identity }.
  Qed.

An isomorphism is cancellable on both sides.
  Theorem iso_monic : ∀ (a b x : O) (to : a ~> b) (from : b ~> a) `{!Iso O a b to from}
      (f g : x ~> a), to ∘ f == to ∘ g → f == g.
  Proof.
    intros a b x to from HI f g Hfg.
    calc f
      == (cat_id ∘ f) { left_identity }
      == (from ∘ to ∘ f) { Iso.hom_inv_id }
      == (from ∘ (to ∘ f)) { Category.comp_assoc }
      == (from ∘ (to ∘ g)) { Hfg }
      == (from ∘ to ∘ g) { Category.comp_assoc }
      == (cat_id ∘ g) { Iso.hom_inv_id }
      == g { left_identity }.
  Qed.
End IsoTheory.
