From Riktig Require Export Structure.Iso Lib.Calc.

Section IsoTheory.
  Context {A B : Type} {eqA : EqRel A} {eqB : EqRel B}
          `{!Equivalence eqA} `{!Equivalence eqB}
          {to : A → B} {from : B → A} `{!Iso A B to from}.

  Theorem to_injective : ∀ a a' : A, to a == to a' → a == a'.
  Proof.
    intros a a' Haa'.
    calc a
      == (from (to a)) { from_to }
      == (from (to a')) { Haa' }
      == a' { from_to }.
  Qed.

  Theorem from_injective : ∀ b b' : B, from b == from b' → b == b'.
  Proof.
    intros b b' Hbb'.
    calc b
      == (to (from b)) { to_from }
      == (to (from b')) { Hbb' }
      == b' { to_from }.
  Qed.

  Theorem to_surjective : ∀ b : B, ∃ a : A, to a == b.
  Proof. intros b. ∃ (from b). exact (to_from (to := to) b). Qed.

End IsoTheory.
