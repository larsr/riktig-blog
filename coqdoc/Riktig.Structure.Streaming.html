From Riktig.Structure Require Export Sequence.

A sequence that can be read: next yields exactly the steps, up to the element's equality. This is the law that lets one parser serve every such sequence.
Module Streaming.
  Class Streaming (C S : Type) {eqC : EqRel C} {eqS : EqRel S}
      {stepping : HasStep C S} {Seq : Sequence C S stepping}
      (next : HasNext C S) := {
    next_decides_step : ∀ (s : S) (c : C) (s' : S),
      step s c s' ↔ ∃ c0 : C, next s = Some (c0, s') ∧ c0 == c;
  }.
End Streaming.
Export Streaming(Streaming).
