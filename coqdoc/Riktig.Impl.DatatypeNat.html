From Riktig.Theory Require Import Signature.
From Riktig.Impl Require Export SignaturePoly CategoryFun.
From Riktig.Lib Require Import Coq.Nat.

Coq's nat carries the datatype of the signature 1 + self: build reads a level as zero or successor, the fold is nat_rect restated on that level, and the laws are its equations.
Instance Nat_HasBuild {K : Type} (den : K → Type) :
    HasBuild Type (lobj (layer den (natS (S := poly K)))) nat :=
  fun s ⇒ match s with inl _ ⇒ O | inr n ⇒ S n end.

Instance Nat_HasCata {K : Type} (den : K → Type) :
    HasCata Type (lobj (layer den (natS (S := poly K)))) nat :=
  fun B h ⇒ nat_step h.

Instance Nat_InitialAlgebra {K : Type} (den : K → Type) :
    InitialAlgebra Type (lobj (layer den (natS (S := poly K)))) nat
      (Nat_HasBuild den) (Nat_HasCata den) :=
  {| InitialAlgebra.functor := layer_functor den (natS (S := poly K));
     InitialAlgebra.cata_proper := fun B h h' Hh ⇒ nat_step_ext B h h' Hh;
     InitialAlgebra.cata_build := fun B h s ⇒ nat_step_build B h s;
     InitialAlgebra.cata_unique := fun B h u Hu ⇒
       nat_step_unique B h u (Hu (inl tt)) (fun n ⇒ Hu (inr n)) |}.
