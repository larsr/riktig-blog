Section BMBounded.
  Context {F U V W : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VSW : VectorSpace F W} {nmW : HasNorm F W} {NSW : NormedSpace F W}.

  Definition bm_bounded (B : BilinearMap F U V W) : Prop :=
    ∃ C : F, 0 ≤ C ∧ ∀ (u : U) (v : V), ‖ B u v ‖ ≤ C × ‖ u ‖ × ‖ v ‖.

End BMBounded.

Section BilinearDerivative.
  Context {F U V W : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VSW : VectorSpace F W} {nmW : HasNorm F W} {NSW : NormedSpace F W}.
  Context (B : BilinearMap F U V W) (u0 : U) (v0 : V).

  Definition bilinear_deriv_fun (p : U × V) : W :=
    B (fst p) v0 + B u0 (snd p).

  #[export] Instance bilinear_deriv_proper : Proper (equ ==> equ) bilinear_deriv_fun.
  Proof.
    intros [du1 dv1] [du2 dv2] [Hu Hv].
    unfold bilinear_deriv_fun; simpl.
    rewrite (bilinear_map_proper B du1 du2 Hu v0 v0 (reflexivity v0)),
      (bilinear_map_proper B u0 u0 (reflexivity u0) dv1 dv2 Hv).
    reflexivity.
  Qed.

  Lemma bilinear_deriv_add : ∀ p q : U × V,
    bilinear_deriv_fun (p + q) == bilinear_deriv_fun p + bilinear_deriv_fun q.
  Proof.
    intros [du1 dv1] [du2 dv2].
    unfold bilinear_deriv_fun; simpl.
    rewrite (bilinear_left_additive B du1 du2 v0), (bilinear_right_additive B u0 dv1 dv2).
    apply add_interchange.
  Qed.

  Lemma bilinear_deriv_scale : ∀ (a : F) (p : U × V),
    bilinear_deriv_fun (a *: p) == a *: bilinear_deriv_fun p.
  Proof.
    intros a [du dv].
    unfold bilinear_deriv_fun; simpl.
    rewrite (bilinear_left_scale B a du v0), (bilinear_right_scale B a u0 dv).
    symmetry.
    exact (left_distributive _ _ _).
  Qed.

  Definition bilinear_deriv_lm : LinearMap F (U × V) W :=
    linear_map_from_laws
      bilinear_deriv_fun
      bilinear_deriv_proper
      bilinear_deriv_add
      bilinear_deriv_scale.

  Lemma bilinear_deriv_bounded : bm_bounded B → linear_map_bounded bilinear_deriv_lm.
  Proof.
    intros [C [HC HCb]].
    ∃ (C × ‖ v0 ‖ + C × ‖ u0 ‖).
    split.
    - transitivity ((0 : F) + 0).
      + apply eq_le. symmetry. exact (left_identity _).
      + apply add_le_mono.
        × apply OrderedField.mul_nonneg_closed; [exact HC | apply NormedSpace.nonneg].
        × apply OrderedField.mul_nonneg_closed; [exact HC | apply NormedSpace.nonneg].
    - intros [du dv].
      unfold lm_fun, bilinear_deriv_lm, bilinear_deriv_fun; simpl.
      unfold vnorm, Prod_HasNorm, prod_norm; simpl.
      calc (‖ B du v0 + B u0 dv ‖)
        == (‖ B du v0 ‖ + ‖ B u0 dv ‖)
        == ((C × ‖ v0 ‖ + C × ‖ u0 ‖) × ‖ du ‖
              + (C × ‖ v0 ‖ + C × ‖ u0 ‖) × ‖ dv ‖)
        == ((C × ‖ v0 ‖ + C × ‖ u0 ‖) × (‖ du ‖ + ‖ dv ‖)).
      + apply NormedSpace.triangle.
      + apply add_le_mono.
        × transitivity (C × ‖ v0 ‖ × ‖ du ‖).
          -- transitivity (C × ‖ du ‖ × ‖ v0 ‖).
             ++ exact (HCb du v0).
             ++ apply eq_le. apply mul_assoc_comm_r.
          -- apply mul_le_mono_r.
             ++ apply NormedSpace.nonneg.
             ++ apply le_add_nonneg_r.
                apply OrderedField.mul_nonneg_closed; [exact HC | apply NormedSpace.nonneg].
        × transitivity (C × ‖ u0 ‖ × ‖ dv ‖).
          -- exact (HCb u0 dv).
          -- apply mul_le_mono_r.
             ++ apply NormedSpace.nonneg.
             ++ apply le_add_nonneg_l.
                apply OrderedField.mul_nonneg_closed; [exact HC | apply NormedSpace.nonneg].
      + apply eq_le. symmetry. exact (left_distributive _ _ _).
  Qed.

  Lemma bilinear_remainder_eq : ∀ (du : U) (dv : V),
    B (u0 + du) (v0 + dv) + - (B u0 v0 + (B du v0 + B u0 dv)) == B du dv.
  Proof.
    intros du dv.
    calc (B (u0 + du) (v0 + dv) + - (B u0 v0 + (B du v0 + B u0 dv)))
      == ((B (u0 + du) v0 + B (u0 + du) dv)
            + - (B u0 v0 + (B du v0 + B u0 dv)))
        { (bilinear_right_additive B (u0 + du) v0 dv) }
      == (((B u0 v0 + B du v0) + B (u0 + du) dv)
            + - (B u0 v0 + (B du v0 + B u0 dv)))
        { (bilinear_left_additive B u0 du v0) }
      == (((B u0 v0 + B du v0) + (B u0 dv + B du dv))
            + - (B u0 v0 + (B du v0 + B u0 dv)))
        { (bilinear_left_additive B u0 du dv) }
      == (B du dv)
        { (four_term_cancel (B u0 v0) (B du v0) (B u0 dv) (B du dv)) }.
  Qed.

  Lemma bilinear_remainder_bound : ∀ C : F, 0 ≤ C →
    (∀ (u : U) (v : V), ‖ B u v ‖ ≤ C × ‖ u ‖ × ‖ v ‖) →
    ∀ eps : F, 0 < eps →
    ∀ (du : U) (dv : V),
      ‖ du ‖ + ‖ dv ‖ < eps × inv (1 + C) →
      ‖ B du dv ‖ ≤ eps × (‖ du ‖ + ‖ dv ‖).
  Proof.
    intros C HC HCb eps Heps du dv Hh.
    assert (Hdu_h : ‖ du ‖ ≤ ‖ du ‖ + ‖ dv ‖).
    { apply le_add_nonneg_r. apply NormedSpace.nonneg. }
    assert (Hdv_h : ‖ dv ‖ ≤ ‖ du ‖ + ‖ dv ‖).
    { apply le_add_nonneg_l. apply NormedSpace.nonneg. }
    assert (Hh_nonneg : 0 ≤ ‖ du ‖ + ‖ dv ‖).
    { transitivity (‖ du ‖).
      - apply NormedSpace.nonneg.
      - exact Hdu_h. }
    calc (‖ B du dv ‖)
      == (C × ‖ du ‖ × ‖ dv ‖)
      == (C × (‖ du ‖ × ‖ dv ‖))
      == (C × ((‖ du ‖ + ‖ dv ‖) × (‖ du ‖ + ‖ dv ‖)))
      == ((C × (‖ du ‖ + ‖ dv ‖)) × (‖ du ‖ + ‖ dv ‖))
      == (eps × (‖ du ‖ + ‖ dv ‖)).
    - exact (HCb du dv).
    - apply eq_le. exact (associative _ _ _).
    - apply mul_le_mono_l.
      + exact HC.
      + transitivity ((‖ du ‖ + ‖ dv ‖) × ‖ dv ‖).
        × apply mul_le_mono_r; [apply NormedSpace.nonneg | exact Hdu_h].
        × apply mul_le_mono_l; [exact Hh_nonneg | exact Hdv_h].
    - apply eq_le. symmetry. exact (associative _ _ _).
    - apply mul_le_mono_r.
      + exact Hh_nonneg.
      + transitivity (C × (eps × inv (1 + C))).
        × apply mul_le_mono_l; [exact HC | apply lt_le; exact Hh].
        × apply mul_pad_div_le; [apply lt_le; exact Heps | exact HC].
  Qed.

  Theorem bilinear_is_deriv : bm_bounded B →
    is_deriv (fun p : U × V ⇒ B (fst p) (snd p)) (u0, v0) bilinear_deriv_lm.
  Proof.
    intros Hbb.
    split.
    - exact (bilinear_deriv_bounded Hbb).
    - destruct Hbb as [C [HC HCb]].
      intros eps Heps.
      ∃ (eps × inv (1 + C)).
      split.
      + apply mul_pos.
        × exact Heps.
        × apply pos_inv. apply one_plus_nonneg_pos. exact HC.
      + intros [du dv] Hh.
        simpl.
        unfold lm_fun, bilinear_deriv_lm, bilinear_deriv_fun; simpl.
        calc (‖ B (u0 + du) (v0 + dv) + - (B u0 v0 + (B du v0 + B u0 dv)) ‖)
          == (‖ B du dv ‖)
          == (eps × ‖ (du, dv) ‖).
        × apply eq_le. apply NormedSpace.vnorm_proper. exact (bilinear_remainder_eq du dv).
        × unfold vnorm, Prod_HasNorm, prod_norm; simpl.
          exact (bilinear_remainder_bound C HC HCb eps Heps du dv Hh).
  Qed.

End BilinearDerivative.
