Section UpdFamilies.
  Context {I V : Type} {eI : EqRel I} {IE : Equivalence (@equ I eI)}
    {Ieq : EquDec I} {eV : EqRel V} {VE : Equivalence (@equ V eV)}.
  Context {Vu : HasUpd I V} {VuL : Upd I V}.

  Lemma upd_val_proper : ∀ (ys : I → V) (i : I) (x y : V),
    x == y → famEq (upd ys i x) (upd ys i y).
  Proof.
    intros ys i x y Hxy j.
    destruct (dec (j == i)) as [Hji | Hji].
    - calc (upd ys i x j)
        == x { (upd_at ys i x j Hji) }
        == y { Hxy }
        == (upd ys i y j) { (symmetry (upd_at ys i y j Hji)) }.
    - assert (Hij : ¬ (i == j)) by (intros Hc; exact (Hji (symmetry Hc))).
      calc (upd ys i x j)
        == (ys j) { (upd_off ys i j x Hij) }
        == (upd ys i y j) { (symmetry (upd_off ys i j y Hij)) }.
  Qed.

  Lemma upd_family_proper : ∀ (ws : I → V),
    Proper (equ ==> equ) ws →
    ∀ (i : I) (x : V), Proper (equ ==> equ) (upd ws i x).
  Proof.
    intros ws Pws i x k k' Hkk'.
    destruct (dec (k == i)) as [Hki | Hki].
    - calc (upd ws i x k)
        == x { (upd_at ws i x k Hki) }
        == (upd ws i x k') { (symmetry (upd_at ws i x k'
                               (transitivity (symmetry Hkk') Hki))) }.
    - assert (Hik : ¬ (i == k)) by (intros Hc; exact (Hki (symmetry Hc))).
      assert (Hik' : ¬ (i == k'))
        by (intros Hc; exact (Hik (transitivity Hc (symmetry Hkk')))).
      calc (upd ws i x k)
        == (ws k) { (upd_off ws i k x Hik) }
        == (ws k') { (Pws k k' Hkk') }
        == (upd ws i x k') { (symmetry (upd_off ws i k' x Hik')) }.
  Qed.

  Lemma u_idem : ∀ (ws : I → V), Proper (equ ==> equ) ws →
    ∀ (i : I) (x : V),
    x == ws i → ∀ k, upd ws i x k == ws k.
  Proof.
    intros ws Pws i x Hx k; destruct (dec (k == i)) as [Hki | Hki].
    - calc (upd ws i x k)
        == x { (upd_at ws i x k Hki) }
        == (ws i) { Hx }
        == (ws k) { (Pws i k (symmetry Hki)) }.
    - assert (Hik : ¬ (i == k)) by (intros Hc; exact (Hki (symmetry Hc))).
      exact (upd_off ws i k x Hik).
  Qed.

  Lemma upd_upd_commutes : ∀ (ws : I → V) (i j : I) (x y : V),
    ¬ (i == j) →
    ∀ k, upd (upd ws i x) j y k == upd (upd ws j y) i x k.
  Proof.
    intros ws i j x y Hij k.
    destruct (dec (k == j)) as [Hkj | Hkj];
      destruct (dec (k == i)) as [Hki | Hki].
    - destruct (Hij (transitivity (symmetry Hki) Hkj)).
    - assert (Hik : ¬ (i == k))
        by (intros Hc; exact (Hki (symmetry Hc))).
      calc (upd (upd ws i x) j y k)
        == y { (upd_at (upd ws i x) j y k Hkj) }
        == (upd ws j y k) { (symmetry (upd_at ws j y k Hkj)) }
        == (upd (upd ws j y) i x k)
             { (symmetry (upd_off (upd ws j y) i k x Hik)) }.
    - assert (Hjk : ¬ (j == k))
        by (intros Hc; exact (Hkj (symmetry Hc))).
      calc (upd (upd ws i x) j y k)
        == (upd ws i x k) { (upd_off (upd ws i x) j k y Hjk) }
        == x { (upd_at ws i x k Hki) }
        == (upd (upd ws j y) i x k)
             { (symmetry (upd_at (upd ws j y) i x k Hki)) }.
    - assert (Hjk : ¬ (j == k))
        by (intros Hc; exact (Hkj (symmetry Hc))).
      assert (Hik : ¬ (i == k))
        by (intros Hc; exact (Hki (symmetry Hc))).
      calc (upd (upd ws i x) j y k)
        == (upd ws i x k) { (upd_off (upd ws i x) j k y Hjk) }
        == (ws k) { (upd_off ws i k x Hik) }
        == (upd ws j y k) { (symmetry (upd_off ws j k y Hjk)) }
        == (upd (upd ws j y) i x k)
             { (symmetry (upd_off (upd ws j y) i k x Hik)) }.
  Qed.

End UpdFamilies.

Section SwapNegates.
  Context {F V : Type} `{VV : VectorSpace F V} {I : Type}
    {eI : EqRel I} {IE : Equivalence (@equ I eI)} {Ieq : EquDec I}.
  Context {Vu : HasUpd I V} {VuL : Upd I V}.
  Context (m : MultilinearForm I F V) (Halt : Alternating m).
  Context (swap : I → I → I → I) {Sw : Swap swap}.

  Definition fswap (vs : I → V) (i j : I) : I → V :=
    fun k ⇒ vs (swap i j k).


  Lemma fswap_at_i : ∀ (vs : I → V), Proper (equ ==> equ) vs →
    ∀ i j : I, fswap vs i j i == vs j.
  Proof.
    intros vs Pvs i j.
    unfold fswap.
    exact (Pvs (swap i j i) j (swap_at_i swap i j i (reflexivity i))).
  Qed.

  Lemma fswap_at_j : ∀ (vs : I → V), Proper (equ ==> equ) vs →
    ∀ i j : I, ¬ (i == j) → fswap vs i j j == vs i.
  Proof.
    intros vs Pvs i j Hij.
    unfold fswap.
    exact (Pvs (swap i j j) i (swap_at_j swap i j j Hij (reflexivity j))).
  Qed.

  Theorem mf_swap_neg : ∀ (vs : I → V), Proper (equ ==> equ) vs →
    ∀ (i j : I),
    ¬ (i == j) → m (fswap vs i j) == - m vs.
  Proof.
    intros vs Pvs i j Hij.
    pose (a := vs i). pose (b := vs j).
    pose (VS' := upd vs i (a + b)).
    assert (PVS' : Proper (equ ==> equ) VS')
      by exact (upd_family_proper vs Pvs i (a + b)).
    assert (Hji : ¬ (j == i)) by (intro Heq; apply Hij; symmetry; exact Heq).
    assert (HW0 : m (upd VS' j a) + m (upd VS' j b) == 0).
    { assert (HWi : upd VS' j (a + b) i == a + b).
      { calc (upd VS' j (a + b) i)
          == (VS' i) { (upd_off VS' j i (a + b) Hji) }
          == (a + b) { (upd_at vs i (a + b) i (reflexivity i)) }. }
      assert (HWj : upd VS' j (a + b) j == a + b)
        by exact (upd_at VS' j (a + b) j (reflexivity j)).
      assert (HW0' : m (upd VS' j (a + b)) == 0).
      { apply (alternating_repeat_zero Halt (upd VS' j (a + b)) i j Hij).
        rewrite HWi, HWj. reflexivity. }
      rewrite <- HW0'.
      symmetry.
      exact (multilinear_form_additive m VS' j a b). }
    assert (HP2val : m (upd VS' j b) == m vs).
    { assert (HVS'j : b == VS' j)
        by (unfold VS'; rewrite (upd_off vs i j (a + b) Hij); reflexivity).
      assert (HP2 : ∀ k, upd VS' j b k == VS' k)
        by (intros k; exact (u_idem VS' PVS' j b HVS'j k)).
      assert (Hm_P2 : m (upd VS' j b) == m VS')
        by (apply (multilinear_form_proper m); exact HP2).
      assert (Hexp2 : m VS' == m (upd vs i a) + m (upd vs i b))
        by exact (multilinear_form_additive m vs i a b).
      assert (HVSia : ∀ k, upd vs i a k == vs k)
        by (intros k; exact (u_idem vs Pvs i a (reflexivity a) k)).
      assert (Hm1 : m (upd vs i a) == m vs)
        by (apply (multilinear_form_proper m); exact HVSia).
      assert (Hm2 : m (upd vs i b) == 0).
      { apply (alternating_repeat_zero Halt (upd vs i b) i j Hij).
        rewrite (upd_at vs i b i (reflexivity i)),
          (upd_off vs i j b Hij).
        reflexivity. }
      calc (m (upd VS' j b))
        == (m VS') { Hm_P2 }
        == (m (upd vs i a) + m (upd vs i b)) { Hexp2 }
        == (m vs + m (upd vs i b)) { Hm1 }
        == (m vs + 0) { Hm2 }
        == (m vs) { (right_identity (m vs)) }. }
    assert (HP1val : m (upd VS' j a) == m (fswap vs i j)).
    { assert (Hreorder : ∀ k, upd VS' j a k == upd (upd vs j a) i (a + b) k).
      { intros k. unfold VS'. exact (upd_upd_commutes vs i j (a + b) a Hij k). }
      assert (Hm_P1 : m (upd VS' j a) == m (upd (upd vs j a) i (a + b)))
        by (apply (multilinear_form_proper m); exact Hreorder).
      assert (Hexp3 : m (upd (upd vs j a) i (a + b))
                    == m (upd (upd vs j a) i a) + m (upd (upd vs j a) i b))
        by exact (multilinear_form_additive m (upd vs j a) i a b).
      assert (HT1 : m (upd (upd vs j a) i a) == 0).
      { apply (alternating_repeat_zero Halt (upd (upd vs j a) i a) i j Hij).
        calc (upd (upd vs j a) i a i)
          == a { (upd_at (upd vs j a) i a i (reflexivity i)) }
          == (upd vs j a j) { (upd_at vs j a j (reflexivity j)) }
          == (upd (upd vs j a) i a j) { (upd_off (upd vs j a) i j a Hij) }. }
      assert (HT2 : ∀ k, upd (upd vs j a) i b k == fswap vs i j k).
      { intros k.
        unfold fswap.
        destruct (dec (k == i)) as [Hki | Hki].
        - calc (upd (upd vs j a) i b k)
            == b { (upd_at (upd vs j a) i b k Hki) }
            == (vs (swap i j k))
                 { (symmetry (Pvs (swap i j k) j
                     (swap_at_i swap i j k Hki))) }.
        - assert (Hik : ¬ (i == k))
            by (intros Hc; exact (Hki (symmetry Hc))).
          destruct (dec (k == j)) as [Hkj | Hkj].
          + calc (upd (upd vs j a) i b k)
              == (upd vs j a k) { (upd_off (upd vs j a) i k b Hik) }
              == a { (upd_at vs j a k Hkj) }
              == (vs (swap i j k))
                   { (symmetry (Pvs (swap i j k) i
                       (swap_at_j swap i j k Hij Hkj))) }.
          + assert (Hjk : ¬ (j == k))
              by (intros Hc; exact (Hkj (symmetry Hc))).
            calc (upd (upd vs j a) i b k)
              == (upd vs j a k) { (upd_off (upd vs j a) i k b Hik) }
              == (vs k) { (upd_off vs j k a Hjk) }
              == (vs (swap i j k))
                   { (symmetry (Pvs (swap i j k) k
                       (swap_ne swap i j k Hki Hkj))) }. }
      assert (HT2m : m (upd (upd vs j a) i b) == m (fswap vs i j))
        by (apply (multilinear_form_proper m); exact HT2).
      calc (m (upd VS' j a))
        == (m (upd (upd vs j a) i (a + b))) { Hm_P1 }
        == (m (upd (upd vs j a) i a) + m (upd (upd vs j a) i b)) { Hexp3 }
        == (0 + m (upd (upd vs j a) i b)) { HT1 }
        == (0 + m (fswap vs i j)) { HT2m }
        == (m (fswap vs i j)) { (left_identity (m (fswap vs i j))) }. }
    assert (Hneg : m (upd VS' j a) == - m (upd VS' j b))
      by exact (neg_unique_l (m (upd VS' j b)) (m (upd VS' j a)) HW0).
    rewrite HP1val, HP2val in Hneg.
    exact Hneg.
  Qed.

End SwapNegates.

Section Reindex.
  Context {F V : Type} `{VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context {Vu : HasUpd I V} {VuL : Upd I V}.

  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma alternating_sign_fixed_outside : ∀ (L : list I) (sigma : I → I),
    Proper (equ ==> equ) sigma → Injective I I sigma →
    (∀ x : I, ¬ InA equ x L → sigma x == x) →
    ∃ s : F, (s == 1 ∨ s == - (1)) ∧
      ∀ m : MultilinearForm I F V, Alternating m →
      ∀ vs : I → V, Proper (equ ==> equ) vs →
        m (fun i ⇒ vs (sigma i)) == s × m vs.
  Proof.
    intros L; induction L as [| a L IH]; intros sigma Ps Hinj Hfix.
    - assert (Hid : ∀ x : I, sigma x == x)
        by (intros x; apply Hfix; intros Hc; inversion Hc).
      ∃ 1. split; [left; reflexivity |].
      intros m Halt vs Pvs.
      calc (m (fun i ⇒ vs (sigma i)))
        == (m vs)
        == ((1 : F) × m vs).
      + apply (multilinear_form_proper m). intros i.
        exact (Pvs (sigma i) i (Hid i)).
      + ring.
    - destruct (dec (sigma a == a)) as [Hsa | Hsa].
      +
        apply (IH sigma Ps Hinj).
        intros x Hx.
        destruct (dec (x == a)) as [Hxa | Hxa].
        × calc (sigma x)
            == (sigma a) { (Ps x a Hxa) }
            == a { Hsa }
            == x { (symmetry Hxa) }.
        × apply Hfix. intros Hc.
          destruct (proj1 (InA_cons equ x a L) Hc) as [Hc' | Hc'];
            [exact (Hxa Hc') | exact (Hx Hc')].
      + assert (Hne : ¬ (a == sigma a)) by (intros Hc; exact (Hsa (symmetry Hc))).
        destruct (swap_exists (I := I)) as [sw Hsw].
        assert (Ps' : Proper (equ ==> equ) (fun x ⇒ sw a (sigma a) (sigma x)))
          by exact (fun x y Hxy ⇒
                      swap_proper sw a (sigma a) (sigma x) (sigma y) (Ps x y Hxy)).
        assert (Hinj' : Injective I I (fun x ⇒ sw a (sigma a) (sigma x)))
          by exact (fun x y Hxy ⇒
                      Hinj x y (swap_inj sw a (sigma a) (sigma x) (sigma y) Hxy)).
        assert (Hfix' : ∀ x : I, ¬ InA equ x L →
                  sw a (sigma a) (sigma x) == x).
        { intros x Hx.
          destruct (dec (x == a)) as [Hxa | Hxa].
          - calc (sw a (sigma a) (sigma x))
              == (sw a (sigma a) (sigma a))
                   { (swap_proper sw a (sigma a) (sigma x) (sigma a) (Ps x a Hxa)) }
              == a { (swap_at_j sw a (sigma a) (sigma a) Hne (reflexivity (sigma a))) }
              == x { (symmetry Hxa) }.
          - assert (Hfx : sigma x == x).
            { apply Hfix. intros Hc.
              destruct (proj1 (InA_cons equ x a L) Hc) as [Hc' | Hc'];
                [exact (Hxa Hc') | exact (Hx Hc')]. }
            assert (Hxsa : ¬ (x == sigma a))
              by (intros Hc; exact (Hxa (Hinj x a (transitivity Hfx Hc)))).
            calc (sw a (sigma a) (sigma x))
              == (sw a (sigma a) x)
                   { (swap_proper sw a (sigma a) (sigma x) x Hfx) }
              == x { (swap_ne sw a (sigma a) x Hxa Hxsa) }. }
        destruct (IH (fun x ⇒ sw a (sigma a) (sigma x)) Ps' Hinj' Hfix')
          as [s [Hs Hsign]].
        ∃ (- s). split.
        × destruct Hs as [Hs1 | Hs1]; [right | left]; rewrite Hs1.
          -- reflexivity.
          -- exact (neg_involutive 1).
        × intros m Halt vs Pvs.
          assert (Pfs : Proper (equ ==> equ) (fswap sw vs a (sigma a)))
            by exact (fun x y Hxy ⇒
                        Pvs (sw a (sigma a) x) (sw a (sigma a) y)
                          (swap_proper sw a (sigma a) x y Hxy)).
          calc (m (fun i ⇒ vs (sigma i)))
            == (m (fun i ⇒ fswap sw vs a (sigma a) (sw a (sigma a) (sigma i))))
            == (s × m (fswap sw vs a (sigma a)))
            == (s × (- m vs))
            == ((- s) × m vs).
          -- apply (multilinear_form_proper m). intros i.
             symmetry.
             exact (Pvs _ _ (swap_invol sw a (sigma a) (sigma i))).
          -- exact (Hsign m Halt (fswap sw vs a (sigma a)) Pfs).
          -- rewrite (mf_swap_neg m Halt sw vs Pvs a (sigma a) Hne). reflexivity.
          -- ring.
  Qed.

Reindexing a family by an injective self-map multiplies every alternating form by one and the same sign.
  Theorem alternating_reindex_sign : ∀ sigma : I → I,
    Proper (equ ==> equ) sigma → Injective I I sigma →
    ∃ s : F, (s == 1 ∨ s == - (1)) ∧
      ∀ m : MultilinearForm I F V, Alternating m →
      ∀ vs : I → V, Proper (equ ==> equ) vs →
        m (fun i ⇒ vs (sigma i)) == s × m vs.
  Proof.
    intros sigma Ps Hinj.
    apply (alternating_sign_fixed_outside enum sigma Ps Hinj).
    intros x Hx. destruct (Hx (enum_complete x)).
  Qed.

End Reindex.

Section AlternatingOnBasis.

          {BL : BigOp F bigop_list}.

  Add Ring F_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma multilinear_slot_zero : ∀ (m : MultilinearForm I F V)
      (vs : I → V) (i : I),
    m (upd vs i 0) == 0.
  Proof.
    intros m vs i.
    calc (m (upd vs i 0))
      == (m (upd vs i ((0 : F) *: (0 : V))))
      == ((0 : F) × m (upd vs i 0))
      == (0 : F).
    - apply (multilinear_form_proper m).
      exact (upd_val_proper vs i 0 ((0 : F) *: (0 : V))
               (symmetry (zero_scale (0 : V)))).
    - exact (multilinear_form_scale m vs i 0 0).
    - ring.
  Qed.

  Lemma multilinear_slot_bigop_list : ∀ (m : MultilinearForm I F V)
      (vs : I → V) (i : I) {J : Type} (g : J → V) (l : list J),
    m (upd vs i (bigop_list_vec g l))
      == bigop_list (fun j ⇒ m (upd vs i (g j))) l.
  Proof.
    intros m vs i J g l; induction l as [| b l IH].
    - calc (m (upd vs i (bigop_list_vec g nil)))
        == (m (upd vs i 0))
        == (0 : F)
        == (bigop_list (fun j ⇒ m (upd vs i (g j))) nil).
      + apply (multilinear_form_proper m).
        exact (upd_val_proper vs i (bigop_list_vec g nil) 0 (bigop_list_nil (bigop_list := @bigop_list_vec) g)).
      + exact (multilinear_slot_zero m vs i).
      + symmetry. exact (bigop_list_nil (fun j ⇒ m (upd vs i (g j)))).
    - calc (m (upd vs i (bigop_list_vec g (b :: l))))
        == (m (upd vs i (g b + bigop_list_vec g l)))
        == (m (upd vs i (g b)) + m (upd vs i (bigop_list_vec g l)))
        == (m (upd vs i (g b)) + bigop_list (fun j ⇒ m (upd vs i (g j))) l)
        == (bigop_list (fun j ⇒ m (upd vs i (g j))) (b :: l)).
      + apply (multilinear_form_proper m).
        exact (upd_val_proper vs i (bigop_list_vec g (b :: l))
                 (g b + bigop_list_vec g l) (bigop_list_cons (bigop_list := @bigop_list_vec) g b l)).
      + exact (multilinear_form_additive m vs i (g b) (bigop_list_vec g l)).
      + rewrite IH. reflexivity.
      + symmetry.
        exact (bigop_list_cons (fun j ⇒ m (upd vs i (g j))) b l).
  Qed.

  Lemma multilinear_slot_bigopI : ∀ (m : MultilinearForm I F V)
      (vs : I → V) (i : I) (g : I → V),
    m (upd vs i (bigop_list_vec g enum)) == bigop_list (fun k ⇒ m (upd vs i (g k))) enum.
  Proof.
    intros m vs i g.
    calc (m (upd vs i (bigop_list_vec g enum)))
      == (m (upd vs i (bigop_list_vec g enum)))
      == (bigop_list (fun k ⇒ m (upd vs i (g k))) enum)
      == (bigop_list (fun k ⇒ m (upd vs i (g k))) enum).
    - apply (multilinear_form_proper m).
      exact (upd_val_proper vs i (bigop_list_vec g enum) (bigop_list_vec g enum) (reflexivity (bigop_list_vec g enum))).
    - exact (multilinear_slot_bigop_list m vs i g enum).
    - symmetry. exact (reflexivity (bigop_list (fun k ⇒ m (upd vs i (g k))) enum)).
  Qed.

One slot of a multilinear form, resolved into the basis.
  Lemma multilinear_slot_coords : ∀ (m : MultilinearForm I F V)
      (vs : I → V) (i : I) (v : V),
    m (upd vs i v)
      == bigop_list (fun k ⇒ Basis.coords B v k × m (upd vs i (Basis.vec B k))) enum.
  Proof.
    intros m vs i v.
    calc (m (upd vs i v))
      == (m (upd vs i (bigop_list_vec (fun k ⇒ Basis.coords B v k *: Basis.vec B k) enum)))
      == (bigop_list (fun k ⇒ m (upd vs i (Basis.coords B v k *: Basis.vec B k))) enum)
      == (bigop_list (fun k ⇒ Basis.coords B v k × m (upd vs i (Basis.vec B k))) enum).
    - apply (multilinear_form_proper m).
      apply (upd_val_proper vs i).
      calc v
        == (lincomb (Basis.vec B) (Basis.coords B v))
             { (symmetry (Basis.reconstruction B v)) }
        == (bigop_list_vec (fun k ⇒ Basis.coords B v k *: Basis.vec B k) enum)
             { lincomb_def }.
    - exact (multilinear_slot_bigopI m vs i
               (fun k ⇒ Basis.coords B v k *: Basis.vec B k)).
    - apply bigopI_ext. intros k.
      exact (multilinear_form_scale m vs i
               (Basis.coords B v k) (Basis.vec B k)).
  Qed.

  Lemma alternating_pair_on_basis : ∀ (u w : MultilinearForm I F V),
    Alternating u → Alternating w →
    ∀ sigma : I → I, Proper (equ ==> equ) sigma →
    u (fun i ⇒ Basis.vec B (sigma i)) × w (Basis.vec B)
      == w (fun i ⇒ Basis.vec B (sigma i)) × u (Basis.vec B).
  Proof.
    intros u w Hu Hw sigma Ps.
    destruct (finindex_identifies_or_injective sigma Ps)
      as [[i [j [Hij Hs]]] | Hinj].
    - assert (Hrep : ∀ m : MultilinearForm I F V, Alternating m →
        m (fun k ⇒ Basis.vec B (sigma k)) == 0).
      { intros m Halt.
        exact (alternating_repeat_zero Halt
                 (fun k ⇒ Basis.vec B (sigma k)) i j Hij
                 (Basis.vec_proper B (sigma i) (sigma j) Hs)). }
      rewrite (Hrep u Hu), (Hrep w Hw). ring.
    - destruct (alternating_reindex_sign (F := F) (V := V) sigma Ps Hinj)
        as [s [_ Hsign]].
      rewrite (Hsign u Hu (Basis.vec B) (Basis.vec_proper B)),
              (Hsign w Hw (Basis.vec B) (Basis.vec_proper B)).
      ring.
  Qed.

  Lemma alternating_pair_basis_outside : ∀ (u w : MultilinearForm I F V),
    Alternating u → Alternating w →
    ∀ (L : list I) (sigma : I → I), Proper (equ ==> equ) sigma →
    ∀ vs : I → V, Proper (equ ==> equ) vs →
    (∀ x : I, ¬ InA equ x L → vs x == Basis.vec B (sigma x)) →
    u vs × w (Basis.vec B) == w vs × u (Basis.vec B).
  Proof.
    intros u w Hu Hw L; induction L as [| a L IH]; intros sigma Ps vs Pvs Hout.
    - assert (Hfam : ∀ x : I, vs x == Basis.vec B (sigma x))
        by (intros x; apply Hout; intros Hc; inversion Hc).
      rewrite (multilinear_form_proper u vs (fun i ⇒ Basis.vec B (sigma i)) Hfam),
              (multilinear_form_proper w vs (fun i ⇒ Basis.vec B (sigma i)) Hfam).
      exact (alternating_pair_on_basis u w Hu Hw sigma Ps).
    - assert (Hexpand : ∀ m : MultilinearForm I F V,
        m vs == bigop_list (fun k ⇒ Basis.coords B (vs a) k
                                   × m (upd vs a (Basis.vec B k))) enum).
      { intros m.
        calc (m vs)
          == (m (upd vs a (vs a)))
          == (bigop_list (fun k ⇒ Basis.coords B (vs a) k
                                 × m (upd vs a (Basis.vec B k))) enum).
        - symmetry. apply (multilinear_form_proper m).
          exact (u_idem vs Pvs a (vs a) (reflexivity (vs a))).
        - exact (multilinear_slot_coords m vs a (vs a)). }
      assert (Hterm : ∀ k : I,
        Basis.coords B (vs a) k × u (upd vs a (Basis.vec B k))
          × w (Basis.vec B)
        == Basis.coords B (vs a) k × w (upd vs a (Basis.vec B k))
          × u (Basis.vec B)).
      { intros k.
        destruct (family_update_exists (A := I) _ sigma a k Ps)
          as [tau [Ptau [Htau_at Htau_off]]].
        assert (Hk : u (upd vs a (Basis.vec B k)) × w (Basis.vec B)
                     == w (upd vs a (Basis.vec B k)) × u (Basis.vec B)).
        { apply (IH tau Ptau (upd vs a (Basis.vec B k))
                   (upd_family_proper vs Pvs a (Basis.vec B k))).
          intros x Hx.
          destruct (dec (x == a)) as [Hxa | Hxa].
          - calc (upd vs a (Basis.vec B k) x)
              == (Basis.vec B k) { (upd_at vs a (Basis.vec B k) x Hxa) }
              == (Basis.vec B (tau x))
                   { (Basis.vec_proper B k (tau x)
                       (symmetry (transitivity (Ptau x a Hxa) Htau_at))) }.
          - assert (Hax : ¬ (a == x)) by (intros Hc; exact (Hxa (symmetry Hc))).
            calc (upd vs a (Basis.vec B k) x)
              == (vs x) { (upd_off vs a x (Basis.vec B k) Hax) }
              == (Basis.vec B (sigma x))
                   { (Hout x (fun Hc ⇒
                        match proj1 (InA_cons equ x a L) Hc with
                        | or_introl Hc' ⇒ Hxa Hc'
                        | or_intror Hc' ⇒ Hx Hc'
                        end)) }
              == (Basis.vec B (tau x))
                   { (Basis.vec_proper B (sigma x) (tau x)
                       (symmetry (Htau_off x Hxa))) }. }
        transitivity (Basis.coords B (vs a) k
                        × (u (upd vs a (Basis.vec B k)) × w (Basis.vec B))).
        { ring. }
        rewrite Hk. ring. }
      calc (u vs × w (Basis.vec B))
        == (bigop_list (fun k ⇒ Basis.coords B (vs a) k
                               × u (upd vs a (Basis.vec B k))) enum
              × w (Basis.vec B))
        == (bigop_list (fun k ⇒ Basis.coords B (vs a) k
                               × u (upd vs a (Basis.vec B k))
                               × w (Basis.vec B)) enum)
        == (bigop_list (fun k ⇒ Basis.coords B (vs a) k
                               × w (upd vs a (Basis.vec B k))
                               × u (Basis.vec B)) enum)
        == (bigop_list (fun k ⇒ Basis.coords B (vs a) k
                               × w (upd vs a (Basis.vec B k))) enum
              × u (Basis.vec B))
        == (w vs × u (Basis.vec B)).
      + rewrite (Hexpand u). reflexivity.
      + symmetry.
        exact (bigopI_mul_r (w (Basis.vec B))
                 (fun k ⇒ Basis.coords B (vs a) k
                             × u (upd vs a (Basis.vec B k)))).
      + apply bigopI_ext. exact Hterm.
      + exact (bigopI_mul_r (u (Basis.vec B))
                 (fun k ⇒ Basis.coords B (vs a) k
                             × w (upd vs a (Basis.vec B k)))).
      + rewrite (Hexpand w). reflexivity.
  Qed.

Two alternating forms over one basis are proportional: each is determined by its value on the basis.
  Theorem alternating_forms_proportional : ∀ (u w : MultilinearForm I F V),
    Alternating u → Alternating w →
    ∀ vs : I → V, Proper (equ ==> equ) vs →
      u vs × w (Basis.vec B) == w vs × u (Basis.vec B).
  Proof.
    intros u w Hu Hw vs Pvs.
    apply (alternating_pair_basis_outside u w Hu Hw enum (fun x ⇒ x)
             (fun x y Hxy ⇒ Hxy) vs Pvs).
    intros x Hx. destruct (Hx (enum_complete x)).
  Qed.

End AlternatingOnBasis.
