SumTo f L: the finite sections of f converge to L.
  Definition SumTo (f : nat → R) (L : R) : Prop :=
    seq_lim (fun n ⇒ bigop n f) L.

sections_bounded f: the finite sections of f stay below one bound — the hypothesis of the existence theorem, and, at the absolute values, the summability the instance runs on.
  Definition sections_bounded (f : nat → R) : Prop :=
    ∃ B : R, ∀ n : nat, bigop n f ≤ B.

  Theorem sum_to_unique : ∀ (f : nat → R) (L M : R),
    SumTo f L → SumTo f M → L == M.
  Proof. intros f L M. exact (lim_unique (fun n ⇒ bigop n f) L M). Qed.

  Theorem sum_to_add : ∀ (f g : nat → R) (L M : R),
    SumTo f L → SumTo g M → SumTo (fun i ⇒ f i + g i) (L + M).
  Proof.
    intros f g L M Hf Hg.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ bigop n f + bigop n g)
                    (fun n ⇒ bigop n (fun i ⇒ f i + g i))
                    (fun n ⇒ symmetry (bigop_add n f g))
                    (L + M) (L + M) (reflexivity (L + M)))
             (lim_add (fun n ⇒ bigop n f) (fun n ⇒ bigop n g) L M Hf Hg)).
  Qed.

  Theorem sum_to_neg : ∀ (f : nat → R) (L : R),
    SumTo f L → SumTo (fun i ⇒ - f i) (- L).
  Proof.
    intros f L Hf.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ - bigop n f)
                    (fun n ⇒ bigop n (fun i ⇒ - f i))
                    (fun n ⇒ symmetry (bigop_neg n f))
                    (- L) (- L) (reflexivity (- L)))
             (lim_neg (fun n ⇒ bigop n f) L Hf)).
  Qed.

  Theorem sum_to_scale : ∀ (a : R) (f : nat → R) (L : R),
    SumTo f L → SumTo (fun i ⇒ a *: f i) (a *: L).
  Proof.
    intros a f L Hf.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ a *: bigop n f)
                    (fun n ⇒ bigop n (fun i ⇒ a *: f i))
                    (fun n ⇒ symmetry (bigop_scale n a f))
                    (a *: L) (a *: L) (reflexivity (a *: L)))
             (lim_scale a (fun n ⇒ bigop n f) L Hf)).
  Qed.

  Theorem sum_to_value : ∀ (f : nat → R) (L M : R),
    L == M → SumTo f L → SumTo f M.
  Proof.
    intros f L M HLM.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ bigop n f) (fun n ⇒ bigop n f)
                    (fun n ⇒ reflexivity (bigop n f)) L M HLM)).
  Qed.

  Theorem sum_to_ext : ∀ (f g : nat → R) (L : R),
    (∀ i : nat, f i == g i) → SumTo f L → SumTo g L.
  Proof.
    intros f g L Hfg.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ bigop n f) (fun n ⇒ bigop n g)
                    (fun n ⇒ bigop_ext n f g Hfg) L L (reflexivity L))).
  Qed.

  Lemma sections_settle : ∀ (f : nat → R) (N : nat),
    (∀ m : nat, (N ≤ m)%nat → f m == 0) →
    ∀ n : nat, (N ≤ n)%nat → bigop n f == bigop N f.
  Proof.
    intros f N Hzero n; induction n as [| n IH]; intros Hn.
    - rewrite (proj1 (Nat.le_0_r N) Hn). reflexivity.
    - destruct (proj1 (Nat.lt_eq_cases N (S n)) Hn) as [Hlt | Heq].
      + assert (HNn : (N ≤ n)%nat) by exact (proj1 (Nat.lt_succ_r N n) Hlt).
        calc (bigop (S n) f)
          == (bigop n f + f n) { (bigop_snoc n f) }
          == (bigop N f + f n) { (IH HNn) }
          == (bigop N f + (0 : R)) { (Hzero n HNn) }
          == (bigop N f) { right_identity }.
      + rewrite Heq. reflexivity.
  Qed.

  Theorem sum_to_supported : ∀ (f : nat → R) (N : nat),
    (∀ m : nat, (N ≤ m)%nat → f m == 0) → SumTo f (bigop N f).
  Proof.
    intros f N Hzero eps Heps.
    ∃ N.
    intros n Hn.
    apply (le_lt_trans _ 0 eps).
    - apply eq_le.
      calc (‖ bigop n f + - bigop N f ‖)
        == (‖ bigop N f + - bigop N f ‖)
             { (sections_settle f N Hzero n Hn) }
        == (0 : R)
             { (norm_sub_self (bigop N f)) }.
    - exact Heps.
  Qed.

  Theorem sections_monotone : ∀ (f : nat → R),
    (∀ i : nat, 0 ≤ f i) →
    ∀ m n : nat, (m ≤ n)%nat → bigop m f ≤ bigop n f.
  Proof.
    intros f Hf m n; induction n as [| n IH]; intros Hmn.
    - rewrite (proj1 (Nat.le_0_r m) Hmn). reflexivity.
    - destruct (proj1 (Nat.lt_eq_cases m (S n)) Hmn) as [Hlt | Heq].
      + apply (le_transitive _ (bigop n f + f n) _).
        × apply (le_transitive _ (bigop n f) _).
          -- exact (IH (proj1 (Nat.lt_succ_r m n) Hlt)).
          -- apply le_add_nonneg_r. exact (Hf n).
        × apply eq_le. symmetry. exact (bigop_snoc n f).
      + rewrite Heq. reflexivity.
  Qed.


  Lemma norm_is_abs : ∀ x : R, ‖ x ‖ == abs x.
  Proof. intros x. reflexivity. Qed.

  Lemma norm_sub_zero : ∀ x : R, ‖ x + - (0 : R) ‖ == abs x.
  Proof. intros x. apply abs_proper. ring. Qed.

section_set f is the set of finite sections of f — the set the completeness field is applied to.
  Definition section_set (f : nat → R) : Pred R :=
    fun x ⇒ ∃ n : nat, x == bigop n f.

  Lemma sections_lub : ∀ f : nat → R, sections_bounded f →
    ∃ s : R,
      (∀ n : nat, bigop n f ≤ s)
      ∧ (∀ b : R, (∀ n : nat, bigop n f ≤ b) → s ≤ b).
  Proof.
    intros f [B HB].
    assert (Hinhabited : ∃ x : R, section_set f x).
    { ∃ (bigop O f). ∃ O. reflexivity. }
    assert (Hbound : ∃ b : R, ∀ x : R, section_set f x → x ≤ b).
    { ∃ B. intros x [n Hn].
      exact (le_rewrite_l (bigop n f) x B (symmetry Hn) (HB n)). }
    destruct (complete (section_set f) Hinhabited Hbound)
      as [s [Hub Hleast]].
    ∃ s.
    split.
    - intros n. apply Hub. ∃ n. reflexivity.
    - intros b Hb.
      apply Hleast.
      intros x Hx.
      destruct Hx as [n Hn].
      exact (le_rewrite_l (bigop n f) x b (symmetry Hn) (Hb n)).
  Qed.

  Lemma lub_section_index : ∀ (f : nat → R) (s d : R),
    (∀ b : R, (∀ n : nat, bigop n f ≤ b) → s ≤ b) →
    0 < d → ∃ N : nat, s + - d ≤ bigop N f.
  Proof.
    intros f s d Hleast Hd.
    destruct (classic (∃ N : nat, s + - d ≤ bigop N f)) as [Hex | Hnone].
    - exact Hex.
    - exfalso.
      apply (lt_irrefl s).
      apply (le_lt_trans s (s + - d) s).
      + apply Hleast.
        intros n.
        destruct (le_total (bigop n f) (s + - d)) as [Hle | Hge].
        × exact Hle.
        × destruct (Hnone (ex_intro _ n Hge)).
      + exact (sub_pos_lt s d Hd).
  Qed.

Monotone convergence: a nonnegative family whose finite sections stay below one bound has a sum, and it is their least upper bound.
  Theorem sum_exists_nonneg : ∀ f : nat → R,
    (∀ i : nat, 0 ≤ f i) → sections_bounded f →
    ∃ L : R, SumTo f L.
  Proof.
    intros f Hnonneg Hbounded.
    destruct (sections_lub f Hbounded) as [s [Hub Hleast]].
    ∃ s.
    intros eps Heps.
    destruct (lub_section_index f s (half eps) Hleast (half_pos eps Heps))
      as [N HN].
    ∃ N.
    intros n Hn.
    apply (le_lt_trans _ (half eps) eps).
    - assert (Hgap : s + - (s + - half eps) == half eps).
      { calc (s + - (s + - half eps))
          == (s + (- s + - - half eps)) { neg_add }
          == (s + (- s + half eps)) { neg_involutive }
          == ((s + - s) + half eps) { associative }
          == ((0 : R) + half eps) { right_inverse }
          == (half eps) { left_identity }. }
      calc (‖ bigop n f + - s ‖)
        == (abs (bigop n f + - s)) { (norm_is_abs (bigop n f + - s)) }
        == (s + - (s + - half eps))
             { (abs_sub_le_of_between (s + - half eps) (bigop n f) s
                  (le_transitive _ _ _ HN (sections_monotone f Hnonneg N n Hn))
                  (Hub n)) }
        == (half eps) { Hgap }.
    - exact (half_lt eps Heps).
  Qed.


abs_summable f: the sections of fun i ⇒ abs (f i) stay below one bound.
  Definition abs_summable (f : nat → R) : Prop :=
    sections_bounded (fun i ⇒ abs (f i)).

  Lemma abs_sections_bounded : ∀ f : nat → R, abs_summable f →
    ∃ S : R, (0 ≤ S) ∧ (∀ n : nat, bigop n (fun i ⇒ abs (f i)) ≤ S).
  Proof.
    intros f [B HB].
    ∃ B.
    split.
    - exact (le_rewrite_l (bigop O (fun i ⇒ abs (f i))) 0 B
               (bigop_0 (fun i ⇒ abs (f i))) (HB O)).
    - exact HB.
  Qed.

  Lemma abs_sum_exists : ∀ f : nat → R, abs_summable f →
    ∃ T : R, SumTo (fun i ⇒ abs (f i)) T.
  Proof.
    intros f Hf.
    exact (sum_exists_nonneg (fun i ⇒ abs (f i))
             (fun i ⇒ abs_nonneg (f i)) Hf).
  Qed.

  Lemma shifted_nonneg : ∀ a : R, 0 ≤ abs a + a.
  Proof.
    intros a.
    calc (0 : R)
      == (abs a + - abs a) { (right_inverse (abs a)) }
      == (abs a + a) { (add_le_monotone_left (- abs a) a (abs a)
                               (neg_abs_le a)) }.
  Qed.

  Lemma shifted_sections_bounded : ∀ f : nat → R, abs_summable f →
    sections_bounded (fun i ⇒ abs (f i) + f i).
  Proof.
    intros f Hf.
    destruct (abs_sections_bounded f Hf) as [S [_ HS]].
    ∃ (S + S).
    intros n.
    calc (bigop n (fun i ⇒ abs (f i) + f i))
      == (bigop n (fun i ⇒ abs (f i) + abs (f i)))
           { (bigop_le n (fun i ⇒ abs (f i) + f i)
                (fun i ⇒ abs (f i) + abs (f i))
                (fun i ⇒ add_le_monotone_left (f i) (abs (f i)) (abs (f i))
                            (le_abs (f i)))) }
      == (bigop n (fun i ⇒ abs (f i)) + bigop n (fun i ⇒ abs (f i)))
           { (bigop_add n (fun i ⇒ abs (f i)) (fun i ⇒ abs (f i))) }
      == (S + S)
           { (add_le_mono (bigop n (fun i ⇒ abs (f i))) S
                (bigop n (fun i ⇒ abs (f i))) S (HS n) (HS n)) }.
  Qed.

  Theorem sum_exists_abs_summable : ∀ f : nat → R, abs_summable f →
    ∃ L : R, SumTo f L.
  Proof.
    intros f Hf.
    destruct (sum_exists_nonneg (fun i ⇒ abs (f i) + f i)
                (fun i ⇒ shifted_nonneg (f i))
                (shifted_sections_bounded f Hf)) as [P HP].
    destruct (abs_sum_exists f Hf) as [T HT].
    ∃ (P + - T).
    apply (sum_to_ext (fun i ⇒ (abs (f i) + f i) + - abs (f i)) f).
    - intros i.
      calc ((abs (f i) + f i) + - abs (f i))
        == ((f i + abs (f i)) + - abs (f i)) { (commutative (abs (f i))) }
        == (f i + (abs (f i) + - abs (f i))) { associative }
        == (f i + (0 : R)) { right_inverse }
        == (f i) { right_identity }.
    - exact (sum_to_add (fun i ⇒ abs (f i) + f i) (fun i ⇒ - abs (f i))
               P (- T) HP (sum_to_neg (fun i ⇒ abs (f i)) T HT)).
  Qed.


  Lemma abs_summable_even : ∀ f : nat → R, abs_summable f →
    abs_summable (fun k ⇒ f (2 × k)%nat).
  Proof.
    intros f [B HB].
    ∃ B.
    intros n.
    calc (bigop n (fun k ⇒ abs (f (2 × k)%nat)))
      == (bigop n (fun k ⇒ abs (f (2 × k)%nat))
            + bigop n (fun k ⇒ abs (f (S (2 × k)%nat))))
           { (le_add_nonneg_r (bigop n (fun k ⇒ abs (f (2 × k)%nat)))
                (bigop n (fun k ⇒ abs (f (S (2 × k)%nat))))
                (bigop_nonneg n (fun k ⇒ abs (f (S (2 × k)%nat)))
                   (fun k ⇒ abs_nonneg (f (S (2 × k)%nat))))) }
      == (bigop (2 × n)%nat (fun i ⇒ abs (f i)))
           { (bigop_parity n (fun i ⇒ abs (f i))) }
      == B { (HB (2 × n)%nat) }.
  Qed.

  Lemma abs_summable_odd : ∀ f : nat → R, abs_summable f →
    abs_summable (fun k ⇒ f (S (2 × k)%nat)).
  Proof.
    intros f [B HB].
    ∃ B.
    intros n.
    calc (bigop n (fun k ⇒ abs (f (S (2 × k)%nat))))
      == (bigop n (fun k ⇒ abs (f (2 × k)%nat))
            + bigop n (fun k ⇒ abs (f (S (2 × k)%nat))))
           { (le_add_nonneg_l (bigop n (fun k ⇒ abs (f (2 × k)%nat)))
                (bigop n (fun k ⇒ abs (f (S (2 × k)%nat))))
                (bigop_nonneg n (fun k ⇒ abs (f (2 × k)%nat))
                   (fun k ⇒ abs_nonneg (f (2 × k)%nat)))) }
      == (bigop (2 × n)%nat (fun i ⇒ abs (f i)))
           { (bigop_parity n (fun i ⇒ abs (f i))) }
      == B { (HB (2 × n)%nat) }.
  Qed.

  Theorem sum_to_parity : ∀ (f : nat → R) (L Le Lo : R),
    SumTo f L →
    SumTo (fun k ⇒ f (2 × k)%nat) Le →
    SumTo (fun k ⇒ f (S (2 × k)%nat)) Lo →
    L == Le + Lo.
  Proof.
    intros f L Le Lo Hf He Ho.
    apply (lim_unique (fun n ⇒ bigop (2 × n)%nat f) L (Le + Lo)).
    - exact (lim_subseq (fun n ⇒ bigop n f) L (fun n ⇒ (2 × n)%nat)
               le_double Hf).
    - apply (proj1 (seq_lim_proper
                      (fun n ⇒ bigop n (fun k ⇒ f (2 × k)%nat)
                                  + bigop n (fun k ⇒ f (S (2 × k)%nat)))
                      (fun n ⇒ bigop (2 × n)%nat f)
                      (fun n ⇒ symmetry (bigop_parity n f))
                      (Le + Lo) (Le + Lo) (reflexivity (Le + Lo)))).
      exact (lim_add (fun n ⇒ bigop n (fun k ⇒ f (2 × k)%nat))
               (fun n ⇒ bigop n (fun k ⇒ f (S (2 × k)%nat))) Le Lo He Ho).
  Qed.


  Lemma mul_shift : ∀ a x b : R, a × x == a × b + a × (x + - b).
  Proof. intros a x b. ring. Qed.

  Lemma abs_bigop_mul_bound : ∀ (f b : nat → R) (Sbound t : R) (m : nat),
    bigop m (fun i ⇒ abs (f i)) ≤ Sbound → 0 ≤ t →
    (∀ i : nat, (i < m)%nat → abs (b i) ≤ t) →
    abs (bigop m (fun i ⇒ f i × b i)) ≤ t × Sbound.
  Proof.
    intros f b Sbound t m HS Ht Hb.
    calc (abs (bigop m (fun i ⇒ f i × b i)))
      == (bigop m (fun i ⇒ abs (f i × b i)))
           { (abs_bigop_le m (fun i ⇒ f i × b i)) }
      == (bigop m (fun i ⇒ abs (f i) × abs (b i)))
           { (bigop_ext m (fun i ⇒ abs (f i × b i))
                (fun i ⇒ abs (f i) × abs (b i))
                (fun i ⇒ abs_mul (f i) (b i))) }
      == (bigop m (fun i ⇒ abs (f i) × t))
           { (bigop_le_lt m (fun i ⇒ abs (f i) × abs (b i))
                (fun i ⇒ abs (f i) × t)
                (fun i Hi ⇒ mul_le_mono_l (abs (b i)) t (abs (f i))
                               (abs_nonneg (f i)) (Hb i Hi))) }
      == (bigop m (fun i ⇒ abs (f i)) × t)
           { (bigop_mul_r m t (fun i ⇒ abs (f i))) }
      == (Sbound × t)
           { (mul_le_mono_r (bigop m (fun i ⇒ abs (f i))) Sbound t Ht HS) }
      == (t × Sbound)
           { (mul_commutative Sbound t) }.
  Qed.

  Lemma cauchy_section_row : ∀ (f g : nat → R) (N : nat),
    bigop N (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat))
      == bigop N (fun i ⇒ f i × bigop (N - i)%nat g).
  Proof.
    intros f g N.
    calc (bigop N (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat)))
      == (bigop N (fun i ⇒ bigop (N - i)%nat (fun j ⇒ f i × g j)))
           { (bigop_triangle N (fun i j ⇒ f i × g j)) }
      == (bigop N (fun i ⇒ f i × bigop (N - i)%nat g))
           { (bigop_ext N (fun i ⇒ bigop (N - i)%nat (fun j ⇒ f i × g j))
                (fun i ⇒ f i × bigop (N - i)%nat g)
                (fun i ⇒ bigop_mul_l (N - i)%nat (f i) g)) }.
  Qed.

  Lemma cauchy_section_split : ∀ (f g : nat → R) (B : R) (N : nat),
    bigop N (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat))
      == bigop N f × B
           + bigop N (fun i ⇒ f i × (bigop (N - i)%nat g + - B)).
  Proof.
    intros f g B N.
    calc (bigop N (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat)))
      == (bigop N (fun i ⇒ f i × bigop (N - i)%nat g))
           { (cauchy_section_row f g N) }
      == (bigop N (fun i ⇒ f i × B
                              + f i × (bigop (N - i)%nat g + - B)))
           { (bigop_ext N (fun i ⇒ f i × bigop (N - i)%nat g)
                (fun i ⇒ f i × B + f i × (bigop (N - i)%nat g + - B))
                (fun i ⇒ mul_shift (f i) (bigop (N - i)%nat g) B)) }
      == (bigop N (fun i ⇒ f i × B)
            + bigop N (fun i ⇒ f i × (bigop (N - i)%nat g + - B)))
           { (bigop_add N (fun i ⇒ f i × B)
                (fun i ⇒ f i × (bigop (N - i)%nat g + - B))) }
      == (bigop N f × B
            + bigop N (fun i ⇒ f i × (bigop (N - i)%nat g + - B)))
           { (bigop_mul_r N B f) }.
  Qed.

  Lemma sum_to_remainder_bounded : ∀ (g : nat → R) (B : R), SumTo g B →
    ∃ M : R, (0 ≤ M) ∧ (∀ m : nat, abs (bigop m g + - B) ≤ M).
  Proof.
    intros g B Hg.
    destruct (lim_bounded (fun n ⇒ bigop n g) B Hg) as [M0 [HM0 HM]].
    ∃ (M0 + abs B).
    split.
    - exact (le_transitive _ _ _ HM0
               (le_add_nonneg_r M0 (abs B) (abs_nonneg B))).
    - intros m.
      calc (abs (bigop m g + - B))
        == (abs (bigop m g) + abs (- B)) { abs_triangle }
        == (abs (bigop m g) + abs B) { (abs_neg B) }
        == (M0 + abs B)
             { (add_le_mono_r (abs (bigop m g)) M0 (abs B) (HM m)) }.
  Qed.

  Lemma abs_block_bound : ∀ (f : nat → R) (T t : R) (m k : nat),
    abs (bigop m (fun i ⇒ abs (f i)) + - T) ≤ half t →
    abs (bigop (m + k)%nat (fun i ⇒ abs (f i)) + - T) ≤ half t →
    bigop k (fun j ⇒ abs (f (m + j)%nat)) ≤ t.
  Proof.
    intros f T t m k Hm Hmk.
    assert (Hsplit : bigop k (fun j ⇒ abs (f (m + j)%nat))
                     == (bigop (m + k)%nat (fun i ⇒ abs (f i)) + - T)
                          + (T + - bigop m (fun i ⇒ abs (f i)))).

  Lemma cauchy_remainder_bound :
    ∀ (f g : nat → R) (B Sbound Mbound t1 t2 : R) (m k : nat),
    (∀ n : nat, bigop n (fun i ⇒ abs (f i)) ≤ Sbound) →
    (∀ j : nat, abs (bigop j g + - B) ≤ Mbound) →
    (∀ j : nat, (k ≤ j)%nat → abs (bigop j g + - B) ≤ t1) →
    0 ≤ t1 → 0 ≤ Mbound →
    bigop k (fun j ⇒ abs (f (m + j)%nat)) ≤ t2 →
    abs (bigop (m + k)%nat
           (fun i ⇒ f i × (bigop ((m + k) - i)%nat g + - B)))
      ≤ t1 × Sbound + Mbound × t2.
  Proof.
    intros f g B Sbound Mbound t1 t2 m k HS HM Ht1 Ht1nonneg HMnonneg Htail.
    calc (abs (bigop (m + k)%nat
                 (fun i ⇒ f i × (bigop ((m + k) - i)%nat g + - B))))
      == (abs (bigop m (fun i ⇒ f i × (bigop ((m + k) - i)%nat g + - B))
                 + bigop k (fun j ⇒ f (m + j)%nat
                              × (bigop ((m + k) - (m + j))%nat g + - B))))
           { (abs_proper _ _
                (bigop_split m k
                   (fun i ⇒ f i × (bigop ((m + k) - i)%nat g + - B)))) }
      == (abs (bigop m (fun i ⇒ f i × (bigop ((m + k) - i)%nat g + - B)))
            + abs (bigop k (fun j ⇒ f (m + j)%nat
                              × (bigop ((m + k) - (m + j))%nat g + - B))))
           { abs_triangle }
      == (t1 × Sbound + Mbound × t2)
           { (add_le_mono _ _ _ _
                (abs_bigop_mul_bound f
                   (fun i ⇒ bigop ((m + k) - i)%nat g + - B) Sbound t1 m
                   (HS m) Ht1nonneg
                   (fun i Hi ⇒ Ht1 ((m + k) - i)%nat
                                  (le_sub_of_lt_sub i (m + k)%nat k
                                     (lt_sub_of_lt i m k Hi))))
                (abs_bigop_mul_bound (fun j ⇒ f (m + j)%nat)
                   (fun j ⇒ bigop ((m + k) - (m + j))%nat g + - B) t2 Mbound k
                   Htail HMnonneg
                   (fun j _ ⇒ HM ((m + k) - (m + j))%nat))) }.
  Qed.

    (∀ k : nat, u (S k) ≤ r × u k) →

  Lemma ratio_sections_bounded : ∀ (u : nat → R) (r : R),
    (∀ k : nat, 0 ≤ u k) → 0 ≤ r → r < 1 →
    (∀ k : nat, u (S k) ≤ r × u k) →
    ∀ n : nat, bigop n u ≤ u O × inv (1 + - r).
  Proof.
    intros u r Hnonneg Hr Hr1 Hstep n.
    assert (Hpos : 0 < 1 + - r) by exact (sub_pos_of_lt r 1 Hr1).
    assert (Hgrow : bigop n u ≤ u O + r × bigop n u)
      by exact (le_transitive _ _ _
                  (sections_monotone u Hnonneg n (S n) (Nat.le_succ_diag_r n))
                  (ratio_sections_step u r n Hstep)).
    assert (Hexpand : (1 + - r) × bigop n u
                      == bigop n u + - (r × bigop n u)) by ring.
    assert (Hcancel : (u O + r × bigop n u) + - (r × bigop n u) == u O)
      by ring.
    apply (mul_le_cancel_l (bigop n u) (u O × inv (1 + - r)) (1 + - r) Hpos).
    calc ((1 + - r) × bigop n u)
      == (bigop n u + - (r × bigop n u))

                (- (r × bigop n u)) Hgrow) }
      == (u O)
           { Hcancel }
      == ((1 + - r) × (u O × inv (1 + - r)))
           { (field_mul_div_cancel (u O) (1 + - r)
                (pos_neq_0 (1 + - r) Hpos)) }.
  Qed.

                  (neg_abs_le (ssum f + - f O))).
    assert (Hswap : - B + f O == f O + - B) by ring.
    assert (Hcancel : (ssum f + - f O) + f O == ssum f) by ring.
    apply (lt_le_trans 0 (f O + - B) (ssum f)).
    - exact (sub_pos_of_lt B (f O) Hhead).
    - calc (f O + - B)
        == (- B + f O) { Hswap }
        == ((ssum f + - f O) + f O)
             { (add_le_mono_r (- B) (ssum f + - f O) (f O) Hlow) }
        == (ssum f) { Hcancel }.
  Qed.


  #[export] Instance reals_ssum_proper : Proper (equ ==> equ) (ssum (A := R)).
  Proof.
    intros f g Hfg.
    destruct (classic (∃ L : R, SumTo f L)) as [[L HL] | Hnone].
    - calc (ssum f)
        == L { (ssum_of_sum_to f L HL) }
        == (ssum g) { (ssum_of_sum_to g L (sum_to_ext f g L Hfg HL)) }.
    - calc (ssum f)
        == (0 : R)
             { (proj2 (reals_sum_spec f)
                  (fun M HM ⇒ Hnone (ex_intro _ M HM))) }
        == (ssum g)
             { (proj2 (reals_sum_spec g)
                  (fun M HM ⇒ Hnone (ex_intro _ M
                     (sum_to_ext g f M (fun i ⇒ symmetry (Hfg i)) HM)))) }.
  Qed.

  #[export] Instance reals_summable_proper :
    Proper (equ ==> iff) (summable (A := R)).
  Proof.
    intros f g Hfg.
    split.
    - exact (abs_summable_ext f g Hfg).
    - exact (abs_summable_ext g f (fun i ⇒ symmetry (Hfg i))).
  Qed.

  Theorem reals_sum_supported : ∀ (f : nat → R) (n : nat),
    (∀ m : nat, (n ≤ m)%nat → f m == 0) → ssum f == bigop n f.
  Proof.
    intros f n Hzero.
    exact (ssum_of_sum_to f (bigop n f) (sum_to_supported f n Hzero)).
  Qed.

  Theorem reals_sum_add : ∀ f g : nat → R,
    summable f → summable g →
    ssum (fun n ⇒ f n + g n) == ssum f + ssum g.
  Proof.
    intros f g Hf Hg.
    exact (ssum_of_sum_to (fun n ⇒ f n + g n) (ssum f + ssum g)
             (sum_to_add f g (ssum f) (ssum g) (ssum_sums f Hf)
                (ssum_sums g Hg))).
  Qed.

  Theorem reals_sum_neg : ∀ f : nat → R,
    summable f → ssum (fun n ⇒ - f n) == - ssum f.
  Proof.
    intros f Hf.
    exact (ssum_of_sum_to (fun n ⇒ - f n) (- ssum f)
             (sum_to_neg f (ssum f) (ssum_sums f Hf))).
  Qed.

  Theorem reals_sum_parity : ∀ f : nat → R,
    summable f →
    ssum f == ssum (fun k ⇒ f (2 × k)%nat)
                + ssum (fun k ⇒ f (S (2 × k)%nat)).
  Proof.
    intros f Hf.
    exact (sum_to_parity f (ssum f)
             (ssum (fun k ⇒ f (2 × k)%nat))
             (ssum (fun k ⇒ f (S (2 × k)%nat)))
             (ssum_sums f Hf)
             (ssum_sums (fun k ⇒ f (2 × k)%nat) (abs_summable_even f Hf))
             (ssum_sums (fun k ⇒ f (S (2 × k)%nat))
                (abs_summable_odd f Hf))).
  Qed.

  Theorem reals_sum_cauchy : ∀ f g : nat → R,
    summable f → summable g →
    ssum f × ssum g
      == ssum (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat)).
  Proof.
    intros f g Hf Hg.
    symmetry.
    exact (ssum_of_sum_to
             (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat))
             (ssum f × ssum g)
             (sum_to_cauchy f g (ssum f) (ssum g) Hf (ssum_sums f Hf)
                (ssum_sums g Hg))).
  Qed.
