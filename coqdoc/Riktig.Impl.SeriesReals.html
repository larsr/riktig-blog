Section SeriesReals.
  Context {R : Type} `{Reals R}.

  #[local] Instance reals_norm : HasNorm R R := abs.

  #[local] Instance reals_normed_space : NormedSpace R R := {
    NormedSpace.vnorm_proper := abs_proper;
    NormedSpace.nonneg := abs_nonneg;
    NormedSpace.definite := fun u ⇒ proj1 (abs_zero_iff u);
    NormedSpace.is_homogeneous := fun a u ⇒ abs_mul a u;
    NormedSpace.triangle := abs_triangle;
  }.

SumTo f L: the finite sections of f converge to L.
  Definition SumTo (f : nat → R) (L : R) : Prop :=
    seq_lim (fun n ⇒ bigop n f) L.

sections_bounded f: the finite sections of f stay below one bound — the hypothesis the existence theorem will need, stated here because the closure laws are about it.
  Definition sections_bounded (f : nat → R) : Prop :=
    ∃ B : R, ∀ n : nat, bigop n f ≤ B.

  Theorem sum_to_unique : ∀ (f : nat → R) (L M : R),
    SumTo f L → SumTo f M → L == M.
  Proof. intros f L M. exact (lim_unique (fun n ⇒ bigop n f) L M). Qed.

  Theorem sum_to_add : ∀ (f g : nat → R) (L M : R),
    SumTo f L → SumTo g M → SumTo (fun i ⇒ f i + g i) (L + M).
  Proof.
    intros f g L M Hf Hg.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ bigop n f + bigop n g)
                    (fun n ⇒ bigop n (fun i ⇒ f i + g i))
                    (fun n ⇒ symmetry (bigop_add n f g))
                    (L + M) (L + M) (reflexivity (L + M)))
             (lim_add (fun n ⇒ bigop n f) (fun n ⇒ bigop n g) L M Hf Hg)).
  Qed.

  Theorem sum_to_neg : ∀ (f : nat → R) (L : R),
    SumTo f L → SumTo (fun i ⇒ - f i) (- L).
  Proof.
    intros f L Hf.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ - bigop n f)
                    (fun n ⇒ bigop n (fun i ⇒ - f i))
                    (fun n ⇒ symmetry (bigop_neg n f))
                    (- L) (- L) (reflexivity (- L)))
             (lim_neg (fun n ⇒ bigop n f) L Hf)).
  Qed.

  Theorem sum_to_scale : ∀ (a : R) (f : nat → R) (L : R),
    SumTo f L → SumTo (fun i ⇒ a *: f i) (a *: L).
  Proof.
    intros a f L Hf.
    exact (proj1 (seq_lim_proper
                    (fun n ⇒ a *: bigop n f)
                    (fun n ⇒ bigop n (fun i ⇒ a *: f i))
                    (fun n ⇒ symmetry (bigop_scale n a f))
                    (a *: L) (a *: L) (reflexivity (a *: L)))
             (lim_scale a (fun n ⇒ bigop n f) L Hf)).
  Qed.

  Lemma sections_settle : ∀ (f : nat → R) (N : nat),
    (∀ m : nat, (N ≤ m)%nat → f m == 0) →
    ∀ n : nat, (N ≤ n)%nat → bigop n f == bigop N f.
  Proof.
    intros f N Hzero n; induction n as [| n IH]; intros Hn.
    - rewrite (proj1 (Nat.le_0_r N) Hn). reflexivity.
    - destruct (proj1 (Nat.lt_eq_cases N (S n)) Hn) as [Hlt | Heq].
      + assert (HNn : (N ≤ n)%nat) by exact (proj1 (Nat.lt_succ_r N n) Hlt).
        calc (bigop (S n) f)
          == (bigop n f + f n) { (bigop_snoc n f) }
          == (bigop N f + f n) { (IH HNn) }
          == (bigop N f + (0 : R)) { (Hzero n HNn) }
          == (bigop N f) { right_identity }.
      + rewrite Heq. reflexivity.
  Qed.

  Theorem sum_to_supported : ∀ (f : nat → R) (N : nat),
    (∀ m : nat, (N ≤ m)%nat → f m == 0) → SumTo f (bigop N f).
  Proof.
    intros f N Hzero eps Heps.
    ∃ N.
    intros n Hn.
    apply (le_lt_trans _ 0 eps).
    - apply eq_le.
      calc (‖ bigop n f + - bigop N f ‖)
        == (‖ bigop N f + - bigop N f ‖)
             { (sections_settle f N Hzero n Hn) }
        == (0 : R)
             { (norm_sub_self (bigop N f)) }.
    - exact Heps.
  Qed.

  Theorem sections_monotone : ∀ (f : nat → R),
    (∀ i : nat, 0 ≤ f i) →
    ∀ m n : nat, (m ≤ n)%nat → bigop m f ≤ bigop n f.
  Proof.
    intros f Hf m n; induction n as [| n IH]; intros Hmn.
    - rewrite (proj1 (Nat.le_0_r m) Hmn). reflexivity.
    - destruct (proj1 (Nat.lt_eq_cases m (S n)) Hmn) as [Hlt | Heq].
      + apply (le_transitive _ (bigop n f + f n) _).
        × apply (le_transitive _ (bigop n f) _).
          -- exact (IH (proj1 (Nat.lt_succ_r m n) Hlt)).
          -- apply le_add_nonneg_r. exact (Hf n).
        × apply eq_le. symmetry. exact (bigop_snoc n f).
      + rewrite Heq. reflexivity.
  Qed.
