The natural numbers of a category: the datatype of the signature 1 + self, one level being zero or the successor of a hole.
Module NaturalNumbers.
  Notation shape O := (shape_of (fun A : O ⇒ A) (natS (S := sig O))).

  Class NaturalNumbers (O : Type) `{Bicartesian O} (N : O)
      (build : HasBuild O (shape O) N)
      (fold : HasCata O (shape O) N) := {
    initial :: InitialAlgebra O (shape O) N build fold;
  }.

  Notation zero := (build (F := shape _) ∘ inj1).
  Notation succ := (build (F := shape _) ∘ inj2).
  Notation fold := (cata (F := shape _)).
  Notation map := (fmap (F := shape _)).
End NaturalNumbers.
