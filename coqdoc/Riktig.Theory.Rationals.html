Section RationalsTheory.
  Context {Q Z : Type} `{Rationals Q Z}.

  Theorem cast_zero : cast 0 == 0.
  Proof. exact (hom_zero cast). Qed.

  Theorem cast_neg : ∀ a : Z, cast (- a) == - cast a.
  Proof. intros a. exact (hom_neg cast a). Qed.

Every rational is an integer over a nonzero integer.
  Theorem fraction : ∀ q : Q, ∃ a b : Z,
    cast b =/= 0 ∧ q == cast a × inv (cast b).
  Proof.
    intros q.
    destruct (is_fraction q) as [a [b [Hb Hq]]].
    ∃ a, b. split; [ exact Hb | ].
    calc q
      == (q × 1) { right_identity }
      == (q × (cast b × inv (cast b))) { (mul_inv_cancel (cast b) Hb) }
      == ((q × cast b) × inv (cast b)) { associative }
      == (cast a × inv (cast b)) { Hq }.
  Qed.

End RationalsTheory.
