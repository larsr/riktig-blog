Library Riktig.Impl.VectZero

From Coq Require Export Morphisms.
From Riktig Require Export Theory.UnitVS Impl.VectCat.

Section VectZero.
  Context {F : Type} `{FF : Field F}.

  Definition VUnit : VectObj F :=
    vectObj F unit.

  Definition vect_emit (V : VectObj F) : VHom VUnit V := linear_map_const0.

  Lemma vect_emit_unique (V : VectObj F) (f : VHom VUnit V) :
    vhom_equ VUnit V f (vect_emit V).
  Proof.
    intros u.
    rewrite (linear_map_proper f u 0 I).
    exact (linear_map_zero f).
  Qed.

  #[export] Instance Vect_Initial : Initial (Vect_Category (F:=F)) := {|
    Initial.init := VUnit;
    Initial.emit := vect_emit;
    Initial.emit_uniqueness := vect_emit_unique
  |}.


  Section Bang.
    Context (V : VectObj F).

    #[export] Instance bang_proper : Proper (equ ==> equ) (fun _ : vo_car V ⇒ tt).
    Proof.
      intros u v Huv.
      exact I.
    Qed.

    Lemma bang_add : ∀ u v : vo_car V, (tt : unit) == tt + tt.
    Proof.
      intros u v.
      exact I.
    Qed.

    Lemma bang_scale : ∀ (a : F) (u : vo_car V), (tt : unit) == a *: tt.
    Proof.
      intros a u.
      exact I.
    Qed.

    Definition vect_bang : VHom V VUnit :=
      linear_map_from_laws
      (fun _ : vo_car V ⇒ tt)
      bang_proper
      bang_add
      bang_scale.

  End Bang.

  Lemma vect_bang_unique (V : VectObj F) (f : VHom V VUnit) :
    vhom_equ V VUnit f (vect_bang V).
  Proof.
    intros u.
    exact I.
  Qed.

  #[export] Instance Vect_Terminal : Terminal (Vect_Category (F:=F)) := {|
    Terminal.term := VUnit;
    Terminal.morphism := vect_bang;
    Terminal.morphism_uniqueness := vect_bang_unique
  |}.

  Definition zero_map (U V : VectObj F) : VHom U V :=
    vcomp U VUnit V (vect_emit V) (vect_bang U).

  Example zero_map_direct (U V : VectObj F) :
    vhom_equ U V (zero_map U V) linear_map_const0.
  Proof.
    intros u.
    reflexivity.
  Qed.

End VectZero.
