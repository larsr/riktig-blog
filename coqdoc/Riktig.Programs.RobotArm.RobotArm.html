Record Joint2 (F : Type) := mkJoint2 {
  joint2_c : F;
  joint2_s : F;
  joint2_length : F
}.

Arguments mkJoint2 {F} joint2_c joint2_s joint2_length.
Arguments joint2_c {F} j.
Arguments joint2_s {F} j.
Arguments joint2_length {F} j.

Definition Chain2 (F : Type) : Type := list (Joint2 F).

Section Chain2Ops.
  Context {F : Type} `{FF : Field F}.

  Definition beam_rest_direction2 : Cl F 2 := ga_vec (plane_vector 1 0).

  Definition chain_direction2 (R : Cl F 2) : Cl F 2 :=
    sandwich R beam_rest_direction2.

  Lemma chain_direction2_rotor : ∀ c s : F,
    chain_direction2 (rotor c s)
    == ga_vec (plane_vector ((c×c + - (s×s)) × 1 + (c×s + c×s) × 0)
                            (- ((c×s + c×s) × 1) + (c×c + - (s×s)) × 0)).
  Proof.
    intros c s. unfold chain_direction2, beam_rest_direction2.
    exact (sandwich_rotor_vec c s 1 0).
  Qed.

  Definition chain_rotor2 (joints : Chain2 F) : Cl F 2 :=
    fold_right (fun j acc ⇒ rotor (joint2_c j) (joint2_s j) × acc) 1 joints.

  Lemma chain_rotor2_nil : chain_rotor2 [] = (1 : Cl F 2).
  Proof. reflexivity. Qed.

  Lemma chain_rotor2_cons : ∀ (j : Joint2 F) (rest : Chain2 F),
    chain_rotor2 (j :: rest) = rotor (joint2_c j) (joint2_s j) × chain_rotor2 rest.
  Proof. reflexivity. Qed.

  Lemma chain_rotor2_is_rotor : ∀ joints : Chain2 F,
    ∃ c s : F, chain_rotor2 joints == rotor c s.
  Proof.
    induction joints as [| j rest [c [s IH]]].
    - ∃ 1, 0. rewrite chain_rotor2_nil. symmetry. exact rotor_one.
    - ∃ (joint2_c j × c + - (joint2_s j × s)), (joint2_c j × s + joint2_s j × c).
      calc (chain_rotor2 (j :: rest))
        == (rotor (joint2_c j) (joint2_s j) × chain_rotor2 rest : Cl F 2)
           { chain_rotor2_cons }
        == (rotor (joint2_c j) (joint2_s j) × rotor c s : Cl F 2)
           { IH }
        == (rotor (joint2_c j × c + - (joint2_s j × s))
                  (joint2_c j × s + joint2_s j × c) : Cl F 2)
           { rotor_mul }.
  Qed.

  Fixpoint chain_orientation_from2 (orientation : Cl F 2) (joints : Chain2 F) : Cl F 2 :=
    match joints with
    | [] ⇒ orientation
    | j :: rest ⇒
        chain_orientation_from2 (orientation × rotor (joint2_c j) (joint2_s j)) rest
    end.

  Lemma chain_orientation_from2_nil : ∀ orientation : Cl F 2,
    chain_orientation_from2 orientation [] = orientation.
  Proof. reflexivity. Qed.

  Lemma chain_orientation_from2_cons : ∀ (orientation : Cl F 2)
      (j : Joint2 F) (rest : Chain2 F),
    chain_orientation_from2 orientation (j :: rest)
    = chain_orientation_from2 (orientation × rotor (joint2_c j) (joint2_s j)) rest.
  Proof. reflexivity. Qed.

  Theorem chain_rotor_is_product2 : ∀ (joints : Chain2 F) (orientation : Cl F 2),
    chain_orientation_from2 orientation joints == orientation × chain_rotor2 joints.
  Proof.
    induction joints as [| j rest IH]; intros orientation.
    - calc (chain_orientation_from2 orientation [])
        == (orientation : Cl F 2)
           { chain_orientation_from2_nil }
        == (orientation × 1 : Cl F 2)
           { right_identity }
        == (orientation × chain_rotor2 [] : Cl F 2)
           { chain_rotor2_nil }.
    - calc (chain_orientation_from2 orientation (j :: rest))
        == (chain_orientation_from2
              (orientation × rotor (joint2_c j) (joint2_s j)) rest : Cl F 2)
           { chain_orientation_from2_cons }
        == ((orientation × rotor (joint2_c j) (joint2_s j)) × chain_rotor2 rest : Cl F 2)
           { IH }
        == (orientation × (rotor (joint2_c j) (joint2_s j) × chain_rotor2 rest) : Cl F 2)
           { associative }
        == (orientation × chain_rotor2 (j :: rest) : Cl F 2)
           { chain_rotor2_cons }.
  Qed.

  Corollary chain_rotor_is_product2_from_identity : ∀ joints : Chain2 F,
    chain_orientation_from2 1 joints == chain_rotor2 joints.
  Proof.
    intros joints. rewrite (chain_rotor_is_product2 joints 1).
    exact (left_identity (chain_rotor2 joints)).
  Qed.

  Fixpoint chain_positions2_from (orientation position : Cl F 2)
           (joints : Chain2 F) : list (Cl F 2) :=
    match joints with
    | [] ⇒ []
    | j :: rest ⇒
        let orientation' := orientation × rotor (joint2_c j) (joint2_s j) in
        let position' := position + joint2_length j *: chain_direction2 orientation' in
        position' :: chain_positions2_from orientation' position' rest
    end.

  Lemma chain_positions2_from_nil : ∀ orientation position : Cl F 2,
    chain_positions2_from orientation position [] = [].
  Proof. reflexivity. Qed.

  Lemma chain_positions2_from_cons : ∀ (orientation position : Cl F 2)
      (j : Joint2 F) (rest : Chain2 F),
    chain_positions2_from orientation position (j :: rest)
    = let orientation' := orientation × rotor (joint2_c j) (joint2_s j) in
      let position' := position + joint2_length j *: chain_direction2 orientation' in
      position' :: chain_positions2_from orientation' position' rest.
  Proof. reflexivity. Qed.

  Definition chain_positions2 (base : Cl F 2) (joints : Chain2 F) : list (Cl F 2) :=
    chain_positions2_from 1 base joints.

End Chain2Ops.

Record Joint3 (F : Type) := mkJoint3 {
  joint3_c : F;
  joint3_b12 : F;
  joint3_b13 : F;
  joint3_b23 : F;
  joint3_length : F
}.

Arguments mkJoint3 {F} joint3_c joint3_b12 joint3_b13 joint3_b23 joint3_length.
Arguments joint3_c {F} j.
Arguments joint3_b12 {F} j.
Arguments joint3_b13 {F} j.
Arguments joint3_b23 {F} j.
Arguments joint3_length {F} j.

Definition Chain3 (F : Type) : Type := list (Joint3 F).

Section Chain3Ops.
  Context {F : Type} `{FF : Field F}.

  Definition joint3_rotor (j : Joint3 F) : Cl F 3 :=
    spinor (joint3_c j) (joint3_b12 j) (joint3_b13 j) (joint3_b23 j).

  Definition beam_rest_direction3 : Fin.t 3 → F := space_vector 1 0 0.

  Definition chain_direction3 (R : Cl F 3) : Cl F 3 :=
    ga_vec (rotate_coordinates R beam_rest_direction3).

  Definition chain_rotor3 (joints : Chain3 F) : Cl F 3 :=
    fold_right (fun j acc ⇒ joint3_rotor j × acc) 1 joints.

  Lemma chain_rotor3_nil : chain_rotor3 [] = (1 : Cl F 3).
  Proof. reflexivity. Qed.

  Lemma chain_rotor3_cons : ∀ (j : Joint3 F) (rest : Chain3 F),
    chain_rotor3 (j :: rest) = joint3_rotor j × chain_rotor3 rest.
  Proof. reflexivity. Qed.

  Lemma unit_even : cl_even (1 : Cl F 3).
  Proof.
    intros D HD.
    rewrite (blade3_case D) in ×.
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D)));
      solve [ discriminate HD | reflexivity ].
  Qed.

  Lemma chain_rotor3_even : ∀ joints : Chain3 F, cl_even (chain_rotor3 joints).
  Proof.
    induction joints as [| j rest IH].
    - rewrite chain_rotor3_nil. exact unit_even.
    - rewrite chain_rotor3_cons.
      apply even_mul_closed.
      + exact (spinor_even (joint3_c j) (joint3_b12 j) (joint3_b13 j) (joint3_b23 j)).
      + exact IH.
  Qed.

  Fixpoint chain_orientation_from3 (orientation : Cl F 3) (joints : Chain3 F) : Cl F 3 :=
    match joints with
    | [] ⇒ orientation
    | j :: rest ⇒ chain_orientation_from3 (orientation × joint3_rotor j) rest
    end.

  Lemma chain_orientation_from3_nil : ∀ orientation : Cl F 3,
    chain_orientation_from3 orientation [] = orientation.
  Proof. reflexivity. Qed.

  Lemma chain_orientation_from3_cons : ∀ (orientation : Cl F 3)
      (j : Joint3 F) (rest : Chain3 F),
    chain_orientation_from3 orientation (j :: rest)
    = chain_orientation_from3 (orientation × joint3_rotor j) rest.
  Proof. reflexivity. Qed.

  Theorem chain_rotor_is_product3 : ∀ (joints : Chain3 F) (orientation : Cl F 3),
    chain_orientation_from3 orientation joints == orientation × chain_rotor3 joints.
  Proof.
    induction joints as [| j rest IH]; intros orientation.
    - calc (chain_orientation_from3 orientation [])
        == (orientation : Cl F 3)
           { chain_orientation_from3_nil }
        == (orientation × 1 : Cl F 3)
           { right_identity }
        == (orientation × chain_rotor3 [] : Cl F 3)
           { chain_rotor3_nil }.
    - calc (chain_orientation_from3 orientation (j :: rest))
        == (chain_orientation_from3 (orientation × joint3_rotor j) rest : Cl F 3)
           { chain_orientation_from3_cons }
        == ((orientation × joint3_rotor j) × chain_rotor3 rest : Cl F 3)
           { IH }
        == (orientation × (joint3_rotor j × chain_rotor3 rest) : Cl F 3)
           { associative }
        == (orientation × chain_rotor3 (j :: rest) : Cl F 3)
           { chain_rotor3_cons }.
  Qed.

  Corollary chain_rotor_is_product3_from_identity : ∀ joints : Chain3 F,
    chain_orientation_from3 1 joints == chain_rotor3 joints.
  Proof.
    intros joints. rewrite (chain_rotor_is_product3 joints 1).
    exact (left_identity (chain_rotor3 joints)).
  Qed.

  Fixpoint chain_positions3_from (orientation position : Cl F 3)
           (joints : Chain3 F) : list (Cl F 3) :=
    match joints with
    | [] ⇒ []
    | j :: rest ⇒
        let orientation' := orientation × joint3_rotor j in
        let position' := position + joint3_length j *: chain_direction3 orientation' in
        position' :: chain_positions3_from orientation' position' rest
    end.

  Lemma chain_positions3_from_nil : ∀ orientation position : Cl F 3,
    chain_positions3_from orientation position [] = [].
  Proof. reflexivity. Qed.

  Lemma chain_positions3_from_cons : ∀ (orientation position : Cl F 3)
      (j : Joint3 F) (rest : Chain3 F),
    chain_positions3_from orientation position (j :: rest)
    = let orientation' := orientation × joint3_rotor j in
      let position' := position + joint3_length j *: chain_direction3 orientation' in
      position' :: chain_positions3_from orientation' position' rest.
  Proof. reflexivity. Qed.

  Definition chain_positions3 (base : Cl F 3) (joints : Chain3 F) : list (Cl F 3) :=
    chain_positions3_from 1 base joints.

End Chain3Ops.


Definition planar_arm_example : Chain2 Q :=
  [ mkJoint2 (1#1)%Q (0#1)%Q (1#1)%Q;
    mkJoint2 (0#1)%Q (1#1)%Q (1#1)%Q ].

Example planar_arm_positions :
  Forall2 equ (chain_positions2 (0 : Cl Q 2) planar_arm_example)
    [ ga_vec (plane_vector (1#1)%Q (0#1)%Q); ga_vec (plane_vector (0#1)%Q (0#1)%Q) ].
Proof. repeat constructor; apply cl2_ext; vm_compute; reflexivity. Qed.

Example planar_arm_cumulative_rotor_is_product :
  chain_orientation_from2 (1 : Cl Q 2) planar_arm_example == chain_rotor2 planar_arm_example.
Proof. exact (chain_rotor_is_product2_from_identity planar_arm_example). Qed.

Definition spatial_arm_example : Chain3 Q :=
  [ mkJoint3 (1#1)%Q (0#1)%Q (0#1)%Q (0#1)%Q (1#1)%Q;
    mkJoint3 (0#1)%Q (1#1)%Q (0#1)%Q (0#1)%Q (1#1)%Q ].

Example spatial_arm_positions :

      ga_vec (space_vector (0#1)%Q (0#1)%Q (0#1)%Q) ].
Proof. repeat constructor; apply cl3_ext; vm_compute; reflexivity. Qed.

Example spatial_arm_cumulative_rotor_is_product :

Proof. exact (chain_rotor_is_product3_from_identity spatial_arm_example). Qed.
