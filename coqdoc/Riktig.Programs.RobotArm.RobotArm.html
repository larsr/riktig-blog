Definition Joint2 (F : Type) : Type := Joint F (Cl F 2).
Definition Chain2 (F : Type) : Type := list (Joint2 F).

Section PlanarArm.
  Context {F : Type} `{FF : Field F}.

  Definition mkJoint2 (c s beam : F) : Joint2 F := mkJoint (rotor c s) beam.

  Definition beam_rest_direction2 : Cl F 2 := ga_vec (plane_vector 1 0).

  Definition chain_direction2 : Cl F 2 → Cl F 2 :=
    beam_direction cl_reverse beam_rest_direction2.

  Definition chain_positions2 : Cl F 2 → Chain2 F → list (Cl F 2) :=
    chain_positions cl_reverse beam_rest_direction2.

A planar joint steers in the plane of the bivector rotor 0 1, whose square is - (1) — so `turn_mul` and `chain_rotor_in_plane` are about these joints.
  Lemma joint2_turns : ∀ c s beam : F,
    joint_rotor (mkJoint2 c s beam) == turn (rotor 0 1) c s.
  Proof. intros c s beam. exact (rotor_two_term c s). Qed.

A chain of even joints accumulates to a rotor: the general `chain_rotor_even` at the plane's parity grading, and in this algebra an even element is a rotor.
  Theorem chain_rotor2_is_rotor : ∀ joints : Chain2 F,
    Forall (fun j ⇒ cl_even (joint_rotor j)) joints →
    ∃ c s : F, chain_rotor joints == rotor c s.
  Proof.
    intros joints Hjoints.
    exact (even_is_rotor (chain_rotor joints)
             (chain_rotor_even cl_even cl_odd joints Hjoints)).
  Qed.

A joint steered by a point of the unit circle is a rotor — Impl/PlaneRotor.v's `rotor_is_rotor`, which is what `beam_keeps_its_length` asks of every joint.
  Lemma joint2_unit : ∀ c s beam : F,
    c×c + s×s == 1 →
    Rotor cl_reverse cl_even (joint_rotor (mkJoint2 c s beam)).
  Proof. intros c s beam Hcs. exact (rotor_is_rotor c s Hcs). Qed.

The beams of a planar arm of unit joints keep their length: the general `beam_keeps_its_length` at this algebra, whose scalar line is faithful.
  Theorem chain_beam_length2 : ∀ (joints : Chain2 F) (w : Fin.t 2 → F),
    Forall (fun j ⇒ Rotor cl_reverse cl_even (joint_rotor j)) joints →
    ga_vec w == chain_direction2 (chain_rotor joints) →
    qform sig2 w == qform sig2 (plane_vector 1 0).
  Proof.
    intros joints w Hjoints Hw.
    exact (beam_keeps_its_length cl_reverse cl_even cl_odd
             (plane_vector 1 0) joints w plane_one_faithful Hjoints Hw).
  Qed.

End PlanarArm.


Definition Joint3 (F : Type) : Type := Joint F (Cl F 3).
Definition Chain3 (F : Type) : Type := list (Joint3 F).

Section SpatialArm.
  Context {F : Type} `{FF : Field F}.

  Definition mkJoint3 (r b12 b13 b23 beam : F) : Joint3 F :=
    mkJoint (spinor r b12 b13 b23) beam.

  Definition beam_rest_direction3 : Cl F 3 := ga_vec (space_vector 1 0 0).

  Definition chain_direction3 : Cl F 3 → Cl F 3 :=
    beam_direction cl_reverse beam_rest_direction3.

  Definition chain_positions3 : Cl F 3 → Chain3 F → list (Cl F 3) :=
    chain_positions cl_reverse beam_rest_direction3.

Even joints accumulate to an even orientation, one dimension up: the same theorem at the space algebra's parity grading.
  Theorem chain_rotor3_even : ∀ joints : Chain3 F,
    Forall (fun j ⇒ cl_even (joint_rotor j)) joints →
    cl_even (chain_rotor joints).
  Proof. exact (chain_rotor_even cl_even cl_odd). Qed.

The same beam-length theorem, one dimension up: the citation is the same, only the algebra and the rest direction's coordinates change.
  Theorem chain_beam_length3 : ∀ (joints : Chain3 F) (w : Fin.t 3 → F),
    Forall (fun j ⇒ Rotor cl_reverse cl_even (joint_rotor j)) joints →
    ga_vec w == chain_direction3 (chain_rotor joints) →
    qform sig3 w == qform sig3 (space_vector 1 0 0).
  Proof.
    intros joints w Hjoints Hw.
    exact (beam_keeps_its_length cl_reverse cl_even cl_odd
             (space_vector 1 0 0) joints w space_one_faithful Hjoints Hw).
  Qed.

End SpatialArm.


Definition planar_arm_example : Chain2 Q :=
  [ mkJoint2 (1#1)%Q (0#1)%Q (1#1)%Q;
    mkJoint2 (0#1)%Q (1#1)%Q (1#1)%Q ].

Example planar_arm_positions :
  Forall2 equ (chain_positions2 (0 : Cl Q 2) planar_arm_example)
    [ ga_vec (plane_vector (1#1)%Q (0#1)%Q); ga_vec (plane_vector (0#1)%Q (0#1)%Q) ].
Proof.
  apply (arm_folds_back cl_reverse beam_rest_direction2
           (cl_reverse_proper 2) (cl_reverse_one 2)).
  - exact rotor_one.
  - apply cl2_ext; vm_compute; reflexivity.
  - reflexivity.
  - apply cl2_ext; vm_compute; reflexivity.
  - apply cl2_ext; vm_compute; reflexivity.
Qed.

Example planar_arm_cumulative_rotor_is_product :
  chain_orientation_from (1 : Cl Q 2) planar_arm_example
  == chain_rotor planar_arm_example.
Proof. exact (chain_rotor_is_product_from_identity planar_arm_example). Qed.

Definition spatial_arm_example : Chain3 Q :=

Example spatial_arm_steers_in_one_plane :
  ∃ c s : Q, chain_rotor spatial_arm_example == turn (spinor 0 1 0 0) c s.
Proof.
  apply (chain_rotor_in_plane (spinor 0 1 0 0)).
  - apply cl3_ext; vm_compute; reflexivity.
  - repeat constructor.
    + ∃ (1#1)%Q, (0#1)%Q. apply cl3_ext; vm_compute; reflexivity.
    + ∃ (0#1)%Q, (1#1)%Q. apply cl3_ext; vm_compute; reflexivity.
Qed.

Example spatial_arm_stays_even : cl_even (chain_rotor spatial_arm_example).
Proof.
  apply chain_rotor3_even.
  repeat constructor; apply spinor_even.
Qed.
