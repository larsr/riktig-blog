Reading lambda terms as the grammar writes them, and only then giving the names their meaning. The reader knows nothing of scope: \x y. body is two named binders, a name is a name, a number a numeral. The change to de Bruijn terms is a separate program in the parser's own monad: a name becomes the index of its innermost binder, or the definition it was given earlier, or a failure; a numeral becomes a Church numeral. A definition is read named and changed at once, so later ones see it.

The grammar, on named terms; natrec is the loop.

  Definition named_step (fuel : N) (rec : P G × P G) : P G :=
    (token fuel (csym backslash) >>= (fun _ ⇒
     ident fuel >>= (fun k0 ⇒
     p_foldl fuel (ident fuel) (fun b : G ⇒ nlam k0 b)
       (fun (w : G → G) k ⇒ fun b ⇒ w (nlam k b)) >>= (fun w ⇒
     token fuel (csym dot_c) >>= (fun _ ⇒
     fst rec >>= (fun b ⇒ ret (w b)))))))
    <|>
    (snd rec >>= (fun a0 ⇒ p_foldl fuel (snd rec) a0 (fun f x ⇒ app f x))).
  Definition natom_step (fuel : N) (rec : P G × P G) : P G :=
    (ident fuel >>= (fun x ⇒ ret (nvar x : G)))
    <|>
    ((number fuel >>= (fun n ⇒ ret (num n : G)))
     <|>
     (token fuel (csym lparen) >>= (fun _ ⇒
      fst rec >>= (fun t ⇒
      token fuel (csym rparen) >>= (fun _ ⇒ ret t))))).
  Definition named_p (fuel : N) : P G :=
    fst (natrec (P G × P G) (fail, fail) (fun _ rec ⇒ (named_step fuel rec, natom_step fuel rec)) fuel).

  Definition to_debruijn (d : D) : G → E → P T :=
    ntermrec (E → P T)
      (fun x env ⇒
         match debruijn x env with
         | Some n ⇒ ret (var n : T)
         | None ⇒ match find d x with Some t ⇒ ret t | None ⇒ fail end
         end)
      (fun n _ ⇒ ret (church n))
      (fun k body env ⇒ body (prepend k env) >>= (fun b ⇒ ret (lam b)))
      (fun f a env ⇒ f env >>= (fun f' ⇒ a env >>= (fun a' ⇒ ret (app f' a')))).

  Definition ndef_p (fuel : N) (d : D) : P (K × T) :=
    ident fuel >>= (fun x ⇒
    token fuel (csym eqsign) >>= (fun _ ⇒
    named_p fuel >>= (fun g ⇒
    to_debruijn d g (empty : E) >>= (fun t ⇒
    token fuel (csym semi) >>= (fun _ ⇒ ret (x, t)))))).
  Definition ndefs_loop (fuel : N) : D → P D :=
    natrec (D → P D)
      (fun d ⇒ ret d)
      (fun _ rec d ⇒
        (ndef_p fuel d >>= (fun kt ⇒ rec (insert (fst kt) (snd kt) d)))
        <|> (ret d))
      fuel.
  Definition ndefs_p (fuel : N) : P D :=
    spaces fuel >>= (fun _ ⇒ ndefs_loop fuel (empty : D)).
