The sine and cosine series at the constructive reals: the terms by recursion, each the previous times - x^2 / ((2k+a)(2k+b)); past the Archimedean index of x^2 the ratio is at most a half, so the partial sums are Cauchy by the ratio test.

  Local Notation "'psum' f n" := (natrec _ 0 (fun k acc ⇒ acc + f k) n) (at level 10, f at level 9, n at level 9).
  Local Notation "'is_cauchy' u" :=
    (∀ eps : Q, 0 < eps →
      {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps})

  Local Notation half := (inv ((1 : Q) + 1)).

The shared bound: a positive product of successors under the inverse eats the Archimedean bound on x^2.
  Lemma inv_prod_ratio_le_half : ∀ (x : R) (m a b : N),
    abs (x × x) ≼ cast (of_nat m × half) → m ≤ a →
    abs (- (cast (inv (of_nat (succ a) × of_nat (succ b))) × (x × x))) ≼ cast half.
  Proof.
    intros x m a b Hx Hma.
    assert (Hs : (0 : Q) < of_nat (succ a) × of_nat (succ b))
      by exact (Theory.Order.mul_pos _ _ (of_nat_succ_pos a) (of_nat_succ_pos b)).
    assert (Hq : (0 : Q) ≤ inv (of_nat (succ a) × of_nat (succ b)))
      by exact (Theory.Order.le_of_lt _ _ (Theory.Order.inv_pos _ Hs)).
    assert (Hmp : of_nat m ≤ of_nat (succ a) × of_nat (succ b)).
    { apply (transitivity (y := of_nat (succ a))).
      - exact (of_nat_le m (succ a) (Theory.NatOrder.le_trans m a (succ a) Hma (Theory.NatOrder.le_succ_diag a))).
      - apply (transitivity (Theory.Order.le_of_eq _ _ (symmetry (right_identity (action := mult) (of_nat (succ a)))))).
        exact (Theory.Order.mul_le_mono_l 1 (of_nat (succ b)) (of_nat (succ a)) (of_nat_nonneg (succ a)) (Theory.Archimedean.one_le_of_nat_succ b)). }
    apply (le_equ_l (abs (cast (inv (of_nat (succ a) × of_nat (succ b))) × (x × x)))).
    { symmetry. exact (Theory.ConstructiveReals.abs_neg _). }
    apply (le_equ_l (cast (inv (of_nat (succ a) × of_nat (succ b))) × abs (x × x))).
    { calc (cast (inv (of_nat (succ a) × of_nat (succ b))) × abs (x × x))
        == (cast (abs (inv (of_nat (succ a) × of_nat (succ b)))) × abs (x × x))
             { (OrderedField.abs_of_nonneg _ Hq) }
        == (abs (cast (inv (of_nat (succ a) × of_nat (succ b)))) × abs (x × x))
             { (abs_cast _) }
        == (abs (cast (inv (of_nat (succ a) × of_nat (succ b))) × (x × x)))
             { (ConstructiveReals.abs_mul _ _) }. }
    apply (Theory.ConstructiveReals.le_trans _ (cast (inv (of_nat (succ a) × of_nat (succ b))) × cast (of_nat m × half))).
    { exact (ConstructiveReals.mul_le_mono_l _ _ _ (cast_nonneg _ Hq) Hx). }
    apply (le_equ_l (cast (inv (of_nat (succ a) × of_nat (succ b)) × (of_nat m × half)))).
    { exact (additive (f := cast) (opA := mult) _ _) . }
    apply cast_le.
    apply (transitivity (Theory.Order.le_of_eq _ _ (symmetry (associative (op := mult) (action := mult) (inv (of_nat (succ a) × of_nat (succ b))) (of_nat m) half)))).
    apply (transitivity (y := 1 × half)).
    { apply (Theory.Order.mul_le_mono_r _ _ half (Theory.GeometricSeries.half_nonneg (F := Q))).
      exact (Theory.Order.inv_mul_le_one (of_nat m) _ Hs Hmp). }
    exact (Theory.Order.le_of_eq _ _ (left_identity (action := mult) half)).
  Qed.

  Lemma sin_term_proper : ∀ x : R, Proper (equ ==> equ) (fun n : N ⇒ sin_term x n).
  Proof.
    intros x. apply (natrec_proper R _ _ x _).
    intros k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
    apply neg_proper. apply op_proper; [ | reflexivity ].
    apply hom_proper. apply inv_proper. apply op_proper; apply of_nat_equ.
    - apply succ_proper. apply succ_proper. exact (op_proper k k' Hk k k' Hk).
    - apply succ_proper. apply succ_proper. apply succ_proper. exact (op_proper k k' Hk k k' Hk).
  Qed.

  Lemma cos_term_proper : ∀ x : R, Proper (equ ==> equ) (fun n : N ⇒ cos_term x n).
  Proof.
    intros x. apply (natrec_proper R _ _ 1 _).
    intros k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
    apply neg_proper. apply op_proper; [ | reflexivity ].
    apply hom_proper. apply inv_proper. apply op_proper; apply of_nat_equ.
    - apply succ_proper. exact (op_proper k k' Hk k k' Hk).
    - apply succ_proper. apply succ_proper. exact (op_proper k k' Hk k k' Hk).
  Qed.

  Theorem sin_summable : ∀ x : R, is_cauchy (fun n ⇒ psum (fun j ⇒ sin_term x j) n).
  Proof.
    intros x.
    destruct (exp_index (x × x)) as [m Hm].
    apply (ratio_summable (fun n ⇒ sin_term x n) (sin_term_proper x) m).
    intros k Hmk.
    apply (le_equ_l (abs (sin_term x k × sin_ratio x k))).
    { apply ConstructiveReals.abs_proper. exact (symmetry (natrec_succ R _ _ x _ k)). }
    apply (le_equ_l (abs (sin_term x k) × abs (sin_ratio x k))).
    { symmetry. exact (ConstructiveReals.abs_mul _ _). }
    apply (le_equ_r _ (abs (sin_term x k) × cast half)).
    { exact (commutative (op := mult) _ _) . }
    apply ConstructiveReals.mul_le_mono_l.
    - exact (Theory.ConstructiveReals.abs_nonneg _).
    - apply (inv_prod_ratio_le_half x m (succ (k + k)) (succ (succ (k + k))) Hm).
      exact (Theory.NatOrder.le_trans m k (succ (k + k))
        Hmk (Theory.NatOrder.le_trans k (k + k) (succ (k + k)) (Theory.NatOrder.le_add_r k k) (Theory.NatOrder.le_succ_diag (k + k)))).
  Defined.

  Theorem cos_summable : ∀ x : R, is_cauchy (fun n ⇒ psum (fun j ⇒ cos_term x j) n).
  Proof.
    intros x.
    destruct (exp_index (x × x)) as [m Hm].
    apply (ratio_summable (fun n ⇒ cos_term x n) (cos_term_proper x) m).
    intros k Hmk.
    apply (le_equ_l (abs (cos_term x k × cos_ratio x k))).
    { apply ConstructiveReals.abs_proper. exact (symmetry (natrec_succ R _ _ 1 _ k)). }
    apply (le_equ_l (abs (cos_term x k) × abs (cos_ratio x k))).
    { symmetry. exact (ConstructiveReals.abs_mul _ _). }
    apply (le_equ_r _ (abs (cos_term x k) × cast half)).
    { exact (commutative (op := mult) _ _) . }
    apply ConstructiveReals.mul_le_mono_l.
    - exact (Theory.ConstructiveReals.abs_nonneg _).
    - apply (inv_prod_ratio_le_half x m (k + k) (succ (k + k)) Hm).
      exact (Theory.NatOrder.le_trans m k (k + k) Hmk (Theory.NatOrder.le_add_r k k)).
  Defined.
