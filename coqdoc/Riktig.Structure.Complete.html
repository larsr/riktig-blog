From Riktig.Structure Require Export Limits.

A complete space: every Cauchy sequence has a limit, computed from its Cauchy evidence.
Module Complete.
  Class Complete (F N V : Type) `{Limits F N V} (limit : HasLim N V) := {
    lim_tendsto : ∀ (u : N → V) (Hu : cauchy u), tendsto u (lim u Hu);
  }.
  Hint Mode Complete - ! ! - - - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Complete.
