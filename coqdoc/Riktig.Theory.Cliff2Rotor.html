Section EvenIso.
  Context {F : Type} `{FF : Field.Field F}.

  Lemma c2_even_add_closed : ∀ x y : C2 F,
    c2_even x → c2_even y → c2_even (x + y).
  Proof.
    intros x y [Hx1 Hx2] [Hy1 Hy2].
    split; simpl.
    - rewrite Hx1, Hy1. exact (idl (0 : F)).
    - rewrite Hx2, Hy2. exact (idl (0 : F)).
  Qed.

  Lemma c2_even_neg_closed : ∀ x : C2 F, c2_even x → c2_even (- x).
  Proof.
    intros x [Hx1 Hx2].
    split; simpl.
    - rewrite Hx1. exact neg_zero.
    - rewrite Hx2. exact neg_zero.
  Qed.


  Definition c2_to_c (x : C2 F) : C F := mkC (c2_s x) (c2_12 x).
  Definition c_to_c2 (z : C F) : C2 F := mkC2 (re z) 0 0 (im z).

  #[global] Instance c2_to_c_proper : Proper (equ ==> equ) c2_to_c.
  Proof.
    intros x y [Hs [H1 [H2 H12]]].
    split; [exact Hs | exact H12].
  Qed.

  #[global] Instance c_to_c2_proper : Proper (equ ==> equ) c_to_c2.
  Proof.
    intros z w [Hr Hi].
    repeat split; simpl; [exact Hr | reflexivity | reflexivity | exact Hi].
  Qed.

  Lemma c_to_c2_to_c : ∀ x : C2 F, c2_even x → c_to_c2 (c2_to_c x) == x.
  Proof.
    intros x [Hx1 Hx2].
    repeat split; simpl.
    - reflexivity.
    - symmetry. exact Hx1.
    - symmetry. exact Hx2.
    - reflexivity.
  Qed.

  Lemma c2_to_c_to_c2 : ∀ z : C F, c2_to_c (c_to_c2 z) == z.
  Proof.
    intros z.
    split; reflexivity.
  Qed.

  Lemma c_to_c2_even : ∀ z : C F, c2_even (c_to_c2 z).
  Proof.
    intros z.
    split; reflexivity.
  Qed.


  Lemma c2_to_c_zero : c2_to_c (0 : C2 F) == 0.
  Proof. reflexivity. Qed.

  Lemma c2_to_c_one : c2_to_c (1 : C2 F) == 1.
  Proof. reflexivity. Qed.

  Lemma c2_to_c_add : ∀ x y : C2 F,
    c2_to_c (x + y) == c2_to_c x + c2_to_c y.
  Proof.
    intros x y.
    split; reflexivity.
  Qed.

  Lemma c2_to_c_neg : ∀ x : C2 F, c2_to_c (- x) == - c2_to_c x.
  Proof.
    intros x.
    split; reflexivity.
  Qed.

  Lemma c2_to_c_mul_even : ∀ x y : C2 F,
    c2_even x → c2_even y → c2_to_c (x × y) == c2_to_c x × c2_to_c y.
  Proof.
    intros x y [Hx1 Hx2] [Hy1 Hy2].
    split; simpl.
    - rewrite Hx1, Hx2, Hy1, Hy2.
      repeat rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
    - rewrite Hx1, Hx2.
      repeat rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
  Qed.

End EvenIso.


Section Rotor.
  Context {F : Type} `{OF : OrderedField.OrderedField F}.

  Definition c2_reverse (x : C2 F) : C2 F :=
    mkC2 (c2_s x) (c2_1 x) (c2_2 x) (- c2_12 x).

  #[global] Instance c2_reverse_proper : Proper (equ ==> equ) c2_reverse.
  Proof.
    intros x y [Hs [H1 [H2 H12]]].
    repeat split; simpl; [exact Hs | exact H1 | exact H2 |].
    exact (Group.neg_proper (c2_12 x) (c2_12 y) H12).
  Qed.

  Lemma cliff2_reorder4 : ∀ a b c d : F,
    a + (b + (c + d)) == b + (a + (d + c)).
  Proof.
    intros a b c d.
    rewrite (add_swap_l a b (c + d)).
    rewrite (comm c d).
    reflexivity.
  Qed.

  Lemma c2_reverse_anti : ∀ x y : C2 F,
    c2_reverse (x × y) == c2_reverse y × c2_reverse x.
  Proof.
    intros x y.
    unfold c2_reverse.
    repeat split; simpl.
    - calc (c2_s x × c2_s y + (c2_1 x × c2_1 y + (c2_2 x × c2_2 y + - (c2_12 x × c2_12 y))))
        == (c2_s y × c2_s x + (c2_1 x × c2_1 y + (c2_2 x × c2_2 y + - (c2_12 x × c2_12 y))))
           { (mul_comm (c2_s x) (c2_s y)) }
        == (c2_s y × c2_s x + (c2_1 y × c2_1 x + (c2_2 x × c2_2 y + - (c2_12 x × c2_12 y))))
           { (mul_comm (c2_1 x) (c2_1 y)) }
        == (c2_s y × c2_s x + (c2_1 y × c2_1 x + (c2_2 y × c2_2 x + - (c2_12 x × c2_12 y))))
           { (mul_comm (c2_2 x) (c2_2 y)) }
        == (c2_s y × c2_s x + (c2_1 y × c2_1 x + (c2_2 y × c2_2 x + - (c2_12 y × c2_12 x))))
           { (mul_comm (c2_12 x) (c2_12 y)) }
        == (c2_s y × c2_s x + (c2_1 y × c2_1 x + (c2_2 y × c2_2 x + - (- c2_12 y × - c2_12 x))))
           { (mul_neg_neg (c2_12 y) (c2_12 x)) }.
    - transitivity
        (c2_1 x × c2_s y + (c2_1 y × c2_s x + (- (c2_12 y × c2_2 x) + c2_2 y × c2_12 x))).
      + calc (c2_s x × c2_1 y + (c2_1 x × c2_s y + (c2_12 x × c2_2 y + - (c2_2 x × c2_12 y))))
          == (c2_1 x × c2_s y + (c2_s x × c2_1 y + (- (c2_2 x × c2_12 y) + c2_12 x × c2_2 y)))
             { (cliff2_reorder4 (c2_s x × c2_1 y) (c2_1 x × c2_s y)
                  (c2_12 x × c2_2 y) (- (c2_2 x × c2_12 y))) }
          == (c2_1 x × c2_s y + (c2_s x × c2_1 y + (- (c2_2 x × c2_12 y) + c2_2 y × c2_12 x)))
             { (mul_comm (c2_2 y) (c2_12 x)) }
          == (c2_1 x × c2_s y + (c2_s x × c2_1 y + (- (c2_12 y × c2_2 x) + c2_2 y × c2_12 x)))
             { (mul_comm (c2_12 y) (c2_2 x)) }
          == (c2_1 x × c2_s y + (c2_1 y × c2_s x + (- (c2_12 y × c2_2 x) + c2_2 y × c2_12 x)))
             { (mul_comm (c2_1 y) (c2_s x)) }.
      + calc (c2_1 x × c2_s y + (c2_1 y × c2_s x + (- (c2_12 y × c2_2 x) + c2_2 y × c2_12 x)))
          == (c2_s y × c2_1 x + (c2_1 y × c2_s x + (- (c2_12 y × c2_2 x) + c2_2 y × c2_12 x)))
             { (mul_comm (c2_1 x) (c2_s y)) }
          == (c2_s y × c2_1 x + (c2_1 y × c2_s x + (- (c2_12 y × c2_2 x) + - - (c2_2 y × c2_12 x))))
             { (neg_involutive (c2_2 y × c2_12 x)) }
          == (c2_s y × c2_1 x + (c2_1 y × c2_s x + (- (c2_12 y × c2_2 x) + - (c2_2 y × - c2_12 x))))
             { (mul_neg_r (c2_2 y) (c2_12 x)) }
          == (c2_s y × c2_1 x + (c2_1 y × c2_s x + (- c2_12 y × c2_2 x + - (c2_2 y × - c2_12 x))))
             { (mul_neg_l (c2_12 y) (c2_2 x)) }.
    - transitivity
        (c2_2 x × c2_s y + (c2_2 y × c2_s x + (- (c2_1 y × c2_12 x) + c2_12 y × c2_1 x))).
      + calc (c2_s x × c2_2 y + (c2_2 x × c2_s y + (c2_1 x × c2_12 y + - (c2_12 x × c2_1 y))))
          == (c2_2 x × c2_s y + (c2_s x × c2_2 y + (- (c2_12 x × c2_1 y) + c2_1 x × c2_12 y)))
             { (cliff2_reorder4 (c2_s x × c2_2 y) (c2_2 x × c2_s y)
                  (c2_1 x × c2_12 y) (- (c2_12 x × c2_1 y))) }
          == (c2_2 x × c2_s y + (c2_s x × c2_2 y + (- (c2_12 x × c2_1 y) + c2_12 y × c2_1 x)))
             { (mul_comm (c2_12 y) (c2_1 x)) }
          == (c2_2 x × c2_s y + (c2_s x × c2_2 y + (- (c2_1 y × c2_12 x) + c2_12 y × c2_1 x)))
             { (mul_comm (c2_1 y) (c2_12 x)) }
          == (c2_2 x × c2_s y + (c2_2 y × c2_s x + (- (c2_1 y × c2_12 x) + c2_12 y × c2_1 x)))
             { (mul_comm (c2_2 y) (c2_s x)) }.
      + calc (c2_2 x × c2_s y + (c2_2 y × c2_s x + (- (c2_1 y × c2_12 x) + c2_12 y × c2_1 x)))
          == (c2_s y × c2_2 x + (c2_2 y × c2_s x + (- (c2_1 y × c2_12 x) + c2_12 y × c2_1 x)))
             { (mul_comm (c2_s y) (c2_2 x)) }
          == (c2_s y × c2_2 x + (c2_2 y × c2_s x + (- (c2_1 y × c2_12 x) + - - (c2_12 y × c2_1 x))))
             { (neg_involutive (c2_12 y × c2_1 x)) }
          == (c2_s y × c2_2 x + (c2_2 y × c2_s x + (- (c2_1 y × c2_12 x) + - (- c2_12 y × c2_1 x))))
             { (mul_neg_l (c2_12 y) (c2_1 x)) }
          == (c2_s y × c2_2 x + (c2_2 y × c2_s x + (c2_1 y × - c2_12 x + - (- c2_12 y × c2_1 x))))
             { (mul_neg_r (c2_1 y) (c2_12 x)) }.
    - transitivity
        (- (c2_12 x × c2_s y) + (- (c2_s x × c2_12 y) + (c2_2 x × c2_1 y + - (c2_2 y × c2_1 x)))).
      + calc (- (c2_s x × c2_12 y + (c2_12 x × c2_s y + (c2_1 x × c2_2 y + - (c2_2 x × c2_1 y)))))
          == (- (c2_s x × c2_12 y) + - (c2_12 x × c2_s y + (c2_1 x × c2_2 y + - (c2_2 x × c2_1 y))))
             { (neg_add (c2_s x × c2_12 y)
                  (c2_12 x × c2_s y + (c2_1 x × c2_2 y + - (c2_2 x × c2_1 y)))) }
          == (- (c2_s x × c2_12 y) + (- (c2_12 x × c2_s y) + - (c2_1 x × c2_2 y + - (c2_2 x × c2_1 y))))
             { (neg_add (c2_12 x × c2_s y) (c2_1 x × c2_2 y + - (c2_2 x × c2_1 y))) }
          == (- (c2_s x × c2_12 y) + (- (c2_12 x × c2_s y) + (- (c2_1 x × c2_2 y) + - - (c2_2 x × c2_1 y))))
             { (neg_add (c2_1 x × c2_2 y) (- (c2_2 x × c2_1 y))) }
          == (- (c2_s x × c2_12 y) + (- (c2_12 x × c2_s y) + (- (c2_1 x × c2_2 y) + c2_2 x × c2_1 y)))
             { (neg_involutive (c2_2 x × c2_1 y)) }
          == (- (c2_12 x × c2_s y) + (- (c2_s x × c2_12 y) + (c2_2 x × c2_1 y + - (c2_1 x × c2_2 y))))
             { (cliff2_reorder4 (- (c2_s x × c2_12 y)) (- (c2_12 x × c2_s y))
                  (- (c2_1 x × c2_2 y)) (c2_2 x × c2_1 y)) }
          == (- (c2_12 x × c2_s y) + (- (c2_s x × c2_12 y) + (c2_2 x × c2_1 y + - (c2_2 y × c2_1 x))))
             { (mul_comm (c2_2 y) (c2_1 x)) }.
      + calc (- (c2_12 x × c2_s y) + (- (c2_s x × c2_12 y) + (c2_2 x × c2_1 y + - (c2_2 y × c2_1 x))))
          == (- (c2_12 x × c2_s y) + (- (c2_s x × c2_12 y) + (c2_1 y × c2_2 x + - (c2_2 y × c2_1 x))))
             { (mul_comm (c2_1 y) (c2_2 x)) }
          == (- (c2_12 x × c2_s y) + (- (c2_12 y × c2_s x) + (c2_1 y × c2_2 x + - (c2_2 y × c2_1 x))))
             { (mul_comm (c2_12 y) (c2_s x)) }
          == (- (c2_s y × c2_12 x) + (- (c2_12 y × c2_s x) + (c2_1 y × c2_2 x + - (c2_2 y × c2_1 x))))
             { (mul_comm (c2_s y) (c2_12 x)) }
          == (- (c2_s y × c2_12 x) + (- c2_12 y × c2_s x + (c2_1 y × c2_2 x + - (c2_2 y × c2_1 x))))
             { (mul_neg_l (c2_12 y) (c2_s x)) }
          == (c2_s y × - c2_12 x + (- c2_12 y × c2_s x + (c2_1 y × c2_2 x + - (c2_2 y × c2_1 x))))
             { (mul_neg_r (c2_s y) (c2_12 x)) }.
  Qed.

  Definition rotor (c s : F) : C2 F := mkC2 c 0 0 s.

  Lemma rotor_even : ∀ c s : F, c2_even (rotor c s).
  Proof.
    intros c s.
    split; reflexivity.
  Qed.

  Lemma rotor_reverse : ∀ c s : F,
    c2_reverse (rotor c s) == mkC2 c 0 0 (- s).
  Proof.
    intros c s.
    unfold c2_reverse, rotor.
    repeat split; reflexivity.
  Qed.

  Lemma rotor_reverse_mul : ∀ c s : F,
    rotor c s × c2_reverse (rotor c s) == rotor (c×c + s×s) 0.
  Proof.
    intros c s.
    unfold rotor, c2_reverse.
    repeat split; simpl.
    - rewrite mul_0_l, mul_neg_r.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      autorewrite with grp.
      rewrite mul_neg_r, (mul_comm s c).
      autorewrite with grp.
      reflexivity.
  Qed.

  Lemma rotor_unit : ∀ c s : F,
    c×c + s×s == 1 → rotor c s × c2_reverse (rotor c s) == 1.
  Proof.
    intros c s Hcs.
    rewrite (rotor_reverse_mul c s).
    unfold rotor.
    repeat split; simpl; [exact Hcs | reflexivity | reflexivity | reflexivity].
  Qed.

  Lemma rotor_unit_necessary : ∀ c s : F,
    rotor c s × c2_reverse (rotor c s) == 1 → c×c + s×s == 1.
  Proof.
    intros c s Hru.
    rewrite (rotor_reverse_mul c s) in Hru.
    unfold rotor in Hru.
    destruct Hru as [Hs _].
    exact Hs.
  Qed.

  Lemma rotor_unit_iff : ∀ c s : F,
    rotor c s × c2_reverse (rotor c s) == 1 ↔ c×c + s×s == 1.
  Proof.
    intros c s.
    split; [exact (rotor_unit_necessary c s) | exact (rotor_unit c s)].
  Qed.


  Definition row2 {A : Type} (a b : A) : Fin.t 2 → A :=
    fun j ⇒ match j with
             | Fin.F1 ⇒ a
             | Fin.FS _ ⇒ b
             end.

  Definition c2_sandwich (R x : C2 F) : C2 F := R × x × c2_reverse R.

  Lemma rotor_mul_vec : ∀ c s v1 v2 : F,
    rotor c s × c2_vec (row2 v1 v2)
    == c2_vec (row2 (c×v1 + s×v2) (c×v2 + - (s×v1))).
  Proof.
    intros c s v1 v2.
    unfold rotor, c2_vec, row2.
    repeat split; simpl.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
  Qed.

  Lemma vec_mul_rotor : ∀ w1 w2 c s : F,
    c2_vec (row2 w1 w2) × rotor c s
    == c2_vec (row2 (w1×c + - (w2×s)) (w2×c + w1×s)).
  Proof.
    intros w1 w2 c s.
    unfold rotor, c2_vec, row2.
    repeat split; simpl.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
  Qed.

  Lemma rotor_reverse_eq_rotor : ∀ c s : F,
    c2_reverse (rotor c s) == rotor c (- s).
  Proof.
    intros c s.
    reflexivity.
  Qed.

  Lemma cliff2_sandwich_coeff1 : ∀ c s v1 v2 : F,
    (c×v1 + s×v2)*c + - ((c×v2 + - (s×v1)) × (- s))
    == (c×c + - (s×s)) × v1 + (c×s + c×s) × v2.
  Proof.
    intros c s v1 v2.
    transitivity (c × v1 × c + s × v2 × c + (c × v2 × s + - (s × v1 × s))).
    - calc ((c×v1 + s×v2)*c + - ((c×v2 + - (s×v1)) × (- s)))
        == ((c×v1 + s×v2)*c + - (c×v2 × (- s) + - (s×v1) × (- s)))
           { (distr (c×v2) (- (s×v1)) (- s)) }
        == ((c×v1 + s×v2)*c + - (- (c×v2×s) + - (s×v1) × (- s)))
           { (mul_neg_r (c×v2) s) }
        == ((c×v1 + s×v2)*c + - (- (c×v2×s) + s×v1×s))
           { (mul_neg_neg (s×v1) s) }
        == ((c×v1 + s×v2)*c + (- - (c×v2×s) + - (s×v1×s)))
           { (neg_add (- (c×v2×s)) (s×v1×s)) }
        == ((c×v1 + s×v2)*c + (c×v2×s + - (s×v1×s)))
           { (neg_involutive (c×v2×s)) }
        == (c×v1×c + s×v2×c + (c×v2×s + - (s×v1×s)))
           { (distr (c×v1) (s×v2) c) }.
    - transitivity (c×c×v1 + - (s×s×v1) + (c×s×v2 + c×s×v2)).
      + calc (c×v1×c + s×v2×c + (c×v2×s + - (s×v1×s)))
          == (c×c×v1 + s×v2×c + (c×v2×s + - (s×v1×s)))
             { (mul_assoc_comm_r c v1 c) }
          == (c×c×v1 + s×c×v2 + (c×v2×s + - (s×v1×s)))
             { (mul_assoc_comm_r s v2 c) }
          == (c×c×v1 + c×s×v2 + (c×v2×s + - (s×v1×s)))
             { (mul_comm s c) }
          == (c×c×v1 + c×s×v2 + (c×s×v2 + - (s×v1×s)))
             { (mul_assoc_comm_r c v2 s) }
          == (c×c×v1 + c×s×v2 + (c×s×v2 + - (s×s×v1)))
             { (mul_assoc_comm_r s v1 s) }
          == (c×c×v1 + - (s×s×v1) + (c×s×v2 + c×s×v2))
             { (add_rotate (c×c×v1) (c×s×v2) (c×s×v2) (- (s×s×v1))) }.
      + calc (c×c×v1 + - (s×s×v1) + (c×s×v2 + c×s×v2))
          == (c×c×v1 + - (s×s) × v1 + (c×s×v2 + c×s×v2))
             { (mul_neg_l (s×s) v1) }
          == (c×c×v1 + - (s×s) × v1 + (c×s + c×s) × v2)
             { (distr (c×s) (c×s) v2) }
          == ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
             { (distr (c×c) (- (s×s)) v1) }.
  Qed.

  Lemma cliff2_sandwich_coeff2 : ∀ c s v1 v2 : F,

  Lemma c2_sandwich_rotor_preserves_dot : ∀ c s v1 v2 : F,
    c×c + s×s == 1 →
    dot (row2 ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)

  Lemma rotor_mul : ∀ c1 s1 c2 s2 : F,
    rotor c1 s1 × rotor c2 s2 == rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2).
  Proof.
    intros c1 s1 c2 s2.
    unfold rotor.
    repeat split; simpl.
    - rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
    - repeat rewrite mul_0_l.
      repeat rewrite mul_0_r.
      autorewrite with grp.
      reflexivity.
    - rewrite mul_0_l.
      autorewrite with grp.
      reflexivity.
  Qed.

  Lemma rotor_mul_to_c : ∀ c1 s1 c2 s2 : F,
    c2_to_c (rotor c1 s1 × rotor c2 s2)
    == c2_to_c (rotor c1 s1) × c2_to_c (rotor c2 s2).
  Proof.
    intros c1 s1 c2 s2.
    exact (c2_to_c_mul_even (rotor c1 s1) (rotor c2 s2)
             (rotor_even c1 s1) (rotor_even c2 s2)).
  Qed.

  Theorem rotor_composition_is_u1 : ∀ c1 s1 c2 s2 : F,
    c2_to_c (rotor c1 s1) × c2_to_c (rotor c2 s2)
    == c2_to_c (rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)).
  Proof.
    intros c1 s1 c2 s2.
    rewrite <- (rotor_mul_to_c c1 s1 c2 s2).
    rewrite (rotor_mul c1 s1 c2 s2).
    reflexivity.
  Qed.
