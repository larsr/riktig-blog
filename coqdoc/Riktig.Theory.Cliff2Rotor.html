Section EvenIso.
  Context {F : Type} `{FF : Field F}.
  Add Ring even_iso_F_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma even_closed_under_add : ∀ x y : Cl2 F,
    Cl2.is_even x → Cl2.is_even y → Cl2.is_even (x + y).
  Proof.
    intros x y [Hx1 Hx2] [Hy1 Hy2].
    split; simpl.
    - rewrite Hx1, Hy1. exact (left_identity (0 : F)).
    - rewrite Hx2, Hy2. exact (left_identity (0 : F)).
  Qed.

  Lemma even_closed_under_neg : ∀ x : Cl2 F, Cl2.is_even x → Cl2.is_even (- x).
  Proof.
    intros x [Hx1 Hx2].
    split; simpl.
    - rewrite Hx1. exact neg_zero.
    - rewrite Hx2. exact neg_zero.
  Qed.

  Definition even_to_complex (x : Cl2 F) : C F := mkC (scalar_part x) (e12_part x).
  Definition complex_to_even (z : C F) : Cl2 F := mkCl2 (re z) 0 0 (im z).

  #[global] Instance even_to_complex_proper : Proper (equ ==> equ) even_to_complex.
  Proof.
    intros x y [Hs [H1 [H2 H12]]].
    split; [exact Hs | exact H12].
  Qed.

  #[global] Instance complex_to_even_proper : Proper (equ ==> equ) complex_to_even.
  Proof.
    intros z w [Hr Hi].
    repeat split; simpl; [exact Hr | reflexivity | reflexivity | exact Hi].
  Qed.

  Lemma even_round_trip : ∀ x : Cl2 F,
    Cl2.is_even x → complex_to_even (even_to_complex x) == x.
  Proof.
    intros x [Hx1 Hx2].
    repeat split; simpl.
    - reflexivity.
    - symmetry. exact Hx1.
    - symmetry. exact Hx2.
    - reflexivity.
  Qed.

  Lemma complex_round_trip : ∀ z : C F, even_to_complex (complex_to_even z) == z.
  Proof.
    intros z.
    split; reflexivity.
  Qed.

  Lemma complex_to_even_is_even : ∀ z : C F, Cl2.is_even (complex_to_even z).
  Proof.
    intros z.
    split; reflexivity.
  Qed.

  Lemma even_to_complex_zero : even_to_complex (0 : Cl2 F) == 0.
  Proof. reflexivity. Qed.

  Lemma even_to_complex_one : even_to_complex (1 : Cl2 F) == 1.
  Proof. reflexivity. Qed.

  Lemma even_to_complex_add : ∀ x y : Cl2 F,
    even_to_complex (x + y) == even_to_complex x + even_to_complex y.
  Proof.
    intros x y.
    split; reflexivity.
  Qed.

  Lemma even_to_complex_neg : ∀ x : Cl2 F,
    even_to_complex (- x) == - even_to_complex x.
  Proof.
    intros x.
    split; reflexivity.
  Qed.

  Lemma even_to_complex_mul : ∀ x y : Cl2 F,
    Cl2.is_even x → Cl2.is_even y →
    even_to_complex (x × y) == even_to_complex x × even_to_complex y.
  Proof.
    intros x y [Hx1 Hx2] [Hy1 Hy2].
    split; simpl; rewrite Hx1, Hx2, Hy1, Hy2; ring.
  Qed.

End EvenIso.

Module Reversal.
  Section Reversal.
    Context {F : Type} `{FF : Field F}.
    Add Ring reversal_F_ring : (ring_theory_of_Ring (R := F) commutative).


    Lemma scalar_component :
      ∀ x_scalar x_e1 x_e2 x_e12 y_scalar y_e1 y_e2 y_e12 : F,
      x_scalar × y_scalar + (x_e1 × y_e1 + (x_e2 × y_e2 + - (x_e12 × y_e12)))
      == y_scalar × x_scalar
         + (y_e1 × x_e1 + (y_e2 × x_e2 + - (- y_e12 × - x_e12))).
    Proof. intros; ring. Qed.

    Lemma e1_component :
      ∀ x_scalar x_e1 x_e2 x_e12 y_scalar y_e1 y_e2 y_e12 : F,
      x_scalar × y_e1 + (x_e1 × y_scalar + (x_e12 × y_e2 + - (x_e2 × y_e12)))
      == y_scalar × x_e1
         + (y_e1 × x_scalar + (- y_e12 × x_e2 + - (y_e2 × - x_e12))).
    Proof. intros; ring. Qed.

    Lemma e2_component :
      ∀ x_scalar x_e1 x_e2 x_e12 y_scalar y_e1 y_e2 y_e12 : F,
      x_scalar × y_e2 + (x_e2 × y_scalar + (x_e1 × y_e12 + - (x_e12 × y_e1)))
      == y_scalar × x_e2
         + (y_e2 × x_scalar + (y_e1 × - x_e12 + - (- y_e12 × x_e1))).
    Proof. intros; ring. Qed.

    Lemma e12_component :
      ∀ x_scalar x_e1 x_e2 x_e12 y_scalar y_e1 y_e2 y_e12 : F,
      - (x_scalar × y_e12 + (x_e12 × y_scalar + (x_e1 × y_e2 + - (x_e2 × y_e1))))
      == y_scalar × - x_e12
         + (- y_e12 × x_scalar + (y_e1 × x_e2 + - (y_e2 × x_e1))).
    Proof. intros; ring. Qed.

  End Reversal.
End Reversal.

Section Rotor.
  Context {F : Type} `{OF : OrderedField F}.
  Add Ring rotor_F_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition cl2_reverse (x : Cl2 F) : Cl2 F :=
    mkCl2 (scalar_part x) (e1_part x) (e2_part x) (- e12_part x).

  #[global] Instance cl2_reverse_proper : Proper (equ ==> equ) cl2_reverse.
  Proof.
    intros x y [Hs [H1 [H2 H12]]].
    repeat split; simpl; [exact Hs | exact H1 | exact H2 |].
    exact (Group.neg_proper (e12_part x) (e12_part y) H12).
  Qed.

  Lemma cl2_reverse_anti : ∀ x y : Cl2 F,
    cl2_reverse (x × y) == cl2_reverse y × cl2_reverse x.
  Proof.
    intros x y.
    unfold cl2_reverse.
    repeat split; simpl.
    - exact (Reversal.scalar_component
               (scalar_part x) (e1_part x) (e2_part x) (e12_part x)
               (scalar_part y) (e1_part y) (e2_part y) (e12_part y)).
    - exact (Reversal.e1_component
               (scalar_part x) (e1_part x) (e2_part x) (e12_part x)
               (scalar_part y) (e1_part y) (e2_part y) (e12_part y)).
    - exact (Reversal.e2_component
               (scalar_part x) (e1_part x) (e2_part x) (e12_part x)
               (scalar_part y) (e1_part y) (e2_part y) (e12_part y)).
    - exact (Reversal.e12_component
               (scalar_part x) (e1_part x) (e2_part x) (e12_part x)
               (scalar_part y) (e1_part y) (e2_part y) (e12_part y)).
  Qed.

  Definition rotor (c s : F) : Cl2 F := mkCl2 c 0 0 s.

  Lemma rotor_even : ∀ c s : F, Cl2.is_even (rotor c s).
  Proof.
    intros c s.
    split; reflexivity.
  Qed.

  Lemma rotor_reverse : ∀ c s : F,
    cl2_reverse (rotor c s) == mkCl2 c 0 0 (- s).
  Proof.
    intros c s.
    unfold cl2_reverse, rotor.
    repeat split; reflexivity.
  Qed.

  Lemma rotor_reverse_mul : ∀ c s : F,
    rotor c s × cl2_reverse (rotor c s) == rotor (c×c + s×s) 0.
  Proof.
    intros c s.
    unfold rotor, cl2_reverse.
    repeat split; simpl; ring.
  Qed.

  Lemma rotor_unit : ∀ c s : F,
    c×c + s×s == 1 → rotor c s × cl2_reverse (rotor c s) == 1.
  Proof.
    intros c s Hcs.
    rewrite (rotor_reverse_mul c s).
    unfold rotor.
    repeat split; simpl; [exact Hcs | reflexivity | reflexivity | reflexivity].
  Qed.

  Lemma rotor_unit_necessary : ∀ c s : F,
    rotor c s × cl2_reverse (rotor c s) == 1 → c×c + s×s == 1.
  Proof.
    intros c s Hru.
    rewrite (rotor_reverse_mul c s) in Hru.
    unfold rotor in Hru.
    destruct Hru as [Hs _].
    exact Hs.
  Qed.

  Lemma rotor_unit_iff : ∀ c s : F,
    rotor c s × cl2_reverse (rotor c s) == 1 ↔ c×c + s×s == 1.
  Proof.
    intros c s.
    split; [exact (rotor_unit_necessary c s) | exact (rotor_unit c s)].
  Qed.

  Definition plane_vector {A : Type} (a b : A) : Fin.t 2 → A :=
    fun j ⇒ match j with
             | Fin.F1 ⇒ a
             | Fin.FS _ ⇒ b
             end.

  Definition cl2_sandwich (R x : Cl2 F) : Cl2 F := R × x × cl2_reverse R.

  Lemma rotor_mul_vec : ∀ c s v1 v2 : F,
    rotor c s × Cl2.vec (plane_vector v1 v2)
    == Cl2.vec (plane_vector (c×v1 + s×v2) (c×v2 + - (s×v1))).
  Proof.
    intros c s v1 v2.
    unfold rotor, Cl2.vec, plane_vector.
    repeat split; simpl; ring.
  Qed.

  Lemma vec_mul_rotor : ∀ w1 w2 c s : F,
    Cl2.vec (plane_vector w1 w2) × rotor c s
    == Cl2.vec (plane_vector (w1×c + - (w2×s)) (w2×c + w1×s)).
  Proof.
    intros w1 w2 c s.
    unfold rotor, Cl2.vec, plane_vector.
    repeat split; simpl; ring.
  Qed.

  Lemma rotor_reverse_eq_rotor : ∀ c s : F,
    cl2_reverse (rotor c s) == rotor c (- s).
  Proof.
    intros c s.
    reflexivity.
  Qed.

  Lemma sandwich_first_coordinate : ∀ c s v1 v2 : F,
    (c×v1 + s×v2)*c + - ((c×v2 + - (s×v1)) × (- s))
    == (c×c + - (s×s)) × v1 + (c×s + c×s) × v2.
  Proof. intros; ring. Qed.

  Lemma sandwich_second_coordinate : ∀ c s v1 v2 : F,
    (c×v2 + - (s×v1))*c + (c×v1 + s×v2) × (- s)
    == - ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2.
  Proof. intros; ring. Qed.

  Lemma cl2_sandwich_rotor_vec : ∀ c s v1 v2 : F,
    cl2_sandwich (rotor c s) (Cl2.vec (plane_vector v1 v2))
    == Cl2.vec (plane_vector
         ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
         (- ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2)).
  Proof.
    intros c s v1 v2.
    unfold cl2_sandwich.
    rewrite (rotor_mul_vec c s v1 v2), (rotor_reverse_eq_rotor c s),
            (vec_mul_rotor (c×v1+s×v2) (c×v2 + - (s×v1)) c (- s)).
    repeat split; simpl.
    - reflexivity.
    - exact (sandwich_first_coordinate c s v1 v2).
    - exact (sandwich_second_coordinate c s v1 v2).
    - reflexivity.
  Qed.

  Lemma dot_plane_vector : ∀ a b : F,
    dot (plane_vector a b) (plane_vector a b) == a×a + b×b.
  Proof.
    intros a b.
    exact (dot_in_coordinates (plane_vector a b)).
  Qed.

  Lemma squared_length_scales : ∀ A B v1 v2 : F,
    (A×v1 + B×v2)*(A×v1 + B×v2) + (- (B×v1) + A×v2)*(- (B×v1) + A×v2)
    == (A×A + B×B) × (v1×v1 + v2×v2).
  Proof. intros; ring. Qed.

  Lemma double_angle_pythagoras : ∀ c s : F,
    (c×c + - (s×s)) × (c×c + - (s×s)) + (c×s + c×s) × (c×s + c×s)
    == (c×c + s×s) × (c×c + s×s).
  Proof. intros; ring. Qed.

  Lemma cl2_sandwich_rotor_preserves_dot : ∀ c s v1 v2 : F,
    c×c + s×s == 1 →
    dot (plane_vector ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)
              (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2))
        (plane_vector ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)
              (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2))
    == dot (plane_vector v1 v2) (plane_vector v1 v2).
  Proof.
    intros c s v1 v2 Hcs.
    calc (dot (plane_vector ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)
              (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2))
        (plane_vector ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)
              (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2)))
      == (((c×c + - (s×s))*v1 + (c×s + c×s)*v2) × ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)
          + (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2) × (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2))
         { (dot_plane_vector ((c×c + - (s×s))*v1 + (c×s + c×s)*v2)
                     (- ((c×s + c×s)*v1) + (c×c + - (s×s))*v2)) }
      == (((c×c + - (s×s)) × (c×c + - (s×s)) + (c×s + c×s) × (c×s + c×s)) × (v1×v1 + v2×v2))
         { (squared_length_scales (c×c + - (s×s)) (c×s + c×s) v1 v2) }
      == ((c×c + s×s) × (c×c + s×s) × (v1×v1 + v2×v2))
         { (double_angle_pythagoras c s) }
      == (1 × 1 × (v1×v1 + v2×v2))
         { Hcs }
      == (1 × (v1×v1 + v2×v2))
         { (left_identity (1 : F)) }
      == (v1×v1 + v2×v2)
         { (left_identity (v1×v1 + v2×v2)) }
      == (dot (plane_vector v1 v2) (plane_vector v1 v2))
         { (dot_plane_vector v1 v2) }.
  Qed.

  Lemma rotor_mul : ∀ c1 s1 c2 s2 : F,
    rotor c1 s1 × rotor c2 s2 == rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2).
  Proof.
    intros c1 s1 c2 s2.
    unfold rotor.
    repeat split; simpl; ring.
  Qed.

  Lemma rotor_mul_to_complex : ∀ c1 s1 c2 s2 : F,
    even_to_complex (rotor c1 s1 × rotor c2 s2)
    == even_to_complex (rotor c1 s1) × even_to_complex (rotor c2 s2).
  Proof.
    intros c1 s1 c2 s2.
    exact (even_to_complex_mul (rotor c1 s1) (rotor c2 s2)
             (rotor_even c1 s1) (rotor_even c2 s2)).
  Qed.

  Theorem rotor_composition_is_u1 : ∀ c1 s1 c2 s2 : F,
    even_to_complex (rotor c1 s1) × even_to_complex (rotor c2 s2)
    == even_to_complex (rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)).
  Proof.
    intros c1 s1 c2 s2.
    rewrite <- (rotor_mul_to_complex c1 s1 c2 s2).
    rewrite (rotor_mul c1 s1 c2 s2).
    reflexivity.
  Qed.

End Rotor.
