From Riktig Require Export Theory.Realizes Theory.Hom.

D is a dual space of V over F when D realizes Hom(V,F), the space of linear maps from V to F.
Module DualSpace.
  Class DualSpace (F V D : Type) `{VectorSpace F V}
      {eD : EqRel D} {zD : Zero D} {aD : HasAdd D} {nD : HasNeg D}
      {sD : HasScale F D} {VD : VectorSpace F D} := {
    represent :: Realizes F (LinearMap F V F) D;
  }.
End DualSpace.
Export DualSpace(DualSpace).
Existing Instance DualSpace.represent.
