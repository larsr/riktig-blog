Library Riktig.Structure.ConstructiveReals

From Riktig.Structure Require Export Ring Order NatOrder RingHom.

The constructive reals over an ordered field Q of rationals and naturals N: a commutative ring with a strict order ≺ carrying evidence -- asymmetric, transitive, cotransitive, respecting equality, and with ≼, its negation, deciding equality both ways; the rationals sit in it by a ring homomorphism that keeps and reflects the order, are dense in it and bound every element; an element apart from zero has an inverse; adding keeps the order and positive elements multiply to a positive one; the absolute value is the least bound of an element and its negation; every element is approximated to any positive rational precision; and every Cauchy sequence converges.
Module ConstructiveReals.
  Class ConstructiveReals (R Q N : Type)
      `{Ring R} `{!Commutative R R mult} `{OrderedField Q} `{NatOrder N}
      (cast : Q → R) (lt : HasLt R) (inv : HasInvApart R) (abs : HasAbs R)
      (approx : HasApprox R Q) := {
    cast_hom :: RingHom Q R cast;
    lt_asym : ∀ x y : R, x ≺ y → y ≺ x → False;
    lt_trans : ∀ x y z : R, x ≺ y → y ≺ z → x ≺ z;
    lt_cotrans : ∀ x y z : R, x ≺ z → ((x ≺ y) + (y ≺ z))%type;
    lt_equ_l : ∀ x x' y : R, x == x' → x ≺ y → x' ≺ y;
    lt_equ_r : ∀ x y y' : R, y == y' → x ≺ y → x ≺ y';
    equ_of_le : ∀ x y : R, x ≼ y → y ≼ x → x == y;
    cast_lt : ∀ q r : Q, q < r → cast q ≺ cast r;
    lt_cast : ∀ q r : Q, cast q ≺ cast r → q < r;
    zero_lt_one : (0 : R) ≺ 1;
    add_lt_mono_l : ∀ x y z : R, y ≺ z → x + y ≺ x + z;
    add_lt_reg_l : ∀ x y z : R, x + y ≺ x + z → y ≺ z;
    mul_pos : ∀ x y : R, 0 ≺ x → 0 ≺ y → 0 ≺ x × y;
    mul_le_mono_l : ∀ c x y : R, 0 ≼ c → x ≼ y → c × x ≼ c × y;
    inv_l : ∀ (x : R) (Hx : ((x ≺ 0) + (0 ≺ x))%type), inv_apart x Hx × x == 1;
    inv_pos : ∀ (x : R) (Hx : ((x ≺ 0) + (0 ≺ x))%type), 0 ≺ x → 0 ≺ inv_apart x Hx;
    dense : ∀ x y : R, x ≺ y → {q : Q & ((x ≺ cast q) × (cast q ≺ y))%type};
    archimedean : ∀ x : R, {q : Q & x ≺ cast q};
    abs_proper :: Proper (equ ==> equ) abs;
    abs_le_iff : ∀ x y : R, (x ≼ y ∧ - x ≼ y) ↔ abs x ≼ y;
    abs_mul : ∀ x y : R, abs (x × y) == abs x × abs y;
    approx_close : ∀ (x : R) (eps : Q), 0 < eps →
      abs (x + - cast (approx x eps)) ≼ cast eps;
    complete : ∀ u : N → R,
      (∀ eps : Q, 0 < eps →
        {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps}) →
      {l : R & ∀ eps : Q, 0 < eps →
        {n : N & ∀ i : N, n ≤ i → abs (u i + - l) ≼ cast eps}};
  }.
End ConstructiveReals.
