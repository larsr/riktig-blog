From Riktig.Structure Require Export Channel.
From Riktig.Theory Require Export Iso.

Section ChannelTheory.
  Context {O : Type} `{Category O} {A W : O}
    (encode : A ~> W) (decode : W ~> A) `{!Channel O A W encode decode}.

Cutting a composite at the channel: the receiver takes decode, the sender takes encode, and the wire between them vanishes.
  Theorem through_channel : ∀ (x y : O) (f : x ~> A) (g : A ~> y),
    (g ∘ decode) ∘ (encode ∘ f) == g ∘ f.
  Proof.
    intros x y f g.
    calc ((g ∘ decode) ∘ (encode ∘ f))
      == (g ∘ (decode ∘ (encode ∘ f))) { Category.comp_assoc }
      == (g ∘ ((decode ∘ encode) ∘ f)) { Category.comp_assoc }
      == (g ∘ (cat_id ∘ f)) { Channel.decode_encode }
      == (g ∘ f) { left_identity }.
  Qed.

What goes in comes out unchanged, so encode identifies nothing.
  Theorem encode_monic : ∀ (x : O) (f g : x ~> A),
    encode ∘ f == encode ∘ g → f == g.
  Proof.
    intros x f g Hfg.
    calc f
      == (cat_id ∘ f) { left_identity }
      == ((decode ∘ encode) ∘ f) { Channel.decode_encode }
      == (decode ∘ (encode ∘ f)) { Category.comp_assoc }
      == (decode ∘ (encode ∘ g)) { Hfg }
      == ((decode ∘ encode) ∘ g) { Category.comp_assoc }
      == (cat_id ∘ g) { Channel.decode_encode }
      == g { left_identity }.
  Qed.

Every value comes back off some wire, so decode is cancellable.
  Theorem decode_epi : ∀ (y : O) (f g : A ~> y),
    f ∘ decode == g ∘ decode → f == g.
  Proof.
    intros y f g Hfg.
    calc f
      == (f ∘ cat_id) { right_identity }
      == (f ∘ (decode ∘ encode)) { Channel.decode_encode }
      == ((f ∘ decode) ∘ encode) { Category.comp_assoc }
      == ((g ∘ decode) ∘ encode) { Hfg }
      == (g ∘ (decode ∘ encode)) { Category.comp_assoc }
      == (g ∘ cat_id) { Channel.decode_encode }
      == g { right_identity }.
  Qed.
End ChannelTheory.

Section ChannelComposition.
  Context {O : Type} `{Category O}.

Channels chain: wrap once more, unwrap in the opposite order.
  #[export] Instance channel_compose (A W V : O)
      (e1 : A ~> W) (d1 : W ~> A) (e2 : W ~> V) (d2 : V ~> W)
      `{!Channel O A W e1 d1} `{!Channel O W V e2 d2} :
    Channel O A V (e2 ∘ e1) (d1 ∘ d2).
  Proof.
    split.
    calc ((d1 ∘ d2) ∘ (e2 ∘ e1))
      == (d1 ∘ (d2 ∘ (e2 ∘ e1))) { Category.comp_assoc }
      == (d1 ∘ ((d2 ∘ e2) ∘ e1)) { Category.comp_assoc }
      == (d1 ∘ (cat_id ∘ e1)) { Channel.decode_encode }
      == (d1 ∘ e1) { left_identity }
      == (cat_id (x := A)) { Channel.decode_encode }.
  Qed.

An isomorphism is a channel in both directions.
  #[export] Instance iso_channel (a b : O) (to : a ~> b) (from : b ~> a)
      `{!Iso O a b to from} : Channel O a b to from :=
    {| Channel.decode_encode := Iso.hom_inv_id |}.

  #[export] Instance iso_channel_back (a b : O) (to : a ~> b) (from : b ~> a)
      `{!Iso O a b to from} : Channel O b a from to :=
    {| Channel.decode_encode := Iso.inv_hom_id |}.
End ChannelComposition.
