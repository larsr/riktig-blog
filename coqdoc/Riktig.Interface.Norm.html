From Coq Require Export Morphisms.
From Riktig Require Export Ops Interface.VectorSpace Interface.Order.

Class HasNorm F V := vnorm : V → F.
Notation "‖ u ‖" := (vnorm u) (at level 20, u at level 200).

#[global] Hint Mode HasNorm - ! : typeclass_instances.

Module NormedSpace.
  Class NormedSpace (F V : Type) `{OrderedField.OrderedField F}
      {nsE : EqRel V} {nsZ : Zero V} {nsA : HasAdd V} {nsN : HasNeg V}
      {nsS : HasScale F V} {VS : VectorSpace F V} {nsNorm : HasNorm F V} := {
    proper : ∀ u v : V, u == v → ‖ u ‖ == ‖ v ‖;
    nonneg : ∀ u : V, 0 ≤ ‖ u ‖;
    definite : ∀ u : V, ‖ u ‖ == 0 → u == 0;
    scale : ∀ (a : F) (u : V), ‖ a *: u ‖ == abs a × ‖ u ‖;
    triangle : ∀ u v : V, ‖ u + v ‖ ≤ ‖ u ‖ + ‖ v ‖
  }.
End NormedSpace.
Notation NormedSpace := NormedSpace.NormedSpace.

#[global] Instance NormedSpace_norm_Proper (F V : Type)
  `{OrderedField.OrderedField F}
  {nsE : EqRel V} {nsZ : Zero V} {nsA : HasAdd V} {nsN : HasNeg V}
  {nsS : HasScale F V} {VS : VectorSpace F V} {nsNorm : HasNorm F V}
  (NS : NormedSpace.NormedSpace F V)
  : Proper (equ ==> equ) (@vnorm F V _) := NormedSpace.proper.
