Library Riktig.Interface.Norm

From Coq Require Export Morphisms.
From Riktig Require Export Interface.VectorSpace Interface.Order.

Class HasNorm F V := vnorm : V → F.
Notation "‖ u ‖" := (vnorm u) (at level 20, u at level 200).

Hint Mode HasNorm - ! : typeclass_instances.

Typeclasses Transparent HasNorm.

Module NormedSpace.
  Class NormedSpace (F V : Type) `{OrderedField F}
      {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V}
      {Vs : HasScale F V} {VS : VectorSpace F V} {Vnorm : HasNorm F V} := {
    proper :: Proper (equ ==> equ) Vnorm;

    nonneg : ∀ u : V, 0 ≤ ‖ u ‖;
    definite : ∀ u : V, ‖ u ‖ == 0 → u == 0;
    scale :: Homogeneous F V F mult abs Vnorm;
    triangle : ∀ u v : V, ‖ u + v ‖ ≤ ‖ u ‖ + ‖ v ‖;
  }.

End NormedSpace.
