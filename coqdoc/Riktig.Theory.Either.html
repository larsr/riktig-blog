Library Riktig.Theory.Either

From Riktig.Structure Require Export Either.
From Riktig.Theory Require Export InitialAlgebra Signature.
From Riktig.Lib Require Import Calc.

Section EitherTheory.
  Context {O : Type} `{Bicartesian O} {K : Type} {den : K → O} {a b : K} {E : O}
    {build : HasBuild O (Either.shape den a b) E} {fold : HasCata O (Either.shape den a b) E}
    {EE : Either O den a b E build fold}.

  Theorem case_build : ∀ (x : O) (h : den a ⨿ den b ~> x),
    cata (F := shape_of den (eitherS (S := sig K) a b)) x h ∘ build == h.
  Proof.
    intros x h. etransitivity.
    - exact (InitialAlgebra.cata_build x h).
    - setoid_rewrite (sum_id (O := O)). apply right_identity.
  Qed.

  Theorem case_left : ∀ (x : O) (h : den a ⨿ den b ~> x),
    cata (F := shape_of den (eitherS (S := sig K) a b)) x h ∘ build ∘ inj1 == h ∘ inj1.
  Proof. intros x h. rewrite (case_build x h). reflexivity. Qed.

  Theorem case_right : ∀ (x : O) (h : den a ⨿ den b ~> x),
    cata (F := shape_of den (eitherS (S := sig K) a b)) x h ∘ build ∘ inj2 == h ∘ inj2.
  Proof. intros x h. rewrite (case_build x h). reflexivity. Qed.

Taking the sum apart is the inverse of building it.
  Theorem build_unbuild_either :
    build ∘ fold (den a ⨿ den b) (fmap (F := shape_of den (eitherS (S := sig K) a b)) build) == cat_id.
  Proof. exact build_unbuild. Qed.

  Theorem unbuild_build_either :
    fold (den a ⨿ den b) (fmap (F := shape_of den (eitherS (S := sig K) a b)) build) ∘ build == cat_id.
  Proof. exact unbuild_build. Qed.
End EitherTheory.
