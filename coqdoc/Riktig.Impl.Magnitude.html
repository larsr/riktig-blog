Definition mag2 {A : Type} {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A}
    {An : HasNeg A} {Ao : One A} {Am : HasMult A} {AR : Ring A}
    {Ag : HasGrade A} (rev : A → A) (a : A) : A :=
  ⟨ rev a × a ⟩_ 0.

#[export] Instance Magnitude_rev {A : Type} {Ae : EqRel A} {Az : Zero A}
    {Aa : HasAdd A} {An : HasNeg A} {Ao : One A} {Am : HasMult A}
    {AR : Ring A} {Ag : HasGrade A} (rev : A → A) {RV : Reversal A rev}
    {RE : Reflexive (@equ A Ae)} :
  Magnitude A rev (mag2 rev).
Proof. intros a. reflexivity. Qed.
