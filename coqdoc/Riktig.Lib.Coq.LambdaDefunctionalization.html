Defunctionalizing the continuations of the CPS translation. The continuations cps ever builds are the initial one and the two closures contf and contv, so a first-order stack dcont of their captured variables represents them all; reify reads a stack back as the term it stands for. Dispatching on the stack instead of applying the term gives the CK machine dstep: a state either evaluates a source term or feeds a source value to the stack. Decoding a state as a CPS application shows every transition is a conversion, and the machine computes exactly the call-by-value runs.
Inductive dcont : Type :=
  | Halt : dcont
  | KArg (a : term) (k : dcont)
  | KFun (f : term) (k : dcont).
Fixpoint reify (k : dcont) : term :=
  match k with
  | Halt ⇒ λ #0
  | KArg a k' ⇒ contf a (reify k')
  | KFun f k' ⇒ contv (psi f) (reify k')
  end.

Each frame, read back, is the continuation it stands for

Lemma reify_halt_joinable : ∀ t : term, joinable ⦅(reify Halt) t⦆ t.
Proof. intros t. apply joinable_bsteps. apply star_one. exact (bstep_beta #0 t). Qed.

Lemma reify_karg_joinable : ∀ (f a : term) (k : dcont),
  joinable ⦅(cps (App f a)) (reify k)⦆ ⦅(cps f) (reify (KArg a k))⦆.
Proof. intros f a k. exact (cps_app_unfold f a (reify k)). Qed.

Lemma reify_kfun_joinable : ∀ (a f : term) (k : dcont),
  joinable ⦅(reify (KArg a k)) (psi f)⦆ ⦅(cps a) (reify (KFun f k))⦆.
Proof. intros a f k. exact (contf_apply a (reify k) (psi f)). Qed.

Lemma reify_call_joinable : ∀ (b v : term) (k : dcont), val v = true →
  joinable ⦅(reify (KFun (Lam b) k)) (psi v)⦆ ⦅(cps (psub (scons v Var) b)) (reify k)⦆.
Proof.
  intros b v k Hv.
  apply (joinable_trans _ ⦅(psi (Lam b)) (psi v) (reify k)⦆).
  { exact (contv_apply (psi (Lam b)) (reify k) (psi v)). }
  apply (joinable_trans _ ⦅(psub (scons (psi v) Var) (cps b)) (reify k)⦆).
  { apply joinable_bsteps. apply star_one. apply bstep_app_l. exact (bstep_beta (cps b) (psi v)). }
  rewrite <- (cps_psub_single b v Hv). apply joinable_refl.
Qed.

The CK machine

Evaluate the function, push the argument; once both are values, contract. Values go to the stack unevaluated -- they are done.
Inductive dstate : Type :=
  | Ev (t : term) (k : dcont)
  | Ap (k : dcont) (v : term).

Inductive dstep : dstate → dstate → Prop :=
  | dstep_app : ∀ (f a : term) (k : dcont), dstep (Ev (App f a) k) (Ev f (KArg a k))
  | dstep_val : ∀ (v : term) (k : dcont), val v = true → dstep (Ev v k) (Ap k v)
  | dstep_arg : ∀ (a f : term) (k : dcont), dstep (Ap (KArg a k) f) (Ev a (KFun f k))
  | dstep_call : ∀ (b v : term) (k : dcont), val v = true →
      dstep (Ap (KFun (Lam b) k) v) (Ev (psub (scons v Var) b) k).

Definition dsteps : dstate → dstate → Prop := star dstep.

Definition druns (t : term) (k : dcont) (v : term) : Prop := dsteps (Ev t k) (Ap Halt v).

Soundness: every transition is a conversion of the decodings

Definition decode (s : dstate) : term :=
  match s with
  | Ev t k ⇒ ⦅(cps t) (reify k)⦆
  | Ap k v ⇒ ⦅(reify k) (psi v)⦆
  end.

Theorem dstep_decode : ∀ s s' : dstate, dstep s s' → joinable (decode s) (decode s').
Proof.
  intros s s' H. destruct H as [f a k | v k Hv | a f k | b v k Hv].
  - exact (reify_karg_joinable f a k).
  - exact (cps_val v (reify k) Hv).
  - exact (reify_kfun_joinable a f k).
  - exact (reify_call_joinable b v k Hv).
Qed.

Theorem dsteps_decode : ∀ s s' : dstate, dsteps s s' → joinable (decode s) (decode s').
Proof.
  intros s s' H. induction H as [s | s y z Hy Hs IH].
  - apply joinable_refl.
  - exact (joinable_trans _ _ _ (dstep_decode s y Hy) IH).
Qed.

A run computes what the translated term hands the empty stack.
Theorem druns_cps : ∀ (t v : term) (k : dcont),
  druns t k v → joinable ⦅(cps t) (reify k)⦆ (psi v).
Proof.
  intros t v k H.
  apply (joinable_trans _ ⦅(reify Halt) (psi v)⦆).
  - exact (dsteps_decode (Ev t k) (Ap Halt v) H).
  - exact (reify_halt_joinable (psi v)).
Qed.

Completeness: the machine performs every call-by-value run

A finished run is inverted one deterministic step at a time, and a cbv step at the head is replayed as machine transitions.
Lemma dsteps_ev_app_inv : ∀ (f a : term) (k : dcont) (w : term),
  dsteps (Ev (App f a) k) (Ap Halt w) → dsteps (Ev f (KArg a k)) (Ap Halt w).
Proof.
  intros f a k w H. inversion H as [| s y z Hstep Hrest]; subst.
  inversion Hstep; subst.
  - exact Hrest.
  - discriminate.
Qed.

Lemma dsteps_ev_val_inv : ∀ (v : term) (k : dcont) (w : term), val v = true →
  dsteps (Ev v k) (Ap Halt w) → dsteps (Ap k v) (Ap Halt w).
Proof.
  intros v k w Hv H. inversion H as [| s y z Hstep Hrest]; subst.
  inversion Hstep; subst.
  - discriminate Hv.
  - exact Hrest.
Qed.

Lemma dsteps_ap_karg_inv : ∀ (a f : term) (k : dcont) (w : term),
  dsteps (Ap (KArg a k) f) (Ap Halt w) → dsteps (Ev a (KFun f k)) (Ap Halt w).
Proof.
  intros a f k w H. inversion H as [| s y z Hstep Hrest]; subst.
  inversion Hstep; subst. exact Hrest.
Qed.

Lemma dsteps_cbv : ∀ t t' : term, cbv t t' →
  ∀ (k : dcont) (w : term), dsteps (Ev t' k) (Ap Halt w) → dsteps (Ev t k) (Ap Halt w).
Proof.
  intros t t' Hc. induction Hc as [b u Hu | f f' a Hc IH | f a a' Hf Hc IH]; intros k w H.
  - apply (star_step dstep _ _ _ (dstep_app (Lam b) u k)).
    apply (star_step dstep _ _ _ (dstep_val (Lam b) (KArg u k) eq_refl)).
    apply (star_step dstep _ _ _ (dstep_arg u (Lam b) k)).
    apply (star_step dstep _ _ _ (dstep_val u (KFun (Lam b) k) Hu)).
    apply (star_step dstep _ _ _ (dstep_call b u k Hu)).
    exact H.
  - apply (star_step dstep _ _ _ (dstep_app f a k)).
    apply IH. exact (dsteps_ev_app_inv f' a k w H).
  - apply (star_step dstep _ _ _ (dstep_app f a k)).
    apply (star_step dstep _ _ _ (dstep_val f (KArg a k) Hf)).
    apply (star_step dstep _ _ _ (dstep_arg a f k)).
    apply IH.
    apply (dsteps_ap_karg_inv a' f k w).
    apply (dsteps_ev_val_inv f (KArg a' k) w Hf).
    exact (dsteps_ev_app_inv f a' k w H).
Qed.

Theorem cbvs_druns : ∀ l v : term, cbvs l v → val v = true → druns l Halt v.
Proof.
  intros l v H. unfold druns. induction H as [t | t x u Hx Hs IH]; intros Hv.
  - apply star_one. exact (dstep_val t Halt Hv).
  - exact (dsteps_cbv t x Hx Halt u (IH Hv)).
Qed.

Running the machine

Because the continuation is data and not a closure, the transition is a function on states and the machine can simply be run: drun n takes up to n steps and stops at a state with no transition. Each step it takes is a step of the relation, so a computed trace is a proof-carrying one.
Definition dstep_fun (s : dstate) : option dstate :=
  match s with
  | Ev t k ⇒ match t with
              | App f a ⇒ Some (Ev f (KArg a k))
              | _ ⇒ Some (Ap k t)
              end
  | Ap k v ⇒ match k with
              | Halt ⇒ None
              | KArg a k' ⇒ Some (Ev a (KFun v k'))
              | KFun f k' ⇒ match f with
                             | Lam b ⇒ if val v then Some (Ev (psub (scons v Var) b) k') else None
                             | _ ⇒ None
                             end
              end
  end.

Lemma dstep_fun_sound : ∀ s s' : dstate, dstep_fun s = Some s' → dstep s s'.
Proof.
  intros [t k | k v] s' H; simpl in H.
  - destruct t as [n | b | f a | x]; injection H as H; subst.
    + exact (dstep_val (Var n) k eq_refl).
    + exact (dstep_val (Lam b) k eq_refl).
    + exact (dstep_app f a k).
    + exact (dstep_val (Const x) k eq_refl).
  - destruct k as [| a k' | f k']; [ discriminate | injection H as H; subst; exact (dstep_arg a v k') | ].
    destruct f as [n | b | f1 f2 | x]; try discriminate.
    destruct (val v) eqn:Hv; [ | discriminate ].
    injection H as H; subst. exact (dstep_call b v k' Hv).
Qed.

Fixpoint drun (n : nat) (s : dstate) : dstate :=
  match n with
  | O ⇒ s
  | S n' ⇒ match dstep_fun s with Some s' ⇒ drun n' s' | None ⇒ s end
  end.

Theorem drun_dsteps : ∀ (n : nat) (s : dstate), dsteps s (drun n s).
Proof.
  induction n as [| n IH]; intros s; simpl.
  - apply star_refl.
  - destruct (dstep_fun s) as [s' |] eqn:Hs; [ | apply star_refl ].
    exact (star_step dstep _ _ _ (dstep_fun_sound s s' Hs) (IH s')).
Qed.
