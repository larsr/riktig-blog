From Riktig Require Export Structure.PeanoArith Lib.Calc.

The recursion equations give the commutative semiring laws, each by induction on the argument the equations recurse on.
Section PeanoArithTheory.
  Context {N : Type} `{PeanoArith N}.

  Theorem add_zero_l : ∀ a : N, 0 + a == a.
  Proof.
    apply (nat_induction (fun a ⇒ 0 + a == a)); [ solve_proper | | ].
    - exact (add_zero 0).
    - intros n IH.
      calc (0 + succ n)
        == (succ (0 + n)) { add_succ }
        == (succ n) { IH }.
  Qed.

  Theorem add_succ_l : ∀ a b : N, succ a + b == succ (a + b).
  Proof.
    intros a.
    apply (nat_induction (fun b ⇒ succ a + b == succ (a + b))); [ solve_proper | | ].
    - calc (succ a + 0)
        == (succ a) { add_zero }
        == (succ (a + 0)) { (add_zero a) }.
    - intros n IH.
      calc (succ a + succ n)
        == (succ (succ a + n)) { add_succ }
        == (succ (succ (a + n))) { IH }
        == (succ (a + succ n)) { (add_succ a n) }.
  Qed.

  Theorem add_assoc : ∀ a b c : N, (a + b) + c == a + (b + c).
  Proof.
    intros a b.
    apply (nat_induction (fun c ⇒ (a + b) + c == a + (b + c))); [ solve_proper | | ].
    - calc ((a + b) + 0)
        == (a + b) { add_zero }
        == (a + (b + 0)) { (add_zero b) }.
    - intros n IH.
      calc ((a + b) + succ n)
        == (succ ((a + b) + n)) { add_succ }
        == (succ (a + (b + n))) { IH }
        == (a + succ (b + n)) { (add_succ a (b + n)) }
        == (a + (b + succ n)) { (add_succ b n) }.
  Qed.

  Theorem add_comm : ∀ a b : N, a + b == b + a.
  Proof.
    intros a.
    apply (nat_induction (fun b ⇒ a + b == b + a)); [ solve_proper | | ].
    - calc (a + 0)
        == a { add_zero }
        == (0 + a) { add_zero_l }.
    - intros n IH.
      calc (a + succ n)
        == (succ (a + n)) { add_succ }
        == (succ (n + a)) { IH }
        == (succ n + a) { add_succ_l }.
  Qed.

  Theorem add_right_comm : ∀ a b c : N, (a + b) + c == (a + c) + b.
  Proof.
    intros a b c.
    calc ((a + b) + c)
      == (a + (b + c)) { add_assoc }
      == (a + (c + b)) { (add_comm b c) }
      == ((a + c) + b) { add_assoc }.
  Qed.

  Theorem succ_eq_add_one : ∀ n : N, succ n == n + 1.
  Proof.
    intros n.
    calc (succ n)
      == (succ (n + 0)) { (add_zero n) }
      == (n + succ 0) { add_succ }
      == (n + 1) { one_succ }.
  Qed.

  Theorem mul_zero_l : ∀ a : N, 0 × a == 0.
  Proof.
    apply (nat_induction (fun a ⇒ 0 × a == 0)); [ solve_proper | | ].
    - exact (mul_zero 0).
    - intros n IH.
      calc (0 × succ n)
        == (0 × n + 0) { mul_succ }
        == (0 × n) { add_zero }
        == (0 : N) { IH }.
  Qed.

  Theorem mul_succ_l : ∀ a b : N, succ a × b == a × b + b.
  Proof.
    intros a.
    apply (nat_induction (fun b ⇒ succ a × b == a × b + b)); [ solve_proper | | ].
    - calc (succ a × 0)
        == (0 : N) { mul_zero }
        == (a × 0) { (mul_zero a) }
        == (a × 0 + 0) { add_zero }.
    - intros n IH.
      calc (succ a × succ n)
        == (succ a × n + succ a) { mul_succ }
        == ((a × n + n) + succ a) { IH }
        == (succ ((a × n + n) + a)) { add_succ }
        == (succ ((a × n + a) + n)) { add_right_comm }
        == ((a × n + a) + succ n) { (add_succ (a × n + a) n) }
        == (a × succ n + succ n) { (mul_succ a n) }.
  Qed.

  Theorem mul_one_r : ∀ a : N, a × 1 == a.
  Proof.
    intros a.
    calc (a × 1)
      == (a × succ 0) { one_succ }
      == (a × 0 + a) { mul_succ }
      == (0 + a) { mul_zero }
      == a { add_zero_l }.
  Qed.

  Theorem mul_comm : ∀ a b : N, a × b == b × a.
  Proof.
    intros a.
    apply (nat_induction (fun b ⇒ a × b == b × a)); [ solve_proper | | ].
    - calc (a × 0)
        == (0 : N) { mul_zero }
        == (0 × a) { mul_zero_l }.
    - intros n IH.
      calc (a × succ n)
        == (a × n + a) { mul_succ }
        == (n × a + a) { IH }
        == (succ n × a) { mul_succ_l }.
  Qed.

  Theorem mul_one_l : ∀ a : N, 1 × a == a.
  Proof.
    intros a.
    calc (1 × a)
      == (a × 1) { mul_comm }
      == a { mul_one_r }.
  Qed.

  Theorem mul_add_distr_l : ∀ a b c : N, a × (b + c) == a × b + a × c.
  Proof.
    intros a b.
    apply (nat_induction (fun c ⇒ a × (b + c) == a × b + a × c)); [ solve_proper | | ].
    - calc (a × (b + 0))
        == (a × b) { add_zero }
        == (a × b + 0) { add_zero }
        == (a × b + a × 0) { (mul_zero a) }.
    - intros n IH.
      calc (a × (b + succ n))
        == (a × succ (b + n)) { add_succ }
        == (a × (b + n) + a) { mul_succ }
        == ((a × b + a × n) + a) { IH }
        == (a × b + (a × n + a)) { add_assoc }
        == (a × b + a × succ n) { (mul_succ a n) }.
  Qed.

  Theorem mul_add_distr_r : ∀ a b c : N, (a + b) × c == a × c + b × c.
  Proof.
    intros a b c.
    calc ((a + b) × c)
      == (c × (a + b)) { mul_comm }
      == (c × a + c × b) { mul_add_distr_l }
      == (a × c + c × b) { (mul_comm c a) }
      == (a × c + b × c) { (mul_comm c b) }.
  Qed.

  Theorem mul_assoc : ∀ a b c : N, (a × b) × c == a × (b × c).
  Proof.
    intros a b.
    apply (nat_induction (fun c ⇒ (a × b) × c == a × (b × c))); [ solve_proper | | ].
    - calc ((a × b) × 0)
        == (0 : N) { mul_zero }
        == (a × 0) { (mul_zero a) }
        == (a × (b × 0)) { (mul_zero b) }.
    - intros n IH.
      calc ((a × b) × succ n)
        == ((a × b) × n + a × b) { mul_succ }
        == (a × (b × n) + a × b) { IH }
        == (a × (b × n + b)) { mul_add_distr_l }
        == (a × (b × succ n)) { (mul_succ b n) }.
  Qed.

End PeanoArithTheory.
