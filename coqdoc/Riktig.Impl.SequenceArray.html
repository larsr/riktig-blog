Library Riktig.Impl.SequenceArray

From Coq Require Import PArray ZArith Arith.
From Riktig Require Export Structure.Sequence Lib.Coq.Parser Lib.Coq.PArray Lib.Coq.Decidable.
Local Close Scope lambda_scope.

A primitive array read through a cursor: the sequence is the array with the position of the next element; a step reads the entry at the position and advances it. Elements are compared by Coq's equality.
Definition ArrayCursor (C : Type) : Type := (array C × nat)%type.

Section SequenceArray.
  Context {C : Type}.

  Definition array_len (a : array C) : nat := Z.to_nat (Uint63.to_Z (PArray.length a)).

  #[export] Instance ArrayCursor_EqRel : EqRel (ArrayCursor C) := eq.
  #[export] Instance ArrayCursor_HasStep : HasStep C (ArrayCursor C) :=
    fun s c s' ⇒ (snd s < array_len (fst s))%nat ∧ c = (fst s).[int_of_nat (snd s)] ∧ s' = (fst s, S (snd s)).
  #[export] Instance ArrayCursor_HasNext : HasNext C (ArrayCursor C) :=
    fun s ⇒ if dec ((snd s < array_len (fst s))%nat)
             then Some ((fst s).[int_of_nat (snd s)], (fst s, S (snd s)))
             else None.

  #[export] Instance ArrayCursor_Sequence : Sequence C (ArrayCursor C) ArrayCursor_HasStep (eqC := eq).
  Proof.
    split.
    - exact eq_equivalence.
    - exact eq_equivalence.
    - intros s s0 Hs c c0 Hc t t0 Ht. unfold equ, ArrayCursor_EqRel in Hs, Ht. unfold equ in Hc. subst. reflexivity.
    - intros s c c' s' s'' [_ [Hc Hs]] [_ [Hc' Hs']]. subst. split; reflexivity.
  Qed.

  Lemma array_next_step : ∀ (s : ArrayCursor C) (c : C) (s' : ArrayCursor C),
    step s c s' ↔ ∃ c0 : C, next s = Some (c0, s') ∧ c0 = c.
  Proof.
    intros s c s'. unfold next, ArrayCursor_HasNext, step, ArrayCursor_HasStep.
    destruct (dec (P := (snd s < array_len (fst s))%nat)) as [Hlt | Hlt].
    - split.
      + intros [_ [Hc Hs]]. subst. ∃ ((fst s).[int_of_nat (snd s)]). split; reflexivity.
      + intros [c0 [Hn Hc]]. injection Hn as Hc0 Hs. subst. split; [ exact Hlt | split; reflexivity ].
    - split; [ intros [Hlt' _]; destruct (Hlt Hlt') | intros [c0 [Hn _]]; discriminate Hn ].
  Qed.

  Definition array_size (s : ArrayCursor C) : nat := array_len (fst s) - snd s.
End SequenceArray.
