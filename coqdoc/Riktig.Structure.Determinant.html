A determinant for a basis is an alternating multilinear form over the basis's index carrier, taking the value 1 on the basis.
Class Determinant (n : nat) (I F V : Type)
    {eI : EqRel I} {Ieq : EquDec I} {Vu : HasUpd I V} {IFin : FinIndex n I}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V} {Vl : HasLincomb I F V} (B : Basis n I F V) := {
  determinant_form : MultilinearForm I F V;
  determinant_is_alternating :: Alternating determinant_form;
  determinant_normalization : determinant_form (Basis.vec B) == 1;
}.

Arguments determinant_form {n I F V eI Ieq Vu IFin Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV Vl B}.
Coercion determinant_form : Determinant >-> MultilinearForm.
Arguments determinant_is_alternating {n I F V eI Ieq Vu IFin Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV Vl B}.
Arguments determinant_normalization {n I F V eI Ieq Vu IFin Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV Vl B}.

Section DeterminantLaws.
  Context {n : nat} {I F V : Type} {eI : EqRel I} {Ieq : EquDec I}
    {Vu : HasUpd I V} {IFin : FinIndex n I}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V} {Vl : HasLincomb I F V} {B : Basis n I F V}.

  Definition determinant_multilinear_form (d : Determinant n I F V B)
    : MultilinearForm I F V := determinant_form d.

  Definition determinant_alternating (d : Determinant n I F V B)
    : Alternating (determinant_form d) := determinant_is_alternating d.

  Definition determinant_basis_normalization (d : Determinant n I F V B)
    : d (Basis.vec B) == 1 := determinant_normalization d.

  Definition determinant_from_laws (m : MultilinearForm I F V)
      (m_alternating : Alternating m)
      (m_normalization : m (Basis.vec B) == 1)
    : Determinant n I F V B :=
    {| determinant_form := m;
       determinant_is_alternating := m_alternating;
       determinant_normalization := m_normalization |}.
End DeterminantLaws.
