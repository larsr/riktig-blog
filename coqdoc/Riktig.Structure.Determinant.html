From Riktig.Structure Require Export Multilinear.

A determinant on a vector space with respect to a basis: a multilinear alternating form on families of vectors that is 1 on the basis.
Module Determinant.
  Class Determinant (F I V : Type) `{VectorSpace F V} {eqI : EqRel I} {update : HasUpd I V}
      (basis : I → V) (determinant : HasDet F I V) := {
    is_multilinear :: Multilinear F I V det;
    is_alternating :: Alternating F I V det;
    det_basis : det basis == 1;
  }.

End Determinant.
