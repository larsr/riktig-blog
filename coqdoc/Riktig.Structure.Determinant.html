From Riktig.Structure Require Export VectorSpace Decidable.

A determinant on a vector space, over an index with decided equality and a distinguished family, the basis: a form on families of vectors that respects equality, is linear in every slot, vanishes when two distinct slots hold the same vector, and is 1 on the basis.
Module Determinant.
  Class Determinant (F I V : Type) `{VectorSpace F V} {eqI : EqRel I} {EI : EquDec I}
      (basis : I → V) (determinant : HasDet F I V) := {
    det_proper : ∀ vs vs' : I → V,
      Proper (equ ==> equ) vs → Proper (equ ==> equ) vs' →
      (∀ i : I, vs i == vs' i) → det vs == det vs';
    det_slot_add : ∀ (vs : I → V) (j : I) (u v : V), Proper (equ ==> equ) vs →
      det (fun i ⇒ if equ_dec i j then u + v else vs i)
        == det (fun i ⇒ if equ_dec i j then u else vs i) + det (fun i ⇒ if equ_dec i j then v else vs i);
    det_slot_smul : ∀ (vs : I → V) (j : I) (a : F) (u : V), Proper (equ ==> equ) vs →
      det (fun i ⇒ if equ_dec i j then a *: u else vs i)
        == a × det (fun i ⇒ if equ_dec i j then u else vs i);
    det_alternating : ∀ (vs : I → V) (j j' : I), Proper (equ ==> equ) vs →
      j =/= j' → vs j == vs j' → det vs == 0;
    det_basis : det basis == 1;
  }.

End Determinant.
