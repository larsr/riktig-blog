  Lemma bcount_cons : ∀ n (b : bool) (A : Blade n),
    bcount (bcons b A) = if b then S (bcount A) else bcount A.
  Proof. reflexivity. Qed.

  Lemma bcount_S : ∀ n (D : Blade (S n)),
    bcount D = if bhead D then S (bcount (btail D)) else bcount (btail D).
  Proof. reflexivity. Qed.

  Lemma bcount_bempty : ∀ n, bcount (@bempty n) = O.
  Proof.
    induction n as [| m IH].
    - reflexivity.
    - exact IH.
  Qed.

  Lemma bcount_le : ∀ n (D : Blade n), (bcount D ≤ n)%nat.
  Proof.
    induction n as [| m IH]; intros D.
    - apply le_0_n.
    - rewrite (blade_S m D), bcount_cons.
      destruct (bhead D).
      + exact (le_n_S (bcount (btail D)) m (IH (btail D))).
      + exact (le_S (bcount (btail D)) m (IH (btail D))).
  Qed.

  Lemma cl_grade_at : ∀ n (k : nat) (x : Cl F n) (D : Blade n),
    cl_grade k x D = if Nat.eqb (bcount D) k then x D else 0.
  Proof. reflexivity. Qed.

  Lemma cl_grade_ext : ∀ n (k : nat) (x y : Cl F n),
    x == y → cl_grade k x == cl_grade k y.
  Proof.
    intros n k x y Hxy D.
    rewrite !cl_grade_at.
    destruct (Nat.eqb (bcount D) k).
    - exact (Hxy D).
    - reflexivity.
  Qed.

  Lemma cl_grade_add : ∀ n (k : nat) (x y : Cl F n),
    cl_grade k (x + y) == cl_grade k x + cl_grade k y.
  Proof.
    intros n k x y D.
    rewrite (cl_grade_at n k (x + y) D),
            (pw_add_at (cl_grade k x) (cl_grade k y) D),
            (cl_grade_at n k x D), (cl_grade_at n k y D).
    destruct (Nat.eqb (bcount D) k).
    - exact (pw_add_at x y D).
    - ring.
  Qed.

  Lemma cl_grade_scale : ∀ n (k : nat) (a : F) (x : Cl F n),
    cl_grade k (a *: x) == a *: cl_grade k x.
  Proof.
    intros n k a x D.
    rewrite (cl_grade_at n k (a *: x) D),
            (pw_scale_at a (cl_grade k x) D), (cl_grade_at n k x D).
    rewrite (scale_is_mult a (if Nat.eqb (bcount D) k then x D else 0)).
    destruct (Nat.eqb (bcount D) k).
    - rewrite (pw_scale_at a x D), (scale_is_mult a (x D)). reflexivity.
    - ring.
  Qed.

  Lemma cl_grade_idempotent : ∀ n (k : nat) (x : Cl F n),
    cl_grade k (cl_grade k x) == cl_grade k x.
  Proof.
    intros n k x D.
    rewrite !cl_grade_at.
    destruct (Nat.eqb (bcount D) k); reflexivity.
  Qed.

  Lemma cl_grade_orthogonal : ∀ n (k l : nat) (x : Cl F n),
    k ≠ l → cl_grade k (cl_grade l x) == 0.
  Proof.
    intros n k l x Hkl D.
    rewrite !cl_grade_at.
    destruct (Nat.eqb (bcount D) k) eqn:Hk; [| reflexivity].
    destruct (Nat.eqb (bcount D) l) eqn:Hl; [| reflexivity].
    apply Nat.eqb_eq in Hk, Hl.
    destruct (Hkl (eq_trans (eq_sym Hk) Hl)).
  Qed.

  Lemma cl_grade_own : ∀ n (x : Cl F n) (D : Blade n),
    cl_grade (bcount D) x D == x D.
  Proof.
    intros n x D.
    rewrite cl_grade_at, Nat.eqb_refl. reflexivity.
  Qed.

  Lemma cl_grade_other : ∀ n (k : nat) (x : Cl F n) (D : Blade n),
    k ≠ bcount D → cl_grade k x D == 0.
  Proof.
    intros n k x D Hk.
    rewrite cl_grade_at.
    destruct (Nat.eqb (bcount D) k) eqn:HD; [| reflexivity].
    apply Nat.eqb_eq in HD.
    destruct (Hk (eq_sym HD)).
  Qed.

  Lemma cl_grade_complete : ∀ n (x : Cl F n),
    bigop (S n) (fun k ⇒ cl_grade k x) == x.
  Proof.
    intros n x D.
    calc (bigop (S n) (fun k ⇒ cl_grade k x) D)
      == (bigop (S n) (fun k ⇒ cl_grade k x D))
        { (bigop_app (S n) (fun k ⇒ cl_grade k x) D) }
      == (cl_grade (bcount D) x D)
        { (bigop_support_singleton (S n) (bcount D) (fun k ⇒ cl_grade k x D)
            (le_n_S (bcount D) n (bcount_le n D))
            (fun k Hk ⇒ cl_grade_other n k x D Hk)) }
      == (x D)
        { (cl_grade_own n x D) }.
  Qed.

  #[export] Instance Cl_HasGrade (n : nat) : HasGrade (Cl F n) := cl_grade.

  #[export] Instance Cl_Graded (n : nat) :
    Graded n F (Cl F n) (@bigop_list (Cl F n) _ _) := {
    grade_linear := fun k ⇒
      {| Linear.proper := fun x y Hxy ⇒ cl_grade_ext n k x y Hxy;
         Linear.is_additive := cl_grade_add n k;
         Linear.is_homogeneous := cl_grade_scale n k |};
    grade_idempotent := fun k x ⇒ cl_grade_idempotent n k x;
    grade_orthogonal := fun k l x Hkl ⇒ cl_grade_orthogonal n k l x Hkl;
    grade_complete := cl_grade_complete n;
  }.


  Section CliffordGradedProducts.
    Context {n : nat} (s : Signature F n).

    Definition cl_wedge (x y : Cl F n) : Cl F n :=
      bigop (S n) (fun r ⇒ bigop (S n) (fun t ⇒
        cl_grade (r + t) (cl_mul s (cl_grade r x) (cl_grade t y)))).

    Definition cl_dot (x y : Cl F n) : Cl F n :=
      bigop (S n) (fun r ⇒ bigop (S n) (fun t ⇒
        cl_grade ((r - t) + (t - r)) (cl_mul s (cl_grade r x) (cl_grade t y)))).

    #[local] Instance Cl_Wedge : HasGradedWedge (Cl F n) := cl_wedge.
    #[local] Instance Cl_Dot : HasGradedDot (Cl F n) := cl_dot.
    #[local] Instance Grade_Mult : HasMult (Cl F n) := cl_mul s.

    #[local] Instance Cl_GradedProducts :
      GradedProducts n F (Cl F n) (@bigop_list (Cl F n) _ _) := {
      wedge_from_grade_parts := fun x y ⇒ reflexivity (cl_wedge x y);
      dot_from_grade_parts := fun x y ⇒ reflexivity (cl_dot x y);
    }.

  End CliffordGradedProducts.


  Definition cl_blade {n : nat} (A : Blade n) : Cl F n :=
    fun D ⇒ if bis_nil (bxor A D) then 1 else 0.

  Definition sig_of_list (l : list F) : Signature F (length l) :=
    Vector.of_list l.

  Lemma ga_vec_not_grade_one : ∀ n (v : Fin.t n → F) (D : Blade n),
    bcount D ≠ 1%nat → ga_vec v D == 0.
  Proof.
    induction n as [| m IH]; intros v D Hc.
    - reflexivity.
    - rewrite (blade_S m D).
      destruct (bhead D) eqn:Hh.
      + rewrite (ga_vec_cons_true m v (btail D)).
        destruct (bis_nil (btail D)) eqn:Hnil; [| reflexivity].
        rewrite (blade_S m D), bcount_cons, Hh,
                (bis_nil_true m (btail D) Hnil), bcount_bempty in Hc.
        destruct (Hc eq_refl).
      + rewrite (ga_vec_cons_false m v (btail D)).
        apply IH.
        rewrite (blade_S m D), bcount_cons, Hh in Hc.
        exact Hc.
  Qed.

  Lemma cl_grade_one_vec : ∀ n (v : Fin.t n → F),
    cl_grade 1%nat (ga_vec v) == ga_vec v.
  Proof.
    intros n v D.
    rewrite cl_grade_at.
    destruct (Nat.eqb (bcount D) 1) eqn:Hk; [ reflexivity |].
    symmetry.
    exact (ga_vec_not_grade_one n v D
             (proj1 (Nat.eqb_neq (bcount D) 1%nat) Hk)).
  Qed.

  Lemma bcount_zero_is_nil : ∀ n (D : Blade n),
    Nat.eqb (bcount D) 0%nat = bis_nil D.
  Proof.
    induction n as [| m IH]; intros D.
    - reflexivity.
    - rewrite (blade_S m D), bis_nil_cons, bcount_cons.
      destruct (bhead D).
      + reflexivity.
      + exact (IH (btail D)).
  Qed.

The unit is homogeneous of grade zero — Structure/GradedGA.v's second field.
  Lemma cl_grade_zero_one : ∀ n : nat,
    cl_grade 0%nat (cl_one : Cl F n) == cl_one.
  Proof.
    intros n D.
    rewrite cl_grade_at, bcount_zero_is_nil.
    unfold cl_one.
    destruct (bis_nil D); reflexivity.
  Qed.

  Lemma bparity_odd : ∀ n (D : Blade n),
    bparity D = Nat.odd (bcount D).
  Proof.
    induction n as [| m IH]; intros D.
    - reflexivity.
    - rewrite (bparity_S m D), (bcount_S m D).
      destruct (bhead D); rewrite (IH (btail D)).
      + rewrite Nat.odd_succ, <- Nat.negb_odd. reflexivity.
      + reflexivity.
  Qed.

  Lemma cl_mul_vec_odd : ∀ n (s : Signature F n) (u v : Fin.t n → F)
      (D : Blade n),
    bparity D = true → cl_mul s (ga_vec u) (ga_vec v) D == 0.
  Proof.
    intros n s u v D HD.
    transitivity (bsum (fun _ : Blade n ⇒ (0 : F)));
      [| exact (bsum_zero n) ].
    apply bsum_ext. intros A.
    destruct (bparity A) eqn:HA.
    - assert (Hxor : bparity (bxor A D) = false).
      { rewrite (bparity_bxor n A D), HA, HD. reflexivity. }
      rewrite (ga_vec_parity n v (bxor A D) Hxor). ring.
    - rewrite (ga_vec_parity n u A HA). ring.
  Qed.

  Lemma cl_grade_odd_vec_mul : ∀ n (s : Signature F n)
      (u v : Fin.t n → F) (k : nat),
    Nat.odd k = true →
    cl_grade k (cl_mul s (ga_vec u) (ga_vec v)) == 0.
  Proof.
    intros n s u v k Hk D.
    rewrite cl_grade_at.
    destruct (Nat.eqb (bcount D) k) eqn:HD; [| reflexivity].
    apply Nat.eqb_eq in HD.
    apply (cl_mul_vec_odd n s u v D).
    rewrite (bparity_odd n D), HD. exact Hk.
  Qed.


  Lemma qform_const_one : ∀ n (u : Fin.t n → F),
    qform (Vector.const 1 n) u == ⟪ u , u ⟫.
  Proof.
    induction n as [| m IH]; intros u.
    - reflexivity.
    -
      transitivity
        (1 × (u Fin.F1 × u Fin.F1)
         + qform (Vector.const 1 m) (fun i ⇒ u (Fin.FS i)));
        [ reflexivity |].
      rewrite (IH (fun i ⇒ u (Fin.FS i))).
      transitivity
        (u Fin.F1 × u Fin.F1
         + bigopF (fun i ⇒ u (Fin.FS i) × u (Fin.FS i))).
      + apply op_proper; [ ring | reflexivity ].
      + symmetry. exact (bigopF_S m (fun i ⇒ u i × u i)).
  Qed.

  Lemma gv_square_inner : ∀ n (v : Fin.t n → F),
    cl_mul (Vector.const 1 n) (ga_vec v) (ga_vec v) == norm2 v *: cl_one.
  Proof.
    intros n v.
    calc (cl_mul (Vector.const 1 n) (ga_vec v) (ga_vec v))
      == ((qform (Vector.const 1 n) v *: cl_one) : Cl F n)
        { (gv_square n (Vector.const 1 n) v) }
      == ((norm2 v *: cl_one) : Cl F n)
        { (qform_const_one n v) }.
  Qed.

  Lemma cl_reverse_one : ∀ n, cl_reverse (cl_one : Cl F n) == cl_one.
  Proof.
    intros n D.
    rewrite (cl_reverse_at n cl_one D). unfold cl_one.
    destruct (bis_nil D) eqn:Hnil.
    - rewrite (bis_nil_true n D Hnil), (bcount_bempty n). simpl. ring.
    - ring.
  Qed.

  Section CliffordInnerGA.
    Context {n : nat}.

    #[local] Instance InnerGA_One : One (Cl F n) := cl_one.
    #[local] Instance InnerGA_Mult : HasMult (Cl F n) :=
      cl_mul (Vector.const 1 n).

    Definition Cl_GA_inner :
        GeometricAlgebra F (Fin.t n → F) (Cl F n) (Q := norm2).
    Proof.
      unshelve econstructor.
      - exact ga_vec.
      - exact (Cl_Ring (Vector.const 1 n)).
      - exact PW_ScaleLaws.
      - intros y. unshelve econstructor.
        + intros x x' Hx.
          exact (cl_mul_proper (Vector.const 1 n) x x' Hx y y (reflexivity y)).
        + intros x x'. exact (cl_mul_dist_r n (Vector.const 1 n) x x' y).
        + intros a x. exact (cl_mul_scale_l n (Vector.const 1 n) a x y).
      - intros x. unshelve econstructor.
        + intros y y' Hy.
          exact (cl_mul_proper (Vector.const 1 n) x x (reflexivity x) y y' Hy).
        + intros y y'. exact (cl_mul_dist_l n (Vector.const 1 n) x y y').
        + intros a y. exact (cl_mul_scale_r n (Vector.const 1 n) a x y).
      - exact Cl_VecLinear.
      - exact (gv_square_inner n).
    Defined.

  End CliffordInnerGA.
