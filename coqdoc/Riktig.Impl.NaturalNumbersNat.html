Coq's nat is the natural numbers of Coq's types: build is O and S, and cata is the recursion.
#[export] Instance Nat_HasBuild : HasBuild Type (NaturalNumbers.shape Type) nat :=
  fun s ⇒ match s with
           | inl _ ⇒ O
           | inr n ⇒ S n
           end.

#[export] Instance Nat_HasCata : HasCata Type (NaturalNumbers.shape Type) nat :=
  fun B h ⇒ fix go (n : nat) : B :=
    match n with
    | O ⇒ h (inl tt)
    | S m ⇒ h (inr (go m))
    end.

#[export] Instance Nat_NaturalNumbers : NaturalNumbers Type nat Nat_HasBuild Nat_HasCata.
Proof.
  split. split.
  - exact (layer_functor (fun A : Type ⇒ A) (natS (S := sig Type))).
  - intros B h h' Hh n. induction n as [| m IH].
    + exact (Hh (inl tt)).
    + exact (eq_trans (f_equal (fun b ⇒ h (inr b)) IH) (Hh (inr _))).
  - intros B h [[] | m]; reflexivity.
  - intros B h u Hu n. induction n as [| m IH].
    + exact (Hu (inl tt)).
    + exact (eq_trans (Hu (inr m)) (f_equal (fun b ⇒ h (inr b)) IH)).
Qed.
