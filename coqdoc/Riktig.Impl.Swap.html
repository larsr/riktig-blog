Library Riktig.Impl.Swap

From Riktig Require Export Theory.Fin Impl.Decidable.
From Riktig Require Import Ops.

Definition swap {I : Type} {eqdec : EqDec I} (i j r : I) : I :=
  if dec (r = i) then j
  else if dec (r = j) then i
  else r.

#[export] Instance swap_Swap {I : Type} {eqdec : EqDec I} :
  Swap (eI := eq) (swap (I := I)).
Proof.
  constructor.
  - intros i j x Hxi.
    unfold swap.
    destruct (dec (x = i)) as [_ | Hne].
    + reflexivity.
    + destruct (Hne Hxi).
  - intros i j x Hij Hxj.
    unfold swap.
    destruct (dec (x = i)) as [Heq | _].
    + destruct (Hij (eq_sym (eq_trans (eq_sym Hxj) Heq))).
    + destruct (dec (x = j)) as [_ | Hne].
      × reflexivity.
      × destruct (Hne Hxj).
  - intros i j x Hxi Hxj.
    unfold swap.
    destruct (dec (x = i)) as [Heq | _].
    + destruct (Hxi Heq).
    + destruct (dec (x = j)) as [Heq2 | _].
      × destruct (Hxj Heq2).
      × reflexivity.
Qed.
