Library Riktig.Theory.Monoid

From Riktig Require Export Structure.Monoid Lib.Calc.

Section MonoidTheory.
  Context {M : Type} `{Monoid M}.

  Theorem id_unique : ∀ e f : M,
    (∀ a, e + a == a) → (∀ a, a + f == a) → e == f.
  Proof.
    intros e f He Hf.
    calc e
      == (e + f) { Hf }
      == f { He }.
  Qed.

  Theorem zero_unique_l : ∀ e : M, (∀ a, e + a == a) → e == 0.
  Proof.
    intros e He.
    calc e
      == (e + 0) { right_identity }
      == (0 : M) { He }.
  Qed.

  Theorem zero_unique_r : ∀ e : M, (∀ a, a + e == a) → e == 0.
  Proof.
    intros e He.
    calc e
      == (0 + e) { left_identity }
      == (0 : M) { He }.
  Qed.

  Lemma add_zero_l {z y : M} : z == 0 → z + y == y.
  Proof.
    intros Hz.
    calc (z + y)
      == (0 + y) { Hz }
      == y { left_identity }.
  Qed.

  Lemma add_zero_r {y z : M} : z == 0 → y + z == y.
  Proof.
    intros Hz.
    calc (y + z)
      == (y + 0) { Hz }
      == y { right_identity }.
  Qed.

End MonoidTheory.

Instance Monoid_LeftIdUnique M `{HM : Monoid M} :
  Unique M (fun e : M ⇒ ∀ a, e + a == a).
Proof.
  intros e f He Hf.
  calc e
    == (0 : M) { zero_unique_l }
    == f { zero_unique_l }.
Qed.
