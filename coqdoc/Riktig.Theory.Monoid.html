From Riktig Require Export Structure.Monoid Lib.Calc.

Section MonoidTheory.
  Context {M : Type} `{Monoid M}.

A left identity and a right identity are the same element.
  Theorem id_unique : ∀ e f : M,
    (∀ a, e + a == a) → (∀ a, a + f == a) → e == f.
  Proof.
    intros e f He Hf.
    calc e
      == (e + f) { Hf }
      == f { He }.
  Qed.

  Theorem zero_unique_l : ∀ e : M, (∀ a, e + a == a) → e == 0.
  Proof.
    intros e He.
    calc e
      == (e + 0) { right_identity }
      == (0 : M) { He }.
  Qed.

  Theorem zero_unique_r : ∀ e : M, (∀ a, a + e == a) → e == 0.
  Proof.
    intros e He.
    calc e
      == (0 + e) { left_identity }
      == (0 : M) { He }.
  Qed.

A summand known to be the identity drops out.
  Theorem add_left_zero : ∀ z a : M, z == 0 → z + a == a.
  Proof.
    intros z a Hz.
    calc (z + a)
      == (0 + a) { Hz }
      == a { left_identity }.
  Qed.

  Theorem add_right_zero : ∀ a z : M, z == 0 → a + z == a.
  Proof.
    intros a z Hz.
    calc (a + z)
      == (a + 0) { Hz }
      == a { right_identity }.
  Qed.

End MonoidTheory.

Section CommMonoidTheory.
  Context {M : Type} `{Monoid M} `{!Commutative M M add}.

  Theorem add_left_comm : ∀ a b c : M, a + (b + c) == b + (a + c).
  Proof.
    intros a b c.
    calc (a + (b + c))
      == ((a + b) + c) { associative }
      == ((b + a) + c) { (commutative a b) }
      == (b + (a + c)) { associative }.
  Qed.

  Theorem add_right_comm : ∀ a b c : M, (a + b) + c == (a + c) + b.
  Proof.
    intros a b c.
    calc ((a + b) + c)
      == (a + (b + c)) { associative }
      == (a + (c + b)) { (commutative b c) }
      == ((a + c) + b) { associative }.
  Qed.

  Theorem add_add_add_comm : ∀ a b c d : M,
    (a + b) + (c + d) == (a + c) + (b + d).
  Proof.
    intros a b c d.
    calc ((a + b) + (c + d))
      == (a + (b + (c + d))) { associative }
      == (a + (c + (b + d))) { (add_left_comm b c d) }
      == ((a + c) + (b + d)) { associative }.
  Qed.

End CommMonoidTheory.
