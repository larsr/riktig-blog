From Riktig.Interface Require Export Group.

An abelian group is a group whose operation is commutative.
Module AbGroup.
  Class AbGroup (A : Type) {eq : EqRel A} {zero : Zero A} {add : HasAdd A}
                {neg : HasNeg A} := {
    group :: Group A;
    is_commutative :: Commutative A A add;
  }.
  Hint Mode AbGroup ! - - - - : typeclass_instances.
  Hint Rewrite @commutative : comm_db.
End AbGroup.
Export AbGroup.
