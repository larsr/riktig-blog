From Riktig Require Export Ops Props Interface.Group.

Module AbGroup.
  Class AbGroup (G : Type) {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G} := {
    equivalence : Equivalence (@equ G _);
    associative : ∀ x y w : G, (x + y) + w == x + (y + w);
    left_identity : ∀ x : G, 0 + x == x;
    right_identity : ∀ x : G, x + 0 == x;
    left_inverse : ∀ x : G, (- x) + x == 0;
    right_inverse : ∀ x : G, x + (- x) == 0;
    commutative : ∀ x y : G, x + y == y + x;
    add_proper : ∀ x y : G, x == y → ∀ u v : G, u == v → (x + u) == (y + v);
    neg_proper : ∀ x y : G, x == y → (- x) == (- y)
  }.
End AbGroup.
Notation AbGroup := AbGroup.AbGroup.

#[global] Instance AbGroup_is_Group G {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G}
  `{H : AbGroup.AbGroup G} : Group G := {|
    Group.equivalence := AbGroup.equivalence;
    Group.associative := AbGroup.associative;
    Group.left_identity := AbGroup.left_identity;
    Group.right_identity := AbGroup.right_identity;
    Group.left_inverse := AbGroup.left_inverse;
    Group.right_inverse := AbGroup.right_inverse;
    Group.add_proper := AbGroup.add_proper;
    Group.neg_proper := AbGroup.neg_proper
  |}.

#[global] Instance AbGroup_Commutative G {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G}
  `{H : AbGroup.AbGroup G} : Commutes G add := AbGroup.commutative.
