Library Riktig.Structure.BinaryArith

From Riktig.Structure Require Export BinaryNumber.

Arithmetic on the binary numbers by its equations, bit by bit: one is the one bit on zero; the successor carries a one bit to the left; addition reads the bits of both summands, adding one where two one bits meet; multiplication doubles and adds. The zero-bit equations hold for nonzero numbers only, since a zero bit on zero is zero again.
Module BinaryArith.
  Class BinaryArith (N : Type) {eq : EqRel N} (zero : Zero N) (bit0 : HasBit0 N) (bit1 : HasBit1 N)
      (succ : HasSucc N) (one : One N) (add : HasAdd N) (mult : HasMult N) := {
    binary :: BinaryNumber N 0 bit0 bit1;
    succ_proper :: Proper (equ ==> equ) succ;
    add_proper :: Proper (equ ==> equ ==> equ) add;
    mul_proper :: Proper (equ ==> equ ==> equ) mult;
    one_bit1 : (1 : N) == bit1 0;
    succ_zero : succ 0 == bit1 0;
    succ_bit0 : ∀ n : N, n =/= 0 → succ (bit0 n) == bit1 n;
    succ_bit1 : ∀ n : N, succ (bit1 n) == bit0 (succ n);
    add_zero_l : ∀ y : N, 0 + y == y;
    add_bit0_zero : ∀ n : N, n =/= 0 → bit0 n + 0 == bit0 n;
    add_bit0_bit0 : ∀ n m : N, n =/= 0 → m =/= 0 → bit0 n + bit0 m == bit0 (n + m);
    add_bit0_bit1 : ∀ n m : N, n =/= 0 → bit0 n + bit1 m == bit1 (n + m);
    add_bit1_zero : ∀ n : N, bit1 n + 0 == bit1 n;
    add_bit1_bit0 : ∀ n m : N, m =/= 0 → bit1 n + bit0 m == bit1 (n + m);
    add_bit1_bit1 : ∀ n m : N, bit1 n + bit1 m == bit0 (succ (n + m));
    mul_zero_l : ∀ y : N, 0 × y == 0;
    mul_bit0 : ∀ n y : N, n =/= 0 → bit0 n × y == bit0 (n × y);
    mul_bit1 : ∀ n y : N, bit1 n × y == bit0 (n × y) + y;
  }.
End BinaryArith.
