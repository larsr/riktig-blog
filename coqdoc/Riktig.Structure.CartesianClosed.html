From Riktig.Structure Require Export Terminal Exponentials.

A cartesian closed category has a terminal object, binary products and exponentials.
Module CartesianClosed.
  Class CartesianClosed (O : Type) `{Category O}
      (terminal : HasTerm O) (to_terminal : HasToTerm O)
      (product : HasProd O) (first : HasPi1 O) (second : HasPi2 O) (lifting : HasLift O)
      (exponential : HasExp O) (evaluation : HasEval O) (currying : HasCurry O) := {
    has_terminal :: Terminal O terminal to_terminal;
    has_products :: Products O product first second lifting;
    has_exponentials :: Exponentials O exponential evaluation currying;
  }.
  Hint Mode CartesianClosed ! - - - - - - - - - - - - - - : typeclass_instances.
End CartesianClosed.
Export CartesianClosed.
