The geometric algebra of the zero space is the field itself: the one vector is 0, and the grade involution is the identity.
Section ScalarGA.
  Context {F : Type} `{Field F}.

  #[export] Instance Scalar_Vec : HasVec unit F := fun _ ⇒ 0.
  #[export] Instance Scalar_Involute : HasInvolute F | 100 := fun a ⇒ a.

  #[export] Instance Scalar_Vec_Linear : Linear F unit F Scalar_Vec :=
    {| Linear.proper := fun _ _ _ ⇒ reflexivity 0;
       Linear.is_additive := fun _ _ ⇒ symmetry (left_identity (action := add) (0 : F));
       Linear.is_homogeneous := fun a _ ⇒ symmetry (mul_zero a) |}.

  #[export] Instance Scalar_GeometricAlgebra : GeometricAlgebra F unit F Scalar_Vec.
  Proof.
    split. exact Scalar_Vec_Linear.
    intros u.
    calc ((0 : F) × 0)
      == (0 : F) { (zero_mul (0 : F)) }
      == ((0 : F) × 1) { (zero_mul (1 : F)) }.
  Qed.

  #[export] Instance Scalar_GradeInvolution : GradeInvolution F unit F Scalar_Involute.
  Proof.
    split.
    - intros a b Hab. exact Hab.
    - intros a b. reflexivity.
    - intros a b. reflexivity.
    - intros a b. reflexivity.
    - reflexivity.
    - intros a. reflexivity.
    - intros u. symmetry. exact neg_zero.
  Qed.
End ScalarGA.
