From Riktig.Programs Require Import NaturalNumbers.

Successor as a channel in the failing category: encoding never fails, decoding peels the level off again, and zero -- which is never sent -- decodes to the error.
Section SuccChannel.
  Context {O : Type} `{Bicartesian O} {N : O}
    {nb : HasBuild O (NaturalNumbers.shape O) N} {nf : HasCata O (NaturalNumbers.shape O) N}
    `{!NaturalNumbers O N nb nf}.

  Let succ_enc : N ~> N ⨿ term := inj1 ∘ NaturalNumbers.succ.
  Let succ_dec : N ~> N ⨿ term := pred.

  Let succ_law : desc succ_dec inj2 ∘ succ_enc == (inj1 : N ~> N ⨿ term).
  Proof.
    unfold succ_enc, succ_dec, pred.
    set (U := NaturalNumbers.fold (NaturalNumbers.shape O N) (NaturalNumbers.map build)).
    assert (HU : U ∘ build == cat_id (x := NaturalNumbers.shape O N)).
    { unfold U. exact (unbuild_build (F := NaturalNumbers.shape O) (t := N)). }
    etransitivity; [ apply (Category.comp_assoc (O := O)) | ].
    etransitivity; [ apply Category.comp_proper;
      [ apply (Coproducts.desc_inj1 (O := O)) | reflexivity ] | ].
    etransitivity; [ symmetry; apply (Category.comp_assoc (O := O)) | ].
    etransitivity; [ apply Category.comp_proper;
      [ reflexivity | apply (Category.comp_assoc (O := O)) ] | ].
    etransitivity; [ apply Category.comp_proper;
      [ reflexivity | apply Category.comp_proper; [ exact HU | reflexivity ] ] | ].
    etransitivity; [ apply Category.comp_proper;
      [ reflexivity | apply left_identity ] | ].
    apply (Coproducts.desc_inj2 (O := O)).
  Qed.

  #[export] Instance succ_channel :
    Channel (Err O term) (hom_equ := Err_EqRel term) N N succ_enc succ_dec.
  Proof. split. exact succ_law. Qed.
End SuccChannel.
