From Riktig Require Export Props.

The lambda terms as a grammar writes them, with recursion: a variable is a name, a binder carries the name it binds, a numeral is a number, and an application is two terms. From one answer per constructor, a function on terms, given by its four equations, which respects equality when the answers do. Names have no meaning here; giving them one is the change to de Bruijn terms (Programs/LambdaNamed).
Module NamedTermRec.
  Class NamedTermRec (K N T : Type) {eqK : EqRel K} {eqN : EqRel N} {eqT : EqRel T}
      (nvarT : HasNamedVar K T) (numT : HasNum N T) (nlamT : HasNamedLam K T) (appT : HasApp T)
      (rec : HasNamedTermRec K N T) := {
    ntermrec_proper : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : K → B) (c : N → B) (l : K → B → B) (a : B → B → B),
      Proper (equ ==> equ) v → Proper (equ ==> equ) c →
      Proper (equ ==> equ ==> equ) l → Proper (equ ==> equ ==> equ) a →
      Proper (equ ==> equ) (ntermrec B v c l a : T → B);
    ntermrec_var : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : K → B) (c : N → B) (l : K → B → B) (a : B → B → B) (x : K),
      ntermrec B v c l a (nvar x : T) == v x;
    ntermrec_num : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : K → B) (c : N → B) (l : K → B → B) (a : B → B → B) (n : N),
      ntermrec B v c l a (num n : T) == c n;
    ntermrec_lam : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : K → B) (c : N → B) (l : K → B → B) (a : B → B → B) (x : K) (b : T),
      ntermrec B v c l a (nlam x b) == l x (ntermrec B v c l a b);
    ntermrec_app : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : K → B) (c : N → B) (l : K → B → B) (a : B → B → B) (f u : T),
      ntermrec B v c l a (app f u) == a (ntermrec B v c l a f) (ntermrec B v c l a u);
  }.
End NamedTermRec.
