From Coq Require Import Morphisms Setoid.
From Riktig Require Export Theory.HomVS Structure.VectObj.

Module AdmissibleArrows.
The arrows a derivative is allowed to be: a sub-collection of the linear maps closed under the vector-space operations and under composition.
  Class AdmissibleArrows (F : Type) {Fe : EqRel F} {Fz : Zero F}
      {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F}
      {Fi : HasInv F} {FF : Field F}
      (Ob : Type) (car : Ob → VectObj F)
      (admissible : ∀ X Y : Ob,
         LinearMap F (vo_car (car X)) (vo_car (car Y)) → Prop) := {
    admissible_zero : ∀ X Y : Ob,
      admissible X Y (0 : LinearMap F (vo_car (car X)) (vo_car (car Y)));
    admissible_identity : ∀ X : Ob, admissible X X linear_map_id;
    admissible_sum : ∀ (X Y : Ob)
        (f g : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      admissible X Y f → admissible X Y g → admissible X Y (f + g);
    admissible_scaled : ∀ (X Y : Ob) (c : F)
        (f : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      admissible X Y f → admissible X Y (c *: f);
    admissible_negated : ∀ (X Y : Ob)
        (f : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      admissible X Y f → admissible X Y (- f);
    admissible_composite : ∀ (X Y Z : Ob)
        (g : LinearMap F (vo_car (car Y)) (vo_car (car Z)))
        (f : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      admissible Y Z g → admissible X Y f →
      admissible X Z (linear_map_comp g f);
  }.
  Hint Mode AdmissibleArrows ! - - - - - - - - ! ! ! : typeclass_instances.
End AdmissibleArrows.
Export AdmissibleArrows.

Module DerivSpec.
A derivative operator assigns to a function, a point and a candidate arrow the proposition that the arrow is the function's derivative there. The laws are the eight facts Theory/Derivative.v's consumers actually cite.
  Class DerivSpec (F : Type) {Fe : EqRel F} {Fz : Zero F}
      {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F}
      {Fi : HasInv F} {FF : Field F}
      (Ob : Type) (car : Ob → VectObj F) := {
    admissible : ∀ X Y : Ob,
      LinearMap F (vo_car (car X)) (vo_car (car Y)) → Prop;
    deriv_arrows :: AdmissibleArrows F Ob car admissible;

    differential : ∀ X Y : Ob,
      (vo_car (car X) → vo_car (car Y)) → vo_car (car X) →
      LinearMap F (vo_car (car X)) (vo_car (car Y)) → Prop;

    deriv_of_linear : ∀ (X Y : Ob)
        (g : LinearMap F (vo_car (car X)) (vo_car (car Y)))
        (a : vo_car (car X)),
      admissible X Y g → differential X Y (LinearMap.map g) a g;
    deriv_of_constant : ∀ (X Y : Ob) (v : vo_car (car Y))
        (a : vo_car (car X)),
      differential X Y (fun _ : vo_car (car X) ⇒ v) a 0;
    deriv_of_sum : ∀ (X Y : Ob)
        (f g : vo_car (car X) → vo_car (car Y)) (a : vo_car (car X))
        (Lf Lg : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      differential X Y f a Lf → differential X Y g a Lg →
      differential X Y (fun u : vo_car (car X) ⇒ f u + g u) a (Lf + Lg);
    deriv_of_scaled : ∀ (X Y : Ob) (c : F)
        (f : vo_car (car X) → vo_car (car Y)) (a : vo_car (car X))
        (Lf : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      differential X Y f a Lf →
      differential X Y (fun u : vo_car (car X) ⇒ c *: f u) a (c *: Lf);
    deriv_of_composite : ∀ (X Y Z : Ob)
        (f : vo_car (car X) → vo_car (car Y))
        (g : vo_car (car Y) → vo_car (car Z)) (a : vo_car (car X))
        (Lf : LinearMap F (vo_car (car X)) (vo_car (car Y)))
        (Lg : LinearMap F (vo_car (car Y)) (vo_car (car Z))),
      Proper (equ ==> equ) g →
      differential X Y f a Lf → differential Y Z g (f a) Lg →
      differential X Z (fun u : vo_car (car X) ⇒ g (f u)) a
        (linear_map_comp Lg Lf);

    deriv_respects_arrow : ∀ (X Y : Ob)
        (f : vo_car (car X) → vo_car (car Y)) (a : vo_car (car X))
        (L1 L2 : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      L1 == L2 → differential X Y f a L1 → differential X Y f a L2;
    deriv_respects_function : ∀ (X Y : Ob)
        (f g : vo_car (car X) → vo_car (car Y)) (a : vo_car (car X))
        (L : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      (∀ x : vo_car (car X), f x == g x) →
      differential X Y f a L → differential X Y g a L;
    differential_unique : ∀ (X Y : Ob)
        (f : vo_car (car X) → vo_car (car Y)) (a : vo_car (car X))
        (L1 L2 : LinearMap F (vo_car (car X)) (vo_car (car Y))),
      differential X Y f a L1 → differential X Y f a L2 → L1 == L2;
  }.
  Hint Mode DerivSpec ! - - - - - - - - ! ! : typeclass_instances.
End DerivSpec.
Export DerivSpec.
