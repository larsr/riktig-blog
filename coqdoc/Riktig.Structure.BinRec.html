Library Riktig.Structure.BinRec

From Riktig.Structure Require Export BinaryNumber.

Recursion on the binary numbers: from a value at zero and a step for each appended bit, a function on the numbers, given by its equations -- the zero-bit equation for nonzero numbers only, since a zero bit on zero is zero again -- which uses its data only through their values when the steps respect the accumulator, and respects equality when the steps do.
Module BinRec.
  Class BinRec (N : Type) `{BinaryNumber N} (binrec : HasBinRec N) := {
    binrec_proper : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f0 f1 : N → A → A),
      Proper (equ ==> equ ==> equ) f0 → Proper (equ ==> equ ==> equ) f1 →
      Proper (equ ==> equ) (binrec A a f0 f1);
    binrec_ext : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a a' : A) (f0 f0' f1 f1' : N → A → A),
      (∀ k : N, Proper (equ ==> equ) (f0 k)) → (∀ k : N, Proper (equ ==> equ) (f1 k)) →
      a == a' → (∀ (k : N) (x : A), f0 k x == f0' k x) → (∀ (k : N) (x : A), f1 k x == f1' k x) →
      ∀ n : N, binrec A a f0 f1 n == binrec A a' f0' f1' n;
    binrec_zero : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f0 f1 : N → A → A),
      binrec A a f0 f1 0 == a;
    binrec_bit0 : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f0 f1 : N → A → A) (n : N), n =/= 0 →
      binrec A a f0 f1 (bit0 n) == f0 n (binrec A a f0 f1 n);
    binrec_bit1 : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f0 f1 : N → A → A) (n : N),
      binrec A a f0 f1 (bit1 n) == f1 n (binrec A a f0 f1 n);
  }.
End BinRec.
