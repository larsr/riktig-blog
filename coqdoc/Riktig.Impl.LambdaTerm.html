From Coq Require Import List.
From Riktig Require Export Structure.Lambda Impl.PeanoArithNat Impl.NaturalsPeanoArith Lib.Coq.Lambda.

The de Bruijn terms are a lambda calculus: reduction is any number of beta steps anywhere, conversion is joinability, and confluence is Takahashi's theorem.
Instance Term_EqRel : EqRel term := joinable.
Instance Term_HasVar : HasVar nat term := Var.
Instance Term_HasLam : HasLam term := Lam.
Instance Term_HasApp : HasApp term := App.
Instance Term_HasSubst : HasSubst term sub := fun t s ⇒ psub s t.
Instance Sub_HasSubId : HasSubId sub := Var.
Instance Sub_HasSubShift : HasSubShift sub := shift.
Instance Sub_HasSubCons : HasSubCons term sub := fun u s ⇒ scons u s.
Instance Sub_HasSubUp : HasSubUp sub := up.
Instance Term_HasReduces : HasReduces term := bsteps.

Instance Term_LambdaCalculus : LambdaCalculus nat term sub
    Term_HasVar Term_HasLam Term_HasApp Term_HasSubst Sub_HasSubId Sub_HasSubShift Sub_HasSubCons Sub_HasSubUp
    Term_HasReduces.
Proof.
  split.
  - intros t u. split; intros Hj; exact Hj.
  - intros t. apply star_refl.
  - intros t u v. apply star_trans.
  - intros b b'. apply bsteps_lam.
  - intros f f' a a'. apply bsteps_app.
  - intros t t' s Ht. exact (bsteps_psub s t t' Ht).
  - intros b u. apply star_one. exact (bstep_beta b u).
  - intros n m [w [H1 H2]].
    apply bsteps_var_inv in H1. apply bsteps_var_inv in H2. subst.
    injection H2 as H2. subst. reflexivity.
  - intros t. apply eq_joinable. apply psub_var.
  - intros u s. apply joinable_refl.
  - intros n u s. apply joinable_refl.
  - intros n. apply joinable_refl.
  - intros s. apply joinable_refl.
  - intros n s. apply eq_joinable. simpl. apply ren_as_psub.
  - intros f a s. apply joinable_refl.
  - intros b s. apply joinable_refl.
  - intros t u s. apply eq_joinable. unfold substitute, Term_HasSubst.
    rewrite psub_psub. apply psub_ext. intros n. reflexivity.
  - intros t s. apply eq_joinable. unfold substitute, Term_HasSubst.
    rewrite !psub_psub. apply psub_ext. intros n. simpl. apply ren_as_psub.
  - intros t u v. apply bsteps_confluent.
Qed.
