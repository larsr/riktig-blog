From Coq Require Import List.
From Riktig Require Export Structure.Lambda Impl.PeanoArithNat Impl.NaturalsPeanoArith Lib.Coq.Lambda.

The de Bruijn terms with parallel substitution are a lambda calculus: terms are compared up to conversion, substitutions are functions from indices to terms.
Instance Term_EqRel : EqRel term := conv.
Instance Term_HasVar : HasVar nat term := Var.
Instance Term_HasLam : HasLam term := Lam.
Instance Term_HasApp : HasApp term := App.
Instance Term_HasSubst : HasSubst term sub := fun t s ⇒ psub s t.
Instance Sub_HasSubId : HasSubId sub := Var.
Instance Sub_HasSubShift : HasSubShift sub := shift.
Instance Sub_HasSubCons : HasSubCons term sub := fun u s ⇒ scons u s.
Instance Sub_HasSubUp : HasSubUp sub := up.

Instance Term_LambdaCalculus : LambdaCalculus nat term sub
    Term_HasVar Term_HasLam Term_HasApp Term_HasSubst Sub_HasSubId Sub_HasSubShift Sub_HasSubCons Sub_HasSubUp.
Proof.
  split.
  - split; [ exact conv_refl | exact conv_sym | exact conv_trans ].
  - intros n n' Hn. unfold equ, Nat_EqRel in Hn. subst. apply conv_refl.
  - intros b b' Hb. apply conv_lam. exact Hb.
  - intros f f' Hf a a' Ha. apply conv_app; assumption.
  - intros t t' s Ht. apply conv_psub. exact Ht.
  - intros t. unfold substitute, Term_HasSubst, sub_id, Sub_HasSubId. rewrite psub_var. apply conv_refl.
  - intros u s. apply conv_refl.
  - intros n u s. apply conv_refl.
  - intros n. apply conv_refl.
  - intros s. apply conv_refl.
  - intros n s. unfold substitute, Term_HasSubst, sub_up, Sub_HasSubUp, sub_shift, Sub_HasSubShift. simpl.
    rewrite ren_as_psub. apply conv_refl.
  - intros f a s. apply conv_refl.
  - intros b s. apply conv_refl.
  - intros t u s. unfold substitute, Term_HasSubst, sub_cons, Sub_HasSubCons, sub_shift, Sub_HasSubShift.
    rewrite psub_psub. apply conv_refl.
  - intros t s. unfold substitute, Term_HasSubst, sub_up, Sub_HasSubUp, sub_shift, Sub_HasSubShift.
    rewrite !psub_psub. simpl. rewrite (psub_ext t (fun n ⇒ ren S (s n)) (fun n ⇒ psub shift (s n))).
    + apply conv_refl.
    + intros n. apply ren_as_psub.
  - intros b u. apply conv_beta.
Qed.
