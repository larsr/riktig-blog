Section AbstractVectors.
  Context {F V : Type} `{VV : VectorSpace F V}.

  Lemma scale_regroup : ∀ (a b : F) (u v : V),
    a *: (u + - v) + b *: v == a *: u + (b + - a) *: v.
  Proof.
    intros a b u v.
    calc (a *: (u + - v) + b *: v)
      == ((a *: u + - (a *: v)) + b *: v) { scale_sub }
      == ((a *: u + b *: v) + - (a *: v)) { add_swap_r }
      == (a *: u + (b *: v + - (a *: v))) { associative }
      == (a *: u + (b + - a) *: v) { sub_scale }.
  Qed.

End AbstractVectors.

Section AbstractOrthogonality.
  Context {F V : Type} `{IP : InnerProductSpace F V}.

  Lemma orthogonal_neg_r : ∀ u v : V,
    ⟪ u , v ⟫ == 0 → orthogonal u (- v).
  Proof.
    intros u v Huv.
    apply orthogonal_def.
    calc (⟪ u , - v ⟫)
      == (- ⟪ u , v ⟫) { inner_right_neg }
      == (- (0 : F)) { Huv }
      == (0 : F) { neg_zero }.
  Qed.

  Theorem norm2_orthogonal_diagonals : ∀ u v : V,
    ⟪ u , v ⟫ == 0 → norm2 (u + - v) == norm2 (u + v).
  Proof.
    intros u v Huv.
    calc (norm2 (u + - v))
      == (norm2 u + norm2 (- v))
           { (pythagoras u (- v) (orthogonal_neg_r u v Huv)) }
      == (norm2 u + norm2 v)
           { norm2_neg }
      == (norm2 (u + v))
           { (pythagoras u v (orthogonal_def u v Huv)) }.
  Qed.

End AbstractOrthogonality.

Section AbstractCoordinates.
  Context {n : nat} {I F V : Type} `{B : Basis n I F V}.
  Context {bigop_list : ∀ J : Type, (J → V) → list J → V}
          {BL : BigOp V bigop_list}.
  Context {VlL : Lincomb n I F V bigop_list}.

  Theorem coords_determine : ∀ u v : V,
    (∀ i : I, Basis.coords B u i == Basis.coords B v i) → u == v.
  Proof.
    intros u v Hc.
    calc u
      == (lincomb (Basis.vec B) (Basis.coords B u))
           { (Basis.reconstruction B u) }
      == (lincomb (Basis.vec B) (Basis.coords B v))
           { (lincomb_proper (Basis.vec B) (Basis.coords B u)
                (Basis.coords B v) Hc) }
      == v
           { (Basis.reconstruction B v) }.
  Qed.

End AbstractCoordinates.

Section Generic.
  Context {G : Type} `{AbGroup G}.

  Theorem swap_cancel : ∀ a b c : G, (a + b) + (c + - a) == b + c.
  Proof.
    intros a b c.
    rewrite (commutative a b), (associative (op := add)), (add_swap_l a c (- a)), right_inverse, right_identity.
    reflexivity.
  Qed.
End Generic.

Example z_swap_cancel : (3 + 5 + (7 + - 3))%Z = (5 + 7)%Z.
Proof. exact (swap_cancel (G := Z) 3%Z 5%Z 7%Z). Qed.


Section Generic.
  Context {F U V : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Theorem lm_combination : ∀ (f : LinearMap F U V) (a b : F) (u v : U),
    f (a *: u + b *: v) == a *: f u + b *: f v.
  Proof.
    intros f a b u v.
    rewrite (linear_map_additive f), (linear_map_scale f), (linear_map_scale f).
    reflexivity.
  Qed.
End Generic.

Section ScaleMap.
  Context {F V : Type} `{VV : VectorSpace F V} (c : F).

  #[export] Instance scale_map_proper : Proper (equ ==> equ) (fun v : V ⇒ c *: v).
  Proof.
    intros u v Huv.
    rewrite Huv.
    reflexivity.
  Qed.

  Lemma scale_map_add : ∀ u v : V, c *: (u + v) == c *: u + c *: v.
  Proof.
    intros u v.
    exact (left_distributive c u v).
  Qed.

  Lemma scale_map_scale : ∀ (a : F) (u : V),
    c *: (a *: u) == a *: (c *: u).
  Proof.
    intros a u.
    rewrite <- (associative c a u), (mul_commutative c a),
      (associative a c u).
    reflexivity.
  Qed.

  Definition scale_map : LinearMap F V V :=
    linear_map_from_laws
      (fun v : V ⇒ c *: v)
      scale_map_proper
      scale_map_add
      scale_map_scale.
End ScaleMap.

Definition triple : LinearMap Q (Fin.t 2 → Q) (Fin.t 2 → Q) :=
  scale_map (3#1)%Q.

Example fin2_combo : ∀ u v : Fin.t 2 → Q,
  triple (2 *: u + 5 *: v) == 2 *: triple u + 5 *: triple v.
Proof. intros u v. exact (lm_combination triple (2#1)%Q (5#1)%Q u v). Qed.


Section Generic.
  Context {O : Type} `{Category O}.

  Theorem iso_absorb : ∀ (x y : O) (f : Iso x y),
    (iso_from f ∘ iso_to f) ∘ iso_from f == iso_from f.
  Proof.
    intros x y f.
    rewrite (iso_fromto f), (left_identity (iso_from f)).
    reflexivity.
  Qed.
End Generic.
