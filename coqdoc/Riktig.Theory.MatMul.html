mat_mul A B is the matrix product contracting the middle index.

Section MatMulAssoc.
  Context {F : Type} `{FR : Ring F}.
  Context {bigop_list : ∀ J' : Type, (J' → F) → list J' → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J'}.
  Context {nJ : nat} {J : Type} {eJ : EqRel J} {dJ : EquDec J}
    {FJ : FinIndex nJ J}
    {mat_mul : ∀ I K : Type, (I → J → F) → (J → K → F) → I → K → F}
    {MMJ : MatMul nJ J F (@bigop_list) mat_mul}.
  Arguments mat_mul {I K}.
  Context {nK : nat} {K : Type} {eK : EqRel K} {dK : EquDec K}
    {FK : FinIndex nK K}
    {mat_mulK : ∀ I L : Type, (I → K → F) → (K → L → F) → I → L → F}
    {MMK : MatMul nK K F (@bigop_list) mat_mulK}.
  Arguments mat_mulK {I L}.
  Context {I L : Type}.

  Theorem mat_mul_assoc : ∀ (A : I → J → F) (B : J → K → F)
      (C : K → L → F),
    
    mat_mulK (mat_mul A B) C == mat_mul A (mat_mulK B C).
  Proof.
    intros A B C i l.
    calc (mat_mulK (mat_mul A B) C i l)
      == (bigop_list (fun k ⇒ mat_mul A B i k × C k l) enum)
           { (mat_mul_def (mat_mul A B) C i l) }
      == (bigop_list (fun k ⇒ bigop_list (fun j ⇒ A i j × B j k) enum × C k l) enum)
           { (bigopI_ext _ _ (fun k ⇒ mul_r_proper (C k l) _ _
                (mat_mul_def A B i k))) }
      == (bigop_list (fun k ⇒ bigop_list (fun j ⇒ A i j × (B j k × C k l)) enum) enum)
           { (bigopI_ext _ _ (fun k ⇒
                transitivity
                  (symmetry (bigopI_mul_r (C k l) (fun j ⇒ A i j × B j k)))
                  (bigopI_ext _ _
                    (fun j ⇒ associative (A i j) (B j k) (C k l))))) }
      == (bigop_list (fun j ⇒ bigop_list (fun k ⇒ A i j × (B j k × C k l)) enum) enum)
           { (bigopI_swap (fun k j ⇒ A i j × (B j k × C k l))) }
      == (bigop_list (fun j ⇒ A i j × bigop_list (fun k ⇒ B j k × C k l) enum) enum)
           { (bigopI_ext _ _ (fun j ⇒
                bigopI_mul_l (A i j) (fun k ⇒ B j k × C k l))) }
      == (bigop_list (fun j ⇒ A i j × mat_mulK B C j l) enum)
           { (bigopI_ext _ _ (fun j ⇒ mul_l_proper (A i j) _ _
                (symmetry (mat_mul_def B C j l)))) }
      == (mat_mul A (mat_mulK B C) i l)
           { (symmetry (mat_mul_def A (mat_mulK B C) i l)) }.
  Qed.

End MatMulAssoc.

Section MatMulRingLaws.
  Context {F : Type} `{FR : Ring F}.
  Context {bigop_list : ∀ J' : Type, (J' → F) → list J' → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J'}.
  Context {nJ : nat} {J : Type} {eJ : EqRel J} {dJ : EquDec J}
    {FJ : FinIndex nJ J}
    {mat_mul : ∀ I K : Type, (I → J → F) → (J → K → F) → I → K → F}
    {MMJ : MatMul nJ J F (@bigop_list) mat_mul}.
  Arguments mat_mul {I K}.
  Context {I K : Type}.

  #[export] Instance mat_mul_proper :
    Proper (equ ==> equ ==> equ) (mat_mul (I := I) (K := K)).
  Proof.
    intros A A' HA B B' HB i k.
    calc (mat_mul A B i k)
      == (bigop_list (fun j ⇒ A i j × B j k) enum) { (mat_mul_def A B i k) }
      == (bigop_list (fun j ⇒ A' i j × B' j k) enum)
           { (bigopI_ext _ _ (fun j ⇒ op_proper (A i j) (A' i j) (HA i j)
                (B j k) (B' j k) (HB j k))) }
      == (mat_mul A' B' i k) { (mat_mul_def A' B' i k) }.
  Qed.

  Theorem mat_mul_add_l : ∀ (A A' : I → J → F) (B : J → K → F),
    mat_mul (A + A') B == mat_mul A B + mat_mul A' B.
  Proof.
    intros A A' B i k.
    calc (mat_mul (A + A') B i k)
      == (bigop_list (fun j ⇒ (A i j + A' i j) × B j k) enum)
           { (mat_mul_def (A + A') B i k) }
      == (bigop_list (fun j ⇒ A i j × B j k + A' i j × B j k) enum)
           { (bigopI_ext _ _ (fun j ⇒
                right_distributive (A i j) (A' i j) (B j k))) }
      == (bigop_list (fun j ⇒ A i j × B j k) enum + bigop_list (fun j ⇒ A' i j × B j k) enum)
           { (bigopI_add (fun j ⇒ A i j × B j k) (fun j ⇒ A' i j × B j k)) }
      == (mat_mul A B i k + bigop_list (fun j ⇒ A' i j × B j k) enum)
           { (symmetry (mat_mul_def A B i k)) }
      == ((mat_mul A B + mat_mul A' B) i k)
           { (symmetry (mat_mul_def A' B i k)) }.
  Qed.

  Theorem mat_mul_add_r : ∀ (A : I → J → F) (B B' : J → K → F),
    mat_mul A (B + B') == mat_mul A B + mat_mul A B'.
  Proof.
    intros A B B' i k.
    calc (mat_mul A (B + B') i k)
      == (bigop_list (fun j ⇒ A i j × (B j k + B' j k)) enum)
           { (mat_mul_def A (B + B') i k) }
      == (bigop_list (fun j ⇒ A i j × B j k + A i j × B' j k) enum)
           { (bigopI_ext _ _ (fun j ⇒
                left_distributive (A i j) (B j k) (B' j k))) }
      == (bigop_list (fun j ⇒ A i j × B j k) enum + bigop_list (fun j ⇒ A i j × B' j k) enum)
           { (bigopI_add (fun j ⇒ A i j × B j k) (fun j ⇒ A i j × B' j k)) }
      == (mat_mul A B i k + bigop_list (fun j ⇒ A i j × B' j k) enum)
           { (symmetry (mat_mul_def A B i k)) }
      == ((mat_mul A B + mat_mul A B') i k)
           { (symmetry (mat_mul_def A B' i k)) }.
  Qed.

End MatMulRingLaws.

Section MatMulIdentity.
  Context {F : Type} `{FR : Ring F}.
  Context {bigop_list : ∀ J' : Type, (J' → F) → list J' → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J'}.
  Context {nJ : nat} {J : Type} {eJ : EqRel J} {dJ : EquDec J}
    {FJ : FinIndex nJ J}
    {mat_mul : ∀ I K : Type, (I → J → F) → (J → K → F) → I → K → F}
    {MMJ : MatMul nJ J F (@bigop_list) mat_mul}
    {delta : J → J → F} {FdL : Delta J F delta}.
  Arguments mat_mul {I K}.
  Context {I K : Type}.

  Theorem mat_mul_left_identity : ∀ A : J → K → F,
    (∀ k : K, Proper (equ ==> equ) (fun j : J ⇒ A j k)) →
    mat_mul delta A == A.
  Proof.
    intros A PA j k.
    calc (mat_mul delta A j k)
      == (bigop_list (fun j' ⇒ delta j j' × A j' k) enum)
           { (mat_mul_def delta A j k) }
      == (A j k)
           { (bigopI_delta_l (fun j' ⇒ A j' k) j (PA k)) }.
  Qed.

  Theorem mat_mul_right_identity : ∀ A : I → J → F,
    (∀ i : I, Proper (equ ==> equ) (A i)) →
    mat_mul A delta == A.
  Proof.
    intros A PA i j.
    calc (mat_mul A delta i j)
      == (bigop_list (fun j' ⇒ A i j' × delta j' j) enum)
           { (mat_mul_def A delta i j) }
      == (A i j)
           { (bigopI_delta (fun j' ⇒ A i j') j (PA i)) }.
  Qed.

End MatMulIdentity.

Section MatMulScale.
  Context {F : Type} `{FF : Field F}.
  Context {bigop_list : ∀ J' : Type, (J' → F) → list J' → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J'}.
  Context {nJ : nat} {J : Type} {eJ : EqRel J} {dJ : EquDec J}
    {FJ : FinIndex nJ J}
    {mat_mul : ∀ I K : Type, (I → J → F) → (J → K → F) → I → K → F}
    {MMJ : MatMul nJ J F (@bigop_list) mat_mul}.
  Arguments mat_mul {I K}.
  Context {I K : Type}.

  Theorem mat_mul_scale_l : ∀ (a : F) (A : I → J → F) (B : J → K → F),
    mat_mul (a *: A) B == a *: mat_mul A B.
  Proof.
    intros a A B i k.
    calc (mat_mul (a *: A) B i k)
      == (bigop_list (fun j ⇒ a × A i j × B j k) enum)
           { (mat_mul_def (a *: A) B i k) }
      == (bigop_list (fun j ⇒ a × (A i j × B j k)) enum)
           { (bigopI_ext _ _ (fun j ⇒ associative a (A i j) (B j k))) }
      == (a × bigop_list (fun j ⇒ A i j × B j k) enum)
           { (bigopI_mul_l a (fun j ⇒ A i j × B j k)) }
      == ((a *: mat_mul A B) i k)
           { (mul_l_proper a _ _ (symmetry (mat_mul_def A B i k))) }.
  Qed.

  Theorem mat_mul_scale_r : ∀ (a : F) (A : I → J → F) (B : J → K → F),
    mat_mul A (a *: B) == a *: mat_mul A B.
  Proof.
    intros a A B i k.
    calc (mat_mul A (a *: B) i k)
      == (bigop_list (fun j ⇒ A i j × (a × B j k)) enum)
           { (mat_mul_def A (a *: B) i k) }
      == (bigop_list (fun j ⇒ a × (A i j × B j k)) enum)
           { (bigopI_ext _ _ (fun j ⇒
                mul_left_commutative (A i j) a (B j k))) }
      == (a × bigop_list (fun j ⇒ A i j × B j k) enum)
           { (bigopI_mul_l a (fun j ⇒ A i j × B j k)) }
      == ((a *: mat_mul A B) i k)
           { (mul_l_proper a _ _ (symmetry (mat_mul_def A B i k))) }.
  Qed.

End MatMulScale.


Section MatMulSplit.
  Context {F : Type} `{FR : Ring F}.
  Context {bigop_list : ∀ J' : Type, (J' → F) → list J' → F}
          {BL : BigOp F bigop_list}.
  Arguments bigop_list {J'}.
  Context {n1 : nat} {J1 : Type} {e1 : EqRel J1} {d1 : EquDec J1}
    {FI1 : FinIndex n1 J1}
    {mat_mul1 : ∀ I K : Type, (I → J1 → F) → (J1 → K → F) → I → K → F}
    {MM1 : MatMul n1 J1 F (@bigop_list) mat_mul1}.
  Arguments mat_mul1 {I K}.
  Context {n2 : nat} {J2 : Type} {e2 : EqRel J2} {d2 : EquDec J2}
    {FI2 : FinIndex n2 J2}
    {mat_mul2 : ∀ I K : Type, (I → J2 → F) → (J2 → K → F) → I → K → F}
    {MM2 : MatMul n2 J2 F (@bigop_list) mat_mul2}.
  Arguments mat_mul2 {I K}.
  Context {nS : nat} {eS : EqRel (J1 + J2)} {dS : EquDec (J1 + J2)}
    {FIS : FinIndex nS (J1 + J2)}
    {mat_mulS : ∀ I K : Type, (I → (J1 + J2) → F) → ((J1 + J2) → K → F) → I → K → F}
    {MMS : MatMul nS (J1 + J2) F (@bigop_list) mat_mulS}.
  Arguments mat_mulS {I K}.
  Context {BS : BigopSum n1 n2 nS J1 J2 F (@bigop_list)}.
  Context {I K : Type}.

  Theorem mat_mul_split : ∀ (A : I → (J1 + J2) → F)
      (B : (J1 + J2) → K → F),
    
    mat_mulS A B
      == mat_mul1 (fun i j ⇒ A i (inl j)) (fun j k ⇒ B (inl j) k)
       + mat_mul2 (fun i j ⇒ A i (inr j)) (fun j k ⇒ B (inr j) k).
  Proof.
    intros A B i k.
    calc (mat_mulS A B i k)
      == (bigop_list (fun j ⇒ A i j × B j k) enum)
           { (mat_mul_def A B i k) }
      == (bigop_list (fun j ⇒ A i (inl j) × B (inl j) k) enum
          + bigop_list (fun j ⇒ A i (inr j) × B (inr j) k) enum)
           { (bigopI_sum (fun j ⇒ A i j × B j k)) }
      == (mat_mul1 (fun i' j ⇒ A i' (inl j)) (fun j k' ⇒ B (inl j) k') i k
          + bigop_list (fun j ⇒ A i (inr j) × B (inr j) k) enum)
           { (mat_mul_def (fun i' j ⇒ A i' (inl j))
                (fun j k' ⇒ B (inl j) k') i k) }
      == ((mat_mul1 (fun i' j ⇒ A i' (inl j)) (fun j k' ⇒ B (inl j) k')
           + mat_mul2 (fun i' j ⇒ A i' (inr j)) (fun j k' ⇒ B (inr j) k')) i k)
           { (mat_mul_def (fun i' j ⇒ A i' (inr j))
                (fun j k' ⇒ B (inr j) k') i k) }.
  Qed.

End MatMulSplit.
