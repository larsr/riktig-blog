From Riktig Require Export Structure.NatOrder Theory.Naturals.

Any natural numbers are ordered by "something added gives the other".
Section NatOrderNaturals.
  Context {N : Type} `{Naturals N}.

  #[export] Instance Naturals_HasLe : HasLe N := fun a b ⇒ ∃ c : N, a + c == b.

  #[export] Instance Naturals_NatOrder : NatOrder N Naturals_HasLe :=
    {| NatOrder.le_proper := le_by_add_proper;
       NatOrder.le_add := fun a b ⇒ iff_refl _ |}.
End NatOrderNaturals.
