Section Projections.
  Context {F A B : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.

  Lemma exl_proper : Proper (equ ==> equ) (fst : A × B → A).
  Proof. intros p q [Hf _]. exact Hf. Qed.

  Lemma exr_proper : Proper (equ ==> equ) (snd : A × B → B).
  Proof. intros p q [_ Hs]. exact Hs. Qed.

  Definition exl : LinearMap F (A × B) A :=
    linear_map_from_laws fst exl_proper
      (fun p q ⇒ reflexivity (fst p + fst q))
      (fun a p ⇒ reflexivity (a *: fst p)).

  Definition exr : LinearMap F (A × B) B :=
    linear_map_from_laws snd exr_proper
      (fun p q ⇒ reflexivity (snd p + snd q))
      (fun a p ⇒ reflexivity (a *: snd p)).

  Lemma inl_proper : Proper (equ ==> equ) (fun a : A ⇒ (a, (0 : B))).
  Proof. intros a a' Ha. split; [ exact Ha | reflexivity ]. Qed.

  Lemma inr_proper : Proper (equ ==> equ) (fun b : B ⇒ ((0 : A), b)).
  Proof. intros b b' Hb. split; [ reflexivity | exact Hb ]. Qed.

  Definition inl : LinearMap F A (A × B) :=
    linear_map_from_laws (fun a : A ⇒ (a, (0 : B))) inl_proper
      (fun a a' ⇒ conj (reflexivity (a + a')) (symmetry (left_identity (0 : B))))
      (fun c a ⇒ conj (reflexivity (c *: a)) (symmetry (scale_zero c))).

  Definition inr : LinearMap F B (A × B) :=
    linear_map_from_laws (fun b : B ⇒ ((0 : A), b)) inr_proper
      (fun b b' ⇒ conj (symmetry (left_identity (0 : A))) (reflexivity (b + b')))
      (fun c b ⇒ conj (symmetry (scale_zero c)) (reflexivity (c *: b))).

End Projections.

Section Pairing.
  Context {F A B X : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.
  Context {eX : EqRel X} {zX : Zero X} {aX : HasAdd X} {nX : HasNeg X} {sX : HasScale F X}.
  Context {VX : VectorSpace F X}.

  Section Fork.
    Context (f : LinearMap F X A) (g : LinearMap F X B).

    Lemma fork_proper : Proper (equ ==> equ) (fun x : X ⇒ (f x, g x)).
    Proof.
      intros x y Hxy.
      split; [ exact (linear_map_proper f _ _ Hxy) | exact (linear_map_proper g _ _ Hxy) ].
    Qed.

    Definition fork : LinearMap F X (A × B) :=
      linear_map_from_laws (fun x : X ⇒ (f x, g x)) fork_proper
        (fun x y ⇒ conj (linear_map_additive f x y) (linear_map_additive g x y))
        (fun c x ⇒ conj (linear_map_scale f c x) (linear_map_scale g c x)).
  End Fork.

  Section Join.
    Context (f : LinearMap F A X) (g : LinearMap F B X).

    Lemma join_proper : Proper (equ ==> equ) (fun p : A × B ⇒ f (fst p) + g (snd p)).
    Proof.
      intros p q [Hf Hs].
      exact (op_proper _ _ (linear_map_proper f _ _ Hf)
                       _ _ (linear_map_proper g _ _ Hs)).
    Qed.

    Lemma join_add : ∀ p q : A × B,
      f (fst (p + q)) + g (snd (p + q))
        == (f (fst p) + g (snd p)) + (f (fst q) + g (snd q)).
    Proof.
      intros p q.
      calc (f (fst p + fst q) + g (snd p + snd q))
        == ((f (fst p) + f (fst q)) + g (snd p + snd q))
             { linear_map_additive }
        == ((f (fst p) + f (fst q)) + (g (snd p) + g (snd q)))
             { linear_map_additive }
        == ((f (fst p) + g (snd p)) + (f (fst q) + g (snd q)))
             { add_interchange }.
    Qed.

    Lemma join_scale : ∀ (c : F) (p : A × B),
      f (fst (c *: p)) + g (snd (c *: p)) == c *: (f (fst p) + g (snd p)).
    Proof.
      intros c p.
      calc (f (c *: fst p) + g (c *: snd p))
        == (c *: f (fst p) + g (c *: snd p)) { linear_map_scale }
        == (c *: f (fst p) + c *: g (snd p)) { linear_map_scale }
        == (c *: (f (fst p) + g (snd p))) { left_distributive }.
    Qed.

    Definition join : LinearMap F (A × B) X :=
      linear_map_from_laws (fun p : A × B ⇒ f (fst p) + g (snd p))
        join_proper join_add join_scale.
  End Join.

End Pairing.

Section BiproductLaws.
  Context {F A B X : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.
  Context {eX : EqRel X} {zX : Zero X} {aX : HasAdd X} {nX : HasNeg X} {sX : HasScale F X}.
  Context {VX : VectorSpace F X}.

  Theorem exl_fork : ∀ (f : LinearMap F X A) (g : LinearMap F X B),
    exl ∘ fork f g == f.
  Proof. intros f g x. reflexivity. Qed.

  Theorem exr_fork : ∀ (f : LinearMap F X A) (g : LinearMap F X B),
    exr ∘ fork f g == g.
  Proof. intros f g x. reflexivity. Qed.

  Theorem fork_unique : ∀ h : LinearMap F X (A × B),
    fork (exl ∘ h) (exr ∘ h) == h.
  Proof. intros h x. split; reflexivity. Qed.

  Theorem join_inl : ∀ (f : LinearMap F A X) (g : LinearMap F B X),
    join f g ∘ inl == f.
  Proof.
    intros f g a.
    calc (f a + g 0)
      == (f a + (0 : X)) { (linear_map_zero g) }
      == (f a) { right_identity }.
  Qed.

  Theorem join_inr : ∀ (f : LinearMap F A X) (g : LinearMap F B X),
    join f g ∘ inr == g.
  Proof.
    intros f g b.
    calc (f 0 + g b)
      == ((0 : X) + g b) { (linear_map_zero f) }
      == (g b) { left_identity }.
  Qed.

  Theorem join_unique : ∀ h : LinearMap F (A × B) X,
    join (h ∘ inl) (h ∘ inr) == h.
  Proof.
    intros h p.
    calc (h (fst p, (0 : B)) + h ((0 : A), snd p))
      == (h ((fst p, (0 : B)) + ((0 : A), snd p)))
           { linear_map_additive }
      == (h p)
           { (linear_map_proper h _ p
                (conj (right_identity (fst p)) (left_identity (snd p)))) }.
  Qed.

  Theorem join_inl_inr : join inl inr == (id : LinearMap F (A × B) (A × B)).
  Proof.
    intros p.
    split; [ exact (right_identity (fst p)) | exact (left_identity (snd p)) ].
  Qed.

End BiproductLaws.

Section MatrixProduct.
  Context {F A B X Y : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.
  Context {eX : EqRel X} {zX : Zero X} {aX : HasAdd X} {nX : HasNeg X} {sX : HasScale F X}.
  Context {VX : VectorSpace F X}.
  Context {eY : EqRel Y} {zY : Zero Y} {aY : HasAdd Y} {nY : HasNeg Y} {sY : HasScale F Y}.
  Context {VY : VectorSpace F Y}.

  Theorem join_fork : ∀ (f : LinearMap F A X) (g : LinearMap F B X)
      (h : LinearMap F Y A) (k : LinearMap F Y B),
    join f g ∘ fork h k == f ∘ h + g ∘ k.
  Proof. intros f g h k y. reflexivity. Qed.

End MatrixProduct.

Section Representation.
  Context {F U V P Q : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eP : EqRel P} {zP : Zero P} {aP : HasAdd P} {nP : HasNeg P} {sP : HasScale F P}.
  Context {VP : VectorSpace F P}.
  Context {eQ : EqRel Q} {zQ : Zero Q} {aQ : HasAdd Q} {nQ : HasNeg Q} {sQ : HasScale F Q}.
  Context {VQ : VectorSpace F Q}.

  Definition rep (i : VSIso F U P) (j : VSIso F V Q) (f : LinearMap F U V)
    : LinearMap F P Q :=
    VSIso.to j ∘ f ∘ VSIso.from i.

End Representation.

Section RepresentationComp.
  Context {F U V W P Q R : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context {eP : EqRel P} {zP : Zero P} {aP : HasAdd P} {nP : HasNeg P} {sP : HasScale F P}.
  Context {VP : VectorSpace F P}.
  Context {eQ : EqRel Q} {zQ : Zero Q} {aQ : HasAdd Q} {nQ : HasNeg Q} {sQ : HasScale F Q}.
  Context {VQ : VectorSpace F Q}.
  Context {eR : EqRel R} {zR : Zero R} {aR : HasAdd R} {nR : HasNeg R} {sR : HasScale F R}.
  Context {VR : VectorSpace F R}.

  Theorem rep_comp : ∀ (i : VSIso F U P) (j : VSIso F V Q) (k : VSIso F W R)
      (g : LinearMap F V W) (f : LinearMap F U V),
    rep i k (g ∘ f) == rep j k g ∘ rep i j f.
  Proof.
    intros i j k g f p. unfold rep. simpl. symmetry.
    calc (VSIso.to k (g (VSIso.from j (VSIso.to j (f (VSIso.from i p))))))
      == (VSIso.to k (g (f (VSIso.from i p))))
           { vsiso_from_to }.
  Qed.

End RepresentationComp.
