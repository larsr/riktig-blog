Library Riktig.Structure.Circular

From Coq Require Import Morphisms.
From Riktig.Structure Require Export VectorSpace Linear.

Class HasCexp A := cexp : A → A.
Class HasCos F := cos : F → F.
Class HasSin F := sin : F → F.

Hint Mode HasCexp ! : typeclass_instances.
Hint Mode HasCos ! : typeclass_instances.
Hint Mode HasSin ! : typeclass_instances.

Typeclasses Transparent HasCexp HasCos HasSin.

A circular structure on an F-algebra A is an exponential cexp carrying sums of commuting elements to products, together with a pair cos, sin on F whose squares sum to 1, the two tied by Euler's identity at every square root of - (1) in A, with cos even, sin odd, and sin not identically zero.
Module Circular.
  Class Circular (F A : Type) `{Field F}
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
      {Ao : One A} {Am : HasMult A} {As : HasScale F A}
      {Ax : HasCexp A} {Fc : HasCos F} {Fs : HasSin F} := {
    ring :: Ring A;
    scaling :: ScaleLaws F A;

    
    mul_linear_l :: ∀ y : A, Linear F A A (fun x ⇒ x × y);
    mul_linear_r :: ∀ x : A, Linear F A A (fun y ⇒ x × y);

    cexp_proper :: Proper (equ ==> equ) Ax;
    cos_proper :: Proper (equ ==> equ) Fc;
    sin_proper :: Proper (equ ==> equ) Fs;

    cexp_add : ∀ x y : A,
      x × y == y × x → cexp (x + y) == cexp x × cexp y;
    pythagoras : ∀ t : F, cos t × cos t + sin t × sin t == 1;
    euler : ∀ (t : F) (e : A),
      e × e == - (1) → cexp (t *: e) == cos t *: 1 + sin t *: e;
    cos_even : ∀ t : F, cos (- t) == cos t;
    sin_odd : ∀ t : F, sin (- t) == - sin t;

    
    sin_nontrivial : ∃ t : F, sin t =/= 0;
  }.
  Hint Mode Circular - ! - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Circular.
