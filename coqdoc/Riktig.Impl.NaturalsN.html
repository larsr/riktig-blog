Library Riktig.Impl.NaturalsN

From Coq Require Import NArith.
From Riktig Require Export Structure.Naturals.

Coq's binary N, as a commutative semiring whose successor adds one.
Instance N_EqRel : EqRel N := eq.
Instance N_Zero : Zero N := 0%N.
Instance N_HasSucc : HasSucc N := N.succ.
Instance N_One : One N := 1%N.
Instance N_HasAdd : HasAdd N := N.add.
Instance N_HasMult : HasMult N := N.mul.

Instance N_Peano : Peano N N_Zero N_HasSucc :=
  {| Peano.equivalence := eq_equivalence;
     Peano.succ_proper := f_equal N.succ;
     Peano.succ_injective := N.succ_inj;
     Peano.succ_not_zero := N.neq_succ_0;
     Peano.nat_induction := fun P _ H0 HS n ⇒ N.peano_ind P H0 HS n |}.

Instance N_add_Magma : Magma N N_HasAdd :=
  {| Magma.equivalence := eq_equivalence; Magma.op_proper := fun x y Hx u v Hu ⇒ f_equal2 N.add Hx Hu |}.
Instance N_add_Semigroup : Semigroup N N_HasAdd :=
  {| Semigroup.magma := N_add_Magma;
     Semigroup.is_associative := fun a b c ⇒ eq_sym (N.add_assoc a b c) |}.
Instance N_add_Monoid : Monoid N N_Zero N_HasAdd :=
  {| Monoid.semigroup := N_add_Semigroup;
     Monoid.is_left_identity := N.add_0_l; Monoid.is_right_identity := N.add_0_r |}.

Instance N_mul_Magma : Magma N N_HasMult :=
  {| Magma.equivalence := eq_equivalence; Magma.op_proper := fun x y Hx u v Hu ⇒ f_equal2 N.mul Hx Hu |}.
Instance N_mul_Semigroup : Semigroup N N_HasMult :=
  {| Semigroup.magma := N_mul_Magma;
     Semigroup.is_associative := fun a b c ⇒ eq_sym (N.mul_assoc a b c) |}.
Instance N_mul_Monoid : Monoid N N_One N_HasMult :=
  {| Monoid.semigroup := N_mul_Semigroup;
     Monoid.is_left_identity := N.mul_1_l; Monoid.is_right_identity := N.mul_1_r |}.

Instance N_Semiring : Semiring N N_Zero N_HasAdd N_One N_HasMult :=
  {| Semiring.add_monoid := N_add_Monoid;
     Semiring.mul_monoid := N_mul_Monoid;
     Semiring.add_commutative := N.add_comm;
     Semiring.is_left_distributive := N.mul_add_distr_l;
     Semiring.is_right_distributive := N.mul_add_distr_r;
     Semiring.mul_zero_left := N.mul_0_l;
     Semiring.mul_zero_right := N.mul_0_r |}.
Instance N_CommSemiring : CommSemiring N N_Zero N_HasAdd N_One N_HasMult :=
  {| CommSemiring.semiring := N_Semiring;
     CommSemiring.mul_commutative := N.mul_comm |}.

Instance N_Naturals : Naturals N N_Zero N_HasSucc N_One N_HasAdd N_HasMult :=
  {| Naturals.peano := N_Peano;
     Naturals.semiring := N_CommSemiring;
     Naturals.succ_add := fun n ⇒ eq_sym (N.add_1_r n) |}.
