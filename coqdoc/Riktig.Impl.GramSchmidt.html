From Riktig Require Export Theory.GramSchmidt Impl.InnerGeometry.

Section GramSchmidtFix.
  Context {F V : Type} `{PosInner F V}.

  Fixpoint gram_schmidt_aux_fix (done : list V) (v : V) : V :=
    match done with
    | nil ⇒ v
    | cons d done' ⇒ gram_schmidt_aux_fix done' (v + - proj d v)
    end.

  Fixpoint gram_schmidt_step_fix (acc vs : list V) : list V :=
    match vs with
    | nil ⇒ acc
    | cons v vs' ⇒
        gram_schmidt_step_fix (acc ++ (gram_schmidt_aux_fix acc v :: nil)) vs'
    end.

  #[export] Instance HasGramSchmidtAux_fix : HasGramSchmidtAux V :=
    gram_schmidt_aux_fix.

  #[export] Instance HasGramSchmidtStep_fix : HasGramSchmidtStep V :=
    gram_schmidt_step_fix.

  #[export] Instance HasGramSchmidt_fix : HasGramSchmidt V :=
    gram_schmidt_step_fix nil.

  #[export] Instance GramSchmidt_fix : GramSchmidt F V.
  Proof.
    split; intros; reflexivity.
  Qed.

End GramSchmidtFix.
