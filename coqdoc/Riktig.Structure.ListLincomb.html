Library Riktig.Structure.ListLincomb

From Coq Require Export List.
From Riktig Require Export Ops Props.

list_lincomb coeffs vecs weights each vector by the scalar facing it.
Class HasListLincomb (F V : Type) :=
  list_lincomb : list F → list V → V.
Hint Mode HasListLincomb - ! : typeclass_instances.
Typeclasses Transparent HasListLincomb.

Module ListLincomb.
  Class ListLincomb (F V : Type) {Ve : EqRel V} {Vz : Zero V}
      {Va : HasAdd V} {Vs : HasScale F V} {Vll : HasListLincomb F V} := {
    is_list_lincomb_nil_l : ∀ vecs : list V,
      list_lincomb nil vecs == (0 : V);
    is_list_lincomb_nil_r : ∀ (c : F) (coeffs' : list F),
      list_lincomb (c :: coeffs') nil == (0 : V);
    is_list_lincomb_cons : ∀ (c : F) (coeffs' : list F) (v : V)
        (vecs' : list V),
      list_lincomb (c :: coeffs') (v :: vecs')
        == c *: v + list_lincomb coeffs' vecs';
  }.
  Hint Mode ListLincomb - ! - - - - - : typeclass_instances.
End ListLincomb.
Export ListLincomb.
