Require Export Riktig.Lambda.Syntax.


Inductive beta_step : tm → tm → Prop :=
  | beta_step_beta : ∀ b u : tm,
      beta_step (App (Abs b) u) (psub (scons u Var) b)
  | beta_step_abs : ∀ b b' : tm,
      beta_step b b' → beta_step (Abs b) (Abs b')
  | beta_step_app1 : ∀ f f' a : tm,
      beta_step f f' → beta_step (App f a) (App f' a)
  | beta_step_app2 : ∀ f a a' : tm,
      beta_step a a' → beta_step (App f a) (App f a').

Lemma step_is_beta_step : ∀ t t' : tm, step t t' → beta_step t t'.
Proof.
  intros t t' Hs.
  induction Hs as [b v Hv | f f' a Hs IH | v a a' Hv Hs IH].
  - rewrite subst_zero_as_psub. apply beta_step_beta.
  - apply beta_step_app1. exact IH.
  - apply beta_step_app2. exact IH.
Qed.

Inductive beta_conv : tm → tm → Prop :=
  | beta_conv_step : ∀ t t' : tm, beta_step t t' → beta_conv t t'
  | beta_conv_refl : ∀ t : tm, beta_conv t t
  | beta_conv_sym : ∀ t u : tm, beta_conv t u → beta_conv u t
  | beta_conv_trans : ∀ t u v : tm,
      beta_conv t u → beta_conv u v → beta_conv t v.


Lemma beta_conv_abs : ∀ b b' : tm,
  beta_conv b b' → beta_conv (Abs b) (Abs b').
Proof.
  intros b b' Hc.
  induction Hc as [t t' Hs | t | t u Hc IH | t u v Hc1 IH1 Hc2 IH2].
  - apply beta_conv_step. apply beta_step_abs. exact Hs.
  - apply beta_conv_refl.
  - apply beta_conv_sym. exact IH.
  - exact (beta_conv_trans _ _ _ IH1 IH2).
Qed.

Lemma beta_conv_app1 : ∀ f f' a : tm,
  beta_conv f f' → beta_conv (App f a) (App f' a).
Proof.
  intros f f' a Hc.
  induction Hc as [t t' Hs | t | t u Hc IH | t u v Hc1 IH1 Hc2 IH2].
  - apply beta_conv_step. apply beta_step_app1. exact Hs.
  - apply beta_conv_refl.
  - apply beta_conv_sym. exact IH.
  - exact (beta_conv_trans _ _ _ IH1 IH2).
Qed.

Lemma beta_conv_app2 : ∀ f a a' : tm,
  beta_conv a a' → beta_conv (App f a) (App f a').
Proof.
  intros f a a' Hc.
  induction Hc as [t t' Hs | t | t u Hc IH | t u v Hc1 IH1 Hc2 IH2].
  - apply beta_conv_step. apply beta_step_app2. exact Hs.
  - apply beta_conv_refl.
  - apply beta_conv_sym. exact IH.
  - exact (beta_conv_trans _ _ _ IH1 IH2).
Qed.

Lemma beta_conv_app : ∀ f f' a a' : tm,
  beta_conv f f' → beta_conv a a' → beta_conv (App f a) (App f' a').
Proof.
  intros f f' a a' Hf Ha.
  exact (beta_conv_trans _ _ _ (beta_conv_app1 f f' a Hf)
                               (beta_conv_app2 f' a a' Ha)).
Qed.


Lemma ren_psub_scons : ∀ (r : nat → nat) (b u : tm),
  ren r (psub (scons u Var) b) = psub (scons (ren r u) Var) (ren (rup r) b).
Proof.
  intros r b u.
  rewrite ren_psub, psub_ren.
  apply psub_ext. intros [| m]; reflexivity.
Qed.

Lemma beta_step_ren : ∀ (t t' : tm) (r : nat → nat),
  beta_step t t' → beta_step (ren r t) (ren r t').
Proof.
  intros t t' r Hs. revert r.
  induction Hs as [b u | b b' Hs IH | f f' a Hs IH | f a a' Hs IH];
    intros r; simpl.
  - rewrite (ren_psub_scons r b u). apply beta_step_beta.
  - apply beta_step_abs. apply IH.
  - apply beta_step_app1. apply IH.
  - apply beta_step_app2. apply IH.
Qed.

Lemma beta_conv_ren : ∀ (t t' : tm) (r : nat → nat),
  beta_conv t t' → beta_conv (ren r t) (ren r t').
Proof.
  intros t t' r Hc. revert r.
  induction Hc as [u u' Hs | u | u v Hc IH | u v w Hc1 IH1 Hc2 IH2]; intros r.
  - apply beta_conv_step. apply beta_step_ren. exact Hs.
  - apply beta_conv_refl.
  - apply beta_conv_sym. apply IH.
  - exact (beta_conv_trans _ _ _ (IH1 r) (IH2 r)).
Qed.


Inductive par_step : tm → tm → Prop :=
  | par_step_var : ∀ n : nat, par_step (Var n) (Var n)
  | par_step_abs : ∀ b b' : tm,
      par_step b b' → par_step (Abs b) (Abs b')
  | par_step_app : ∀ f f' a a' : tm,
      par_step f f' → par_step a a' → par_step (App f a) (App f' a')
  | par_step_beta : ∀ b b' a a' : tm,
      par_step b b' → par_step a a' →
      par_step (App (Abs b) a) (psub (scons a' Var) b').

Lemma par_step_refl : ∀ t : tm, par_step t t.
Proof.
  induction t as [n | b IH | f IHf a IHa].
  - apply par_step_var.
  - apply par_step_abs. exact IH.
  - apply par_step_app.
    + exact IHf.
    + exact IHa.
Qed.

Lemma beta_step_par_step : ∀ t t' : tm, beta_step t t' → par_step t t'.
Proof.
  intros t t' Hs.
  induction Hs as [b u | b b' Hs IH | f f' a Hs IH | f a a' Hs IH].
  - apply par_step_beta.
    + apply par_step_refl.
    + apply par_step_refl.
  - apply par_step_abs. exact IH.
  - apply par_step_app.
    + exact IH.
    + apply par_step_refl.
  - apply par_step_app.
    + apply par_step_refl.
    + exact IH.
Qed.

Lemma par_step_ren : ∀ (t t' : tm) (r : nat → nat),
  par_step t t' → par_step (ren r t) (ren r t').
Proof.
  intros t t' r Hp. revert r.
  induction Hp as [n | b b' Hp IH | f f' a a' Hf IHf Ha IHa
                   | b b' a a' Hb IHb Ha IHa]; intros r; simpl.
  - apply par_step_var.
  - apply par_step_abs. apply IH.
  - apply par_step_app.
    + apply IHf.
    + apply IHa.
  - rewrite (ren_psub_scons r b' a').
    apply par_step_beta.
    + apply IHb.
    + apply IHa.
Qed.

Lemma psub_scons_psub : ∀ (b a : tm) (s : nat → tm),
  psub (scons (psub s a) Var) (psub (pup s) b) = psub s (psub (scons a Var) b).
Proof.
  intros b a s.
  rewrite !psub_psub.
  apply psub_ext. intros [| m]; simpl.
  - reflexivity.
  - rewrite psub_scons_ren_shift, psub_var. reflexivity.
Qed.

Lemma par_step_pup : ∀ s s' : nat → tm,
  (∀ n : nat, par_step (s n) (s' n)) →
  ∀ n : nat, par_step (pup s n) (pup s' n).
Proof.
  intros s s' Hs [| m]; simpl.
  - apply par_step_var.
  - apply par_step_ren. apply Hs.
Qed.

Lemma par_step_psub : ∀ (t t' : tm) (s s' : nat → tm),
  par_step t t' → (∀ n : nat, par_step (s n) (s' n)) →
  par_step (psub s t) (psub s' t').
Proof.
  intros t t' s s' Hp. revert s s'.
  induction Hp as [n | b b' Hp IH | f f' a a' Hf IHf Ha IHa
                   | b b' a a' Hb IHb Ha IHa]; intros s s' Hs; simpl.
  - apply Hs.
  - apply par_step_abs. apply IH. apply par_step_pup. exact Hs.
  - apply par_step_app.
    + apply IHf. exact Hs.
    + apply IHa. exact Hs.
  - rewrite <- (psub_scons_psub b' a' s').
    apply par_step_beta.
    + apply IHb. apply par_step_pup. exact Hs.
    + apply IHa. exact Hs.
Qed.

Lemma par_step_scons : ∀ (a a' : tm),
  par_step a a' → ∀ n : nat, par_step (scons a Var n) (scons a' Var n).
Proof.
  intros a a' Ha [| m]; simpl.
  - exact Ha.
  - apply par_step_var.
Qed.


Fixpoint develop (t : tm) : tm :=
  match t with
  | Var n ⇒ Var n
  | Abs b ⇒ Abs (develop b)
  | App f a ⇒
      match f with
      | Abs b ⇒ psub (scons (develop a) Var) (develop b)
      | _ ⇒ App (develop f) (develop a)
      end
  end.

Lemma par_step_var_inv : ∀ (n : nat) (t : tm),
  par_step (Var n) t → t = Var n.
Proof.
  intros n t Hp. inversion Hp. reflexivity.
Qed.

Lemma par_step_abs_inv : ∀ (b t : tm),
  par_step (Abs b) t → ∃ b' : tm, t = Abs b' ∧ par_step b b'.
Proof.
  intros b t Hp. inversion Hp as [| b0 b' Hb Eb Et | |].
  ∃ b'. split.
  - reflexivity.
  - exact Hb.
Qed.

Theorem par_step_triangle : ∀ t u : tm,
  par_step t u → par_step u (develop t).
Proof.
  intros t u Hp.
  induction Hp as [n | b b' Hp IH | f f' a a' Hf IHf Ha IHa
                   | b b' a a' Hb IHb Ha IHa].
  - apply par_step_var.
  - apply par_step_abs. exact IH.
  - destruct f as [n | b | f1 f2].
    + rewrite (par_step_var_inv n f' Hf). simpl.
      apply par_step_app.
      × apply par_step_var.
      × exact IHa.
    + destruct (par_step_abs_inv b f' Hf) as [b' [Ef' Hb]].
      rewrite Ef'. simpl.
      apply par_step_beta.
      × rewrite Ef' in IHf.
        destruct (par_step_abs_inv b' (Abs (develop b)) IHf)
          as [b'' [Eb'' Hb'']].
        injection Eb'' as <-. exact Hb''.
      × exact IHa.
    + simpl. apply par_step_app.
      × exact IHf.
      × exact IHa.
  - simpl. apply par_step_psub.
    + exact IHb.
    + apply par_step_scons. exact IHa.
Qed.

Corollary par_step_diamond : ∀ t u v : tm,
  par_step t u → par_step t v →
  ∃ w : tm, par_step u w ∧ par_step v w.
Proof.
  intros t u v Hu Hv.
  ∃ (develop t). split.
  - exact (par_step_triangle t u Hu).
  - exact (par_step_triangle t v Hv).
Qed.


Inductive par_star : tm → tm → Prop :=
  | par_star_refl : ∀ t : tm, par_star t t
  | par_star_step : ∀ t u v : tm,
      par_step t u → par_star u v → par_star t v.

Lemma par_star_one : ∀ t u : tm, par_step t u → par_star t u.
Proof.
  intros t u Hp. exact (par_star_step t u u Hp (par_star_refl u)).
Qed.

Lemma par_star_trans : ∀ t u v : tm,
  par_star t u → par_star u v → par_star t v.
Proof.
  intros t u v H1. induction H1 as [t0 | t0 u0 v0 Hp Hs IH]; intros H2.
  - exact H2.
  - exact (par_star_step t0 u0 v (Hp) (IH H2)).
Qed.

Lemma par_star_strip : ∀ t u v : tm,
  par_star t u → par_step t v →
  ∃ w : tm, par_star v w ∧ par_step u w.
Proof.
  intros t u v H1. revert v.
  induction H1 as [t0 | t0 u0 v0 Hp Hs IH]; intros v Hv.
  - ∃ v. split.
    + apply par_star_refl.
    + exact Hv.
  - destruct (par_step_diamond t0 u0 v Hp Hv) as [w1 [Hu1 Hv1]].
    destruct (IH w1 Hu1) as [w [Hw1 Hw2]].
    ∃ w. split.
    + exact (par_star_step v w1 w Hv1 Hw1).
    + exact Hw2.
Qed.

Theorem par_star_confluent : ∀ t u v : tm,
  par_star t u → par_star t v →
  ∃ w : tm, par_star u w ∧ par_star v w.
Proof.
  intros t u v H1. revert v.
  induction H1 as [t0 | t0 u0 v0 Hp Hs IH]; intros v Hv.
  - ∃ v. split.
    + exact Hv.
    + apply par_star_refl.
  - destruct (par_star_strip t0 v u0 Hv Hp) as [w1 [Hw1 Hw2]].
    destruct (IH w1 Hw1) as [w [Hw3 Hw4]].
    ∃ w. split.
    + exact Hw3.
    + exact (par_star_step v w1 w Hw2 Hw4).
Qed.

Theorem church_rosser : ∀ t u : tm,
  beta_conv t u → ∃ w : tm, par_star t w ∧ par_star u w.
Proof.
  intros t u Hc.
  induction Hc as [t0 t1 Hs | t0 | t0 u0 Hc IH | t0 u0 v0 Hc1 IH1 Hc2 IH2].
  - ∃ t1. split.
    + apply par_star_one. apply beta_step_par_step. exact Hs.
    + apply par_star_refl.
  - ∃ t0. split.
    + apply par_star_refl.
    + apply par_star_refl.
  - destruct IH as [w [Hw1 Hw2]]. ∃ w. split.
    + exact Hw2.
    + exact Hw1.
  - destruct IH1 as [w1 [H1a H1b]]. destruct IH2 as [w2 [H2a H2b]].
    destruct (par_star_confluent u0 w1 w2 H1b H2a) as [w [Hw1 Hw2]].
    ∃ w. split.
    + exact (par_star_trans t0 w1 w H1a Hw1).
    + exact (par_star_trans v0 w2 w H2b Hw2).
Qed.


Lemma par_star_stuck : ∀ t v : tm, par_star t v →
  (∀ u : tm, par_step t u → u = t) → v = t.
Proof.
  intros t v Hs.
  induction Hs as [t0 | t0 u0 v0 Hp Hst IH]; intros Hstuck.
  - reflexivity.
  - rewrite (Hstuck u0 Hp) in IH. exact (IH Hstuck).
Qed.

Theorem var_not_beta_conv : ∀ n m : nat,
  n ≠ m → ¬ beta_conv (Var n) (Var m).
Proof.
  intros n m Hnm Hc.
  destruct (church_rosser (Var n) (Var m) Hc) as [w [Hn Hm]].
  assert (E : Var n = Var m).
  { transitivity w.
    - exact (eq_sym (par_star_stuck (Var n) w Hn (par_step_var_inv n))).
    - exact (par_star_stuck (Var m) w Hm (par_step_var_inv m)). }
  injection E as E'. exact (Hnm E').
Qed.


Definition self_application : tm := Abs (App (Var 0) (Var 0)).
Definition omega : tm := App self_application self_application.

Lemma par_step_app_var_inv : ∀ t : tm,
  par_step (App (Var 0) (Var 0)) t → t = App (Var 0) (Var 0).
Proof.
  intros t Hp.
  inversion Hp as [| | f f' a a' Hf Ha Ef Et |].
  rewrite (par_step_var_inv 0 f' Hf), (par_step_var_inv 0 a' Ha).
  reflexivity.
Qed.

Lemma par_step_self_application_inv : ∀ t : tm,
  par_step self_application t → t = self_application.
Proof.
  intros t Hp.
  destruct (par_step_abs_inv (App (Var 0) (Var 0)) t Hp) as [b' [Et Hb]].
  rewrite Et, (par_step_app_var_inv b' Hb). reflexivity.
Qed.

Lemma par_step_omega_inv : ∀ t : tm,
  par_step omega t → t = omega.
Proof.
  intros t Hp.
  inversion Hp as [| | f f' a a' Hf Ha Ef Et | b b' a a' Hb Ha Eb Et].
  - rewrite (par_step_self_application_inv f' Hf),
            (par_step_self_application_inv a' Ha).
    reflexivity.
  - rewrite (par_step_app_var_inv b' Hb),
            (par_step_self_application_inv a' Ha).
    reflexivity.
Qed.

Theorem omega_not_beta_conv_var : ∀ n : nat, ¬ beta_conv omega (Var n).
Proof.
  intros n Hc.
  destruct (church_rosser omega (Var n) Hc) as [w [Ho Hv]].
  assert (E : omega = Var n).
  { transitivity w.
    - exact (eq_sym (par_star_stuck omega w Ho par_step_omega_inv)).
    - exact (par_star_stuck (Var n) w Hv (par_step_var_inv n)). }
  discriminate E.
Qed.
