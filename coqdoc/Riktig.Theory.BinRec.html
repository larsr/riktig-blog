The programs of Programs/BinaryNumber, spelled as recursions: their equations bit by bit, and that they respect equality. They are what makes any binary numbers with recursion a BinaryArith (Impl/BinaryArithBinRec).
Section BinRecTheory.
  Context {N : Type} `{BinRec N}.

  Local Notation bin_case z f0 f1 := (binrec _ z (fun m _ ⇒ f0 m) (fun m _ ⇒ f1 m)).
  Local Notation bin_succ := (binrec N (bit1 0) (fun m _ ⇒ bit1 m) (fun _ s ⇒ bit0 s)).
  Local Notation bin_add :=
    (binrec (N → N) (fun y ⇒ y)
      (fun n rec y ⇒ bin_case (bit0 n) (fun m ⇒ bit0 (rec m)) (fun m ⇒ bit1 (rec m)) y)
      (fun n rec y ⇒ bin_case (bit1 n) (fun m ⇒ bit1 (rec m)) (fun m ⇒ bit0 (bin_succ (rec m))) y)).
  Local Notation bin_mult x y := (binrec N (0 : N) (fun _ p ⇒ bit0 p) (fun _ p ⇒ bin_add (bit0 p) y) x).

Reading the top bit

  Lemma bin_case_proper : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (z : A) (f0 f1 : N → A),
    Proper (equ ==> equ) f0 → Proper (equ ==> equ) f1 → Proper (equ ==> equ) (bin_case z f0 f1).
  Proof.
    intros A eA EA z f0 f1 Hf0 Hf1.
    exact (BinRec.binrec_proper A eA EA z _ _ (fun m m' Hm _ _ _ ⇒ Hf0 m m' Hm) (fun m m' Hm _ _ _ ⇒ Hf1 m m' Hm)).
  Qed.

  Lemma bin_case_ext : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (z z' : A) (f0 f0' f1 f1' : N → A),
    z == z' → (∀ m : N, f0 m == f0' m) → (∀ m : N, f1 m == f1' m) →
    ∀ n : N, bin_case z f0 f1 n == bin_case z' f0' f1' n.
  Proof.
    intros A eA EA z z' f0 f0' f1 f1' Hz E0 E1.
    exact (BinRec.binrec_ext A eA EA z z' _ _ _ _ (fun _ _ _ _ ⇒ reflexivity _) (fun _ _ _ _ ⇒ reflexivity _)
             Hz (fun m _ ⇒ E0 m) (fun m _ ⇒ E1 m)).
  Qed.

  Lemma bin_case_zero : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (z : A) (f0 f1 : N → A),
    bin_case z f0 f1 0 == z.
  Proof. intros A eA EA z f0 f1. exact (BinRec.binrec_zero A eA EA z _ _). Qed.

  Lemma bin_case_bit0 : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (z : A) (f0 f1 : N → A) (n : N),
    n =/= 0 → bin_case z f0 f1 (bit0 n) == f0 n.
  Proof. intros A eA EA z f0 f1 n Hn. exact (BinRec.binrec_bit0 A eA EA z _ _ n Hn). Qed.

  Lemma bin_case_bit1 : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (z : A) (f0 f1 : N → A) (n : N),
    bin_case z f0 f1 (bit1 n) == f1 n.
  Proof. intros A eA EA z f0 f1 n. exact (BinRec.binrec_bit1 A eA EA z _ _ n). Qed.

The successor

  #[export] Instance bin_succ_proper : Proper (equ ==> equ) (bin_succ : N → N).
  Proof.
    exact (BinRec.binrec_proper N _ _ (bit1 0) _ _
             (fun m m' Hm _ _ _ ⇒ BinaryNumber.bit1_proper m m' Hm) (fun _ _ _ s s' Hs ⇒ BinaryNumber.bit0_proper s s' Hs)).
  Qed.

  Lemma bin_succ_zero : bin_succ (0 : N) == bit1 0.
  Proof. exact (BinRec.binrec_zero N _ _ (bit1 0) _ _). Qed.

  Lemma bin_succ_bit0 : ∀ n : N, n =/= 0 → bin_succ (bit0 n) == bit1 n.
  Proof. intros n Hn. exact (BinRec.binrec_bit0 N _ _ (bit1 0) _ _ n Hn). Qed.

  Lemma bin_succ_bit1 : ∀ n : N, bin_succ (bit1 n) == bit0 (bin_succ n).
  Proof. intros n. exact (BinRec.binrec_bit1 N _ _ (bit1 0) _ _ n). Qed.

  Theorem bin_succ_not_zero : ∀ n : N, bin_succ n =/= 0.
  Proof.
    apply (BinaryNumber.bin_induction (fun n ⇒ bin_succ n =/= 0)).
    - intros n n' Hn. exact (not_iff_morphism _ _ (equiv_iff (bin_succ_proper n n' Hn) (reflexivity 0))).
    - intros Hz. apply (BinaryNumber.bit1_not_zero (0 : N)).
      calc (bit1 (0 : N)) == (bin_succ (0 : N)) { bin_succ_zero } == (0 : N) { Hz }.
    - intros n IH. destruct (zero_or_not n) as [Hz | Hn].
      + intros Hs. apply IH.
        calc (bin_succ n) == (bin_succ (bit0 n)) { (bin_succ_proper _ _ (symmetry (bit0_zero_of n Hz))) } == (0 : N) { Hs }.
      + intros Hs. apply (BinaryNumber.bit1_not_zero n).
        calc (bit1 n) == (bin_succ (bit0 n)) { (bin_succ_bit0 n Hn) } == (0 : N) { Hs }.
    - intros n IH Hs. apply (bit0_not_zero (bin_succ n) IH).
      calc (bit0 (bin_succ n)) == (bin_succ (bit1 n)) { (bin_succ_bit1 n) } == (0 : N) { Hs }.
  Qed.

Addition: functions of the second summand compare pointwise.

  Local Instance fun_eqrel : EqRel (N → N) := fun f g ⇒ ∀ y : N, f y == g y.
  Local Instance fun_equivalence : Equivalence (@equ (N → N) fun_eqrel).
  Proof.
    split.
    - intros f y. reflexivity.
    - intros f g Hfg y. exact (symmetry (Hfg y)).
    - intros f g h Hfg Hgh y. exact (transitivity (Hfg y) (Hgh y)).
  Qed.

  Lemma bin_add_zero_l : ∀ y : N, bin_add (0 : N) y == y.
  Proof. intros y. exact (BinRec.binrec_zero (N → N) fun_eqrel fun_equivalence (fun y ⇒ y) _ _ y). Qed.

  Lemma bin_add_bit0 : ∀ n : N, n =/= 0 → ∀ y : N,
    bin_add (bit0 n) y == bin_case (bit0 n) (fun m ⇒ bit0 (bin_add n m)) (fun m ⇒ bit1 (bin_add n m)) y.
  Proof. intros n Hn y. exact (BinRec.binrec_bit0 (N → N) fun_eqrel fun_equivalence (fun y ⇒ y) _ _ n Hn y). Qed.

  Lemma bin_add_bit1 : ∀ n y : N,
    bin_add (bit1 n) y == bin_case (bit1 n) (fun m ⇒ bit1 (bin_add n m)) (fun m ⇒ bit0 (bin_succ (bin_add n m))) y.
  Proof. intros n y. exact (BinRec.binrec_bit1 (N → N) fun_eqrel fun_equivalence (fun y ⇒ y) _ _ n y). Qed.

  Lemma bin_add_step0_proper : Proper (equ ==> equ ==> equ)
    (fun (n : N) (rec : N → N) (y : N) ⇒ bin_case (bit0 n) (fun m ⇒ bit0 (rec m)) (fun m ⇒ bit1 (rec m)) y).
  Proof.
    intros n n' Hn rec rec' Hrec y.
    exact (bin_case_ext N _ _ _ _ _ _ _ _ (BinaryNumber.bit0_proper n n' Hn)
             (fun m ⇒ BinaryNumber.bit0_proper _ _ (Hrec m)) (fun m ⇒ BinaryNumber.bit1_proper _ _ (Hrec m)) y).
  Qed.

  Lemma bin_add_step1_proper : Proper (equ ==> equ ==> equ)
    (fun (n : N) (rec : N → N) (y : N) ⇒ bin_case (bit1 n) (fun m ⇒ bit1 (rec m)) (fun m ⇒ bit0 (bin_succ (rec m))) y).
  Proof.
    intros n n' Hn rec rec' Hrec y.
    exact (bin_case_ext N _ _ _ _ _ _ _ _ (BinaryNumber.bit1_proper n n' Hn)
             (fun m ⇒ BinaryNumber.bit1_proper _ _ (Hrec m))
             (fun m ⇒ BinaryNumber.bit0_proper _ _ (bin_succ_proper _ _ (Hrec m))) y).
  Qed.

  Lemma bin_add_proper_l : ∀ x x' : N, x == x' → ∀ y : N, bin_add x y == bin_add x' y.
  Proof.
    intros x x' Hx.
    exact (BinRec.binrec_proper (N → N) fun_eqrel fun_equivalence (fun y ⇒ y) _ _ bin_add_step0_proper bin_add_step1_proper x x' Hx).
  Qed.

  Lemma bin_add_proper_r : ∀ x : N, Proper (equ ==> equ) (fun y ⇒ bin_add x y).
  Proof.
    apply (BinaryNumber.bin_induction (fun x ⇒ Proper (equ ==> equ) (fun y ⇒ bin_add x y))).
    - intros x x' Hx. split; intros HP y y' Hy.
      + calc (bin_add x' y) == (bin_add x y) { (bin_add_proper_l x x' Hx y) } == (bin_add x y') { (HP y y' Hy) }
          == (bin_add x' y') { (bin_add_proper_l x x' Hx y') }.
      + calc (bin_add x y) == (bin_add x' y) { (bin_add_proper_l x x' Hx y) } == (bin_add x' y') { (HP y y' Hy) }
          == (bin_add x y') { (bin_add_proper_l x x' Hx y') }.
    - intros y y' Hy. calc (bin_add (0 : N) y) == y { (bin_add_zero_l y) } == y' { Hy } == (bin_add (0 : N) y') { (bin_add_zero_l y') }.
    - intros n IH y y' Hy. destruct (zero_or_not n) as [Hz | Hn].
      + calc (bin_add (bit0 n) y)
          == (bin_add n y) { (bin_add_proper_l _ _ (bit0_zero_of n Hz) y) }
          == (bin_add n y') { (IH y y' Hy) }
          == (bin_add (bit0 n) y') { (bin_add_proper_l _ _ (bit0_zero_of n Hz) y') }.
      + calc (bin_add (bit0 n) y)
          == (bin_case (bit0 n) (fun m ⇒ bit0 (bin_add n m)) (fun m ⇒ bit1 (bin_add n m)) y) { (bin_add_bit0 n Hn y) }
          == (bin_case (bit0 n) (fun m ⇒ bit0 (bin_add n m)) (fun m ⇒ bit1 (bin_add n m)) y')
               { (bin_case_proper N _ _ _ _ _ (fun m m' Hm ⇒ BinaryNumber.bit0_proper _ _ (IH m m' Hm))
                    (fun m m' Hm ⇒ BinaryNumber.bit1_proper _ _ (IH m m' Hm)) y y' Hy) }
          == (bin_add (bit0 n) y') { (bin_add_bit0 n Hn y') }.
    - intros n IH y y' Hy.
      calc (bin_add (bit1 n) y)
        == (bin_case (bit1 n) (fun m ⇒ bit1 (bin_add n m)) (fun m ⇒ bit0 (bin_succ (bin_add n m))) y) { (bin_add_bit1 n y) }
        == (bin_case (bit1 n) (fun m ⇒ bit1 (bin_add n m)) (fun m ⇒ bit0 (bin_succ (bin_add n m))) y')
             { (bin_case_proper N _ _ _ _ _ (fun m m' Hm ⇒ BinaryNumber.bit1_proper _ _ (IH m m' Hm))
                  (fun m m' Hm ⇒ BinaryNumber.bit0_proper _ _ (bin_succ_proper _ _ (IH m m' Hm))) y y' Hy) }
        == (bin_add (bit1 n) y') { (bin_add_bit1 n y') }.
  Qed.

  #[export] Instance bin_add_proper : Proper (equ ==> equ ==> equ) (bin_add : N → N → N).
  Proof. intros x x' Hx y y' Hy. exact (transitivity (bin_add_proper_l x x' Hx y) (bin_add_proper_r x' y y' Hy)). Qed.
The equations of addition, bit by bit.
  Lemma bin_add_bit0_zero : ∀ n : N, n =/= 0 → bin_add (bit0 n) 0 == bit0 n.
  Proof.
    intros n Hn.
    calc (bin_add (bit0 n) 0)
      == (bin_case (bit0 n) (fun m ⇒ bit0 (bin_add n m)) (fun m ⇒ bit1 (bin_add n m)) 0) { (bin_add_bit0 n Hn 0) }
      == (bit0 n) { (bin_case_zero N _ _ _ _ _) }.
  Qed.

  Lemma bin_add_bit0_bit0 : ∀ n m : N, n =/= 0 → m =/= 0 → bin_add (bit0 n) (bit0 m) == bit0 (bin_add n m).
  Proof.
    intros n m Hn Hm.
    calc (bin_add (bit0 n) (bit0 m))
      == (bin_case (bit0 n) (fun m ⇒ bit0 (bin_add n m)) (fun m ⇒ bit1 (bin_add n m)) (bit0 m)) { (bin_add_bit0 n Hn (bit0 m)) }
      == (bit0 (bin_add n m)) { (bin_case_bit0 N _ _ _ _ _ m Hm) }.
  Qed.

  Lemma bin_add_bit0_bit1 : ∀ n m : N, n =/= 0 → bin_add (bit0 n) (bit1 m) == bit1 (bin_add n m).
  Proof.
    intros n m Hn.
    calc (bin_add (bit0 n) (bit1 m))
      == (bin_case (bit0 n) (fun m ⇒ bit0 (bin_add n m)) (fun m ⇒ bit1 (bin_add n m)) (bit1 m)) { (bin_add_bit0 n Hn (bit1 m)) }
      == (bit1 (bin_add n m)) { (bin_case_bit1 N _ _ _ _ _ m) }.
  Qed.

  Lemma bin_add_bit1_zero : ∀ n : N, bin_add (bit1 n) 0 == bit1 n.
  Proof.
    intros n.
    calc (bin_add (bit1 n) 0)
      == (bin_case (bit1 n) (fun m ⇒ bit1 (bin_add n m)) (fun m ⇒ bit0 (bin_succ (bin_add n m))) 0) { (bin_add_bit1 n 0) }
      == (bit1 n) { (bin_case_zero N _ _ _ _ _) }.
  Qed.

  Lemma bin_add_bit1_bit0 : ∀ n m : N, m =/= 0 → bin_add (bit1 n) (bit0 m) == bit1 (bin_add n m).
  Proof.
    intros n m Hm.
    calc (bin_add (bit1 n) (bit0 m))
      == (bin_case (bit1 n) (fun m ⇒ bit1 (bin_add n m)) (fun m ⇒ bit0 (bin_succ (bin_add n m))) (bit0 m)) { (bin_add_bit1 n (bit0 m)) }
      == (bit1 (bin_add n m)) { (bin_case_bit0 N _ _ _ _ _ m Hm) }.
  Qed.

  Lemma bin_add_bit1_bit1 : ∀ n m : N, bin_add (bit1 n) (bit1 m) == bit0 (bin_succ (bin_add n m)).
  Proof.
    intros n m.
    calc (bin_add (bit1 n) (bit1 m))
      == (bin_case (bit1 n) (fun m ⇒ bit1 (bin_add n m)) (fun m ⇒ bit0 (bin_succ (bin_add n m))) (bit1 m)) { (bin_add_bit1 n (bit1 m)) }
      == (bit0 (bin_succ (bin_add n m))) { (bin_case_bit1 N _ _ _ _ _ m) }.
  Qed.

Multiplication

  Lemma bin_mult_proper_l : ∀ x x' : N, x == x' → ∀ y : N, bin_mult x y == bin_mult x' y.
  Proof.
    intros x x' Hx y.
    exact (BinRec.binrec_proper N _ _ 0 _ _ (fun _ _ _ p p' Hp ⇒ BinaryNumber.bit0_proper p p' Hp)
             (fun _ _ _ p p' Hp ⇒ bin_add_proper _ _ (BinaryNumber.bit0_proper p p' Hp) y y (reflexivity y)) x x' Hx).
  Qed.

  Lemma bin_mult_zero_l : ∀ y : N, bin_mult (0 : N) y == (0 : N).
  Proof. intros y. exact (BinRec.binrec_zero N _ _ 0 _ _). Qed.

  Lemma bin_mult_bit0 : ∀ n y : N, n =/= 0 → bin_mult (bit0 n) y == bit0 (bin_mult n y).
  Proof. intros n y Hn. exact (BinRec.binrec_bit0 N _ _ 0 _ _ n Hn). Qed.

  Lemma bin_mult_bit1 : ∀ n y : N, bin_mult (bit1 n) y == bin_add (bit0 (bin_mult n y)) y.
  Proof. intros n y. exact (BinRec.binrec_bit1 N _ _ 0 _ _ n). Qed.

  Lemma bin_mult_proper_r : ∀ x : N, Proper (equ ==> equ) (fun y ⇒ bin_mult x y).
  Proof.
    apply (BinaryNumber.bin_induction (fun x ⇒ Proper (equ ==> equ) (fun y ⇒ bin_mult x y))).
    - intros x x' Hx. split; intros HP y y' Hy.
      + calc (bin_mult x' y) == (bin_mult x y) { (bin_mult_proper_l x x' Hx y) } == (bin_mult x y') { (HP y y' Hy) }
          == (bin_mult x' y') { (bin_mult_proper_l x x' Hx y') }.
      + calc (bin_mult x y) == (bin_mult x' y) { (bin_mult_proper_l x x' Hx y) } == (bin_mult x' y') { (HP y y' Hy) }
          == (bin_mult x y') { (bin_mult_proper_l x x' Hx y') }.
    - intros y y' Hy. calc (bin_mult (0 : N) y) == (0 : N) { (bin_mult_zero_l y) } == (bin_mult (0 : N) y') { (bin_mult_zero_l y') }.
    - intros n IH y y' Hy. destruct (zero_or_not n) as [Hz | Hn].
      + calc (bin_mult (bit0 n) y)
          == (bin_mult n y) { (bin_mult_proper_l _ _ (bit0_zero_of n Hz) y) }
          == (bin_mult n y') { (IH y y' Hy) }
          == (bin_mult (bit0 n) y') { (bin_mult_proper_l _ _ (bit0_zero_of n Hz) y') }.
      + calc (bin_mult (bit0 n) y)
          == (bit0 (bin_mult n y)) { (bin_mult_bit0 n y Hn) }
          == (bit0 (bin_mult n y')) { (IH y y' Hy) }
          == (bin_mult (bit0 n) y') { (bin_mult_bit0 n y' Hn) }.
    - intros n IH y y' Hy.
      calc (bin_mult (bit1 n) y)
        == (bin_add (bit0 (bin_mult n y)) y) { (bin_mult_bit1 n y) }
        == (bin_add (bit0 (bin_mult n y')) y') { (bin_add_proper _ _ (BinaryNumber.bit0_proper _ _ (IH y y' Hy)) y y' Hy) }
        == (bin_mult (bit1 n) y') { (bin_mult_bit1 n y') }.
  Qed.

  #[export] Instance bin_mult_proper : Proper (equ ==> equ ==> equ) (fun x y : N ⇒ bin_mult x y).
  Proof. intros x x' Hx y y' Hy. exact (transitivity (bin_mult_proper_l x x' Hx y) (bin_mult_proper_r x' y y' Hy)). Qed.
