Over an associative algebra the commutator x × y + - (y × x) is bilinear, vanishes on a repeated argument, and satisfies the Jacobi identity.

  #[local] Instance commutator_vector_space : VectorSpace F A :=
    { vectors := abgroup; scaling := SL }.

  Definition commutator (x y : A) : A := x × y + - (y × x).

  #[local] Instance Commutator_HasBracket : HasBracket A := commutator.


  Lemma commutator_proper :
    Proper (equ ==> equ ==> equ) (bracket (V := A)).
  Proof.
    intros x x' Hx y y' Hy.
    unfold bracket, Commutator_HasBracket, commutator.
    rewrite Hx, Hy.
    reflexivity.
  Qed.

  Lemma commutator_left_additive : ∀ x x' y : A,
    [ x + x' , y ] == [ x , y ] + [ x' , y ].
  Proof.
    intros x x' y.
    calc (((x + x') × y + - (y × (x + x'))) : A)
      == (((x × y + x' × y) + - (y × (x + x'))) : A)
         { (right_distributive x x' y) }
      == (((x × y + x' × y) + - (y × x + y × x')) : A)
         { (left_distributive y x x') }
      == (((x × y + x' × y) + (- (y × x) + - (y × x'))) : A)
         { (neg_add (y × x) (y × x')) }
      == (((x × y + - (y × x)) + (x' × y + - (y × x'))) : A)
         { add_interchange }.
  Qed.

  Lemma commutator_right_additive : ∀ x y y' : A,
    [ x , y + y' ] == [ x , y ] + [ x , y' ].
  Proof.
    intros x y y'.
    calc ((x × (y + y') + - ((y + y') × x)) : A)
      == (((x × y + x × y') + - ((y + y') × x)) : A)
         { (left_distributive x y y') }
      == (((x × y + x × y') + - (y × x + y' × x)) : A)
         { (right_distributive y y' x) }
      == (((x × y + x × y') + (- (y × x) + - (y' × x))) : A)
         { (neg_add (y × x) (y' × x)) }
      == (((x × y + - (y × x)) + (x × y' + - (y' × x))) : A)
         { add_interchange }.
  Qed.

  Lemma commutator_left_scale : ∀ (a : F) (x y : A),
    [ a *: x , y ] == a *: [ x , y ].
  Proof.
    intros a x y.
    calc (((a *: x) × y + - (y × (a *: x))) : A)
      == ((a *: (x × y) + - (y × (a *: x))) : A)
         { (scale_mul_l a x y) }
      == ((a *: (x × y) + - (a *: (y × x))) : A)
         { (scale_mul_r a y x) }
      == ((a *: (x × y) + a *: - (y × x)) : A)
         { (scale_neg a (y × x)) }
      == ((a *: (x × y + - (y × x))) : A)
         { (left_distributive a (x × y) (- (y × x))) }.
  Qed.

  Lemma commutator_right_scale : ∀ (a : F) (x y : A),
    [ x , a *: y ] == a *: [ x , y ].
  Proof.
    intros a x y.
    calc ((x × (a *: y) + - ((a *: y) × x)) : A)
      == ((a *: (x × y) + - ((a *: y) × x)) : A)
         { (scale_mul_r a x y) }
      == ((a *: (x × y) + - (a *: (y × x))) : A)
         { (scale_mul_l a y x) }
      == ((a *: (x × y) + a *: - (y × x)) : A)
         { (scale_neg a (y × x)) }
      == ((a *: (x × y + - (y × x))) : A)
         { (left_distributive a (x × y) (- (y × x))) }.
  Qed.

  #[local] Instance Commutator_Bilinear : Bilinear F A A A bracket :=
    {| Bilinear.proper := commutator_proper;
       Bilinear.linear_l := fun v ⇒
         {| Linear.proper := fun x x' Hx ⇒
              commutator_proper x x' Hx v v (reflexivity v);
            Linear.is_additive := fun x x' ⇒ commutator_left_additive x x' v;
            Linear.is_homogeneous := fun a x ⇒ commutator_left_scale a x v |};
       Bilinear.linear_r := fun u ⇒
         {| Linear.proper := fun y y' Hy ⇒
              commutator_proper u u (reflexivity u) y y' Hy;
            Linear.is_additive := fun y y' ⇒ commutator_right_additive u y y';
            Linear.is_homogeneous := fun a y ⇒ commutator_right_scale a u y |} |}.

The bracket vanishes on a repeated argument: an element commutes with itself, so the two products cancel.
  Lemma commutator_self_zero : ∀ x : A, [ x , x ] == 0.
  Proof. intros x. exact (add_right_inverse (x × x)). Qed.

  #[local] Instance Commutator_Alternates : Alternates A bracket :=
    commutator_self_zero.


  Lemma commutator_leibniz_left : ∀ x u v : A,
    [ x , u ] × v == x × (u × v) + - (u × (x × v)).
  Proof.
    intros x u v.
    calc (((x × u + - (u × x)) × v) : A)
      == (((x × u) × v + (- (u × x)) × v) : A)
         { (right_distributive (x × u) (- (u × x)) v) }
      == (((x × u) × v + - ((u × x) × v)) : A)
         { (mul_neg_l (u × x) v) }
      == ((x × (u × v) + - ((u × x) × v)) : A)
         { (associative x u v) }
      == ((x × (u × v) + - (u × (x × v))) : A)
         { (associative u x v) }.
  Qed.

  Lemma commutator_leibniz_right : ∀ x u v : A,
    u × [ x , v ] == u × (x × v) + - ((u × v) × x).
  Proof.
    intros x u v.
    calc ((u × (x × v + - (v × x))) : A)
      == ((u × (x × v) + u × - (v × x)) : A)
         { (left_distributive u (x × v) (- (v × x))) }
      == ((u × (x × v) + - (u × (v × x))) : A)
         { (mul_neg_r u (v × x)) }
      == ((u × (x × v) + - ((u × v) × x)) : A)
         { (associative u v x) }.
  Qed.

  Theorem commutator_leibniz : ∀ x u v : A,
    [ x , u × v ] == [ x , u ] × v + u × [ x , v ].
  Proof.
    intros x u v.
    symmetry.
    calc (([ x , u ] × v + u × [ x , v ]) : A)
      == ((x × (u × v) + - (u × (x × v))) + u × [ x , v ] : A)
         { (commutator_leibniz_left x u v) }
      == ((x × (u × v) + - (u × (x × v)))
            + (u × (x × v) + - ((u × v) × x)) : A)
         { (commutator_leibniz_right x u v) }
      == ((x × (u × v) + - ((u × v) × x)) : A)
         { (sub_add_mid_cancel (x × (u × v)) (u × (x × v)) (- ((u × v) × x))) }.
  Qed.


The commutator brackets like a derivation: Leibniz for the associative product, read at a product that is itself a bracket.
  Theorem commutator_derivation : ∀ x y z : A,
    [ x , [ y , z ] ] == [ [ x , y ] , z ] + [ y , [ x , z ] ].
  Proof.
    intros x y z.
    calc ([ x , y × z + - (z × y) ] : A)
      == ([ x , y × z ] + [ x , - (z × y) ] : A)
         { (commutator_right_additive x (y × z) (- (z × y))) }
      == ([ x , y × z ] + - [ x , z × y ] : A)
         { (bracket_neg_r x (z × y)) }
      == (([ x , y ] × z + y × [ x , z ]) + - [ x , z × y ] : A)
         { (commutator_leibniz x y z) }
      == (([ x , y ] × z + y × [ x , z ])
            + - ([ x , z ] × y + z × [ x , y ]) : A)
         { (commutator_leibniz x z y) }
      == (([ x , y ] × z + y × [ x , z ])
            + - (z × [ x , y ] + [ x , z ] × y) : A)
         { (add_commutative ([ x , z ] × y) (z × [ x , y ])) }
      == (([ x , y ] × z + - (z × [ x , y ]))
            + (y × [ x , z ] + - ([ x , z ] × y)) : A)
         { sub_pair }.
  Qed.

Jacobi, cited from Theory/LieAlgebra.v rather than expanded: with alternation in hand, the derivation identity is the whole content.
  Definition commutator_jacobi : ∀ x y z : A,
    [ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ] == 0 :=
    jacobi_from_derivation commutator_derivation.

The second instance of the Lie algebra class: every associative algebra is one under the commutator.
  Definition commutator_lie_algebra : LieAlgebra F A :=
    {| LieAlgebra.is_bilinear := Commutator_Bilinear;
       LieAlgebra.is_alternating := Commutator_Alternates;
       LieAlgebra.jacobi := commutator_jacobi |}.

End Commutator.


Section MatrixCommutator.
  Context {F : Type} `{Field F} (n : nat).

  #[local] Instance matrix_mul_linear_l (B : Matrix n n F) :
    Linear F (Matrix n n F) (Matrix n n F) (fun A ⇒ A × B) :=
    {| Linear.proper := fun A A' HA ⇒
         mat_mul_proper A A' HA B B (reflexivity B);
       Linear.is_additive := fun A A' ⇒ mat_mul_add_l A A' B;
       Linear.is_homogeneous := fun a A ⇒ mat_mul_scale_l a A B |}.

  #[local] Instance matrix_mul_linear_r (A : Matrix n n F) :
    Linear F (Matrix n n F) (Matrix n n F) (fun B ⇒ A × B) :=
    {| Linear.proper := fun B B' HB ⇒
         mat_mul_proper A A (reflexivity A) B B' HB;
       Linear.is_additive := fun B B' ⇒ mat_mul_add_r A B B';
       Linear.is_homogeneous := fun a B ⇒ mat_mul_scale_r a A B |}.

  #[local] Instance Matrix_HasBracket : HasBracket (Matrix n n F) :=
    commutator.

  Definition matrix_lie_algebra : LieAlgebra F (Matrix n n F) :=
    commutator_lie_algebra.

End MatrixCommutator.

  Example mat2_commutator_value :
    mat2_commutator == mat2 (-4 # 1)%Q (-12 # 1)%Q (12 # 1)%Q (4 # 1)%Q.
  Proof.
    intros i k; revert k; revert i.
    refine (fin2_case _ _ _); [ | intros i1 ]; refine (fin2_case _ _ _);
      intros; vm_compute; reflexivity.
  Qed.

  Theorem mat2_commutator_nonzero : mat2_commutator =/= (0 : Matrix 2 2 Q).
  Proof.
    intros Hzero.
    generalize (Hzero Fin.F1 Fin.F1).
    vm_compute.
    discriminate.
  Qed.

  Theorem mat2_jacobi : ∀ x y z : Matrix 2 2 Q,
    [ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ] == 0.
  Proof. exact jacobi. Qed.
