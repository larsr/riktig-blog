The zero space: unit is a vector space over any field, every operation returning its one element.
Local Ltac unit_law :=
  repeat (hnf; intros); repeat match goal with u : unit |- _ ⇒ destruct u end; reflexivity.

Section UnitVS.
  Context {F : Type} `{Field F}.

  #[export] Instance Unit_EqRel : EqRel unit := eq.
  #[export] Instance Unit_Zero : Zero unit := tt.
  #[export] Instance Unit_HasAdd : HasAdd unit := fun _ _ ⇒ tt.
  #[export] Instance Unit_HasNeg : HasNeg unit := fun _ ⇒ tt.
  #[export] Instance Unit_HasScale : HasScale F unit := fun _ _ ⇒ tt.

  #[export] Instance Unit_Magma : Magma unit Unit_HasAdd.
  Proof. split. exact eq_equivalence. unit_law. Qed.
  #[export] Instance Unit_Semigroup : Semigroup unit Unit_HasAdd.
  Proof. split. exact Unit_Magma. unit_law. Qed.
  #[export] Instance Unit_Monoid : Monoid unit Unit_Zero Unit_HasAdd.
  Proof. split. exact Unit_Semigroup. all: unit_law. Qed.
  #[export] Instance Unit_Group : Group unit Unit_Zero Unit_HasAdd Unit_HasNeg.
  Proof. split. exact Unit_Monoid. all: unit_law. Qed.
  #[export] Instance Unit_AbGroup : AbGroup unit Unit_Zero Unit_HasAdd Unit_HasNeg.
  Proof. split. exact Unit_Group. unit_law. Qed.
  #[export] Instance Unit_ScaleLaws : ScaleLaws F unit Unit_HasAdd Unit_HasScale.
  Proof. split. all: unit_law. Qed.
  #[export] Instance Unit_VectorSpace : VectorSpace F unit Unit_Zero Unit_HasAdd Unit_HasNeg Unit_HasScale :=
    {| VectorSpace.vectors := Unit_AbGroup; VectorSpace.scaling := Unit_ScaleLaws |}.
End UnitVS.
