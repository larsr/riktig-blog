b is bilinear when it is linear in each argument separately.

  Class Bilinear (F U V W : Type)

      {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}

      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
      {VW : VectorSpace F W}
      (b : U → V → W) := {
    proper :: Proper (equ ==> equ ==> equ) b;

    linear_l :: ∀ v : V, Linear F U W (fun u ⇒ b u v);
    linear_r :: ∀ u : U, Linear F V W (b u);
  }.

End Bilinear.

A bilinear map is a function of two arguments, linear in each.
Module BilinearMap.
  Class BilinearMap (F U V W : Type)
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
      {VU : VectorSpace F U}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
      {VW : VectorSpace F W} := {
    map : U → V → W;
    bilinear :: Bilinear F U V W map;
  }.
  Arguments map {F U V W Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV eW zW aW nW sW VW}.
  Arguments bilinear {F U V W Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV eW zW aW nW sW VW}.
End BilinearMap.

Section BilinearMapLaws.
  Context {F U V W : Type}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
    {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}
    {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
    {VW : VectorSpace F W}.

  Definition bilinear_map_proper (b : BilinearMap F U V W)
    : Proper (equ ==> equ ==> equ) (BilinearMap.map b) :=
    Bilinear.proper (b := BilinearMap.map b).

  Definition bilinear_left_additive (b : BilinearMap F U V W)
    : ∀ (u u' : U) (v : V), b (u + u') v == b u v + b u' v :=
    fun u u' v ⇒ additive (f := fun x ⇒ BilinearMap.map b x v) u u'.

  Definition bilinear_left_scale (b : BilinearMap F U V W)
    : ∀ (a : F) (u : U) (v : V), b (a *: u) v == a *: b u v :=
    fun a u v ⇒ homogeneous (f := fun x ⇒ BilinearMap.map b x v) a u.

  Definition bilinear_right_additive (b : BilinearMap F U V W)
    : ∀ (u : U) (v v' : V), b u (v + v') == b u v + b u v' :=
    fun u v v' ⇒ additive (f := BilinearMap.map b u) v v'.

  Definition bilinear_right_scale (b : BilinearMap F U V W)
    : ∀ (a : F) (u : U) (v : V), b u (a *: v) == a *: b u v :=
    fun a u v ⇒ homogeneous (f := BilinearMap.map b u) a v.

  Definition bilinear_map_from_laws (b : U → V → W)
      (b_proper : Proper (equ ==> equ ==> equ) b)
      (b_left_additive : ∀ (u u' : U) (v : V), b (u + u') v == b u v + b u' v)
      (b_left_scale : ∀ (a : F) (u : U) (v : V), b (a *: u) v == a *: b u v)
      (b_right_additive : ∀ (u : U) (v v' : V), b u (v + v') == b u v + b u v')
      (b_right_scale : ∀ (a : F) (u : U) (v : V), b u (a *: v) == a *: b u v)
    : BilinearMap F U V W :=
    {| BilinearMap.map := b;
       BilinearMap.bilinear :=
         {| Bilinear.proper := b_proper;
            Bilinear.linear_l := fun v ⇒
              {| Linear.proper := fun x x' Hx ⇒ b_proper x x' Hx v v (reflexivity v);
                 Linear.is_additive := fun x x' ⇒ b_left_additive x x' v;
                 Linear.is_homogeneous := fun a x ⇒ b_left_scale a x v |};
            Bilinear.linear_r := fun u ⇒
              {| Linear.proper := fun y y' Hy ⇒ b_proper u u (reflexivity u) y y' Hy;
                 Linear.is_additive := fun y y' ⇒ b_right_additive u y y';
                 Linear.is_homogeneous := fun a y ⇒ b_right_scale a u y |} |} |}.
End BilinearMapLaws.

Definition famEq {V : Type} {EV : EqRel V} {I : Type}
    (vs ws : I → V) : Prop :=
  ∀ i : I, vs i == ws i.

upd vs i x is the family vs with slot i replaced by x.
Class HasUpd (I V : Type) := upd : (I → V) → I → V → I → V.
Hint Mode HasUpd ! ! : typeclass_instances.
Typeclasses Transparent HasUpd.

Module Upd.
  Class Upd (I V : Type) {eI : EqRel I} {eV : EqRel V}
      {Vu : HasUpd I V} := {
    upd_at : ∀ (vs : I → V) (i : I) (x : V) (j : I),
                j == i → upd vs i x j == x;
    upd_off : ∀ (vs : I → V) (i j : I) (x : V),
                ¬ (i == j) → upd vs i x j == vs j;
  }.
  Hint Mode Upd ! ! - - - : typeclass_instances.
End Upd.

form is multilinear when it is linear in each of its arguments separately.
Module Multilinear.
  Class Multilinear (I F V : Type) {eI : EqRel I} {Ieq : EquDec I}
      {Vu : HasUpd I V}
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      (form : (I → V) → F) := {
    proper :: Proper (@famEq V eV I ==> equ) form;

    is_additive :: ∀ (vs : I → V) (i : I),
      Additive V F add add (fun x ⇒ form (upd vs i x));
    is_homogeneous :: ∀ (vs : I → V) (i : I),
      Homogeneous F V F mult (fun a ⇒ a) (fun x ⇒ form (upd vs i x));
  }.
  Hint Mode Multilinear - - - - - - - - - - - - - - - - - - - - ! : typeclass_instances.
End Multilinear.
Export Multilinear.

A multilinear form on V over the index carrier I is a function of an I-family of vectors, linear in each argument.
Module MultilinearForm.
  Class MultilinearForm (I F V : Type) {eI : EqRel I} {Ieq : EquDec I}
      {Vu : HasUpd I V}
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V} := {
    map : (I → V) → F;
    multilinear :: Multilinear I F V map;
  }.
  Arguments map {I F V eI Ieq Vu Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.
  Arguments multilinear {I F V eI Ieq Vu Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV}.
End MultilinearForm.

Section MultilinearFormLaws.
  Context {I F V : Type} {eI : EqRel I} {Ieq : EquDec I} {Vu : HasUpd I V}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}.

  Definition multilinear_form_proper (m : MultilinearForm I F V)
    : Proper (@famEq V eV I ==> equ) (MultilinearForm.map m) :=
    Multilinear.proper (form := MultilinearForm.map m).

  Definition multilinear_form_additive (m : MultilinearForm I F V)
    : ∀ (vs : I → V) (i : I) (x y : V),
        m (upd vs i (x + y)) == m (upd vs i x) + m (upd vs i y) :=
    fun vs i ⇒ additive (f := fun x ⇒ MultilinearForm.map m (upd vs i x)).

  Definition multilinear_form_scale (m : MultilinearForm I F V)
    : ∀ (vs : I → V) (i : I) (a : F) (x : V),
        m (upd vs i (a *: x)) == a × m (upd vs i x) :=
    fun vs i ⇒ homogeneous (f := fun x ⇒ MultilinearForm.map m (upd vs i x)).

  Definition multilinear_form_from_laws (form : (I → V) → F)
      (form_proper : Proper (@famEq V eV I ==> equ) form)
      (form_additive : ∀ (vs : I → V) (i : I) (x y : V),
         form (upd vs i (x + y)) == form (upd vs i x) + form (upd vs i y))
      (form_scale : ∀ (vs : I → V) (i : I) (a : F) (x : V),
         form (upd vs i (a *: x)) == a × form (upd vs i x))
    : MultilinearForm I F V :=
    {| MultilinearForm.map := form;
       MultilinearForm.multilinear :=
         {| Multilinear.proper := form_proper;
            Multilinear.is_additive := form_additive;
            Multilinear.is_homogeneous := form_scale |} |}.
End MultilinearFormLaws.

A form is alternating when it vanishes on every family with two equal entries.
Module Alternating.
  Class Alternating {I F V : Type} {eI : EqRel I}
      {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
      {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VV : VectorSpace F V}
      (form : (I → V) → F) :=
    repeat_zero : ∀ (vs : I → V) (i j : I),
      ¬ (i == j) → vs i == vs j → form vs == 0.
  Hint Mode Alternating - - - - - - - - - - - - - - - - - - ! : typeclass_instances.
End Alternating.
Notation Alternating := Alternating.Alternating.

Section AlternatingLaw.
  Context {I F V : Type} {eI : EqRel I} {Ieq : EquDec I} {Vu : HasUpd I V}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}.

  Definition alternating_repeat_zero {m : MultilinearForm I F V}
      (alt : Alternating m)
    : ∀ (vs : I → V) (i j : I),
        ¬ (i == j) → vs i == vs j → m vs == 0 := alt.

  Definition alternating_from_law (form : (I → V) → F)
      (form_repeat_zero : ∀ (vs : I → V) (i j : I),
         ¬ (i == j) → vs i == vs j → form vs == 0)
    : Alternating form := form_repeat_zero.
End AlternatingLaw.
