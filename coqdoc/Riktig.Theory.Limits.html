Library Riktig.Theory.Limits

From Coq Require Export Morphisms PeanoNat.
From Riktig Require Export Theory.Norm Lib.NatFacts.

Section Eventually.
  Context {F V : Type} `{NormedSpace F V}.
  Context {I : Type}.

  Definition ev_weakens (Ev : (I → Prop) → Prop) : Prop :=
    ∀ P Q : I → Prop, (∀ i : I, P i → Q i) → Ev P → Ev Q.

  Definition ev_meets (Ev : (I → Prop) → Prop) : Prop :=
    ∀ P Q : I → Prop, Ev P → Ev Q → Ev (fun i : I ⇒ P i ∧ Q i).

  Definition ev_inhabited (Ev : (I → Prop) → Prop) : Prop :=
    ∀ P : I → Prop, Ev P → ∃ i : I, P i.

  Definition tendsto (Ev : (I → Prop) → Prop) (x : I → V) (L : V) : Prop :=
    ∀ eps : F, 0 < eps → Ev (fun i : I ⇒ ‖ x i + - L ‖ < eps).

  Theorem tendsto_unique : ∀ Ev : (I → Prop) → Prop,
    ev_meets Ev → ev_inhabited Ev →
    ∀ (x : I → V) (L1 L2 : V),
      tendsto Ev x L1 → tendsto Ev x L2 → L1 == L2.
  Proof.
    intros Ev Hmeets Hinhab x L1 L2 Hx1 Hx2.
    apply sub_eq_zero.
    apply definite.
    apply nonneg_below_all_pos_is_zero.
    - apply nonneg.
    - intros eps Heps.
      destruct (Hinhab _ (Hmeets _ _
          (Hx1 (half eps) (half_pos eps Heps))
          (Hx2 (half eps) (half_pos eps Heps))))
        as [i [Hi1 Hi2]].
      apply lt_le.
      apply (le_lt_trans (‖ L1 + - L2 ‖)
        (‖ L1 + - x i ‖ + ‖ x i + - L2 ‖) eps).
      + apply norm_sub_triangle.
      + apply add_half_lt.
        × rewrite (norm_sub_swap L1 (x i)). exact Hi1.
        × exact Hi2.
  Qed.

  Theorem tendsto_add : ∀ Ev : (I → Prop) → Prop,
    ev_weakens Ev → ev_meets Ev →
    ∀ (x y : I → V) (L M : V),
      tendsto Ev x L → tendsto Ev y M →
      tendsto Ev (fun i : I ⇒ x i + y i) (L + M).
  Proof.
    intros Ev Hweak Hmeets x y L M Hx Hy eps Heps.
    apply (Hweak (fun i : I ⇒
      ‖ x i + - L ‖ < half eps ∧ ‖ y i + - M ‖ < half eps)).
    - intros i [Hi1 Hi2].
      apply (le_lt_trans _ (‖ x i + - L ‖ + ‖ y i + - M ‖) eps).
      + apply norm_sub_pair.
      + apply add_half_lt.
        × exact Hi1.
        × exact Hi2.
    - exact (Hmeets _ _ (Hx (half eps) (half_pos eps Heps))
               (Hy (half eps) (half_pos eps Heps))).
  Qed.

  Theorem tendsto_scale : ∀ Ev : (I → Prop) → Prop,
    ev_weakens Ev →
    ∀ (a : F) (x : I → V) (L : V),
      tendsto Ev x L → tendsto Ev (fun i : I ⇒ a *: x i) (a *: L).
  Proof.
    intros Ev Hweak a x L Hx eps Heps.
    apply (Hweak (fun i : I ⇒ ‖ x i + - L ‖ < eps × inv (1 + abs a))).
    - intros i Hi.
      apply (le_lt_trans _ (abs a × ‖ x i + - L ‖) eps).
      + apply eq_le. apply norm_scale_sub.
      + apply mul_lt_of_lt_pad_div.
        × apply abs_nonneg.
        × apply nonneg.
        × exact Hi.
    - exact (Hx (eps × inv (1 + abs a))
               (pad_div_pos eps (abs a) Heps (abs_nonneg a))).
  Qed.

  Theorem tendsto_neg : ∀ Ev : (I → Prop) → Prop,
    ev_weakens Ev →
    ∀ (x : I → V) (L : V),
      tendsto Ev x L → tendsto Ev (fun i : I ⇒ - x i) (- L).
  Proof.
    intros Ev Hweak x L Hx eps Heps.
    apply (Hweak (fun i : I ⇒ ‖ x i + - L ‖ < eps)).
    - intros i Hi.
      apply (le_lt_trans _ (‖ x i + - L ‖) eps).
      + apply eq_le. apply norm_neg_add.
      + exact Hi.
    - exact (Hx eps Heps).
  Qed.

End Eventually.

Section Limits.
  Context {F V : Type} `{NormedSpace F V}.

  Definition seq_eventually (P : nat → Prop) : Prop :=
    ∃ N : nat, ∀ n : nat, (N ≤ n)%nat → P n.

  Lemma seq_ev_weakens : ev_weakens seq_eventually.
  Proof.
    intros P Q HPQ [N HN].
    ∃ N.
    intros n Hn.
    exact (HPQ n (HN n Hn)).
  Qed.

  Lemma seq_ev_meets : ev_meets seq_eventually.
  Proof.
    intros P Q [N1 HN1] [N2 HN2].
    ∃ (Nat.max N1 N2).
    intros n Hn.
    split.
    - exact (HN1 n (max_le_l N1 N2 n Hn)).
    - exact (HN2 n (max_le_r N1 N2 n Hn)).
  Qed.

  Lemma seq_ev_inhabited : ev_inhabited seq_eventually.
  Proof.
    intros P [N HN].
    ∃ N.
    exact (HN N (Nat.le_refl N)).
  Qed.

  Definition seq_lim (x : nat → V) (L : V) : Prop :=
    tendsto seq_eventually x L.

  Theorem lim_const : ∀ v : V, seq_lim (fun _ ⇒ v) v.
  Proof.
    intros v eps Heps.
    ∃ 0%nat.
    intros n Hn.
    apply (le_lt_trans _ 0 eps).
    - apply eq_le. apply norm_sub_self.
    - exact Heps.
  Qed.

  Theorem lim_unique : ∀ (x : nat → V) (L1 L2 : V),
    seq_lim x L1 → seq_lim x L2 → L1 == L2.
  Proof.
    exact (tendsto_unique seq_eventually seq_ev_meets seq_ev_inhabited).
  Qed.

  Theorem lim_add : ∀ (x y : nat → V) (L M : V),
    seq_lim x L → seq_lim y M → seq_lim (fun n ⇒ x n + y n) (L + M).
  Proof.
    exact (tendsto_add seq_eventually seq_ev_weakens seq_ev_meets).
  Qed.

  Theorem lim_scale : ∀ (a : F) (x : nat → V) (L : V),
    seq_lim x L → seq_lim (fun n ⇒ a *: x n) (a *: L).
  Proof.
    exact (tendsto_scale seq_eventually seq_ev_weakens).
  Qed.

  Theorem lim_neg : ∀ (x : nat → V) (L : V),
    seq_lim x L → seq_lim (fun n ⇒ - x n) (- L).
  Proof.
    exact (tendsto_neg seq_eventually seq_ev_weakens).
  Qed.

  Theorem lim_subseq : ∀ (x : nat → V) (L : V) (phi : nat → nat),
    (∀ n : nat, (n ≤ phi n)%nat) →
    seq_lim x L → seq_lim (fun n ⇒ x (phi n)) L.
  Proof.
    intros x L phi Hphi Hx eps Heps.
    destruct (Hx eps Heps) as [N HN].
    ∃ N.
    intros n Hn.
    exact (HN (phi n) (Nat.le_trans N n (phi n) Hn (Hphi n))).
  Qed.

  Theorem lim_bounded : ∀ (x : nat → V) (L : V), seq_lim x L →
    ∃ M : F, (0 ≤ M) ∧ (∀ n : nat, ‖ x n ‖ ≤ M).
  Proof.
    intros x L Hx.
    destruct (Hx 1 lt_0_1) as [N HN].
    destruct (finite_bound (fun n ⇒ ‖ x n ‖) N) as [M0 [HM0 Hearly]].
    assert (Htail : 0 ≤ ‖ L ‖ + 1)
      by exact (le_transitive _ _ _ (nonneg L) (le_add_nonneg_r (‖ L ‖) 1
                  (lt_le 0 1 lt_0_1))).
    ∃ (M0 + (‖ L ‖ + 1)).
    split.
    - exact (le_transitive _ _ _ HM0 (le_add_nonneg_r M0 (‖ L ‖ + 1) Htail)).
    - intros n.
      destruct (Nat.le_gt_cases N n) as [Hlate | Hearly'].
      + calc (‖ x n ‖)
          == (‖ x n + - L ‖ + ‖ L ‖) { norm_le_sub_add }
          == ((1 : F) + ‖ L ‖)
               { (add_le_mono_r (‖ x n + - L ‖) 1 (‖ L ‖)
                    (lt_le _ _ (HN n Hlate))) }
          == (‖ L ‖ + (1 : F)) { add_commutative }
          == (M0 + (‖ L ‖ + 1)) { (le_add_nonneg_l M0 (‖ L ‖ + 1) HM0) }.
      + calc (‖ x n ‖)
          == M0 { (Hearly n Hearly') }
          == (M0 + (‖ L ‖ + 1)) { (le_add_nonneg_r M0 (‖ L ‖ + 1) Htail) }.
  Qed.

End Limits.

Instance seq_lim_proper {F V : Type} `{NormedSpace F V} :
  Proper (pointwise_relation nat equ ==> equ ==> iff) (seq_lim (F:=F) (V:=V)).
Proof.
  intros x y Hxy L M HLM.
  split.
  - intros Hx eps Heps.
    destruct (Hx eps Heps) as [N HN].
    ∃ N.
    intros n Hn.
    rewrite <- (Hxy n), <- HLM.
    apply HN.
    exact Hn.
  - intros Hy eps Heps.
    destruct (Hy eps Heps) as [N HN].
    ∃ N.
    intros n Hn.
    rewrite (Hxy n), HLM.
    apply HN.
    exact Hn.
Qed.
