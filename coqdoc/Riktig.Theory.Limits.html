Section LimitsTheory.
  Context {F N V : Type} `{Limits F N V}.

  Theorem tendsto_sub : ∀ (u v : N → V) (l m : V),
    tendsto u l → tendsto v m → tendsto (fun n ⇒ u n + - v n) (l + - m).
  Proof. intros u v l m Hu Hv. exact (tendsto_add u (fun n ⇒ - v n) l (- m) Hu (tendsto_neg v m Hv)). Defined.

  Theorem tendsto_const_add : ∀ (c : V) (u : N → V) (l : V),
    tendsto u l → tendsto (fun n ⇒ c + u n) (c + l).
  Proof. intros c u l Hu. exact (tendsto_add (fun _ ⇒ c) u c l (tendsto_const c) Hu). Defined.

  Theorem tendsto_zero_of_sub : ∀ (u : N → V) (l : V),
    tendsto (fun n ⇒ u n + - l) 0 → tendsto u l.
  Proof.
    intros u l Hd.
    refine (tendsto_proper (fun n ⇒ l + (u n + - l)) u (l + 0) l _ _
      (tendsto_const_add l (fun n ⇒ u n + - l) 0 Hd)).
    - intros n. exact (add_add_neg_cancel l (u n)).
    - exact (right_identity (action := add) l).
  Defined.

  Theorem cauchy_sub : ∀ u v : N → V, cauchy u → cauchy v → cauchy (fun n ⇒ u n + - v n).
  Proof. intros u v Hu Hv. exact (cauchy_add u (fun n ⇒ - v n) Hu (cauchy_neg v Hv)). Defined.

  Theorem cauchy_const : ∀ c : V, cauchy (fun _ : N ⇒ c).
  Proof. intros c. exact (cauchy_of_tendsto (fun _ ⇒ c) c (tendsto_const c)). Defined.
End LimitsTheory.

Section CompleteTheory.
  Context {F N V : Type} `{Complete F N V}.

The limit is the limit: whatever a Cauchy sequence converges to is its limit.
  Theorem lim_eq : ∀ (u : N → V) (Hu : cauchy u) (l : V), tendsto u l → lim u Hu == l.
  Proof. intros u Hu l Hl. exact (tendsto_unique u (lim u Hu) l (lim_tendsto u Hu) Hl). Qed.

  Theorem lim_proper : ∀ (u u' : N → V) (Hu : cauchy u) (Hu' : cauchy u'),
    (∀ n : N, u n == u' n) → lim u Hu == lim u' Hu'.
  Proof.
    intros u u' Hu Hu' Huu'.
    apply lim_eq.
    exact (tendsto_proper u' u (lim u' Hu') (lim u' Hu') (fun n ⇒ symmetry (Huu' n)) (reflexivity _)
             (lim_tendsto u' Hu')).
  Qed.

  Theorem lim_const : ∀ (c : V) (Hc : cauchy (fun _ : N ⇒ c)), lim (fun _ ⇒ c) Hc == c.
  Proof. intros c Hc. exact (lim_eq (fun _ ⇒ c) Hc c (tendsto_const c)). Qed.

  Theorem lim_add : ∀ (u v : N → V) (Hu : cauchy u) (Hv : cauchy v) (Huv : cauchy (fun n ⇒ u n + v n)),
    lim (fun n ⇒ u n + v n) Huv == lim u Hu + lim v Hv.
  Proof.
    intros u v Hu Hv Huv. apply lim_eq.
    exact (tendsto_add u v (lim u Hu) (lim v Hv) (lim_tendsto u Hu) (lim_tendsto v Hv)).
  Qed.

  Theorem lim_neg : ∀ (u : N → V) (Hu : cauchy u) (Hn : cauchy (fun n ⇒ - u n)),
    lim (fun n ⇒ - u n) Hn == - lim u Hu.
  Proof. intros u Hu Hn. apply lim_eq. exact (tendsto_neg u (lim u Hu) (lim_tendsto u Hu)). Qed.

  Theorem lim_smul : ∀ (a : F) (u : N → V) (Hu : cauchy u) (Ha : cauchy (fun n ⇒ a *: u n)),
    lim (fun n ⇒ a *: u n) Ha == a *: lim u Hu.
  Proof. intros a u Hu Ha. apply lim_eq. exact (tendsto_smul a u (lim u Hu) (lim_tendsto u Hu)). Qed.

  Theorem lim_tail : ∀ (u : N → V) (Hu : cauchy u) (Ht : cauchy (fun n ⇒ u (succ n))),
    lim (fun n ⇒ u (succ n)) Ht == lim u Hu.
  Proof. intros u Hu Ht. apply lim_eq. exact (tendsto_tail u (lim u Hu) (lim_tendsto u Hu)). Qed.
End CompleteTheory.
