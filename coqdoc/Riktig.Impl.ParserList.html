The list-of-successes parsers over Coq lists implement the Parser structure: the meaning of a parser is membership in its list of readings.
Section ParserList.
  Context {C : Type}.

  #[export] Instance ListParser_EqRel (A : Type) : EqRel (parser C A) :=
    fun p q ⇒ ∀ (s : list C) (a : A) (s' : list C), p_parses p s a s' ↔ p_parses q s a s'.
  #[export] Instance ListParser_HasReturn : HasReturn (parser C) := @p_ret C.
  #[export] Instance ListParser_HasBind : HasBind (parser C) := @p_bind C.
  #[export] Instance ListParser_HasFail : HasFail (parser C) := @p_fail C.
  #[export] Instance ListParser_HasAlt : HasAlt (parser C) := @p_alt C.
  #[export] Instance ListParser_HasItem : HasItem C (parser C) := p_item.
  #[export] Instance ListParser_HasParses : HasParses (list C) (parser C) := @p_parses C.

  #[export] Instance ListParser : Parser C (list C) (parser C) (eqC := eq)
      ListParser_HasReturn ListParser_HasBind ListParser_HasFail ListParser_HasAlt ListParser_HasItem ListParser_HasParses.
  Proof.
    split.
    - intros A p q. split; intros Hpq; exact Hpq.
    - intros A p s s' Hs a a' Ha t t' Ht. unfold equ, List_EqRel in Hs, Ht. subst. reflexivity.
    - intros A a s a' s'. exact (p_parses_ret A a s a' s').
    - intros A B p f s b s''. exact (p_parses_bind A B p f s b s'').
    - intros A s a s'. exact (p_parses_fail A s a s').
    - intros A p q s a s'. exact (p_parses_alt A p q s a s').
    - intros s c s'. exact (p_parses_item s c s').
  Qed.
End ParserList.
