From Riktig.Structure Require Export ListRec.
From Riktig.Theory Require Import InitialAlgebra List.
From Riktig.Impl Require Export CategoryFun SignaturePoly.

Any list datatype at Coq's types recurses: the empty list and prepending are the two constructors, the recursion is the fold, and the equations are the fold's own, read at one level.
Section S.
  Context {X L : Type} {eqX : EqRel X}
    {lb : HasBuild Type (List.shape (fun T0 : Type ⇒ T0) X) L}
    {lf : HasCata Type (List.shape (fun T0 : Type ⇒ T0) X) L}
    `{!List Type (fun T0 : Type ⇒ T0) X L lb lf}.

  #[export] Instance List_HasEmpty : HasEmpty L := lb (inl tt).
  #[export] Instance List_HasPrepend : HasPrepend X L := fun x l ⇒ lb (inr (x, l)).
  #[export] Instance List_HasListRec : HasListRec X L :=
    fun B n c ⇒ lf B (fun s ⇒ match s with inl _ ⇒ n | inr (x, r) ⇒ c x r end).

  #[export] Instance List_ListRec :
      ListRec X L (eqL := eq) List_HasEmpty List_HasPrepend List_HasListRec.
  Proof.
    split.
    - intros B eB EB n c Hc l l' Hl. rewrite Hl. reflexivity.
    - intros B eB EB n c.
      assert (E : list_rec B n c (empty : L) = n).
      { exact (InitialAlgebra.cata_build (t := L) B _ (inl tt)). }
      rewrite E. reflexivity.
    - intros B eB EB n c x l.
      assert (E : list_rec B n c (prepend x l) = c x (list_rec B n c l)).
      { exact (InitialAlgebra.cata_build (t := L) B _ (inr (x, l))). }
      rewrite E. reflexivity.
  Qed.
End S.
