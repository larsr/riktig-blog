Lemma swap_fin_hit : ∀ (n : nat) (i j : Fin.t n), swap_fin i j i = j.
Proof.
  intros n i j.
  unfold swap_fin.
  destruct (dec (i = i)) as [_ | Hne].
  - reflexivity.
  - destruct (Hne eq_refl).
Qed.

Lemma swap_fin_FSFS_F1 : ∀ (n : nat) (i j : Fin.t n),
  swap_fin (Fin.FS i) (Fin.FS j) Fin.F1 = Fin.F1.
Proof.
  intros n i j.
  unfold swap_fin.
  destruct (dec ((Fin.F1 : Fin.t (S n)) = Fin.FS i)) as [He1 | _].
  - destruct (fin_F1_neq_FS n i He1).
  - destruct (dec ((Fin.F1 : Fin.t (S n)) = Fin.FS j)) as [He2 | _].
    + destruct (fin_F1_neq_FS n j He2).
    + reflexivity.
Qed.

Lemma swap_fin_FSFS_FS : ∀ (n : nat) (i j s : Fin.t n),
  swap_fin (Fin.FS i) (Fin.FS j) (Fin.FS s) = Fin.FS (swap_fin i j s).
Proof.
  intros n i j s.
  unfold swap_fin.
  destruct (dec (Fin.FS s = Fin.FS i)) as [HeqI | HneI].
  - destruct (dec (s = i)) as [_ | Hni].
    + reflexivity.
    + destruct (Hni (Fin.FS_inj s i HeqI)).
  - destruct (dec (s = i)) as [Heqi | _].
    + destruct (HneI (f_equal Fin.FS Heqi)).
    + destruct (dec (Fin.FS s = Fin.FS j)) as [HeqJ | HneJ].
      × destruct (dec (s = j)) as [_ | Hnj].
        -- reflexivity.
        -- destruct (Hnj (Fin.FS_inj s j HeqJ)).
      × destruct (dec (s = j)) as [Heqj | _].
        -- destruct (HneJ (f_equal Fin.FS Heqj)).
        -- reflexivity.
Qed.

  Lemma cancel_unit : ∀ f g x : F, f × g == 1 → f × x == 0 → x == 0.
  Proof.
    intros f g x Hfg Hfx.
    calc x
      == ((1 : F) × x) { (left_identity x) }
      == ((f × g) × x) { Hfg }
      == ((g × f) × x) { (mul_commutative f g) }
      == (g × (f × x)) { (associative g f x) }
      == (g × (0 : F)) { Hfx }
      == (0 : F) { (mul_0_r g) }.
  Qed.

  Theorem det_zero_row : ∀ (n : nat) (A : Matrix n n F) (i : Fin.t n),
    (∀ j : Fin.t n, A i j == 0) → detM A == 0.
  Proof.
    intros n A i Hz.
    destruct n as [| p].
    - exact (Fin.case0 (fun i0 : Fin.t 0 ⇒ detM A == 0) i).
    - calc (detM A)
        == (detM (upd_row A i (fun k ⇒ 0 × A i k)))
        == ((0 : F) × detM (upd_row A i (A i)))
        == (0 : F).
      + apply (detM_proper (S p)).
        intros r c.
        destruct (Fin.eq_dec r i) as [Heq | Hne].
        × rewrite Heq, (upd_row_at (S p) A i (fun k ⇒ 0 × A i k) c), (Hz c).
          ring.
        × symmetry.
          exact (upd_row_ne (S p) A i r (fun k ⇒ 0 × A i k) c Hne).
      + exact (det_rowI_scale p i A 0 (A i)).
      + ring.
  Qed.

  Lemma mat_mul_zero_row : ∀ (n p : nat) (A : Matrix n n F)
      (B : Matrix n p F) (i : Fin.t n),
    (∀ j : Fin.t n, A i j == 0) →
    ∀ c : Fin.t p, mat_mul A B i c == 0.
  Proof.
    intros n p A B i Hz c.
    calc (mat_mul A B i c)
      == (bigopF (fun k : Fin.t n ⇒ (0 : F)))
      == (0 : F).
    - apply bigopF_ext. intros k.
      calc (A i k × B k c)
        == ((0 : F) × B k c) { (Hz k) }
        == (0 : F) { (mul_0_l (B k c)) }.
    - exact (bigopF_zero n).
  Qed.

  Theorem det_upper_triangular : ∀ (n : nat) (A : Matrix n n F),
    (∀ i j : Fin.t n, (fin_to_nat j < fin_to_nat i)%nat → A i j == 0) →
    detM A == prodF (fun i ⇒ A i i).
  Proof.
    intros n; induction n as [| p IH]; intros A Hup.
    - reflexivity.
    - assert (Hcol : ∀ s : Fin.t p, A (Fin.FS s) Fin.F1 == 0).
      { intros s. apply (Hup (Fin.FS s) Fin.F1).
        rewrite (fin_to_nat_F1 p), (fin_to_nat_FS p s). lia. }
      rewrite (detM_S p A),
        (bigopF_S p (fun i ⇒ sign_at i × (A i Fin.F1 × detM (minor A i)))),
        (bigopF_ext p
           (fun s : Fin.t p ⇒ sign_at (Fin.FS s)
              × (A (Fin.FS s) Fin.F1 × detM (minor A (Fin.FS s))))
           (fun _ : Fin.t p ⇒ (0 : F))
           (fun s ⇒ det_term_col0 p A (Fin.FS s) (Hcol s))),
        (bigopF_zero p),
        (sign_at_F1 p),
        (left_identity (A Fin.F1 Fin.F1 × detM (minor A Fin.F1))),
        (right_identity (A Fin.F1 Fin.F1 × detM (minor A Fin.F1))).
      assert (Hmin : ∀ s c : Fin.t p,
        (fin_to_nat c < fin_to_nat s)%nat → minor A Fin.F1 s c == 0).
      { intros s c Hlt.
        rewrite (minor_def p A Fin.F1 s c), (skip_F1 p s).
        apply (Hup (Fin.FS s) (Fin.FS c)).
        rewrite (fin_to_nat_FS p s), (fin_to_nat_FS p c). lia. }
      rewrite (IH (minor A Fin.F1) Hmin).
      assert (Hpe : ∀ s : Fin.t p,
        minor A Fin.F1 s s == A (Fin.FS s) (Fin.FS s)).
      { intros s.
        rewrite (minor_def p A Fin.F1 s s), (skip_F1 p s).
        reflexivity. }
      rewrite (prodF_ext p (fun s ⇒ minor A Fin.F1 s s)
                 (fun s ⇒ A (Fin.FS s) (Fin.FS s)) Hpe).
      symmetry.
      exact (prodF_S p (fun i ⇒ A i i)).
  Qed.

  Inductive ElemOp (n : nat) : Type :=
  | elem_op_swap (i j : Fin.t n) (Hij : i ≠ j)
  | elem_op_scale (i : Fin.t n) (c d : F) (Hcd : c × d == 1)
  | elem_op_add (i j : Fin.t n) (c : F) (Hij : i ≠ j).

  Definition elem_op_mat {n : nat} (op : ElemOp n) : Matrix n n F :=
    match op with
    | elem_op_swap _ i j _ ⇒ elem_swap i j
    | elem_op_scale _ i c _ _ ⇒ elem_scale i c
    | elem_op_add _ i j c _ ⇒ elem_add i j c
    end.

  Definition elem_op_factor {n : nat} (op : ElemOp n) : F :=
    match op with
    | elem_op_swap _ _ _ _ ⇒ - (1)
    | elem_op_scale _ _ c _ _ ⇒ c
    | elem_op_add _ _ _ _ _ ⇒ 1
    end.

  Definition elem_op_cofactor {n : nat} (op : ElemOp n) : F :=
    match op with
    | elem_op_swap _ _ _ _ ⇒ - (1)
    | elem_op_scale _ _ _ d _ ⇒ d
    | elem_op_add _ _ _ _ _ ⇒ 1
    end.

  Definition elem_op_apply {n : nat} (op : ElemOp n) (M : Matrix n n F)
      : Matrix n n F :=
    mat_mul (elem_op_mat op) M.

  Lemma elem_op_factor_cofactor : ∀ (n : nat) (op : ElemOp n),
    elem_op_factor op × elem_op_cofactor op == 1.
  Proof.
    intros n op; destruct op as [i j Hij | i c d Hcd | i j c Hij];
      cbn [elem_op_factor elem_op_cofactor].
    - ring.
    - exact Hcd.
    - ring.
  Qed.

  Lemma elem_op_add_at : ∀ (n : nat) (i j : Fin.t n) (c : F) (Hij : i ≠ j)
      (M : Matrix n n F) (k : Fin.t n),
    elem_op_apply (elem_op_add n i j c Hij) M i k == M i k + c × M j k.
  Proof.
    intros n i j c Hij M k.
    exact (elem_add_mul_eq n n i j c M k).
  Qed.

  Lemma elem_op_add_ne : ∀ (n : nat) (i j : Fin.t n) (c : F) (Hij : i ≠ j)
      (M : Matrix n n F) (r k : Fin.t n),
    r ≠ i → elem_op_apply (elem_op_add n i j c Hij) M r k == M r k.
  Proof.
    intros n i j c Hij M r k Hr.
    exact (elem_add_mul_ne n n i j r c M k Hr).
  Qed.

  Lemma elem_scale_mul_eq_upd_row : ∀ (n : nat) (i : Fin.t n) (c : F)
      (M : Matrix n n F),
    mat_mul (elem_scale i c) M == upd_row M i (fun k ⇒ c × M i k).
  Proof.
    intros n i c M r0 k.
    destruct (Fin.eq_dec r0 i) as [Heq | Hne].
    - rewrite Heq.
      calc (mat_mul (elem_scale i c) M i k)
        == (c × M i k)
           { (elem_scale_mul_eq n n i c M k) }
        == (upd_row M i (fun k0 ⇒ c × M i k0) i k)
           { (upd_row_at n M i (fun k0 ⇒ c × M i k0) k) }.
    - calc (mat_mul (elem_scale i c) M r0 k)
        == (M r0 k)
           { (elem_scale_mul_ne n n i r0 c M k Hne) }
        == (upd_row M i (fun k0 ⇒ c × M i k0) r0 k)
           { (upd_row_ne n M i r0 (fun k0 ⇒ c × M i k0) k Hne) }.
  Qed.

  Lemma det_eop_scale : ∀ (n : nat) (i : Fin.t n) (c : F)
      (M : Matrix n n F),
    detM (mat_mul (elem_scale i c) M) == c × detM M.
  Proof.
    intros n i c M.
    destruct n as [| p].
    - exact (Fin.case0
        (fun i0 : Fin.t 0 ⇒ detM (mat_mul (elem_scale i0 c) M) == c × detM M)
        i).
    - calc (detM (mat_mul (elem_scale i c) M))
        == (detM (upd_row M i (fun k ⇒ c × M i k)))
             { (detM_proper (S p) _ _ (elem_scale_mul_eq_upd_row (S p) i c M)) }
        == (c × detM (upd_row M i (M i)))
             { (det_rowI_scale p i M c (M i)) }
        == (c × detM M)
             { (detM_proper (S p) _ _ (upd_row_self (S p) M i)) }.
  Qed.

  Theorem det_eop_apply : ∀ (n : nat) (op : ElemOp n) (M : Matrix n n F),
    detM (elem_op_apply op M) == elem_op_factor op × detM M.
  Proof.
    intros n op M; destruct op as [i j Hij | i c d Hcd | i j c Hij];
      cbn [elem_op_factor elem_op_mat elem_op_apply].
    - calc (detM (mat_mul (elem_swap i j) M))
        == (- detM M) { (det_elem_swap n i j Hij M) }
        == (- (1) × detM M) { tac: ring }.
    - exact (det_eop_scale n i c M).
    - calc (detM (mat_mul (elem_add i j c) M))
        == (detM M) { (det_elem_add n i j c M Hij) }
        == (1 × detM M) { (left_identity (detM M)) }.
  Qed.

  Fixpoint elem_ops_apply {n : nat} (l : list (ElemOp n)) (M : Matrix n n F)
      {struct l} : Matrix n n F :=
    match l with
    | nil ⇒ M
    | op :: l' ⇒ elem_ops_apply l' (elem_op_apply op M)
    end.

  Fixpoint elem_ops_factor {n : nat} (l : list (ElemOp n)) {struct l} : F :=
    match l with
    | nil ⇒ 1
    | op :: l' ⇒ elem_op_factor op × elem_ops_factor l'
    end.

  Fixpoint elem_ops_cofactor {n : nat} (l : list (ElemOp n)) {struct l} : F :=
    match l with
    | nil ⇒ 1
    | op :: l' ⇒ elem_op_cofactor op × elem_ops_cofactor l'
    end.

  Lemma elem_op_apply_proper : ∀ (n : nat) (op : ElemOp n)
      (M M' : Matrix n n F),
    M == M' → elem_op_apply op M == elem_op_apply op M'.
  Proof.
    intros n op M M' HM.
    exact (mat_mul_proper n n n (elem_op_mat op) (elem_op_mat op)
             (fun i j ⇒ reflexivity (elem_op_mat op i j)) M M' HM).
  Qed.

  Lemma elem_ops_apply_proper : ∀ (n : nat) (l : list (ElemOp n))
      (M M' : Matrix n n F),
    M == M' → elem_ops_apply l M == elem_ops_apply l M'.
  Proof.
    intros n l; induction l as [| op l' IH]; intros M M' HM.
    - exact HM.
    - exact (IH (elem_op_apply op M) (elem_op_apply op M')
              (elem_op_apply_proper n op M M' HM)).
  Qed.

  Lemma elem_ops_apply_app : ∀ (n : nat) (l1 l2 : list (ElemOp n))
      (M : Matrix n n F),
    elem_ops_apply (l1 ++ l2) M = elem_ops_apply l2 (elem_ops_apply l1 M).
  Proof.
    intros n l1; induction l1 as [| op l1' IH]; intros l2 M.
    - reflexivity.
    - exact (IH l2 (elem_op_apply op M)).
  Qed.

  Theorem det_eops_apply : ∀ (n : nat) (l : list (ElemOp n))
      (M : Matrix n n F),
    detM (elem_ops_apply l M) == elem_ops_factor l × detM M.
  Proof.
    intros n l; induction l as [| op l' IH]; intros M.
    - symmetry. exact (left_identity (detM M)).
    - cbn [elem_ops_apply elem_ops_factor].
      calc (detM (elem_ops_apply l' (elem_op_apply op M)))
        == (elem_ops_factor l' × detM (elem_op_apply op M))
             { (IH (elem_op_apply op M)) }
        == (elem_ops_factor l' × (elem_op_factor op × detM M))
             { (det_eop_apply n op M) }
        == (elem_op_factor op × (elem_ops_factor l' × detM M))
             { (mul_left_commutative (elem_ops_factor l') (elem_op_factor op) (detM M)) }
        == ((elem_op_factor op × elem_ops_factor l') × detM M)
             { (associative (elem_op_factor op) (elem_ops_factor l') (detM M)) }.
  Qed.

  Lemma elem_ops_factor_cofactor : ∀ (n : nat) (l : list (ElemOp n)),
    elem_ops_factor l × elem_ops_cofactor l == 1.
  Proof.
    intros n l; induction l as [| op l' IH].
    - exact (left_identity (1 : F)).
    - cbn [elem_ops_factor elem_ops_cofactor].
      calc ((elem_op_factor op × elem_ops_factor l') × (elem_op_cofactor op × elem_ops_cofactor l'))
        == ((elem_op_factor op × elem_op_cofactor op) × (elem_ops_factor l' × elem_ops_cofactor l'))
        == (1 : F).
      + ring.
      + rewrite (elem_op_factor_cofactor n op), IH.
        ring.
  Qed.

  Lemma elem_ops_apply_mul : ∀ (n : nat) (l : list (ElemOp n))
      (A B : Matrix n n F),
    elem_ops_apply l (mat_mul A B) == mat_mul (elem_ops_apply l A) B.
  Proof.
    intros n l; induction l as [| op l' IH]; intros A B.
    - reflexivity.
    - transitivity (elem_ops_apply l' (mat_mul (elem_op_apply op A) B)).
      + apply (elem_ops_apply_proper n l').
        intros i k.
        symmetry.
        exact (mat_mul_assoc n n n n (elem_op_mat op) A B i k).
      + exact (IH (elem_op_apply op A) B).
  Qed.

  Definition subm {p : nat} (M : Matrix (S p) (S p) F) : Matrix p p F :=
    fun s c ⇒ M (Fin.FS s) (Fin.FS c).

  Definition lift_eop {p : nat} (op : ElemOp p) : ElemOp (S p) :=
    match op with
    | elem_op_swap _ i j Hij ⇒
        elem_op_swap (S p) (Fin.FS i) (Fin.FS j)
          (fun Heq ⇒ Hij (Fin.FS_inj i j Heq))
    | elem_op_scale _ i c d Hcd ⇒ elem_op_scale (S p) (Fin.FS i) c d Hcd
    | elem_op_add _ i j c Hij ⇒
        elem_op_add (S p) (Fin.FS i) (Fin.FS j) c
          (fun Heq ⇒ Hij (Fin.FS_inj i j Heq))
    end.

  Lemma lift_eop_row_F1 : ∀ (p : nat) (op : ElemOp p)
      (M : Matrix (S p) (S p) F) (j : Fin.t (S p)),
    elem_op_apply (lift_eop op) M Fin.F1 j == M Fin.F1 j.
  Proof.
    intros p op M j; destruct op as [i1 i2 Hij | i1 c d Hcd | i1 i2 c Hij].
    - transitivity (M (swap_fin (Fin.FS i1) (Fin.FS i2) Fin.F1) j).
      + exact (elem_swap_mul (S p) (S p) (Fin.FS i1) (Fin.FS i2) M Fin.F1 j).
      + rewrite (swap_fin_FSFS_F1 p i1 i2). reflexivity.
    - exact (elem_scale_mul_ne (S p) (S p) (Fin.FS i1) Fin.F1 c M j
               (fin_F1_neq_FS p i1)).
    - exact (elem_add_mul_ne (S p) (S p) (Fin.FS i1) (Fin.FS i2) Fin.F1 c M j
               (fin_F1_neq_FS p i1)).
  Qed.

  Lemma lift_eop_sub : ∀ (p : nat) (op : ElemOp p)
      (M : Matrix (S p) (S p) F) (s c : Fin.t p),
    elem_op_apply (lift_eop op) M (Fin.FS s) (Fin.FS c) == elem_op_apply op (subm M) s c.
  Proof.
    intros p op M s c; destruct op as [i1 i2 Hij | i1 c0 d Hcd | i1 i2 c0 Hij].
    - transitivity (M (swap_fin (Fin.FS i1) (Fin.FS i2) (Fin.FS s)) (Fin.FS c)).
      + exact (elem_swap_mul (S p) (S p) (Fin.FS i1) (Fin.FS i2) M
                 (Fin.FS s) (Fin.FS c)).
      + rewrite (swap_fin_FSFS_FS p i1 i2 s).
        symmetry.
        exact (elem_swap_mul p p i1 i2 (subm M) s c).
    - destruct (Fin.eq_dec s i1) as [Heq | Hne].
      + rewrite Heq.
        transitivity (c0 × M (Fin.FS i1) (Fin.FS c)).
        × exact (elem_scale_mul_eq (S p) (S p) (Fin.FS i1) c0 M (Fin.FS c)).
        × symmetry. exact (elem_scale_mul_eq p p i1 c0 (subm M) c).
      + transitivity (M (Fin.FS s) (Fin.FS c)).
        × exact (elem_scale_mul_ne (S p) (S p) (Fin.FS i1) (Fin.FS s) c0 M
                   (Fin.FS c) (fun Heq ⇒ Hne (Fin.FS_inj s i1 Heq))).
        × symmetry. exact (elem_scale_mul_ne p p i1 s c0 (subm M) c Hne).
    - destruct (Fin.eq_dec s i1) as [Heq | Hne].
      + rewrite Heq.
        transitivity (M (Fin.FS i1) (Fin.FS c) + c0 × M (Fin.FS i2) (Fin.FS c)).
        × exact (elem_add_mul_eq (S p) (S p) (Fin.FS i1) (Fin.FS i2) c0 M
                   (Fin.FS c)).
        × symmetry. exact (elem_add_mul_eq p p i1 i2 c0 (subm M) c).
      + transitivity (M (Fin.FS s) (Fin.FS c)).
        × exact (elem_add_mul_ne (S p) (S p) (Fin.FS i1) (Fin.FS i2)
                   (Fin.FS s) c0 M (Fin.FS c)
                   (fun Heq ⇒ Hne (Fin.FS_inj s i1 Heq))).
        × symmetry. exact (elem_add_mul_ne p p i1 i2 s c0 (subm M) c Hne).
  Qed.

  Lemma lift_eop_col0 : ∀ (p : nat) (op : ElemOp p)
      (M : Matrix (S p) (S p) F),
    (∀ t : Fin.t p, M (Fin.FS t) Fin.F1 == 0) →
    ∀ s : Fin.t p, elem_op_apply (lift_eop op) M (Fin.FS s) Fin.F1 == 0.
  Proof.
    intros p op M Hz s; destruct op as [i1 i2 Hij | i1 c d Hcd | i1 i2 c Hij].
    - transitivity (M (swap_fin (Fin.FS i1) (Fin.FS i2) (Fin.FS s)) Fin.F1).
      + exact (elem_swap_mul (S p) (S p) (Fin.FS i1) (Fin.FS i2) M
                 (Fin.FS s) Fin.F1).
      + rewrite (swap_fin_FSFS_FS p i1 i2 s).
        exact (Hz (swap_fin i1 i2 s)).
    - destruct (Fin.eq_dec s i1) as [Heq | Hne].
      + rewrite Heq.
        transitivity (c × M (Fin.FS i1) Fin.F1).
        × exact (elem_scale_mul_eq (S p) (S p) (Fin.FS i1) c M Fin.F1).
        × rewrite (Hz i1). ring.
      + transitivity (M (Fin.FS s) Fin.F1).
        × exact (elem_scale_mul_ne (S p) (S p) (Fin.FS i1) (Fin.FS s) c M Fin.F1
                   (fun Heq ⇒ Hne (Fin.FS_inj s i1 Heq))).
        × exact (Hz s).
    - destruct (Fin.eq_dec s i1) as [Heq | Hne].
      + rewrite Heq.
        transitivity (M (Fin.FS i1) Fin.F1 + c × M (Fin.FS i2) Fin.F1).
        × exact (elem_add_mul_eq (S p) (S p) (Fin.FS i1) (Fin.FS i2) c M Fin.F1).
        × rewrite (Hz i1), (Hz i2). ring.
      + transitivity (M (Fin.FS s) Fin.F1).
        × exact (elem_add_mul_ne (S p) (S p) (Fin.FS i1) (Fin.FS i2)
                   (Fin.FS s) c M Fin.F1
                   (fun Heq ⇒ Hne (Fin.FS_inj s i1 Heq))).
        × exact (Hz s).
  Qed.

  Lemma lift_eops_row_F1 : ∀ (p : nat) (l : list (ElemOp p))
      (M : Matrix (S p) (S p) F) (j : Fin.t (S p)),
    elem_ops_apply (map lift_eop l) M Fin.F1 j == M Fin.F1 j.
  Proof.
    intros p l; induction l as [| op l' IH]; intros M j.
    - reflexivity.
    - transitivity (elem_op_apply (lift_eop op) M Fin.F1 j).
      + exact (IH (elem_op_apply (lift_eop op) M) j).
      + exact (lift_eop_row_F1 p op M j).
  Qed.

  Lemma lift_eops_sub : ∀ (p : nat) (l : list (ElemOp p))
      (M : Matrix (S p) (S p) F) (s c : Fin.t p),
    elem_ops_apply (map lift_eop l) M (Fin.FS s) (Fin.FS c)
      == elem_ops_apply l (subm M) s c.
  Proof.
    intros p l; induction l as [| op l' IH]; intros M s c.
    - reflexivity.
    - transitivity (elem_ops_apply l' (subm (elem_op_apply (lift_eop op) M)) s c).
      + exact (IH (elem_op_apply (lift_eop op) M) s c).
      + exact (elem_ops_apply_proper p l' (subm (elem_op_apply (lift_eop op) M))
                 (elem_op_apply op (subm M))
                 (fun s0 c0 ⇒ lift_eop_sub p op M s0 c0) s c).
  Qed.

  Lemma lift_eops_col0 : ∀ (p : nat) (l : list (ElemOp p))
      (M : Matrix (S p) (S p) F),
    (∀ t : Fin.t p, M (Fin.FS t) Fin.F1 == 0) →
    ∀ s : Fin.t p, elem_ops_apply (map lift_eop l) M (Fin.FS s) Fin.F1 == 0.
  Proof.
    intros p l; induction l as [| op l' IH]; intros M Hz s.
    - exact (Hz s).
    - exact (IH (elem_op_apply (lift_eop op) M) (lift_eop_col0 p op M Hz) s).
  Qed.

  Fixpoint clear_col_ops {p : nat} (M0 : Matrix (S p) (S p) F)
      (l : list (Fin.t p)) {struct l} : list (ElemOp (S p)) :=
    match l with
    | nil ⇒ nil
    | s :: l' ⇒
        elem_op_add (S p) (Fin.FS s) Fin.F1 (- M0 (Fin.FS s) Fin.F1)
          (fin_FS_neq_F1 p s)
        :: clear_col_ops M0 l'
    end.

  Lemma clear_col_spec : ∀ (p : nat) (l : list (Fin.t p))
      (M0 M : Matrix (S p) (S p) F),
    NoDup l →
    M Fin.F1 Fin.F1 == 1 →
    (∀ s : Fin.t p, In s l → M (Fin.FS s) Fin.F1 == M0 (Fin.FS s) Fin.F1) →
    (∀ j : Fin.t (S p),
       elem_ops_apply (clear_col_ops M0 l) M Fin.F1 j == M Fin.F1 j)
    ∧ (∀ s : Fin.t p, In s l →
          elem_ops_apply (clear_col_ops M0 l) M (Fin.FS s) Fin.F1 == 0)
    ∧ (∀ (s : Fin.t p) (j : Fin.t (S p)), ¬ In s l →
          elem_ops_apply (clear_col_ops M0 l) M (Fin.FS s) j == M (Fin.FS s) j).
  Proof.
    intros p l; induction l as [| s0 l' IH]; intros M0 M Hnd H11 Horig.
    - split; [intros j; reflexivity |
        split; [intros s Hin; destruct Hin | intros s j _; reflexivity]].
    - inversion Hnd as [| x xs Hnotin Hnd' Hxs]; subst.
      set (M1 := elem_op_apply
        (elem_op_add (S p) (Fin.FS s0) Fin.F1 (- M0 (Fin.FS s0) Fin.F1)
           (fin_FS_neq_F1 p s0)) M).
      assert (HM1F1 : ∀ j : Fin.t (S p), M1 Fin.F1 j == M Fin.F1 j).
      { intros j.
        exact (elem_op_add_ne (S p) (Fin.FS s0) Fin.F1 (- M0 (Fin.FS s0) Fin.F1)
                 (fin_FS_neq_F1 p s0) M Fin.F1 j (fin_F1_neq_FS p s0)). }
      assert (HM1ne : ∀ (s : Fin.t p) (j : Fin.t (S p)),
        s ≠ s0 → M1 (Fin.FS s) j == M (Fin.FS s) j).
      { intros s j Hs.
        exact (elem_op_add_ne (S p) (Fin.FS s0) Fin.F1 (- M0 (Fin.FS s0) Fin.F1)
                 (fin_FS_neq_F1 p s0) M (Fin.FS s) j
                 (fun Heq ⇒ Hs (Fin.FS_inj s s0 Heq))). }
      assert (HM1s0 : M1 (Fin.FS s0) Fin.F1 == 0).
      { transitivity (M (Fin.FS s0) Fin.F1
                      + (- M0 (Fin.FS s0) Fin.F1) × M Fin.F1 Fin.F1).
        - exact (elem_op_add_at (S p) (Fin.FS s0) Fin.F1 (- M0 (Fin.FS s0) Fin.F1)
                   (fin_FS_neq_F1 p s0) M Fin.F1).
        - rewrite H11, (Horig s0 (in_eq s0 l')).
          ring. }
      assert (H11' : M1 Fin.F1 Fin.F1 == 1).
      { transitivity (M Fin.F1 Fin.F1); [exact (HM1F1 Fin.F1) | exact H11]. }
      assert (Horig' : ∀ s : Fin.t p, In s l' →
        M1 (Fin.FS s) Fin.F1 == M0 (Fin.FS s) Fin.F1).
      { intros s Hin.
        assert (Hs : s ≠ s0)
          by (intros He; apply Hnotin; rewrite <- He; exact Hin).
        transitivity (M (Fin.FS s) Fin.F1).
        - exact (HM1ne s Fin.F1 Hs).
        - exact (Horig s (in_cons s0 s l' Hin)). }
      destruct (IH M0 M1 Hnd' H11' Horig') as [IH1 [IH2 IH3]].
      split; [| split].
      + intros j.
        transitivity (M1 Fin.F1 j); [exact (IH1 j) | exact (HM1F1 j)].
      + intros s Hin. destruct Hin as [Heq | Hin'].
        × rewrite <- Heq.
          transitivity (M1 (Fin.FS s0) Fin.F1);
            [exact (IH3 s0 Fin.F1 Hnotin) | exact HM1s0].
        × exact (IH2 s Hin').
      + intros s j Hnin.
        assert (Hs : s ≠ s0)
          by (intros He; apply Hnin; rewrite He; exact (in_eq s0 l')).
        assert (Hnin' : ¬ In s l')
          by (intros Hi; apply Hnin; exact (in_cons s0 s l' Hi)).
        transitivity (M1 (Fin.FS s) j);
          [exact (IH3 s j Hnin') | exact (HM1ne s j Hs)].
  Qed.

  Fixpoint clear_row_ops {p : nat} (M0 : Matrix (S p) (S p) F)
      (l : list (Fin.t p)) {struct l} : list (ElemOp (S p)) :=
    match l with
    | nil ⇒ nil
    | c :: l' ⇒
        elem_op_add (S p) Fin.F1 (Fin.FS c) (- M0 Fin.F1 (Fin.FS c))
          (fin_F1_neq_FS p c)
        :: clear_row_ops M0 l'
    end.

  Lemma clear_row_spec : ∀ (p : nat) (l : list (Fin.t p))
      (M0 M : Matrix (S p) (S p) F),
    NoDup l →
    (∀ (s : Fin.t p) (j : Fin.t (S p)),
       M (Fin.FS s) j == @mat_id F _ _ (S p) (Fin.FS s) j) →
    (∀ c : Fin.t p, In c l → M Fin.F1 (Fin.FS c) == M0 Fin.F1 (Fin.FS c)) →
    (∀ (s : Fin.t p) (j : Fin.t (S p)),
       elem_ops_apply (clear_row_ops M0 l) M (Fin.FS s) j == M (Fin.FS s) j)
    ∧ (elem_ops_apply (clear_row_ops M0 l) M Fin.F1 Fin.F1 == M Fin.F1 Fin.F1)
    ∧ (∀ c : Fin.t p, In c l →
          elem_ops_apply (clear_row_ops M0 l) M Fin.F1 (Fin.FS c) == 0)
    ∧ (∀ c : Fin.t p, ¬ In c l →
          elem_ops_apply (clear_row_ops M0 l) M Fin.F1 (Fin.FS c)
            == M Fin.F1 (Fin.FS c)).
  Proof.
    intros p l; induction l as [| c0 l' IH]; intros M0 M Hnd Hid Horig.
    - split; [intros s j; reflexivity |
        split; [reflexivity |
          split; [intros c Hin; destruct Hin | intros c _; reflexivity]]].
    - inversion Hnd as [| x xs Hnotin Hnd' Hxs]; subst.
      set (M1 := elem_op_apply
        (elem_op_add (S p) Fin.F1 (Fin.FS c0) (- M0 Fin.F1 (Fin.FS c0))
           (fin_F1_neq_FS p c0)) M).
      assert (HM1FS : ∀ (s : Fin.t p) (j : Fin.t (S p)),
        M1 (Fin.FS s) j == M (Fin.FS s) j).
      { intros s j.
        exact (elem_op_add_ne (S p) Fin.F1 (Fin.FS c0) (- M0 Fin.F1 (Fin.FS c0))
                 (fin_F1_neq_FS p c0) M (Fin.FS s) j (fin_FS_neq_F1 p s)). }
      assert (HM1F1 : ∀ j : Fin.t (S p),
        M1 Fin.F1 j == M Fin.F1 j + (- M0 Fin.F1 (Fin.FS c0)) × M (Fin.FS c0) j).
      { intros j.
        exact (elem_op_add_at (S p) Fin.F1 (Fin.FS c0) (- M0 Fin.F1 (Fin.FS c0))
                 (fin_F1_neq_FS p c0) M j). }
      assert (HM1_11 : M1 Fin.F1 Fin.F1 == M Fin.F1 Fin.F1).
      { rewrite (HM1F1 Fin.F1), (Hid c0 Fin.F1), (mat_id_col_0 p c0).
        ring. }
      assert (HM1_at : M1 Fin.F1 (Fin.FS c0) == 0).
      { rewrite (HM1F1 (Fin.FS c0)), (Hid c0 (Fin.FS c0)),
          (mat_id_diag (S p) (Fin.FS c0)), (Horig c0 (in_eq c0 l')).
        ring. }
      assert (HM1_ne : ∀ c : Fin.t p, c ≠ c0 →
        M1 Fin.F1 (Fin.FS c) == M Fin.F1 (Fin.FS c)).
      { intros c Hc.
        rewrite (HM1F1 (Fin.FS c)), (Hid c0 (Fin.FS c)),
          (e_at_neq (S p) (Fin.FS c0) (Fin.FS c)
             (fun Heq ⇒ Hc (eq_sym (Fin.FS_inj c0 c Heq)))).
        ring. }
      assert (Hid' : ∀ (s : Fin.t p) (j : Fin.t (S p)),
        M1 (Fin.FS s) j == @mat_id F _ _ (S p) (Fin.FS s) j).
      { intros s j.
        transitivity (M (Fin.FS s) j); [exact (HM1FS s j) | exact (Hid s j)]. }
      assert (Horig' : ∀ c : Fin.t p, In c l' →
        M1 Fin.F1 (Fin.FS c) == M0 Fin.F1 (Fin.FS c)).
      { intros c Hin.
        assert (Hc : c ≠ c0)
          by (intros He; apply Hnotin; rewrite <- He; exact Hin).
        transitivity (M Fin.F1 (Fin.FS c)).
        - exact (HM1_ne c Hc).
        - exact (Horig c (in_cons c0 c l' Hin)). }
      destruct (IH M0 M1 Hnd' Hid' Horig') as [IH1 [IH2 [IH3 IH4]]].
      split; [| split; [| split]].
      + intros s j.
        transitivity (M1 (Fin.FS s) j); [exact (IH1 s j) | exact (HM1FS s j)].
      + transitivity (M1 Fin.F1 Fin.F1); [exact IH2 | exact HM1_11].
      + intros c Hin. destruct Hin as [Heq | Hin'].
        × rewrite <- Heq.
          transitivity (M1 Fin.F1 (Fin.FS c0));
            [exact (IH4 c0 Hnotin) | exact HM1_at].
        × exact (IH3 c Hin').
      + intros c Hnin.
        assert (Hc : c ≠ c0)
          by (intros He; apply Hnin; rewrite He; exact (in_eq c0 l')).
        assert (Hnin' : ¬ In c l')
          by (intros Hi; apply Hnin; exact (in_cons c0 c l' Hi)).
        transitivity (M1 Fin.F1 (Fin.FS c));
          [exact (IH4 c Hnin') | exact (HM1_ne c Hc)].
  Qed.

  Definition find_nonzero (dec : DecEq F) :
    ∀ (n : nat) (v : Fin.t n → F),
      {i : Fin.t n | v i =/= 0} + {∀ i : Fin.t n, v i == 0}.
  Proof.
    intros n; induction n as [| m IH]; intros v.
    - right. intros i. exact (Fin.case0 (fun i0 : Fin.t 0 ⇒ v i0 == 0) i).
    - destruct (dec (v Fin.F1) 0) as [Hz | Hnz].
      + destruct (IH (fun s ⇒ v (Fin.FS s))) as [[i Hi] | Hall].
        × left. ∃ (Fin.FS i). exact Hi.
        × right. intros i.
          apply (Fin.caseS' i (fun i0 ⇒ v i0 == 0)).
          -- exact Hz.
          -- intros s. exact (Hall s).
      + left. ∃ Fin.F1. exact Hnz.
  Defined.

  Theorem gauss_reduce : ∀ (dec : DecEq F) (n : nat) (A : Matrix n n F),
    ∃ l : list (ElemOp n),
      (elem_ops_apply l A == @mat_id F _ _ n)
      ∨ (∃ i : Fin.t n, ∀ j : Fin.t n, elem_ops_apply l A i j == 0).
  Proof.
    intros dec n; induction n as [| p IH]; intros A.
    - ∃ (@nil (ElemOp 0)). left.
      intros i.
      exact (Fin.case0
        (fun i0 : Fin.t 0 ⇒ equ (elem_ops_apply nil A i0) (@mat_id F _ _ 0 i0)) i).
    - destruct (find_nonzero dec (S p) (fun i ⇒ A i Fin.F1)) as
        [[i0 Hi0] | Hall].
      +
        assert (Hstep1 : ∃ l1 : list (ElemOp (S p)),
          elem_ops_apply l1 A Fin.F1 Fin.F1 =/= 0).
        { destruct (Fin.eq_dec i0 Fin.F1) as [He | Hne].
          - ∃ (@nil (ElemOp (S p))).
            rewrite He in Hi0. exact Hi0.
          - assert (Hne' : Fin.F1 ≠ i0)
              by (intros He'; apply Hne; exact (eq_sym He')).
            ∃ (elem_op_swap (S p) Fin.F1 i0 Hne' :: nil).
            intros Hzero. apply Hi0.
            transitivity (elem_op_apply (elem_op_swap (S p) Fin.F1 i0 Hne') A
                            Fin.F1 Fin.F1).
            + symmetry.
              transitivity (A (swap_fin Fin.F1 i0 Fin.F1) Fin.F1).
              × exact (elem_swap_mul (S p) (S p) Fin.F1 i0 A Fin.F1 Fin.F1).
              × rewrite (swap_fin_hit (S p) Fin.F1 i0). reflexivity.
            + exact Hzero. }
        destruct Hstep1 as [l1 Hpiv].
        set (A1 := elem_ops_apply l1 A).
        set (op2 := elem_op_scale (S p) Fin.F1 (inv (A1 Fin.F1 Fin.F1))
                      (A1 Fin.F1 Fin.F1)
                      ((left_inverse_on (defined_on := nonzero _)) (A1 Fin.F1 Fin.F1) Hpiv)).
        set (A2 := elem_op_apply op2 A1).
        assert (H2 : A2 Fin.F1 Fin.F1 == 1).
        { transitivity (inv (A1 Fin.F1 Fin.F1) × A1 Fin.F1 Fin.F1).
          - exact (elem_scale_mul_eq (S p) (S p) Fin.F1
                     (inv (A1 Fin.F1 Fin.F1)) A1 Fin.F1).
          - exact ((left_inverse_on (defined_on := nonzero _)) (A1 Fin.F1 Fin.F1) Hpiv). }
        destruct (clear_col_spec p (fin_all p) A2 A2 (fin_all_NoDup p) H2
                    (fun s _ ⇒ reflexivity (A2 (Fin.FS s) Fin.F1)))
          as [HC1 [HC2 HC3]].
        set (A3 := elem_ops_apply (clear_col_ops A2 (fin_all p)) A2).
        assert (H3_11 : A3 Fin.F1 Fin.F1 == 1).
        { transitivity (A2 Fin.F1 Fin.F1); [exact (HC1 Fin.F1) | exact H2]. }
        assert (H3col : ∀ s : Fin.t p, A3 (Fin.FS s) Fin.F1 == 0).
        { intros s. exact (HC2 s (fin_all_complete p s)). }
        destruct (IH (subm A3)) as [lsub [Hsid | [isub Hszero]]].
        ×
          set (A4 := elem_ops_apply (map lift_eop lsub) A3).
          assert (H4F1 : ∀ j : Fin.t (S p), A4 Fin.F1 j == A3 Fin.F1 j).
          { intros j. exact (lift_eops_row_F1 p lsub A3 j). }
          assert (H4id : ∀ (s : Fin.t p) (j : Fin.t (S p)),
            A4 (Fin.FS s) j == @mat_id F _ _ (S p) (Fin.FS s) j).
          { intros s j.
            apply (Fin.caseS' j (fun j0 ⇒
              A4 (Fin.FS s) j0 == @mat_id F _ _ (S p) (Fin.FS s) j0)).
            - transitivity (0 : F).
              + exact (lift_eops_col0 p lsub A3 H3col s).
              + symmetry. exact (mat_id_col_0 p s).
            - intros c.
              transitivity (elem_ops_apply lsub (subm A3) s c).
              + exact (lift_eops_sub p lsub A3 s c).
              + transitivity (@mat_id F _ _ p s c).
                × exact (Hsid s c).
                × symmetry. exact (e_FS p s c). }
          destruct (clear_row_spec p (fin_all p) A4 A4 (fin_all_NoDup p) H4id
                      (fun c _ ⇒ reflexivity (A4 Fin.F1 (Fin.FS c))))
            as [HR1 [HR2 [HR3 HR4]]].
          ∃ (l1 ++ (op2 :: nil) ++ clear_col_ops A2 (fin_all p)
                  ++ map lift_eop lsub ++ clear_row_ops A4 (fin_all p)).
          left.

            elem_ops_apply (l1 ++ (op2 :: nil) ++ clear_col_ops A2 (fin_all p)
                        ++ map lift_eop lsub ++ clear_row_ops A4 (fin_all p)) A

            equ (elem_ops_apply (clear_row_ops A4 (fin_all p)) A4 r0)
                (@mat_id F _ _ (S p) r0))).
          -- intros j.
             apply (Fin.caseS' j (fun j0 ⇒
               elem_ops_apply (clear_row_ops A4 (fin_all p)) A4 Fin.F1 j0
                 == @mat_id F _ _ (S p) Fin.F1 j0)).
             ++ transitivity (A4 Fin.F1 Fin.F1); [exact HR2 |].
                transitivity (A3 Fin.F1 Fin.F1); [exact (H4F1 Fin.F1) |].
                transitivity (1 : F); [exact H3_11 |].
                symmetry. exact (mat_id_diag (S p) Fin.F1).
             ++ intros c.
                transitivity (0 : F);
                  [exact (HR3 c (fin_all_complete p c)) |].
                symmetry.
                exact (e_at_neq (S p) Fin.F1 (Fin.FS c) (fin_F1_neq_FS p c)).
          -- intros s j.
             transitivity (A4 (Fin.FS s) j);
               [exact (HR1 s j) | exact (H4id s j)].
        ×
          ∃ (l1 ++ (op2 :: nil) ++ clear_col_ops A2 (fin_all p)
                  ++ map lift_eop lsub).
          right. ∃ (Fin.FS isub).
          assert (Etotal :

                        ++ map lift_eop lsub) A

          { repeat rewrite (elem_ops_apply_app (S p)). reflexivity. }

            elem_ops_apply (map lift_eop lsub) A3 (Fin.FS isub) j0 == 0)).
          -- exact (lift_eops_col0 p lsub A3 H3col isub).
          -- intros c.
             transitivity (elem_ops_apply lsub (subm A3) isub c).
             ++ exact (lift_eops_sub p lsub A3 isub c).
             ++ exact (Hszero c).
      +
        destruct (IH (subm A)) as [lsub [Hsid | [isub Hszero]]].
        ×
          set (A4 := elem_ops_apply (map lift_eop lsub) A).
          assert (H4F1 : ∀ j : Fin.t (S p), A4 Fin.F1 j == A Fin.F1 j).
          { intros j. exact (lift_eops_row_F1 p lsub A j). }
          assert (H4id : ∀ (s : Fin.t p) (j : Fin.t (S p)),
            A4 (Fin.FS s) j == @mat_id F _ _ (S p) (Fin.FS s) j).
          { intros s j.
            apply (Fin.caseS' j (fun j0 ⇒
              A4 (Fin.FS s) j0 == @mat_id F _ _ (S p) (Fin.FS s) j0)).
            - transitivity (0 : F).
              + exact (lift_eops_col0 p lsub A (fun t ⇒ Hall (Fin.FS t)) s).
              + symmetry. exact (mat_id_col_0 p s).
            - intros c.
              transitivity (elem_ops_apply lsub (subm A) s c).
              + exact (lift_eops_sub p lsub A s c).
              + transitivity (@mat_id F _ _ p s c).
                × exact (Hsid s c).
                × symmetry. exact (e_FS p s c). }
          destruct (clear_row_spec p (fin_all p) A4 A4 (fin_all_NoDup p) H4id
                      (fun c _ ⇒ reflexivity (A4 Fin.F1 (Fin.FS c))))
            as [HR1 [HR2 [HR3 HR4]]].
          ∃ (map lift_eop lsub ++ clear_row_ops A4 (fin_all p)).
          right. ∃ Fin.F1.
          assert (Etotal :
            elem_ops_apply (map lift_eop lsub ++ clear_row_ops A4 (fin_all p)) A
            = elem_ops_apply (clear_row_ops A4 (fin_all p)) A4).
          { repeat rewrite (elem_ops_apply_app (S p)). reflexivity. }
          rewrite Etotal.
          intros j.
          apply (Fin.caseS' j (fun j0 ⇒
            elem_ops_apply (clear_row_ops A4 (fin_all p)) A4 Fin.F1 j0 == 0)).
          -- transitivity (A4 Fin.F1 Fin.F1); [exact HR2 |].
             transitivity (A Fin.F1 Fin.F1); [exact (H4F1 Fin.F1) |].
             exact (Hall Fin.F1).
          -- intros c. exact (HR3 c (fin_all_complete p c)).
        ×
          ∃ (map lift_eop lsub).
          right. ∃ (Fin.FS isub).
          intros j.
          apply (Fin.caseS' j (fun j0 ⇒
            elem_ops_apply (map lift_eop lsub) A (Fin.FS isub) j0 == 0)).
          -- exact (lift_eops_col0 p lsub A (fun t ⇒ Hall (Fin.FS t)) isub).
          -- intros c.
             transitivity (elem_ops_apply lsub (subm A) isub c).
             ++ exact (lift_eops_sub p lsub A isub c).
             ++ exact (Hszero c).
  Qed.

  Theorem detM_mul_dec : ∀ (dec : DecEq F) (n : nat)
      (A B : Matrix n n F),
    detM (mat_mul A B) == detM A × detM B.
  Proof.
    intros dec n A B.
    destruct (gauss_reduce dec n A) as [l [Hid | [i Hz]]].
    -
      assert (HfA : elem_ops_factor l × detM A == 1).
      { calc (elem_ops_factor l × detM A)
          == (detM (elem_ops_apply l A))
             { (det_eops_apply n l A) }
          == (detM (@mat_id F _ _ n))
             { (detM_proper n (elem_ops_apply l A) mat_id Hid) }
          == (1 : F)
             { (Impl.Det.det_id n) }. }
      assert (HfAB : elem_ops_factor l × detM (mat_mul A B) == detM B).
      { calc (elem_ops_factor l × detM (mat_mul A B))
          == (detM (elem_ops_apply l (mat_mul A B)))
             { (det_eops_apply n l (mat_mul A B)) }
          == (detM (mat_mul (elem_ops_apply l A) B))
             { (detM_proper n _ _ (elem_ops_apply_mul n l A B)) }
          == (detM (mat_mul (@mat_id F _ _ n) B))
             { (detM_proper n _ _ (mat_mul_proper n n n (elem_ops_apply l A) mat_id Hid
                  B B (fun i0 k ⇒ reflexivity (B i0 k)))) }
          == (detM B)
             { (detM_proper n _ _ (mat_mul_left_identity n n B)) }. }
      calc (detM (mat_mul A B))
        == ((elem_ops_factor l × detM A) × detM (mat_mul A B))
        == (detM A × (elem_ops_factor l × detM (mat_mul A B)))
        == (detM A × detM B).
      + rewrite HfA. ring.
      + ring.
      + rewrite HfAB. reflexivity.
    -
      assert (HfA : elem_ops_factor l × detM A == 0).
      { calc (elem_ops_factor l × detM A)
          == (detM (elem_ops_apply l A))
             { (det_eops_apply n l A) }
          == (0 : F)
             { (det_zero_row n (elem_ops_apply l A) i Hz) }. }
      assert (HfAB : elem_ops_factor l × detM (mat_mul A B) == 0).
      { calc (elem_ops_factor l × detM (mat_mul A B))
          == (detM (elem_ops_apply l (mat_mul A B)))
             { (det_eops_apply n l (mat_mul A B)) }
          == (detM (mat_mul (elem_ops_apply l A) B))
             { (detM_proper n _ _ (elem_ops_apply_mul n l A B)) }
          == (0 : F)
             { (det_zero_row n (mat_mul (elem_ops_apply l A) B) i
                  (mat_mul_zero_row n n (elem_ops_apply l A) B i Hz)) }. }
      assert (HA0 : detM A == 0).
      { exact (cancel_unit (elem_ops_factor l) (elem_ops_cofactor l) (detM A)
                 (elem_ops_factor_cofactor n l) HfA). }
      assert (HAB0 : detM (mat_mul A B) == 0).
      { exact (cancel_unit (elem_ops_factor l) (elem_ops_cofactor l)
                 (detM (mat_mul A B)) (elem_ops_factor_cofactor n l) HfAB). }
      rewrite HAB0, HA0.
      ring.
  Qed.

End DetMul.

Corollary detM_mul_Q : ∀ (n : nat) (A B : Matrix n n Q),
  detM (mat_mul A B) == detM A × detM B.
Proof.
  exact (detM_mul_dec Q_DecEq).
Qed.

Corollary det_comp_Q : ∀ (U : Type)
    (eU : EqRel U) (zU : Zero U) (aU : HasAdd U) (nU : HasNeg U)

  det B (linear_map_comp f g) == det B f × det B g.
