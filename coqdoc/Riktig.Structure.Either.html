From Riktig.Structure Require Export Adt.

A sum of two objects named by a and b: the datatype of the constant signature const a + const b, with no recursion.
Module Either.
  Notation shape den a b := (shape_of den (eitherS (S := sig _) a b)).

  Class Either (O : Type) `{Bicartesian O} {K : Type}
      (den : K → O) (a b : K) (E : O)
      (build : HasBuild O (shape den a b) E)
      (fold : HasCata O (shape den a b) E) := {
    initial :: InitialAlgebra O (shape den a b) E build fold;
  }.

  Notation left := (build (F := shape _ _ _) ∘ inj1).
  Notation right := (build (F := shape _ _ _) ∘ inj2).
  Notation case := (cata (F := shape _ _ _)).
End Either.
Export Either(Either).
