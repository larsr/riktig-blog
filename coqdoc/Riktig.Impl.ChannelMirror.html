From Riktig.Structure Require Import Channel.
From Riktig.Theory Require Import Tree InitialAlgebra Products Coproducts.
From Riktig.Programs Require Import Tree.

The mirror image is its own decoder: a tree survives the round trip across its reflection, by initiality.
Section MirrorChannel.
  Context {O : Type} `{Bicartesian O} {Kc : Type} {den : Kc → O} {x : Kc} {T : O}
    {tb : HasBuild O (Tree.shape den x) T} {tf : HasCata O (Tree.shape den x) T}
    `{!Tree O den x T tb tf}.

  #[export] Instance mirror_channel : Channel O T T mirror mirror.
  Proof.
    split.
    cbv [mirror].
    set (sw := lift (pi2 (a := T) (b := T)) pi1).
    set (K := (tb ∘ inj2) ∘ sw).
    set (M := cata (F := Tree.shape den x) (t := T) T (desc (tb ∘ inj1) K)).
    assert (Hs : ∀ f : T ~> T, sw ∘ (f ×× f) == (f ×× f) ∘ sw).
    { intros f. unfold sw.
      rewrite (lift_comp (O := O)). rewrite (pi2_par (O := O)).
      rewrite (pi1_par (O := O)). rewrite (par_lift (O := O)). reflexivity. }
    assert (Hss : sw ∘ sw == cat_id (x := T ⨯ T)).
    { unfold sw. exact (swap_swap (O := O) T T). }
    assert (Hcb : M ∘ build == desc (tb ∘ inj1) K ∘ fmap (F := Tree.shape den x) M).
    { unfold M. exact (InitialAlgebra.cata_build (F := Tree.shape den x) (t := T) T (desc (tb ∘ inj1) K)). }
    etransitivity; [ | apply (cata_id (F := Tree.shape den x) (t := T)) ].
    apply (cata_fusion (F := Tree.shape den x) T T (desc (tb ∘ inj1) K) build M).
        setoid_rewrite (comp_desc (O := O)).
    apply Coproducts.desc_proper.
    -
      etransitivity; [ apply (Category.comp_assoc (O := O)) | ].
      etransitivity; [ apply Category.comp_proper; [ exact Hcb | reflexivity ] | ].
      etransitivity; [ symmetry; apply (Category.comp_assoc (O := O)) | ].
      etransitivity; [ apply Category.comp_proper;
        [ reflexivity | apply (Coproducts.desc_inj1 (O := O)) ] | ].
      etransitivity; [ apply Category.comp_proper; [ reflexivity | apply right_identity ] | ].
      etransitivity; [ apply (Coproducts.desc_inj1 (O := O)) | ].
      apply Category.comp_proper; [ reflexivity | symmetry; apply right_identity ].
    -
      etransitivity; [ apply (Category.comp_assoc (O := O)) | ].
      etransitivity; [ apply Category.comp_proper;
        [ apply (Category.comp_assoc (O := O)) | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ apply Category.comp_proper; [ exact Hcb | reflexivity ] | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ symmetry; apply (Category.comp_assoc (O := O)) | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ apply Category.comp_proper;
          [ reflexivity | apply (Coproducts.desc_inj2 (O := O)) ]
        | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ apply (Category.comp_assoc (O := O)) | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ apply Category.comp_proper; [ apply (Coproducts.desc_inj2 (O := O)) | reflexivity ]
        | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ symmetry; apply (Category.comp_assoc (O := O)) | reflexivity ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ apply Category.comp_proper; [ reflexivity | exact (Hs M) ] | reflexivity ] | ].
      etransitivity; [ symmetry; apply (Category.comp_assoc (O := O)) | ].
      etransitivity; [ apply Category.comp_proper;
        [ reflexivity | symmetry; apply (Category.comp_assoc (O := O)) ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ reflexivity | apply Category.comp_proper; [ reflexivity | exact Hss ] ] | ].
      etransitivity; [ apply Category.comp_proper;
        [ reflexivity | apply right_identity ] | ].
      symmetry. apply (Category.comp_assoc (O := O)).
  Qed.
End MirrorChannel.
