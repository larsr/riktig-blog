Section VectorBlade.
  Context {K : Type} `{Boolean K} {n : nat}.

  #[export] Instance VectorBlade_EqRel : EqRel (Vector.t K n) :=
    fun v w ⇒ ∀ i : Fin.t n, Vector.nth v i == Vector.nth w i.
  #[export] Instance VectorBlade_Zero : Zero (Vector.t K n) :=
    Vector.const 0 n.
  #[export] Instance VectorBlade_One : One (Vector.t K n) :=
    Vector.const 1 n.
  #[export] Instance VectorBlade_Add : HasAdd (Vector.t K n) :=
    Vector.map2 add.
  #[export] Instance VectorBlade_Mult : HasMult (Vector.t K n) :=
    Vector.map2 mult.
  #[export] Instance VectorBlade_Neg : HasNeg (Vector.t K n) :=
    Vector.map neg.

  #[export] Instance VectorBlade_Equivalence :
    Equivalence (@equ (Vector.t K n) _).
  Proof.
    constructor.
    - intros v i. reflexivity.
    - intros v w Hvw i. symmetry. exact (Hvw i).
    - intros u v w Huv Hvw i.
      transitivity (Vector.nth v i); [ exact (Huv i) | exact (Hvw i) ].
  Qed.

  Definition vector_member (v : Vector.t K n) (i : Fin.t n) : K :=
    Vector.nth v i.

  #[export] Instance vector_member_proper :
    Proper (equ ==> eq ==> equ) vector_member.
  Proof.
    intros v w Hvw i i' Hii'. unfold vector_member.
    rewrite <- Hii'. exact (Hvw i).
  Qed.

  Lemma vector_member_faithful : ∀ v w : Vector.t K n,
    (∀ i : Fin.t n, vector_member v i == vector_member w i) → v == w.
  Proof. intros v w Hvw. exact Hvw. Qed.

  Lemma vector_member_empty : ∀ i : Fin.t n,
    vector_member 0 i == (0 : K).
  Proof.
    intros i.
    change (vector_member 0 i) with (Vector.nth (Vector.const (0 : K) n) i).
    rewrite (const_nth K 0 n i). reflexivity.
  Qed.

  Lemma vector_member_full : ∀ i : Fin.t n,
    vector_member 1 i == (1 : K).
  Proof.
    intros i.
    change (vector_member 1 i) with (Vector.nth (Vector.const (1 : K) n) i).
    rewrite (const_nth K 1 n i). reflexivity.
  Qed.

  Lemma vector_member_add : ∀ (v w : Vector.t K n) (i : Fin.t n),
    vector_member (v + w) i == vector_member v i + vector_member w i.
  Proof.
    intros v w i.
    change (vector_member (v + w) i)
      with (Vector.nth (Vector.map2 add v w) i).
    rewrite (nth_map2 add v w i i i eq_refl eq_refl). reflexivity.
  Qed.

  Lemma vector_member_mult : ∀ (v w : Vector.t K n) (i : Fin.t n),
    vector_member (v × w) i == vector_member v i × vector_member w i.
  Proof.
    intros v w i.
    change (vector_member (v × w) i)
      with (Vector.nth (Vector.map2 mult v w) i).
    rewrite (nth_map2 mult v w i i i eq_refl eq_refl). reflexivity.
  Qed.

  Lemma vector_member_neg : ∀ (v : Vector.t K n) (i : Fin.t n),
    vector_member (- v) i == - vector_member v i.
  Proof.
    intros v i.
    change (vector_member (- v) i) with (Vector.nth (Vector.map neg v) i).
    rewrite (nth_map neg v i i eq_refl). reflexivity.
  Qed.

  #[export] Instance VectorBlade_Blade :
    Blade (Fin.t n) K (Vector.t K n) :=
    {| Blade.setoid := VectorBlade_Equivalence;
       Blade.member := vector_member;
       Blade.member_proper := vector_member_proper;
       Blade.member_faithful := vector_member_faithful;
       Blade.member_empty := vector_member_empty;
       Blade.member_full := vector_member_full;
       Blade.member_symmetric_difference := vector_member_add;
       Blade.member_intersection := vector_member_mult;
       Blade.member_complement := vector_member_neg |}.

End VectorBlade.

Section VectorBladeComputes.

  Definition generator (i : Fin.t 3) : Vector.t bool 3 :=
    Vector.replace (0 : Vector.t bool 3) i true.

  Notation e1 := (generator Fin.F1).
  Notation e2 := (generator (Fin.FS Fin.F1)).
  Notation e3 := (generator (Fin.FS (Fin.FS Fin.F1))).

  Lemma equ_of_eq : ∀ (v w : Vector.t bool 3), v = w → v == w.
  Proof. intros v w Hvw i. rewrite Hvw. reflexivity. Qed.

  Example member_e1 : member e1 Fin.F1 == (1 : bool).
  Proof. reflexivity. Qed.

  Example member_e1_second : member e1 (Fin.FS Fin.F1) == (0 : bool).
  Proof. reflexivity. Qed.

  Example e1_plus_e2_holds_both : member (e1 + e2) (Fin.FS Fin.F1) == (1 : bool).
  Proof. reflexivity. Qed.

  Example e1_meet_e2_is_empty : e1 × e2 == 0.
  Proof. apply equ_of_eq. reflexivity. Qed.

  Example e1_self_cancels : e1 + e1 == 0.
  Proof. apply equ_of_eq. reflexivity. Qed.

  Example e1_self_cancels_by_law : e1 + e1 == 0.
  Proof. exact (add_self_zero e1). Qed.

  Example e1_shift : e1 + (e1 + e2) == e2.
  Proof. exact (blade_shift e1 e2). Qed.

End VectorBladeComputes.
