Parsing and checking the simply typed calculus: A, A → B, (A → B) → A → B for the types, \x : T. body for the terms. The checker infers the type, and its soundness theorem hands back a typing derivation for the erased term.
Section TypeGrammar.
  Context {St : Type} {NX : HasNext ascii St} (size : St → N).

  Definition tok {A : Type} (p : parser St A) : parser St A := p <* p_many size (p_sat is_space).
  Definition tident : parser St string := string_of_list_ascii <$> p_many1 size (p_sat is_alpha).
  Definition arrow_tok : parser St ascii := sym "-"%char *> tok (sym ">"%char).

  Definition ty_step (rec : parser St sty × parser St sty) : parser St sty :=
    snd rec >>= fun a ⇒ (arrow_tok *> fst rec >>= fun c ⇒ ret (SArrow a c)) <|> ret a.
  Definition ty_atom_step (rec : parser St sty × parser St sty) : parser St sty :=
    (SBase <$> tok tident)
    <|> (tok (sym "("%char) *> fst rec <* tok (sym ")"%char)).
  Definition ty_p (fuel : N) : parser St sty :=
    fst (natrec _ (p_fail, p_fail) (fun _ rec ⇒ (ty_step rec, ty_atom_step rec)) fuel).

  Definition tterm_step (rec : parser St tnamed × parser St tnamed) : parser St tnamed :=
    ((tok (sym "092"%char) *> tok tident) >>= fun x ⇒
       (tok (sym ":"%char) *> ty_p 16%N) >>= fun A ⇒
       (tok (sym "."%char) *> fst rec) >>= fun b ⇒ ret (TNLam x A b))
    <|> (p_many1 size (snd rec) >>= fun atoms ⇒
           match atoms with [] ⇒ fail | a :: rest ⇒ ret (fold_left TNApp rest a) end).
  Definition tterm_atom_step (rec : parser St tnamed × parser St tnamed) : parser St tnamed :=
    (TNVar <$> tok tident)
    <|> (tok (sym "("%char) *> fst rec <* tok (sym ")"%char)).
  Definition tterm_p (fuel : N) : parser St tnamed :=
    fst (natrec _ (p_fail, p_fail) (fun _ rec ⇒ (tterm_step rec, tterm_atom_step rec)) fuel).
End TypeGrammar.

Definition lsize (s : list ascii) : N := N.of_nat (length s).

Definition parse_ty (s : string) : option sty :=
  match (p_many lsize (p_sat is_space) *> ty_p lsize 16%N) (list_ascii_of_string s) with
  | (A, []) :: _ ⇒ Some A
  | _ ⇒ None
  end.

Definition parse_tterm (s : string) : option tterm :=
  match (p_many lsize (p_sat is_space) *> tterm_p lsize 32%N) (list_ascii_of_string s) with
  | (t, []) :: _ ⇒ Some (tresolve [] t)
  | _ ⇒ None
  end.

Eval vm_compute in parse_ty "(A -> B) -> A -> B".
Eval vm_compute in parse_tterm "\f : A -> B. \x : A. f x".

Definition check (s : string) : option sty :=
  match parse_tterm s with Some t ⇒ infer [] t | None ⇒ None end.

Eval vm_compute in check "\f : A -> B. \x : A. f x".

Example compose_checks :
  check "\f : A -> B. \x : A. f x" = Some ((SBase "A" --> SBase "B") --> SBase "A" --> SBase "B").
Proof. vm_compute. reflexivity. Qed.

Example self_application_rejected : check "\f : A. f f" = None.
Proof. vm_compute. reflexivity. Qed.

The checker's yes is a typing derivation, by soundness.
Example compose_typed :
  has_ty (ctx_of []) (erase (TLam (SBase "A" --> SBase "B") (TLam (SBase "A") (TApp (TVar 1) (TVar 0)))))
    ((SBase "A" --> SBase "B") --> SBase "A" --> SBase "B").
Proof. apply infer_sound. vm_compute. reflexivity. Qed.
