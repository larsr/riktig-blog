Library Riktig.Theory.NatOrder

From Riktig.Structure Require Export NatOrder.
From Riktig.Theory Require Export Naturals.

Section NatOrderTheory.
  Context {N : Type} `{NatOrder N}.

  Theorem le_refl : ∀ a : N, a ≤ a.
  Proof. intros a. apply NatOrder.le_add. ∃ 0. exact (right_identity a). Qed.

  Theorem le_trans : ∀ a b c : N, a ≤ b → b ≤ c → a ≤ c.
  Proof.
    intros a b c Hab Hbc.
    destruct (proj1 (NatOrder.le_add a b) Hab) as [x Hx].
    destruct (proj1 (NatOrder.le_add b c) Hbc) as [y Hy].
    apply NatOrder.le_add. ∃ (x + y).
    calc (a + (x + y))
      == ((a + x) + y) { associative }
      == (b + y) { Hx }
      == c { Hy }.
  Qed.

  Theorem le_antisym : ∀ a b : N, a ≤ b → b ≤ a → a == b.
  Proof.
    intros a b Hab Hba.
    destruct (proj1 (NatOrder.le_add a b) Hab) as [x Hx].
    destruct (proj1 (NatOrder.le_add b a) Hba) as [y Hy].
    assert (Hxy : x + y == 0).
    { apply (eq_zero_of_add_eq_self a).
      calc (a + (x + y))
        == ((a + x) + y) { associative }
        == (b + y) { Hx }
        == a { Hy }. }
    assert (Hx0 : x == 0) by exact (eq_zero_of_add_eq_zero x y Hxy).
    calc a
      == (a + 0) { right_identity }
      == (a + x) { Hx0 }
      == b { Hx }.
  Qed.

  Theorem zero_le : ∀ a : N, 0 ≤ a.
  Proof. intros a. apply NatOrder.le_add. ∃ a. exact (left_identity a). Qed.

  Theorem le_add_r : ∀ a c : N, a ≤ a + c.
  Proof. intros a c. apply NatOrder.le_add. ∃ c. reflexivity. Qed.

  Theorem le_add_l : ∀ a b : N, b ≤ a + b.
  Proof.
    intros a b.
    apply (NatOrder.le_proper b b (reflexivity b) (b + a) (a + b) (commutative b a)).
    exact (le_add_r b a).
  Qed.

  Theorem le_succ_diag : ∀ a : N, a ≤ succ a.
  Proof. intros a. apply NatOrder.le_add. ∃ 1. symmetry. exact (Naturals.succ_add a). Qed.

  Theorem succ_le_succ : ∀ a b : N, a ≤ b → succ a ≤ succ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (NatOrder.le_add a b) Hab) as [c Hc].
    apply NatOrder.le_add. ∃ c.
    calc (succ a + c)
      == (succ (a + c)) { add_succ_l }
      == (succ b) { Hc }.
  Qed.

  Theorem le_of_succ_le_succ : ∀ a b : N, succ a ≤ succ b → a ≤ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (NatOrder.le_add (succ a) (succ b)) Hab) as [c Hc].
    apply NatOrder.le_add. ∃ c.
    apply (injective (f := succ)).
    calc (succ (a + c))
      == (succ a + c) { add_succ_l }
      == (succ b) { Hc }.
  Qed.

  Theorem succ_le_of_lt : ∀ a b : N, a < b → succ a ≤ b.
  Proof.
    intros a b [Hab Hne].
    destruct (proj1 (NatOrder.le_add a b) Hab) as [c Hc].
    destruct (zero_or_succ c) as [Hc0 | [c' Hc']].
    - exfalso. apply Hne.
      calc a == (a + 0) { (add_zero a) } == (a + c) { Hc0 } == b { Hc }.
    - apply (proj2 (NatOrder.le_add (succ a) b)). ∃ c'.
      calc (succ a + c') == (succ (a + c')) { (add_succ_l a c') } == (a + succ c') { (add_succ a c') }
        == (a + c) { Hc' } == b { Hc }.
  Qed.

  Theorem le_total : ∀ a b : N, a ≤ b ∨ b ≤ a.
  Proof.
    apply (Peano.nat_induction (fun a ⇒ ∀ b : N, a ≤ b ∨ b ≤ a)).
    - intros a a' Ha.
      exact (all_iff_morphism _ _ (fun b ⇒ or_iff_morphism _ _ (NatOrder.le_proper a a' Ha b b (reflexivity b))
                                                             _ _ (NatOrder.le_proper b b (reflexivity b) a a' Ha))).
    - intros b. left. exact (zero_le b).
    - intros a IH b.
      destruct (Theory.Naturals.zero_or_succ b) as [Hb | [b' Hb]].
      + right. exact (proj2 (NatOrder.le_proper b 0 Hb (succ a) (succ a) (reflexivity _)) (zero_le (succ a))).
      + destruct (IH b') as [Hab | Hba].
        × left. exact (proj2 (NatOrder.le_proper (succ a) (succ a) (reflexivity _) b (succ b') Hb) (succ_le_succ a b' Hab)).
        × right. exact (proj2 (NatOrder.le_proper b (succ b') Hb (succ a) (succ a) (reflexivity _)) (succ_le_succ b' a Hba)).
  Qed.

  Theorem not_lt_zero : ∀ a : N, ¬ a < 0.
  Proof.
    intros a [Ha Hne].
    destruct (proj1 (NatOrder.le_add a 0) Ha) as [c Hc].
    apply Hne. exact (eq_zero_of_add_eq_zero a c Hc).
  Qed.

  Theorem lt_succ_diag : ∀ a : N, a < succ a.
  Proof.
    intros a. split; [ exact (le_succ_diag a) | ].
    intros Ha. apply (succ_ne_self a). symmetry. exact Ha.
  Qed.

  Theorem zero_lt_one : (0 : N) < 1.
  Proof.
    split; [ exact (zero_le 1) | ].
    intros H01. apply (Peano.succ_not_zero (0 : N)).
    calc (succ (0 : N))
      == (1 : N) { one_succ }
      == (0 : N) { H01 }.
  Qed.

  Theorem eq_zero_of_lt_one : ∀ k : N, k < 1 → k == 0.
  Proof.
    intros k [Hk Hne].
    destruct (proj1 (NatOrder.le_add k 1) Hk) as [c Hc].
    destruct (zero_or_succ k) as [Hz | [m Hm]]; [ exact Hz | ].
    exfalso. apply Hne.
    assert (Hmc : m + c == 0).
    { apply (injective (f := succ)).
      calc (succ (m + c))
        == (succ m + c) { add_succ_l }
        == (k + c) { Hm }
        == (1 : N) { Hc }
        == (succ (0 : N)) { one_succ }. }
    calc k
      == (succ m) { Hm }
      == (succ (0 : N)) { (eq_zero_of_add_eq_zero m c Hmc) }
      == (1 : N) { one_succ }.
  Qed.

  Theorem lt_proper : Proper (equ ==> equ ==> iff) (fun a b : N ⇒ a < b).
  Proof.
    intros a a' Ha b b' Hb.
    exact (and_iff_morphism _ _ (NatOrder.le_proper a a' Ha b b' Hb) _ _ (not_iff_morphism _ _ (equiv_iff Ha Hb))).
  Qed.

  Theorem le_of_lt : ∀ a b : N, a < b → a ≤ b.
  Proof. intros a b [Hle _]. exact Hle. Qed.

  Theorem lt_of_lt_of_le : ∀ a b c : N, a < b → b ≤ c → a < c.
  Proof.
    intros a b c [Hab Hne] Hbc. split.
    - exact (le_trans a b c Hab Hbc).
    - intros Hac. apply Hne. apply le_antisym; [ exact Hab | ].
      rewrite <- Hac in Hbc. exact Hbc.
  Qed.

  Theorem lt_irrefl : ∀ a : N, ¬ a < a.
  Proof. intros a [_ Hne]. exact (Hne (reflexivity a)). Qed.

  Theorem lt_of_le_of_ne : ∀ a b : N, a ≤ b → a =/= b → a < b.
  Proof. intros a b Hle Hne. split; [ exact Hle | exact Hne ]. Qed.

  Theorem le_of_lt_succ : ∀ a b : N, a < succ b → a ≤ b.
  Proof.
    intros a b [Hle Hne].
    destruct (proj1 (NatOrder.le_add a (succ b)) Hle) as [c Hc].
    destruct (zero_or_succ c) as [Hz | [d Hd]].
    - exfalso. apply Hne.
      calc a
        == (a + 0) { right_identity }
        == (a + c) { Hz }
        == (succ b) { Hc }.
    - apply NatOrder.le_add. ∃ d. apply (injective (f := succ)).
      calc (succ (a + d))
        == (a + succ d) { add_succ }
        == (a + c) { Hd }
        == (succ b) { Hc }.
  Qed.

  Theorem lt_succ_iff : ∀ a b : N, a < succ b ↔ a < b ∨ a == b.
  Proof.
    intros a b. split.
    - intros Hlt. destruct (eq_or_ne a b) as [Heq | Hne].
      + right. exact Heq.
      + left. exact (lt_of_le_of_ne a b (le_of_lt_succ a b Hlt) Hne).
    - intros [Hlt | Heq].
      + exact (lt_of_lt_of_le a b (succ b) Hlt (le_succ_diag b)).
      + exact (proj2 (lt_proper a b Heq (succ b) (succ b) (reflexivity (succ b))) (lt_succ_diag b)).
  Qed.

  Theorem le_succ_inv : ∀ a b : N, succ a ≤ succ b → a ≤ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (NatOrder.le_add (succ a) (succ b)) Hab) as [c Hc].
    apply NatOrder.le_add. ∃ c.
    apply Peano.succ_injective.
    calc (succ (a + c))
      == (succ a + c) { (symmetry (Theory.Naturals.add_succ_l a c)) }
      == (succ b) { Hc }.
  Qed.

  Theorem le_split : ∀ a b : N, a ≤ b → a == b ∨ succ a ≤ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (NatOrder.le_add a b) Hab) as [c Hc].
    destruct (zero_or_succ c) as [Hc0 | [d Hd]].
    - left.
      calc a
        == (a + 0) { (symmetry (Theory.Naturals.add_zero a)) }
        == (a + c) { Hc0 }
        == b { Hc }.
    - right. apply NatOrder.le_add. ∃ d.
      calc (succ a + d)
        == (succ (a + d)) { (Theory.Naturals.add_succ_l a d) }
        == (a + succ d) { (symmetry (Theory.Naturals.add_succ a d)) }
        == (a + c) { Hd }
        == b { Hc }.
  Qed.

  Theorem le_add_mono_r : ∀ a b c : N, a ≤ b → a + c ≤ b + c.
  Proof.
    intros a b c Hab.
    destruct (proj1 (NatOrder.le_add a b) Hab) as [d Hd].
    apply NatOrder.le_add. ∃ d.
    calc ((a + c) + d)
      == (a + (c + d)) { (associative (op := add) (action := add) a c d) }
      == (a + (d + c)) { (Magma.op_proper a a (reflexivity a) _ _ (commutative (op := add) c d)) }
      == ((a + d) + c) { (associative (op := add) (action := add) a d c) }
      == (b + c) { (Magma.op_proper _ _ Hd c c (reflexivity c)) }.
  Qed.

  Theorem le_of_eq : ∀ a b : N, a == b → a ≤ b.
  Proof. intros a b Hab. rewrite Hab. exact (le_refl b). Qed.

  Theorem le_of_succ_le : ∀ a b : N, succ a ≤ b → a ≤ b.
  Proof. intros a b Hab. exact (le_trans a (succ a) b (le_succ_diag a) Hab). Qed.

  Theorem lt_trans : ∀ a b c : N, a < b → b < c → a < c.
  Proof. intros a b c Hab Hbc. exact (lt_of_lt_of_le a b c Hab (le_of_lt b c Hbc)). Qed.

  Theorem lt_of_succ_le : ∀ a b : N, succ a ≤ b → a < b.
  Proof. intros a b Hab. exact (lt_of_lt_of_le a (succ a) b (lt_succ_diag a) Hab). Qed.

Induction below a bound, and between a base and a bound: the predicate is stated without its side conditions, which the principle carries.
  Theorem nat_induction_le : ∀ (m : N) (P : N → Prop), Proper (equ ==> iff) P →
    P 0 → (∀ n : N, n < m → P n → P (succ n)) → ∀ n : N, n ≤ m → P n.
  Proof.
    intros m P HP Hz HS.
    apply (Peano.nat_induction (fun n ⇒ n ≤ m → P n)).
    - intros k k' Hk. split; intros Hp Hle; apply (HP k k' Hk); apply Hp;
        apply (NatOrder.le_proper k k' Hk m m (reflexivity m)); exact Hle.
    - intros _. exact Hz.
    - intros k IH Hle. exact (HS k (lt_of_succ_le k m Hle) (IH (le_of_succ_le k m Hle))).
  Qed.

  Theorem nat_induction_between : ∀ (b m : N) (P : N → Prop), Proper (equ ==> iff) P →
    P (succ b) → (∀ n : N, b < n → n < m → P n → P (succ n)) →
    ∀ n : N, b < n → n ≤ m → P n.
  Proof.
    intros b m P HP Hb HS n Hlt Hle.
    refine (nat_induction_le m (fun n ⇒ b < n → P n) _ _ _ n Hle Hlt).
    - intros k k' Hk. split; intros Hp Hbk; apply (HP k k' Hk); apply Hp;
        apply (lt_proper b b (reflexivity b) k k' Hk); exact Hbk.
    - intros Hb0. exfalso. exact (not_lt_zero b Hb0).
    - intros k Hkm IH Hbk.
      exact (or_ind (fun Hbk' ⇒ HS k Hbk' Hkm (IH Hbk'))
                    (fun Heq ⇒ proj1 (HP (succ b) (succ k) (Peano.succ_proper b k Heq)) Hb)
                    (proj1 (lt_succ_iff b k) Hbk)).
  Qed.

  Theorem nat_induction_le2 : ∀ (m1 m2 : N) (P : N → Prop), Proper (equ ==> iff) P →
    P 0 → (∀ n : N, n < m1 → n < m2 → P n → P (succ n)) →
    ∀ n : N, n ≤ m1 → n ≤ m2 → P n.
  Proof.
    intros m1 m2 P HP Hz HS n Hn1 Hn2.
    refine (nat_induction_le m1 (fun n ⇒ n ≤ m2 → P n) _ _ _ n Hn1 Hn2).
    - intros k k' Hk. split; intros Hp Hk2; apply (HP k k' Hk); apply Hp;
        apply (NatOrder.le_proper k k' Hk m2 m2 (reflexivity m2)); exact Hk2.
    - intros _. exact Hz.
    - intros k Hk1 IH Hsk2. exact (HS k Hk1 (lt_of_succ_le k m2 Hsk2) (IH (le_of_succ_le k m2 Hsk2))).
  Qed.

Induction from a base index on: the predicate holds at the base and is kept by the successor from there.
  Theorem nat_induction_from : ∀ (b : N) (P : N → Prop), Proper (equ ==> iff) P →
    P b → (∀ n : N, b ≤ n → P n → P (succ n)) → ∀ n : N, b ≤ n → P n.
  Proof.
    intros b P HP Hb HS n Hbn.
    destruct (proj1 (NatOrder.le_add b n) Hbn) as [c Hc].
    apply (proj1 (HP (b + c) n Hc)).
    apply (Peano.nat_induction (fun c ⇒ P (b + c))).
    - intros k k' Hk. exact (HP (b + k) (b + k') (Magma.op_proper b b (reflexivity b) k k' Hk)).
    - exact (proj1 (HP b (b + 0) (symmetry (Naturals.add_zero b))) Hb).
    - intros k IH.
      exact (proj1 (HP (succ (b + k)) (b + succ k) (symmetry (Naturals.add_succ b k)))
                   (HS (b + k) (le_add_r b k) IH)).
  Qed.

Two numbers compare one of three ways, and one of two.
  Theorem succ_le_or_le : ∀ a b : N, succ a ≤ b ∨ b ≤ a.
  Proof.
    intros a b. destruct (le_total a b) as [Hab | Hba]; [ | right; exact Hba ].
    destruct (le_split a b Hab) as [Heq | Hs]; [ right; exact (le_of_eq b a (symmetry Heq)) | left; exact Hs ].
  Qed.

  Theorem lt_eq_or_gt : ∀ a b : N, a < b ∨ a == b ∨ b < a.
  Proof.
    intros a b. destruct (le_total a b) as [Hab | Hba].
    - destruct (le_split a b Hab) as [Heq | Hs].
      + exact (or_intror (or_introl Heq)).
      + exact (or_introl (lt_of_succ_le a b Hs)).
    - destruct (le_split b a Hba) as [Heq | Hs].
      + exact (or_intror (or_introl (symmetry Heq))).
      + exact (or_intror (or_intror (lt_of_succ_le b a Hs))).
  Qed.

  #[export] Instance le_reflexive : Reflexive (le : N → N → Prop) := le_refl.
  #[export] Instance le_transitive : Transitive (le : N → N → Prop) := le_trans.

Multiplication and the order, and the bound of a digit pair

  Theorem mul_le_mono_l : ∀ a b c : N, b ≤ c → a × b ≤ a × c.
  Proof.
    intros a b c Hbc. destruct (proj1 (NatOrder.le_add b c) Hbc) as [d Hd].
    apply NatOrder.le_add. ∃ (a × d).
    calc (a × b + a × d) == (a × (b + d)) { (left_distributive a b d) } == (a × c) { Hd }.
  Qed.

  Theorem lt_of_digits : ∀ a b1 c b2 : N, a < b1 → c < b2 → a + b1 × c < b1 × b2.
  Proof.
    intros a b1 c b2 Ha Hc. apply lt_of_succ_le.
    calc (succ (a + b1 × c))
      == (succ a + b1 × c) { (add_succ_l a (b1 × c)) }
      ≤ (b1 + b1 × c) { (le_add_mono_r (succ a) b1 (b1 × c) (succ_le_of_lt a b1 Ha)) }
      == (b1 × c + b1) { (commutative (op := add) b1 (b1 × c)) }
      == (b1 × succ c) { (mul_succ b1 c) }
      ≤ (b1 × b2) { (mul_le_mono_l b1 (succ c) b2 (succ_le_of_lt c b2 Hc)) }.
  Qed.
