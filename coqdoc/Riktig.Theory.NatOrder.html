Library Riktig.Theory.NatOrder

From Riktig Require Export Structure.NatOrder Theory.Naturals.

Section NatOrderTheory.
  Context {N : Type} `{NatOrder N}.

  Theorem le_refl : ∀ a : N, a ≤ a.
  Proof. intros a. apply le_add. ∃ 0. exact (right_identity a). Qed.

  Theorem le_trans : ∀ a b c : N, a ≤ b → b ≤ c → a ≤ c.
  Proof.
    intros a b c Hab Hbc.
    destruct (proj1 (le_add a b) Hab) as [x Hx].
    destruct (proj1 (le_add b c) Hbc) as [y Hy].
    apply le_add. ∃ (x + y).
    calc (a + (x + y))
      == ((a + x) + y) { associative }
      == (b + y) { Hx }
      == c { Hy }.
  Qed.

  Theorem le_antisym : ∀ a b : N, a ≤ b → b ≤ a → a == b.
  Proof.
    intros a b Hab Hba.
    destruct (proj1 (le_add a b) Hab) as [x Hx].
    destruct (proj1 (le_add b a) Hba) as [y Hy].
    assert (Hxy : x + y == 0).
    { apply (eq_zero_of_add_eq_self a).
      calc (a + (x + y))
        == ((a + x) + y) { associative }
        == (b + y) { Hx }
        == a { Hy }. }
    assert (Hx0 : x == 0) by exact (eq_zero_of_add_eq_zero x y Hxy).
    calc a
      == (a + 0) { right_identity }
      == (a + x) { Hx0 }
      == b { Hx }.
  Qed.

  Theorem zero_le : ∀ a : N, 0 ≤ a.
  Proof. intros a. apply le_add. ∃ a. exact (left_identity a). Qed.

  Theorem le_add_r : ∀ a c : N, a ≤ a + c.
  Proof. intros a c. apply le_add. ∃ c. reflexivity. Qed.

  Theorem le_succ_diag : ∀ a : N, a ≤ succ a.
  Proof. intros a. apply le_add. ∃ 1. symmetry. exact (succ_add a). Qed.

  Theorem succ_le_succ : ∀ a b : N, a ≤ b → succ a ≤ succ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (le_add a b) Hab) as [c Hc].
    apply le_add. ∃ c.
    calc (succ a + c)
      == (succ (a + c)) { add_succ_l }
      == (succ b) { Hc }.
  Qed.

  Theorem le_of_succ_le_succ : ∀ a b : N, succ a ≤ succ b → a ≤ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (le_add (succ a) (succ b)) Hab) as [c Hc].
    apply le_add. ∃ c.
    apply (injective (f := succ)).
    calc (succ (a + c))
      == (succ a + c) { add_succ_l }
      == (succ b) { Hc }.
  Qed.

  Theorem not_lt_zero : ∀ a : N, ¬ a < 0.
  Proof.
    intros a [Ha Hne].
    destruct (proj1 (le_add a 0) Ha) as [c Hc].
    apply Hne. exact (eq_zero_of_add_eq_zero a c Hc).
  Qed.

  Theorem lt_succ_diag : ∀ a : N, a < succ a.
  Proof.
    intros a. split; [ exact (le_succ_diag a) | ].
    intros Ha. apply (succ_ne_self a). symmetry. exact Ha.
  Qed.

  Theorem zero_lt_one : (0 : N) < 1.
  Proof.
    split; [ exact (zero_le 1) | ].
    intros H01. apply (succ_not_zero (0 : N)).
    calc (succ (0 : N))
      == (1 : N) { one_succ }
      == (0 : N) { H01 }.
  Qed.

  Theorem eq_zero_of_lt_one : ∀ k : N, k < 1 → k == 0.
  Proof.
    intros k [Hk Hne].
    destruct (proj1 (le_add k 1) Hk) as [c Hc].
    destruct (zero_or_succ k) as [Hz | [m Hm]]; [ exact Hz | ].
    exfalso. apply Hne.
    assert (Hmc : m + c == 0).
    { apply (injective (f := succ)).
      calc (succ (m + c))
        == (succ m + c) { add_succ_l }
        == (k + c) { Hm }
        == (1 : N) { Hc }
        == (succ (0 : N)) { one_succ }. }
    calc k
      == (succ m) { Hm }
      == (succ (0 : N)) { (eq_zero_of_add_eq_zero m c Hmc) }
      == (1 : N) { one_succ }.
  Qed.

  Theorem lt_proper : Proper (equ ==> equ ==> iff) (fun a b : N ⇒ a < b).
  Proof.
    intros a a' Ha b b' Hb. split; intros [Hle Hne]; split.
    - exact (proj1 (le_proper a a' Ha b b' Hb) Hle).
    - intros Hab. apply Hne. rewrite Ha, Hb. exact Hab.
    - exact (proj2 (le_proper a a' Ha b b' Hb) Hle).
    - intros Hab. apply Hne. rewrite <- Ha, <- Hb. exact Hab.
  Qed.

  Theorem le_of_lt : ∀ a b : N, a < b → a ≤ b.
  Proof. intros a b [Hle _]. exact Hle. Qed.

  Theorem lt_of_lt_of_le : ∀ a b c : N, a < b → b ≤ c → a < c.
  Proof.
    intros a b c [Hab Hne] Hbc. split.
    - exact (le_trans a b c Hab Hbc).
    - intros Hac. apply Hne. apply le_antisym; [ exact Hab | ].
      rewrite <- Hac in Hbc. exact Hbc.
  Qed.

  Theorem lt_irrefl : ∀ a : N, ¬ a < a.
  Proof. intros a [_ Hne]. exact (Hne (reflexivity a)). Qed.

  Theorem lt_of_le_of_ne : ∀ a b : N, a ≤ b → a =/= b → a < b.
  Proof. intros a b Hle Hne. split; [ exact Hle | exact Hne ]. Qed.

  Theorem le_of_lt_succ : ∀ a b : N, a < succ b → a ≤ b.
  Proof.
    intros a b [Hle Hne].
    destruct (proj1 (le_add a (succ b)) Hle) as [c Hc].
    destruct (zero_or_succ c) as [Hz | [d Hd]].
    - exfalso. apply Hne.
      calc a
        == (a + 0) { right_identity }
        == (a + c) { Hz }
        == (succ b) { Hc }.
    - apply le_add. ∃ d. apply (injective (f := succ)).
      calc (succ (a + d))
        == (a + succ d) { add_succ }
        == (a + c) { Hd }
        == (succ b) { Hc }.
  Qed.

  Theorem lt_succ_iff : ∀ a b : N, a < succ b ↔ a < b ∨ a == b.
  Proof.
    intros a b. split.
    - intros Hlt.
      destruct (eq_or_ne a b) as [Heq | Hne]; [ right; exact Heq | ].
      left. exact (lt_of_le_of_ne a b (le_of_lt_succ a b Hlt) Hne).
    - intros [[Hle Hne] | Heq].
      + split.
        × exact (le_trans a b (succ b) Hle (le_succ_diag b)).
        × intros Hab. apply (succ_ne_self b). symmetry.
          apply le_antisym; [ exact (le_succ_diag b) | rewrite <- Hab; exact Hle ].
      + destruct (lt_succ_diag b) as [Hle Hne]. split.
        × setoid_rewrite Heq. exact Hle.
        × intros Hab. apply Hne. transitivity a; [ symmetry; exact Heq | exact Hab ].
  Qed.

End NatOrderTheory.
