From Riktig Require Export Structure.NatOrder Theory.Naturals.

Section NatOrderTheory.
  Context {N : Type} `{NatOrder N}.

  Theorem le_refl : ∀ a : N, a ≤ a.
  Proof. intros a. apply le_add. ∃ 0. exact (right_identity a). Qed.

  Theorem le_trans : ∀ a b c : N, a ≤ b → b ≤ c → a ≤ c.
  Proof.
    intros a b c Hab Hbc.
    destruct (proj1 (le_add a b) Hab) as [x Hx].
    destruct (proj1 (le_add b c) Hbc) as [y Hy].
    apply le_add. ∃ (x + y).
    calc (a + (x + y))
      == ((a + x) + y) { associative }
      == (b + y) { Hx }
      == c { Hy }.
  Qed.

  Theorem le_antisym : ∀ a b : N, a ≤ b → b ≤ a → a == b.
  Proof.
    intros a b Hab Hba.
    destruct (proj1 (le_add a b) Hab) as [x Hx].
    destruct (proj1 (le_add b a) Hba) as [y Hy].
    assert (Hxy : x + y == 0).
    { apply (eq_zero_of_add_eq_self a).
      calc (a + (x + y))
        == ((a + x) + y) { associative }
        == (b + y) { Hx }
        == a { Hy }. }
    assert (Hx0 : x == 0) by exact (eq_zero_of_add_eq_zero x y Hxy).
    calc a
      == (a + 0) { right_identity }
      == (a + x) { Hx0 }
      == b { Hx }.
  Qed.

  Theorem zero_le : ∀ a : N, 0 ≤ a.
  Proof. intros a. apply le_add. ∃ a. exact (left_identity a). Qed.

  Theorem le_add_r : ∀ a c : N, a ≤ a + c.
  Proof. intros a c. apply le_add. ∃ c. reflexivity. Qed.

  Theorem le_succ_diag : ∀ a : N, a ≤ succ a.
  Proof. intros a. apply le_add. ∃ 1. symmetry. exact (succ_add a). Qed.

  Theorem succ_le_succ : ∀ a b : N, a ≤ b → succ a ≤ succ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (le_add a b) Hab) as [c Hc].
    apply le_add. ∃ c.
    calc (succ a + c)
      == (succ (a + c)) { add_succ_l }
      == (succ b) { Hc }.
  Qed.

  Theorem le_of_succ_le_succ : ∀ a b : N, succ a ≤ succ b → a ≤ b.
  Proof.
    intros a b Hab.
    destruct (proj1 (le_add (succ a) (succ b)) Hab) as [c Hc].
    apply le_add. ∃ c.
    apply (injective (f := succ)).
    calc (succ (a + c))
      == (succ a + c) { add_succ_l }
      == (succ b) { Hc }.
  Qed.

  Theorem not_lt_zero : ∀ a : N, ¬ a < 0.
  Proof.
    intros a [Ha Hne].
    destruct (proj1 (le_add a 0) Ha) as [c Hc].
    apply Hne. exact (eq_zero_of_add_eq_zero a c Hc).
  Qed.

  Theorem lt_succ_diag : ∀ a : N, a < succ a.
  Proof.
    intros a. split; [ exact (le_succ_diag a) | ].
    intros Ha. apply (succ_ne_self a). symmetry. exact Ha.
  Qed.

  Theorem zero_lt_one : (0 : N) < 1.
  Proof.
    split; [ exact (zero_le 1) | ].
    intros H01. apply (succ_not_zero (0 : N)).
    calc (succ (0 : N))
      == (1 : N) { one_succ }
      == (0 : N) { H01 }.
  Qed.

  Theorem eq_zero_of_lt_one : ∀ k : N, k < 1 → k == 0.
  Proof.
    intros k [Hk Hne].
    destruct (proj1 (le_add k 1) Hk) as [c Hc].
    destruct (zero_or_succ k) as [Hz | [m Hm]]; [ exact Hz | ].
    exfalso. apply Hne.
    assert (Hmc : m + c == 0).
    { apply (injective (f := succ)).
      calc (succ (m + c))
        == (succ m + c) { add_succ_l }
        == (k + c) { Hm }
        == (1 : N) { Hc }
        == (succ (0 : N)) { one_succ }. }
    calc k
      == (succ m) { Hm }
      == (succ (0 : N)) { (eq_zero_of_add_eq_zero m c Hmc) }
      == (1 : N) { one_succ }.
  Qed.

End NatOrderTheory.
