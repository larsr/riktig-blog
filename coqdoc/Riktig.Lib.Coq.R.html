From Coq Require Import Reals.
Open Scope R_scope.

What RIneq and Raxioms state in other shapes, in the shape the ordered field and the reals ask for.
Lemma Rle_wd : ∀ a a' : R, a = a' → ∀ b b' : R, b = b' → (a ≤ b ↔ a' ≤ b').
Proof. intros a a' Ha b b' Hb. rewrite Ha, Hb. reflexivity. Qed.

Lemma Rle_total : ∀ a b : R, a ≤ b ∨ b ≤ a.
Proof.
  intros a b.
  destruct (Rle_or_lt a b) as [H | H]; [ left; exact H | right; exact (Rlt_le b a H) ].
Qed.

Lemma Rle_stable : ∀ a b : R, ¬ ¬ a ≤ b → a ≤ b.
Proof.
  intros a b Hnn.
  destruct (Rle_dec a b) as [H | H]; [ exact H | destruct (Hnn H) ].
Qed.

Lemma Rplus_le_mono_l : ∀ a b c : R, a ≤ b → c + a ≤ c + b.
Proof. intros a b c Hab. exact (Rplus_le_compat_l c a b Hab). Qed.

Lemma R_complete : ∀ P : R → Prop,
  (∃ x : R, P x) →
  (∃ b : R, ∀ x : R, P x → x ≤ b) →
  ∃ s : R,
    (∀ x : R, P x → x ≤ s) ∧
    (∀ b : R, (∀ x : R, P x → x ≤ b) → s ≤ b).
Proof.
  intros P Hinh [b Hb].
  destruct (completeness P (ex_intro _ b Hb) Hinh) as [s [Hub Hleast]].
  ∃ s. split.
  - exact Hub.
  - intros c Hc. exact (Hleast c Hc).
Qed.
