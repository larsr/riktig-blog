The twisted group algebra of an involutive group: coefficient families over an index group B, multiplied by convolution with a two-cocycle in front. Associativity is the cocycle equation, the units are its normalization, and bilinearity is the enumeration's linearity — no representation of B appears in any of them, and no order on its generators.

  Lemma add_self : ∀ a : B, a + a == 0.
  Proof. exact alternates. Qed.

  Lemma add_swap_mid : ∀ a b c : B, (a + b) + c == b + (a + c).
  Proof.
    intros a b c.
    calc ((a + b) + c)
      == (a + (b + c)) { (symmetry (associative a b c)) }
      == (b + (a + c)) { (add_swap_front a b c) }.
  Qed.

  Lemma sigma_cocycle : ∀ a b c : B,
    sigma a b × sigma (a + b) c == sigma b c × sigma a (b + c).
  Proof. exact cocycle_equation. Qed.

  Lemma sigma_zero_l : ∀ a : B, sigma 0 a == 1.
  Proof. exact cocycle_left_normal. Qed.

  Lemma sigma_zero_r : ∀ a : B, sigma a 0 == 1.
  Proof. exact cocycle_right_normal. Qed.

  Lemma add_left_proper : ∀ b : B, Proper (equ ==> equ) (fun a ⇒ b + a).
  Proof.
    intros b a a' Haa'. exact (op_proper b b (reflexivity b) a a' Haa').
  Qed.


  Definition twist (x y : B → F) : B → F :=
    fun d ⇒ bigopI (fun a ⇒ sigma a (a + d) × (x a × y (a + d))).

  Definition twist_one : B → F := fun a ⇒ if dec (a == 0) then 1 else 0.


  #[export] Instance sigma_left_proper : ∀ b : B,
    Proper (equ ==> equ) (sigma b) :=
    fun b ⇒ cocycle_proper b b (reflexivity b).

  #[export] Instance twist_one_proper : Proper (equ ==> equ) twist_one.
  Proof.
    intros a a' Haa'. unfold twist_one.
    destruct (dec (a == 0)) as [Ha | Ha]; destruct (dec (a' == 0)) as [Ha' | Ha'].
    - reflexivity.
    - exfalso. exact (Ha' (transitivity (symmetry Haa') Ha)).
    - exfalso. exact (Ha (transitivity Haa' Ha')).
    - reflexivity.
  Qed.

  Lemma twist_assoc_summand_proper : ∀ (b d : B) (x y z : B → F),
    Proper (equ ==> equ) y → Proper (equ ==> equ) z →
    Proper (equ ==> equ)
      (fun a ⇒ sigma a (a + d) × (sigma b (b + a) × (x b × y (b + a)))
                × z (a + d)).
  Proof.
    intros b d x y z Hy Hz a a' Haa'.
    assert (Hleft : b + a == b + a') by (rewrite Haa'; reflexivity).
    assert (Hright : a + d == a' + d) by (rewrite Haa'; reflexivity).
    rewrite (cocycle_proper a a' Haa' _ _ Hright),
            (cocycle_proper b b (reflexivity b) _ _ Hleft),
            (Hy _ _ Hleft), (Hz _ _ Hright).
    reflexivity.
  Qed.

  Lemma twist_unit_summand_proper_l : ∀ (d : B) (x : B → F),
    Proper (equ ==> equ) x →
    Proper (equ ==> equ) (fun a ⇒ sigma a (a + d) × (twist_one a × x (a + d))).
  Proof.
    intros d x Hx a a' Haa'.
    assert (Hright : a + d == a' + d) by (rewrite Haa'; reflexivity).
    rewrite (cocycle_proper a a' Haa' _ _ Hright),
            (twist_one_proper a a' Haa'), (Hx _ _ Hright).
    reflexivity.
  Qed.

  Lemma twist_unit_summand_proper_r : ∀ (d : B) (x : B → F),
    Proper (equ ==> equ) x →
    Proper (equ ==> equ) (fun a ⇒ sigma a (a + d) × (x a × twist_one (a + d))).
  Proof.
    intros d x Hx a a' Haa'.
    assert (Hright : a + d == a' + d) by (rewrite Haa'; reflexivity).
    rewrite (cocycle_proper a a' Haa' _ _ Hright),
            (Hx a a' Haa'), (twist_one_proper _ _ Hright).
    reflexivity.
  Qed.


  Lemma twist_assoc : ∀ x y z : B → F,
    Proper (equ ==> equ) y → Proper (equ ==> equ) z →
    twist (twist x y) z == twist x (twist y z).
  Proof.
    intros x y z Hy_respects Hz_respects d. unfold twist.
    transitivity (bigopI (fun a ⇒ bigopI (fun b ⇒
      sigma a (a + d) × (sigma b (b + a) × (x b × y (b + a))) × z (a + d)))).
    { apply bigopI_ext. intros a.
      transitivity (bigopI (fun b ⇒
        sigma a (a + d)
        × (sigma b (b + a) × (x b × y (b + a)) × z (a + d)))).
      - rewrite (bigopI_mul_l (sigma a (a + d))
          (fun b ⇒ sigma b (b + a) × (x b × y (b + a)) × z (a + d))).
        rewrite (bigopI_mul_r (z (a + d))
          (fun b ⇒ sigma b (b + a) × (x b × y (b + a)))).
        ring.
      - apply bigopI_ext. intros b. ring. }
    
    transitivity (bigopI (fun b ⇒ bigopI (fun c ⇒
      sigma b (b + d) × (x b × (sigma c (c + (b + d))
        × (y c × z (c + (b + d)))))))).
    2: { apply bigopI_ext. intros b.
         rewrite (bigopI_mul_l (sigma b (b + d))
           (fun c ⇒ x b × (sigma c (c + (b + d))
                            × (y c × z (c + (b + d)))))).
         rewrite (bigopI_mul_l (x b)
           (fun c ⇒ sigma c (c + (b + d)) × (y c × z (c + (b + d))))).
         reflexivity. }
    rewrite (bigopI_swap (fun a b ⇒
      sigma a (a + d) × (sigma b (b + a) × (x b × y (b + a))) × z (a + d))).
    apply bigopI_ext. intros b.
    rewrite <- (bigopI_reindex
      (fun a ⇒ sigma a (a + d) × (sigma b (b + a) × (x b × y (b + a)))
                × z (a + d))
      (fun a ⇒ b + a) (fun a ⇒ b + a)
      (twist_assoc_summand_proper b d x y z Hy_respects Hz_respects)
      (add_left_proper b) (add_left_proper b)
      (add_shift b) (add_shift b)).
    apply bigopI_ext. intros c.
    rewrite (add_shift b c).
    pose proof (sigma_cocycle b c ((b + c) + d)) as Hcocycle.
    assert (Hcb : c + (b + c) == b).
    { calc (c + (b + c))
        == (c + (c + b)) { (commutative b c) }
        == b { (add_shift c b) }. }
    assert (Hce : c + ((b + c) + d) == b + d).
    { calc (c + ((b + c) + d))
        == ((c + (b + c)) + d) { (associative c (b + c) d) }
        == (b + d) { Hcb }. }
    rewrite Hce in Hcocycle.
    assert (Hcoefficient : sigma (b + c) ((b + c) + d) × sigma b c
                           == sigma b (b + d) × sigma c ((b + c) + d)).
    { transitivity (sigma b c × sigma (b + c) ((b + c) + d)); [ ring |].
      rewrite Hcocycle. ring. }
    
    assert (Hz : c + (b + d) == (b + c) + d)
      by (symmetry; apply add_swap_mid).
    rewrite Hz.
    transitivity ((sigma (b + c) ((b + c) + d) × sigma b c)
                  × (x b × (y c × z ((b + c) + d)))); [ ring |].
    rewrite Hcoefficient. ring.
  Qed.


  Lemma twist_one_at_zero : twist_one 0 == 1.
  Proof.
    unfold twist_one. destruct (dec ((0 : B) == 0)) as [_ | Hne].
    - reflexivity.
    - exfalso. exact (Hne (reflexivity 0)).
  Qed.

  Lemma twist_one_off_zero : ∀ a : B, ¬ (a == 0) → twist_one a == 0.
  Proof.
    intros a Ha. unfold twist_one.
    destruct (dec (a == 0)) as [Heq | _]; [ exfalso; exact (Ha Heq) | reflexivity ].
  Qed.

  Lemma twist_one_l : ∀ x : B → F,
    Proper (equ ==> equ) x → twist twist_one x == x.
  Proof.
    intros x Hx d. unfold twist.
    transitivity (sigma 0 ((0 : B) + d) × (twist_one 0 × x ((0 : B) + d))).
    { apply (bigopI_support_singleton 0
        (fun a ⇒ sigma a (a + d) × (twist_one a × x (a + d)))
        (twist_unit_summand_proper_l d x Hx)).
      intros a Ha.
      rewrite (twist_one_off_zero a Ha). ring. }
    rewrite twist_one_at_zero, (left_identity (B := B) d),
            (sigma_zero_l d).
    ring.
  Qed.

  Lemma twist_one_r : ∀ x : B → F,
    Proper (equ ==> equ) x → twist x twist_one == x.
  Proof.
    intros x Hx d. unfold twist.
    transitivity (sigma d (d + d) × (x d × twist_one (d + d))).
    { apply (bigopI_support_singleton d
        (fun a ⇒ sigma a (a + d) × (x a × twist_one (a + d)))
        (twist_unit_summand_proper_r d x Hx)).
      intros a Ha.
      rewrite (twist_one_off_zero (a + d)
        (fun Hzero ⇒ Ha (add_zero_unique a d Hzero))).
      ring. }
    rewrite (add_self d), twist_one_at_zero,
            (sigma_zero_r d).
    ring.
  Qed.


  Lemma twist_proper : Proper (equ ==> equ ==> equ) twist.
  Proof.
    intros x x' Hx y y' Hy d. unfold twist.
    apply bigopI_ext. intros a.
    rewrite (Hx a), (Hy (a + d)). reflexivity.
  Qed.

  Lemma twist_dist_l : ∀ x y z : B → F,
    twist x (y + z) == twist x y + twist x z.
  Proof.
    intros x y z d. unfold twist.
    transitivity (bigopI (fun a ⇒
      sigma a (a + d) × (x a × y (a + d))
      + sigma a (a + d) × (x a × z (a + d)))).
    - apply bigopI_ext. intros a.
      rewrite (pw_add_at y z (a + d)). ring.
    - apply (bigopI_add
        (fun a ⇒ sigma a (a + d) × (x a × y (a + d)))
        (fun a ⇒ sigma a (a + d) × (x a × z (a + d)))).
  Qed.

  Lemma twist_dist_r : ∀ x y z : B → F,
    twist (x + y) z == twist x z + twist y z.
  Proof.
    intros x y z d. unfold twist.
    transitivity (bigopI (fun a ⇒
      sigma a (a + d) × (x a × z (a + d))
      + sigma a (a + d) × (y a × z (a + d)))).
    - apply bigopI_ext. intros a.
      rewrite (pw_add_at x y a). ring.
    - apply (bigopI_add
        (fun a ⇒ sigma a (a + d) × (x a × z (a + d)))
        (fun a ⇒ sigma a (a + d) × (y a × z (a + d)))).
  Qed.

  Lemma twist_scale_l : ∀ (k : F) (x y : B → F),
    twist (k *: x) y == k *: twist x y.
  Proof.
    intros k x y d. unfold twist.
    transitivity (k × bigopI (fun a ⇒ sigma a (a + d) × (x a × y (a + d)))).
    2: reflexivity.
    rewrite <- (bigopI_mul_l k
      (fun a ⇒ sigma a (a + d) × (x a × y (a + d)))).
    apply bigopI_ext. intros a.
    rewrite (pw_scale_at k x a), (scale_is_mult k (x a)). ring.
  Qed.

  Lemma twist_scale_r : ∀ (k : F) (x y : B → F),
    twist x (k *: y) == k *: twist x y.
  Proof.
    intros k x y d. unfold twist.
    transitivity (k × bigopI (fun a ⇒ sigma a (a + d) × (x a × y (a + d)))).
    2: reflexivity.
    rewrite <- (bigopI_mul_l k
      (fun a ⇒ sigma a (a + d) × (x a × y (a + d)))).
    apply bigopI_ext. intros a.
    rewrite (pw_scale_at k y (a + d)), (scale_is_mult k (y (a + d))). ring.
  Qed.

  Context (families_proper : ∀ f : B → F, Proper (equ ==> equ) f).

  #[local] Instance Twist_One : One (B → F) := twist_one.
  #[local] Instance Twist_Mult : HasMult (B → F) := twist.

  #[local] Instance twist_mult_proper :
    Proper (equ ==> equ ==> equ) (@mult (B → F) _) := twist_proper.
  #[local] Instance twist_associative :
    Associative (B → F) (B → F) mult mult :=
    fun x y z ⇒ twist_assoc x y z (families_proper y) (families_proper z).
  #[local] Instance twist_left_identity :
    LeftIdentity (B → F) (B → F) mult 1 :=
    fun x ⇒ twist_one_l x (families_proper x).
  #[local] Instance twist_right_identity :
    RightIdentity (B → F) (B → F) mult 1 :=
    fun x ⇒ twist_one_r x (families_proper x).
  #[local] Instance twist_left_distributive :
    LeftDistributive (B → F) (B → F) (B → F) mult := twist_dist_l.
  #[local] Instance twist_right_distributive :
    RightDistributive (B → F) (B → F) (B → F) mult := twist_dist_r.
