A doubling of a graded geometric algebra is one again: the grades, the scalar part and the reversal of the doubled algebra agree with its geometric structure, the top grade one higher.
Section GradedGADouble.
  Context {F V A N : Type} `{GradedGA N F V A} (s : F).

  Local Notation "'low' k b" := (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨b⟩_j) k) (at level 10, k at level 9, b at level 9).

  #[export] Instance Double_HasScalarPart : HasScalarPart F (Double s A) := fun x ⇒ scalar_part (fst x).

  #[export] Instance Double_GradedGA :
    GradedGA N F (Extend s V) (Double s A) (Double_HasGrade s) (succ top_grade) Double_HasScalarPart (Double_HasReverse s).
  Proof.
    split.
    - exact (Double_Graded s).
    - exact (Double_Reversal s).
    - split; simpl.
      + exact one_grade_zero.
      + exact (low_zero (0 : A)).
    - intros [a b]. split; simpl.
      + exact (grade_zero_scalar a).
      + calc (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨b⟩_j) 0)
          == (0 : A) { (low_zero b) }
          == (scalar_part a *: (0 : A)) { (symmetry (smul_zero (scalar_part a))) }.
    - intros [v c]. split; simpl.
      + exact (vec_grade_one v).
      + calc (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨c *: (1 : A)⟩_j) (succ 0))
          == (⟨c *: (1 : A)⟩_(0 : N)) { (low_succ 0 (c *: (1 : A))) }
          == (c *: (1 : A)) { (scalar_grade_zero c) }.
    -
      intros [v c] [a b].
      assert (Hx0 : (⟨((a, b) : Double s A)⟩_(0 : N) : Double s A) == (⟨a⟩_(0 : N), (0 : A))).
      { split; simpl; [ reflexivity | exact (low_zero b) ]. }
      assert (HP1 : vec v × ⟨a⟩_(0 : N) + s *: ((c *: (1 : A)) × involute (0 : A)) == vec v × ⟨a⟩_(0 : N)).
      { calc (vec v × ⟨a⟩_(0 : N) + s *: ((c *: (1 : A)) × involute (0 : A)))
          == (vec v × ⟨a⟩_(0 : N) + s *: ((c *: (1 : A)) × (0 : A)))
               { (op_proper _ _ (reflexivity _) _ _ (scale_proper _ _ (reflexivity s) _ _ (op_proper _ _ (reflexivity _) _ _ involute_zero))) }
          == (vec v × ⟨a⟩_(0 : N) + s *: (0 : A))
               { (op_proper _ _ (reflexivity _) _ _ (scale_proper _ _ (reflexivity s) _ _ (Theory.Ring.mul_zero (c *: (1 : A))))) }
          == (vec v × ⟨a⟩_(0 : N) + (0 : A))
               { (op_proper _ _ (reflexivity _) _ _ (smul_zero s)) }
          == (vec v × ⟨a⟩_(0 : N))
               { (right_identity (action := add) (vec v × ⟨a⟩_(0 : N))) }. }
      assert (HP2 : vec v × (0 : A) + (c *: (1 : A)) × involute ⟨a⟩_(0 : N) == c *: ⟨involute a⟩_(0 : N)).
      { calc (vec v × (0 : A) + (c *: (1 : A)) × involute ⟨a⟩_(0 : N))
          == ((0 : A) + (c *: (1 : A)) × involute ⟨a⟩_(0 : N))
               { (op_proper _ _ (Theory.Ring.mul_zero (vec v)) _ _ (reflexivity _)) }
          == ((c *: (1 : A)) × involute ⟨a⟩_(0 : N)) { (left_identity (action := add) _) }
          == (c *: involute ⟨a⟩_(0 : N)) { (smul_one_mul c (involute ⟨a⟩_(0 : N))) }
          == (c *: ⟨involute a⟩_(0 : N)) { (scale_proper _ _ (reflexivity c) _ _ (involute_grade 0 a)) }. }
      assert (HX : ((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(0 : N), (0 : A)) : Double s A)
                   == ⟨((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(0 : N), (0 : A)) : Double s A)⟩_(succ (0 : N))).
      { split; simpl.
        + calc (vec v × ⟨a⟩_(0 : N) + s *: ((c *: (1 : A)) × involute (0 : A)))
            == (vec v × ⟨a⟩_(0 : N)) { HP1 }
            == (⟨vec v × ⟨a⟩_(0 : N)⟩_(succ (0 : N))) { (vec_mul_grade_zero v a) }
            == (⟨vec v × ⟨a⟩_(0 : N) + s *: ((c *: (1 : A)) × involute (0 : A))⟩_(succ (0 : N)))
                 { (grade_proper (succ 0) _ _ (symmetry HP1)) }.
        + calc (vec v × (0 : A) + (c *: (1 : A)) × involute ⟨a⟩_(0 : N))
            == (c *: ⟨involute a⟩_(0 : N)) { HP2 }
            == (⟨c *: ⟨involute a⟩_(0 : N)⟩_(0 : N)) { (grade_pure_smul 0 c (involute a)) }
            == (⟨vec v × (0 : A) + (c *: (1 : A)) × involute ⟨a⟩_(0 : N)⟩_(0 : N))
                 { (grade_proper 0 _ _ (symmetry HP2)) }
            == (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨vec v × (0 : A) + (c *: (1 : A)) × involute ⟨a⟩_(0 : N)⟩_j) (succ 0))
                 { (symmetry (low_succ 0 _)) }. }
      calc ((vec ((v, c) : Extend s V) : Double s A) × ⟨((a, b) : Double s A)⟩_(0 : N))
        == ((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(0 : N), (0 : A)))
             { (op_proper _ _ (reflexivity _) _ _ Hx0) }
        == (⟨(vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(0 : N), (0 : A))⟩_(succ (0 : N)))
             { HX }
        == (⟨(vec ((v, c) : Extend s V) : Double s A) × ⟨((a, b) : Double s A)⟩_(0 : N)⟩_(succ (0 : N)))
             { (grade_proper (F := F) (succ 0) _ _ (op_proper _ _ (reflexivity _) _ _ (symmetry Hx0))) }.
    -
      intros [v c] [a b] k.
      assert (Hx1 : (⟨((a, b) : Double s A)⟩_(succ k) : Double s A) == (⟨a⟩_(succ k), ⟨b⟩_k)).
      { split; simpl; [ reflexivity | exact (low_succ k b) ]. }
      assert (HT2 : s *: ((c *: (1 : A)) × involute ⟨b⟩_k) == (s × c) *: ⟨involute b⟩_k).
      { calc (s *: ((c *: (1 : A)) × involute ⟨b⟩_k))
          == (s *: (c *: involute ⟨b⟩_k)) { (scale_proper _ _ (reflexivity s) _ _ (smul_one_mul c _)) }
          == (s *: (c *: ⟨involute b⟩_k)) { (scale_proper _ _ (reflexivity s) _ _ (scale_proper _ _ (reflexivity c) _ _ (involute_grade k b))) }
          == ((s × c) *: ⟨involute b⟩_k) { (symmetry (associative (op := mult) (action := scale) s c _)) }. }
      assert (HS2 : (c *: (1 : A)) × involute ⟨a⟩_(succ k) == c *: ⟨involute a⟩_(succ k)).
      { calc ((c *: (1 : A)) × involute ⟨a⟩_(succ k))
          == (c *: involute ⟨a⟩_(succ k)) { (smul_one_mul c _) }
          == (c *: ⟨involute a⟩_(succ k)) { (scale_proper _ _ (reflexivity c) _ _ (involute_grade (succ k) a)) }. }
      assert (Hsplit2 : s *: ((c *: (1 : A)) × involute ⟨b⟩_k)
                        == ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k + ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k))).
      { assert (HT2k : ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k == s *: ((c *: (1 : A)) × involute ⟨b⟩_k)).
        { calc (⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k)
            == (⟨(s × c) *: ⟨involute b⟩_k⟩_k) { (grade_proper k _ _ HT2) }
            == ((s × c) *: ⟨involute b⟩_k) { (symmetry (grade_pure_smul k (s × c) (involute b))) }
            == (s *: ((c *: (1 : A)) × involute ⟨b⟩_k)) { (symmetry HT2) }. }
        assert (HT2ss : ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k)) == 0).
        { calc (⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k)))
            == (⟨(s × c) *: ⟨involute b⟩_k⟩_(succ (succ k))) { (grade_proper (succ (succ k)) _ _ HT2) }
            == (0 : A) { (grade_pure_orth (succ (succ k)) k (s × c) (involute b) (ne_succ_succ' k)) }. }
        calc (s *: ((c *: (1 : A)) × involute ⟨b⟩_k))
          == (⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k) { (symmetry HT2k) }
          == (⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k + 0) { (symmetry (right_identity (action := add) _)) }
          == (⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k + ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k)))
               { (op_proper _ _ (reflexivity _) _ _ (symmetry HT2ss)) }. }
      assert (Hfst : vec v × ⟨a⟩_(succ k) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k)
                     == ⟨vec v × ⟨a⟩_(succ k) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k
                        + ⟨vec v × ⟨a⟩_(succ k) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k))).
      { calc (vec v × ⟨a⟩_(succ k) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k))
          == ((⟨vec v × ⟨a⟩_(succ k)⟩_k + ⟨vec v × ⟨a⟩_(succ k)⟩_(succ (succ k))) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k))
               { (op_proper _ _ (vec_mul_grade_succ v a k) _ _ (reflexivity _)) }
          == ((⟨vec v × ⟨a⟩_(succ k)⟩_k + ⟨vec v × ⟨a⟩_(succ k)⟩_(succ (succ k)))
              + (⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k + ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k))))
               { (op_proper _ _ (reflexivity _) _ _ Hsplit2) }
          == ((⟨vec v × ⟨a⟩_(succ k)⟩_k + ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k)
              + (⟨vec v × ⟨a⟩_(succ k)⟩_(succ (succ k)) + ⟨s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k))))
               { (add_add_add_comm _ _ _ _) }
          == (⟨vec v × ⟨a⟩_(succ k) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_k
              + ⟨vec v × ⟨a⟩_(succ k) + s *: ((c *: (1 : A)) × involute ⟨b⟩_k)⟩_(succ (succ k)))
               { (op_proper _ _ (symmetry (grade_add k _ _)) _ _ (symmetry (grade_add (succ (succ k)) _ _))) }. }
      assert (Hsnd : vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k)
                     == low k (vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
                        + ⟨vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k)).
      { assert (HlowS2 : low k ((c *: (1 : A)) × involute ⟨a⟩_(succ k)) == 0).
        { calc (low k ((c *: (1 : A)) × involute ⟨a⟩_(succ k)))
            == (low k (c *: ⟨involute a⟩_(succ k))) { (low_proper k _ _ HS2) }
            == (0 : A) { (low_of_pure_above k c (involute a)) }. }
        assert (HgrS2 : ⟨(c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k) == (c *: (1 : A)) × involute ⟨a⟩_(succ k)).
        { calc (⟨(c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k))
            == (⟨c *: ⟨involute a⟩_(succ k)⟩_(succ k)) { (grade_proper (succ k) _ _ HS2) }
            == (c *: ⟨involute a⟩_(succ k)) { (symmetry (grade_pure_smul (succ k) c (involute a))) }
            == ((c *: (1 : A)) × involute ⟨a⟩_(succ k)) { (symmetry HS2) }. }
        calc (vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
          == ((low k (vec v × ⟨b⟩_k) + ⟨vec v × ⟨b⟩_k⟩_(succ k)) + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
               { (op_proper _ _ (vec_mul_grade_low v b k) _ _ (reflexivity _)) }
          == ((low k (vec v × ⟨b⟩_k) + ⟨vec v × ⟨b⟩_k⟩_(succ k))
              + (low k ((c *: (1 : A)) × involute ⟨a⟩_(succ k)) + ⟨(c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k)))
               { (op_proper _ _ (reflexivity _) _ _ (symmetry (transitivity (op_proper _ _ HlowS2 _ _ HgrS2) (left_identity (action := add) _)))) }
          == ((low k (vec v × ⟨b⟩_k) + low k ((c *: (1 : A)) × involute ⟨a⟩_(succ k)))
              + (⟨vec v × ⟨b⟩_k⟩_(succ k) + ⟨(c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k)))
               { (add_add_add_comm _ _ _ _) }
          == (low k (vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
              + ⟨vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k))
               { (op_proper _ _ (symmetry (low_add k _ _)) _ _ (symmetry (grade_add (succ k) _ _))) }. }
      assert (HX : ((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(succ k), ⟨b⟩_k) : Double s A)
                   == ⟨((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(succ k), ⟨b⟩_k) : Double s A)⟩_k
                      + ⟨((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(succ k), ⟨b⟩_k) : Double s A)⟩_(succ (succ k))).
      { split; simpl.
        + exact Hfst.
        + calc (vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
            == (low k (vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
                + ⟨vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_(succ k))
                 { Hsnd }
            == (low k (vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k))
                + natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨vec v × ⟨b⟩_k + (c *: (1 : A)) × involute ⟨a⟩_(succ k)⟩_j) (succ (succ k)))
                 { (op_proper _ _ (reflexivity _) _ _ (symmetry (low_succ (succ k) _))) }. }
      calc ((vec ((v, c) : Extend s V) : Double s A) × ⟨((a, b) : Double s A)⟩_(succ k))
        == ((vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(succ k), ⟨b⟩_k))
             { (op_proper _ _ (reflexivity _) _ _ Hx1) }
        == (⟨(vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(succ k), ⟨b⟩_k)⟩_k
            + ⟨(vec ((v, c) : Extend s V) : Double s A) × (⟨a⟩_(succ k), ⟨b⟩_k)⟩_(succ (succ k)))
             { HX }
        == (⟨(vec ((v, c) : Extend s V) : Double s A) × ⟨((a, b) : Double s A)⟩_(succ k)⟩_k
            + ⟨(vec ((v, c) : Extend s V) : Double s A) × ⟨((a, b) : Double s A)⟩_(succ k)⟩_(succ (succ k)))
             { (op_proper _ _ (grade_proper (F := F) k _ _ (op_proper _ _ (reflexivity _) _ _ (symmetry Hx1))) _ _ (grade_proper (F := F) (succ (succ k)) _ _ (op_proper _ _ (reflexivity _) _ _ (symmetry Hx1)))) }.
    -
      intros k [a b].
      assert (Hinv : involute (low k b) == low k (involute b)).
      { destruct (zero_or_succ k) as [Hk | [m Hk]].
        - calc (involute (low k b))
            == (involute (low 0 b)) { (involute_proper _ _ (low_index_proper k 0 b Hk)) }
            == (involute (0 : A)) { (involute_proper _ _ (low_zero b)) }
            == (0 : A) { involute_zero }
            == (low 0 (involute b)) { (symmetry (low_zero (involute b))) }
            == (low k (involute b)) { (low_index_proper 0 k (involute b) (symmetry Hk)) }.
        - calc (involute (low k b))
            == (involute (low (succ m) b)) { (involute_proper _ _ (low_index_proper k (succ m) b Hk)) }
            == (involute ⟨b⟩_m) { (involute_proper _ _ (low_succ m b)) }
            == (⟨involute b⟩_m) { (involute_grade m b) }
            == (low (succ m) (involute b)) { (symmetry (low_succ m (involute b))) }
            == (low k (involute b)) { (low_index_proper (succ m) k (involute b) (symmetry Hk)) }. }
      split; simpl.
      + exact (involute_grade k a).
      + calc (- involute (low k b))
          == (- (low k (involute b))) { (neg_proper _ _ Hinv) }
          == (low k (- involute b)) { (symmetry (low_neg k (involute b))) }.
    -
      intros k [a b].
      assert (Hrev : involute (reverse (low k b)) == low k (involute (reverse b))).
      { destruct (zero_or_succ k) as [Hk | [m Hk]].
        - calc (involute (reverse (low k b)))
            == (involute (reverse (low 0 b))) { (involute_proper _ _ (reverse_proper _ _ (low_index_proper k 0 b Hk))) }
            == (involute (reverse (0 : A))) { (involute_proper _ _ (reverse_proper _ _ (low_zero b))) }
            == (involute (0 : A)) { (involute_proper _ _ reverse_zero) }
            == (0 : A) { involute_zero }
            == (low 0 (involute (reverse b))) { (symmetry (low_zero _)) }
            == (low k (involute (reverse b))) { (low_index_proper 0 k _ (symmetry Hk)) }.
        - calc (involute (reverse (low k b)))
            == (involute (reverse (low (succ m) b))) { (involute_proper _ _ (reverse_proper _ _ (low_index_proper k (succ m) b Hk))) }
            == (involute (reverse ⟨b⟩_m)) { (involute_proper _ _ (reverse_proper _ _ (low_succ m b))) }
            == (involute ⟨reverse b⟩_m) { (involute_proper _ _ (reverse_grade m b)) }
            == (⟨involute (reverse b)⟩_m) { (involute_grade m (reverse b)) }
            == (low (succ m) (involute (reverse b))) { (symmetry (low_succ m _)) }
            == (low k (involute (reverse b))) { (low_index_proper (succ m) k _ (symmetry Hk)) }. }
      split; simpl.
      + exact (reverse_grade k a).
      + exact Hrev.
    -
      intros x [a b]. split; simpl.
      + exact (homogeneous (f := reverse) x a).
      + calc (involute (reverse (x *: b)))
          == (involute (x *: reverse b)) { (involute_proper _ _ (homogeneous (f := reverse) x b)) }
          == (x *: involute (reverse b)) { (homogeneous (f := involute) x (reverse b)) }.
    -
      intros [v c]. split; simpl.
      + exact (reverse_vec v).
      + calc (involute (reverse (c *: (1 : A))))
          == (involute (c *: reverse (1 : A))) { (involute_proper _ _ (homogeneous (f := reverse) c (1 : A))) }
          == (involute (c *: (1 : A))) { (involute_proper _ _ (scale_proper _ _ (reflexivity c) _ _ reverse_one)) }
          == (c *: (1 : A)) { (involute_smul_one c) }.
    -
      intros [a b]. split; simpl.
      + exact (reverse_involute a).
      + calc (involute (reverse (- involute b)))
          == (involute (- reverse (involute b))) { (involute_proper _ _ (reverse_neg (involute b))) }
          == (- involute (reverse (involute b))) { (involute_neg (reverse (involute b))) }
          == (- involute (involute (reverse b))) { (neg_proper _ _ (involute_proper _ _ (reverse_involute b))) }.
  Qed.
End GradedGADouble.
