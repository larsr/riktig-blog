From Riktig.Structure Require Export Functor.

A natural transformation between two functors gives every object an arrow between its two images, and the square with the images of any arrow commutes.
Module NaturalTransformation.
  Class NaturalTransformation (C D : Type) `{Category C} `{Category D}
      (F G : C → D) {mapF : HasFmap C D F} {mapG : HasFmap C D G}
      (eta : ∀ x : C, F x ~> G x) := {
    source :: Functor C D F mapF;
    target :: Functor C D G mapG;
    naturality : ∀ (x y : C) (f : x ~> y), fmap (F := G) f ∘ eta x == eta y ∘ fmap (F := F) f;
  }.
  Hint Mode NaturalTransformation ! ! - - - - - - - - - - ! ! - - ! : typeclass_instances.
End NaturalTransformation.
Export NaturalTransformation.
