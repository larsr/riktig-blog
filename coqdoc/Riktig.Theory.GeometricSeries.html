Powers by recursion on the exponent, and the geometric series of halves: its partial sums have a closed form, so its tails from an index on are below any positive element of an Archimedean field.
Section GeometricSeries.
  Context {F N : Type} `{Archimedean F N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Local Notation pow r n := (natrec F 1 (fun _ acc ⇒ acc × r) n).
  Local Notation psum g n := (natrec F 0 (fun k acc ⇒ acc + g k) n).
  Local Notation two := ((1 : F) + 1).
  Local Notation half := (inv ((1 : F) + 1)).

  Theorem pow_zero : ∀ r : F, pow r (0 : N) == 1.
  Proof. intros r. exact (natrec_zero F _ _ 1 _). Qed.

  Theorem pow_succ : ∀ (r : F) (n : N), pow r (succ n) == pow r n × r.
  Proof. intros r n. exact (natrec_succ F _ _ 1 _ n). Qed.

  Theorem pow_proper : ∀ r : F, Proper (equ ==> equ) (fun n : N ⇒ pow r n).
  Proof.
    intros r. apply (natrec_proper F _ _ 1 _).
    intros k k' _ acc acc' Hacc. apply op_proper; [ exact Hacc | reflexivity ].
  Qed.

  Theorem pow_nonneg : ∀ r : F, 0 ≤ r → ∀ n : N, 0 ≤ pow r n.
  Proof.
    intros r Hr.
    apply (nat_induction (fun n ⇒ 0 ≤ pow r n)).
    - intros n n' Hn. setoid_rewrite (pow_proper r n n' Hn). reflexivity.
    - setoid_rewrite (pow_zero r). exact zero_le_one.
    - intros n IH. setoid_rewrite (pow_succ r n). exact (mul_nonneg _ _ IH Hr).
  Qed.

  Theorem pow_add : ∀ (r : F) (i k : N), pow r (i + k) == pow r i × pow r k.
  Proof.
    intros r i.
    apply (nat_induction (fun k ⇒ pow r (i + k) == pow r i × pow r k)).
    - intros k k' Hk.
      setoid_rewrite (pow_proper r (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk)).
      setoid_rewrite (pow_proper r k k' Hk). reflexivity.
    - calc (pow r (i + 0))
        == (pow r i) { (pow_proper r (i + 0) i (Theory.Naturals.add_zero i)) }
        == (pow r i × 1) { (right_identity (action := mult) (pow r i)) }
        == (pow r i × pow r (0 : N)) { (pow_zero r) }.
    - intros k IH.
      calc (pow r (i + succ k))
        == (pow r (succ (i + k))) { (pow_proper r (i + succ k) (succ (i + k)) (Theory.Naturals.add_succ i k)) }
        == (pow r (i + k) × r) { (pow_succ r (i + k)) }
        == ((pow r i × pow r k) × r) { IH }
        == (pow r i × (pow r k × r)) { (associative (op := mult) (action := mult) (pow r i) (pow r k) r) }
        == (pow r i × pow r (succ k)) { (pow_succ r k) }.
  Qed.

  Lemma two_ne_zero : two =/= 0.
  Proof. exact (fun H0 ⇒ proj2 two_pos (symmetry H0)). Qed.

  Lemma half_mul_two : half × two == 1.
  Proof. exact (left_inverse_on (defined_on := nonzero F) two two_ne_zero). Qed.

  Lemma half_nonneg : 0 ≤ half.
  Proof. exact (le_of_lt _ _ (inv_pos two two_pos)). Qed.

Halves and twos cancel power by power.
  Theorem pow_half_mul_pow_two : ∀ n : N, pow half n × pow two n == 1.
  Proof.
    apply (nat_induction (fun n ⇒ pow half n × pow two n == 1)).
    - intros n n' Hn. setoid_rewrite (pow_proper half n n' Hn). setoid_rewrite (pow_proper two n n' Hn). reflexivity.
    - calc (pow half (0 : N) × pow two (0 : N))
        == ((1 : F) × pow two (0 : N)) { (pow_zero half) }
        == ((1 : F) × 1) { (pow_zero two) }
        == (1 : F) { (left_identity (action := mult) (1 : F)) }.
    - intros n IH.
      calc (pow half (succ n) × pow two (succ n))
        == ((pow half n × half) × pow two (succ n)) { (pow_succ half n) }
        == ((pow half n × half) × (pow two n × two)) { (pow_succ two n) }
        == ((pow half n × pow two n) × (half × two)) { (mul_mul_mul_comm (pow half n) half (pow two n) two) }
        == ((1 : F) × (half × two)) { IH }
        == ((1 : F) × 1) { half_mul_two }
        == (1 : F) { (left_identity (action := mult) (1 : F)) }.
  Qed.

  Theorem pow_two_pos : ∀ n : N, 0 < pow two n.
  Proof.
    apply (nat_induction (fun n ⇒ 0 < pow two n)).
    - intros n n' Hn. exact (lt_proper 0 0 (reflexivity _) _ _ (pow_proper two n n' Hn)).
    - apply (proj2 (lt_proper 0 0 (reflexivity _) (pow two (0 : N)) 1 (pow_zero two))). exact zero_lt_one.
    - intros n IH.
      apply (proj2 (lt_proper 0 0 (reflexivity _) (pow two (succ n)) (pow two n × two) (pow_succ two n))).
      exact (mul_pos _ _ IH two_pos).
  Qed.

  Theorem pow_half_eq_inv : ∀ n : N, pow half n == inv (pow two n).
  Proof.
    intros n.
    assert (Hne : pow two n =/= 0) by exact (fun H0 ⇒ proj2 (pow_two_pos n) (symmetry H0)).
    calc (pow half n)
      == (pow half n × 1) { (right_identity (action := mult) (pow half n)) }
      == (pow half n × (pow two n × inv (pow two n))) { (mul_inv_cancel (pow two n) Hne) }
      == ((pow half n × pow two n) × inv (pow two n)) { (associative (op := mult) (action := mult) (pow half n) (pow two n) (inv (pow two n))) }
      == ((1 : F) × inv (pow two n)) { (pow_half_mul_pow_two n) }
      == (inv (pow two n)) { (left_identity (action := mult) (inv (pow two n))) }.
  Qed.

Bernoulli for two: 2^n is at least n + 1.
  Theorem of_nat_succ_le_pow_two : ∀ n : N, of_nat (succ n) ≤ pow two n.
  Proof.
    apply (nat_induction (fun n ⇒ of_nat (succ n) ≤ pow two n)).
    - intros n n' Hn.
      setoid_rewrite (of_nat_equ (succ n) (succ n') (succ_proper n n' Hn)).
      setoid_rewrite (pow_proper two n n' Hn). reflexivity.
    - setoid_rewrite (pow_zero two). setoid_rewrite (of_nat_succ 0). setoid_rewrite of_nat_zero.
      exact (le_of_eq _ _ (left_identity (action := add) (1 : F))).
    - intros n IH.
      assert (Hone : (1 : F) ≤ pow two n).
      { apply (transitivity (le_of_eq _ _ (symmetry (left_identity (action := add) (1 : F))))).
        apply (transitivity (add_le_mono_r 0 (of_nat n) 1 (of_nat_nonneg n))).
        apply (transitivity (le_of_eq _ _ (symmetry (of_nat_succ n)))). exact IH. }
      assert (Hdouble : pow two n + pow two n == pow two n × two).
      { calc (pow two n + pow two n)
          == (pow two n × 1 + pow two n × 1) { (right_identity (action := mult) (pow two n)) }
          == (pow two n × two) { (left_distributive (op := mult) (pow two n) 1 1) }. }
      setoid_rewrite (of_nat_succ (succ n)). setoid_rewrite (pow_succ two n).
      apply (transitivity (add_le_mono _ _ _ _ IH Hone)).
      exact (le_of_eq _ _ Hdouble).
  Qed.

  Theorem inv_le_inv : ∀ a b : F, 0 < a → a ≤ b → inv b ≤ inv a.
  Proof.
    intros a b Ha Hab. apply le_of_not_lt. intros Hlt.
    assert (Hb : 0 < b) by exact (lt_of_lt_of_le 0 a b Ha Hab).
    pose proof (inv_lt_inv (inv a) (inv b) (inv_pos a Ha) Hlt) as H'.
    assert (Hane : a =/= 0) by exact (fun H0 : a == 0 ⇒ proj2 Ha (symmetry H0)).
    assert (Hbne : b =/= 0) by exact (fun H0 : b == 0 ⇒ proj2 Hb (symmetry H0)).
    pose proof (proj1 (lt_proper (inv (inv b)) b (inv_inv b Hbne) (inv (inv a)) a (inv_inv a Hane)) H') as H''.
    exact (proj2 H'' (antisymmetry (proj1 H'') Hab)).
  Qed.

A power of a half is at most the inverse of the successor.
  Theorem pow_half_le_inv_succ : ∀ n : N, pow half n ≤ inv (of_nat (succ n)).
  Proof.
    intros n.
    setoid_rewrite (pow_half_eq_inv n).
    exact (inv_le_inv (of_nat (succ n)) (pow two n) (of_nat_succ_pos n) (of_nat_succ_le_pow_two n)).
  Qed.

The closed form of the partial sums of halves.
  Theorem psum_pow_half : ∀ c : N, psum (fun k ⇒ pow half k) c == two × (1 + - pow half c).
  Proof.
    apply (nat_induction (fun c ⇒ psum (fun k ⇒ pow half k) c == two × (1 + - pow half c))).
    - intros c c' Hc.
      setoid_rewrite (psum_proper (fun k ⇒ pow half k) (pow_proper half) c c' Hc).
      setoid_rewrite (pow_proper half c c' Hc). reflexivity.
    - calc (psum (fun k ⇒ pow half k) (0 : N))
        == (0 : F) { (psum_zero (fun k : N ⇒ pow half k)) }
        == (two × 0) { (mul_zero two) }
        == (two × (1 + - (1 : F))) { (right_inverse (op := add) (1 : F)) }
        == (two × (1 + - pow half (0 : N))) { (pow_zero half) }.
    - intros c IH.
      assert (Hstep : pow half c == two × pow half (succ c)).
      { calc (pow half c)
          == (pow half c × 1) { (right_identity (action := mult) (pow half c)) }
          == (pow half c × (half × two)) { half_mul_two }
          == ((pow half c × half) × two) { (associative (op := mult) (action := mult) (pow half c) half two) }
          == (two × (pow half c × half)) { (commutative (op := mult) (pow half c × half) two) }
          == (two × pow half (succ c)) { (pow_succ half c) }. }
      calc (psum (fun k ⇒ pow half k) (succ c))
        == (psum (fun k ⇒ pow half k) c + pow half c) { (psum_succ (fun k : N ⇒ pow half k) c) }
        == (two × (1 + - pow half c) + pow half c) { IH }
        == (two × (1 + - pow half c) + two × pow half (succ c)) { Hstep }
        == (two × ((1 + - pow half c) + pow half (succ c))) { (left_distributive (op := mult) two (1 + - pow half c) (pow half (succ c))) }
        == (two × (1 + - pow half (succ c))) { _ }.
      apply op_proper; [ reflexivity | ].
      calc ((1 + - pow half c) + pow half (succ c))
        == ((1 + - (two × pow half (succ c))) + pow half (succ c)) { Hstep }
        == ((1 + - (pow half (succ c) + pow half (succ c))) + pow half (succ c)) { (two_mul (pow half (succ c))) }
        == ((1 + (- pow half (succ c) + - pow half (succ c))) + pow half (succ c)) { (neg_add (pow half (succ c)) (pow half (succ c))) }
        == (1 + ((- pow half (succ c) + - pow half (succ c)) + pow half (succ c))) { associative }
        == (1 + (- pow half (succ c) + (- pow half (succ c) + pow half (succ c)))) { associative }
        == (1 + (- pow half (succ c) + 0)) { (left_inverse (op := add) (pow half (succ c))) }
        == (1 + - pow half (succ c)) { (right_identity (action := add) (- pow half (succ c))) }.
  Qed.

  Theorem psum_pow_half_le_two : ∀ c : N, psum (fun k ⇒ pow half k) c ≤ two.
  Proof.
    intros c.
    setoid_rewrite (psum_pow_half c).
    apply (transitivity (y := two × 1)).
    - apply (mul_le_mono_l _ _ two (le_of_lt _ _ two_pos)).
      apply (transitivity (y := 1 + - (0 : F))).
      + apply add_le_mono_l. apply neg_le_mono. exact (pow_nonneg half half_nonneg c).
      + apply le_of_eq. apply (transitivity (y := (1 : F) + 0)).
        × apply op_proper; [ reflexivity | exact neg_zero ].
        × exact (right_identity (action := add) (1 : F)).
    - exact (le_of_eq _ _ (right_identity (action := mult) two)).
  Qed.

A partial sum of a scaled sequence is the scaled partial sum.
  Theorem psum_mul_l : ∀ (a : F) (g : N → F), Proper (equ ==> equ) g →
    ∀ c : N, psum (fun k ⇒ a × g k) c == a × psum g c.
  Proof.
    intros a g Hg.
    assert (Hag : Proper (equ ==> equ) (fun k ⇒ a × g k))
      by (intros k k' Hk; apply op_proper; [ reflexivity | apply Hg; exact Hk ]).
    apply (nat_induction (fun c ⇒ psum (fun k ⇒ a × g k) c == a × psum g c)).
    - intros c c' Hc.
      setoid_rewrite (psum_proper _ Hag c c' Hc). setoid_rewrite (psum_proper g Hg c c' Hc). reflexivity.
    - calc (psum (fun k ⇒ a × g k) (0 : N))
        == (0 : F) { (psum_zero (fun k : N ⇒ a × g k)) }
        == (a × 0) { (mul_zero a) }
        == (a × psum g (0 : N)) { (psum_zero g) }.
    - intros c IH.
      calc (psum (fun k ⇒ a × g k) (succ c))
        == (psum (fun k ⇒ a × g k) c + a × g c) { (psum_succ (fun k : N ⇒ a × g k) c) }
        == (a × psum g c + a × g c) { IH }
        == (a × (psum g c + g c)) { (left_distributive (op := mult) a (psum g c) (g c)) }
        == (a × psum g (succ c)) { (psum_succ g c) }.
  Qed.

  Theorem psum_nonneg : ∀ g : N → F, Proper (equ ==> equ) g → (∀ k : N, 0 ≤ g k) →
    ∀ c : N, 0 ≤ psum g c.
  Proof.
    intros g Hg Hpos.
    apply (nat_induction (fun c ⇒ 0 ≤ psum g c)).
    - intros c c' Hc. setoid_rewrite (psum_proper g Hg c c' Hc). reflexivity.
    - setoid_rewrite (psum_zero g). reflexivity.
    - intros c IH. setoid_rewrite (psum_succ g c).
      apply (transitivity (le_of_eq _ _ (symmetry (right_identity (action := add) (0 : F))))).
      exact (add_le_mono _ _ _ _ IH (Hpos c)).
  Qed.

The tail of the halves from index i is at most twice the i-th half.
  Theorem psum_pow_half_tail : ∀ i c : N,
    psum (fun k ⇒ pow half (i + k)) c ≤ two × pow half i.
  Proof.
    intros i c.
    assert (Hshift : psum (fun k ⇒ pow half (i + k)) c == pow half i × psum (fun k ⇒ pow half k) c).
    { apply (transitivity (y := psum (fun k ⇒ pow half i × pow half k) c)).
      - apply psum_ext.
        + intros k k' Hk. exact (pow_proper half (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk)).
        + intros k k' Hk. apply op_proper; [ reflexivity | exact (pow_proper half k k' Hk) ].
        + intros k. exact (pow_add half i k).
      - exact (psum_mul_l (pow half i) (fun k ⇒ pow half k) (pow_proper half) c). }
    setoid_rewrite Hshift.
    apply (transitivity (y := pow half i × two)).
    - exact (mul_le_mono_l _ _ (pow half i) (pow_nonneg half half_nonneg i) (psum_pow_half_le_two c)).
    - exact (le_of_eq _ _ (commutative (op := mult) (pow half i) two)).
  Qed.

  Theorem pow_le_one : ∀ r : F, 0 ≤ r → r ≤ 1 → ∀ n : N, pow r n ≤ 1.
  Proof.
    intros r Hr0 Hr1.
    apply (nat_induction (fun n ⇒ pow r n ≤ 1)).
    - intros n n' Hn. setoid_rewrite (pow_proper r n n' Hn). reflexivity.
    - setoid_rewrite (pow_zero r). reflexivity.
    - intros n IH. setoid_rewrite (pow_succ r n).
      apply (transitivity (mul_le_mono_r _ _ r Hr0 IH)).
      apply (transitivity (le_of_eq (1 × r) r (left_identity (action := mult) r))).
      exact Hr1.
  Qed.

  Lemma half_le_one : half ≤ 1.
  Proof.
    apply (transitivity (y := half × two)).
    - apply (transitivity (le_of_eq _ _ (symmetry (right_identity (action := mult) half)))).
      apply (mul_le_mono_l _ _ half half_nonneg).
      apply (transitivity (le_of_eq _ _ (symmetry (right_identity (action := add) (1 : F))))).
      exact (add_le_mono_l 0 1 1 zero_le_one).
    - exact (le_of_eq _ _ half_mul_two).
  Qed.

Powers of a half decrease.
  Theorem pow_half_antitone : ∀ n i : N, n ≤ i → pow half i ≤ pow half n.
  Proof.
    intros n i Hni.
    destruct (proj1 (le_add n i) Hni) as [c Hc].
    setoid_rewrite <- (pow_proper half (n + c) i Hc). setoid_rewrite (pow_add half n c).
    apply (transitivity (mul_le_mono_l _ _ (pow half n) (pow_nonneg half half_nonneg n)
                           (pow_le_one half half_nonneg half_le_one c))).
    exact (le_of_eq _ _ (right_identity (action := mult) (pow half n))).
  Qed.

The least index past which twice the power of a half is below eps, found by stepping up from 0 as long as it is not, for as many steps as the Archimedean bound allows. The bound is what makes the search terminate; the steps are what keep the index small.
  Section IndexSearch.
    Context {LD : @LtDec F eq0 le0}.

    Let up (eps : F) (_ : N) (acc : N) : N :=
      match lt_dec (two × pow half acc) eps with left _ ⇒ acc | right _ ⇒ succ acc end.

    Lemma up_proper : ∀ eps : F, Proper (equ ==> equ ==> equ) (up eps).
    Proof.
      intros eps k k' _ acc acc' Hacc. unfold up.
      assert (He : two × pow half acc == two × pow half acc')
        by (apply op_proper; [ reflexivity | exact (pow_proper half acc acc' Hacc) ]).
      destruct (lt_dec (two × pow half acc) eps) as [Hl | Hl];
      destruct (lt_dec (two × pow half acc') eps) as [Hl' | Hl'].
      - exact Hacc.
      - destruct (Hl' (proj1 (lt_proper _ _ He eps eps (reflexivity eps)) Hl)).
      - destruct (Hl (proj2 (lt_proper _ _ He eps eps (reflexivity eps)) Hl')).
      - apply succ_proper. exact Hacc.
    Qed.

    Lemma up_step : ∀ (eps : F) (k acc : N),
      (two × pow half acc < eps ∧ up eps k acc == acc)
      ∨ (¬ two × pow half acc < eps ∧ up eps k acc == succ acc).
    Proof.
      intros eps k acc. unfold up.
      destruct (lt_dec (two × pow half acc) eps) as [Hl | Hn]; [ left | right ]; split; try assumption; reflexivity.
    Qed.

After c steps the accumulator is either good or equal to c.
    Lemma search_invariant : ∀ (eps : F) (c : N),
      two × pow half (natrec N 0 (up eps) c) < eps ∨ natrec N 0 (up eps) c == c.
    Proof.
      intros eps.
      apply (nat_induction (fun c ⇒ two × pow half (natrec N 0 (up eps) c) < eps ∨ natrec N 0 (up eps) c == c)).
      - intros c c' Hc.
        assert (He : natrec N 0 (up eps) c == natrec N 0 (up eps) c')
          by exact (natrec_proper N _ _ 0 _ (up_proper eps) c c' Hc).
        split; intros [Hl | Heq].
        + left. exact (proj1 (lt_proper _ _ (op_proper two two (reflexivity two) _ _ (pow_proper half _ _ He)) eps eps (reflexivity eps)) Hl).
        + right. exact (transitivity (symmetry He) (transitivity Heq Hc)).
        + left. exact (proj2 (lt_proper _ _ (op_proper two two (reflexivity two) _ _ (pow_proper half _ _ He)) eps eps (reflexivity eps)) Hl).
        + right. exact (transitivity He (transitivity Heq (symmetry Hc))).
      - right. exact (natrec_zero N _ _ 0 _).
      - intros c IH.
        pose proof (natrec_succ N _ _ 0 (up eps) c) as Hs.
        destruct (up_step eps c (natrec N 0 (up eps) c)) as [[Hl Hu] | [Hn Hu]].
        + left.
          apply (proj2 (lt_proper _ _ (op_proper two two (reflexivity two) _ _ (pow_proper half _ _ (transitivity Hs Hu))) eps eps (reflexivity eps))).
          exact Hl.
        + destruct IH as [Hl | Heq]; [ destruct (Hn Hl) | ].
          right. apply (transitivity (transitivity Hs Hu)). apply succ_proper. exact Heq.
    Qed.

The search: an index where twice the power of a half is below eps.
    Theorem half_index : ∀ eps : F, 0 < eps → {n : N & two × pow half n < eps}.
    Proof.
      intros eps Heps.
      destruct (archimedean_inv (N := N) (of_nat := of_nat) (eps × half) (half_pos eps Heps)) as [b Hb].
      ∃ (natrec N 0 (up eps) b).
      destruct (search_invariant eps b) as [Hl | Heq]; [ exact Hl | ].
      apply (proj2 (lt_proper _ _ (op_proper two two (reflexivity two) _ _ (pow_proper half _ _ Heq)) eps eps (reflexivity eps))).
      apply (lt_of_le_of_lt _ (two × inv (of_nat (succ b)))).
      - exact (mul_le_mono_l _ _ two (le_of_lt _ _ two_pos) (pow_half_le_inv_succ b)).
      - apply (proj2 (lt_proper _ _ (reflexivity (two × inv (of_nat (succ b)))) _ _ (symmetry (mul_mul_inv_cancel_left eps two two_ne_zero)))).
        exact (mul_lt_mono_l _ _ two two_pos Hb).
    Defined.

From an index on, every tail of the halves is below any given positive element.
    Theorem half_tail_small : ∀ eps : F, 0 < eps →
      {n : N & ∀ i c : N, n ≤ i → psum (fun k ⇒ pow half (i + k)) c ≤ eps}.
    Proof.
      intros eps Heps.
      destruct (half_index eps Heps) as [n Hn]. ∃ n.
      intros i c Hi.
      apply (transitivity (psum_pow_half_tail i c)).
      apply le_of_lt.
      apply (lt_of_le_of_lt _ (two × pow half n)); [ | exact Hn ].
      exact (mul_le_mono_l _ _ two (le_of_lt _ _ two_pos) (pow_half_antitone n i Hi)).
    Defined.
  End IndexSearch.
