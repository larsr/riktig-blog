Library Riktig.Interface.Reals

From Coq Require Import Morphisms.
From Riktig Require Import Ops Interface.Field.

Module Reals.
  Class Reals (R : Type) `{Field.Field R} := {
    
    le : Rel R;
    le_proper : ∀ x y : R, x == y →
                 ∀ u v : R, u == v → (le x u ↔ le y v);
    le_refl : ∀ x : R, le x x;
    le_trans : ∀ x y z : R, le x y → le y z → le x z;
    le_antisym : ∀ x y : R, le x y → le y x → x == y;
    le_total : ∀ x y : R, le x y ∨ le y x;

    
    le_add_compat : ∀ x y z : R, le x y → le (x + z) (y + z);
    le_mul_compat : ∀ x y : R, le 0 x → le 0 y → le 0 (x × y);

    
    complete : ∀ P : R → Prop,
      (∃ x : R, P x) →
      (∃ b : R, ∀ x : R, P x → le x b) →
      ∃ s : R,
        (∀ x : R, P x → le x s) ∧
        (∀ b : R, (∀ x : R, P x → le x b) → le s b)
  }.
End Reals.
Notation Reals := Reals.Reals.

#[global] Instance Reals_le_Proper (R : Type) `{Field.Field R}
  (H : Reals.Reals R) : Proper (equ ==> equ ==> iff) Reals.le
  := Reals.le_proper.

#[global] Hint Mode Reals.Reals ! - - - - - - - - : typeclass_instances.
