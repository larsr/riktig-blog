Section SwapSignFlip.
  Context {F : Type} `{FF : Field F}.

  Theorem swap_sign_flip_adjacent_gen : ∀ (n0 : nat) (i j : Fin.t (S n0)),
    fin_to_nat j = S (fin_to_nat i) →
    ∀ g : Fin.t (S n0) → Fin.t (S n0),
    (∀ x y : Fin.t (S n0), g x = g y → x = y) →
    (perm_sign (perm_of (fun x ⇒ swap_fin i j (g x))) : F)
      == - perm_sign (perm_of g).
  Proof.
    intros n0 i j Hgap g Hinj.
    pose proof (fin_to_nat_bound (S n0) j) as Hbound.
    assert (Hlt : (fin_to_nat i < n0)%nat) by lia.
    destruct (fin_weaken_surj n0 i Hlt) as [k Hik].
    assert (Hjk : j = Fin.FS k).
    { apply (fin_to_nat_inj (S n0)).
      rewrite (fin_to_nat_FS n0 k), <- (fin_to_nat_fweaken n0 k), <- Hik.
      exact Hgap. }
    rewrite Hik, Hjk.
    exact (perm_sign_swap_adjacent n0 k g Hinj).
  Qed.

  Theorem swap_sign_flip_gap : ∀ (gap N : nat) (i j : Fin.t N),
    fin_to_nat j = (fin_to_nat i + S gap)%nat →
    ∀ g : Fin.t N → Fin.t N,
    (∀ x y : Fin.t N, g x = g y → x = y) →
    (perm_sign (perm_of (fun x ⇒ swap_fin i j (g x))) : F)
      == - perm_sign (perm_of g).
  Proof.
    induction gap as [| gap IH]; intros N i j Hgap g Hinj.
    - assert (Hgap' : fin_to_nat j = S (fin_to_nat i)) by lia.
      destruct N as [| n0].
      + exact (Fin.case0 (fun i0 : Fin.t 0 ⇒
          (perm_sign (perm_of (fun x ⇒ swap_fin i0 j (g x))) : F)
            == - perm_sign (perm_of g)) i).
      + exact (swap_sign_flip_adjacent_gen n0 i j Hgap' g Hinj).
    - destruct N as [| n0].
      + exact (Fin.case0 (fun i0 : Fin.t 0 ⇒
          (perm_sign (perm_of (fun x ⇒ swap_fin i0 j (g x))) : F)
            == - perm_sign (perm_of g)) i).
      + pose proof (fin_to_nat_bound (S n0) j) as Hbound.
        assert (Hlt : (fin_to_nat i < n0)%nat) by lia.
        destruct (fin_weaken_surj n0 i Hlt) as [i' Hi'].
        set (k := Fin.FS i').
        assert (Hik_adj : fin_to_nat k = S (fin_to_nat i)).
        { unfold k. rewrite (fin_to_nat_FS n0 i'), Hi', (fin_to_nat_fweaken n0 i').
          reflexivity. }
        assert (Hkj : fin_to_nat j = (fin_to_nat k + S gap)%nat) by lia.
        assert (Hik : i ≠ k).
        { intros Heq. rewrite Heq in Hik_adj. lia. }
        assert (Hjk : j ≠ k).
        { intros Heq. rewrite Heq, Hik_adj in Hgap. lia. }
        assert (Hij : i ≠ j).
        { intros Heq. rewrite Heq in Hgap. lia. }
        assert (HinjA : ∀ x y : Fin.t (S n0),
            swap_fin i k (g x) = swap_fin i k (g y) → x = y).
        { intros x y Heq. exact (Hinj x y (swap_fin_inj (S n0) i k (g x) (g y) Heq)). }
        assert (HinjBA : ∀ x y : Fin.t (S n0),
            swap_fin k j (swap_fin i k (g x)) = swap_fin k j (swap_fin i k (g y)) → x = y).
        { intros x y Heq.
          exact (HinjA x y (swap_fin_inj (S n0) k j (swap_fin i k (g x)) (swap_fin i k (g y)) Heq)). }
        calc (perm_sign (perm_of (fun x ⇒ swap_fin i j (g x))) : F)
          == (perm_sign (perm_of (fun x ⇒ swap_fin i k (swap_fin k j (swap_fin i k (g x))))) : F)
               { (perm_of_ext (S n0)
                   (fun x ⇒ swap_fin i j (g x))
                   (fun x ⇒ swap_fin i k (swap_fin k j (swap_fin i k (g x))))
                   (fun x ⇒ swap_fin_conj (S n0) i j k Hij Hik Hjk (g x))) }
          == (- perm_sign (perm_of (fun x ⇒ swap_fin k j (swap_fin i k (g x)))) : F)
               { (swap_sign_flip_adjacent_gen n0 i k Hik_adj
                   (fun x ⇒ swap_fin k j (swap_fin i k (g x))) HinjBA) }
          == (- - perm_sign (perm_of (fun x ⇒ swap_fin i k (g x))) : F)
               { (IH (S n0) k j Hkj (fun x ⇒ swap_fin i k (g x)) HinjA) }
          == (- - - perm_sign (perm_of g) : F)
               { (swap_sign_flip_adjacent_gen n0 i k Hik_adj g Hinj) }
          == (- perm_sign (perm_of g) : F)
               { neg_involutive }.
  Qed.

  Theorem perm_sign_postcompose_swap : ∀ (N : nat) (i j : Fin.t N),
    i ≠ j → ∀ sigma : Perm N,
    (perm_sign (perm_postcompose sigma (swap_fin i j)) : F) == - perm_sign sigma.
  Proof.
    intros N i j Hij sigma.
    destruct (Nat.lt_trichotomy (fin_to_nat i) (fin_to_nat j)) as [Hlt | [Heq | Hgt]].
    - assert (Hgap : fin_to_nat j = (fin_to_nat i + S (fin_to_nat j - fin_to_nat i - 1))%nat)
        by lia.
      calc (perm_sign (perm_postcompose sigma (swap_fin i j)) : F)
        == (- perm_sign (perm_of (perm_apply sigma)) : F)
             { (swap_sign_flip_gap (fin_to_nat j - fin_to_nat i - 1) N i j Hgap
                 (perm_apply sigma) (perm_apply_inj N sigma)) }
        == (- perm_sign sigma : F)
             { (perm_of_perm_apply N sigma) }.
    - destruct (Hij (fin_to_nat_inj N i j Heq)).
    - assert (Hgap : fin_to_nat i = (fin_to_nat j + S (fin_to_nat i - fin_to_nat j - 1))%nat)
        by lia.
      calc (perm_sign (perm_postcompose sigma (swap_fin i j)) : F)
        == (perm_sign (perm_postcompose sigma (swap_fin j i)) : F)
             { (perm_postcompose_ext N sigma (swap_fin i j) (swap_fin j i)
                 (swap_fin_symmetric N i j)) }
        == (- perm_sign (perm_of (perm_apply sigma)) : F)
             { (swap_sign_flip_gap (fin_to_nat i - fin_to_nat j - 1) N j i Hgap
                 (perm_apply sigma) (perm_apply_inj N sigma)) }
        == (- perm_sign sigma : F)
             { (perm_of_perm_apply N sigma) }.
  Qed.

End SwapSignFlip.

Section DetSwapGeneral.
  Context {F : Type} `{FF : Field F}.

  Theorem detM_leibniz_swap_general : ∀ (N : nat) (i j : Fin.t N),
    i ≠ j → ∀ A : Matrix N N F,
    detM_leibniz (row_swap A i j) == - detM_leibniz A.
  Proof.
    intros N i j Hij A.
    calc (detM_leibniz (row_swap A i j))
      == (psum (fun sigma : Perm N ⇒ perm_sign sigma ×
            prodF (fun c ⇒ A (swap_fin i j (perm_apply sigma c)) c)))
      == (psum (fun sigma : Perm N ⇒ - (perm_sign sigma × perm_prod A sigma)))
      == (- detM_leibniz A).
    - unfold detM_leibniz at 1.
      apply psum_ext. intros sigma.
      apply op_proper.
      + reflexivity.
      + exact (perm_prod_row_swap N A i j sigma).
    - rewrite (psum_reindex N
        (fun sigma : Perm N ⇒ perm_sign sigma ×
          prodF (fun c ⇒ A (swap_fin i j (perm_apply sigma c)) c))
        (swap_fin i j) (swap_fin i j)
        (swap_fin_invol N i j) (swap_fin_invol N i j)).
      apply psum_ext. intros sigma.
      transitivity ((- perm_sign sigma) ×
        prodF (fun c ⇒ A (swap_fin i j
          (perm_apply (perm_postcompose sigma (swap_fin i j)) c)) c)).
      + apply op_proper.
        × exact (perm_sign_postcompose_swap N i j Hij sigma).
        × reflexivity.
      + rewrite (mul_neg_l (perm_sign sigma)
          (prodF (fun c ⇒ A (swap_fin i j
            (perm_apply (perm_postcompose sigma (swap_fin i j)) c)) c))).
        apply neg_proper.
        apply op_proper.
        × reflexivity.
        × unfold perm_prod.
          apply prodF_ext. intros c.
          rewrite (perm_apply_postcompose N sigma (swap_fin i j)
                     (swap_fin_inj N i j) c),
             (swap_fin_invol N i j (perm_apply sigma c)).
          reflexivity.
    - exact (psum_neg N (fun sigma ⇒ perm_sign sigma × perm_prod A sigma)).
  Qed.

  Theorem detM_swap_general : ∀ (N : nat) (i j : Fin.t N),
    i ≠ j → ∀ A : Matrix N N F,
    detM (row_swap A i j) == - detM A.
  Proof.
    intros N i j Hij A.
    calc (detM (row_swap A i j))
      == (detM_leibniz (row_swap A i j)) { (detM_leibniz_eq_detM N (row_swap A i j)) }
      == (- detM_leibniz A) { (detM_leibniz_swap_general N i j Hij A) }
      == (- detM A) { (detM_leibniz_eq_detM N A) }.
  Qed.

End DetSwapGeneral.

Section DetElemSwap.
  Context {F : Type} `{FF : Field F}.

  Theorem det_elem_swap : ∀ (n : nat) (i j : Fin.t n),
    i ≠ j → ∀ A : Matrix n n F,
    detM (mat_mul (elem_swap i j) A) == - detM A.
  Proof.
    intros n i j Hij A.
    calc (detM (mat_mul (elem_swap i j) A))
      == (detM (row_swap A i j))
      == (- detM A).
    - apply detM_proper. intros r c.
      calc (mat_mul (elem_swap i j) A r c)
        == (A (swap_fin i j r) c) { (elem_swap_mul n n i j A r c) }
        == (row_swap A i j r c) { (row_swap_eq_swap_fin n A i j r c) }.
    - exact (detM_swap_general n i j Hij A).
  Qed.

End DetElemSwap.
