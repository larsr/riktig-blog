The product of two inner product spaces: the inner product is the sum of the inner products of the components.
#[local] Existing Instances Prod_EqRel Prod_Zero Prod_HasAdd Prod_HasNeg Prod_HasScale Prod_Equivalence Prod_Magma Prod_Semigroup Prod_Monoid Prod_Group Prod_AbGroup Prod_ScaleLaws Prod_VectorSpace.

Section ProdInner.
  Context {F V W : Type} `{InnerProductSpace F V}
    {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}
    {VSW : VectorSpace F W zW aW nW sW} {iW : HasInner F W} {IPW : InnerProductSpace F W}.

  #[local] Instance Prod_HasInner : HasInner F (V × W) :=
    fun x y ⇒ ⟪ fst x , fst y ⟫ + ⟪ snd x , snd y ⟫.

  #[local] Instance Prod_InnerProductSpace : InnerProductSpace F (V × W).
  Proof.
    split.
    - intros x x' [Hx1 Hx2] y y' [Hy1 Hy2]. unfold inner, Prod_HasInner.
      apply op_proper; apply inner_proper; assumption.
    - intros w u v. unfold inner, Prod_HasInner. simpl.
      calc (⟪ fst u + fst v , fst w ⟫ + ⟪ snd u + snd v , snd w ⟫)
        == ((⟪ fst u , fst w ⟫ + ⟪ fst v , fst w ⟫) + ⟪ snd u + snd v , snd w ⟫)
             { (additive (f := fun x ⇒ ⟪ x , fst w ⟫) (fst u) (fst v)) }
        == ((⟪ fst u , fst w ⟫ + ⟪ fst v , fst w ⟫) + (⟪ snd u , snd w ⟫ + ⟪ snd v , snd w ⟫))
             { (additive (f := fun x ⇒ ⟪ x , snd w ⟫) (snd u) (snd v)) }
        == ((⟪ fst u , fst w ⟫ + ⟪ snd u , snd w ⟫) + (⟪ fst v , fst w ⟫ + ⟪ snd v , snd w ⟫))
             { add_add_add_comm }.
    - intros w a u. unfold inner, Prod_HasInner. simpl.
      calc (⟪ a *: fst u , fst w ⟫ + ⟪ a *: snd u , snd w ⟫)
        == (a × ⟪ fst u , fst w ⟫ + ⟪ a *: snd u , snd w ⟫)
             { (homogeneous (f := fun x ⇒ ⟪ x , fst w ⟫) a (fst u)) }
        == (a × ⟪ fst u , fst w ⟫ + a × ⟪ snd u , snd w ⟫)
             { (homogeneous (f := fun x ⇒ ⟪ x , snd w ⟫) a (snd u)) }
        == (a × (⟪ fst u , fst w ⟫ + ⟪ snd u , snd w ⟫))
             { left_distributive }.
    - intros u v. unfold inner, Prod_HasInner.
      calc (⟪ fst u , fst v ⟫ + ⟪ snd u , snd v ⟫)
        == (⟪ fst v , fst u ⟫ + ⟪ snd u , snd v ⟫) { (commutative (op := inner) (fst u) (fst v)) }
        == (⟪ fst v , fst u ⟫ + ⟪ snd v , snd u ⟫) { (commutative (op := inner) (snd u) (snd v)) }.
  Qed.
End ProdInner.
