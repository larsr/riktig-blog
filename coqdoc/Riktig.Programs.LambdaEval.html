The call-by-value evaluator with the delta rule, the loop factored out of the program: one step does one layer of the term, handing every subterm to the function it is given for the recursive call; natrec is the loop, and the fuel is any naturals -- pass a binary one for big budgets. The fuel bounds the recursion depth, since a term need not terminate.
Definition eval_step (G : ctx) (rec : list value → term → option value)
    (env : list value) (t : term) : option value :=
  match t with
  | Var n ⇒ nth_error env n
  | Lam b ⇒ Some (Clo env b)
  | Const x ⇒ match lookup x G with
               | Some t' ⇒ rec [] t'
               | None ⇒ Some (Neu x [])
               end
  | App f a ⇒
    match rec env f with
    | Some vf ⇒
      match rec env a with
      | Some va ⇒
        match vf with
        | Clo env' b ⇒ rec (va :: env') b
        | Neu x args ⇒ Some (Neu x (args ++ [va]))
        end
      | None ⇒ None
      end
    | None ⇒ None
    end
  end.

Definition eval {N : Type} {natrecN : HasNatRec N} (G : ctx) (fuel : N)
    : list value → term → option value :=
  natrec (list value → term → option value) (fun _ _ ⇒ None) (fun _ rec ⇒ eval_step G rec) fuel.

Reading a numeral back: apply the value to the inert constants S and Z and count.
Definition eval_numeral {N : Type} {natrecN : HasNatRec N} (G : ctx) (fuel : N) (t : term) : option nat :=
  match eval G fuel [] (App (App t (Const "S")) (Const "Z")) with
  | Some v ⇒ count_S v
  | None ⇒ None
  end.
