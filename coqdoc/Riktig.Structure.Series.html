From Coq Require Import Morphisms.

From Riktig Require Import Theory.BigOp.

ssum f is the sum of the family f over all of nat.
Class HasSsum (A : Type) := ssum : (nat → A) → A.

summable f is the domain the sum laws hold on.
Class HasSummable (A : Type) := summable : Pred (nat → A).

Hint Mode HasSsum ! : typeclass_instances.
Hint Mode HasSummable ! : typeclass_instances.

Typeclasses Transparent HasSsum HasSummable.

Summation on a ring: the sum of a family supported below n is its n-th finite section, summable families are closed under the operations the laws rearrange, and on them the sum is additive, respects negation, splits into its even and odd halves, and turns a product of sums into the sum of the Cauchy products.
Module Series.
  Class Series (A : Type) `{Ring A} {Ab : HasBigop A}
      {Ss : HasSsum A} {Sm : HasSummable A} := {
    ssum_proper :: Proper (equ ==> equ) Ss;
    summable_proper :: Proper (equ ==> iff) Sm;

    summable_add : ∀ f g : nat → A,
      summable f → summable g → summable (fun n ⇒ f n + g n);
    summable_neg : ∀ f : nat → A,
      summable f → summable (fun n ⇒ - f n);
    summable_even : ∀ f : nat → A,
      summable f → summable (fun k ⇒ f (2 × k)%nat);
    summable_odd : ∀ f : nat → A,
      summable f → summable (fun k ⇒ f (S (2 × k)%nat));
    summable_cauchy : ∀ f g : nat → A,
      summable f → summable g →
      summable (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat));

    sum_supported : ∀ (f : nat → A) (n : nat),
      (∀ m : nat, (n ≤ m)%nat → f m == 0) → ssum f == bigop n f;
    sum_add : ∀ f g : nat → A,
      summable f → summable g →
      ssum (fun n ⇒ f n + g n) == ssum f + ssum g;
    sum_neg : ∀ f : nat → A,
      summable f → ssum (fun n ⇒ - f n) == - ssum f;
    sum_parity : ∀ f : nat → A,
      summable f →
      ssum f == ssum (fun k ⇒ f (2 × k)%nat) + ssum (fun k ⇒ f (S (2 × k)%nat));
    sum_cauchy : ∀ f g : nat → A,
      summable f → summable g →
      ssum f × ssum g == ssum (fun n ⇒ bigop (S n) (fun k ⇒ f k × g (n - k)%nat));
  }.
  Hint Mode Series ! - - - - - - - - - - : typeclass_instances.
End Series.
Export Series.

Summation on an F-algebra against summation on F: a scalar comes out of a sum of vectors, and a sum of scalars comes out of a family of multiples of one vector.
Module ScaleSeries.
  Class ScaleSeries (F A : Type) `{Field F}
      {Fb : HasBigop F} {Fs : HasSsum F} {Fm : HasSummable F}
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
      {Ao : One A} {Amu : HasMult A} {Asc : HasScale F A}
      {AR : Ring A} {ASL : ScaleLaws F A}
      {Ab : HasBigop A} {As : HasSsum A} {Am : HasSummable A} := {
    vectors :: Series A;
    scalars :: Series F;

    summable_scale : ∀ (a : F) (f : nat → A),
      summable f → summable (fun n ⇒ a *: f n);
    summable_scale_r : ∀ (c : nat → F) (v : A),
      summable c → summable (fun k ⇒ c k *: v);

    sum_scale : ∀ (a : F) (f : nat → A),
      summable f → ssum (fun n ⇒ a *: f n) == a *: ssum f;
    sum_scale_r : ∀ (c : nat → F) (v : A),
      summable c → ssum (fun k ⇒ c k *: v) == ssum c *: v;
  }.
  Hint Mode ScaleSeries - ! - - - - - - - - - - - - - - - - - - - - - - - :
    typeclass_instances.
End ScaleSeries.
Export ScaleSeries.
