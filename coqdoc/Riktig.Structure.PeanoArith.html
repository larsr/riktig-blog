From Riktig.Structure Require Export Peano.

Arithmetic on the Peano numbers by its recursion equations: one is the successor of zero, and adding or multiplying by zero and by a successor are given by the two equations each.
Module PeanoArith.
  Class PeanoArith (N : Type) {eq : EqRel N} (zero : Zero N) (succ : HasSucc N)
      (one : One N) (add : HasAdd N) (mult : HasMult N) := {
    peano :: Peano N 0 succ;
    add_proper :: Proper (equ ==> equ ==> equ) add;
    mul_proper :: Proper (equ ==> equ ==> equ) mult;
    one_succ : (1 : N) == succ 0;
    add_zero : ∀ a : N, a + 0 == a;
    add_succ : ∀ a b : N, a + succ b == succ (a + b);
    mul_zero : ∀ a : N, a × 0 == 0;
    mul_succ : ∀ a b : N, a × succ b == a × b + a;
  }.
  Hint Mode PeanoArith ! - - - - - - : typeclass_instances.
End PeanoArith.
Export PeanoArith.
