From Riktig.Structure Require Export Finite Decidable Upd.
From Riktig.Theory Require Export NatOrder.
From Riktig.Lib Require Export Calc.

Section FiniteTheory.
  Context {I N : Type} `{Finite I N}.

  Theorem index_injective : ∀ i j : I, index i == index j → i == j.
  Proof.
    intros i j Hij.
    calc i
      == (elem (index i) : I) { Finite.elem_index }
      == (elem (index j) : I) { Hij }
      == j { Finite.elem_index }.
  Qed.

  Theorem elem_injective : ∀ k l : N,
    k < card → l < card → (elem k : I) == elem l → k == l.
  Proof.
    intros k l Hk Hl Hkl.
    calc k
      == (index (elem k : I)) { (Finite.index_elem k Hk) }
      == (index (elem l : I)) { Hkl }
      == l { (Finite.index_elem l Hl) }.
  Qed.

A number names i exactly when it is i's index.
  Theorem elem_eq_of_index : ∀ (i : I) (n : N), n == index i → (elem n : I) == i.
  Proof.
    intros i n Hn.
    calc (elem n : I)
      == (elem (index i) : I) { Hn }
      == i { Finite.elem_index }.
  Qed.

  Theorem index_eq_of_elem : ∀ (i : I) (n : N), n < card → (elem n : I) == i → n == index i.
  Proof.
    intros i n Hn Hi.
    calc n
      == (index (elem n : I)) { (Finite.index_elem n Hn) }
      == (index i) { Hi }.
  Qed.

Equality on a finite index is decided, through the numbering.
  Theorem eq_or_ne_index : ∀ i j : I, i == j ∨ i =/= j.
  Proof.
    intros i j.
    destruct (eq_or_ne (index i) (index j)) as [Heq | Hne].
    - left. exact (index_injective i j Heq).
    - right. intros Hij. exact (Hne (Finite.index_proper i j Hij)).
  Qed.

End FiniteTheory.

Families over a finite index with decided equality: replacing a slot with equal elements gives equal families, and the replaced family reads through elem as the laws say.
Section FamilyUpdate.
  Context {I N V : Type} `{Finite I N} {EI : EquDec I} {eV : EqRel V} {EqvV : Equivalence (A := V) equ}
    {update : HasUpd I V} {U : Upd I V update}.

  Lemma upd_ext : ∀ (vs : I → V) (j : I) (u u' : V), u == u' → ∀ i : I, upd vs j u i == upd vs j u' i.
  Proof.
    intros vs j u u' Hu i. destruct (EquDec.equ_dec i j) as [Hij | Hij].
    - calc (upd vs j u i) == u { (Upd.upd_at vs j u i Hij) } == u' { Hu } == (upd vs j u' i) { (symmetry (Upd.upd_at vs j u' i Hij)) }.
    - calc (upd vs j u i) == (vs i) { (Upd.upd_off vs j u i Hij) } == (upd vs j u' i) { (symmetry (Upd.upd_off vs j u' i Hij)) }.
  Qed.

  Lemma upd_at_elem : ∀ (vs : I → V) (j : I) (u : V), upd vs j u (elem (index j)) == u.
  Proof. intros vs j u. exact (Upd.upd_at vs j u (elem (index j)) (Finite.elem_index j)). Qed.

  Lemma upd_off_elem : ∀ (vs : I → V) (j : I) (u : V) (k : N),
    k < card → k =/= index j → upd vs j u (elem k) == vs (elem k).
  Proof.
    intros vs j u k Hk Hkj. apply Upd.upd_off. intros Hej. apply Hkj.
    calc k == (index (I := I) (N := N) (elem (I := I) (N := N) k)) { (Finite.index_elem (I := I) (N := N) k Hk) }
      == (index j) { (Finite.index_proper (elem k) j Hej) }.
  Qed.
End FamilyUpdate.
