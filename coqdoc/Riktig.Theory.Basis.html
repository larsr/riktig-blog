From Coq Require Export RelationClasses.
From Riktig Require Export Theory.FieldVS Theory.BigOp.

Section Lincomb.
  Context {F V : Type}
    {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F} {Fo : One F}
    {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {Ieq : EqDec I} {IFin : FinIndex n I}.

  Definition lincomb (b : I → V) (c : I → F) : V :=
    bigopI (fun i ⇒ c i *: b i).

  Lemma lincomb_def : ∀ (b : I → V) (c : I → F),
    lincomb b c == bigopI (fun i ⇒ c i *: b i).
  Proof.
    intros b c. reflexivity.
  Qed.

  Lemma lincomb_proper : ∀ (b : I → V) (c c' : I → F),
    (∀ i : I, c i == c' i) → lincomb b c == lincomb b c'.
  Proof.
    intros b c c' Hc.
    exact (bigopI_ext (fun i ⇒ c i *: b i) (fun i ⇒ c' i *: b i)
             (fun i ⇒ scale_proper (c i) (c' i) (Hc i)
                         (b i) (b i) (reflexivity (b i)))).
  Qed.

  Lemma lincomb_add : ∀ (b : I → V) (c d : I → F),
    lincomb b (c + d) == lincomb b c + lincomb b d.
  Proof.
    intros b c d.
    calc (lincomb b (c + d))
      == (bigopI (fun i ⇒ (c + d) i *: b i))
           { lincomb_def }
      == (bigopI (fun i ⇒ c i *: b i + d i *: b i))
           { (bigopI_ext (fun i ⇒ (c + d) i *: b i)
                (fun i ⇒ c i *: b i + d i *: b i)
                (fun i ⇒ right_distributive (c i) (d i) (b i))) }
      == (lincomb b c + lincomb b d)
           { bigopI_add }.
  Qed.

  Lemma lincomb_scale : ∀ (a : F) (b : I → V) (c : I → F),
    lincomb b (a *: c) == a *: lincomb b c.
  Proof.
    intros a b c.
    calc (lincomb b (a *: c))
      == (bigopI (fun i ⇒ (a *: c) i *: b i))
           { lincomb_def }
      == (bigopI (fun i ⇒ a *: (c i *: b i)))
           { (bigopI_ext (fun i ⇒ (a *: c) i *: b i)
                (fun i ⇒ a *: (c i *: b i))
                (fun i ⇒ associative a (c i) (b i))) }
      == (a *: lincomb b c)
           { bigopI_scale }.
  Qed.

  Lemma lincomb_delta : ∀ (b : I → V) (j : I),
    lincomb b (delta j) == b j.
  Proof.
    intros b j.
    calc (lincomb b (delta j))
      == (delta j j *: b j)
           { (bigopI_support_singleton j (fun i ⇒ delta j i *: b i)
               (fun i Hne ⇒
                 transitivity
                   (scale_proper (delta j i) 0
                     (delta_at_neq j i (fun Heq ⇒ Hne (eq_sym Heq)))
                     (b i) (b i) (reflexivity (b i)))
                   (zero_scale (b i)))) }
      == ((1 : F) *: b j) { (delta_at_eq j j eq_refl) }
      == (b j) { left_identity }.
  Qed.

End Lincomb.
