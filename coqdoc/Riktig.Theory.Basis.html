  Lemma lincomb_proper : ∀ (b : I → V) (c c' : I → F),
    (∀ i : I, c i == c' i) → lincomb b c == lincomb b c'.
  Proof.
    intros b c c' Hc.
    calc (lincomb b c)
      == (bigop_list (fun i ⇒ c i *: b i) enum)
           { lincomb_def }
      == (bigop_list (fun i ⇒ c' i *: b i) enum)
           { (bigopI_ext _ _ (fun i ⇒ scale_proper _ _ (Hc i) _ _ (reflexivity (b i)))) }
      == (lincomb b c')
           { lincomb_def }.
  Qed.

  Lemma lincomb_add : ∀ (b : I → V) (c d : I → F),
    lincomb b (c + d) == lincomb b c + lincomb b d.
  Proof.
    intros b c d.
    calc (lincomb b (c + d))
      == (bigop_list (fun i ⇒ (c + d) i *: b i) enum)
           { lincomb_def }
      == (bigop_list (fun i ⇒ c i *: b i + d i *: b i) enum)
           { (bigopI_ext _ _ (fun i ⇒ right_distributive (c i) (d i) (b i))) }
      == (bigop_list (fun i ⇒ c i *: b i) enum + bigop_list (fun i ⇒ d i *: b i) enum)
           { bigopI_add }
      == (lincomb b c + bigop_list (fun i ⇒ d i *: b i) enum)
           { lincomb_def }
      == (lincomb b c + lincomb b d)
           { lincomb_def }.
  Qed.

  Lemma lincomb_scale : ∀ (a : F) (b : I → V) (c : I → F),
    lincomb b (a *: c) == a *: lincomb b c.
  Proof.
    intros a b c.
    calc (lincomb b (a *: c))
      == (bigop_list (fun i ⇒ (a *: c) i *: b i) enum)
           { lincomb_def }
      == (bigop_list (fun i ⇒ a *: (c i *: b i)) enum)
           { (bigopI_ext _ _ (fun i ⇒ associative a (c i) (b i))) }
      == (a *: bigop_list (fun i ⇒ c i *: b i) enum)
           { bigopI_scale }
      == (a *: lincomb b c)
           { lincomb_def }.
  Qed.

What the three lemmas above amount to: a linear combination is a linear map from the coefficient space to the vectors. Stating it lets a caller cite `additivelincomb b` and `homogeneouslincomb b` — the general projections — rather than three names local to bases.
  #[export] Instance lincomb_linear (b : I → V) :
    Linear F (I → F) V (lincomb b) := {
    proper := lincomb_proper b;
    is_additive := lincomb_add b;
    is_homogeneous := fun a c ⇒ lincomb_scale a b c;
  }.

  Lemma lincomb_delta : ∀ (b : I → V) (j : I),
    Proper (equ ==> equ) b →
    lincomb b (delta j) == b j.
  Proof.
    intros b j Pb.
    calc (lincomb b (delta j))
      == (bigop_list (fun i ⇒ delta j i *: b i) enum) { lincomb_def }
      == (b j) { (bigopI_delta_scale b j Pb) }.
  Qed.
