From Coq Require Export RelationClasses.
From Coq.Vectors Require Fin.
From Riktig Require Export Ops Interface.Field Interface.VectorSpace
  Theory.FieldVS Theory.BigOp Lib.Calc.


Section Lincomb.
  Context {F V : Type} `{FF : Field.Field F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Definition lincomb {n : nat} (b : Fin.t n → V) (c : Fin.t n → F) : V :=
    bigopF (fun i ⇒ c i *: b i).

  Lemma lincomb_def : ∀ (n : nat) (b : Fin.t n → V) (c : Fin.t n → F),
    lincomb b c == bigopF (fun i ⇒ c i *: b i).
  Proof.
    intros n b c. reflexivity.
  Qed.

  Lemma lincomb_proper : ∀ (n : nat) (b : Fin.t n → V)
      (c c' : Fin.t n → F),
    (∀ i : Fin.t n, c i == c' i) → lincomb b c == lincomb b c'.
  Proof.
    intros n b c c' Hc.
    exact (bigopF_ext n (fun i ⇒ c i *: b i) (fun i ⇒ c' i *: b i)
             (fun i ⇒ scale_proper (c i) (c' i) (Hc i)
                         (b i) (b i) (reflexivity (b i)))).
  Qed.

  Lemma lincomb_add : ∀ (n : nat) (b : Fin.t n → V)
      (c d : Fin.t n → F),
    lincomb b (c + d) == lincomb b c + lincomb b d.
  Proof.
    intros n b c d.
    calc (lincomb b (c + d))
      == (bigopF (fun i ⇒ (c + d) i *: b i))
           { (lincomb_def n b (c + d)) }
      == (bigopF (fun i ⇒ c i *: b i + d i *: b i))
           { (bigopF_ext n (fun i ⇒ (c + d) i *: b i)
                (fun i ⇒ c i *: b i + d i *: b i)
                (fun i ⇒ scale_distributes_over_scalar_add (c i) (d i) (b i))) }
      == (lincomb b c + lincomb b d)
           { (bigopF_add n (fun i ⇒ c i *: b i) (fun i ⇒ d i *: b i)) }.
  Qed.

  Lemma lincomb_scale : ∀ (n : nat) (a : F) (b : Fin.t n → V)
      (c : Fin.t n → F),
    lincomb b (a *: c) == a *: lincomb b c.
  Proof.
    intros n a b c.
    calc (lincomb b (a *: c))
      == (bigopF (fun i ⇒ (a *: c) i *: b i))
           { (lincomb_def n b (a *: c)) }
      == (bigopF (fun i ⇒ a *: (c i *: b i)))
           { (bigopF_ext n (fun i ⇒ (a *: c) i *: b i)
                (fun i ⇒ a *: (c i *: b i))
                (fun i ⇒ scale_associative a (c i) (b i))) }
      == (a *: lincomb b c)
           { (bigopF_scale n a (fun i ⇒ c i *: b i)) }.
  Qed.

End Lincomb.
