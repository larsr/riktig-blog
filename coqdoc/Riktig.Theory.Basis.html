lincomb b c is the linear combination of the family b with coefficients c.
Class HasLincomb (I F V : Type) := lincomb : (I → V) → (I → F) → V.
Hint Mode HasLincomb ! - ! : typeclass_instances.
Typeclasses Transparent HasLincomb.

Module Lincomb.
  Class Lincomb (n : nat) (I F V : Type)
      {eI : EqRel I} {Ieq : EquDec I} {IFin : FinIndex n I}
      {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {VM : Monoid V}
      {Vs : HasScale F V} {Vbi : HasBigopI I V} {Vl : HasLincomb I F V} :=
    lincomb_def : ∀ (b : I → V) (c : I → F),
      lincomb b c == bigopI (fun i ⇒ c i *: b i).
  Hint Mode Lincomb - ! ! ! - - - - - - - - - - : typeclass_instances.
End Lincomb.
Export Lincomb.

    {VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context {Vbl : HasBigopList V} {VblL : BigopList V}.
  Context {Vbi : HasBigopI I V} {VbiL : BigopI n I V}.
  Context {Vl : HasLincomb I F V} {VlL : Lincomb n I F V}.

  Lemma lincomb_proper : ∀ (b : I → V) (c c' : I → F),
    (∀ i : I, c i == c' i) → lincomb b c == lincomb b c'.
  Proof.
    intros b c c' Hc.
    calc (lincomb b c)
      == (bigopI (fun i ⇒ c i *: b i))
           { lincomb_def }
      == (bigopI (fun i ⇒ c' i *: b i))
           { (bigopI_ext (fun i ⇒ c i *: b i) (fun i ⇒ c' i *: b i)
               (fun i ⇒ scale_proper (c i) (c' i) (Hc i)
                           (b i) (b i) (reflexivity (b i)))) }
      == (lincomb b c')
           { lincomb_def }.
  Qed.

  Lemma lincomb_add : ∀ (b : I → V) (c d : I → F),
    lincomb b (c + d) == lincomb b c + lincomb b d.
  Proof.
    intros b c d.
    calc (lincomb b (c + d))
      == (bigopI (fun i ⇒ (c + d) i *: b i))
           { lincomb_def }
      == (bigopI (fun i ⇒ c i *: b i + d i *: b i))
           { (bigopI_ext (fun i ⇒ (c + d) i *: b i)
                (fun i ⇒ c i *: b i + d i *: b i)
                (fun i ⇒ right_distributive (c i) (d i) (b i))) }
      == (bigopI (fun i ⇒ c i *: b i) + bigopI (fun i ⇒ d i *: b i))
           { bigopI_add }
      == (lincomb b c + bigopI (fun i ⇒ d i *: b i))
           { lincomb_def }
      == (lincomb b c + lincomb b d)
           { lincomb_def }.
  Qed.

  Lemma lincomb_scale : ∀ (a : F) (b : I → V) (c : I → F),
    lincomb b (a *: c) == a *: lincomb b c.
  Proof.
    intros a b c.
    calc (lincomb b (a *: c))
      == (bigopI (fun i ⇒ (a *: c) i *: b i))
           { lincomb_def }
      == (bigopI (fun i ⇒ a *: (c i *: b i)))
           { (bigopI_ext (fun i ⇒ (a *: c) i *: b i)
                (fun i ⇒ a *: (c i *: b i))
                (fun i ⇒ associative a (c i) (b i))) }
      == (a *: bigopI (fun i ⇒ c i *: b i))
           { bigopI_scale }
      == (a *: lincomb b c)
           { lincomb_def }.
  Qed.

  Lemma delta_scale_proper : ∀ (b : I → V) (j : I),
    Proper (equ ==> equ) b →
    Proper (equ ==> equ) (fun i ⇒ delta j i *: b i).
  Proof.
    intros b j Pb i i' Hii'.
    exact (scale_proper (delta j i) (delta j i')
             (delta_proper j j (reflexivity j) i i' Hii')
             (b i) (b i') (Pb i i' Hii')).
  Qed.

  Lemma lincomb_delta : ∀ (b : I → V) (j : I),
    Proper (equ ==> equ) b →
    lincomb b (delta j) == b j.
  Proof.
    intros b j Pb.
    calc (lincomb b (delta j))
      == (bigopI (fun i ⇒ delta j i *: b i))
           { lincomb_def }
      == (delta j j *: b j)
           { (bigopI_support_singleton j (fun i ⇒ delta j i *: b i)
               (delta_scale_proper b j Pb)
               (fun i Hne ⇒
                 transitivity
                   (scale_proper (delta j i) 0
                     (delta_at_neq j i (fun Heq ⇒ Hne (symmetry Heq)))
                     (b i) (b i) (reflexivity (b i)))
                   (zero_scale (b i)))) }
      == ((1 : F) *: b j) { (delta_at_eq j j (reflexivity j)) }
      == (b j) { left_identity }.
  Qed.

End LincombTheory.
