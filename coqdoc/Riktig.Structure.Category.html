A category is a collection of objects with hom-setoids, an identity morphism on each object, and an associative composition for which the identities are neutral.
Module Category.
  Class Category (obj : Type) := {
    hom : obj → obj → Type;
    hom_equ :: ∀ x y, EqRel (hom x y);
    hom_equiv :: ∀ x y, Equivalence (hom_equ x y);
    id : ∀ {x : obj}, hom x x;
    comp : ∀ {x y z : obj}, hom y z → hom x y → hom x z;
    comp_proper :: ∀ x y z, Proper (equ ==> equ ==> equ) (@comp x y z);
    is_left_identity :: ∀ x y, LeftIdentity (hom y y) (hom x y) comp id;
    is_right_identity :: ∀ x y, RightIdentity (hom x x) (hom x y) comp id;
    comp_associative : ∀ (w x y z : obj) (f : hom y z) (g : hom x y) (h : hom w x),
        comp f (comp g h) == comp (comp f g) h
  }.

  Definition comp_left_identity {obj : Type} `{Category obj}
    (x y : obj) (f : hom x y) : comp id f == f := left_identity f.

  Definition comp_right_identity {obj : Type} `{Category obj}
    (x y : obj) (f : hom x y) : comp f id == f := right_identity f.
End Category.
Export Category.
Notation hom := Category.hom.
Notation cat_id := Category.id.
Notation comp := Category.comp.

Notation "x ~> y" := (Category.hom x y) (at level 70, right associativity).
Infix "∘" := Category.comp (at level 40, left associativity).

A functor sends objects to objects and morphisms to morphisms, preserving identities and composition.
Module Functor.
  Class Functor {C D : Type} (cC : Category C) (cD : Category D) := {
    fobj : C → D;
    fmap : ∀ {x y : C}, (x ~> y) → (fobj x ~> fobj y);
    fmap_proper :: ∀ x y : C, Proper (equ ==> equ) (@fmap x y);
    fmap_id : ∀ x : C, fmap (@cat_id C cC x) == cat_id;
    fmap_comp : ∀ (x y z : C) (f : y ~> z) (g : x ~> y),
        fmap (f ∘ g) == fmap f ∘ fmap g
  }.
End Functor.
Notation Functor := Functor.Functor.
Notation fobj := Functor.fobj.
Notation fmap := Functor.fmap.

A natural transformation from F to G gives each object a morphism between its images, commuting with the image of every morphism.
Module NaturalTransformation.
  Class NaturalTransformation {C D : Type} {cC : Category C} {cD : Category D}
      (F G : Functor cC cD) := {
    eta : ∀ x : C, fobj (Functor:=F) x ~> fobj (Functor:=G) x;
    naturality : ∀ (x y : C) (f : x ~> y),
        fmap (Functor:=G) f ∘ eta x == eta y ∘ fmap (Functor:=F) f
  }.
End NaturalTransformation.
Notation NaturalTransformation := NaturalTransformation.NaturalTransformation.
Notation eta := NaturalTransformation.eta.

An initial object has exactly one morphism to each object.

    emit_uniqueness : ∀ (x : O) (f : init ~> x), f == emit x
  }.
End Initial.

A product of a and b is an object with projections to a and b through which every pair of morphisms from a common source factors uniquely.
Module HasProducts.
  Class HasProducts {O : Type} (C : Category O) := {
    prod_obj : O → O → O;
    pi1 : ∀ {a b : O}, prod_obj a b ~> a;
    pi2 : ∀ {a b : O}, prod_obj a b ~> b;
    pair_morphism : ∀ {x a b : O}, (x ~> a) → (x ~> b) → (x ~> prod_obj a b);
    pair_proper :: ∀ x a b : O,
        Proper (equ ==> equ ==> equ) (@pair_morphism x a b);
    pi1_pair : ∀ (x a b : O) (f : x ~> a) (g : x ~> b),
        pi1 ∘ pair_morphism f g == f;
    pi2_pair : ∀ (x a b : O) (f : x ~> a) (g : x ~> b),
        pi2 ∘ pair_morphism f g == g;
    uniqueness : ∀ (x a b : O) (f : x ~> a) (g : x ~> b)
                        (h : x ~> prod_obj a b),
        pi1 ∘ h == f → pi2 ∘ h == g → h == pair_morphism f g
  }.
End HasProducts.
Export HasProducts.
Notation prod_obj := HasProducts.prod_obj.
Notation pi1 := HasProducts.pi1.
Notation pi2 := HasProducts.pi2.
Notation pair_morphism := HasProducts.pair_morphism.

Definition par_morphism {O : Type} {C : Category O} {P : HasProducts C}
  {a b c d : O} (f : a ~> c) (g : b ~> d) : prod_obj a b ~> prod_obj c d :=
  pair_morphism (f ∘ pi1) (g ∘ pi2).

Notation "f ×× g" := (par_morphism f g) (at level 41).

A coproduct of a and b is an object with injections from a and b through which every pair of morphisms to a common target factors uniquely.
Module HasCoproducts.
  Class HasCoproducts {O : Type} (C : Category O) := {
    sum_obj : O → O → O;
    inj1 : ∀ {a b : O}, a ~> sum_obj a b;
    inj2 : ∀ {a b : O}, b ~> sum_obj a b;
    case_morphism : ∀ {x a b : O}, (a ~> x) → (b ~> x) → (sum_obj a b ~> x);
    case_proper :: ∀ x a b : O,
        Proper (equ ==> equ ==> equ) (@case_morphism x a b);
    case_inj1 : ∀ (x a b : O) (f : a ~> x) (g : b ~> x),
        case_morphism f g ∘ inj1 == f;
    case_inj2 : ∀ (x a b : O) (f : a ~> x) (g : b ~> x),
        case_morphism f g ∘ inj2 == g;
    uniqueness : ∀ (x a b : O) (f : a ~> x) (g : b ~> x)
                        (h : sum_obj a b ~> x),
        h ∘ inj1 == f → h ∘ inj2 == g → h == case_morphism f g
  }.
End HasCoproducts.
Notation HasCoproducts := HasCoproducts.HasCoproducts.
Notation sum_obj := HasCoproducts.sum_obj.
Notation inj1 := HasCoproducts.inj1.
Notation inj2 := HasCoproducts.inj2.
Notation case_morphism := HasCoproducts.case_morphism.

An exponential of a and b is an object exp_obj a b with an evaluation morphism through which every morphism prod_obj x a ~> b factors uniquely by currying.
Module HasExponentials.
  Class HasExponentials {O : Type} (C : Category O) {P : HasProducts C} := {
    exp_obj : O → O → O;
    eval_morphism : ∀ {a b : O}, prod_obj (exp_obj a b) a ~> b;
    curry : ∀ {x a b : O}, (prod_obj x a ~> b) → (x ~> exp_obj a b);
    curry_proper :: ∀ x a b : O,
        Proper (equ ==> equ) (@curry x a b);
    curry_beta : ∀ (x a b : O) (h : prod_obj x a ~> b),
        eval_morphism ∘ (curry h ×× cat_id) == h;
    curry_uniqueness : ∀ (x a b : O) (h : prod_obj x a ~> b)
                              (k : x ~> exp_obj a b),
        eval_morphism ∘ (k ×× cat_id) == h → k == curry h
  }.
End HasExponentials.
Export HasExponentials.
Notation exp_obj := HasExponentials.exp_obj.
Notation eval_morphism := HasExponentials.eval_morphism.
Notation curry := HasExponentials.curry.

A cartesian closed category has a terminal object, binary products, and exponentials.
Module CCC.
  Class CCC {O : Type} (C : Category O) := {
    terminal :: Terminal C;
    products :: HasProducts C;
    exponentials :: @HasExponentials O C products
  }.
End CCC.
Export CCC.
