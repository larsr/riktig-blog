A cast of the naturals into R: zero goes to zero and each successor adds one.

Section NatCastUnique.
  Context {R : Type} {eR : EqRel R} {zR : Zero R} {oR : One R}
    {aR : HasAdd R} {MR : Monoid R}.

The two equations reach every natural, so they leave no freedom: any two casts agree everywhere. This is what licenses reading "the" ring map in the header of Impl/TrigSeries.v.
  Theorem cast_unique : ∀ (c d : nat → R),
    NatCast R c → NatCast R d → ∀ n : nat, c n == d n.
  Proof.
    intros c d Nc Nd n.
    induction n as [| n IH].
    - calc (c O)
        == (0 : R) { cast_zero }
        == (d O) { cast_zero }.
    - calc (c (S n))
        == (1 + c n) { cast_succ }
        == (1 + d n) { (op_proper 1 1 (reflexivity (1 : R)) _ _ IH) }
        == (d (S n)) { cast_succ }.
  Qed.

End NatCastUnique.

Section NatCastOrder.
  Context {F : Type} `{OrderedField F}.
  Context (c : nat → F) {NC : NatCast F c}.

Every natural lands on the nonnegative side.
  Lemma cast_nonneg : ∀ n : nat, 0 ≤ c n.
  Proof.
    intros n; induction n as [| n IH].
    - rewrite cast_zero. reflexivity.
    - calc (0 : F)
        == (c n) { IH }
        == (1 + c n) { (le_add_nonneg_l 1 (c n) (lt_le 0 1 lt_0_1)) }
        == (c (S n)) { cast_succ }.
  Qed.

  Lemma cast_pos : ∀ n : nat, 0 < c (S n).
  Proof.
    intros n. rewrite (cast_succ n).
    exact (one_plus_nonneg_pos (c n) (cast_nonneg n)).
  Qed.

The cast is monotone: it takes the order on nat to the order on F.
  Lemma cast_le_mono : ∀ m n : nat, (m ≤ n)%nat → c m ≤ c n.
  Proof.
    intros m n; induction n as [| n IH]; intros Hmn.
    - rewrite (proj1 (Nat.le_0_r m) Hmn). reflexivity.
    - destruct (proj1 (Nat.lt_eq_cases m (S n)) Hmn) as [Hlt | Heq].
      + calc (c m)
          == (c n) { (IH (proj1 (Nat.lt_succ_r m n) Hlt)) }
          == (1 + c n) { (le_add_nonneg_l 1 (c n) (lt_le 0 1 lt_0_1)) }
          == (c (S n)) { cast_succ }.
      + rewrite Heq. reflexivity.
  Qed.

Two. Reading this off a Fixpoint took a reflexivity that rewrite could not see through; off the two equations it is three ordinary steps.
  Lemma cast_two : c 2 == (two : F).
  Proof.
    calc (c 2)
      == (1 + c 1) { cast_succ }
      == (1 + (1 + c O)) { (op_proper 1 1 (reflexivity (1 : F)) _ _ (cast_succ O)) }
      == ((1 : F) + (1 + 0)) { (op_proper 1 1 (reflexivity (1 : F)) _ _ cast_zero) }
      == ((1 : F) + 1) { right_identity }
      == (two : F) { two_eq }.
  Qed.

Past the second natural the reciprocal is at most a half.
  Lemma inv_cast_le_half : ∀ n : nat,
    (2 ≤ n)%nat → inv (c n) ≤ half (1 : F).
  Proof.
    intros n Hn.
    assert (Hpos : 0 < c 2) by (rewrite cast_two; exact two_pos).
    calc (inv (c n))
      == (inv (c 2)) { (inv_le_mono (c 2) (c n) Hpos (cast_le_mono 2 n Hn)) }
      == (inv (two : F)) { cast_two }
      == (half (1 : F)) { inv_two_is_half_one }.
  Qed.

End NatCastOrder.
