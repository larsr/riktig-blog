Library Riktig.Interface.DecTotalOrder

From Coq Require Import RelationClasses.
From Riktig Require Import Interface.Decidable.

Class HasOlt (A : Type) := olt : A → A → Prop.
Class HasOle (A : Type) := ole : A → A → Prop.

Hint Mode HasOlt ! : typeclass_instances.
Hint Mode HasOle ! : typeclass_instances.

Typeclasses Transparent HasOlt HasOle.

A decidable total order is a strict order with decidable comparison in which any two elements are strictly ordered or equal, and ole x y holds exactly when olt y x fails.
Module DecTotalOrder.
  Class DecTotalOrder (A : Type) {olt : HasOlt A} {ole : HasOle A} := {
    olt_dec :: ∀ x y : A, Decidable (olt x y);
    olt_strict :: StrictOrder olt;

    olt_total : ∀ x y : A, olt x y ∨ x = y ∨ olt y x;
    ole_not_lt : ∀ x y : A, ole x y ↔ ¬ olt y x;
  }.
  Hint Mode DecTotalOrder ! - - : typeclass_instances.
End DecTotalOrder.
Export DecTotalOrder.

Definition ogt {A} `{HasOlt A} (x y : A) : Prop := olt y x.
Definition oge {A} `{HasOle A} (x y : A) : Prop := ole y x.

Declare Scope order_scope.
Infix "<" := olt : order_scope.
Infix "≤" := ole : order_scope.
Infix ">" := ogt : order_scope.
Infix "≥" := oge : order_scope.
