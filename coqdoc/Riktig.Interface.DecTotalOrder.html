Library Riktig.Interface.DecTotalOrder


From Coq Require Import RelationClasses.
From Riktig Require Import Interface.Decidable.

Class HasOlt (A : Type) := olt : A → A → Prop.
Class HasOle (A : Type) := ole : A → A → Prop.

#[global] Hint Mode HasOlt ! : typeclass_instances.
#[global] Hint Mode HasOle ! : typeclass_instances.

Module DecTotalOrder.
  Class DecTotalOrder (A : Type) {Alt : HasOlt A} {Ale : HasOle A} := {
    olt_dec : ∀ x y : A, Decidable (olt x y);
    olt_strict : StrictOrder (@olt A Alt);
    olt_total : ∀ x y : A, olt x y ∨ x = y ∨ olt y x;
    ole_not_lt : ∀ x y : A, ole x y ↔ ¬ olt y x
  }.
End DecTotalOrder.
Notation DecTotalOrder := DecTotalOrder.DecTotalOrder.

#[global] Instance DecTotalOrder_olt_dec (A : Type) {Alt : HasOlt A} {Ale : HasOle A}
  {O : DecTotalOrder.DecTotalOrder A} : ∀ x y : A, Decidable (olt x y)
  := DecTotalOrder.olt_dec.

#[global] Instance DecTotalOrder_StrictOrder (A : Type) {Alt : HasOlt A} {Ale : HasOle A}
  {O : DecTotalOrder.DecTotalOrder A} : StrictOrder (@olt A Alt)
  := DecTotalOrder.olt_strict.

#[global] Hint Mode DecTotalOrder.DecTotalOrder ! - - : typeclass_instances.

Definition ogt {A} `{HasOlt A} (x y : A) : Prop := olt y x.
Definition oge {A} `{HasOle A} (x y : A) : Prop := ole y x.

Declare Scope order_scope.
Infix "<" := olt : order_scope.
Infix "≤" := ole : order_scope.
Infix ">" := ogt : order_scope.
Infix "≥" := oge : order_scope.
