Definition Sub (V : Type) (P : V → Prop) := { v : V | P v }.

#[global] Instance Sub_EqRel {V : Type} `{EqRel V} (P : V → Prop) :
  EqRel (Sub V P) :=
  fun x y ⇒ proj1_sig x == proj1_sig y.

#[global] Instance Sub_Zero {F V : Type} `{VectorSpace F V}
    (P : V → Prop) {SPP : SubspacePred F V P} : Zero (Sub V P) :=
  exist P 0 SubspacePred.zero.

#[global] Instance Sub_Add {F V : Type} `{VectorSpace F V}
    (P : V → Prop) {SPP : SubspacePred F V P} : HasAdd (Sub V P) :=
  fun x y ⇒
    exist P (proj1_sig x + proj1_sig y)
      (SubspacePred.add_closed (proj1_sig x) (proj1_sig y) (proj2_sig x) (proj2_sig y)).

#[global] Instance Sub_Neg {F V : Type} `{VectorSpace F V}
    (P : V → Prop) {SPP : SubspacePred F V P} : HasNeg (Sub V P) :=
  fun x ⇒
    exist P ((- (1 : F)) *: proj1_sig x)
      (SubspacePred.scale_closed (- (1 : F)) (proj1_sig x) (proj2_sig x)).

#[global] Instance Sub_Scale {F V : Type} `{VectorSpace F V}
    (P : V → Prop) {SPP : SubspacePred F V P} : HasScale F (Sub V P) :=
  fun a x ⇒ exist P (a *: proj1_sig x) (SubspacePred.scale_closed a (proj1_sig x) (proj2_sig x)).

#[global] Instance Sub_Equivalence {V : Type} {e : EqRel V} {E : Equivalence e}
    (P : V → Prop) : Equivalence (@Sub_EqRel V e P).
Proof.
  apply Build_Equivalence.
  - intros x. unfold Sub_EqRel. reflexivity.
  - intros x y Hxy. unfold Sub_EqRel in ×. symmetry. exact Hxy.
  - intros x y z Hxy Hyz. unfold Sub_EqRel in ×. rewrite Hxy. exact Hyz.
Qed.

Section SubMonoid.
  Context {F V : Type} `{VectorSpace F V}.
  Context (P : V → Prop) {SPP : SubspacePred F V P}.

  Lemma sub_assoc : ∀ x y z : Sub V P, (x + y) + z == x + (y + z).
  Proof.
    intros x y z.
    exact (associative (proj1_sig x) (proj1_sig y) (proj1_sig z)).
  Qed.

  Lemma sub_idl : ∀ x : Sub V P, 0 + x == x.
  Proof. intros x. exact (left_identity (B := V) (proj1_sig x)). Qed.

  Lemma sub_idr : ∀ x : Sub V P, x + 0 == x.
  Proof. intros x. exact (right_identity (proj1_sig x)). Qed.

  #[global] Instance sub_add_proper : Proper (equ ==> equ ==> equ) (@add (Sub V P) _).
  Proof.
    intros x x' Hx y y' Hy.
    change (proj1_sig x == proj1_sig x') in Hx.
    change (proj1_sig y == proj1_sig y') in Hy.
    exact (Magma.op_proper (proj1_sig x) (proj1_sig x') Hx
             (proj1_sig y) (proj1_sig y') Hy).
  Qed.

End SubMonoid.

Section SubGroup.
  Context {F V : Type} `{VectorSpace F V}.
  Context (P : V → Prop) {SPP : SubspacePred F V P}.

  Lemma sub_invl : ∀ x : Sub V P, (- x) + x == 0.
  Proof.
    intros x.
    change ((- (1 : F)) *: proj1_sig x + proj1_sig x == 0).
    rewrite (neg1_scale (proj1_sig x)).
    exact (left_inverse _).
  Qed.

  Lemma sub_invr : ∀ x : Sub V P, x + (- x) == 0.
  Proof.
    intros x.
    change (proj1_sig x + (- (1 : F)) *: proj1_sig x == 0).
    rewrite (neg1_scale (proj1_sig x)).
    exact (right_inverse _).
  Qed.

  #[global] Instance sub_neg_proper : Proper (equ ==> equ) (@neg (Sub V P) _).
  Proof.
    intros x x' Hx.
    change ((- (1 : F)) *: proj1_sig x == (- (1 : F)) *: proj1_sig x').
    apply ScaleLaws.scale_proper; [reflexivity | exact Hx].
  Qed.

  #[global] Instance Sub_Group : Group (Sub V P) := {| Group.monoid := {| Monoid.semigroup := {| Semigroup.magma := {| Magma.equivalence := Sub_Equivalence P;
                         Magma.op_proper := sub_add_proper P |};
   Semigroup.is_associative := sub_assoc P |};
   Monoid.is_left_identity := sub_idl P;
   Monoid.is_right_identity := sub_idr P |};
   Group.neg_proper := sub_neg_proper;
   Group.is_left_inverse := sub_invl;
   Group.is_right_inverse := sub_invr |}.

End SubGroup.

Section SubVectorSpace.
  Context {F V : Type} `{VectorSpace F V}.
  Context (P : V → Prop) {SPP : SubspacePred F V P}.

  #[global] Instance sub_scale_proper : Proper (equ ==> equ ==> equ) (@scale F (Sub V P) _).
  Proof.
    intros a a' Ha x x' Hx.
    change (a *: proj1_sig x == a' *: proj1_sig x').
    apply ScaleLaws.scale_proper; [exact Ha | exact Hx].
  Qed.

  Lemma sub_scale_add : ∀ (a : F) (x y : Sub V P),
    a *: (x + y) == a *: x + a *: y.
  Proof.
    intros a x y.
    exact (left_distributive (C := V) a (proj1_sig x) (proj1_sig y)).
  Qed.

  Lemma sub_add_scale : ∀ (a b : F) (x : Sub V P),
    (a + b) *: x == a *: x + b *: x.
  Proof.
    intros a b x.
    exact (right_distributive (C := V) a b (proj1_sig x)).
  Qed.

  Lemma sub_scale_scale : ∀ (a b : F) (x : Sub V P),
    (a × b) *: x == a *: (b *: x).
  Proof.
    intros a b x.
    exact (associative (B := V) a b (proj1_sig x)).
  Qed.

  Lemma sub_scale_1 : ∀ x : Sub V P, 1 *: x == x.
  Proof. intros x. exact (left_identity (B := V) (proj1_sig x)). Qed.

  #[global] Instance Sub_ScaleLaws : ScaleLaws F (Sub V P) :=
    {| ScaleLaws.scale_proper := sub_scale_proper;
       ScaleLaws.is_left_distributive := sub_scale_add;
       ScaleLaws.is_right_distributive := sub_add_scale;
       ScaleLaws.is_associative := sub_scale_scale;
       ScaleLaws.is_left_identity := sub_scale_1 |}.

  #[global] Instance Sub_VectorSpace : VectorSpace F (Sub V P) :=
    {| VectorSpace.vectors := Sub_AbGroup P; VectorSpace.scaling := Sub_ScaleLaws |}.

End SubVectorSpace.

Section SubInject.
  Context {F V : Type} `{VectorSpace F V}.
  Context (P : V → Prop) {SPP : SubspacePred F V P}.

  #[global] Instance sub_inject_proper : Proper (equ ==> equ) (fun x : Sub V P ⇒ proj1_sig x).
  Proof. intros x y Hxy. exact Hxy. Qed.

  Lemma sub_inject_add : ∀ x y : Sub V P,
    proj1_sig (x + y) == proj1_sig x + proj1_sig y.
  Proof. intros x y. reflexivity. Qed.

  Lemma sub_inject_scale : ∀ (a : F) (x : Sub V P),
    proj1_sig (a *: x) == a *: proj1_sig x.
  Proof. intros a x. reflexivity. Qed.

  Definition sub_inject : LinearMap F (Sub V P) V :=
    linear_map_from_laws
      (fun x ⇒ proj1_sig x)
      sub_inject_proper
      sub_inject_add
      sub_inject_scale.

End SubInject.
