From Coq Require Export NArith.
From Coq Require Import RelationClasses.

What Impl/BinaryNumberN needs of Coq's binary naturals: the two bit appendings are injective and apart, a zero bit on zero is zero, and N.binary_rect is the recursion, with its equations.

Definition binrec_N (A : Type) (z : A) (f0 f1 : N → A → A) (n : N) : A :=
  N.binary_rect (fun _ ⇒ A) z f0 f1 n.

Lemma binrec_N_zero : ∀ (A : Type) (z : A) (f0 f1 : N → A → A), binrec_N A z f0 f1 0 = z.
Proof. reflexivity. Qed.

Lemma binrec_N_double : ∀ (A : Type) (z : A) (f0 f1 : N → A → A) (n : N),
  n ≠ 0%N → binrec_N A z f0 f1 (N.double n) = f0 n (binrec_N A z f0 f1 n).
Proof. intros A z f0 f1 [| p] Hn; [ destruct (Hn eq_refl) | reflexivity ]. Qed.

Lemma binrec_N_succ_double : ∀ (A : Type) (z : A) (f0 f1 : N → A → A) (n : N),
  binrec_N A z f0 f1 (N.succ_double n) = f1 n (binrec_N A z f0 f1 n).
Proof. intros A z f0 f1 [| p]; reflexivity. Qed.

Lemma pos_not_zero_N : ∀ p : positive, N.pos p ≠ 0%N.
Proof. intros p; discriminate. Qed.

Lemma binrec_N_ext : ∀ (A : Type) (R : A → A → Prop) (z z' : A) (f0 f0' f1 f1' : N → A → A),
  Reflexive R → Transitive R →
  (∀ k x y, R x y → R (f0 k x) (f0 k y)) → (∀ k x y, R x y → R (f1 k x) (f1 k y)) →
  R z z' → (∀ k x, R (f0 k x) (f0' k x)) → (∀ k x, R (f1 k x) (f1' k x)) →
  ∀ n : N, R (binrec_N A z f0 f1 n) (binrec_N A z' f0' f1' n).
Proof.
  intros A R z z' f0 f0' f1 f1' Hrefl Htrans Hf0 Hf1 Hz E0 E1.
  apply N.binary_ind.
  - exact Hz.
  - intros [| p] IH; [ exact Hz | ].
    rewrite (binrec_N_double A z f0 f1 (N.pos p) (pos_not_zero_N p)).
    rewrite (binrec_N_double A z' f0' f1' (N.pos p) (pos_not_zero_N p)).
    exact (Htrans _ _ _ (Hf0 _ _ _ IH) (E0 _ _)).
  - intros n IH. rewrite binrec_N_succ_double, binrec_N_succ_double.
    exact (Htrans _ _ _ (Hf1 _ _ _ IH) (E1 _ _)).
Qed.

Lemma succ_double_not_zero_N : ∀ n : N, N.succ_double n ≠ 0%N.
Proof. intros [| p]; discriminate. Qed.

Lemma double_ne_succ_double_N : ∀ n m : N, N.double n ≠ N.succ_double m.
Proof. intros [| p] [| q]; discriminate. Qed.
