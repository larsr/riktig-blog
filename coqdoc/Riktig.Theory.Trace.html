Section MatrixTrace.
  Context {F : Type} `{FF : Field F}.

  Definition matrix_trace {n : nat} (A : Matrix n n F) : F :=
    bigopF (fun i ⇒ A i i).

  #[export] Instance matrix_trace_proper : ∀ n : nat, Proper (equ ==> equ) (@matrix_trace n).
  Proof.
    intros n A A' HA.
    exact (bigopF_ext n (fun i ⇒ A i i) (fun i ⇒ A' i i)
             (fun i ⇒ HA i i)).
  Qed.

  Lemma matrix_trace_mul_expand : ∀ (m n : nat) (A : Matrix m n F)
      (B : Matrix n m F),
    matrix_trace (mat_mul A B)
      == bigopF (fun i : Fin.t m ⇒ bigopF (fun j : Fin.t n ⇒ A i j × B j i)).
  Proof.
    intros m n A B. reflexivity.
  Qed.

  Theorem tr_mul_comm_rect : ∀ (m n : nat) (A : Matrix m n F)
      (B : Matrix n m F),
    matrix_trace (mat_mul A B) == matrix_trace (mat_mul B A).
  Proof.
    intros m n A B.
    rewrite (matrix_trace_mul_expand m n A B), (matrix_trace_mul_expand n m B A).
    calc (bigopF (fun i : Fin.t m ⇒ bigopF (fun j : Fin.t n ⇒ A i j × B j i)))
      == (bigopF (fun j : Fin.t n ⇒ bigopF (fun i : Fin.t m ⇒ A i j × B j i)))
      == (bigopF (fun i : Fin.t n ⇒ bigopF (fun j : Fin.t m ⇒ B i j × A j i))).
    - exact (bigopF_swap m n (fun i j ⇒ A i j × B j i)).
    - apply bigopF_ext. intros j.
      apply bigopF_ext. intros i.
      exact (mul_comm (A i j) (B j i)).
  Qed.

  Theorem tr_mul_comm : ∀ (n : nat) (A B : Matrix n n F),
    matrix_trace (mat_mul A B) == matrix_trace (mat_mul B A).
  Proof.
    intros n A B.
    exact (tr_mul_comm_rect n n A B).
  Qed.

End MatrixTrace.

Section MatOfId.
  Context {F U : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {n : nat} (B : Basis n F U).

  Theorem mat_of_id : mat_of B B (linear_map_id (F:=F) (U:=U)) == mat_id.
  Proof.
    intros i j.
    calc (mat_of B B (linear_map_id (F:=F) (U:=U)) i j)
      == (@e F _ _ n j i)
      == (@mat_id F _ _ n i j).
    - exact (coords_vec B j i).
    - exact (e_sym n j i).
  Qed.

End MatOfId.


Section ChangeOfBasis.
  Context {F U : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {n : nat}.

  Definition change_of_basis (B B' : Basis n F U) : Matrix n n F :=
    mat_of B' B (linear_map_id (F:=F) (U:=U)).

  Lemma lm_comp_id_id :
    linear_map_comp (linear_map_id (F:=F) (U:=U)) (linear_map_id (F:=F) (U:=U)) == linear_map_id (F:=F) (U:=U).
  Proof.
    intros u. reflexivity.
  Qed.

  Lemma lm_id_comp_expand : ∀ f : LinearMap F U U,
    f == linear_map_comp (linear_map_id (F:=F) (U:=U)) (linear_map_comp f (linear_map_id (F:=F) (U:=U))).
  Proof.
    intros f u. reflexivity.
  Qed.

  Theorem change_of_basis_inv : ∀ B B' : Basis n F U,
    mat_mul (change_of_basis B' B) (change_of_basis B B') == mat_id.
  Proof.
    intros B B'.
    calc (mat_mul (change_of_basis B' B) (change_of_basis B B'))
      == (mat_of B' B' (linear_map_comp (linear_map_id (F:=F) (U:=U)) (linear_map_id (F:=F) (U:=U))))
      == (mat_of B' B' (linear_map_id (F:=F) (U:=U)))
      == (@mat_id F _ _ n).
    - symmetry.
      exact (mat_of_comp B' B B' (linear_map_id (F:=F) (U:=U)) (linear_map_id (F:=F) (U:=U))).
    - exact (mat_of_proper B' B'
               (linear_map_comp (linear_map_id (F:=F) (U:=U)) (linear_map_id (F:=F) (U:=U)))
               (linear_map_id (F:=F) (U:=U)) lm_comp_id_id).
    - exact (mat_of_id B').
  Qed.

  Theorem mat_of_conj : ∀ (B B' : Basis n F U) (f : LinearMap F U U),
    mat_of B' B' f
      == mat_mul (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B')).
  Proof.
    intros B B' f.
    calc (mat_of B' B' f)
      == (mat_of B' B' (linear_map_comp (linear_map_id (F:=F) (U:=U)) (linear_map_comp f (linear_map_id (F:=F) (U:=U)))))
      == (mat_mul (change_of_basis B' B) (mat_of B' B (linear_map_comp f (linear_map_id (F:=F) (U:=U)))))
      == (mat_mul (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B'))).
    - exact (mat_of_proper B' B' f
               (linear_map_comp (linear_map_id (F:=F) (U:=U)) (linear_map_comp f (linear_map_id (F:=F) (U:=U))))
               (lm_id_comp_expand f)).
    - exact (mat_of_comp B' B B' (linear_map_id (F:=F) (U:=U))
               (linear_map_comp f (linear_map_id (F:=F) (U:=U)))).
    - exact (mat_mul_proper n n n
               (change_of_basis B' B) (change_of_basis B' B) (reflexivity (change_of_basis B' B))
               (mat_of B' B (linear_map_comp f (linear_map_id (F:=F) (U:=U))))
               (mat_mul (mat_of B B f) (change_of_basis B B'))
               (mat_of_comp B' B B f (linear_map_id (F:=F) (U:=U)))).
  Qed.

End ChangeOfBasis.

Section TraceBasisIndep.
  Context {F U : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {n : nat}.

  Theorem tr_basis_independent : ∀ (B B' : Basis n F U)
      (f : LinearMap F U U),
    matrix_trace (mat_of B' B' f) == matrix_trace (mat_of B B f).
  Proof.
    intros B B' f.
    calc (matrix_trace (mat_of B' B' f))
      == (matrix_trace (mat_mul (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B'))))
      == (matrix_trace (mat_mul (mat_mul (mat_of B B f) (change_of_basis B B')) (change_of_basis B' B)))
      == (matrix_trace (mat_mul (mat_of B B f) (mat_mul (change_of_basis B B') (change_of_basis B' B))))
      == (matrix_trace (mat_mul (mat_of B B f) (@mat_id F _ _ n)))
      == (matrix_trace (mat_of B B f)).
    - exact (matrix_trace_proper n
               (mat_of B' B' f)
               (mat_mul (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B')))
               (mat_of_conj B B' f)).
    - exact (tr_mul_comm n (change_of_basis B' B)
               (mat_mul (mat_of B B f) (change_of_basis B B'))).
    - exact (matrix_trace_proper n
               (mat_mul (mat_mul (mat_of B B f) (change_of_basis B B')) (change_of_basis B' B))
               (mat_mul (mat_of B B f) (mat_mul (change_of_basis B B') (change_of_basis B' B)))
               (mat_mul_assoc n n n n (mat_of B B f)
                  (change_of_basis B B') (change_of_basis B' B))).
    - exact (matrix_trace_proper n
               (mat_mul (mat_of B B f) (mat_mul (change_of_basis B B') (change_of_basis B' B)))
               (mat_mul (mat_of B B f) (@mat_id F _ _ n))
               (mat_mul_proper n n n
                  (mat_of B B f) (mat_of B B f)
                  (reflexivity (mat_of B B f))
                  (mat_mul (change_of_basis B B') (change_of_basis B' B)) (@mat_id F _ _ n)
                  (change_of_basis_inv B' B))).
    - exact (matrix_trace_proper n
               (mat_mul (mat_of B B f) (@mat_id F _ _ n))
               (mat_of B B f)
               (mat_mul_right_identity n n (mat_of B B f))).
  Qed.

  Definition tr (B : Basis n F U) (f : LinearMap F U U) : F :=
    matrix_trace (mat_of B B f).

  Theorem tr_indep : ∀ (B B' : Basis n F U) (f : LinearMap F U U),
    tr B' f == tr B f.
  Proof.
    intros B B' f.
    exact (tr_basis_independent B B' f).
  Qed.

End TraceBasisIndep.

Example tr_check (F : Type) `{FF : Field F} (n : nat) : F :=
  tr (FinVec_Basis (F := F) n) linear_map_id.
