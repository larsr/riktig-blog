Library Riktig.Lib.NatFacts

Require Import Arith Lia.

Lemma half_le d : d / 2 ≤ d.
Proof. apply Nat.Div0.div_le_upper_bound. lia. Qed.

Lemma two_mul_succ : ∀ k : nat, (2 × S k)%nat = S (S (2 × k)).
Proof. intros k. lia. Qed.


Lemma double_add : ∀ n : nat, (2 × n)%nat = (n + n)%nat.
Proof. intros n. lia. Qed.

Lemma succ_double_add : ∀ n : nat, S (2 × n) = S (n + n).
Proof. intros n. lia. Qed.


Lemma le_double : ∀ n : nat, (n ≤ 2 × n)%nat.
Proof. intros n. lia. Qed.

Lemma sub_add_split : ∀ n k : nat, (k ≤ n)%nat → n = (n - k + k)%nat.
Proof. intros n k Hk. lia. Qed.

Lemma le_of_add_le : ∀ a k n : nat, (a + k ≤ n)%nat → (k ≤ n)%nat.
Proof. intros a k n Hn. lia. Qed.

Lemma le_sub_of_add_le : ∀ a k n : nat,
  (a + k ≤ n)%nat → (a ≤ n - k)%nat.
Proof. intros a k n Hn. lia. Qed.

Lemma le_add_of_le : ∀ a b k : nat, (a ≤ b)%nat → (a ≤ b + k)%nat.
Proof. intros a b k Hab. lia. Qed.

Lemma lt_sub_of_lt : ∀ i m k : nat, (i < m)%nat → (i < m + k - k)%nat.
Proof. intros i m k Hi. lia. Qed.

Lemma le_sub_of_lt_sub : ∀ i n k : nat,
  (i < n - k)%nat → (k ≤ n - i)%nat.
Proof. intros i n k Hi. lia. Qed.

Lemma max_le_l : ∀ a b n : nat,
  (Nat.max a b ≤ n)%nat → (a ≤ n)%nat.
Proof.
  intros a b n Hn.
  exact (Nat.le_trans a (Nat.max a b) n (Nat.le_max_l a b) Hn).
Qed.

Lemma max_le_r : ∀ a b n : nat,
  (Nat.max a b ≤ n)%nat → (b ≤ n)%nat.
Proof.
  intros a b n Hn.
  exact (Nat.le_trans b (Nat.max a b) n (Nat.le_max_r a b) Hn).
Qed.

Lemma nat_leb_total : ∀ x y : nat,
  Nat.leb x y = false → Nat.leb y x = true.
Proof.
  intros x y H. apply Nat.leb_gt in H. apply Nat.leb_le. lia.
Qed.
