The inner product is positive definite: ⟨ u , u ⟩ is nonnegative and vanishes only at u == 0.
Module PosInner.
  Class PosInner (F V : Type) `{OrderedField F}
      {piE : EqRel V} {piZ : Zero V} {piA : HasAdd V} {piN : HasNeg V}
      {piS : HasScale F V} {piI : HasInner F V}
      {VS : VectorSpace F V} {IP : InnerProductSpace F V} := {
    nonneg : ∀ u : V, 0 ≤ ⟨ u , u ⟩;
    definite : ∀ u : V, ⟨ u , u ⟩ == 0 → u == 0
  }.
  Hint Mode PosInner - ! - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End PosInner.
Export PosInner.
