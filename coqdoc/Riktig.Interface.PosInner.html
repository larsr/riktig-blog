Module PosInner.
  Class PosInner (F V : Type) `{OrderedField.OrderedField F}
      {piE : EqRel V} {piZ : Zero V} {piA : HasAdd V} {piN : HasNeg V}
      {piS : HasScale F V} {piI : HasInner F V}

    nonneg : ∀ u : V, 0 ≤ ⟨ u , u ⟩;

  }.
End PosInner.
