Library Riktig.Structure.Semiring

From Riktig.Structure Require Export Monoid.

A semiring is a commutative monoid under addition and a monoid under multiplication, with multiplication distributing over addition on both sides and annihilating zero on both sides.
Module Semiring.
  Class Semiring (R : Type) {eq : EqRel R} (zero : Zero R) (add : HasAdd R)
                 (one : One R) (mult : HasMult R) := {
    add_monoid :: Monoid R 0 add;
    add_commutative :: Commutative R R add;
    mul_monoid :: Monoid R 1 mult;
    is_left_distributive :: LeftDistributive R R R mult;
    is_right_distributive :: RightDistributive R R R mult;
    mul_zero_left :: LeftAbsorbing R R mult 0 0;
    mul_zero_right :: RightAbsorbing R R mult 0 0;
  }.
  Hint Mode Semiring ! - - - - - : typeclass_instances.
End Semiring.
Export Semiring.

A semiring whose multiplication is commutative.
Module CommSemiring.
  Class CommSemiring (R : Type) {eq : EqRel R} (zero : Zero R) (add : HasAdd R)
                     (one : One R) (mult : HasMult R) := {
    semiring :: Semiring R zero add one mult;
    mul_commutative :: Commutative R R mult;
  }.
  Hint Mode CommSemiring ! - - - - - : typeclass_instances.
End CommSemiring.
Export CommSemiring.
