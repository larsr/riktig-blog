The chain rule across three spaces: the derivative of a composite is the composite of the derivatives.
Module ChainRule.
  Class ChainRule (U V W : Type) (dUV : HasDifferential U V) (dVW : HasDifferential V W)
      (dUW : HasDifferential U W) := {
    differential_comp : ∀ (f : U → V) (g : V → W) (a : U) (Lf : U → V) (Lg : V → W),
      differential f a Lf → differential g (f a) Lg →
      differential (fun x ⇒ g (f x)) a (fun h ⇒ Lg (Lf h));
  }.
  Hint Mode ChainRule ! ! ! - - - : typeclass_instances.
End ChainRule.
