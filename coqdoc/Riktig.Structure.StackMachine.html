From Riktig.Structure Require Export Monoid.

A stack machine. The code is a monoid -- the empty program 0 and concatenation + -- generated by two instructions: push_instr v pushes the value v, and op_instr replaces the top two cells of the stack by their binop. Running code is a stack transformer, and its first two laws say that the empty program does nothing and that running a concatenation runs one program after the other. That second law is the whole of the associativity in this development: it carries the code monoid into the stack transformers, reversing the order against composition and preserving it against sequencing, and it says nothing at all about binop, which is not assumed associative (Wand, "Deriving target code as a representation of continuation semantics", 1982; Gibbons, arXiv:2111.10413, section 5).
Module StackMachine.
  Class StackMachine (V C S : Type)
      {eqV : EqRel V} {eqC : EqRel C} {eqS : EqRel S} {zC : Zero C} {aC : HasAdd C}
      (binop : V → V → V) (push_instr : V → C) (op_instr : C)
      (exec : C → S → S) (push : V → S → S) := {
    code :: Monoid C 0 add;
    values :: Equivalence (@equ V eqV);
    stacks :: Equivalence (@equ S eqS);
    binop_proper :: Proper (equ ==> equ ==> equ) binop;
    exec_proper :: Proper (equ ==> equ ==> equ) exec;
    push_proper :: Proper (equ ==> equ ==> equ) push;
    
    exec_empty : ∀ s : S, exec 0 s == s;
    exec_antiadditive :: Antiadditive C (S → S) add (fun g h s ⇒ g (h s)) exec
                           (eqB := fun f g ⇒ ∀ s : S, f s == g s);
    
    exec_push : ∀ (v : V) (s : S), exec (push_instr v) s == push v s;
    exec_op : ∀ (a b : V) (s : S), exec op_instr (push b (push a s)) == push (binop a b) s;
  }.
End StackMachine.
