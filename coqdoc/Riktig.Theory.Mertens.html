From Riktig.Theory Require Export CauchyProduct ConstructiveRealsLimits.

Mertens' theorem: if one series converges absolutely and the other converges, the Cauchy-product series converges to the product of the sums. The partial sums of the product are, by the triangle exchange, sum_j a_j × B (n - j); writing B m = Lb + r m with r the remainder, the error term sum_j a_j × r (n - j) splits at n - nr: below the split the remainders are small, above it the window of |a| is small, and each side is worth half of epsilon.
Section Mertens.
  Context {R Q N : Type} `{ConstructiveReals R Q N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Local Notation "'psum' f n" := (natrec _ 0 (fun k acc ⇒ acc + f k) n) (at level 10, f at level 9, n at level 9).
  Local Notation "'is_cauchy' u" :=
    (∀ eps : Q, 0 < eps →
      {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps})
    (at level 10, u at level 9).
  Local Notation "'converges' u l" :=
    (∀ eps : Q, 0 < eps → {n : N & ∀ i : N, n ≤ i → abs (u i + - l) ≼ cast eps})
    (at level 10, u at level 9, l at level 9).
  Local Notation half eps := (eps × inv (1 + 1)).
  Local Notation "'pred' n" := (natrec N 0 (fun k _ ⇒ k) n) (at level 10, n at level 9).
  Local Notation "m ∸ j" := (natrec N m (fun _ acc ⇒ pred acc) j) (at level 50, left associativity).

  Variables a b : N → R.
  Hypothesis Ha : Proper (equ ==> equ) a.
  Hypothesis Hb : Proper (equ ==> equ) b.
  Variables La Lb : R.
  Hypothesis Hca : converges (fun n ⇒ psum a n) La.
  Hypothesis Hcb : converges (fun n ⇒ psum b n) Lb.
  Hypothesis Haabs : is_cauchy (fun n ⇒ psum (fun k ⇒ abs (a k)) n).

  Local Notation r m := (psum b m + - Lb).

  Lemma r_proper : Proper (equ ==> equ) (fun m : N ⇒ r m).
  Proof. intros m m' Hm. apply Magma.op_proper; [ exact (psum_proper b Hb m m' Hm) | reflexivity ]. Qed.

  Lemma aabs_proper : Proper (equ ==> equ) (fun k : N ⇒ abs (a k)).
  Proof. intros k k' Hk. apply ConstructiveReals.abs_proper. apply Ha. exact Hk. Qed.

  Lemma err_proper : ∀ n : N, Proper (equ ==> equ) (fun j : N ⇒ a j × r (n ∸ j)).
  Proof.
    intros n j j' Hj. apply Magma.op_proper; [ apply Ha; exact Hj | ].
    exact (r_proper _ _ (monus_index_proper n j j' Hj)).
  Qed.

  Lemma errabs_proper : ∀ n : N, Proper (equ ==> equ) (fun j : N ⇒ abs (a j) × abs (r (n ∸ j))).
  Proof.
    intros n j j' Hj. apply Magma.op_proper; [ exact (aabs_proper j j' Hj) | ].
    apply ConstructiveReals.abs_proper. exact (r_proper _ _ (monus_index_proper n j j' Hj)).
  Qed.

  Lemma mertens_error_small : ∀ eps : Q, 0 < eps →
    {n2 : N & ∀ n : N, n2 ≤ n → abs (psum (fun j ⇒ a j × r (n ∸ j)) n) ≼ cast eps}.
  Proof.
    intros eps Heps.
    destruct (Hcb 1 (Theory.Order.zero_lt_one (F := Q))) as [nr1 Hnr1].
    destruct (bound_of_bound_tail (fun m ⇒ r m) r_proper nr1 1 (Theory.Order.zero_lt_one (F := Q)) Hnr1) as [Mr [HMr HrM]].
    destruct (cauchy_bound (fun n ⇒ psum (fun k ⇒ abs (a k)) n) (psum_proper _ aabs_proper) Haabs) as [A [HA HSA]].
    assert (HneA : A =/= 0) by exact (fun H0 ⇒ proj2 HA (symmetry H0)).
    assert (HneMr : Mr =/= 0) by exact (fun H0 ⇒ proj2 HMr (symmetry H0)).
    pose (eps1 := half eps × inv A).
    assert (Heps1 : 0 < eps1)
      by exact (Theory.Order.mul_pos _ _ (Theory.Order.half_pos eps Heps) (Theory.Order.inv_pos A HA)).
    pose (eps2 := half eps × inv Mr).
    assert (Heps2 : 0 < eps2)
      by exact (Theory.Order.mul_pos _ _ (Theory.Order.half_pos eps Heps) (Theory.Order.inv_pos Mr HMr)).
    destruct (Hcb eps1 Heps1) as [nr Hnr].
    destruct (Haabs eps2 Heps2) as [n1 Hn1].
    ∃ ((n1 + nr) + nr). intros n Hn.
    assert (Hnrn : nr ≤ n) by exact (le_of_add_le_r (n1 + nr) nr n Hn).
    assert (Hn1p : n1 ≤ n ∸ nr).
    { apply (le_monus n nr n1).
      apply (proj1 (NatOrder.le_proper (n1 + nr) (nr + n1) (commutative (op := add) n1 nr) n n (reflexivity n))).
      exact (le_of_add_le_l (n1 + nr) nr n Hn). }
    assert (Hsmall : ∀ j : N, succ j ≤ n ∸ nr → abs (r (n ∸ j)) ≼ cast eps1).
    { intros j Hj.
      assert (Hjn : succ j + nr ≤ n).
      { apply (proj1 (NatOrder.le_proper (succ j + nr) (succ j + nr) (reflexivity _) ((n ∸ nr) + nr) n (monus_add_cancel n nr Hnrn))).
        exact (Theory.NatOrder.le_add_mono_r (succ j) (n ∸ nr) nr Hj). }
      apply (Hnr (n ∸ j)).
      apply (le_monus n j nr).
      apply (Theory.NatOrder.le_trans (j + nr) (succ (j + nr)) n (Theory.NatOrder.le_succ_diag (j + nr))).
      exact (proj1 (NatOrder.le_proper (succ j + nr) (succ (j + nr)) (Theory.Naturals.add_succ_l j nr) n n (reflexivity n)) Hjn). }
    assert (Hshiftp : Proper (equ ==> equ) (fun k : N ⇒ abs (a ((n ∸ nr) + k))))
      by (intros k k' Hk; apply aabs_proper; apply Magma.op_proper; [ reflexivity | exact Hk ]).
    assert (Hbelow : psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) (n ∸ nr) ≼ cast (half eps)).
    { assert (Hterm : ∀ k : N, succ k ≤ n ∸ nr → abs (a k) × abs (r (n ∸ k)) ≼ abs (a k) × cast eps1)
        by (intros k Hk; exact (ConstructiveReals.mul_le_mono_l _ _ _ (Theory.ConstructiveReals.abs_nonneg _) (Hsmall k Hk))).
      assert (Hcoef : Proper (equ ==> equ) (fun j : N ⇒ abs (a j) × cast eps1))
        by (intros j j' Hj; apply Magma.op_proper; [ exact (aabs_proper j j' Hj) | reflexivity ]).
      assert (Hbound : psum (fun k ⇒ abs (a k)) (n ∸ nr) ≼ cast A).
      { calc (psum (fun k ⇒ abs (a k)) (n ∸ nr))
          == (abs (psum (fun k ⇒ abs (a k)) (n ∸ nr))) { (ConstructiveReals.abs_of_nonneg _ (psum_abs_nonneg a Ha (n ∸ nr))) }
          ≼ (cast A) { (HSA (n ∸ nr)) }. }
      calc (psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) (n ∸ nr))
        ≼ (psum (fun j ⇒ abs (a j) × cast eps1) (n ∸ nr)) { (psum_le_below _ _ (errabs_proper n) Hcoef (n ∸ nr) Hterm) }
        == (psum (fun k ⇒ abs (a k)) (n ∸ nr) × cast eps1) { (psum_mul_r _ (cast eps1) aabs_proper (n ∸ nr)) }
        ≼ (cast A × cast eps1)
             { (Theory.ConstructiveReals.mul_le_mono_r _ _ _ (cast_nonneg _ (Theory.Order.le_of_lt _ _ Heps1)) Hbound) }
        == (cast (A × eps1)) { (additive (f := cast) (opA := mult) A eps1) }
        == (cast (half eps)) { (RingHom.hom_proper (h := cast) _ _ (mul_mul_inv_cancel_left (half eps) A HneA)) }. }
    
    assert (Hwin : psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr ≼ cast eps2).
    { assert (Hwindow : psum (fun k ⇒ abs (a k)) ((n ∸ nr) + nr) + - psum (fun k ⇒ abs (a k)) (n ∸ nr)
                        == psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr).
      { calc (psum (fun k ⇒ abs (a k)) ((n ∸ nr) + nr) + - psum (fun k ⇒ abs (a k)) (n ∸ nr))
          == ((psum (fun k ⇒ abs (a k)) (n ∸ nr) + psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr)
                + - psum (fun k ⇒ abs (a k)) (n ∸ nr)) { (psum_shift _ aabs_proper (n ∸ nr) nr) }
          == (psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr) { (add_sub_cancel_left (psum (fun k ⇒ abs (a k)) (n ∸ nr)) (psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr)) }. }
      calc (psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr)
        == (abs (psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr))
             { (ConstructiveReals.abs_of_nonneg _ (psum_abs_nonneg (fun k ⇒ a ((n ∸ nr) + k))
                  (fun k k' Hk ⇒ Ha _ _ (Magma.op_proper _ _ (reflexivity _) k k' Hk)) nr)) }
        == (abs (psum (fun k ⇒ abs (a k)) ((n ∸ nr) + nr) + - psum (fun k ⇒ abs (a k)) (n ∸ nr)))
             { (ConstructiveReals.abs_proper _ _ Hwindow) }
        ≼ (cast eps2)
             { (Hn1 ((n ∸ nr) + nr) (n ∸ nr)
                  (Theory.NatOrder.le_trans n1 (n ∸ nr) ((n ∸ nr) + nr) Hn1p (Theory.NatOrder.le_add_r (n ∸ nr) nr)) Hn1p) }. }
    assert (Habove : psum (fun k ⇒ abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k)))) nr ≼ cast (half eps)).
    { assert (Hterm : ∀ k : N,
          abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k))) ≼ abs (a ((n ∸ nr) + k)) × cast Mr)
        by (intros k; exact (ConstructiveReals.mul_le_mono_l _ _ _ (Theory.ConstructiveReals.abs_nonneg _) (HrM (n ∸ ((n ∸ nr) + k))))).
      assert (Hf : Proper (equ ==> equ) (fun k : N ⇒ abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k))))).
      { intros k k' Hk. apply Magma.op_proper; [ exact (Hshiftp k k' Hk) | ].
        apply ConstructiveReals.abs_proper. apply r_proper.
        exact (monus_index_proper n _ _ (Magma.op_proper (n ∸ nr) (n ∸ nr) (reflexivity (n ∸ nr)) k k' Hk)). }
      assert (Hg : Proper (equ ==> equ) (fun k : N ⇒ abs (a ((n ∸ nr) + k)) × cast Mr))
        by (intros k k' Hk; apply Magma.op_proper; [ exact (Hshiftp k k' Hk) | reflexivity ]).
      calc (psum (fun k ⇒ abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k)))) nr)
        ≼ (psum (fun k ⇒ abs (a ((n ∸ nr) + k)) × cast Mr) nr) { (psum_le_mono _ _ Hf Hg Hterm nr) }
        == (psum (fun k ⇒ abs (a ((n ∸ nr) + k))) nr × cast Mr) { (psum_mul_r _ (cast Mr) Hshiftp nr) }
        ≼ (cast eps2 × cast Mr)
             { (Theory.ConstructiveReals.mul_le_mono_r _ _ _ (cast_nonneg _ (Theory.Order.le_of_lt _ _ HMr)) Hwin) }
        == (cast (eps2 × Mr)) { (additive (f := cast) (opA := mult) eps2 Mr) }
        == (cast (half eps))
             { (RingHom.hom_proper (h := cast) _ _
                  (transitivity (commutative (op := mult) eps2 Mr) (mul_mul_inv_cancel_left (half eps) Mr HneMr))) }. }
    
    assert (Hsplit : psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) n
      == psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) (n ∸ nr)
         + psum (fun k ⇒ abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k)))) nr).
    { calc (psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) n)
        == (psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) ((n ∸ nr) + nr))
             { (psum_proper _ (errabs_proper n) n ((n ∸ nr) + nr) (symmetry (monus_add_cancel n nr Hnrn))) }
        == (psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) (n ∸ nr)
              + psum (fun k ⇒ abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k)))) nr)
             { (psum_shift _ (errabs_proper n) (n ∸ nr) nr) }. }
    calc (abs (psum (fun j ⇒ a j × r (n ∸ j)) n))
      ≼ (psum (fun j ⇒ abs (a j × r (n ∸ j))) n) { (abs_psum_le_self _ (err_proper n) n) }
      == (psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) n)
           { (psum_ext _ _ (fun j j' Hj ⇒ ConstructiveReals.abs_proper _ _ (err_proper n j j' Hj)) (errabs_proper n)
                (fun j ⇒ ConstructiveReals.abs_mul (a j) (r (n ∸ j))) n) }
      == (psum (fun j ⇒ abs (a j) × abs (r (n ∸ j))) (n ∸ nr)
            + psum (fun k ⇒ abs (a ((n ∸ nr) + k)) × abs (r (n ∸ ((n ∸ nr) + k)))) nr) { Hsplit }
      ≼ (cast (half eps) + cast (half eps)) { (add_le_mono _ _ _ _ Hbelow Habove) }
      == (cast eps) { (cast_half eps) }.
  Defined.

Mertens' theorem.
  Theorem mertens :
    converges (fun n ⇒ psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) n) (La × Lb).
  Proof.
    assert (Hrb : ∀ m : N, psum b m == Lb + r m).
    { intros m. symmetry.
      calc (Lb + (psum b m + - Lb))
        == ((psum b m + - Lb) + Lb) { (commutative (op := add) Lb (psum b m + - Lb)) }
        == (psum b m + (- Lb + Lb)) { (associative (op := add) (action := add) (psum b m) (- Lb) Lb) }
        == (psum b m + 0) { (left_inverse (op := add) Lb) }
        == (psum b m) { (right_identity (action := add) (psum b m)) }. }
    assert (Hdecomp : ∀ n : N,
      psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) n
        == psum a n × Lb + psum (fun j ⇒ a j × r (n ∸ j)) n).
    { intros n.
      calc (psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) n)
        == (psum (fun j ⇒ a j × psum b (n ∸ j)) n)
             { (cauchy_product_exchange a b Ha Hb n) }
        == (psum (fun j ⇒ a j × Lb + a j × r (n ∸ j)) n)
             { (psum_ext _ _
                 (fun j j' Hj ⇒ Magma.op_proper _ _ (Ha j j' Hj) _ _ (psum_proper b Hb _ _ (monus_index_proper n j j' Hj)))
                 (fun j j' Hj ⇒ Magma.op_proper _ _ (Magma.op_proper _ _ (Ha j j' Hj) Lb Lb (reflexivity Lb)) _ _
                    (Magma.op_proper _ _ (Ha j j' Hj) _ _ (r_proper _ _ (monus_index_proper n j j' Hj))))
                 (fun j ⇒ transitivity
                    (Magma.op_proper (a j) (a j) (reflexivity (a j)) _ _ (Hrb (n ∸ j)))
                    (left_distributive (a j) Lb (r (n ∸ j))))
                 n) }
        == (psum (fun j ⇒ a j × Lb) n + psum (fun j ⇒ a j × r (n ∸ j)) n)
             { (psum_add _ _
                 (fun j j' Hj ⇒ Magma.op_proper _ _ (Ha j j' Hj) Lb Lb (reflexivity Lb))
                 (err_proper n) n) }
        == (psum a n × Lb + psum (fun j ⇒ a j × r (n ∸ j)) n)
             { (Magma.op_proper _ _ (psum_mul_r a Lb Ha n) _ _ (reflexivity _)) }. }
    intros eps Heps.
    destruct (converges_mul (fun n ⇒ psum a n) (fun _ ⇒ Lb) La Lb Hca (converges_const Lb) eps Heps) as [na Hna].
    destruct (mertens_error_small eps Heps) as [n2 Hn2].
    destruct (converges_mul (fun n ⇒ psum a n) (fun _ ⇒ Lb) La Lb Hca (converges_const Lb)
                (half eps) (Theory.Order.half_pos eps Heps)) as [nb Hnb].
    destruct (mertens_error_small (half eps) (Theory.Order.half_pos eps Heps)) as [nc Hnc].
    ∃ (nb + nc). intros i Hi.
    assert (Hib : nb ≤ i) by exact (le_of_add_le_l nb nc i Hi).
    assert (Hic : nc ≤ i) by exact (le_of_add_le_r nb nc i Hi).
    apply (le_equ_l (abs ((psum a i × Lb + - (La × Lb)) + psum (fun j ⇒ a j × r (i ∸ j)) i))).
    { apply ConstructiveReals.abs_proper.
      calc ((psum a i × Lb + - (La × Lb)) + psum (fun j ⇒ a j × r (i ∸ j)) i)
        == (psum a i × Lb + (- (La × Lb) + psum (fun j ⇒ a j × r (i ∸ j)) i))
             { (associative (op := add) (action := add) (psum a i × Lb) (- (La × Lb)) (psum (fun j ⇒ a j × r (i ∸ j)) i)) }
        == (psum a i × Lb + (psum (fun j ⇒ a j × r (i ∸ j)) i + - (La × Lb)))
             { (Magma.op_proper _ _ (reflexivity (psum a i × Lb)) _ _ (commutative (op := add) (- (La × Lb)) (psum (fun j ⇒ a j × r (i ∸ j)) i))) }
        == ((psum a i × Lb + psum (fun j ⇒ a j × r (i ∸ j)) i) + - (La × Lb))
             { (associative (op := add) (action := add) (psum a i × Lb) (psum (fun j ⇒ a j × r (i ∸ j)) i) (- (La × Lb))) }
        == (psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) i + - (La × Lb))
             { (Magma.op_proper _ _ (symmetry (Hdecomp i)) _ _ (reflexivity (- (La × Lb)))) }. }
    apply (le_trans _ (abs (psum a i × Lb + - (La × Lb)) + abs (psum (fun j ⇒ a j × r (i ∸ j)) i)));
      [ exact (abs_triangle _ _) | ].
    apply (le_equ_r _ (cast (half eps) + cast (half eps))); [ exact (cast_half eps) | ].
    exact (add_le_mono _ _ _ _ (Hnb i Hib) (Hnc i Hic)).
  Defined.
End Mertens.
