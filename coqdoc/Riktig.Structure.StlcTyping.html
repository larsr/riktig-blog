Library Riktig.Structure.StlcTyping

From Riktig.Structure Require Export StlcCalculus NatOrder.

Typing for the simply typed calculus: a context G lists the types of the variables in scope -- looked up by index, extended under a binder, empty at the top, its length the bound -- and the three rules type each constructor. The laws a typing must satisfy beyond the rules: it respects equality of types, reduction cannot break it, and a closed term of arrow type reduces to an abstraction. Those last two are proven by each instantiation, as the calculus's confluence is.
Module StlcTyping.
  Class StlcTyping (N Ty T Sb G : Type) `{StlcCalculus N Ty T Sb}
      {leN : HasLe N} `{!NatOrder N leN} {arrow_ : HasArrow Ty}
      (lookup_ : G → N → Ty) (extend_ : Ty → G → G) (empty_ : G) (bound_ : G → N)
      (typing : G → T → Ty → Prop) := {
    typing_proper : ∀ (g : G) (t : T) (A A' : Ty),
      A == A' → typing g t A → typing g t A';
    
    typed_var : ∀ (g : G) (n : N) (A : Ty),
      succ n ≤ bound_ g → lookup_ g n == A → typing g (var n) A;
    typed_tylam : ∀ (g : G) (A C : Ty) (b : T),
      typing (extend_ A g) b C → typing g (tylam A b) (A --> C);
    typed_app : ∀ (g : G) (A C : Ty) (f a : T),
      typing g f (A --> C) → typing g a A → typing g (app f a) C;
    
    lookup_extend_zero : ∀ (g : G) (A : Ty), lookup_ (extend_ A g) 0 == A;
    lookup_extend_succ : ∀ (g : G) (A : Ty) (n : N),
      lookup_ (extend_ A g) (succ n) == lookup_ g n;
    bound_extend : ∀ (g : G) (A : Ty), bound_ (extend_ A g) == succ (bound_ g);
    bound_empty : bound_ empty_ == 0;
    
    preservation : ∀ (g : G) (t t' : T) (A : Ty),
      typing g t A → t -->* t' → typing g t' A;
    
    typed_reduces : ∀ (t : T) (A C : Ty),
      typing empty_ t (A --> C) → ∃ (A' : Ty) (b : T), t -->* tylam A' b;
  }.
End StlcTyping.
