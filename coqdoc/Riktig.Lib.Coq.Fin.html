Library Riktig.Lib.Coq.Fin

From Coq Require Import Arith Lia.
From Coq.Vectors Require Import Fin.

Coq's Fin.t (S n) numbered by nat: fin_index reads an element as its number, fin_elem takes the element at a number, sending numbers past the last to the first.
Definition fin_index {n} (i : Fin.t (S n)) : nat := proj1_sig (Fin.to_nat i).

Definition fin_elem {n} (k : nat) : Fin.t (S n) :=
  match lt_dec k (S n) with
  | left H ⇒ Fin.of_nat_lt H
  | right _ ⇒ Fin.F1
  end.

Lemma fin_index_lt : ∀ n (i : Fin.t (S n)), fin_index i ≤ S n ∧ fin_index i ≠ S n.
Proof. intros n i. unfold fin_index. destruct (Fin.to_nat i) as [k Hk]. simpl. lia. Qed.

Lemma fin_elem_index : ∀ n (i : Fin.t (S n)), fin_elem (fin_index i) = i.
Proof.
  intros n i. unfold fin_elem, fin_index.
  destruct (lt_dec (proj1_sig (Fin.to_nat i)) (S n)) as [H | H].
  - rewrite (Fin.of_nat_ext H (proj2_sig (Fin.to_nat i))). apply Fin.of_nat_to_nat_inv.
  - exfalso. apply H. exact (proj2_sig (Fin.to_nat i)).
Qed.

Lemma fin_index_elem : ∀ n (k : nat), k ≤ S n ∧ k ≠ S n →
  fin_index (fin_elem k : Fin.t (S n)) = k.
Proof.
  intros n k [Hle Hne]. unfold fin_elem, fin_index.
  destruct (lt_dec k (S n)) as [H | H]; [ | lia ].
  rewrite Fin.to_nat_of_nat. reflexivity.
Qed.
