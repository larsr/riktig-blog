The sign the reversal applies at grade k: false for fixed, true for negated.
Module RevSign.
  Class RevSign (rev_sign : nat → bool) := {
    rev_sign_zero : rev_sign 0%nat = false;
    rev_sign_succ : ∀ k : nat, rev_sign (S k) = xorb (Nat.odd k) (rev_sign k);
  }.
  Hint Mode RevSign ! : typeclass_instances.
End RevSign.
Export RevSign.

A reversal that acts grade by grade, through that sign.
Module GradedReversal.
  Class GradedReversal (A : Type) {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A}
      {An : HasNeg A} {Ao : One A} {Am : HasMult A} {AR : Ring A}
      {Ag : HasGrade A}
      (rev : A → A) {RV : Reversal A rev}
      (rev_sign : nat → bool) {RS : RevSign rev_sign} :=
    rev_grade : ∀ (k : nat) (a : A),
      rev (⟨ a ⟩_ k) == (if rev_sign k then - ⟨ a ⟩_ k else ⟨ a ⟩_ k).
  Hint Mode GradedReversal ! - - - - - - - - ! - ! - : typeclass_instances.
End GradedReversal.
Export GradedReversal.
