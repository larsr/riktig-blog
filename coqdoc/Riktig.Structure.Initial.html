From Riktig.Structure Require Export Category.

An initial object sends exactly one arrow to every object.
Module Initial.
  Class Initial (O : Type) `{Category O} (initial : HasInit O) (from_initial : HasFromInit O) := {
    from_init_unique : ∀ (x : O) (f : init ~> x), f == from_init x;
  }.
  Hint Mode Initial ! - - - - - - - : typeclass_instances.
End Initial.
Export Initial.
