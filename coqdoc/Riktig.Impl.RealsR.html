  {| Monoid.semigroup := R_add_Semigroup;

  {| Group.monoid := R_add_Monoid; Group.neg_proper := f_equal Ropp;

Instance R_AbGroup : AbGroup R R_Zero R_HasAdd R_HasNeg :=
  {| AbGroup.group := R_Group; AbGroup.is_commutative := Rplus_comm |}.

Instance R_mul_Magma : Magma R R_HasMult :=
