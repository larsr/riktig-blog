From Riktig Require Export Structure.Finite Impl.NatOrderNat Lib.Coq.Two.

The two-element index two is finite of count two, numbered by nat.
Instance Two_EqRel : EqRel two := eq.

Instance Two_HasCard : HasCard two nat := 2.
Instance Two_HasIndex : HasIndex two nat := two_index.
Instance Two_HasElem : HasElem two nat := two_elem.

Instance Two_Finite : Finite two nat Two_HasCard Two_HasIndex Two_HasElem (eqI := eq) :=
  {| Finite.equivalence := eq_equivalence;
     Finite.index_proper := f_equal two_index;
     Finite.elem_proper := f_equal two_elem;
     Finite.index_lt := two_index_lt;
     Finite.elem_index := two_elem_index;
     Finite.index_elem := two_index_elem |}.
