What holds of any binary numbers: a number is zero, a zero bit on a nonzero number, or a one bit. And of any binary arithmetic: the value of a number in any naturals -- doubling as the bits go by -- respects successor, addition and multiplication, is faithful and onto, which is what makes a binary arithmetic a naturals (Impl/NaturalsBinary).
Section BinaryNumberTheory.
  Context {N : Type} `{BinaryNumber N}.
  Lemma bit0_not_zero : ∀ n : N, n =/= 0 → bit0 n =/= 0.
  Proof.
    intros n Hn Hb. apply Hn. apply (injective (f := bit0)).
    calc (bit0 n) == (0 : N) { Hb } == (bit0 (0 : N)) { BinaryNumber.bit0_zero }.
  Qed.

  Lemma bit0_zero_of : ∀ n : N, n == 0 → bit0 n == n.
  Proof. intros n Hz. calc (bit0 n) == (bit0 (0 : N)) { Hz } == (0 : N) { BinaryNumber.bit0_zero } == n { Hz }. Qed.

  Theorem zero_or_not : ∀ n : N, n == 0 ∨ n =/= 0.
  Proof.
    apply (BinaryNumber.bin_induction (fun n ⇒ n == 0 ∨ n =/= 0)).
    - intros n n' Hn.
      exact (or_iff_morphism _ _ (equiv_iff Hn (reflexivity 0)) _ _ (not_iff_morphism _ _ (equiv_iff Hn (reflexivity 0)))).
    - left. reflexivity.
    - intros n [Hz | Hn].
      + left. calc (bit0 n) == (bit0 (0 : N)) { Hz } == (0 : N) { BinaryNumber.bit0_zero }.
      + right. exact (bit0_not_zero n Hn).
    - intros n _. right. exact (BinaryNumber.bit1_not_zero n).
  Qed.

Every number has one of three shapes.
  Theorem bin_cases : ∀ n : N,
    n == 0 ∨ (∃ m : N, m =/= 0 ∧ n == bit0 m) ∨ (∃ m : N, n == bit1 m).
  Proof.
    intros n. destruct (zero_or_not n) as [Hz | Hn]; [ left; exact Hz | right ].
    revert Hn.
    apply (BinaryNumber.bin_induction (fun n ⇒ n =/= 0 → (∃ m : N, m =/= 0 ∧ n == bit0 m) ∨ (∃ m : N, n == bit1 m))).
    - intros k k' Hk.
      exact (iff_iff_iff_impl_morphism _ _ (not_iff_morphism _ _ (equiv_iff Hk (reflexivity 0))) _ _
               (or_iff_morphism _ _ (ex_iff_morphism _ _ (fun m ⇒ and_iff_morphism _ _ (iff_refl _) _ _ (equiv_iff Hk (reflexivity (bit0 m)))))
                                _ _ (ex_iff_morphism _ _ (fun m ⇒ equiv_iff Hk (reflexivity (bit1 m)))))).
    - intros Hz. destruct (Hz (reflexivity 0)).
    - intros k _ Hk. left. ∃ k. split; [ | reflexivity ].
      intros Hz. apply Hk. calc (bit0 k) == (bit0 (0 : N)) { Hz } == (0 : N) { BinaryNumber.bit0_zero }.
    - intros k _ _. right. ∃ k. reflexivity.
  Qed.

The value of a number in any naturals

  Section Value.
    Context {binrecN : HasBinRec N} {BR : BinRec N binrecN}
      {succN : HasSucc N} {oneN : One N} {addN : HasAdd N} {mulN : HasMult N}
      {BA : BinaryArith N 0 bit0 bit1 succ 1 add mult}.
    Context {N0 : Type} `{Naturals N0}.
    Local Notation val n := (binrec N0 (0 : N0) (fun _ v ⇒ v + v) (fun _ v ⇒ v + v + 1) n).

    #[export] Instance bin_val_proper : Proper (equ ==> equ) (fun n : N ⇒ val n).
    Proof.
      exact (BinRec.binrec_proper N0 _ _ 0 _ _ (fun _ _ _ v v' Hv ⇒ Magma.op_proper v v' Hv v v' Hv)
               (fun _ _ _ v v' Hv ⇒ Magma.op_proper _ _ (Magma.op_proper v v' Hv v v' Hv) 1 1 (reflexivity 1))).
    Qed.

    Lemma bin_val_zero : val (0 : N) == (0 : N0).
    Proof. exact (BinRec.binrec_zero N0 _ _ 0 _ _). Qed.

    Lemma bin_val_zero_of : ∀ n : N, n == 0 → val n == (0 : N0).
    Proof. intros n Hz. calc (val n) == (val (0 : N)) { Hz } == (0 : N0) { bin_val_zero }. Qed.

    Lemma bin_val_bit1 : ∀ n : N, val (bit1 n) == val n + val n + 1.
    Proof. intros n. exact (BinRec.binrec_bit1 N0 _ _ 0 _ _ n). Qed.

    Lemma bin_val_bit0 : ∀ n : N, val (bit0 n) == val n + val n.
    Proof.
      intros n. destruct (zero_or_not n) as [Hz | Hn].
      - calc (val (bit0 n))
          == (val n) { (bit0_zero_of n Hz) }
          == (0 : N0) { (bin_val_zero_of n Hz) }
          == ((0 : N0) + 0) { (left_identity (action := add) (0 : N0)) }
          == (val n + val n) { (bin_val_zero_of n Hz) }.
      - exact (BinRec.binrec_bit0 N0 _ _ 0 _ _ n Hn).
    Qed.

    Lemma bin_val_one : val (bit1 (0 : N)) == (1 : N0).
    Proof.
      calc (val (bit1 (0 : N)))
        == (val (0 : N) + val (0 : N) + 1) { (bin_val_bit1 0) }
        == ((0 : N0) + 0 + 1) { bin_val_zero }
        == ((0 : N0) + 1) { (left_identity (action := add) (0 : N0)) }
        == (1 : N0) { (left_identity (action := add) (1 : N0)) }.
    Qed.

    Theorem bin_val_succ : ∀ n : N, val (succ n) == succ (val n).
    Proof.
      apply (BinaryNumber.bin_induction (fun n ⇒ val (succ n) == succ (val n))).
      - intros n n' Hn. exact (equiv_iff (bin_val_proper _ _ (BinaryArith.succ_proper _ _ Hn)) (Peano.succ_proper _ _ (bin_val_proper _ _ Hn))).
      - calc (val (succ (0 : N)))
          == (val (bit1 (0 : N))) { BinaryArith.succ_zero }
          == (1 : N0) { bin_val_one }
          == (succ (0 : N0)) { one_succ }
          == (succ (val (0 : N))) { bin_val_zero }.
      - intros n IH. destruct (zero_or_not n) as [Hz | Hn].
        + calc (val (succ (bit0 n)))
            == (val (succ n)) { (bit0_zero_of n Hz) }
            == (succ (val n)) { IH }
            == (succ (val (bit0 n))) { (bit0_zero_of n Hz) }.
        + calc (val (succ (bit0 n)))
            == (val (bit1 n)) { (BinaryArith.succ_bit0 n Hn) }
            == (val n + val n + 1) { (bin_val_bit1 n) }
            == (succ (val n + val n)) { (Naturals.succ_add (val n + val n)) }
            == (succ (val (bit0 n))) { (bin_val_bit0 n) }.
      - intros n IH.
        calc (val (succ (bit1 n)))
          == (val (bit0 (succ n))) { (BinaryArith.succ_bit1 n) }
          == (val (succ n) + val (succ n)) { (bin_val_bit0 (succ n)) }
          == (succ (val n) + succ (val n)) { IH }
          == (succ (succ (val n + val n))) { (add_succ_succ (val n) (val n)) }
          == (succ (val n + val n + 1)) { (Naturals.succ_add (val n + val n)) }
          == (succ (val (bit1 n))) { (bin_val_bit1 n) }.
    Qed.

    Theorem bin_val_add : ∀ x y : N, val (x + y) == val x + val y.
    Proof.
      apply (BinaryNumber.bin_induction (fun x ⇒ ∀ y : N, val (x + y) == val x + val y)).
      - intros x x' Hx.
        exact (all_iff_morphism _ _ (fun y ⇒ equiv_iff (bin_val_proper _ _ (BinaryArith.add_proper x x' Hx y y (reflexivity y)))
                                                        (Magma.op_proper _ _ (bin_val_proper _ _ Hx) _ _ (reflexivity (val y))))).
      - intros y.
        calc (val ((0 : N) + y))
          == (val y) { (BinaryArith.add_zero_l y) }
          == ((0 : N0) + val y) { (left_identity (action := add) (val y)) }
          == (val (0 : N) + val y) { bin_val_zero }.
      - intros n IH y. destruct (zero_or_not n) as [Hz | Hn].
        + calc (val (bit0 n + y))
            == (val (n + y)) { (bit0_zero_of n Hz) }
            == (val n + val y) { (IH y) }
            == (val (bit0 n) + val y) { (bit0_zero_of n Hz) }.
        + destruct (bin_cases y) as [Hy | [[m [Hm Hy]] | [m Hy]]]; rewrite Hy.
          × calc (val (bit0 n + 0))
              == (val (bit0 n)) { (BinaryArith.add_bit0_zero n Hn) }
              == (val (bit0 n) + 0) { (right_identity (action := add) (val (bit0 n))) }
              == (val (bit0 n) + val (0 : N)) { bin_val_zero }.
          × calc (val (bit0 n + bit0 m))
              == (val (bit0 (n + m))) { (BinaryArith.add_bit0_bit0 n m Hn Hm) }
              == (val (n + m) + val (n + m)) { (bin_val_bit0 (n + m)) }
              == ((val n + val m) + (val n + val m)) { (IH m) }
              == ((val n + val n) + (val m + val m)) { (add_add_add_comm (val n) (val m) (val n) (val m)) }
              == (val (bit0 n) + (val m + val m)) { (bin_val_bit0 n) }
              == (val (bit0 n) + val (bit0 m)) { (bin_val_bit0 m) }.
          × calc (val (bit0 n + bit1 m))
              == (val (bit1 (n + m))) { (BinaryArith.add_bit0_bit1 n m Hn) }
              == (val (n + m) + val (n + m) + 1) { (bin_val_bit1 (n + m)) }
              == ((val n + val m) + (val n + val m) + 1) { (IH m) }
              == ((val n + val n) + (val m + val m) + 1) { (add_add_add_comm (val n) (val m) (val n) (val m)) }
              == ((val n + val n) + (val m + val m + 1))
                   { (associative (op := add) (action := add) (val n + val n) (val m + val m) 1) }
              == (val (bit0 n) + (val m + val m + 1)) { (bin_val_bit0 n) }
              == (val (bit0 n) + val (bit1 m)) { (bin_val_bit1 m) }.
      - intros n IH y. destruct (bin_cases y) as [Hy | [[m [Hm Hy]] | [m Hy]]]; rewrite Hy.
        + calc (val (bit1 n + 0))
            == (val (bit1 n)) { (BinaryArith.add_bit1_zero n) }
            == (val (bit1 n) + 0) { (right_identity (action := add) (val (bit1 n))) }
            == (val (bit1 n) + val (0 : N)) { bin_val_zero }.
        + calc (val (bit1 n + bit0 m))
            == (val (bit1 (n + m))) { (BinaryArith.add_bit1_bit0 n m Hm) }
            == (val (n + m) + val (n + m) + 1) { (bin_val_bit1 (n + m)) }
            == ((val n + val m) + (val n + val m) + 1) { (IH m) }
            == ((val n + val n) + (val m + val m) + 1) { (add_add_add_comm (val n) (val m) (val n) (val m)) }
            == ((val n + val n + 1) + (val m + val m)) { (add_right_comm (val n + val n) (val m + val m) 1) }
            == (val (bit1 n) + (val m + val m)) { (bin_val_bit1 n) }
            == (val (bit1 n) + val (bit0 m)) { (bin_val_bit0 m) }.
        + calc (val (bit1 n + bit1 m))
            == (val (bit0 (succ (n + m)))) { (BinaryArith.add_bit1_bit1 n m) }
            == (val (succ (n + m)) + val (succ (n + m))) { (bin_val_bit0 (succ (n + m))) }
            == (succ (val (n + m)) + succ (val (n + m))) { (bin_val_succ (n + m)) }
            == (succ (succ (val (n + m) + val (n + m)))) { (add_succ_succ (val (n + m)) (val (n + m))) }
            == (succ (succ ((val n + val m) + (val n + val m)))) { (IH m) }
            == ((val n + val n + 1) + (val m + val m + 1)) { (succ_succ_double (val n) (val m)) }
            == (val (bit1 n) + val (bit1 m)) { (Magma.op_proper _ _ (bin_val_bit1 n) _ _ (bin_val_bit1 m)) }.
    Qed.

    Theorem bin_val_mult : ∀ x y : N, val (x × y) == val x × val y.
    Proof.
      apply (BinaryNumber.bin_induction (fun x ⇒ ∀ y : N, val (x × y) == val x × val y)).
      - intros x x' Hx.
        exact (all_iff_morphism _ _ (fun y ⇒ equiv_iff (bin_val_proper _ _ (BinaryArith.mul_proper x x' Hx y y (reflexivity y)))
                                                        (Magma.op_proper _ _ (bin_val_proper _ _ Hx) _ _ (reflexivity (val y))))).
      - intros y.
        calc (val ((0 : N) × y))
          == (val (0 : N)) { (BinaryArith.mul_zero_l y) }
          == (0 : N0) { bin_val_zero }
          == ((0 : N0) × val y) { (left_absorbing (val y)) }
          == (val (0 : N) × val y) { bin_val_zero }.
      - intros n IH y. destruct (zero_or_not n) as [Hz | Hn].
        + calc (val (bit0 n × y))
            == (val (n × y)) { (bit0_zero_of n Hz) }
            == (val n × val y) { (IH y) }
            == (val (bit0 n) × val y) { (bit0_zero_of n Hz) }.
        + calc (val (bit0 n × y))
            == (val (bit0 (n × y))) { (BinaryArith.mul_bit0 n y Hn) }
            == (val (n × y) + val (n × y)) { (bin_val_bit0 (n × y)) }
            == (val n × val y + val n × val y) { (IH y) }
            == ((val n + val n) × val y) { (right_distributive (val n) (val n) (val y)) }
            == (val (bit0 n) × val y) { (bin_val_bit0 n) }.
      - intros n IH y.
        calc (val (bit1 n × y))
          == (val (bit0 (n × y) + y)) { (BinaryArith.mul_bit1 n y) }
          == (val (bit0 (n × y)) + val y) { (bin_val_add (bit0 (n × y)) y) }
          == (val (n × y) + val (n × y) + val y) { (bin_val_bit0 (n × y)) }
          == (val n × val y + val n × val y + val y) { (IH y) }
          == ((val n + val n) × val y + val y) { (right_distributive (val n) (val n) (val y)) }
          == ((val n + val n + 1) × val y) { (double_succ_mul (val n) (val y)) }
          == (val (bit1 n) × val y) { (bin_val_bit1 n) }.
    Qed.

The value is faithful and onto

    Lemma bin_val_eq_zero : ∀ n : N, val n == 0 → n == 0.
    Proof.
      apply (BinaryNumber.bin_induction (fun n ⇒ val n == 0 → n == 0)).
      - intros n n' Hn.
        exact (iff_iff_iff_impl_morphism _ _ (equiv_iff (bin_val_proper _ _ Hn) (reflexivity 0)) _ _ (equiv_iff Hn (reflexivity 0))).
      - intros _. reflexivity.
      - intros n IH Hv.
        assert (Hz : n == 0).
        { apply IH. apply (eq_zero_of_add_eq_zero (val n) (val n)).
          calc (val n + val n) == (val (bit0 n)) { (bin_val_bit0 n) } == (0 : N0) { Hv }. }
        calc (bit0 n) == n { (bit0_zero_of n Hz) } == (0 : N) { Hz }.
      - intros n _ Hv. exfalso. apply (Peano.succ_not_zero (val n + val n)).
        calc (succ (val n + val n))
          == (val n + val n + 1) { (Naturals.succ_add (val n + val n)) }
          == (val (bit1 n)) { (bin_val_bit1 n) }
          == (0 : N0) { Hv }.
    Qed.

    Lemma bin_val_bit0_bit1 : ∀ n m : N, val (bit0 n) =/= val (bit1 m).
    Proof.
      intros n m Hv. apply (double_ne_succ_double (val n) (val m)).
      calc (val n + val n)
        == (val (bit0 n)) { (bin_val_bit0 n) }
        == (val (bit1 m)) { Hv }
        == (val m + val m + 1) { (bin_val_bit1 m) }
        == (succ (val m + val m)) { (Naturals.succ_add (val m + val m)) }.
    Qed.

    Theorem bin_val_injective : ∀ x y : N, val x == val y → x == y.
    Proof.
      apply (BinaryNumber.bin_induction (fun x ⇒ ∀ y : N, val x == val y → x == y)).
      - intros x x' Hx.
        exact (all_iff_morphism _ _ (fun y ⇒ iff_iff_iff_impl_morphism _ _
                 (equiv_iff (bin_val_proper _ _ Hx) (reflexivity (val y))) _ _ (equiv_iff Hx (reflexivity y)))).
      - intros y Hv. symmetry. apply bin_val_eq_zero.
        calc (val y) == (val (0 : N)) { Hv } == (0 : N0) { bin_val_zero }.
      - intros n IH y Hv. destruct (zero_or_not n) as [Hz | Hn].
        + calc (bit0 n) == n { (bit0_zero_of n Hz) } == y { (IH y (transitivity (symmetry (bin_val_proper _ _ (bit0_zero_of n Hz))) Hv)) }.
        + destruct (bin_cases y) as [Hy | [[m [Hm Hy]] | [m Hy]]].
          × exfalso. apply (bit0_not_zero n Hn). apply bin_val_eq_zero.
            calc (val (bit0 n)) == (val y) { Hv } == (val (0 : N)) { Hy } == (0 : N0) { bin_val_zero }.
          × assert (Hnm : val n + val n == val m + val m).
            { calc (val n + val n) == (val (bit0 n)) { (bin_val_bit0 n) } == (val y) { Hv }
                == (val (bit0 m)) { Hy } == (val m + val m) { (bin_val_bit0 m) }. }
            calc (bit0 n) == (bit0 m) { (IH m (double_injective (val n) (val m) Hnm)) } == y { Hy }.
          × exfalso. apply (bin_val_bit0_bit1 n m).
            calc (val (bit0 n)) == (val y) { Hv } == (val (bit1 m)) { Hy }.
      - intros n IH y Hv. destruct (bin_cases y) as [Hy | [[m [Hm Hy]] | [m Hy]]].
        + exfalso. apply (BinaryNumber.bit1_not_zero n). apply bin_val_eq_zero.
          calc (val (bit1 n)) == (val y) { Hv } == (val (0 : N)) { Hy } == (0 : N0) { bin_val_zero }.
        + exfalso. apply (bin_val_bit0_bit1 m n).
          calc (val (bit0 m)) == (val y) { Hy } == (val (bit1 n)) { Hv }.
        + assert (Hnm : val n + val n == val m + val m).
          { apply (injective (f := succ)).
            calc (succ (val n + val n))
              == (val n + val n + 1) { (Naturals.succ_add (val n + val n)) }
              == (val (bit1 n)) { (bin_val_bit1 n) }
              == (val y) { Hv }
              == (val (bit1 m)) { Hy }
              == (val m + val m + 1) { (bin_val_bit1 m) }
              == (succ (val m + val m)) { (Naturals.succ_add (val m + val m)) }. }
          calc (bit1 n) == (bit1 m) { (IH m (double_injective (val n) (val m) Hnm)) } == y { Hy }.
    Qed.

    Theorem bin_val_onto : ∀ v : N0, ∃ n : N, val n == v.
    Proof.
      apply (Peano.nat_induction (fun v ⇒ ∃ n : N, val n == v)).
      - intros v v' Hv. exact (ex_iff_morphism _ _ (fun n ⇒ equiv_iff (reflexivity (val n)) Hv)).
      - ∃ 0. exact bin_val_zero.
      - intros v [n Hn]. ∃ (succ n).
        calc (val (succ n)) == (succ (val n)) { (bin_val_succ n) } == (succ v) { Hn }.
    Qed.
  End Value.
