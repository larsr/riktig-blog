Definition dual_apply {F V D : Type} `{DualSpace F V D} (d : D) (v : V) : F :=
  VSIso.to DualSpace.represent d v.

Notation "⟨ d , v ⟩" := (dual_apply d v).

Section Pairing.
  Context {F V D : Type} `{DualSpace F V D}.

  #[export] Instance dual_apply_bilinear : Bilinear F D V F dual_apply :=
    BilinearMap.bilinear (uncurry (VSIso.to DualSpace.represent)).

  Theorem dual_apply_linear : ∀ d : D, Linear F V F (dual_apply d).
  Proof. intros d. exact (linear_r (b := dual_apply) d). Qed.

  Theorem dual_apply_linear_l : ∀ v : V,
    Linear F D F (fun d ⇒ ⟨ d , v ⟩).
  Proof. intros v. exact (linear_l (b := dual_apply) v). Qed.

  Theorem dual_apply_add_l : ∀ (d d' : D) (v : V),
    (⟨ d + d' , v ⟩) == (⟨ d , v ⟩) + (⟨ d' , v ⟩).
  Proof.
    intros d d' v. exact (additive[fun x : D ⇒ ⟨ x , v ⟩] d d').
  Qed.

  Theorem dual_apply_scale_l : ∀ (a : F) (d : D) (v : V),
    (⟨ a *: d , v ⟩) == a × (⟨ d , v ⟩).
  Proof.
    intros a d v.
    exact (homogeneous[fun x : D ⇒ ⟨ x , v ⟩] a d).
  Qed.

  Theorem dual_apply_add_r : ∀ (d : D) (v v' : V),
    (⟨ d , v + v' ⟩) == (⟨ d , v ⟩) + (⟨ d , v' ⟩).
  Proof.
    intros d v v'. exact (additive[dual_apply d] v v').
  Qed.

  Theorem dual_apply_scale_r : ∀ (a : F) (d : D) (v : V),
    (⟨ d , a *: v ⟩) == a × (⟨ d , v ⟩).
  Proof.
    intros a d v.
    exact (homogeneous[dual_apply d] a v).
  Qed.

  Theorem dual_apply_nondegenerate : ∀ d : D,
    (∀ v : V, (⟨ d , v ⟩) == 0) → d == 0.
  Proof.
    intros d Hz.
    calc d
      == (VSIso.from DualSpace.represent (VSIso.to DualSpace.represent d))
           { vsiso_from_to }
      == (VSIso.from DualSpace.represent 0)
           { (Linear.proper (f := VSIso.from DualSpace.represent) _ 0 Hz) }
      == (0 : D)
           { (additive_zero (VSIso.from DualSpace.represent)) }.
  Qed.

End Pairing.
