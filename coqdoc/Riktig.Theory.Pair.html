Library Riktig.Theory.Pair

From Riktig.Structure Require Export Pair.
From Riktig.Theory Require Export InitialAlgebra Signature.
From Riktig.Lib Require Import Calc.

Section PairTheory.
  Context {O : Type} `{Bicartesian O} {K : Type} {den : K → O} {a b : K} {P : O}
    {build : HasBuild O (Pair.shape den a b) P} {fold : HasCata O (Pair.shape den a b) P}
    {PP : Pair O den a b P build fold}.

  Theorem cata_mk : ∀ (x : O) (h : (den a × den b)%object ~> x),
    cata (F := shape_of den (pairS (S := sig K) a b)) x h ∘ build == h.
  Proof.
    intros x h. etransitivity.
    - exact (InitialAlgebra.cata_build x h).
    - setoid_rewrite (par_id (O := O)). apply right_identity.
  Qed.

  Theorem first_mk : cata (F := shape_of den (pairS (S := sig K) a b)) (den a) pi1 ∘ build == (pi1 : (den a × den b)%object ~> den a).
  Proof. exact (cata_mk (den a) pi1). Qed.

  Theorem second_mk : cata (F := shape_of den (pairS (S := sig K) a b)) (den b) pi2 ∘ build == (pi2 : (den a × den b)%object ~> den b).
  Proof. exact (cata_mk (den b) pi2). Qed.

Taking the pair apart is the inverse of building it.
  Theorem mk_unmk_pair :
    build ∘ fold ((den a × den b)%object) (fmap (F := shape_of den (pairS (S := sig K) a b)) build) == cat_id.
  Proof. exact build_unbuild. Qed.

  Theorem unmk_mk_pair :
    fold ((den a × den b)%object) (fmap (F := shape_of den (pairS (S := sig K) a b)) build) ∘ build == cat_id.
  Proof. exact unbuild_build. Qed.
End PairTheory.
