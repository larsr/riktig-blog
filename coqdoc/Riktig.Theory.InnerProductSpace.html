From Riktig.Structure Require Export InnerProductSpace.
From Riktig.Theory Require Import Ring AbGroup Field.
From Riktig.Lib Require Import Calc.

The inner product of a pair space whose second coordinate is weighed by a scalar s: bilinear and symmetric, one coordinate at a time.
Section InnerProductSpaceTheory.
  Context {F V : Type} `{InnerProductSpace F V} (s : F).

  Lemma extend_inner_add : ∀ (u v w : V) (b c d : F),
    ⟪ u + v , w ⟫ + s × ((b + c) × d)
      == (⟪ u , w ⟫ + s × (b × d)) + (⟪ v , w ⟫ + s × (c × d)).
  Proof.
    intros u v w b c d.
    calc (⟪ u + v , w ⟫ + s × ((b + c) × d))
      == ((⟪ u , w ⟫ + ⟪ v , w ⟫) + s × ((b + c) × d))
           { (additive (f := fun x ⇒ ⟪ x , w ⟫) u v) }
      == ((⟪ u , w ⟫ + ⟪ v , w ⟫) + s × (b × d + c × d))
           { (right_distributive (op := mult) b c d) }
      == ((⟪ u , w ⟫ + ⟪ v , w ⟫) + (s × (b × d) + s × (c × d)))
           { (left_distributive (op := mult) s (b × d) (c × d)) }
      == ((⟪ u , w ⟫ + s × (b × d)) + (⟪ v , w ⟫ + s × (c × d)))
           { add_add_add_comm }.
  Qed.

  Lemma extend_inner_smul : ∀ (u w : V) (a b d : F),
    ⟪ a *: u , w ⟫ + s × ((a × b) × d) == a × (⟪ u , w ⟫ + s × (b × d)).
  Proof.
    intros u w a b d.
    calc (⟪ a *: u , w ⟫ + s × ((a × b) × d))
      == (a × ⟪ u , w ⟫ + s × ((a × b) × d))
           { (homogeneous (f := fun x ⇒ ⟪ x , w ⟫) a u) }
      == (a × ⟪ u , w ⟫ + s × (a × (b × d)))
           { (associative (op := mult) (action := mult) a b d) }
      == (a × ⟪ u , w ⟫ + a × (s × (b × d)))
           { (mul_left_comm s a (b × d)) }
      == (a × (⟪ u , w ⟫ + s × (b × d)))
           { (left_distributive (op := mult) a ⟪ u , w ⟫ (s × (b × d))) }.
  Qed.

  Lemma extend_inner_comm : ∀ (u v : V) (b c : F),
    ⟪ u , v ⟫ + s × (b × c) == ⟪ v , u ⟫ + s × (c × b).
  Proof.
    intros u v b c.
    calc (⟪ u , v ⟫ + s × (b × c))
      == (⟪ v , u ⟫ + s × (b × c)) { (commutative (op := inner) u v) }
      == (⟪ v , u ⟫ + s × (c × b)) { (commutative (op := mult) b c) }.
  Qed.
End InnerProductSpaceTheory.
