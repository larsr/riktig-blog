Library Riktig.Theory.InnerProductSpace

From Coq Require Export Morphisms.
From Riktig Require Export Interface.InnerProductSpace Theory.AbGroup.

Section InnerProductTheory.
  Context {F V : Type} `{InnerProductSpace F V}.

  Definition inner_left_additive (u v w : V) : ⟨ u + v , w ⟩ == ⟨ u , w ⟩ + ⟨ v , w ⟩ :=
    additive (f := fun x : V ⇒ ⟨ x , w ⟩) u v.

  Definition inner_left_scale (a : F) (u v : V) : ⟨ a *: u , v ⟩ == a × ⟨ u , v ⟩ :=
    homogeneous (f := fun x : V ⇒ ⟨ x , v ⟩) a u.

  Theorem norm2_spec : ∀ u : V, norm2 u == ⟨ u , u ⟩.
  Proof.
    intros u. reflexivity.
  Qed.

  #[export] Instance inner_proper : Proper (equ ==> equ ==> equ) (@inner F V _).
  Proof.
    intros x x' Hx y y' Hy.
    calc (⟨ x , y ⟩)
      == (⟨ x' , y ⟩) { (Linear.proper (f := fun u ⇒ ⟨ u , y ⟩) x x' Hx) }
      == (⟨ y , x' ⟩) { (commutative x' y) }
      == (⟨ y' , x' ⟩) { (Linear.proper (f := fun u ⇒ ⟨ u , x' ⟩) y y' Hy) }
      == (⟨ x' , y' ⟩) { (commutative y' x') }.
  Qed.

  #[export] Instance norm2_proper : Proper (equ ==> equ) (norm2 (F:=F) (V:=V)).
  Proof.
    intros x y Hxy.
    unfold norm2.
    rewrite Hxy.
    reflexivity.
  Qed.

  Theorem inner_right_additive : ∀ u v w : V,
    ⟨ u , v + w ⟩ == ⟨ u , v ⟩ + ⟨ u , w ⟩.
  Proof.
    intros u v w.
    calc (⟨ u , v + w ⟩)
      == (⟨ v + w , u ⟩) { (commutative u (v + w)) }
      == (⟨ v , u ⟩ + ⟨ w , u ⟩) { (inner_left_additive v w u) }
      == (⟨ u , v ⟩ + ⟨ w , u ⟩) { (commutative v u) }
      == (⟨ u , v ⟩ + ⟨ u , w ⟩) { (commutative w u) }.
  Qed.

  Theorem inner_right_scale : ∀ (a : F) (u v : V),
    ⟨ u , a *: v ⟩ == a × ⟨ u , v ⟩.
  Proof.
    intros a u v.
    calc (⟨ u , a *: v ⟩)
      == (⟨ a *: v , u ⟩) { (commutative u (a *: v)) }
      == (a × ⟨ v , u ⟩) { (inner_left_scale a v u) }
      == (a × ⟨ u , v ⟩) { (commutative v u) }.
  Qed.

  #[export] Instance inner_left_proper : ∀ v : V, Proper (equ ==> equ) (fun w : V ⇒ ⟨ w , v ⟩).
  Proof.
    intros v x y Hxy.
    rewrite Hxy.
    reflexivity.
  Qed.

  Theorem inner_left_zero : ∀ u : V, ⟨ (0 : V) , u ⟩ == 0.
  Proof.
    intros u.
    exact (hom_zero (fun w : V ⇒ ⟨ w , u ⟩) (fun x y ⇒ inner_left_additive x y u)).
  Qed.

  Theorem inner_right_zero : ∀ u : V, ⟨ u , (0 : V) ⟩ == 0.
  Proof.
    intros u.
    calc (⟨ u , (0 : V) ⟩)
      == (⟨ (0 : V) , u ⟩) { (commutative u 0) }
      == (0 : F) { (inner_left_zero u) }.
  Qed.

  Theorem inner_left_neg : ∀ u v : V, ⟨ - u , v ⟩ == - ⟨ u , v ⟩.
  Proof.
    intros u v.
    exact (hom_neg (fun w : V ⇒ ⟨ w , v ⟩) (fun x y ⇒ inner_left_additive x y v) u).
  Qed.

  Theorem inner_right_neg : ∀ u v : V, ⟨ u , - v ⟩ == - ⟨ u , v ⟩.
  Proof.
    intros u v.
    calc (⟨ u , - v ⟩)
      == (⟨ - v , u ⟩) { (commutative u (- v)) }
      == (- ⟨ v , u ⟩) { (inner_left_neg v u) }
      == (- ⟨ u , v ⟩) { (commutative v u) }.
  Qed.

  Theorem norm2_zero : norm2 (0 : V) == 0.
  Proof.
    calc (norm2 (0 : V))
      == (⟨ (0 : V) , (0 : V) ⟩) { (norm2_spec 0) }
      == (0 : F) { (inner_left_zero 0) }.
  Qed.

  Theorem inner_expand : ∀ u v : V,
    ⟨ u + v , u + v ⟩ == (⟨ u , u ⟩ + ⟨ v , v ⟩) + (⟨ u , v ⟩ + ⟨ v , u ⟩).
  Proof.
    intros u v.
    calc (⟨ u + v , u + v ⟩)
      == (⟨ u , u + v ⟩ + ⟨ v , u + v ⟩)
        { (inner_left_additive u v (u + v)) }
      == ((⟨ u , u ⟩ + ⟨ u , v ⟩) + ⟨ v , u + v ⟩)
        { (inner_right_additive u u v) }
      == ((⟨ u , u ⟩ + ⟨ u , v ⟩) + (⟨ v , u ⟩ + ⟨ v , v ⟩))
        { (inner_right_additive v u v) }
      == ((⟨ u , u ⟩ + ⟨ v , v ⟩) + (⟨ u , v ⟩ + ⟨ v , u ⟩))
        { (add_rotate ⟨ u , u ⟩ ⟨ u , v ⟩ ⟨ v , u ⟩ ⟨ v , v ⟩) }.
  Qed.

  Theorem norm2_add : ∀ u v : V,
    norm2 (u + v) == (norm2 u + norm2 v) + (⟨ u , v ⟩ + ⟨ v , u ⟩).
  Proof.
    intros u v.
    calc (norm2 (u + v))
      == (⟨ u + v , u + v ⟩)
        { (norm2_spec (u + v)) }
      == ((⟨ u , u ⟩ + ⟨ v , v ⟩) + (⟨ u , v ⟩ + ⟨ v , u ⟩))
        { (inner_expand u v) }
      == ((norm2 u + ⟨ v , v ⟩) + (⟨ u , v ⟩ + ⟨ v , u ⟩))
        { (norm2_spec u) }
      == ((norm2 u + norm2 v) + (⟨ u , v ⟩ + ⟨ v , u ⟩))
        { (norm2_spec v) }.
  Qed.

  Theorem norm2_scale : ∀ (a : F) (u : V),
    norm2 (a *: u) == (a × a) × norm2 u.
  Proof.
    intros a u.
    calc (norm2 (a *: u))
      == (⟨ a *: u , a *: u ⟩) { (norm2_spec (a *: u)) }
      == (a × ⟨ u , a *: u ⟩) { (inner_left_scale a u (a *: u)) }
      == (a × (a × ⟨ u , u ⟩)) { (inner_right_scale a u u) }
      == ((a × a) × ⟨ u , u ⟩) { (associative a a ⟨ u , u ⟩) }
      == ((a × a) × norm2 u) { (norm2_spec u) }.
  Qed.

End InnerProductTheory.
