The simply typed lambda calculus over named base types: the types, the annotated terms, their erasure into the untyped calculus, the typing relation, and the checker with its soundness.
Inductive sty : Type :=
  | SBase (x : string)
  | SArrow (a b : sty).

Definition sty_eq_dec : ∀ a b : sty, {a = b} + {a ≠ b}.
Proof. decide equality; apply string_dec. Defined.

Fixpoint sty_eqb (a b : sty) : bool :=
  match a, b with
  | SBase x, SBase y ⇒ String.eqb x y
  | SArrow a1 c1, SArrow a2 c2 ⇒ sty_eqb a1 a2 && sty_eqb c1 c2
  | _, _ ⇒ false
  end.

Lemma sty_eqb_eq : ∀ a b : sty, sty_eqb a b = true ↔ a = b.
Proof.
  intros a. induction a as [x | a1 IH1 c1 IH2]; intros [y | a2 c2]; simpl.
  - rewrite String.eqb_eq. split; intros H; [ subst; reflexivity | injection H as H; subst; reflexivity ].
  - split; discriminate.
  - split; discriminate.
  - rewrite andb_true_iff, IH1, IH2. split.
    + intros [H1 H2]. subst. reflexivity.
    + intros H. injection H as H1 H2. subst. split; reflexivity.
Qed.

Annotated terms: the binder carries its type, so checking is syntax-directed.
Inductive tterm : Type :=
  | TVar (n : nat)
  | TLam (A : sty) (b : tterm)
  | TApp (f a : tterm).

Fixpoint erase (t : tterm) : term :=
  match t with
  | TVar n ⇒ Var n
  | TLam A b ⇒ Lam (erase b)
  | TApp f a ⇒ App (erase f) (erase a)
  end.

The typing relation, on the untyped terms, over total contexts: every index has a type.
Definition tctx : Type := nat → sty.

Inductive has_ty : tctx → term → sty → Prop :=
  | ty_var : ∀ (G : tctx) (n : nat), has_ty G (Var n) (G n)
  | ty_lam : ∀ (G : tctx) (A C : sty) (b : term),
      has_ty (scons A G) b C → has_ty G (Lam b) (SArrow A C)
  | ty_app : ∀ (G : tctx) (A C : sty) (f a : term),
      has_ty G f (SArrow A C) → has_ty G a A → has_ty G (App f a) C.

A finite context, read totally: a default base type past the end.
Definition ctx_of (G : list sty) : tctx := fun n ⇒ nth n G (SBase "?").

The checker: infer the type of an annotated term.
Fixpoint infer (G : list sty) (t : tterm) : option sty :=
  match t with
  | TVar n ⇒ nth_error G n
  | TLam A b ⇒ match infer (A :: G) b with
                | Some C ⇒ Some (SArrow A C)
                | None ⇒ None
                end
  | TApp f a ⇒ match infer G f, infer G a with
                | Some (SArrow A C), Some A' ⇒ if sty_eqb A A' then Some C else None
                | _, _ ⇒ None
                end
  end.

Theorem infer_sound : ∀ (t : tterm) (G : list sty) (A : sty),
  infer G t = Some A → has_ty (ctx_of G) (erase t) A.
Proof.
  intros t. induction t as [n | A b IH | f IHf a IHa]; intros G C H; simpl in ×.
  - rewrite <- (nth_error_nth G n (SBase "?") H). apply ty_var.
  - destruct (infer (A :: G) b) as [C0 |] eqn:Hb; [ | discriminate ].
    injection H as H. subst. apply ty_lam.
    assert (Hc : ∀ n, scons A (ctx_of G) n = ctx_of (A :: G) n) by (intros [| k]; reflexivity).
    revert Hc. generalize (scons A (ctx_of G)). generalize (ctx_of (A :: G)) (IH _ _ Hb).
    clear. intros G1 H1 G2 Hc.
    assert (Hfun : G2 = G1) by (apply FunctionalExtensionality.functional_extensionality; intros k; apply Hc).
    rewrite Hfun. exact H1.
  - destruct (infer G f) as [[x | A0 C0] |] eqn:Hf; try discriminate.
    destruct (infer G a) as [A' |] eqn:Ha; [ | discriminate ].
    destruct (sty_eqb A0 A') eqn:He; [ | discriminate ].
    injection H as H. subst.
    apply sty_eqb_eq in He. subst.
    exact (ty_app (ctx_of G) A' C (erase f) (erase a) (IHf _ _ Hf) (IHa _ _ Ha)).
Qed.

Annotated terms with names, as written, and their resolution.
Inductive tnamed : Type :=
  | TNVar (x : string)
  | TNLam (x : string) (A : sty) (b : tnamed)
  | TNApp (f a : tnamed).

Fixpoint tresolve (bs : list string) (t : tnamed) : tterm :=
  match t with
  | TNVar x ⇒ TVar (match index_of x bs with Some n ⇒ n | None ⇒ O end)
  | TNLam x A b ⇒ TLam A (tresolve (x :: bs) b)
  | TNApp f a ⇒ TApp (tresolve bs f) (tresolve bs a)
  end.

Subject reduction

Typing survives renaming, substitution by well-typed terms, and therefore reduction.
Lemma has_ty_ren : ∀ (G : tctx) (t : term) (A : sty), has_ty G t A →
  ∀ (G' : tctx) (r : nat → nat), (∀ n : nat, G' (r n) = G n) → has_ty G' (ren r t) A.
Proof.
  intros G t A H.
  induction H as [G n | G A C b Hb IH | G A C f a Hf IHf Ha IHa]; intros G' r Hr; simpl.
  - rewrite <- (Hr n). apply ty_var.
  - apply ty_lam. apply IH. intros [| k]; simpl; [ reflexivity | apply Hr ].
  - exact (ty_app G' A C _ _ (IHf G' r Hr) (IHa G' r Hr)).
Qed.

Lemma has_ty_psub : ∀ (G : tctx) (t : term) (A : sty), has_ty G t A →
  ∀ (G' : tctx) (s : sub), (∀ n : nat, has_ty G' (s n) (G n)) → has_ty G' (psub s t) A.
Proof.
  intros G t A H.
  induction H as [G n | G A C b Hb IH | G A C f a Hf IHf Ha IHa]; intros G' s Hs; simpl.
  - apply Hs.
  - apply ty_lam. apply IH. intros [| k]; simpl.
    + exact (ty_var (scons A G') 0).
    + apply (has_ty_ren G' (s k) (G k) (Hs k) (scons A G') S). intros m. reflexivity.
  - exact (ty_app G' A C _ _ (IHf G' s Hs) (IHa G' s Hs)).
Qed.

Lemma has_ty_beta : ∀ (G : tctx) (A C : sty) (b u : term),
  has_ty (scons A G) b C → has_ty G u A → has_ty G (psub (scons u Var) b) C.
Proof.
  intros G A C b u Hb Hu. apply (has_ty_psub _ _ _ Hb). intros [| k]; simpl.
  - exact Hu.
  - exact (ty_var G k).
Qed.

Lemma has_ty_lam_inv : ∀ (G : tctx) (b : term) (T : sty), has_ty G (Lam b) T →
  ∃ A C : sty, T = SArrow A C ∧ has_ty (scons A G) b C.
Proof. intros G b T H. inversion H; subst. ∃ A, C. split; [ reflexivity | assumption ]. Qed.

Lemma has_ty_app_inv : ∀ (G : tctx) (f a : term) (C : sty), has_ty G (App f a) C →
  ∃ A : sty, has_ty G f (SArrow A C) ∧ has_ty G a A.
Proof. intros G f a C H. inversion H; subst. ∃ A. split; assumption. Qed.

Theorem has_ty_bstep : ∀ t t' : term, bstep t t' →
  ∀ (G : tctx) (A : sty), has_ty G t A → has_ty G t' A.
Proof.
  intros t t' H.
  induction H as [b u | b b' H IH | f f' a H IH | f a a' H IH]; intros G T Hty.
  - destruct (has_ty_app_inv _ _ _ _ Hty) as [A [Hf Ha]].
    destruct (has_ty_lam_inv _ _ _ Hf) as [A' [C [HE Hb]]].
    injection HE as HA HC. subst. exact (has_ty_beta _ _ _ _ _ Hb Ha).
  - destruct (has_ty_lam_inv _ _ _ Hty) as [A [C [HE Hb]]]. subst.
    apply ty_lam. exact (IH _ _ Hb).
  - destruct (has_ty_app_inv _ _ _ _ Hty) as [A [Hf Ha]].
    exact (ty_app _ _ _ _ _ (IH _ _ Hf) Ha).
  - destruct (has_ty_app_inv _ _ _ _ Hty) as [A [Hf Ha]].
    exact (ty_app _ _ _ _ _ Hf (IH _ _ Ha)).
Qed.

Theorem has_ty_bsteps : ∀ (G : tctx) (t u : term) (A : sty),
  has_ty G t A → bsteps t u → has_ty G u A.
Proof.
  intros G t u A Hty H. revert Hty. induction H as [t | t x u Hx Hs IH]; intros Hty.
  - exact Hty.
  - apply IH. exact (has_ty_bstep _ _ Hx _ _ Hty).
Qed.
