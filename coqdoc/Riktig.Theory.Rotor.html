Rotors: the unit is one, they are closed under products, and -- when the reversal commutes with the grade involution -- under reversal. In a plane spanned by two orthonormal vectors, the exponential cos t + sin t (e1 e2) is a rotor for every angle.

  Theorem rotor_one : Rotor F V A (1 : A).
  Proof.
    split.
    - exact involute_one.
    - calc ((1 : A) × reverse 1) == ((1 : A) × 1) { (op_proper _ _ (reflexivity _) _ _ reverse_one) } == (1 : A) { (left_identity (action := mult) (1 : A)) }.
    - calc (reverse (1 : A) × 1) == ((1 : A) × 1) { (op_proper _ _ reverse_one _ _ (reflexivity _)) } == (1 : A) { (left_identity (action := mult) (1 : A)) }.
  Qed.

  Theorem rotor_mul : ∀ R S : A, Rotor F V A R → Rotor F V A S → Rotor F V A (R × S).
  Proof.
    intros R S HR HS. split.
    - calc (involute (R × S))
        == (involute R × involute S) { (additive (opA := mult) (opB := mult) (f := involute) R S) }
        == (R × S) { (op_proper _ _ (rotor_even (R := R)) _ _ (rotor_even (R := S))) }.
    - calc ((R × S) × reverse (R × S))
        == ((R × S) × (reverse S × reverse R)) { (op_proper _ _ (reflexivity _) _ _ (reverse_anti R S)) }
        == (R × (S × (reverse S × reverse R))) { (associative (op := mult) (action := mult) R S _) }
        == (R × ((S × reverse S) × reverse R)) { (op_proper _ _ (reflexivity _) _ _ (symmetry (associative (op := mult) (action := mult) S (reverse S) (reverse R)))) }
        == (R × ((1 : A) × reverse R)) { (op_proper _ _ (reflexivity _) _ _ (op_proper _ _ (rotor_unit_r (R := S)) _ _ (reflexivity _))) }
        == (R × reverse R) { (op_proper _ _ (reflexivity _) _ _ (left_identity (action := mult) (reverse R))) }
        == (1 : A) { (rotor_unit_r (R := R)) }.
    - calc (reverse (R × S) × (R × S))
        == ((reverse S × reverse R) × (R × S)) { (op_proper _ _ (reverse_anti R S) _ _ (reflexivity _)) }
        == (reverse S × (reverse R × (R × S))) { (associative (op := mult) (action := mult) (reverse S) (reverse R) _) }
        == (reverse S × ((reverse R × R) × S)) { (op_proper _ _ (reflexivity _) _ _ (symmetry (associative (op := mult) (action := mult) (reverse R) R S))) }
        == (reverse S × ((1 : A) × S)) { (op_proper _ _ (reflexivity _) _ _ (op_proper _ _ (rotor_unit_l (R := R)) _ _ (reflexivity _))) }
        == (reverse S × S) { (op_proper _ _ (reflexivity _) _ _ (left_identity (action := mult) S)) }
        == (1 : A) { (rotor_unit_l (R := S)) }.
  Qed.

  Theorem rotor_reverse : (∀ x : A, reverse (involute x) == involute (reverse x)) →
    ∀ R : A, Rotor F V A R → Rotor F V A (reverse R).
  Proof.
    intros Hri R HR. split.
    - calc (involute (reverse R)) == (reverse (involute R)) { (symmetry (Hri R)) } == (reverse R) { (reverse_proper _ _ (rotor_even (R := R))) }.
    - calc (reverse R × reverse (reverse R)) == (reverse R × R) { (op_proper _ _ (reflexivity _) _ _ (reverse_involutive R)) } == (1 : A) { (rotor_unit_l (R := R)) }.
    - calc (reverse (reverse R) × reverse R) == (R × reverse R) { (op_proper _ _ (reverse_involutive R) _ _ (reflexivity _)) } == (1 : A) { (rotor_unit_r (R := R)) }.
  Qed.

The rotors of a plane

  Section Plane.

    Context (reverse_smul : ∀ (a : F) (x : A), reverse (a *: x) == a *: reverse x)
            (reverse_vec : ∀ u : V, reverse (vec u) == vec u).
    Context (e1 e2 : V) (H11 : ⟪ e1 , e1 ⟫ == 1) (H22 : ⟪ e2 , e2 ⟫ == 1) (H12 : ⟪ e1 , e2 ⟫ == 0).
    Local Notation B := (vec e1 × vec e2).

    Lemma plane_bivector_sq : B × B == - (1 : A).
    Proof.
      assert (Hprod : ⟪ e1 , e1 ⟫ × ⟪ e2 , e2 ⟫ == 1).
      { calc (⟪ e1 , e1 ⟫ × ⟪ e2 , e2 ⟫) == ((1 : F) × ⟪ e2 , e2 ⟫) { H11 } == ((1 : F) × 1) { H22 } == (1 : F) { (left_identity (action := mult) (1 : F)) }. }
      calc (B × B)
        == (- (⟪ e1 , e1 ⟫ × ⟪ e2 , e2 ⟫) *: (1 : A)) { (orthogonal_mul_sq e1 e2 H12) }
        == (- (1 : F) *: (1 : A)) { Hprod }
        == (- (1 : A)) { (neg_one_smul (1 : A)) }.
    Qed.

    Lemma plane_bivector_even : involute B == B.
    Proof.
      calc (involute (vec e1 × vec e2))
        == (involute (vec e1) × involute (vec e2)) { (additive (opA := mult) (opB := mult) (f := involute) (vec e1) (vec e2)) }
        == (- vec e1 × - vec e2) { (op_proper _ _ (involute_vec e1) _ _ (involute_vec e2)) }
        == (vec e1 × vec e2) { (neg_mul_neg (vec e1) (vec e2)) }.
    Qed.

    Lemma plane_bivector_reverse : reverse B == - B.
    Proof.
      calc (reverse (vec e1 × vec e2))
        == (reverse (vec e2) × reverse (vec e1)) { (reverse_anti (vec e1) (vec e2)) }
        == (vec e2 × vec e1) { (op_proper _ _ (reverse_vec e2) _ _ (reverse_vec e1)) }
        == (- (vec e1 × vec e2)) { (orthogonal_anticommute e2 e1 (transitivity (commutative (op := inner) e2 e1) H12)) }.
    Qed.

    Lemma reverse_plane_rotor : ∀ t : F, reverse (rotor B t) == cos t *: (1 : A) + (- sin t) *: B.
    Proof.
      intros t.
      calc (reverse (cos t *: (1 : A) + sin t *: B))
        == (reverse (cos t *: (1 : A)) + reverse (sin t *: B)) { (additive (opA := add) (opB := add) (f := reverse) _ _) }
        == (cos t *: reverse (1 : A) + sin t *: reverse B) { (op_proper _ _ (reverse_smul _ _) _ _ (reverse_smul _ _)) }
        == (cos t *: (1 : A) + sin t *: (- B)) { (op_proper _ _ (scale_proper _ _ (reflexivity _) _ _ reverse_one) _ _ (scale_proper _ _ (reflexivity _) _ _ plane_bivector_reverse)) }
        == (cos t *: (1 : A) + - (sin t *: B)) { (op_proper _ _ (reflexivity _) _ _ (smul_neg (sin t) B)) }
        == (cos t *: (1 : A) + (- sin t) *: B) { (op_proper _ _ (reflexivity _) _ _ (symmetry (neg_smul (sin t) B))) }.
    Qed.

    Theorem plane_rotor : ∀ t : F, Rotor F V A (rotor B t).
    Proof.
      intros t.
      assert (Hcc : cos t × cos t + - (sin t × - sin t) == 1).
      { calc (cos t × cos t + - (sin t × - sin t))
          == (cos t × cos t + - - (sin t × sin t)) { (mul_neg (sin t) (sin t)) }
          == (cos t × cos t + sin t × sin t) { (neg_neg (sin t × sin t)) }
          == (1 : F) { (cos_sq_add_sin_sq t) }. }
      assert (Hcc' : cos t × cos t + - (- sin t × sin t) == 1).
      { calc (cos t × cos t + - (- sin t × sin t))
          == (cos t × cos t + - - (sin t × sin t)) { (neg_mul (sin t) (sin t)) }
          == (cos t × cos t + sin t × sin t) { (neg_neg (sin t × sin t)) }
          == (1 : F) { (cos_sq_add_sin_sq t) }. }
      assert (Hcs : cos t × - sin t + sin t × cos t == 0).
      { calc (cos t × - sin t + sin t × cos t)
          == (- (cos t × sin t) + sin t × cos t) { (mul_neg (cos t) (sin t)) }
          == (- (cos t × sin t) + cos t × sin t) { (commutative (op := mult) (sin t) (cos t)) }
          == (0 : F) { (left_inverse (cos t × sin t)) }. }
      assert (Hcs' : cos t × sin t + - sin t × cos t == 0).
      { calc (cos t × sin t + - sin t × cos t)
          == (cos t × sin t + - (sin t × cos t)) { (neg_mul (sin t) (cos t)) }
          == (cos t × sin t + - (cos t × sin t)) { (commutative (op := mult) (sin t) (cos t)) }
          == (0 : F) { (right_inverse (cos t × sin t)) }. }
      split.
      - calc (involute (cos t *: (1 : A) + sin t *: B))
          == (involute (cos t *: (1 : A)) + involute (sin t *: B)) { (additive (opA := add) (opB := add) (f := involute) _ _) }
          == (cos t *: involute (1 : A) + sin t *: involute B) { (op_proper _ _ (homogeneous (f := involute) _ _) _ _ (homogeneous (f := involute) _ _)) }
          == (cos t *: (1 : A) + sin t *: B) { (op_proper _ _ (scale_proper _ _ (reflexivity _) _ _ involute_one) _ _ (scale_proper _ _ (reflexivity _) _ _ plane_bivector_even)) }.
      - calc (rotor B t × reverse (rotor B t))
          == ((cos t *: (1 : A) + sin t *: B) × (cos t *: (1 : A) + (- sin t) *: B)) { (op_proper _ _ (reflexivity _) _ _ (reverse_plane_rotor t)) }
          == ((cos t × cos t + - (sin t × - sin t)) *: (1 : A) + (cos t × - sin t + sin t × cos t) *: B)
               { (combination_mul B plane_bivector_sq (cos t) (sin t) (cos t) (- sin t)) }
          == ((1 : F) *: (1 : A) + (0 : F) *: B) { (op_proper _ _ (scale_proper _ _ Hcc _ _ (reflexivity _)) _ _ (scale_proper _ _ Hcs _ _ (reflexivity _))) }
          == ((1 : A) + 0) { (op_proper _ _ (left_identity (action := scale) (1 : A)) _ _ (zero_smul B)) }
          == (1 : A) { (right_identity (action := add) (1 : A)) }.
      - calc (reverse (rotor B t) × rotor B t)
          == ((cos t *: (1 : A) + (- sin t) *: B) × (cos t *: (1 : A) + sin t *: B)) { (op_proper _ _ (reverse_plane_rotor t) _ _ (reflexivity _)) }
          == ((cos t × cos t + - (- sin t × sin t)) *: (1 : A) + (cos t × sin t + - sin t × cos t) *: B)
               { (combination_mul B plane_bivector_sq (cos t) (- sin t) (cos t) (sin t)) }
          == ((1 : F) *: (1 : A) + (0 : F) *: B) { (op_proper _ _ (scale_proper _ _ Hcc' _ _ (reflexivity _)) _ _ (scale_proper _ _ Hcs' _ _ (reflexivity _))) }
          == ((1 : A) + 0) { (op_proper _ _ (left_identity (action := scale) (1 : A)) _ _ (zero_smul B)) }
          == (1 : A) { (right_identity (action := add) (1 : A)) }.
    Qed.
  End Plane.
