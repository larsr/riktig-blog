From Riktig.Structure Require Export Order.

An ordered field with approximate square roots: for a nonnegative argument and a positive error, a nonnegative element whose square is at most the argument, and which, once the error is added, squares to at least the argument.
Module ApproxSqrt.
  Class ApproxSqrt (F : Type) `{OrderedField F} (approx_sqrt : HasApproxSqrt F) := {
    approx_sqrt_nonneg : ∀ x eps : F, 0 ≤ approx_sqrt x eps;
    approx_sqrt_le : ∀ x eps : F, 0 ≤ x → 0 < eps →
      approx_sqrt x eps × approx_sqrt x eps ≤ x;
    approx_sqrt_close : ∀ x eps : F, 0 ≤ x → 0 < eps →
      x ≤ (approx_sqrt x eps + eps) × (approx_sqrt x eps + eps);
  }.
  Hint Mode ApproxSqrt ! - - - - - - - - - - - - : typeclass_instances.
End ApproxSqrt.
Export ApproxSqrt.
