Class HasGradedWedge (A : Type) := graded_wedge : A → A → A.
Infix "∧" := graded_wedge (at level 40, left associativity).
Hint Mode HasGradedWedge ! : typeclass_instances.
Typeclasses Transparent HasGradedWedge.

Class HasGradedDot (A : Type) := graded_dot : A → A → A.
Infix "⋅" := graded_dot (at level 40, left associativity).
Hint Mode HasGradedDot ! : typeclass_instances.
Typeclasses Transparent HasGradedDot.

The wedge of a grade-r part with a grade-s part is the grade-r + s selection of their product; the dot keeps grade |r - s| instead, spelled (r - s) + (s - r) in truncated nat subtraction. Bilinearity is not restated: it is what the double bounded sum says.
Module GradedProducts.
  Class GradedProducts (n : nat) (F A : Type)
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {Am : HasMult A}
      {As : HasScale F A} {AM : Monoid A} {Ag : HasGrade A}
      {Aw : HasGradedWedge A} {Ad : HasGradedDot A}
      (bigop_list : ∀ J : Type, (J → A) → list J → A)
      {BL : BigOp A bigop_list} := {
    wedge_from_grade_parts : ∀ x y : A,
      x ∧ y == bigop_list _ (fun r ⇒ bigop_list _ (fun s ⇒
                 ⟨ ⟨ x ⟩_ r × ⟨ y ⟩_ s ⟩_ (r + s)) (seq 0 (S n)))
                 (seq 0 (S n));
    dot_from_grade_parts : ∀ x y : A,
      x ⋅ y == bigop_list _ (fun r ⇒ bigop_list _ (fun s ⇒
                 ⟨ ⟨ x ⟩_ r × ⟨ y ⟩_ s ⟩_ ((r - s) + (s - r))) (seq 0 (S n)))
                 (seq 0 (S n));
  }.

End GradedProducts.
