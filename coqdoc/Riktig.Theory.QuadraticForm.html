Section FieldAsQuadraticForm.
  Context {F : Type} `{Field F}.

  Add Ring square_form_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition square_form : F → F := fun u ⇒ u × u.

  Lemma polar_square : ∀ u v : F, polar square_form u v == u × v + u × v.
  Proof.
    intros u v. unfold polar, square_form. ring.
  Qed.

  #[export] Instance square_form_proper : Proper (equ ==> equ) square_form.
  Proof.
    intros x y Hxy. unfold square_form.
    rewrite Hxy. reflexivity.
  Qed.

  Lemma square_form_scale : ∀ a u : F, square_form (a *: u) == (a × a) × square_form u.
  Proof.
    intros a u. unfold square_form, scale, Field_Scale. ring.
  Qed.

  #[export] Instance Field_QuadraticForm : QuadraticForm F F square_form.
  Proof.
    unshelve econstructor.
    - exact square_form_proper.
    - exact (fun a u ⇒ square_form_scale a u).
    - intros v. unshelve econstructor.
      + intros x y Hxy. rewrite !polar_square. rewrite Hxy. reflexivity.
      + intros x y.
        calc (polar square_form (x + y) v)
          == ((x + y) × v + (x + y) × v : F) { polar_square }
          == ((x × v + y × v) + (x × v + y × v) : F) { (right_distributive x y v) }
          == ((x × v + x × v) + (y × v + y × v) : F) { add_interchange }
          == [fold_polar] (polar square_form x v + polar square_form y v : F).
        [fold_polar]: rewrite !polar_square; reflexivity.
      + intros a x.
        unfold scale, Field_Scale.
        calc (polar square_form (a × x) v)
          == ((a × x) × v + (a × x) × v : F) { polar_square }
          == (a × (x × v) + a × (x × v) : F) { (associative (op := mult) a x v) }
          == (a × (x × v + x × v) : F) { (left_distributive a (x × v) (x × v)) }
          == (a × polar square_form x v : F) { polar_square }.
  Qed.
End FieldAsQuadraticForm.

Section NegatedQuadraticForm.
  Context {F V : Type} `{VectorSpace F V} (Q : V → F) `{!QuadraticForm F V Q}.
  Add Ring neg_form_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition neg_form : V → F := fun u ⇒ - Q u.

  Lemma polar_neg_form : ∀ u v : V, polar neg_form u v == - polar Q u v.
  Proof.
    intros u v. unfold polar, neg_form. ring.
  Qed.

  #[export] Instance Neg_QuadraticForm : QuadraticForm F V neg_form.
  Proof.
    unshelve econstructor.
    - intros x y Hxy. unfold neg_form. rewrite Hxy. reflexivity.
    - intros a u. unfold neg_form.
      calc (- Q (a *: u))
        == (- ((a × a) × Q u) : F) { (homogeneous (f := Q) a u) }
        == ((a × a) × - Q u : F) { mul_neg_r }.
    - intros v. unshelve econstructor.
      + intros x y Hxy. rewrite !polar_neg_form.
        rewrite (proper (f := fun u ⇒ polar Q u v) x y Hxy). reflexivity.
      + intros x y.
        calc (polar neg_form (x + y) v)
          == (- polar Q (x + y) v : F) { polar_neg_form }
          == (- (polar Q x v + polar Q y v) : F) { (additive (f := fun u ⇒ polar Q u v) x y) }
          == (- polar Q x v + - polar Q y v : F) { neg_add }
          == [fold_polar] (polar neg_form x v + polar neg_form y v : F).
        [fold_polar]: rewrite !polar_neg_form; reflexivity.
      + intros a x.
        calc (polar neg_form (a *: x) v)
          == (- polar Q (a *: x) v : F) { polar_neg_form }
          == (- (a × polar Q x v) : F) { (homogeneous (f := fun u ⇒ polar Q u v) a x) }
          == (a × - polar Q x v : F) { mul_neg_r }
          == (a × polar neg_form x v : F) { polar_neg_form }.
  Qed.
End NegatedQuadraticForm.

Section InnerProductQuadraticForm.
  Context {F V : Type} `{InnerProductSpace F V}.
  Add Ring polar_norm2_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma polar_norm2 : ∀ u v : V, polar norm2 u v == ⟨ u , v ⟩ + ⟨ u , v ⟩.
  Proof.
    intros u v. unfold polar.
    rewrite (norm2_add u v), <- (commutative u v). ring.
  Qed.

  #[export] Instance Inner_QuadraticForm : QuadraticForm F V norm2.
  Proof.
    unshelve econstructor.
    - exact norm2_proper.
    - intros a u. unfold norm2.
      calc (⟨ a *: u , a *: u ⟩)
        == (a × ⟨ u , a *: u ⟩ : F) { (inner_left_scale a u (a *: u)) }
        == (a × ⟨ a *: u , u ⟩ : F) { (commutative u (a *: u)) }
        == (a × (a × ⟨ u , u ⟩) : F) { (inner_left_scale a u u) }
        == ((a × a) × ⟨ u , u ⟩ : F) { (associative (action := mult) (op := mult) a a (⟨ u , u ⟩)) }.
    - intros v. unshelve econstructor.
      + intros x y Hxy. rewrite !polar_norm2. rewrite Hxy. reflexivity.
      + intros x y.
        calc (polar norm2 (x + y) v)
          == (⟨ x + y , v ⟩ + ⟨ x + y , v ⟩ : F) { polar_norm2 }
          == ((⟨ x , v ⟩ + ⟨ y , v ⟩) + (⟨ x , v ⟩ + ⟨ y , v ⟩) : F) { (inner_left_additive x y v) }
          == ((⟨ x , v ⟩ + ⟨ x , v ⟩) + (⟨ y , v ⟩ + ⟨ y , v ⟩) : F) { add_interchange }
          == [fold_polar] (polar norm2 x v + polar norm2 y v : F).
        [fold_polar]: rewrite !polar_norm2; reflexivity.
      + intros a x.
        calc (polar norm2 (a *: x) v)
          == (⟨ a *: x , v ⟩ + ⟨ a *: x , v ⟩ : F) { polar_norm2 }
          == (a × ⟨ x , v ⟩ + a × ⟨ x , v ⟩ : F) { (inner_left_scale a x v) }
          == (a × (⟨ x , v ⟩ + ⟨ x , v ⟩) : F) { (left_distributive a (⟨ x , v ⟩) (⟨ x , v ⟩)) }
          == (a × polar norm2 x v : F) { polar_norm2 }.
  Qed.
End InnerProductQuadraticForm.
