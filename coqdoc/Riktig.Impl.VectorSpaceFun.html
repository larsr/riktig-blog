Library Riktig.Impl.VectorSpaceFun

From Riktig Require Export Structure.TableSpace Impl.TableFun.

Functions I → F into a field as a vector space, with every operation taken pointwise; as tables they form a table space.
Section FunVS.
  Context {F I : Type} `{Field F}.

  #[export] Instance Fun_Zero : Zero (I → F) := fun _ ⇒ 0.
  #[export] Instance Fun_HasAdd : HasAdd (I → F) := fun f g i ⇒ f i + g i.
  #[export] Instance Fun_HasNeg : HasNeg (I → F) := fun f i ⇒ - f i.
  #[export] Instance Fun_HasScale : HasScale F (I → F) := fun a f i ⇒ a × f i.

  Local Ltac pointwise := unfold Fun_Zero, Fun_HasAdd, Fun_HasNeg, Fun_HasScale; simpl.

  #[export] Instance Fun_Equivalence : Equivalence (@equ (I → F) Fun_EqRel).
  Proof.
    split.
    - intros f i. reflexivity.
    - intros f g Hfg i. symmetry. exact (Hfg i).
    - intros f g h Hfg Hgh i. transitivity (g i); [ exact (Hfg i) | exact (Hgh i) ].
  Qed.

  #[export] Instance Fun_Magma : Magma (I → F) Fun_HasAdd.
  Proof.
    split.
    - exact Fun_Equivalence.
    - intros f f' Hf g g' Hg i. pointwise. apply op_proper; [ exact (Hf i) | exact (Hg i) ].
  Qed.
  #[export] Instance Fun_Semigroup : Semigroup (I → F) Fun_HasAdd.
  Proof. split. exact Fun_Magma. intros f g h i. pointwise. apply associative. Qed.
  #[export] Instance Fun_Monoid : Monoid (I → F) Fun_Zero Fun_HasAdd.
  Proof.
    split. exact Fun_Semigroup.
    - intros f i. pointwise. apply left_identity.
    - intros f i. pointwise. apply right_identity.
  Qed.
  #[export] Instance Fun_Group : Group (I → F) Fun_Zero Fun_HasAdd Fun_HasNeg.
  Proof.
    split. exact Fun_Monoid.
    - intros f f' Hf i. pointwise. apply neg_proper. exact (Hf i).
    - intros f i. pointwise. apply left_inverse.
    - intros f i. pointwise. apply right_inverse.
  Qed.
  #[export] Instance Fun_AbGroup : AbGroup (I → F) Fun_Zero Fun_HasAdd Fun_HasNeg.
  Proof. split. exact Fun_Group. intros f g i. pointwise. apply commutative. Qed.
  #[export] Instance Fun_ScaleLaws : ScaleLaws F (I → F) Fun_HasAdd Fun_HasScale.
  Proof.
    split.
    - intros a a' Ha f f' Hf i. pointwise. apply op_proper; [ exact Ha | exact (Hf i) ].
    - intros a f g i. pointwise. apply left_distributive.
    - intros a b f i. pointwise. apply right_distributive.
    - intros a b f i. pointwise. apply (associative (op := mult) (action := mult)).
    - intros f i. pointwise. apply left_identity.
  Qed.
  #[export] Instance Fun_VectorSpace : VectorSpace F (I → F) Fun_Zero Fun_HasAdd Fun_HasNeg Fun_HasScale :=
    {| VectorSpace.vectors := Fun_AbGroup; VectorSpace.scaling := Fun_ScaleLaws |}.

  #[export] Instance Fun_TableSpace : TableSpace F I (I → F) (eqI := eq).
  Proof. split; intros; pointwise; reflexivity. Qed.
End FunVS.
