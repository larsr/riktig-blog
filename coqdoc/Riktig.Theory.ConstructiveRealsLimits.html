Convergence and the Cauchy property of a sequence of constructive reals, with a modulus: for every positive rational precision an index from which on the terms are that close. What the limit structure asks of them.

  Lemma le_of_add_le_l : ∀ n m i : N, n + m ≤ i → n ≤ i.
  Proof. intros n m i Hi. exact (Theory.NatOrder.le_trans n (n + m) i (Theory.NatOrder.le_add_r n m) Hi). Qed.

  Lemma le_of_add_le_r : ∀ n m i : N, n + m ≤ i → m ≤ i.
  Proof. intros n m i Hi. exact (Theory.NatOrder.le_trans m (n + m) i (Theory.NatOrder.le_add_l n m) Hi). Qed.

  Lemma abs_sub_comm : ∀ x y : R, abs (x + - y) == abs (y + - x).
  Proof.
    intros x y.
    calc (abs (x + - y))
      == (abs (- (x + - y))) { (abs_neg (x + - y)) }
      == (abs (- x + - - y)) { (neg_add x (- y)) }
      == (abs (- x + y)) { (neg_neg y) }
      == (abs (y + - x)) { (commutative (op := add) (- x) y) }.
  Qed.

The distance between two points is at most the distances to a third.
  Lemma abs_sub_le_add : ∀ x y z : R, abs (x + - y) ≼ abs (x + - z) + abs (y + - z).
  Proof.
    intros x y z.
    apply (le_equ_l (abs ((x + - z) + - (y + - z)))).
    - apply abs_proper.
      calc ((x + - z) + - (y + - z))
        == ((x + - z) + (- y + - - z)) { (neg_add y (- z)) }
        == ((x + - z) + (- y + z)) { (neg_neg z) }
        == ((x + - z) + (z + - y)) { (commutative (op := add) (- y) z) }
        == (x + (- z + (z + - y))) { (associative (op := add) (action := add) x (- z) (z + - y)) }
        == (x + - y) { (neg_add_cancel_left z (- y)) }.
    - apply (le_trans _ (abs (x + - z) + abs (- (y + - z)))); [ exact (abs_triangle _ _) | ].
      apply add_le_mono_l. apply le_of_equ. exact (abs_neg _).
  Qed.

  Theorem converges_proper : ∀ (u u' : N → R) (l l' : R),
    (∀ n : N, u n == u' n) → l == l' → converges u l → converges u' l'.
  Proof.
    intros u u' l l' Huu' Hll' Hu eps Heps.
    destruct (Hu eps Heps) as [n Hn]. ∃ n. intros i Hi.
    apply (le_equ_l (abs (u i + - l))); [ | exact (Hn i Hi) ].
    apply abs_proper. apply op_proper; [ exact (Huu' i) | apply neg_proper; exact Hll' ].
  Defined.

  Theorem converges_unique : ∀ (u : N → R) (l l' : R),
    converges u l → converges u l' → l == l'.
  Proof.
    intros u l l' Hl Hl'. apply eq_of_sub_eq_zero. apply eq_zero_of_abs_le_all.
    intros eps Heps.
    destruct (Hl (half eps) (Theory.Order.half_pos eps Heps)) as [n Hn].
    destruct (Hl' (half eps) (Theory.Order.half_pos eps Heps)) as [m Hm].
    apply (le_equ_r _ (cast (half eps) + cast (half eps))); [ exact (cast_half eps) | ].
    apply (le_trans _ (abs (l + - u (n + m)) + abs (l' + - u (n + m)))); [ exact (abs_sub_le_add l l' (u (n + m))) | ].
    apply add_le_mono.
    - apply (le_equ_l (abs (u (n + m) + - l))); [ exact (abs_sub_comm _ _) | ].
      exact (Hn (n + m) (Theory.NatOrder.le_add_r n m)).
    - apply (le_equ_l (abs (u (n + m) + - l'))); [ exact (abs_sub_comm _ _) | ].
      exact (Hm (n + m) (Theory.NatOrder.le_add_l n m)).
  Qed.

  Theorem converges_const : ∀ c : R, converges (fun _ : N ⇒ c) c.
  Proof.
    intros c eps Heps. ∃ 0. intros i _.
    apply (le_equ_l 0); [ | exact (le_of_lt _ _ (cast_pos eps Heps)) ].
    symmetry.
    calc (abs (c + - c))
      == (abs (0 : R)) { (right_inverse (op := add) c) }
      == (0 : R) { abs_zero }.
  Defined.

  Theorem converges_add : ∀ (u v : N → R) (l m : R),
    converges u l → converges v m → converges (fun n ⇒ u n + v n) (l + m).
  Proof.
    intros u v l m Hu Hv eps Heps.
    destruct (Hu (half eps) (Theory.Order.half_pos eps Heps)) as [n Hn].
    destruct (Hv (half eps) (Theory.Order.half_pos eps Heps)) as [n' Hn'].
    ∃ (n + n'). intros i Hi.
    apply (le_equ_r _ (cast (half eps) + cast (half eps))); [ exact (cast_half eps) | ].
    apply (le_trans _ (abs (u i + - l) + abs (v i + - m)));
      [ | exact (add_le_mono _ _ _ _ (Hn i (le_of_add_le_l n n' i Hi)) (Hn' i (le_of_add_le_r n n' i Hi))) ].
    apply (le_equ_l (abs ((u i + - l) + (v i + - m)))); [ | exact (abs_triangle _ _) ].
    apply abs_proper. symmetry. exact (add_sub_add_comm (u i) (v i) l m).
  Defined.

  Theorem converges_neg : ∀ (u : N → R) (l : R),
    converges u l → converges (fun n ⇒ - u n) (- l).
  Proof.
    intros u l Hu eps Heps.
    destruct (Hu eps Heps) as [n Hn]. ∃ n. intros i Hi.
    apply (le_equ_l (abs (u i + - l))); [ | exact (Hn i Hi) ].
    calc (abs (u i + - l))
      == (abs (- (u i + - l))) { (abs_neg (u i + - l)) }
      == (abs (- u i + - - l)) { (neg_add (u i) (- l)) }.
  Defined.

  Theorem converges_smul : ∀ (a : Q) (u : N → R) (l : R),
    converges u l → converges (fun n ⇒ cast a × u n) (cast a × l).
  Proof.
    intros a u l Hu eps Heps.
    assert (Hpos : 0 < abs a + 1)
      by exact (Theory.Order.lt_of_le_of_lt 0 (abs a) (abs a + 1) (Theory.Order.abs_nonneg a) (Theory.Order.lt_add_one (abs a))).
    assert (Hne : abs a + 1 =/= 0) by exact (fun H0 ⇒ proj2 Hpos (symmetry H0)).
    pose (eps' := eps × inv (abs a + 1)).
    assert (Heps' : 0 < eps') by exact (Theory.Order.mul_pos eps (inv (abs a + 1)) Heps (Theory.Order.inv_pos (abs a + 1) Hpos)).
    destruct (Hu eps' Heps') as [n Hn]. ∃ n. intros i Hi.
    apply (le_equ_r _ (cast ((abs a + 1) × eps')));
      [ apply hom_proper; exact (mul_mul_inv_cancel_left eps (abs a + 1) Hne) | ].
    apply (le_equ_r _ (cast (abs a + 1) × cast eps'));
      [ symmetry; exact (additive (f := cast) (opA := mult) (abs a + 1) eps') | ].
    apply (le_equ_l (cast (abs a) × abs (u i + - l))).
    - calc (cast (abs a) × abs (u i + - l))
        == (abs (cast a) × abs (u i + - l)) { (abs_cast a) }
        == (abs (cast a × (u i + - l))) { (abs_mul (cast a) (u i + - l)) }
        == (abs (cast a × u i + - (cast a × l))) { (mul_sub (cast a) (u i) l) }.
    - apply (le_trans _ (cast (abs a) × cast eps')).
      + exact (mul_le_mono_l (cast (abs a)) _ _ (cast_nonneg (abs a) (Theory.Order.abs_nonneg a)) (Hn i Hi)).
      + apply mul_le_mono_r; [ exact (le_of_lt _ _ (cast_pos eps' Heps')) | ].
        exact (cast_le (abs a) (abs a + 1) (Theory.Order.le_of_lt _ _ (Theory.Order.lt_add_one (abs a)))).
  Defined.

  Theorem converges_tail : ∀ (u : N → R) (l : R),
    converges u l → converges (fun n ⇒ u (succ n)) l.

    intros u l Hu eps Heps.

  Defined.

  Theorem cauchy_proper : ∀ u u' : N → R,
    (∀ n : N, u n == u' n) → is_cauchy u → is_cauchy u'.
  Proof.
    intros u u' Huu' Hu eps Heps.
    destruct (Hu eps Heps) as [n Hn]. ∃ n. intros i j Hi Hj.
    apply (le_equ_l (abs (u i + - u j))); [ | exact (Hn i j Hi Hj) ].
    apply abs_proper. apply op_proper; [ exact (Huu' i) | apply neg_proper; exact (Huu' j) ].
  Defined.

  Theorem cauchy_of_converges : ∀ (u : N → R) (l : R), converges u l → is_cauchy u.
  Proof.
    intros u l Hu eps Heps.
    destruct (Hu (half eps) (Theory.Order.half_pos eps Heps)) as [n Hn].
    ∃ n. intros i j Hi Hj.
    apply (le_equ_r _ (cast (half eps) + cast (half eps))); [ exact (cast_half eps) | ].
    apply (le_trans _ (abs (u i + - l) + abs (u j + - l))); [ exact (abs_sub_le_add (u i) (u j) l) | ].
    exact (add_le_mono _ _ _ _ (Hn i Hi) (Hn j Hj)).
  Defined.

  Theorem cauchy_add : ∀ u v : N → R,
    is_cauchy u → is_cauchy v → is_cauchy (fun n ⇒ u n + v n).
  Proof.
    intros u v Hu Hv eps Heps.
    destruct (Hu (half eps) (Theory.Order.half_pos eps Heps)) as [n Hn].
    destruct (Hv (half eps) (Theory.Order.half_pos eps Heps)) as [n' Hn'].
    ∃ (n + n'). intros i j Hi Hj.
    apply (le_equ_r _ (cast (half eps) + cast (half eps))); [ exact (cast_half eps) | ].
    apply (le_trans _ (abs (u i + - u j) + abs (v i + - v j)));
      [ | exact (add_le_mono _ _ _ _
                   (Hn i j (le_of_add_le_l n n' i Hi) (le_of_add_le_l n n' j Hj))
                   (Hn' i j (le_of_add_le_r n n' i Hi) (le_of_add_le_r n n' j Hj))) ].
    apply (le_equ_l (abs ((u i + - u j) + (v i + - v j)))); [ | exact (abs_triangle _ _) ].
    apply abs_proper. symmetry. exact (add_sub_add_comm (u i) (v i) (u j) (v j)).
  Defined.

A sequence whose tail from some index is Cauchy is Cauchy.
  Theorem cauchy_of_shift : ∀ (u : N → R) (m : N), Proper (equ ==> equ) u →
    is_cauchy (fun c ⇒ u (m + c)) → is_cauchy u.
  Proof.
    intros u m Hu Hc eps Heps.
    destruct (Hc eps Heps) as [n Hn]. ∃ (m + n). intros i j Hi Hj.
    destruct (proj1 (le_add (m + n) i) Hi) as [ci Hci].
    destruct (proj1 (le_add (m + n) j) Hj) as [cj Hcj].
    assert (Hi' : u i == u (m + (n + ci))).
    { apply Hu. symmetry. apply (transitivity (symmetry (associative (op := add) (action := add) m n ci))). exact Hci. }
    assert (Hj' : u j == u (m + (n + cj))).
    { apply Hu. symmetry. apply (transitivity (symmetry (associative (op := add) (action := add) m n cj))). exact Hcj. }
    apply (le_equ_l (abs (u (m + (n + ci)) + - u (m + (n + cj))))).
    - apply abs_proper. apply op_proper; [ symmetry; exact Hi' | apply neg_proper; symmetry; exact Hj' ].
    - exact (Hn (n + ci) (n + cj) (Theory.NatOrder.le_add_r n ci) (Theory.NatOrder.le_add_r n cj)).
  Defined.

  Theorem cauchy_tail : ∀ u : N → R, is_cauchy u → is_cauchy (fun n ⇒ u (succ n)).
  Proof.
    intros u Hu eps Heps.
    destruct (Hu eps Heps) as [n Hn]. ∃ n. intros i j Hi Hj.
    exact (Hn (succ i) (succ j)
             (Theory.NatOrder.le_trans n i (succ i) Hi (Theory.NatOrder.le_succ_diag i))
             (Theory.NatOrder.le_trans n j (succ j) Hj (Theory.NatOrder.le_succ_diag j))).
  Defined.

  Lemma archimedean_abs_pos : ∀ (x : R) (q : Q), abs x ≺ cast q → (0 : Q) < q.
  Proof.
    intros x q Hq. apply (lt_cast 0 q).
    apply (lt_equ_l 0 (cast 0)); [ symmetry; exact (hom_zero cast) | ].
    exact (le_lt_trans 0 (abs x) (cast q) (Theory.ConstructiveReals.abs_nonneg x) Hq).
  Qed.

A convergent sequence is bounded from some index on, by a rational just above the limit.
  Theorem converges_bound_tail : ∀ (u : N → R) (l : R), converges u l →
    {n : N & {C : Q & ((0 : Q) < C) × (∀ i : N, n ≤ i → abs (u i) ≼ cast C)}}%type.
  Proof.
    intros u l Hu.
    destruct (ConstructiveReals.archimedean (abs l)) as [q Hq].
    destruct (Hu 1 (Theory.Order.zero_lt_one (F := Q))) as [n Hn].
    ∃ n. ∃ (1 + q). split.
    - apply (proj1 (Theory.Order.lt_proper 0 0 (reflexivity 0) (q + 1) (1 + q) (commutative (op := add) q 1))).
      exact (Theory.Order.lt_trans 0 q (q + 1) (archimedean_abs_pos l q Hq) (Theory.Order.lt_add_one q)).
    - intros i Hi.
      apply (le_equ_l (abs ((u i + - l) + l))).
      { apply abs_proper.
        calc ((u i + - l) + l)
          == (u i + (- l + l)) { (associative (op := add) (action := add) (u i) (- l) l) }
          == (u i + 0) { (left_inverse (op := add) l) }
          == (u i) { (right_identity (action := add) (u i)) }. }
      apply (le_trans _ (abs (u i + - l) + abs l)); [ exact (abs_triangle _ _) | ].
      apply (le_equ_r _ (cast 1 + cast q));
        [ symmetry; exact (additive (f := cast) (opA := add) 1 q) | ].
      exact (add_le_mono _ _ _ _ (Hn i Hi) (le_of_lt _ _ Hq)).
  Defined.

Products of convergent sequences converge to the product: split u v - l m as u (v - m) + (u - l) m, bound u by its tail bound and m by a rational, and give each part half.
  Theorem converges_mul : ∀ (u v : N → R) (l m : R),
    converges u l → converges v m → converges (fun n ⇒ u n × v n) (l × m).
  Proof.
    intros u v l m Hu Hv eps Heps.
    destruct (converges_bound_tail u l Hu) as [n0 [C [HC HuC]]].
    destruct (ConstructiveReals.archimedean (abs m)) as [q Hq].
    assert (Hq0 : (0 : Q) < q) by exact (archimedean_abs_pos m q Hq).
    assert (HneC : C =/= 0) by exact (fun H0 ⇒ proj2 HC (symmetry H0)).
    assert (Hneq : q =/= 0) by exact (fun H0 ⇒ proj2 Hq0 (symmetry H0)).
    pose (eps1 := half eps × inv C).
    assert (Heps1 : 0 < eps1)
      by exact (Theory.Order.mul_pos _ _ (Theory.Order.half_pos eps Heps) (Theory.Order.inv_pos C HC)).
    pose (eps2 := half eps × inv q).
    assert (Heps2 : 0 < eps2)
      by exact (Theory.Order.mul_pos _ _ (Theory.Order.half_pos eps Heps) (Theory.Order.inv_pos q Hq0)).
    destruct (Hv eps1 Heps1) as [n1 Hn1].
    destruct (Hu eps2 Heps2) as [n2 Hn2].
    ∃ ((n0 + n1) + n2). intros i Hi.
    assert (Hi0 : n0 ≤ i) by exact (le_of_add_le_l n0 n1 i (le_of_add_le_l (n0 + n1) n2 i Hi)).
    assert (Hi1 : n1 ≤ i) by exact (le_of_add_le_r n0 n1 i (le_of_add_le_l (n0 + n1) n2 i Hi)).
    assert (Hi2 : n2 ≤ i) by exact (le_of_add_le_r (n0 + n1) n2 i Hi).
    apply (le_equ_l (abs (u i × (v i + - m) + (u i + - l) × m))).
    { apply abs_proper.
      calc (u i × (v i + - m) + (u i + - l) × m)
        == (u i × v i + - (u i × m) + (u i + - l) × m) { (mul_sub (u i) (v i) m) }
        == (u i × v i + - (u i × m) + (u i × m + - (l × m))) { (sub_mul (u i) l m) }
        == (u i × v i + (- (u i × m) + (u i × m + - (l × m)))) { (associative (op := add) (action := add) (u i × v i) (- (u i × m)) (u i × m + - (l × m))) }
        == (u i × v i + (- (u i × m) + u i × m + - (l × m))) { (associative (op := add) (action := add) (- (u i × m)) (u i × m) (- (l × m))) }
        == (u i × v i + (0 + - (l × m))) { (left_inverse (op := add) (u i × m)) }
        == (u i × v i + - (l × m)) { (left_identity (action := add) (- (l × m))) }. }
    apply (le_trans _ (abs (u i × (v i + - m)) + abs ((u i + - l) × m))); [ exact (abs_triangle _ _) | ].
    apply (le_equ_l (abs (u i) × abs (v i + - m) + abs (u i + - l) × abs m)).
    { apply op_proper; symmetry; [ exact (ConstructiveReals.abs_mul (u i) (v i + - m)) | exact (ConstructiveReals.abs_mul (u i + - l) m) ]. }
    apply (le_equ_r _ (cast (half eps) + cast (half eps))); [ exact (cast_half eps) | ].
    apply add_le_mono.
    - apply (le_equ_r _ (cast (C × eps1))).
      { apply hom_proper. exact (mul_mul_inv_cancel_left (half eps) C HneC). }
      apply (le_equ_r _ (cast C × cast eps1)); [ symmetry; exact (additive (f := cast) (opA := mult) C eps1) | ].
      apply Theory.ConstructiveReals.mul_le_mono.
      + exact (Theory.ConstructiveReals.abs_nonneg _).
      + exact (HuC i Hi0).
      + exact (Theory.ConstructiveReals.abs_nonneg _).
      + exact (Hn1 i Hi1).
    - apply (le_equ_r _ (cast (q × eps2))).
      { apply hom_proper. exact (mul_mul_inv_cancel_left (half eps) q Hneq). }
      apply (le_equ_r _ (cast eps2 × cast q)).
      { apply (transitivity (y := cast (eps2 × q))).
        - symmetry. exact (additive (f := cast) (opA := mult) eps2 q).
        - apply hom_proper. exact (commutative (op := mult) eps2 q). }
      apply Theory.ConstructiveReals.mul_le_mono.
      + exact (Theory.ConstructiveReals.abs_nonneg _).
      + exact (Hn2 i Hi2).
      + exact (Theory.ConstructiveReals.abs_nonneg _).
      + exact (Theory.ConstructiveReals.le_of_lt _ _ Hq).
  Defined.

The comparison test: a sequence of reals bounded term by term by rationals whose tail sums are small has Cauchy partial sums.

  Local Notation "'psum' f n" := (natrec _ 0 (fun k acc ⇒ acc + f k) n) (at level 10, f at level 9, n at level 9).
  Local Notation "'is_cauchy' u" :=
    (∀ eps : Q, 0 < eps →
      {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps})
    (at level 10, u at level 9).

  Lemma abs_psum_le : ∀ (f : N → R) (g : N → Q),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    (∀ k : N, abs (f k) ≼ cast (g k)) →
    ∀ c : N, abs (psum f c) ≼ cast (psum g c).
  Proof.
    intros f g Hf Hg Hfg.
    apply (nat_induction (fun c ⇒ abs (psum f c) ≼ cast (psum g c))).
    - intros c c' Hc. split; intros Hle.
      + apply (le_equ_l (abs (psum f c))); [ apply abs_proper; exact (psum_proper f Hf c c' Hc) | ].
        apply (le_equ_r _ (cast (psum g c))); [ apply hom_proper; exact (psum_proper g Hg c c' Hc) | exact Hle ].
      + apply (le_equ_l (abs (psum f c'))); [ apply abs_proper; exact (psum_proper f Hf c' c (symmetry Hc)) | ].
        apply (le_equ_r _ (cast (psum g c'))); [ apply hom_proper; exact (psum_proper g Hg c' c (symmetry Hc)) | exact Hle ].
    - apply (le_equ_l (0 : R)).
      + symmetry.
        calc (abs (psum f 0)) == (abs (0 : R)) { (psum_zero f) } == (0 : R) { abs_zero }.
      + apply (le_equ_r _ (cast 0)).
        × apply hom_proper. symmetry. exact (psum_zero g).
        × apply (le_equ_r _ 0); [ symmetry; exact (hom_zero cast) | exact (le_refl 0) ].
    - intros c IH.
      apply (le_equ_l (abs (psum f c + f c))); [ apply abs_proper; symmetry; exact (psum_succ f c) | ].
      apply (le_equ_r _ (cast (psum g c + g c))); [ apply hom_proper; symmetry; exact (psum_succ g c) | ].
      apply (le_equ_r _ (cast (psum g c) + cast (g c))); [ symmetry; exact (additive (f := cast) (psum g c) (g c)) | ].
      apply (le_trans _ (abs (psum f c) + abs (f c))); [ exact (abs_triangle _ _) | ].
      exact (add_le_mono _ _ _ _ IH (Hfg c)).
  Qed.

  Theorem cauchy_psum_of_abs_le : ∀ (f : N → R) (g : N → Q),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    (∀ k : N, abs (f k) ≼ cast (g k)) →
    (∀ eps : Q, 0 < eps → {n : N & ∀ i c : N, n ≤ i → psum (fun k ⇒ g (i + k)) c ≤ eps}) →
    is_cauchy (fun n ⇒ psum f n).
  Proof.
    intros f g Hf Hg Hfg Htail eps Heps.
    destruct (Htail eps Heps) as [n Hn]. ∃ n. intros i j Hi Hj.
    assert (Hcase : ∀ i' j' : N, n ≤ i' → i' ≤ j' → abs (psum f j' + - psum f i') ≼ cast eps).
    { intros i' j' Hi' Hij. destruct (proj1 (le_add i' j') Hij) as [c Hc].
      assert (Hfs : Proper (equ ==> equ) (fun k ⇒ f (i' + k)))
        by (intros k k' Hk; apply Hf; apply op_proper; [ reflexivity | exact Hk ]).
      assert (Hgs : Proper (equ ==> equ) (fun k ⇒ g (i' + k)))
        by (intros k k' Hk; apply Hg; apply op_proper; [ reflexivity | exact Hk ]).
      apply (le_equ_l (abs (psum (fun k ⇒ f (i' + k)) c))).
      - apply abs_proper.
        calc (psum (fun k ⇒ f (i' + k)) c)
          == ((psum f i' + psum (fun k ⇒ f (i' + k)) c) + - psum f i') { (add_sub_cancel_left (psum f i') (psum (fun k ⇒ f (i' + k)) c)) }
          == (psum f (i' + c) + - psum f i') { (psum_shift f Hf i' c) }
          == (psum f j' + - psum f i') { (psum_proper f Hf (i' + c) j' Hc) }.
      - apply (le_trans _ (cast (psum (fun k ⇒ g (i' + k)) c)));
          [ exact (abs_psum_le _ _ Hfs Hgs (fun k ⇒ Hfg (i' + k)) c) | ].
        exact (cast_le _ _ (Hn i' c Hi')). }
    destruct (Theory.NatOrder.le_total i j) as [Hij | Hji].
    - apply (le_equ_l (abs (psum f j + - psum f i))); [ exact (abs_sub_comm _ _) | exact (Hcase i j Hi Hij) ].
    - exact (Hcase j i Hj Hji).
  Defined.
End ConstructiveRealsSeries.
