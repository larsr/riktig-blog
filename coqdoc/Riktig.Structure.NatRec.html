From Riktig.Structure Require Export Peano.

Recursion on the Peano numbers: from a value at zero and a step, a function on the numbers, given by its two equations, which respects equality when the step does.
Module NatRec.
  Class NatRec (N : Type) `{Peano N} (natrec : HasNatRec N) := {
    natrec_proper : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f : N → A → A), Proper (equ ==> equ ==> equ) f →
      Proper (equ ==> equ) (natrec A a f);
    natrec_zero : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f : N → A → A),
      natrec A a f 0 == a;
    natrec_succ : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
        (a : A) (f : N → A → A) (n : N),
      natrec A a f (succ n) == f n (natrec A a f n);
  }.
  Hint Mode NatRec ! - - - - - : typeclass_instances.
End NatRec.
Export NatRec.
