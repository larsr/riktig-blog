Lemma vector_bool_equ_eq : ∀ (n : nat) (v w : Vector.t bool n),
  v == w → v = w.
Proof.
  intros n v w Hvw.
  apply Vector.eq_nth_iff. intros p q Hpq. rewrite <- Hpq. exact (Hvw p).
Qed.

Lemma vector_bool_eq_equ : ∀ (n : nat) (v w : Vector.t bool n),
  v = w → v == w.
Proof. intros n v w Hvw i. rewrite Hvw. reflexivity. Qed.

Definition vector_bool_eq_dec (n : nat) (v w : Vector.t bool n)
    : {v = w} + {v ≠ w} :=
  Vector.eq_dec bool Bool.eqb Bool.eqb_true_iff n v w.

Instance VectorBlade_EquDec (n : nat) : EquDec (Vector.t bool n).
Proof.
  intros v w.
  destruct (vector_bool_eq_dec n v w) as [Heq | Hne].
  - left. exact (vector_bool_eq_equ n v w Heq).
  - right. intros Hequ. exact (Hne (vector_bool_equ_eq n v w Hequ)).
Defined.


Fixpoint blade_enum (n : nat) : list (Vector.t bool n) :=
  match n with
  | O ⇒ Vector.nil bool :: nil
  | S m ⇒ List.map (Vector.cons bool false m) (blade_enum m)
             ++ List.map (Vector.cons bool true m) (blade_enum m)
  end.

Lemma blade_enum_S : ∀ m : nat,
  blade_enum (S m)
  = List.map (Vector.cons bool false m) (blade_enum m)
      ++ List.map (Vector.cons bool true m) (blade_enum m).
Proof. reflexivity. Qed.

Lemma vector_cons_injective : ∀ (b : bool) (n : nat)
    (v w : Vector.t bool n),
  Vector.cons bool b n v = Vector.cons bool b n w → v = w.
Proof. intros b n v w Hvw. exact (f_equal (@Vector.tl bool n) Hvw). Qed.

Lemma blade_enum_length : ∀ n : nat, length (blade_enum n) = (2 ^ n)%nat.
Proof.
  induction n as [| m IH]; [ reflexivity |].
  rewrite (blade_enum_S m), length_app, !length_map, IH.
  simpl (2 ^ S m)%nat. lia.
Qed.

Lemma blade_enum_complete : ∀ (n : nat) (v : Vector.t bool n),
  In v (blade_enum n).
Proof.
  induction n as [| m IH]; intros v.
  - apply (Vector.case0 (fun w ⇒ In w (blade_enum 0))). left. reflexivity.
  - apply (Vector.caseS' v (fun w ⇒ In w (blade_enum (S m)))).
    intros b t. rewrite (blade_enum_S m). apply in_or_app.
    destruct b.
    + right. apply in_map. exact (IH t).
    + left. apply in_map. exact (IH t).
Qed.

Lemma blade_enum_NoDup : ∀ n : nat, NoDup (blade_enum n).
Proof.
  induction n as [| m IH].
  - constructor; [ intros [] | constructor ].
  - rewrite (blade_enum_S m).
    apply NoDup_app.
    + apply (NoDup_map_injective (Vector.cons bool false m) (blade_enum m));
        [ exact (vector_cons_injective false m) | exact IH ].
    + apply (NoDup_map_injective (Vector.cons bool true m) (blade_enum m));
        [ exact (vector_cons_injective true m) | exact IH ].
    + intros v Hfalse Htrue.
      apply in_map_iff in Hfalse as [x [Hx _]].
      apply in_map_iff in Htrue as [y [Hy _]].
      rewrite <- Hx in Hy.
      exact (Bool.diff_true_false (f_equal (@Vector.hd bool m) Hy)).
Qed.

Lemma blade_enum_NoDupA : ∀ n : nat,
  NoDupA (@equ (Vector.t bool n) _) (blade_enum n).
Proof.
  intros n.
  exact (NoDupA_of_NoDup_discrete _ _ _ (vector_bool_equ_eq n)
           (blade_enum_NoDup n)).
Qed.

Lemma blade_enum_completeA : ∀ (n : nat) (v : Vector.t bool n),
  InA equ v (blade_enum n).
Proof.
  intros n v.
  exact (@In_InA _ equ VectorBlade_Equivalence _ v (blade_enum_complete n v)).
Qed.

Instance VectorBlade_FinIndex (n : nat) :
  FinIndex (2 ^ n)%nat (Vector.t bool n) :=
  {| FinIndex.enum_equivalence := VectorBlade_Equivalence;
     FinIndex.enum := blade_enum n;
     FinIndex.enum_NoDup := blade_enum_NoDupA n;
     FinIndex.enum_complete := blade_enum_completeA n;
     FinIndex.enum_length := blade_enum_length n |}.

Lemma vector_blade_families_proper {F : Type} {eF : EqRel F}
    {RF : Reflexive (@equ F eF)} (n : nat) :
  ∀ f : Vector.t bool n → F, Proper (equ ==> equ) f.
Proof.
  intros f v w Hvw. rewrite (vector_bool_equ_eq n v w Hvw). reflexivity.
Qed.

Example vector_blade_finindex_check (n : nat) :
  FinIndex (2 ^ n)%nat (Vector.t bool n) := _.

Example vector_blade_bigop_check (F : Type) `{Field F} (n : nat) :
  (Vector.t bool n → F) → F := fun f ⇒ bigopI f.

Example blade_enum_at_two : blade_enum 2 = Vector.of_list (false :: false :: nil)
  :: Vector.of_list (false :: true :: nil)
  :: Vector.of_list (true :: false :: nil)
  :: Vector.of_list (true :: true :: nil) :: nil.
Proof. vm_compute. reflexivity. Qed.
