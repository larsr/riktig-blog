Continuation-passing style. cps t is Fischer's call-by-value translation: every intermediate result is named and every call is the whole body -- the head position. triv are the trivial terms (no work to do), tail the terms whose every call is in head position; psi strips a value's translation back off.
Renaming that inserts one fresh variable at depth k.
Definition liftr (k : nat) : nat → nat := fun i ⇒ if Nat.ltb i k then i else S i.
Definition lift (k : nat) (t : term) : term := ren (liftr k) t.

Fixpoint cps (t : term) : term :=
  match t with
  | Var n ⇒ λ ⦅#0 (Var (S n))⦆
  | Const x ⇒ λ ⦅#0 (Const x)⦆
  | Lam b ⇒ λ ⦅#0 (λ (lift 1 (cps b)))⦆
  | App f a ⇒ λ ⦅(lift 0 (cps f)) (λ ⦅(lift 0 (lift 0 (cps a))) (λ ⦅#1 #0 #2⦆)⦆)⦆
  end.

Definition psi (v : term) : term :=
  match v with
  | Lam b ⇒ λ (cps b)
  | other ⇒ other
  end.

The head-position grammar

Inductive triv : term → Prop :=
  | triv_var : ∀ n : nat, triv (Var n)
  | triv_const : ∀ x : string, triv (Const x)
  | triv_lam : ∀ b : term, tail b → triv (Lam b)
with tail : term → Prop :=
  | tail_triv : ∀ v : term, triv v → tail v
  | tail_call : ∀ f v : term, triv f → triv v → tail ⦅f v⦆
  | tail_call2 : ∀ f v w : term, triv f → triv v → triv w → tail ⦅f v w⦆.

Scheme triv_tail_ind := Induction for triv Sort Prop
  with tail_triv_ind := Induction for tail Sort Prop.

The grammar is stable under renaming.
Lemma triv_ren : ∀ t : term, triv t → ∀ r : nat → nat, triv (ren r t).
Proof.
  apply (triv_tail_ind (fun t _ ⇒ ∀ r, triv (ren r t)) (fun t _ ⇒ ∀ r, tail (ren r t))).
  - intros n r. simpl. apply triv_var.
  - intros x r. simpl. apply triv_const.
  - intros b Hb IH r. simpl. apply triv_lam. apply IH.
  - intros v Hv IH r. apply tail_triv. apply IH.
  - intros f v Hf IHf Hv IHv r. simpl. apply tail_call; [ apply IHf | apply IHv ].
  - intros f v w Hf IHf Hv IHv Hw IHw r. simpl. apply tail_call2; [ apply IHf | apply IHv | apply IHw ].
Qed.

Lemma tail_ren : ∀ t : term, tail t → ∀ r : nat → nat, tail (ren r t).
Proof.
  apply (tail_triv_ind (fun t _ ⇒ ∀ r, triv (ren r t)) (fun t _ ⇒ ∀ r, tail (ren r t))).
  - intros n r. simpl. apply triv_var.
  - intros x r. simpl. apply triv_const.
  - intros b Hb IH r. simpl. apply triv_lam. apply IH.
  - intros v Hv IH r. apply tail_triv. apply IH.
  - intros f v Hf IHf Hv IHv r. simpl. apply tail_call; [ apply IHf | apply IHv ].
  - intros f v w Hf IHf Hv IHv Hw IHw r. simpl. apply tail_call2; [ apply IHf | apply IHv | apply IHw ].
Qed.

Every translated term is trivial, and its body is all tail calls: the recursive calls of a CPS program sit in head position.
Theorem cps_trivial : ∀ t : term, triv (cps t).
Proof.
  intros t. induction t as [n | b IH | f IHf a IHa | x]; simpl.
  - apply triv_lam. apply tail_call; [ apply triv_var | apply triv_var ].
  - apply triv_lam. apply tail_call; [ apply triv_var | ].
    apply triv_lam. unfold lift. apply tail_ren. apply tail_triv. exact IH.
  - apply triv_lam. apply tail_call.
    + unfold lift. apply triv_ren. exact IHf.
    + apply triv_lam. apply tail_call.
      × unfold lift. apply triv_ren. apply triv_ren. exact IHa.
      × apply triv_lam. apply tail_call2; [ apply triv_var | apply triv_var | apply triv_var ].
  - apply triv_lam. apply tail_call; [ apply triv_var | apply triv_const ].
Qed.

Applying a translated value to a continuation

One contraction hands the continuation the stripped value: the inserted lift is cancelled by the substitution.
Lemma psub_lift : ∀ (t k : term), psub (scons k Var) (lift 0 t) = t.
Proof.
  intros t k. unfold lift. rewrite psub_ren.
  rewrite (psub_ext t _ Var); [ apply psub_var | ].
  intros n. unfold liftr. simpl. reflexivity.
Qed.

Lemma psub_lift1 : ∀ (t k : term), psub (up (scons k Var)) (lift 1 t) = t.
Proof.
  intros t k. unfold lift. rewrite psub_ren.
  rewrite (psub_ext t _ Var); [ apply psub_var | ].
  intros [| n]; unfold liftr; simpl; reflexivity.
Qed.

Theorem cps_value : ∀ v k : term, triv v → joinable ⦅(cps v) k⦆ ⦅k (psi v)⦆.
Proof.
  intros v k Hv. destruct Hv as [n | x | b Hb]; simpl.
  - apply joinable_bsteps. apply star_one.
    exact (bstep_beta ⦅#0 (Var (S n))⦆ k).
  - apply joinable_bsteps. apply star_one.
    exact (bstep_beta ⦅#0 (Const x)⦆ k).
  - apply (joinable_trans _ (psub (scons k Var) ⦅#0 (λ (lift 1 (cps b)))⦆)).
    { apply joinable_bsteps. apply star_one. exact (bstep_beta _ k). }
    change (psub (scons k Var) ⦅#0 (λ (lift 1 (cps b)))⦆)
      with ⦅k (λ (psub (up (scons k Var)) (lift 1 (cps b))))⦆.
    rewrite psub_lift1. apply joinable_refl.
Qed.

Values, and how the translation renames

cps commutes with renaming on the nose, so does psi on values, and the translation of any value is one call away from handing its continuation the stripped value -- no grammar hypothesis needed.
Definition val (t : term) : bool := match t with App _ _ ⇒ false | _ ⇒ true end.

Lemma val_ren : ∀ (t : term) (r : nat → nat), val (ren r t) = val t.
Proof. intros [n | b | f a | x] r; reflexivity. Qed.

Lemma cps_ren : ∀ (t : term) (r : nat → nat), cps (ren r t) = ren r (cps t).
Proof.
  intros t. induction t as [n | b IH | f IHf a IHa | x]; intros r; simpl.
  - reflexivity.
  - rewrite IH. unfold lift. rewrite !ren_ren.
    apply f_equal. apply f_equal2; [ reflexivity | ].
    apply f_equal. apply ren_ext. intros [| j]; reflexivity.
  - rewrite IHf, IHa. unfold lift. rewrite !ren_ren.
    apply f_equal. apply f_equal2.
    + apply ren_ext. intros j. reflexivity.
    + apply f_equal. apply f_equal2.
      × apply ren_ext. intros j. reflexivity.
      × reflexivity.
  - reflexivity.
Qed.

Lemma psi_ren : ∀ (v : term) (r : nat → nat), val v = true → psi (ren r v) = ren r (psi v).
Proof.
  intros [n | b | f a | x] r H; simpl; try reflexivity; try discriminate.
  rewrite cps_ren. reflexivity.
Qed.

Lemma cps_val_eta : ∀ v : term, val v = true → cps v = λ ⦅#0 (ren S (psi v))⦆.
Proof.
  intros [n | b | f a | x] H; simpl; try reflexivity; try discriminate.
  unfold lift. apply f_equal. apply f_equal2; [ reflexivity | ].
  apply f_equal. apply ren_ext. intros [| j]; reflexivity.
Qed.

Lemma cps_val : ∀ v k : term, val v = true → joinable ⦅(cps v) k⦆ ⦅k (psi v)⦆.
Proof.
  intros v k Hv. rewrite (cps_val_eta v Hv).
  apply (joinable_trans _ (psub (scons k Var) ⦅#0 (ren S (psi v))⦆)).
  { apply joinable_bsteps. apply star_one. exact (bstep_beta _ k). }
  change (psub (scons k Var) ⦅#0 (ren S (psi v))⦆)
    with ⦅k (psub (scons k Var) (ren S (psi v)))⦆.
  rewrite psub_ren.
  rewrite (psub_ext (psi v) _ Var); [ rewrite psub_var; apply joinable_refl | intros n; reflexivity ].
Qed.

The translation commutes with substitution by values

The heart of the correctness argument: substituting values and then translating is translating and then substituting the stripped values.
Definition vals (s : sub) : Prop := ∀ n : nat, val (s n) = true.

Lemma vals_up : ∀ s : sub, vals s → vals (up s).
Proof. intros s Hs [| n]; simpl; [ reflexivity | rewrite val_ren; apply Hs ]. Qed.

Lemma psi_up : ∀ (s : sub) (n : nat), vals s → psi (up s n) = up (fun m ⇒ psi (s m)) n.
Proof. intros s [| n] Hs; simpl; [ reflexivity | rewrite psi_ren; [ reflexivity | apply Hs ] ]. Qed.
