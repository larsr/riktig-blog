The grading of a doubled algebra: the grade-k part of (a, b) is the grade-k part of a and the grade-k - 1 part of b (nothing at k = 0), since the second component carries the new generator. The top grade goes up by one.
Section GradedDouble.
  Context {F V A N : Type} `{GradeInvolution F V A} (s : F)
    `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}
    {grading : HasGrade N A} {top_grade : N} {GA : Graded N F A grading top_grade}.

  Local Notation "'low' k b" := (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨b⟩_j) k) (at level 10, k at level 9, b at level 9).

  #[export] Instance Double_HasGrade : HasGrade N (Double s A) :=
    fun k x ⇒ (⟨fst x⟩_k, low k (snd x)).

  #[export] Instance Double_grade_Linear : ∀ k : N, Linear F (Double s A) (Double s A) (grade_part k).
  Proof.
    intros k. split.
    - intros x x' [Hx1 Hx2]. split; simpl; [ exact (grade_proper k _ _ Hx1) | exact (low_proper k _ _ Hx2) ].
    - intros x y. split; simpl; [ exact (grade_add k _ _) | exact (low_add k _ _) ].
    - intros a x. split; simpl; [ exact (grade_smul k a _) | exact (low_smul k a _) ].
  Qed.

  #[export] Instance Double_Graded : Graded N F (Double s A) Double_HasGrade (succ top_grade).
  Proof.
    split.
    - exact Double_grade_Linear.
    - intros k k' x Hk. split; simpl; [ exact (Graded.grade_index_proper k k' _ Hk) | exact (low_index_proper k k' _ Hk) ].
    - intros k x. split; simpl; [ exact (Graded.grade_idempotent k _) | exact (low_low k _) ].
    - intros k l x Hkl. split; simpl; [ exact (Graded.grade_orthogonal k l _ Hkl) | exact (low_low_ne k l _ Hkl) ].
    - intros x. destruct x as [a b].
      assert (Hpair : ∀ m : N,
        natrec (Double s A) 0 (fun k acc ⇒ acc + ⟨(a, b)⟩_k) m
        == (natrec A 0 (fun k acc ⇒ acc + ⟨a⟩_k) m, natrec A 0 (fun k acc ⇒ acc + low k b) m)).
      { assert (HsD : Proper (equ ==> equ ==> equ) (fun (k : N) (acc : Double s A) ⇒ acc + ⟨(a, b)⟩_k)).
        { intros k k' Hk acc acc' Hacc. apply Magma.op_proper; [ exact Hacc | ].
          split; simpl; [ exact (Graded.grade_index_proper k k' a Hk) | exact (low_index_proper k k' b Hk) ]. }
        assert (Hsa : Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc + ⟨a⟩_k))
          by (apply sum_step_proper; exact (grade_fun_proper a)).
        assert (Hsb : Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc + low k b))
          by (apply sum_step_proper; intros k k' Hk; exact (low_index_proper k k' b Hk)).
        intros m. split; simpl.
        + exact (natrec_fusion (Double s A) A _ _ _ _ (@fst A A) 0 _ _
                   (fun x y Hxy ⇒ proj1 Hxy) HsD Hsa (fun k x ⇒ reflexivity _) m).
        + exact (natrec_fusion (Double s A) A _ _ _ _ (@snd A A) 0 _ _
                   (fun x y Hxy ⇒ proj2 Hxy) HsD Hsb (fun k x ⇒ reflexivity _) m). }
      rewrite (Hpair (succ (succ top_grade))). split; simpl.
      + calc (natrec A 0 (fun k acc ⇒ acc + ⟨a⟩_k) (succ (succ top_grade)))
          == (natrec A 0 (fun k acc ⇒ acc + ⟨a⟩_k) (succ top_grade) + ⟨a⟩_(succ top_grade)) { (NatRec.natrec_succ A _ _ 0 _ _) }
          == (a + 0) { (Magma.op_proper _ _ (Graded.grade_complete a) _ _ (grade_above a)) }
          == a { (right_identity (action := add) a) }.
      + calc (natrec A 0 (fun k acc ⇒ acc + low k b) (succ (succ top_grade)))
          == (natrec A 0 (fun k acc ⇒ acc + ⟨b⟩_k) (succ top_grade)) { (low_sum b (succ top_grade)) }
          == b { (Graded.grade_complete b) }.
  Qed.
End GradedDouble.
