From Coq Require Import String List.
From Riktig Require Export Structure.TypedLambda Impl.LambdaTerm Impl.LambdaTypeTree.
Import ListNotations.

The de Bruijn terms with the typing relation are well-typed lambda terms: contexts are total -- a type for every index -- extended at the front.
Instance Tctx_EqRel : EqRel tctx := fun G G' ⇒ ∀ n, G n = G' n.

Instance Term_TypedLambda : TypedLambda string sty nat term sub tctx
    (fun G n ⇒ G n) scons has_ty.
Proof.
  split.
  - intros G t A A' HA Hty. unfold equ, Sty_EqRel in HA. subst. exact Hty.
  - intros G n. exact (ty_var G n).
  - intros G A C b Hb. exact (ty_lam G A C b Hb).
  - intros G A C f a Hf Ha. exact (ty_app G A C f a Hf Ha).
  - intros G A. reflexivity.
  - intros G A n. reflexivity.
  - intros G t t' A Hty Hr. exact (has_ty_bsteps G t t' A Hty Hr).
Qed.
