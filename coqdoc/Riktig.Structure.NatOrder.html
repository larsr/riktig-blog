From Riktig.Structure Require Export Naturals.

The order of the natural numbers: one number is at most another exactly when something added to it gives the other.
Module NatOrder.
  Class NatOrder (N : Type) `{Naturals N} (le : HasLe N) := {
    le_proper :: Proper (equ ==> equ ==> iff) le;
    le_add : ∀ a b : N, a ≤ b ↔ ∃ c : N, a + c == b;
  }.
End NatOrder.
Export NatOrder.
