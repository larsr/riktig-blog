From Riktig.Structure Require Export Category.

A terminal object receives exactly one arrow from every object.
Module Terminal.
  Class Terminal (O : Type) `{Category O} (terminal : HasTerm O) (to_terminal : HasToTerm O) := {
    to_term_unique : ∀ (x : O) (f : x ~> term), f == to_term x;
  }.
  Hint Mode Terminal ! - - - - - - - : typeclass_instances.
End Terminal.
Export Terminal.
