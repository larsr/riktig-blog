Section RingTheory.
  Context {R : Type} `{Ring R}.

  Theorem zero_mul : ∀ a : R, 0 × a == 0.
  Proof.
    intros a.
    apply (add_left_cancel (0 × a)).
    calc (0 × a + 0 × a)
      == ((0 + 0) × a) { right_distributive }
      == (0 × a) { (right_identity (action := add) (0 : R)) }
      == (0 × a + 0) { right_identity }.
  Qed.

  Theorem mul_zero : ∀ a : R, a × 0 == 0.
  Proof.
    intros a.
    apply (add_left_cancel (a × 0)).
    calc (a × 0 + a × 0)
      == (a × (0 + 0)) { left_distributive }
      == (a × 0) { (right_identity (action := add) (0 : R)) }
      == (a × 0 + 0) { right_identity }.
  Qed.

  Theorem two_mul : ∀ a : R, a + a == (1 + 1) × a.
  Proof.
    intros a.
    calc (a + a)
      == (1 × a + 1 × a) { (left_identity (action := mult) a) }
      == ((1 + 1) × a) { right_distributive }.
  Qed.

  Theorem neg_mul : ∀ a b : R, - a × b == - (a × b).
  Proof.
    intros a b.
    apply eq_neg_of_add_eq_zero_left.
    calc (- a × b + a × b)
      == ((- a + a) × b) { right_distributive }
      == (0 × b) { (left_inverse (op := add) a) }
      == (0 : R) { zero_mul }.
  Qed.

  Theorem mul_neg : ∀ a b : R, a × - b == - (a × b).
  Proof.
    intros a b.
    apply eq_neg_of_add_eq_zero_left.
    calc (a × - b + a × b)
      == (a × (- b + b)) { left_distributive }
      == (a × 0) { (left_inverse (op := add) b) }
      == (0 : R) { mul_zero }.
  Qed.

  Theorem neg_mul_neg : ∀ a b : R, - a × - b == a × b.
  Proof.
    intros a b.
    calc (- a × - b)
      == (- (a × - b)) { neg_mul }
      == (- - (a × b)) { mul_neg }
      == (a × b) { neg_neg }.
  Qed.

  Theorem neg_one_mul : ∀ a : R, - (1) × a == - a.
  Proof.
    intros a.
    calc (- (1) × a)
      == (- (1 × a)) { neg_mul }
      == (- a) { left_identity }.
  Qed.

  Theorem mul_neg_one : ∀ a : R, a × - (1) == - a.
  Proof.
    intros a.
    calc (a × - (1))
      == (- (a × 1)) { mul_neg }
      == (- a) { right_identity }.
  Qed.

  Theorem mul_sub : ∀ a b c : R, a × (b + - c) == a × b + - (a × c).
  Proof.
    intros a b c.
    calc (a × (b + - c))
      == (a × b + a × - c) { left_distributive }
      == (a × b + - (a × c)) { mul_neg }.
  Qed.

  Theorem sub_mul : ∀ a b c : R, (a + - b) × c == a × c + - (b × c).
  Proof.
    intros a b c.
    calc ((a + - b) × c)
      == (a × c + - b × c) { right_distributive }
      == (a × c + - (b × c)) { neg_mul }.
  Qed.

A product of sums, in a ring that need not be commutative: the cross terms keep their order.
  Theorem add_mul_add : ∀ a b c d : R,
    (a + b) × (c + d) == (a × c + a × d) + (b × c + b × d).
  Proof.
    intros a b c d.
    calc ((a + b) × (c + d))
      == (a × (c + d) + b × (c + d)) { right_distributive }
      == ((a × c + a × d) + b × (c + d)) { left_distributive }
      == ((a × c + a × d) + (b × c + b × d)) { left_distributive }.
  Qed.

End RingTheory.
