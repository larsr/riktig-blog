Recursions with pointwise equal steps, or from equal initial values, agree: induction on the Peano numbers alone.
Section NatRecPeano.
  Context {N : Type} `{Peano N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Theorem natrec_ext : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (a : A) (f g : N → A → A),
    Proper (equ ==> equ ==> equ) f → Proper (equ ==> equ ==> equ) g →
    (∀ (k : N) (x : A), f k x == g k x) →
    ∀ n : N, natrec A a f n == natrec A a g n.
  Proof.
    intros A eA EA a f g Hf Hg Hfg.
    apply (Peano.nat_induction (fun n ⇒ natrec A a f n == natrec A a g n)).
    - intros n n' Hn.
      exact (equiv_iff (NatRec.natrec_proper A eA EA a f Hf n n' Hn) (NatRec.natrec_proper A eA EA a g Hg n n' Hn)).
    - calc (natrec A a f 0) == a { (NatRec.natrec_zero A eA EA a f) } == (natrec A a g 0) { (NatRec.natrec_zero A eA EA a g) }.
    - intros n IH.
      calc (natrec A a f (succ n))
        == (f n (natrec A a f n)) { (NatRec.natrec_succ A eA EA a f n) }
        == (f n (natrec A a g n)) { (Hf n n (reflexivity n) _ _ IH) }
        == (g n (natrec A a g n)) { (Hfg n (natrec A a g n)) }
        == (natrec A a g (succ n)) { (NatRec.natrec_succ A eA EA a g n) }.
  Qed.

Recursions from equal initial values agree.
  Theorem natrec_init : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (a a' : A) (f : N → A → A),
    Proper (equ ==> equ ==> equ) f → a == a' → ∀ n : N, natrec A a f n == natrec A a' f n.
  Proof.
    intros A eA EA a a' f Hf Haa'.
    apply (Peano.nat_induction (fun n ⇒ natrec A a f n == natrec A a' f n)).
    - intros n n' Hn.
      exact (equiv_iff (NatRec.natrec_proper A eA EA a f Hf n n' Hn) (NatRec.natrec_proper A eA EA a' f Hf n n' Hn)).
    - calc (natrec A a f 0) == a { (NatRec.natrec_zero A eA EA a f) } == a' { Haa' } == (natrec A a' f 0) { (NatRec.natrec_zero A eA EA a' f) }.
    - intros n IH.
      calc (natrec A a f (succ n))
        == (f n (natrec A a f n)) { (NatRec.natrec_succ A eA EA a f n) }
        == (f n (natrec A a' f n)) { (Hf n n (reflexivity n) _ _ IH) }
        == (natrec A a' f (succ n)) { (NatRec.natrec_succ A eA EA a' f n) }.
  Qed.
End NatRecPeano.

A recursion past an index continues from its value there, and the shifted recursion is a recursion.
Section NatRecFolds.
  Context {N : Type} `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Theorem natrec_shift : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (a : A) (f : N → A → A),
    Proper (equ ==> equ ==> equ) f →
    ∀ i c : N, natrec A a f (i + c) == natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c.
  Proof.
    intros A eA EA a f Hf i.
    assert (Hg : Proper (equ ==> equ ==> equ) (fun k acc ⇒ f (i + k) acc))
      by (intros k k' Hk x x' Hx; apply Hf; [ apply Magma.op_proper; [ reflexivity | exact Hk ] | exact Hx ]).
    apply (Peano.nat_induction (fun c ⇒ natrec A a f (i + c) == natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c)).
    - intros c c' Hc.
      setoid_rewrite (NatRec.natrec_proper A eA EA a f Hf (i + c) (i + c') (Magma.op_proper i i (reflexivity i) c c' Hc)).
      setoid_rewrite (NatRec.natrec_proper A eA EA _ _ Hg c c' Hc). reflexivity.
    - calc (natrec A a f (i + 0))
        == (natrec A a f i) { (NatRec.natrec_proper A eA EA a f Hf (i + 0) i (Theory.Naturals.add_zero i)) }
        == (natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) 0)
             { (NatRec.natrec_zero A eA EA (natrec A a f i) (fun k acc ⇒ f (i + k) acc)) }.
    - intros c IH.
      calc (natrec A a f (i + succ c))
        == (natrec A a f (succ (i + c))) { (NatRec.natrec_proper A eA EA a f Hf (i + succ c) (succ (i + c)) (Theory.Naturals.add_succ i c)) }
        == (f (i + c) (natrec A a f (i + c))) { (NatRec.natrec_succ A eA EA a f (i + c)) }
        == (f (i + c) (natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c)) { (Hf _ _ (reflexivity _) _ _ IH) }
        == (natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) (succ c))
             { (NatRec.natrec_succ A eA EA (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c) }.
  Qed.

A map commuting with the step carries one recursion to another.
  Theorem natrec_fusion : ∀ (A B : Type) (eA : EqRel A) (eB : EqRel B)
      (EA : Equivalence eA) (EB : Equivalence eB)
      (h : A → B) (a : A) (f : N → A → A) (g : N → B → B),
    Proper (equ ==> equ) h →
    Proper (equ ==> equ ==> equ) f → Proper (equ ==> equ ==> equ) g →
    (∀ (k : N) (x : A), h (f k x) == g k (h x)) →
    ∀ n : N, h (natrec A a f n) == natrec B (h a) g n.
  Proof.
    intros A B eA eB EA EB h a f g Hh Hf Hg Hfg.
    apply (Peano.nat_induction (fun n ⇒ h (natrec A a f n) == natrec B (h a) g n)).
    - intros n n' Hn.
      setoid_rewrite (Hh _ _ (NatRec.natrec_proper A eA EA a f Hf n n' Hn)).
      setoid_rewrite (NatRec.natrec_proper B eB EB (h a) g Hg n n' Hn). reflexivity.
    - calc (h (natrec A a f 0))
        == (h a) { (Hh _ _ (NatRec.natrec_zero A eA EA a f)) }
        == (natrec B (h a) g 0) { (NatRec.natrec_zero B eB EB (h a) g) }.
    - intros n IH.
      calc (h (natrec A a f (succ n)))
        == (h (f n (natrec A a f n))) { (Hh _ _ (NatRec.natrec_succ A eA EA a f n)) }
        == (g n (h (natrec A a f n))) { (Hfg n (natrec A a f n)) }
        == (g n (natrec B (h a) g n)) { (Hg n n (reflexivity n) _ _ IH) }
        == (natrec B (h a) g (succ n)) { (NatRec.natrec_succ B eB EB (h a) g n) }.
  Qed.

End NatRecFolds.
