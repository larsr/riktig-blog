Recursions as folds over the naturals: pointwise equal steps agree, and a recursion past an index continues from its value there.
Section NatRecFolds.
  Context {N : Type} `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Theorem natrec_ext : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (a : A) (f g : N → A → A),
    Proper (equ ==> equ ==> equ) f → Proper (equ ==> equ ==> equ) g →
    (∀ (k : N) (x : A), f k x == g k x) →
    ∀ n : N, natrec A a f n == natrec A a g n.
  Proof.
    intros A eA EA a f g Hf Hg Hfg.
    apply (nat_induction (fun n ⇒ natrec A a f n == natrec A a g n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper A eA EA a f Hf n n' Hn).
      setoid_rewrite (natrec_proper A eA EA a g Hg n n' Hn). reflexivity.
    - calc (natrec A a f 0) == a { (natrec_zero A eA EA a f) } == (natrec A a g 0) { (natrec_zero A eA EA a g) }.
    - intros n IH.
      calc (natrec A a f (succ n))
        == (f n (natrec A a f n)) { (natrec_succ A eA EA a f n) }
        == (f n (natrec A a g n)) { (Hf n n (reflexivity n) _ _ IH) }
        == (g n (natrec A a g n)) { (Hfg n (natrec A a g n)) }
        == (natrec A a g (succ n)) { (natrec_succ A eA EA a g n) }.
  Qed.

  Theorem natrec_shift : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (a : A) (f : N → A → A),
    Proper (equ ==> equ ==> equ) f →
    ∀ i c : N, natrec A a f (i + c) == natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c.
  Proof.
    intros A eA EA a f Hf i.
    assert (Hg : Proper (equ ==> equ ==> equ) (fun k acc ⇒ f (i + k) acc))
      by (intros k k' Hk x x' Hx; apply Hf; [ apply op_proper; [ reflexivity | exact Hk ] | exact Hx ]).
    apply (nat_induction (fun c ⇒ natrec A a f (i + c) == natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c)).
    - intros c c' Hc.
      setoid_rewrite (natrec_proper A eA EA a f Hf (i + c) (i + c') (op_proper i i (reflexivity i) c c' Hc)).
      setoid_rewrite (natrec_proper A eA EA _ _ Hg c c' Hc). reflexivity.
    - calc (natrec A a f (i + 0))
        == (natrec A a f i) { (natrec_proper A eA EA a f Hf (i + 0) i (Theory.Naturals.add_zero i)) }
        == (natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) 0)
             { (natrec_zero A eA EA (natrec A a f i) (fun k acc ⇒ f (i + k) acc)) }.
    - intros c IH.
      calc (natrec A a f (i + succ c))
        == (natrec A a f (succ (i + c))) { (natrec_proper A eA EA a f Hf (i + succ c) (succ (i + c)) (Theory.Naturals.add_succ i c)) }
        == (f (i + c) (natrec A a f (i + c))) { (natrec_succ A eA EA a f (i + c)) }
        == (f (i + c) (natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c)) { (Hf _ _ (reflexivity _) _ _ IH) }
        == (natrec A (natrec A a f i) (fun k acc ⇒ f (i + k) acc) (succ c))
             { (natrec_succ A eA EA (natrec A a f i) (fun k acc ⇒ f (i + k) acc) c) }.
  Qed.

Recursions from equal initial values agree.
  Theorem natrec_init : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA) (a a' : A) (f : N → A → A),
    Proper (equ ==> equ ==> equ) f → a == a' → ∀ n : N, natrec A a f n == natrec A a' f n.
  Proof.
    intros A eA EA a a' f Hf Haa'.
    apply (nat_induction (fun n ⇒ natrec A a f n == natrec A a' f n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper A eA EA a f Hf n n' Hn).
      setoid_rewrite (natrec_proper A eA EA a' f Hf n n' Hn). reflexivity.
    - calc (natrec A a f 0) == a { (natrec_zero A eA EA a f) } == a' { Haa' } == (natrec A a' f 0) { (natrec_zero A eA EA a' f) }.
    - intros n IH.
      calc (natrec A a f (succ n))
        == (f n (natrec A a f n)) { (natrec_succ A eA EA a f n) }
        == (f n (natrec A a' f n)) { (Hf n n (reflexivity n) _ _ IH) }
        == (natrec A a' f (succ n)) { (natrec_succ A eA EA a' f n) }.
  Qed.
End NatRecFolds.
