From Coq Require Import String.
From Riktig Require Export Structure.LambdaType Structure.Decidable Lib.Coq.Stlc.

The type trees of the simply typed lambda calculus are simple types over the named base types, compared by Coq's equality.
Instance Sty_EqRel : EqRel sty := eq.
Instance Sty_HasBase : HasBase string sty := SBase.
Instance Sty_HasArrow : HasArrow sty := SArrow.
Instance Sty_EquDec : EquDec sty := sty_eq_dec.

Instance Sty_LambdaType : LambdaType string sty Sty_HasBase Sty_HasArrow (eqB := eq).
Proof.
  split.
  - exact eq_equivalence.
  - exact eq_equivalence.
  - intros x y H. rewrite H. reflexivity.
  - intros a a' Ha c c' Hc. rewrite Ha, Hc. reflexivity.
  - intros x y H. injection H as H. exact H.
  - intros A A' C C' H. injection H as H1 H2. split; assumption.
  - intros x A C H. discriminate H.
Qed.
