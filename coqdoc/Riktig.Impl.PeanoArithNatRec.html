From Riktig.Structure Require Export PeanoArith NatRec.
From Riktig.Theory Require Export NatRecFold.

Any Peano numbers with recursion have their arithmetic: one is the successor of zero, addition counts the second summand's successors onto the first, and multiplication adds the first factor as many times as the second says. The recursion equations are the PeanoArith laws, and respecting equality is natrec's.
Section PeanoArithNatRec.
  Context {N : Type} {eq0 : EqRel N} {zero0 : Zero N} {succ0 : HasSucc N} {P : Peano N zero0 succ0}
    {natrec0 : HasNatRec N} {NR : NatRec N natrec0}.

  #[export] Instance NatRec_One : One N := succ 0.
  #[export] Instance NatRec_HasAdd : HasAdd N := fun a b ⇒ natrec N a (fun _ acc ⇒ succ acc) b.
  #[export] Instance NatRec_HasMult : HasMult N := fun a b ⇒ natrec N 0 (fun _ acc ⇒ acc + a) b.

  #[export] Instance NatRec_add_proper : Proper (equ ==> equ ==> equ) (add : N → N → N) :=
    fun a a' Ha b b' Hb ⇒
      transitivity (natrec_init N _ _ a a' (fun _ acc ⇒ succ acc)
                      (fun _ _ _ x y Hxy ⇒ Peano.succ_proper x y Hxy) Ha b)
                   (NatRec.natrec_proper N _ _ a' (fun _ acc ⇒ succ acc)
                      (fun _ _ _ x y Hxy ⇒ Peano.succ_proper x y Hxy) b b' Hb).

  #[export] Instance NatRec_PeanoArith : PeanoArith N zero0 succ0 NatRec_One NatRec_HasAdd NatRec_HasMult := {|
    PeanoArith.peano := P;
    PeanoArith.add_proper := NatRec_add_proper;
    PeanoArith.mul_proper := fun a a' Ha b b' Hb ⇒
      transitivity (natrec_ext N _ _ 0 (fun _ acc ⇒ acc + a) (fun _ acc ⇒ acc + a')
                      (fun _ _ _ x y Hxy ⇒ NatRec_add_proper x y Hxy a a (reflexivity a))
                      (fun _ _ _ x y Hxy ⇒ NatRec_add_proper x y Hxy a' a' (reflexivity a'))
                      (fun _ acc ⇒ NatRec_add_proper acc acc (reflexivity acc) a a' Ha) b)
                   (NatRec.natrec_proper N _ _ 0 (fun _ acc ⇒ acc + a')
                      (fun _ _ _ x y Hxy ⇒ NatRec_add_proper x y Hxy a' a' (reflexivity a')) b b' Hb);
    PeanoArith.one_succ := reflexivity (succ 0);
    PeanoArith.add_zero := fun a ⇒ NatRec.natrec_zero N _ _ a _;
    PeanoArith.add_succ := fun a b ⇒ NatRec.natrec_succ N _ _ a _ b;
    PeanoArith.mul_zero := fun a ⇒ NatRec.natrec_zero N _ _ 0 _;
    PeanoArith.mul_succ := fun a b ⇒ NatRec.natrec_succ N _ _ 0 _ b;
  |}.
End PeanoArithNatRec.
