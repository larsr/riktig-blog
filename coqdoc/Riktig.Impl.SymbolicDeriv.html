Section Symbolic.
  Context {F V : Type} `{VS : VectorSpace F V}.

  Inductive Expr : Type :=
  | ExprLinear (g : LinearMap F V V)
  | ExprConst (v : V)
  | ExprSum (e1 e2 : Expr)
  | ExprScaled (c : F) (e : Expr)
  | ExprComp (e1 e2 : Expr).

  Fixpoint denote (e : Expr) : V → V :=
    match e with
    | ExprLinear g ⇒ LinearMap.map g
    | ExprConst v ⇒ fun _ : V ⇒ v
    | ExprSum e1 e2 ⇒ fun u : V ⇒ denote e1 u + denote e2 u
    | ExprScaled c e0 ⇒ fun u : V ⇒ c *: denote e0 u
    | ExprComp e1 e2 ⇒ fun u : V ⇒ denote e1 (denote e2 u)
    end.

The derivative read off the syntax: one clause per law, and the composite clause is where the point moves.
  Fixpoint symbolic_deriv (e : Expr) (a : V) : LinearMap F V V :=
    match e with
    | ExprLinear g ⇒ g
    | ExprConst _ ⇒ 0
    | ExprSum e1 e2 ⇒ symbolic_deriv e1 a + symbolic_deriv e2 a
    | ExprScaled c e0 ⇒ c *: symbolic_deriv e0 a
    | ExprComp e1 e2 ⇒ linear_map_comp (symbolic_deriv e1 (denote e2 a))
                                         (symbolic_deriv e2 a)
    end.

  Fixpoint expr_admissible (adm : LinearMap F V V → Prop) (e : Expr) : Prop :=
    match e with
    | ExprLinear g ⇒ adm g
    | ExprConst _ ⇒ True
    | ExprSum e1 e2 ⇒ expr_admissible adm e1 ∧ expr_admissible adm e2
    | ExprScaled _ e0 ⇒ expr_admissible adm e0
    | ExprComp e1 e2 ⇒ expr_admissible adm e1 ∧ expr_admissible adm e2
    end.

  Lemma denote_proper : ∀ e : Expr, Proper (equ ==> equ) (denote e).
  Proof.
    intros e.
    induction e as [g | v | e1 IH1 e2 IH2 | c e0 IH0 | e1 IH1 e2 IH2];
      intros u w Huw.
    - exact (linear_map_proper g u w Huw).
    - reflexivity.
    - exact (op_proper _ _ (IH1 u w Huw) _ _ (IH2 u w Huw)).
    - exact (scale_proper c c (reflexivity c) _ _ (IH0 u w Huw)).
    - exact (IH1 _ _ (IH2 u w Huw)).
  Qed.

End Symbolic.

Section SymbolicIsADerivative.
  Context {F : Type} `{FF : Field F}.
  Context {Ob : Type} {car : Ob → VectObj F} {D : DerivSpec F Ob car}.

  Theorem symbolic_deriv_is_a_derivative : ∀ (X : Ob)
      (e : Expr F (vo_car (car X))),
    expr_admissible (admissible X X) e →
    ∀ a : vo_car (car X),
      differential X X (denote e) a (symbolic_deriv e a).
  Proof.
    intros X e.
    induction e as [g | v | e1 IH1 e2 IH2 | c e0 IH0 | e1 IH1 e2 IH2];
      intros Hadm a.
    - exact (deriv_of_linear X X g a Hadm).
    - exact (deriv_of_constant X X v a).
    - exact (deriv_of_sum X X (denote e1) (denote e2) a
               (symbolic_deriv e1 a) (symbolic_deriv e2 a)
               (IH1 (proj1 Hadm) a) (IH2 (proj2 Hadm) a)).
    - exact (deriv_of_scaled X X c (denote e0) a (symbolic_deriv e0 a)
               (IH0 Hadm a)).
    - exact (deriv_of_composite X X X (denote e2) (denote e1) a
               (symbolic_deriv e2 a) (symbolic_deriv e1 (denote e2 a))
               (denote_proper e1) (IH2 (proj2 Hadm) a)
               (IH1 (proj1 Hadm) (denote e2 a))).
  Qed.

End SymbolicIsADerivative.

Section DerivativeOperatorsAgree.
  Context {F : Type} `{FF : Field F}.
  Context {Ob : Type} {car : Ob → VectObj F}.

Two operators satisfying the laws agree wherever the laws reach, and not because anyone compared them — exactly as Impl/CliffordDet.v's det_ga_is_detM does not compare the geometric product with the cofactor expansion. The recursion is the common value both are forced onto; differential_unique does the rest. This is what having uniqueness as a law buys, and the reason it is worth assuming when it cannot be proved.
Corollary derivative_operators_agree :
  ∀ (D1 D2 : DerivSpec F Ob car) (X : Ob)
    (e : Expr F (vo_car (car X))) (a : vo_car (car X))
    (L1 L2 : LinearMap F (vo_car (car X)) (vo_car (car X))),
    expr_admissible (admissible (DerivSpec := D1) X X) e →
    expr_admissible (admissible (DerivSpec := D2) X X) e →
    differential (DerivSpec := D1) X X (denote e) a L1 →
    differential (DerivSpec := D2) X X (denote e) a L2 →
    L1 == L2.
Proof.
  intros D1 D2 X e a L1 L2 Hadm1 Hadm2 H1 H2.
  transitivity (symbolic_deriv e a).
  - exact (differential_unique X X (denote e) a L1 (symbolic_deriv e a) H1
             (symbolic_deriv_is_a_derivative (D := D1) X e Hadm1 a)).
  - symmetry.
    exact (differential_unique X X (denote e) a L2 (symbolic_deriv e a) H2
             (symbolic_deriv_is_a_derivative (D := D2) X e Hadm2 a)).
Qed.

End DerivativeOperatorsAgree.


Section SymbolicIsFrechet.
  Context {F : Type} `{OF : OrderedField F}.
  Context {V : Type} {eV : EqRel V} {zV : Zero V} {aV : HasAdd V}
    {nV : HasNeg V} {sV : HasScale F V} {VSV : VectorSpace F V}
    {nmV : HasNorm F V} {NSV : NormedSpace F V}.

The side condition the recursion cannot supply: every linear leaf is bounded.
  Definition expr_bounded (e : Expr F V) : Prop :=
    expr_admissible linear_map_bounded e.

  Theorem symbolic_deriv_is_deriv : ∀ e : Expr F V,
    expr_bounded e → ∀ a : V, is_deriv (denote e) a (symbolic_deriv e a).
  Proof.
    intros e He a.
    exact (symbolic_deriv_is_a_derivative
             (D := Frechet_DerivSpec) (normObj F V) e He a).
  Qed.

End SymbolicIsFrechet.


Section Polynomials.
  Context {F : Type} `{OF : OrderedField F}.

  Add Ring polynomial_ring : (ring_theory_of_Ring (R := F) commutative).

  Inductive Polynomial : Type :=
  | PolyConst (c : F)
  | PolyVar
  | PolySum (p q : Polynomial)
  | PolyProduct (p q : Polynomial)
  | PolyCompose (p q : Polynomial).

  Fixpoint poly_eval (p : Polynomial) (x : F) : F :=
    match p with
    | PolyConst c ⇒ c
    | PolyVar ⇒ x
    | PolySum p1 p2 ⇒ poly_eval p1 x + poly_eval p2 x
    | PolyProduct p1 p2 ⇒ poly_eval p1 x × poly_eval p2 x
    | PolyCompose p1 p2 ⇒ poly_eval p1 (poly_eval p2 x)
    end.

The slope, by the same recursion: the constant rule, the variable rule, the sum rule, the product rule and the chain rule, in that order.
  Fixpoint poly_deriv (p : Polynomial) (x : F) : F :=
    match p with
    | PolyConst _ ⇒ 0
    | PolyVar ⇒ 1
    | PolySum p1 p2 ⇒ poly_deriv p1 x + poly_deriv p2 x
    | PolyProduct p1 p2 ⇒
        poly_deriv p1 x × poly_eval p2 x + poly_eval p1 x × poly_deriv p2 x
    | PolyCompose p1 p2 ⇒ poly_deriv p1 (poly_eval p2 x) × poly_deriv p2 x
    end.


  Lemma lm_scale_map_zero : (lm_scale_map 0 : LinearMap F F F) == 0.
  Proof. intros h. change (0 × h == (0 : F)). ring. Qed.

  #[export] Instance lm_scale_map_proper :
    Proper (equ ==> equ) (lm_scale_map (F := F)).
  Proof. intros c d Hcd h. exact (mul_r_proper h c d Hcd). Qed.

  Lemma lm_scale_map_add : ∀ c d : F,
    (lm_scale_map c + lm_scale_map d : LinearMap F F F) == lm_scale_map (c + d).
  Proof. intros c d h. change (c × h + d × h == (c + d) × h). ring. Qed.

  Lemma is_deriv_identity : ∀ a : F,
    is_deriv (fun x : F ⇒ x) a (lm_scale_map 1).
  Proof.
    intros a.
    apply (is_deriv_fun_equ (scale_map 1) (fun x : F ⇒ x) a (lm_scale_map 1)).
    - intros x. exact (left_identity x).
    - exact (is_deriv_scale_map 1 a).
  Qed.

  Lemma poly_eval_proper : ∀ p : Polynomial,
    Proper (equ ==> equ) (poly_eval p).
  Proof.
    intros p.
    induction p as [c | | p1 IH1 p2 IH2 | p1 IH1 p2 IH2 | p1 IH1 p2 IH2];
      intros x y Hxy.
    - reflexivity.
    - exact Hxy.
    - exact (op_proper _ _ (IH1 x y Hxy) _ _ (IH2 x y Hxy)).
    - exact (op_proper _ _ (IH1 x y Hxy) _ _ (IH2 x y Hxy)).
    - exact (IH1 _ _ (IH2 x y Hxy)).
  Qed.

The slope the recursion computes is the Fréchet derivative's scale map. Five citations and no analysis.
  Theorem poly_deriv_is_deriv : ∀ (p : Polynomial) (a : F),
    is_deriv (poly_eval p) a (lm_scale_map (poly_deriv p a)).
  Proof.
    intros p.
    induction p as [c | | p1 IH1 p2 IH2 | p1 IH1 p2 IH2 | p1 IH1 p2 IH2];
      intros a.
    - exact (is_deriv_L_equ (fun _ : F ⇒ c) a 0 (lm_scale_map 0)
               (symmetry lm_scale_map_zero) (deriv_const c a)).
    - exact (is_deriv_identity a).
    - exact (is_deriv_L_equ (fun x : F ⇒ poly_eval p1 x + poly_eval p2 x) a
               (lm_scale_map (poly_deriv p1 a) + lm_scale_map (poly_deriv p2 a))
               (lm_scale_map (poly_deriv p1 a + poly_deriv p2 a))
               (lm_scale_map_add (poly_deriv p1 a) (poly_deriv p2 a))
               (deriv_add (poly_eval p1) (poly_eval p2) a _ _ (IH1 a) (IH2 a))).
    - exact (deriv_mul (poly_eval p1) (poly_eval p2) a
               (poly_deriv p1 a) (poly_deriv p2 a) (IH1 a) (IH2 a)).
    - exact (is_deriv_L_equ (fun x : F ⇒ poly_eval p1 (poly_eval p2 x)) a
               (linear_map_comp (lm_scale_map (poly_deriv p1 (poly_eval p2 a)))
                  (lm_scale_map (poly_deriv p2 a)))
               (lm_scale_map (poly_deriv p1 (poly_eval p2 a) × poly_deriv p2 a))
               (lm_comp_scale_scale (poly_deriv p2 a)
                  (poly_deriv p1 (poly_eval p2 a)))
               (deriv_comp (poly_eval p2) (poly_eval p1) a _ _
                  (poly_eval_proper p1) (IH2 a) (IH1 (poly_eval p2 a)))).
  Qed.

End Polynomials.
