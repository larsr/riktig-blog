A model of the untyped lambda calculus: a carrier T whose conversion is the setoid equality, together with a carrier Sb of substitutions. Substitution is an operation with equations — the variable computations under sub_cons, sub_shift and sub_up, the cancellations, and value-guarded beta — so consequences follow equationally, with no induction over any syntax.
Module LambdaModel.
  Class LambdaModel (T Sb : Type) {Te : EqRel T} {var : HasVar T}
      {lam : HasLam T} {app : HasApp T} {substitute : HasSub T Sb}
      {sub_id : HasSubId Sb} {sub_shift : HasSubShift Sb}
      {sub_cons : HasSubCons T Sb} {sub_up : HasSubUp Sb}
      {val : HasVal T} := {
    equivalence :: Equivalence Te;
    lam_proper :: Proper (equ ==> equ) lam;
    app_proper :: Proper (equ ==> equ ==> equ) app;
    
    shift_respects : ∀ t t' : T, t == t' → t.[sub_shift] == t'.[sub_shift];
    substitute_id : ∀ t : T, t.[sub_id] == t;
    var_zero_cons : ∀ (u : T) (s : Sb), (var O).[u .: s] == u;
    app_substitute : ∀ (f a : T) (s : Sb),
        (app f a).[s] == app (f.[s]) (a.[s]);
    lam_substitute : ∀ (b : T) (s : Sb), (lam b).[s] == lam (b.[sub_up s]);
    var_zero_up : ∀ s : Sb, (var O).[sub_up s] == var O;
    var_succ_up : ∀ (n : nat) (s : Sb),
        (var (S n)).[sub_up s] == ((var n).[s]).[sub_shift];
    var_shift : ∀ n : nat, (var n).[sub_shift] == var (S n);
    shift_cons : ∀ (t u : T) (s : Sb), (t.[sub_shift]).[u .: s] == t.[s];
    shift_up : ∀ (t : T) (s : Sb),
        (t.[sub_shift]).[sub_up s] == (t.[s]).[sub_shift];
    beta : ∀ b u : T, val u → app (lam b) u == b.[u .: sub_id];
    val_lam : ∀ b : T, val (lam b);
  }.
  Hint Mode LambdaModel ! - - - - - - - - - - - : typeclass_instances.
End LambdaModel.

The Church encodings, spelled once over any carrier with the term-forming operations (Theory/Lambda.v derives their laws).

Definition church_true {T : Type} {var : HasVar T} {lam : HasLam T} : T :=
  lam (lam (var (S O))).

Definition church_false {T : Type} {var : HasVar T} {lam : HasLam T} : T :=
  lam (lam (var O)).

Definition church_if {T : Type} {app : HasApp T} (b g h : T) : T :=
  app (app b g) h.

Definition church_pair {T : Type} {var : HasVar T} {lam : HasLam T}
    {app : HasApp T} : T :=
  lam (lam (lam (app (app (var O) (var (S (S O)))) (var (S O))))).
