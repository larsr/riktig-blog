From Riktig.Structure Require Export Naturals.

The lambda calculus: terms T built from three constructors -- variables indexed by the naturals N, abstraction and application with a reduction t -->* u. Reduction is a congruent preorder that contracts a redex (beta) and passes under substitution; the substitution laws say how a substitution meets each constructor. Two terms are equal when they reduce to a common term, and variables stay distinct. That this == is transitive -- an equivalence at all -- is exactly confluence, the last law.
Module LambdaCalculus.
  Class LambdaCalculus (N T Sb : Type) `{Naturals N} {eqT : EqRel T}
      (var_ : HasVar N T) (lam_ : HasLam T) (app_ : HasApp T) (subst_ : HasSubst T Sb)
      (id_ : HasSubId Sb) (shift_ : HasSubShift Sb) (cons_ : HasSubCons T Sb) (up_ : HasSubUp Sb)
      (reduction : HasReduces T) := {
    
    equ_join : ∀ t u : T, t == u ↔ (∃ w : T, t -->* w ∧ u -->* w);
    
    reduces_refl : ∀ t : T, t -->* t;
    reduces_trans : ∀ t u v : T, t -->* u → u -->* v → t -->* v;
    reduces_lam : ∀ b b' : T, b -->* b' → lam b -->* lam b';
    reduces_app : ∀ f f' a a' : T, f -->* f' → a -->* a' → app f a -->* app f' a';
    reduces_subst : ∀ (t t' : T) (s : Sb), t -->* t' → t.[s] -->* t'.[s];
    
    beta : ∀ b u : T, app (lam b) u -->* b.[u .: sub_id];
    
    var_injective : ∀ n m : N, (var n : T) == var m → n == m;
    
    subst_id : ∀ t : T, t.[sub_id] == t;
    var_zero_cons : ∀ (u : T) (s : Sb), (var 0 : T).[u .: s] == u;
    var_succ_cons : ∀ (n : N) (u : T) (s : Sb), (var (succ n) : T).[u .: s] == (var n : T).[s];
    var_shift : ∀ n : N, (var n : T).[sub_shift] == var (succ n);
    var_zero_up : ∀ s : Sb, (var 0 : T).[sub_up s] == var 0;
    var_succ_up : ∀ (n : N) (s : Sb), (var (succ n) : T).[sub_up s] == ((var n : T).[s]).[sub_shift];
    app_subst : ∀ (f a : T) (s : Sb), (app f a).[s] == app (f.[s]) (a.[s]);
    lam_subst : ∀ (b : T) (s : Sb), (lam b).[s] == lam (b.[sub_up s]);
    shift_cons : ∀ (t u : T) (s : Sb), (t.[sub_shift]).[u .: s] == t.[s];
    shift_up : ∀ (t : T) (s : Sb), (t.[sub_shift]).[sub_up s] == (t.[s]).[sub_shift];
    
    confluence : ∀ t u v : T, t -->* u → t -->* v → ∃ w : T, u -->* w ∧ v -->* w;
  }.
  Hint Mode LambdaCalculus - ! - - - - - - - - - - - - - - - - - - : typeclass_instances.
End LambdaCalculus.
