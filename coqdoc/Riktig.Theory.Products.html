The arrow between products that acts as f on the first factor and as g on the second.
Notation "f ×× g" := (lift (f ∘ pi1) (g ∘ pi2)) (at level 41, left associativity).

Section ProductsTheory.
  Context {O : Type} `{Products O}.

  Theorem lift_comp : ∀ (x y a b : O) (f : y ~> a) (g : y ~> b) (h : x ~> y),
    lift f g ∘ h == lift (f ∘ h) (g ∘ h).
  Proof.
    intros x y a b f g h. apply lift_unique.
    - calc (pi1 ∘ (lift f g ∘ h))
        == ((pi1 ∘ lift f g) ∘ h) { comp_assoc }
        == (f ∘ h) { pi1_lift }.
    - calc (pi2 ∘ (lift f g ∘ h))
        == ((pi2 ∘ lift f g) ∘ h) { comp_assoc }
        == (g ∘ h) { pi2_lift }.
  Qed.

  Theorem lift_pi : ∀ a b : O, lift pi1 pi2 == cat_id (x := a ⨯ b).
  Proof.
    intros a b. symmetry. apply lift_unique; apply right_identity.
  Qed.

  Theorem pi1_par : ∀ (a b c d : O) (f : a ~> c) (g : b ~> d), pi1 ∘ (f ×× g) == f ∘ pi1.
  Proof. intros a b c d f g. apply pi1_lift. Qed.

  Theorem pi2_par : ∀ (a b c d : O) (f : a ~> c) (g : b ~> d), pi2 ∘ (f ×× g) == g ∘ pi2.
  Proof. intros a b c d f g. apply pi2_lift. Qed.

  Theorem par_lift : ∀ (x a b c d : O) (f : a ~> c) (g : b ~> d) (h : x ~> a) (k : x ~> b),
    (f ×× g) ∘ lift h k == lift (f ∘ h) (g ∘ k).
  Proof.
    intros x a b c d f g h k.
    calc ((f ×× g) ∘ lift h k)
      == (lift ((f ∘ pi1) ∘ lift h k) ((g ∘ pi2) ∘ lift h k)) { lift_comp }
      == (lift (f ∘ (pi1 ∘ lift h k)) ((g ∘ pi2) ∘ lift h k)) { comp_assoc }
      == (lift (f ∘ (pi1 ∘ lift h k)) (g ∘ (pi2 ∘ lift h k))) { comp_assoc }
      == (lift (f ∘ h) (g ∘ (pi2 ∘ lift h k))) { pi1_lift }
      == (lift (f ∘ h) (g ∘ k)) { pi2_lift }.
  Qed.

  Theorem par_id : ∀ a b : O, (cat_id ×× cat_id) == cat_id (x := a ⨯ b).
  Proof.
    intros a b. symmetry. apply lift_unique.
    - calc (pi1 (a := a) (b := b) ∘ cat_id) == (pi1 (a := a) (b := b)) { right_identity } == (cat_id ∘ pi1 (a := a) (b := b)) { left_identity }.
    - calc (pi2 (a := a) (b := b) ∘ cat_id) == (pi2 (a := a) (b := b)) { right_identity } == (cat_id ∘ pi2 (a := a) (b := b)) { left_identity }.
  Qed.

  Theorem par_comp : ∀ (a b c d e e' : O) (f : c ~> e) (g : d ~> e') (h : a ~> c) (k : b ~> d),
    (f ×× g) ∘ (h ×× k) == (f ∘ h) ×× (g ∘ k).
  Proof.
    intros a b c d e e' f g h k.
    calc ((f ×× g) ∘ (h ×× k))
      == (lift (f ∘ (h ∘ pi1)) (g ∘ (k ∘ pi2))) { par_lift }
      == (lift ((f ∘ h) ∘ pi1) (g ∘ (k ∘ pi2))) { (comp_assoc _ _ _ _ f h pi1) }
      == (lift ((f ∘ h) ∘ pi1) ((g ∘ k) ∘ pi2)) { (comp_assoc _ _ _ _ g k pi2) }.
  Qed.

Swapping the factors twice is the identity.
  Theorem swap_swap : ∀ a b : O,
    (lift pi2 pi1 : b ⨯ a ~> a ⨯ b) ∘ (lift pi2 pi1 : a ⨯ b ~> b ⨯ a) == cat_id.
  Proof.
    intros a b.
    calc ((lift pi2 pi1 : b ⨯ a ~> a ⨯ b) ∘ (lift pi2 pi1 : a ⨯ b ~> b ⨯ a))
      == (lift (pi2 ∘ lift pi2 pi1) (pi1 ∘ (lift pi2 pi1 : a ⨯ b ~> b ⨯ a))) { lift_comp }
      == (lift pi1 (pi1 ∘ (lift pi2 pi1 : a ⨯ b ~> b ⨯ a))) { pi2_lift }
      == (lift (pi1 (a := a) (b := b)) pi2) { pi1_lift }
      == (cat_id (x := a ⨯ b)) { lift_pi }.
  Qed.
End ProductsTheory.
