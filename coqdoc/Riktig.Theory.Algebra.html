Section AlgebraTheory.
  Context {F A : Type} `{Algebra F A}.

  Theorem smul_one_mul : ∀ (a : F) (x : A), (a *: 1) × x == a *: x.
  Proof.
    intros a x.
    calc ((a *: (1 : A)) × x)
      == (a *: (1 × x)) { smul_mul_assoc }
      == (a *: x) { (left_identity (action := mult) x) }.
  Qed.

  Theorem mul_smul_one : ∀ (a : F) (x : A), x × (a *: 1) == a *: x.
  Proof.
    intros a x.
    calc (x × (a *: (1 : A)))
      == (a *: (x × 1)) { mul_smul_comm }
      == (a *: x) { (right_identity (action := mult) x) }.
  Qed.

  Theorem smul_one_mul_smul_one : ∀ a b : F, (a *: (1 : A)) × (b *: 1) == (a × b) *: 1.
  Proof.
    intros a b.
    calc ((a *: (1 : A)) × (b *: 1))
      == (a *: (b *: (1 : A))) { smul_one_mul }
      == ((a × b) *: (1 : A)) { associative }.
  Qed.

  Section SquareRootOfMinusOne.
    Context (B : A) (HB : B × B == - (1)).

    Lemma smul_one_mul_smul : ∀ c d : F, (c *: (1 : A)) × (d *: B) == (c × d) *: B.
    Proof.
      intros c d.
      calc ((c *: (1 : A)) × (d *: B))
        == (c *: (d *: B)) { smul_one_mul }
        == ((c × d) *: B) { (associative (op := mult) (action := scale) c d B) }.
    Qed.

    Lemma smul_mul_smul_one : ∀ c d : F, (c *: B) × (d *: (1 : A)) == (c × d) *: B.
    Proof.
      intros c d.
      calc ((c *: B) × (d *: (1 : A)))
        == (d *: (c *: B)) { mul_smul_one }
        == ((d × c) *: B) { (associative (op := mult) (action := scale) d c B) }
        == ((c × d) *: B) { (commutative (op := mult) d c) }.
    Qed.

    Lemma smul_mul_smul : ∀ c d : F, (c *: B) × (d *: B) == - ((c × d) *: (1 : A)).
    Proof.
      intros c d.
      calc ((c *: B) × (d *: B))
        == (c *: (B × (d *: B))) { (smul_mul_assoc c B (d *: B)) }
        == (c *: (d *: (B × B))) { (mul_smul_comm d B B) }
        == ((c × d) *: (B × B)) { (associative (op := mult) (action := scale) c d (B × B)) }
        == ((c × d) *: (- (1 : A))) { HB }
        == (- ((c × d) *: (1 : A))) { (smul_neg (c × d) (1 : A)) }.
    Qed.

The product of two combinations of 1 and B.
    Lemma combination_mul_expand : ∀ a b c d : F,
      (a *: (1 : A) + b *: B) × (c *: 1 + d *: B)
        == ((a × c) *: (1 : A) + (a × d) *: B) + ((b × c) *: B + - ((b × d) *: (1 : A))).
    Proof.
      intros a b c d.
      calc ((a *: (1 : A) + b *: B) × (c *: 1 + d *: B))
        == (((a *: (1 : A)) × (c *: 1) + (a *: (1 : A)) × (d *: B))
            + ((b *: B) × (c *: (1 : A)) + (b *: B) × (d *: B))) { add_mul_add }
        == (((a × c) *: (1 : A) + (a *: (1 : A)) × (d *: B))
            + ((b *: B) × (c *: (1 : A)) + (b *: B) × (d *: B))) { smul_one_mul_smul_one }
        == (((a × c) *: (1 : A) + (a × d) *: B)
            + ((b *: B) × (c *: (1 : A)) + (b *: B) × (d *: B))) { smul_one_mul_smul }
        == (((a × c) *: (1 : A) + (a × d) *: B) + ((b × c) *: B + (b *: B) × (d *: B)))
             { smul_mul_smul_one }
        == (((a × c) *: (1 : A) + (a × d) *: B) + ((b × c) *: B + - ((b × d) *: (1 : A))))
             { smul_mul_smul }.
    Qed.

    Lemma combination_mul : ∀ a b c d : F,
      (a *: (1 : A) + b *: B) × (c *: 1 + d *: B)
        == (a × c + - (b × d)) *: (1 : A) + (a × d + b × c) *: B.
    Proof.
      intros a b c d.
      calc ((a *: (1 : A) + b *: B) × (c *: 1 + d *: B))
        == (((a × c) *: (1 : A) + (a × d) *: B) + ((b × c) *: B + - ((b × d) *: (1 : A))))
             { combination_mul_expand }
        == (((a × c) *: (1 : A) + (a × d) *: B) + (- ((b × d) *: (1 : A)) + (b × c) *: B))
             { (commutative (op := add) ((b × c) *: B) (- ((b × d) *: (1 : A)))) }
        == (((a × c) *: (1 : A) + - ((b × d) *: 1)) + ((a × d) *: B + (b × c) *: B))
             { (add_add_add_comm ((a × c) *: (1 : A)) ((a × d) *: B) (- ((b × d) *: (1 : A))) ((b × c) *: B)) }
        == ((a × c + - (b × d)) *: (1 : A) + ((a × d) *: B + (b × c) *: B))
             { (sub_smul (a × c) (b × d) (1 : A)) }
        == ((a × c + - (b × d)) *: (1 : A) + (a × d + b × c) *: B)
             { (right_distributive (op := scale) (a × d) (b × c) B) }.
    Qed.
  End SquareRootOfMinusOne.
