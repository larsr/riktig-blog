  Lemma scale_mul_l : ∀ (a : F) (x y : A), (a *: x) × y == a *: (x × y).
  Proof. intros a x y. exact (homogeneous[fun z ⇒ z × y] a x). Qed.

  Lemma scale_mul_r : ∀ (a : F) (x y : A), x × (a *: y) == a *: (x × y).
  Proof. intros a x y. exact (homogeneous[fun z ⇒ x × z] a y). Qed.

  Lemma scale_mul_scale : ∀ (a b : F) (x y : A),
    (a *: x) × (b *: y) == (a × b) *: (x × y).
  Proof.
    intros a b x y.
    calc (((a *: x) × (b *: y)) : A)
      == ((a *: (x × (b *: y))) : A) { scale_mul_l }
      == ((a *: (b *: (x × y))) : A) { scale_mul_r }
      == (((a × b) *: (x × y)) : A) { associative }.
  Qed.

Two scalings of one element commute, which discharges the guard on cexp_add for arguments on a common line.
  Lemma collinear_commute : ∀ (t u : F) (x : A),
    (t *: x) × (u *: x) == (u *: x) × (t *: x).
  Proof.
    intros t u x.
    calc (((t *: x) × (u *: x)) : A)
      == (((t × u) *: (x × x)) : A) { scale_mul_scale }
      == (((u × t) *: (x × x)) : A) { (mul_commutative t u) }
      == (((u *: x) × (t *: x)) : A) { scale_mul_scale }.
  Qed.


  Lemma scale_one_mul_scale : ∀ (a c : F) (x : A),
    (a *: (1 : A)) × (c *: x) == (a × c) *: x.
  Proof.
    intros a c x.
    calc (((a *: (1 : A)) × (c *: x)) : A)
      == (((a × c) *: ((1 : A) × x)) : A) { scale_mul_scale }
      == (((a × c) *: x) : A) { left_identity }.
  Qed.

  Lemma scale_mul_scale_one : ∀ (a c : F) (x : A),
    (a *: x) × (c *: (1 : A)) == (a × c) *: x.
  Proof.
    intros a c x.
    calc (((a *: x) × (c *: (1 : A))) : A)
      == (((a × c) *: (x × (1 : A))) : A) { scale_mul_scale }
      == (((a × c) *: x) : A) { right_identity }.
  Qed.

  Lemma root_mul_root : ∀ (b d : F) (e : A),
    e × e == - (1) → (b *: e) × (d *: e) == (- (b × d)) *: (1 : A).
  Proof.
    intros b d e Hroot.
    calc (((b *: e) × (d *: e)) : A)
      == (((b × d) *: (e × e)) : A) { scale_mul_scale }
      == (((b × d) *: (- (1 : A))) : A) { Hroot }
      == ((- ((b × d) *: (1 : A))) : A) { scale_neg }
      == (((- (b × d)) *: (1 : A)) : A) { neg_scale }.
  Qed.

Two anticommuting elements whose squares are on the scalar line have a product whose square is there too, at minus the product of the two scalars. The bivector facts — e12 × e12 == - (1) and its relatives — are this theorem, and none of them is about a dimension.
  Theorem anticommuting_square : ∀ (x y : A) (p q : F),
    x × y == - (y × x) → x × x == p *: 1 → y × y == q *: 1 →
    (x × y) × (x × y) == (- (p × q)) *: 1.
  Proof.
    intros x y p q Hanti Hx Hy.
    assert (Hanti' : y × x == - (x × y)).
    { rewrite Hanti. symmetry. exact (neg_involutive (y × x)). }
    assert (Hinner : (x × y) × (x × y) == - ((x × x) × (y × y))).
    { calc (((x × y) × (x × y)) : A)
        == ((x × (y × (x × y))) : A) { associative }
        == ((x × ((y × x) × y)) : A) { associative }
        == ((x × ((- (x × y)) × y)) : A) { Hanti' }
        == ((x × (- ((x × y) × y))) : A) { mul_neg_l }
        == ((x × (- (x × (y × y)))) : A) { associative }
        == ((- (x × (x × (y × y)))) : A) { mul_neg_r }
        == ((- ((x × x) × (y × y))) : A) { associative }. }
    calc (((x × y) × (x × y)) : A)
      == ((- ((x × x) × (y × y))) : A) { Hinner }
      == ((- ((p *: (1 : A)) × (y × y))) : A) { Hx }
      == ((- ((p *: (1 : A)) × (q *: 1))) : A) { Hy }
      == ((- ((p × q) *: (1 : A))) : A) { scale_one_mul_scale }
      == (((- (p × q)) *: (1 : A)) : A) { neg_scale }.
  Qed.

  Lemma scale_one_mul_two_term : ∀ (a c d : F) (x : A),
    (a *: (1 : A)) × (c *: 1 + d *: x) == (a × c) *: 1 + (a × d) *: x.
  Proof.
    intros a c d x.
    calc (((a *: (1 : A)) × (c *: 1 + d *: x)) : A)
      == (((a *: (1 : A)) × (c *: 1) + (a *: (1 : A)) × (d *: x)) : A)
         { left_distributive }
      == (((a × c) *: (1 : A) + (a *: (1 : A)) × (d *: x)) : A)
         { scale_one_mul_scale }
      == (((a × c) *: (1 : A) + (a × d) *: x) : A)
         { scale_one_mul_scale }.
  Qed.

  Lemma root_mul_two_term : ∀ (b c d : F) (e : A),
    e × e == - (1) →
    (b *: e) × (c *: 1 + d *: e) == (b × c) *: e + (- (b × d)) *: 1.
  Proof.
    intros b c d e Hroot.
    calc (((b *: e) × (c *: 1 + d *: e)) : A)
      == (((b *: e) × (c *: (1 : A)) + (b *: e) × (d *: e)) : A)
         { left_distributive }
      == (((b × c) *: e + (b *: e) × (d *: e)) : A)
         { scale_mul_scale_one }
      == (((b × c) *: e + (- (b × d)) *: (1 : A)) : A)
         { (root_mul_root b d e Hroot) }.
  Qed.

The unit as a point of the plane: coefficient one on 1 and zero on the other coordinate.
  Lemma scale_one_scale_zero : ∀ x : A,
    (1 : F) *: (1 : A) + (0 : F) *: x == 1.
  Proof.
    intros x.
    calc (((1 : F) *: (1 : A) + (0 : F) *: x) : A)
      == (((1 : A) + (0 : F) *: x) : A) { left_identity }
      == (((1 : A) + (0 : A)) : A) { zero_scale }
      == ((1 : A)) { right_identity }.
  Qed.

  Lemma two_term_collect : ∀ (p q r s : F) (x : A),
    (p *: 1 + q *: x) + (r *: x + s *: 1) == (p + s) *: 1 + (q + r) *: x.
  Proof.
    intros p q r s x.
    calc (((p *: (1 : A) + q *: x) + (r *: x + s *: 1)) : A)
      == (((p *: (1 : A) + s *: 1) + (q *: x + r *: x)) : A)
         { add_rotate }
      == (((p + s) *: (1 : A) + (q *: x + r *: x)) : A)
         { right_distributive }
      == (((p + s) *: (1 : A) + (q + r) *: x) : A)
         { right_distributive }.
  Qed.

The complex multiplication table, over any square root of - (1).
  Theorem two_term_product : ∀ (a b c d : F) (e : A),
    e × e == - (1) →
    (a *: 1 + b *: e) × (c *: 1 + d *: e)
    == (a × c + - (b × d)) *: 1 + (a × d + b × c) *: e.
  Proof.
    intros a b c d e Hroot.
    calc (((a *: 1 + b *: e) × (c *: 1 + d *: e)) : A)
      == (((a *: (1 : A)) × (c *: 1 + d *: e)
             + (b *: e) × (c *: 1 + d *: e)) : A)
         { right_distributive }
      == ((((a × c) *: (1 : A) + (a × d) *: e)
             + (b *: e) × (c *: 1 + d *: e)) : A)
         { scale_one_mul_two_term }
      == ((((a × c) *: (1 : A) + (a × d) *: e)
             + ((b × c) *: e + (- (b × d)) *: 1)) : A)
         { (root_mul_two_term b c d e Hroot) }
      == (((a × c + - (b × d)) *: (1 : A) + (a × d + b × c) *: e) : A)
         { two_term_collect }.
  Qed.

End ScalarPlane.
