Library Riktig.Lib.Coq.Bool

From Coq Require Import Arith Lia.

bool numbered by nat: false is 0, true is 1.
Definition bool_index (b : bool) : nat := if b then 1 else 0.
Definition bool_elem (k : nat) : bool := match k with 0 ⇒ false | _ ⇒ true end.

Lemma bool_index_lt : ∀ b : bool, bool_index b ≤ 2 ∧ bool_index b ≠ 2.
Proof. intros [|]; simpl; lia. Qed.

Lemma bool_elem_index : ∀ b : bool, bool_elem (bool_index b) = b.
Proof. intros [|]; reflexivity. Qed.

Lemma bool_index_elem : ∀ k : nat, k ≤ 2 ∧ k ≠ 2 → bool_index (bool_elem k) = k.
Proof. intros [| [| k]] [H1 H2]; [ reflexivity | reflexivity | lia ]. Qed.
