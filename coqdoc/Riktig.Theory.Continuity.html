From Coq Require Export Morphisms.
From Riktig Require Export Theory.LinearMap Theory.Limits.

Section Continuity.
  Context {F U V : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.

  Definition near (a : U) (P : U → Prop) : Prop :=
    ∃ delta : F, 0 < delta ∧ ∀ u : U, ‖ u + - a ‖ < delta → P u.

  Lemma near_weakens : ∀ a : U, ev_weakens (near a).
  Proof.
    intros a P Q HPQ [delta [Hdelta Hclose]].
    ∃ delta.
    split.
    - exact Hdelta.
    - intros u Hu. exact (HPQ u (Hclose u Hu)).
  Qed.

  Lemma near_meets : ∀ a : U, ev_meets (near a).
  Proof.
    intros a P Q [d1 [Hd1 HP]] [d2 [Hd2 HQ]].
    destruct (pos_min_exists d1 d2 Hd1 Hd2) as [d [Hd [Hdd1 Hdd2]]].
    ∃ d.
    split.
    - exact Hd.
    - intros u Hu.
      split.
      + exact (HP u (lt_le_trans (‖ u + - a ‖) d d1 Hu Hdd1)).
      + exact (HQ u (lt_le_trans (‖ u + - a ‖) d d2 Hu Hdd2)).
  Qed.

  Lemma near_inhabited : ∀ a : U, ev_inhabited (near a).
  Proof.
    intros a P [delta [Hdelta Hclose]].
    ∃ a.
    exact (Hclose a (norm_sub_self_lt a delta Hdelta)).
  Qed.

  Definition limit_at (f : U → V) (a : U) (L : V) : Prop :=
    tendsto (near a) f L.

  Theorem limit_at_unique : ∀ (f : U → V) (a : U) (L1 L2 : V),
    limit_at f a L1 → limit_at f a L2 → L1 == L2.
  Proof.
    intros f a.
    exact (tendsto_unique (near a) (near_meets a) (near_inhabited a) f).
  Qed.

  Definition continuous_at (f : U → V) (a : U) : Prop :=
    limit_at f a (f a).

  Definition continuous (f : U → V) : Prop :=
    ∀ a : U, continuous_at f a.

  Theorem limit_at_add : ∀ (f g : U → V) (a : U) (L M : V),
    limit_at f a L → limit_at g a M →
    limit_at (fun u : U ⇒ f u + g u) a (L + M).
  Proof.
    intros f g a.
    exact (tendsto_add (near a) (near_weakens a) (near_meets a) f g).
  Qed.

  Theorem limit_at_scale : ∀ (c : F) (f : U → V) (a : U) (L : V),
    limit_at f a L → limit_at (fun u : U ⇒ c *: f u) a (c *: L).
  Proof.
    intros c f a.
    exact (tendsto_scale (near a) (near_weakens a) c f).
  Qed.

  Theorem limit_at_neg : ∀ (f : U → V) (a : U) (L : V),
    limit_at f a L → limit_at (fun u : U ⇒ - f u) a (- L).
  Proof.
    intros f a.
    exact (tendsto_neg (near a) (near_weakens a) f).
  Qed.

  Lemma linear_map_norm_sub : ∀ (f : LinearMap F U V) (u v : U),
    ‖ f u + - f v ‖ == ‖ f (u + - v) ‖.
  Proof.
    intros f u v.
    calc (‖ f u + - f v ‖)
      == (‖ f u + f (- v) ‖) { linear_map_neg }
      == (‖ f (u + - v) ‖) { linear_map_additive }.
  Qed.

  Definition linear_map_bounded (f : LinearMap F U V) : Prop :=
    ∃ C : F, 0 ≤ C ∧ ∀ u : U, ‖ f u ‖ ≤ C × ‖ u ‖.

  Theorem bounded_continuous : ∀ f : LinearMap F U V,
    linear_map_bounded f → continuous (LinearMap.map f).
  Proof.
    intros f [C [HC Hb]] a eps Heps.
    ∃ (eps × inv (1 + C)).
    split.
    - exact (pad_div_pos eps C Heps HC).
    - intros u Hu.
      rewrite (linear_map_norm_sub f u a).
      apply (le_lt_trans _ (C × ‖ u + - a ‖) eps).
      + exact (Hb (u + - a)).
      + apply mul_lt_of_lt_pad_div.
        × exact HC.
        × apply nonneg.
        × exact Hu.
  Qed.

  Theorem continuous_at_seq_lim : ∀ (f : U → V) (a : U) (x : nat → U),
    continuous_at f a → seq_lim x a →
    seq_lim (fun n : nat ⇒ f (x n)) (f a).
  Proof.
    intros f a x Hf Hx eps Heps.
    destruct (Hf eps Heps) as [delta [Hdelta Hclose]].
    destruct (Hx delta Hdelta) as [N HN].
    ∃ N.
    intros n Hn.
    apply Hclose.
    apply HN.
    exact Hn.
  Qed.

End Continuity.

Instance limit_at_proper {F U V : Type} `{NormedSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V}
    {sV : HasScale F V} {VSV : VectorSpace F V} {nmV : HasNorm F V}
    {NSV : NormedSpace F V} :
  Proper (pointwise_relation U equ ==> equ ==> equ ==> iff)
    (limit_at (F:=F) (U:=U) (V:=V)).
Proof.
  intros f g Hfg a b Hab L M HLM.
  split.
  - intros Hf eps Heps.
    destruct (Hf eps Heps) as [delta [Hdelta Hclose]].
    ∃ delta.
    split.
    + exact Hdelta.
    + intros u Hu.
      rewrite <- (Hfg u), <- HLM.
      apply Hclose.
      rewrite Hab.
      exact Hu.
  - intros Hg eps Heps.
    destruct (Hg eps Heps) as [delta [Hdelta Hclose]].
    ∃ delta.
    split.
    + exact Hdelta.
    + intros u Hu.
      rewrite (Hfg u), HLM.
      apply Hclose.
      rewrite <- Hab.
      exact Hu.
Qed.

Section ContinuityComp.
  Context {F U V W : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VSW : VectorSpace F W} {nmW : HasNorm F W} {NSW : NormedSpace F W}.

  Theorem continuous_comp : ∀ (f : U → V) (g : V → W) (a : U),
    continuous_at f a → continuous_at g (f a) →
    continuous_at (fun u : U ⇒ g (f u)) a.
  Proof.
    intros f g a Hf Hg eps Heps.
    destruct (Hg eps Heps) as [d1 [Hd1 Hg1]].
    destruct (Hf d1 Hd1) as [d2 [Hd2 Hf2]].
    ∃ d2.
    split.
    - exact Hd2.
    - intros u Hu.
      apply Hg1.
      apply Hf2.
      exact Hu.
  Qed.

End ContinuityComp.
