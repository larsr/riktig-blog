Lemma bcount_bfull : ∀ n : nat, bcount (bfull n) = n.
Proof.
  induction n as [| m IH]; [ reflexivity |].
  rewrite bfull_cons, bcount_cons, IH. reflexivity.
Qed.

Lemma bcount_bxor_le : ∀ (n : nat) (A B : Blade n),
  (bcount (bxor A B) ≤ bcount A + bcount B)%nat.
Proof.
  induction n as [| m IH]; intros A B.
  - apply le_0_n.
  - rewrite (blade_S m A), (blade_S m B), bxor_cons, !bcount_cons.
    specialize (IH (btail A) (btail B)).
    destruct (bhead A); destruct (bhead B); simpl; lia.
Qed.

Lemma bcount_of_bis_nil : ∀ (n : nat) (D : Blade n),
  bis_nil D = true → bcount D = O.
Proof.
  intros n D Hnil.
  rewrite (bis_nil_true n D Hnil).
  exact (bcount_bempty n).
Qed.


Definition cl_lift {F : Type} {zF : Zero F} {n : nat} (x : Cl F n)
  : Cl F (S n) := fun D ⇒ if bhead D then 0 else x (btail D).

Section CliffordLift.
  Context {F : Type} `{FF : Field F}.
  Add Ring clifford_det_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma cl_lift_cons : ∀ (n : nat) (x : Cl F n) (b : bool) (A : Blade n),
    cl_lift x (bcons b A) = if b then 0 else x A.
  Proof. reflexivity. Qed.

  Lemma sigprod_bempty : ∀ (n : nat) (s : Signature F n),
    sigprod s (@bempty n) == 1.
  Proof.
    induction n as [| m IH]; intros s; [ reflexivity |].
    rewrite sigprod_S, bhead_bempty, btail_bempty, (IH (Vector.tl s)).
    ring.
  Qed.

  Lemma ga_vec_zero_family : ∀ (n : nat) (w : Fin.t n → F),
    (∀ j : Fin.t n, w j == 0) → ∀ D : Blade n, ga_vec w D == 0.
  Proof.
    induction n as [| m IH]; intros w Hw D.
    - rewrite (ga_vec_nil w D). reflexivity.
    - rewrite (blade_S m D).
      destruct (bhead D).
      + rewrite (ga_vec_cons_true m w (btail D)).
        destruct (bis_nil (btail D)); [ exact (Hw Fin.F1) | reflexivity ].
      + rewrite (ga_vec_cons_false m w (btail D)).
        exact (IH (fun i ⇒ w (Fin.FS i)) (fun j ⇒ Hw (Fin.FS j)) (btail D)).
  Qed.

  Lemma cl_lift_one : ∀ (n : nat) (D : Blade (S n)),
    cl_lift (cl_one : Cl F n) D == cl_one D.
  Proof.
    intros n D.
    rewrite (blade_S n D), cl_lift_cons.
    destruct (bhead D); reflexivity.
  Qed.

  Lemma ga_vec_lift : ∀ (n : nat) (v : Fin.t (S n) → F),
    v Fin.F1 == 0 →
    ∀ D : Blade (S n),
      ga_vec v D == cl_lift (ga_vec (fun j ⇒ v (Fin.FS j))) D.
  Proof.
    intros n v Hv D.
    rewrite (blade_S n D), cl_lift_cons.
    destruct (bhead D) eqn:Hh.
    - rewrite (ga_vec_cons_true n v (btail D)).
      destruct (bis_nil (btail D)); [ exact Hv | reflexivity ].
    - rewrite (ga_vec_cons_false n v (btail D)). reflexivity.
  Qed.

  Lemma cl_mul_lift_true : ∀ (n : nat) (s : Signature F (S n))
      (x y : Cl F n) (D : Blade n),
    cl_mul s (cl_lift x) (cl_lift y) (bcons true D) == 0.
  Proof.
    intros n s x y D.
    unfold cl_mul.
    transitivity (bsum (fun _ : Blade (S n) ⇒ (0 : F)));
      [| exact (bsum_zero (S n)) ].
    apply bsum_ext. intros A.
    rewrite (blade_S n A), bxor_cons, !cl_lift_cons.
    destruct (bhead A); simpl; ring.
  Qed.

  Lemma cl_mul_lift_false : ∀ (n : nat) (s : Signature F (S n))
      (x y : Cl F n) (D : Blade n),
    cl_mul s (cl_lift x) (cl_lift y) (bcons false D)
      == cl_mul (Vector.tl s) x y D.
  Proof.
    intros n s x y D.
    unfold cl_mul at 1.
    rewrite bsum_S.
    rewrite (bsum_ext n (fun A ⇒ bsign (bcons true A) (bxor (bcons true A) (bcons false D))
                                  × sigprod s (bdiff (bcons true A) (bcons false D))
                                  × (cl_lift x (bcons true A)
                                     × cl_lift y (bxor (bcons true A) (bcons false D))))
               (fun _ ⇒ 0))
      by (intros A; rewrite cl_lift_cons; ring).
    rewrite (bsum_ext n (fun A ⇒ bsign (bcons false A) (bxor (bcons false A) (bcons false D))
                                  × sigprod s (bdiff (bcons false A) (bcons false D))
                                  × (cl_lift x (bcons false A)
                                     × cl_lift y (bxor (bcons false A) (bcons false D))))
               (fun A ⇒ bsign A (bxor A D) × sigprod (Vector.tl s) (bdiff A D)
                         × (x A × y (bxor A D)))).
    - rewrite bsum_zero. unfold cl_mul. ring.
    - intros A.
      rewrite bxor_cons, bdiff_cons, !cl_lift_cons, sigprod_cons.
      unfold bsign. rewrite bpar_cons. simpl. ring.
  Qed.

  Lemma cl_mul_head_lift : ∀ (n : nat) (s : Signature F (S n))
      (v : Fin.t (S n) → F) (y : Cl F n) (A : Blade n),
    v Fin.F1 == 1 → (∀ j : Fin.t n, v (Fin.FS j) == 0) →
    cl_mul s (ga_vec v) (cl_lift y) (bcons true A) == y A.
  Proof.
    intros n s v y A Hone Hzero.
    assert (Hvec : ∀ B : Blade (S n),
              ga_vec v B == (if bis_nil (bxor (bcons true (@bempty n)) B) then 1 else 0)).
    { intros B.
      rewrite (blade_S n B), bxor_cons.
      destruct (bhead B) eqn:Hh.
      - rewrite (ga_vec_cons_true n v (btail B)).
        rewrite bis_nil_cons, bxor_bempty_l. simpl.
        destruct (bis_nil (btail B)); [ exact Hone | reflexivity ].
      - rewrite (ga_vec_cons_false n v (btail B)).
        rewrite bis_nil_cons. simpl.
        exact (ga_vec_zero_family n (fun i ⇒ v (Fin.FS i)) Hzero (btail B)). }
    unfold cl_mul.
    transitivity (bsum (fun B : Blade (S n) ⇒
      (if bis_nil (bxor (bcons true (@bempty n)) B) then 1 else 0)
      × (bsign B (bxor B (bcons true A)) × sigprod s (bdiff B (bcons true A))
         × cl_lift y (bxor B (bcons true A))))).
    { apply bsum_ext. intros B. rewrite (Hvec B). ring. }
    rewrite (bsum_delta_at (S n) (bcons true (@bempty n))
               (fun B ⇒ bsign B (bxor B (bcons true A)) × sigprod s (bdiff B (bcons true A))
                         × cl_lift y (bxor B (bcons true A)))).
    assert (Hxor : bxor (bcons true (@bempty n)) (bcons true A) = bcons false A)
      by (rewrite bxor_cons, bxor_bempty_l; reflexivity).
    assert (Hdiff : bdiff (bcons true (@bempty n)) (bcons true A)
                    = bcons false (@bempty n))
      by (rewrite bdiff_cons, bdiff_bempty_l; reflexivity).
    rewrite Hxor, Hdiff, cl_lift_cons, sigprod_cons.
    unfold bsign. rewrite bpar_cons, bpar_bempty_l.
    rewrite (sigprod_bempty n (Vector.tl s)).
    unfold parity_sign. simpl. ring.
  Qed.

End CliffordLift.


Definition cl_product {F : Type} {zF : Zero F} {aF : HasAdd F} {oF : One F}
    {nF : HasNeg F} {mF : HasMult F} {n : nat} (s : Signature F n)
    (l : list (Fin.t n → F)) : Cl F n :=
  fold_right (fun v acc ⇒ cl_mul s (ga_vec v) acc) cl_one l.

Section CliffordProductFacts.
  Context {F : Type} `{FF : Field F}.
  Add Ring clifford_product_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma cl_product_nil : ∀ (n : nat) (s : Signature F n),
    cl_product s [] = cl_one.
  Proof. reflexivity. Qed.

  Lemma cl_product_cons : ∀ (n : nat) (s : Signature F n)
      (v : Fin.t n → F) (l : list (Fin.t n → F)),
    cl_product s (v :: l) = cl_mul s (ga_vec v) (cl_product s l).
  Proof. reflexivity. Qed.

  Lemma cl_mul_ext : ∀ (n : nat) (s : Signature F n) (x x' y y' : Cl F n),
    (∀ A : Blade n, x A == x' A) → (∀ A : Blade n, y A == y' A) →
    ∀ D : Blade n, cl_mul s x y D == cl_mul s x' y' D.
  Proof.
    intros n s x x' y y' Hx Hy D.
    unfold cl_mul.
    apply bsum_ext. intros A.
    rewrite (Hx A), (Hy (bxor A D)). reflexivity.
  Qed.

  Lemma cl_mul_lift : ∀ (n : nat) (s : Signature F (S n))
      (x y : Cl F n) (D : Blade (S n)),
    cl_mul s (cl_lift x) (cl_lift y) D == cl_lift (cl_mul (Vector.tl s) x y) D.
  Proof.
    intros n s x y D.
    rewrite (blade_S n D).
    destruct (bhead D).
    - rewrite cl_mul_lift_true, cl_lift_cons. reflexivity.
    - rewrite cl_mul_lift_false, cl_lift_cons. reflexivity.
  Qed.

A product of k vectors has no component on a blade of more than k generators.
  Theorem cl_product_grade_bound : ∀ (n : nat) (s : Signature F n)
      (l : list (Fin.t n → F)) (D : Blade n),
    (length l < bcount D)%nat → cl_product s l D == 0.
  Proof.
    intros n s l; induction l as [| v l IH]; intros D Hlen.
    - assert (Hnil : bis_nil D = false).
      { destruct (bis_nil D) eqn:Hd; [| reflexivity ].
        exfalso.
        rewrite (bcount_of_bis_nil n D Hd) in Hlen.
        simpl in Hlen. lia. }
      rewrite cl_product_nil. unfold cl_one. rewrite Hnil. reflexivity.
    - rewrite cl_product_cons. unfold cl_mul.
      transitivity (bsum (fun _ : Blade n ⇒ (0 : F)));
        [| exact (bsum_zero n) ].
      apply bsum_ext. intros A.
      destruct (Nat.eq_dec (bcount A) 1%nat) as [Hc1 | Hc1].
      + assert (Hrest : cl_product s l (bxor A D) == 0).
        { apply IH.
          pose proof (bcount_bxor_le n A (bxor A D)) as Hle.
          rewrite (bxor_involutive n A D) in Hle.
          simpl in Hlen. lia. }
        rewrite Hrest. ring.
      + rewrite (ga_vec_not_grade_one n v A Hc1). ring.
  Qed.

  Lemma cl_product_e_FS : ∀ (n : nat) (s : Signature F (S n))
      (l : list (Fin.t n)) (D : Blade (S n)),
    cl_product s (map (fun j ⇒ e (Fin.FS j)) l) D
      == cl_lift (cl_product (Vector.tl s) (map (fun j ⇒ e j) l)) D.
  Proof.
    intros n s l; induction l as [| j l IH]; intros D.
    - symmetry. exact (cl_lift_one n D).
    - simpl (map _ (j :: l)).
      rewrite !cl_product_cons.
      transitivity (cl_mul s (cl_lift (ga_vec (e j)))
                      (cl_lift (cl_product (Vector.tl s) (map (fun k ⇒ e k) l))) D).
      + apply cl_mul_ext; [| exact IH ].
        intros A.
        transitivity
          (cl_lift (ga_vec (fun i : Fin.t n ⇒ e (Fin.FS j) (Fin.FS i) : F)) A).
        × apply (ga_vec_lift n (e (Fin.FS j))).
          exact (e_at_neq (S n) (Fin.FS j) Fin.F1 (fin_FS_neq_F1 n j)).
        × rewrite (blade_S n A), !cl_lift_cons.
          destruct (bhead A); [ reflexivity |].
          exact (ga_vec_ext n (fun i ⇒ e (Fin.FS j) (Fin.FS i)) (e j)
                   (fun i ⇒ e_FS n j i) (btail A)).
      + exact (cl_mul_lift n s (ga_vec (e j))
                 (cl_product (Vector.tl s) (map (fun k ⇒ e k) l)) D).
  Qed.

The product of the standard basis, read at the full blade, is 1: the generators come in increasing order, so no sign, and they are distinct, so no signature factor.
  Theorem cl_product_e_top : ∀ (n : nat) (s : Signature F n),
    cl_product s (map (fun i ⇒ e i) (fin_all n)) (bfull n) == 1.
  Proof.
    induction n as [| m IH]; intros s; [ reflexivity |].
    simpl (fin_all (S m)).
    rewrite map_cons, map_map, cl_product_cons, bfull_cons.
    transitivity (cl_mul s (ga_vec (e Fin.F1))
                    (cl_lift (cl_product (Vector.tl s) (map (fun j ⇒ e j) (fin_all m))))
                    (bcons true (bfull m))).
    - apply cl_mul_ext; [ reflexivity |].
      intros A. exact (cl_product_e_FS m s (fin_all m) A).
    - rewrite (cl_mul_head_lift m s (e Fin.F1)
                 (cl_product (Vector.tl s) (map (fun j ⇒ e j) (fin_all m)))
                 (bfull m)
                 (e_at_eq (S m) Fin.F1 Fin.F1 eq_refl)
                 (fun j ⇒ e_at_neq (S m) Fin.F1 (Fin.FS j)
                             (fin_F1_neq_FS m j))).
      exact (IH (Vector.tl s)).
  Qed.

End CliffordProductFacts.


Definition cl_top {F : Type} {n : nat} (x : Cl F n) : F := x (bfull n).

Section CliffordDeterminantInstance.
  Context {F : Type} `{FF : Field F} {n : nat} (s : Signature F n).

  #[local] Instance Det_One : One (Cl F n) := cl_one.
  #[local] Instance Det_Mult : HasMult (Cl F n) := cl_mul s.
  #[local] Instance Det_Ring : Ring (Cl F n) := Cl_Ring s.
  #[local] Instance Det_QF :
    QuadraticForm F (Fin.t n → F) (qform s) := Cl_QuadraticForm s.
  #[local] Instance Det_GA :
    GeometricAlgebra F (Fin.t n → F) (Cl F n) (Q := qform s) := Cl_GA s.

  Lemma cl_product_one : cl_product s [] == 1.
  Proof. reflexivity. Qed.

  Lemma cl_product_mul : ∀ (v : Fin.t n → F) (l : list (Fin.t n → F)),
    cl_product s (v :: l) == ga_vec v × cl_product s l.
  Proof. intros v l. reflexivity. Qed.

  #[local] Instance cl_top_linear : Linear F (Cl F n) F cl_top.
  Proof.
    constructor.
    - intros x y Hxy. exact (Hxy (bfull n)).
    - intros x y. reflexivity.
    - intros a x. reflexivity.
  Qed.

  Lemma cl_top_short : ∀ l : list (Fin.t n → F),
    (length l < n)%nat → cl_top (cl_product s l) == 0.
  Proof.
    intros l Hl.
    apply (cl_product_grade_bound n s l (bfull n)).
    rewrite bcount_bfull. exact Hl.
  Qed.

  Lemma cl_top_basis : cl_top (cl_product s (map (Basis.vec (FinVec_Basis n)) enum)) == 1.
  Proof. exact (cl_product_e_top n s). Qed.

The computing determinant is a Determinant because Theory/Exterior.v's general theorem says so.
  Definition det_ga_Determinant :
      Determinant n (Fin.t n) F (Fin.t n → F) (FinVec_Basis n) :=
    ga_top_determinant (Qform := qform s) (GA := Det_GA)
      (cl_product s) cl_product_one cl_product_mul
      (fun (i j : Fin.t n) (H : i = j) ⇒ H) cl_top cl_top_short
      (FinVec_Basis n) cl_top_basis.

The determinant of a matrix: multiply its columns, embedded as vectors, and read the product at the full blade.
  Definition det_ga (A : Matrix n n F) : F :=
    cl_top (cl_product s (map (fun j i ⇒ A i j) (fin_all n))).

  Example det_ga_is_the_determinant_form : ∀ A : Matrix n n F,
    det_ga A = det_ga_Determinant (fun j i ⇒ A i j).
  Proof. reflexivity. Qed.

The two implementations agree, and not because anyone compared them: `Theory/Determinant.v`'s `determinant_unique` says a basis admits one alternating normalized form, so the geometric product and the cofactor expansion — which share no code — compute the same function. The transpose is bookkeeping: `det_ga` reads the columns, `detM` reads the family as rows.
  Theorem det_ga_is_detM : ∀ A : Matrix n n F,
    det_ga A == detM (fun j i ⇒ A i j).
  Proof.
    intros A.
    exact (detM_is_the_determinant n det_ga_Determinant (fun j i ⇒ A i j)).
  Qed.

End CliffordDeterminantInstance.


Definition sig_euclid {F : Type} {oF : One F} (n : nat) : Signature F n :=
  Vector.const 1 n.

Local Open Scope Q_scope.

Definition m3 : Matrix 3 3 Q :=
  fun i j ⇒
    match i, j with
    | Fin.F1, Fin.F1 ⇒ 1#1 | Fin.F1, Fin.FS Fin.F1 ⇒ 2#1 | Fin.F1, _ ⇒ 3#1
    | Fin.FS Fin.F1, Fin.F1 ⇒ 4#1 | Fin.FS Fin.F1, Fin.FS Fin.F1 ⇒ 5#1
    | Fin.FS Fin.F1, _ ⇒ 6#1
    | _, Fin.F1 ⇒ 7#1 | _, Fin.FS Fin.F1 ⇒ 8#1 | _, _ ⇒ 10#1
    end.

Definition m3sing : Matrix 3 3 Q :=
  fun i j ⇒ match j with Fin.FS (Fin.FS _) ⇒ m3 i Fin.F1 | _ ⇒ m3 i j end.
