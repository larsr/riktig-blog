From Riktig.Structure Require Export Equiv.
From Riktig.Theory Require Export Equiv.

Equivalences compose.
Section EquivCompose.
  Context {A B C : Type} {eqA : EqRel A} {eqB : EqRel B} {eqC : EqRel C}
          `{!Equivalence eqA} `{!Equivalence eqB} `{!Equivalence eqC}
          {f : A → B} {g : B → A} {h : B → C} {k : C → B}
          `{!Equiv A B f g} `{!Equiv B C h k}.

  #[export] Instance Equiv_compose : Equiv A C (fun a ⇒ h (f a)) (fun c ⇒ g (k c)).
  Proof.
    split.
    - intros a a' Haa'. rewrite Haa'. reflexivity.
    - intros c c' Hcc'. rewrite Hcc'. reflexivity.
    - intros a. rewrite (Equiv.from_to (to := h) (f a)). exact (Equiv.from_to (to := f) a).
    - intros c. rewrite (Equiv.to_from (to := f) (k c)). exact (Equiv.to_from (to := h) c).
  Qed.
End EquivCompose.
