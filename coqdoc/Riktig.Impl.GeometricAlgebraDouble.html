Adding one generator e with e × e == s to a geometric algebra with a grade involution. The new algebra is the pairs (a, b) standing for a + b e: since e anticommutes with vectors, e × x == involute x × e, so (a + b e) (c + d e) == (a c + s b involute d) + (a d + b involute c) e. The new vector space is V × F, its last coordinate along e, and the inner product weighs that coordinate by s. The carriers are named by the signature, since the same pairs carry every one.
Definition Extend {F : Type} (s : F) (V : Type) : Type := (V × F)%type.
Definition Double {F : Type} (s : F) (A : Type) : Type := (A × A)%type.

  #[export] Instance Extend_InnerProductSpace : InnerProductSpace F (Extend s V).
  Proof.
    split.
    - intros u u' [Hu1 Hu2] w w' [Hw1 Hw2]. unfold inner, Extend_HasInner.
      apply op_proper; [ apply inner_proper; assumption | ].
      apply op_proper; [ reflexivity | apply op_proper; assumption ].
    - intros w u v. unfold inner, Extend_HasInner. simpl.
      calc (⟪ fst u + fst v , fst w ⟫ + s × ((snd u + snd v) × snd w))
        == ((⟪ fst u , fst w ⟫ + ⟪ fst v , fst w ⟫) + s × ((snd u + snd v) × snd w))
             { (additive (f := fun x ⇒ ⟪ x , fst w ⟫) (fst u) (fst v)) }
        == ((⟪ fst u , fst w ⟫ + ⟪ fst v , fst w ⟫) + s × (snd u × snd w + snd v × snd w))
             { (right_distributive (op := mult) (snd u) (snd v) (snd w)) }
        == ((⟪ fst u , fst w ⟫ + ⟪ fst v , fst w ⟫) + (s × (snd u × snd w) + s × (snd v × snd w)))
             { (left_distributive (op := mult) s (snd u × snd w) (snd v × snd w)) }
        == ((⟪ fst u , fst w ⟫ + s × (snd u × snd w)) + (⟪ fst v , fst w ⟫ + s × (snd v × snd w)))
             { add_add_add_comm }.
    - intros w a u. unfold inner, Extend_HasInner. simpl.
      calc (⟪ a *: fst u , fst w ⟫ + s × ((a × snd u) × snd w))
        == (a × ⟪ fst u , fst w ⟫ + s × ((a × snd u) × snd w))
             { (homogeneous (f := fun x ⇒ ⟪ x , fst w ⟫) a (fst u)) }
        == (a × ⟪ fst u , fst w ⟫ + s × (a × (snd u × snd w)))
             { (associative (op := mult) (action := mult) a (snd u) (snd w)) }
        == (a × ⟪ fst u , fst w ⟫ + a × (s × (snd u × snd w)))
             { (mul_left_comm s a (snd u × snd w)) }
        == (a × (⟪ fst u , fst w ⟫ + s × (snd u × snd w)))
             { (left_distributive (op := mult) a ⟪ fst u , fst w ⟫ (s × (snd u × snd w))) }.
    - intros u v. unfold inner, Extend_HasInner.
      calc (⟪ fst u , fst v ⟫ + s × (snd u × snd v))
        == (⟪ fst v , fst u ⟫ + s × (snd u × snd v)) { (commutative (op := inner) (fst u) (fst v)) }
        == (⟪ fst v , fst u ⟫ + s × (snd v × snd u)) { (commutative (op := mult) (snd u) (snd v)) }.
  Qed.

  #[export] Instance Double_mul_Magma : Magma (Double s A) Double_HasMult.
  Proof.
    split. exact Double_Equivalence.
    intros [a b] [a' b'] [Ha Hb] [c d] [c' d'] [Hc Hd]. split; simpl.
    - setoid_rewrite Ha. setoid_rewrite Hb. setoid_rewrite Hc. setoid_rewrite Hd. reflexivity.
    - setoid_rewrite Ha. setoid_rewrite Hb. setoid_rewrite Hc. setoid_rewrite Hd. reflexivity.
  Qed.

  #[export] Instance Double_mul_Semigroup : Semigroup (Double s A) Double_HasMult.

    split. exact Double_mul_Magma.
    intros [a b] [c d] [e f]. split; simpl.
    - exact (double_assoc (V := V) s a b c d e f).
    - exact (double_assoc_snd (V := V) s a b c d e f).
  Qed.

  #[export] Instance Double_mul_Monoid : Monoid (Double s A) Double_One Double_HasMult.
  Proof.
    split. exact Double_mul_Semigroup.
    - intros [c d]. split; simpl.
      + calc ((1 : A) × c + s *: (0 × involute d))
          == (c + s *: (0 × involute d)) { (left_identity (action := mult) c) }
          == (c + s *: (0 : A)) { (zero_mul (involute d)) }
          == (c + 0) { (smul_zero (V := A) s) }
          == c { (right_identity (action := add) c) }.
      + calc ((1 : A) × d + 0 × involute c)
          == (d + 0 × involute c) { (left_identity (action := mult) d) }
          == (d + 0) { (zero_mul (involute c)) }
          == d { (right_identity (action := add) d) }.
    - intros [a b]. split; simpl.
      + calc (a × 1 + s *: (b × involute 0))
          == (a + s *: (b × involute (0 : A))) { (right_identity (action := mult) a) }
          == (a + s *: (b × 0)) { involute_zero }
          == (a + s *: (0 : A)) { (mul_zero b) }
          == (a + 0) { (smul_zero (V := A) s) }
          == a { (right_identity (action := add) a) }.
      + calc (a × 0 + b × involute 1)
          == (0 + b × involute (1 : A)) { (mul_zero a) }
          == (0 + b × 1) { involute_one }
          == (0 + b) { (right_identity (action := mult) b) }
          == b { (left_identity (action := add) b) }.
  Qed.

  #[export] Instance Double_Ring : Ring (Double s A) 0 add neg 1 mult.
  Proof.
    split. exact Double_AbGroup. exact Double_mul_Monoid.
    - intros [a b] [c d] [e f]. split; simpl.
      + calc (a × (c + e) + s *: (b × involute (d + f)))
          == ((a × c + a × e) + s *: (b × involute (d + f))) { (left_distributive (op := mult) a c e) }
          == ((a × c + a × e) + s *: (b × (involute d + involute f))) { (additive (f := involute) d f) }
          == ((a × c + a × e) + s *: (b × involute d + b × involute f)) { (left_distributive (op := mult) b (involute d) (involute f)) }
          == ((a × c + a × e) + (s *: (b × involute d) + s *: (b × involute f))) { (left_distributive (op := scale) s (b × involute d) (b × involute f)) }
          == ((a × c + s *: (b × involute d)) + (a × e + s *: (b × involute f))) { (add_add_add_comm (a × c) (a × e) (s *: (b × involute d)) (s *: (b × involute f))) }.
      + calc (a × (d + f) + b × involute (c + e))
          == ((a × d + a × f) + b × involute (c + e)) { (left_distributive (op := mult) a d f) }
          == ((a × d + a × f) + b × (involute c + involute e)) { (additive (f := involute) c e) }
          == ((a × d + a × f) + (b × involute c + b × involute e)) { (left_distributive (op := mult) b (involute c) (involute e)) }
          == ((a × d + b × involute c) + (a × f + b × involute e)) { (add_add_add_comm (a × d) (a × f) (b × involute c) (b × involute e)) }.
    - intros [a b] [c d] [e f]. split; simpl.
      + calc ((a + c) × e + s *: ((b + d) × involute f))
          == ((a × e + c × e) + s *: ((b + d) × involute f)) { (right_distributive (op := mult) a c e) }
          == ((a × e + c × e) + s *: (b × involute f + d × involute f)) { (right_distributive (op := mult) b d (involute f)) }
          == ((a × e + c × e) + (s *: (b × involute f) + s *: (d × involute f))) { (left_distributive (op := scale) s (b × involute f) (d × involute f)) }
          == ((a × e + s *: (b × involute f)) + (c × e + s *: (d × involute f))) { (add_add_add_comm (a × e) (c × e) (s *: (b × involute f)) (s *: (d × involute f))) }.
      + calc ((a + c) × f + (b + d) × involute e)
          == ((a × f + c × f) + (b + d) × involute e) { (right_distributive (op := mult) a c f) }
          == ((a × f + c × f) + (b × involute e + d × involute e)) { (right_distributive (op := mult) b d (involute e)) }
          == ((a × f + b × involute e) + (c × f + d × involute e)) { (add_add_add_comm (a × f) (c × f) (b × involute e) (d × involute e)) }.
  Qed.

  #[export] Instance Double_Algebra : Algebra F (Double s A) 0 add neg 1 mult scale.
  Proof.
    split. exact Double_VectorSpace. exact Double_Ring.
    - intros r [a b] [c d]. split; simpl.
      + calc ((r *: a) × c + s *: ((r *: b) × involute d))
          == (r *: (a × c) + s *: ((r *: b) × involute d)) { (smul_mul_assoc r a c) }
          == (r *: (a × c) + s *: (r *: (b × involute d))) { (smul_mul_assoc r b (involute d)) }
          == (r *: (a × c) + r *: (s *: (b × involute d))) { (smul_comm s r (b × involute d)) }
          == (r *: (a × c + s *: (b × involute d))) { (left_distributive (op := scale) r (a × c) (s *: (b × involute d))) }.
      + calc ((r *: a) × d + (r *: b) × involute c)
          == (r *: (a × d) + (r *: b) × involute c) { (smul_mul_assoc r a d) }
          == (r *: (a × d) + r *: (b × involute c)) { (smul_mul_assoc r b (involute c)) }
          == (r *: (a × d + b × involute c)) { (left_distributive (op := scale) r (a × d) (b × involute c)) }.
    - intros r [a b] [c d]. split; simpl.
      + calc (a × (r *: c) + s *: (b × involute (r *: d)))
          == (r *: (a × c) + s *: (b × involute (r *: d))) { (mul_smul_comm r a c) }
          == (r *: (a × c) + s *: (b × (r *: involute d))) { (homogeneous (f := involute) r d) }
          == (r *: (a × c) + s *: (r *: (b × involute d))) { (mul_smul_comm r b (involute d)) }
          == (r *: (a × c) + r *: (s *: (b × involute d))) { (smul_comm s r (b × involute d)) }
          == (r *: (a × c + s *: (b × involute d))) { (left_distributive (op := scale) r (a × c) (s *: (b × involute d))) }.
      + calc (a × (r *: d) + b × involute (r *: c))
          == (r *: (a × d) + b × involute (r *: c)) { (mul_smul_comm r a d) }
          == (r *: (a × d) + b × (r *: involute c)) { (homogeneous (f := involute) r c) }
          == (r *: (a × d) + r *: (b × involute c)) { (mul_smul_comm r b (involute c)) }
          == (r *: (a × d + b × involute c)) { (left_distributive (op := scale) r (a × d) (b × involute c)) }.
  Qed.

  #[export] Instance Double_Vec_Linear : Linear F (Extend s V) (Double s A) Double_Vec.
  Proof.
    split.
    - intros [u x] [v y] [Hu Hx]. split; simpl.
      + apply (proper (f := vec)). exact Hu.
      + setoid_rewrite Hx. reflexivity.
    - intros [u x] [v y]. split; simpl.
      + exact (additive (f := vec) u v).
      + exact (right_distributive (op := scale) x y (1 : A)).
    - intros r [u x]. split; simpl.
      + exact (homogeneous (f := vec) r u).
      + exact (associative (op := mult) (action := scale) r x (1 : A)).
  Qed.

  #[export] Instance Double_GeometricAlgebra : GeometricAlgebra F (Extend s V) (Double s A) Double_Vec.
  Proof.
    split. exact Double_Vec_Linear.
    intros [u x]. split; simpl.
    - calc (vec u × vec u + s *: ((x *: 1) × involute (x *: 1)))
        == (⟪ u , u ⟫ *: (1 : A) + s *: ((x *: 1) × involute (x *: 1))) { vec_sq }
        == (⟪ u , u ⟫ *: (1 : A) + s *: ((x *: 1) × (x *: 1))) { involute_smul_one }
        == (⟪ u , u ⟫ *: (1 : A) + s *: ((x × x) *: 1)) { smul_one_mul_smul_one }
        == (⟪ u , u ⟫ *: (1 : A) + (s × (x × x)) *: 1) { (associative (op := mult) (action := scale) s (x × x) (1 : A)) }
        == ((⟪ u , u ⟫ + s × (x × x)) *: (1 : A)) { (right_distributive (op := scale) ⟪ u , u ⟫ (s × (x × x)) (1 : A)) }.
    - calc (vec u × (x *: 1) + (x *: 1) × involute (vec u))
        == (x *: vec u + (x *: 1) × involute (vec u)) { mul_smul_one }
        == (x *: vec u + (x *: 1) × - vec u) { involute_vec }
        == (x *: vec u + - ((x *: 1) × vec u)) { (mul_neg (x *: (1 : A)) (vec u)) }
        == (x *: vec u + - (x *: vec u)) { smul_one_mul }
        == (0 : A) { (right_inverse (op := add) (x *: vec u)) }
        == ((⟪ u , u ⟫ + s × (x × x)) *: (0 : A)) { (smul_zero (V := A) (⟪ u , u ⟫ + s × (x × x))) }.
  Qed.

  #[export] Instance Double_GradeInvolution :
    GradeInvolution F (Extend s V) (Double s A) Double_Involute.
  Proof.
    split.
    - intros [a b] [a' b'] [Ha Hb]. split; simpl.
      + setoid_rewrite Ha. reflexivity.
      + setoid_rewrite Hb. reflexivity.
    - intros [a b] [c d]. split; simpl.
      + exact (additive (f := involute) a c).
      + calc (- involute (b + d))
          == (- (involute b + involute d)) { (additive (f := involute) b d) }
          == (- involute b + - involute d) { (neg_add (involute b) (involute d)) }.
    - intros [a b] [c d]. split; simpl.
      + transitivity (involute a × involute c + s *: (involute b × d)).
        × calc (involute (a × c + s *: (b × involute d)))
            == (involute (a × c) + involute (s *: (b × involute d))) { (additive (f := involute) (a × c) (s *: (b × involute d))) }
            == (involute a × involute c + involute (s *: (b × involute d))) { (additive (f := involute) (opA := mult) a c) }
            == (involute a × involute c + s *: involute (b × involute d)) { (homogeneous (f := involute) s (b × involute d)) }
            == (involute a × involute c + s *: (involute b × involute (involute d))) { (additive (f := involute) (opA := mult) b (involute d)) }
            == (involute a × involute c + s *: (involute b × d)) { involute_involute }.
        × calc (involute a × involute c + s *: (involute b × d))
            == (involute a × involute c + s *: (- involute b × - d)) { (neg_mul_neg (involute b) d) }
            == (involute a × involute c + s *: (- involute b × - involute (involute d))) { involute_involute }
            == (involute a × involute c + s *: (- involute b × involute (- involute d))) { (involute_neg (involute d)) }.
      + transitivity (- (involute a × involute d + involute b × c)).
        × calc (- involute (a × d + b × involute c))
            == (- (involute (a × d) + involute (b × involute c))) { (additive (f := involute) (a × d) (b × involute c)) }
            == (- (involute a × involute d + involute (b × involute c))) { (additive (f := involute) (opA := mult) a d) }
            == (- (involute a × involute d + involute b × involute (involute c))) { (additive (f := involute) (opA := mult) b (involute c)) }
            == (- (involute a × involute d + involute b × c)) { involute_involute }.
        × calc (- (involute a × involute d + involute b × c))
            == (- (involute a × involute d) + - (involute b × c)) { (neg_add (involute a × involute d) (involute b × c)) }
            == (involute a × - involute d + - (involute b × c)) { (mul_neg (involute a) (involute d)) }
            == (involute a × - involute d + - involute b × c) { (neg_mul (involute b) c) }
            == (involute a × - involute d + - involute b × involute (involute c)) { involute_involute }.
    - intros r [a b]. split; simpl.
      + exact (homogeneous (f := involute) r a).
      + calc (- involute (r *: b))
          == (- (r *: involute b)) { (homogeneous (f := involute) r b) }
          == (r *: - involute b) { (smul_neg r (involute b)) }.
    - split; simpl.
      + exact involute_one.
      + calc (- involute (0 : A)) == (- (0 : A)) { involute_zero } == (0 : A) { neg_zero }.
    - intros [a b]. split; simpl.
      + exact (involute_involute a).
      + calc (- involute (- involute b))
          == (- - involute (involute b)) { involute_neg }
          == (involute (involute b)) { (neg_neg (involute (involute b))) }
          == b { involute_involute }.
    - intros [u x]. split; simpl.
      + exact (involute_vec u).
      + calc (- involute (x *: (1 : A)))
          == (- (x *: (1 : A))) { involute_smul_one }.
  Qed.
