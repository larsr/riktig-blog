Library Riktig.Impl.EquOption

From Riktig Require Export Props.
From Riktig.Lib Require Export Calc.

Options compared by constructor, the values inside by their own equality.
#[export] Instance EqOption (A : Type) {eA : EqRel A} : EqRel (option A) :=
  fun o o' ⇒ match o, o' with
              | Some a, Some a' ⇒ a == a'
              | None, None ⇒ True
              | _, _ ⇒ False
              end.

#[export] Instance EqOption_Equivalence (A : Type) {eA : EqRel A}
  {EA : Equivalence (@equ A eA)} : Equivalence (@equ (option A) (EqOption A)).
Proof.
  destruct EA as [Rf Sy Tr]. split.
  - intros [a |]; [ exact (Rf a) | exact I ].
  - intros [a |] [b |] H; try contradiction; [ exact (Sy _ _ H) | exact I ].
  - intros [a |] [b |] [c |] H1 H2; try contradiction;
      [ exact (Tr _ _ _ H1 H2) | exact I ].
Qed.
