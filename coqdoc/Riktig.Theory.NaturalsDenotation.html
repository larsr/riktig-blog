From Riktig Require Export Structure.Naturals.

A setoid that denotes into a naturals model is a naturals model: if every element denotes some number, denotation is functional and faithful, every number is denoted, and zero, one, successor, addition and multiplication are tracked, then every law of the model transports along the denotation. An implementation -- the Church numerals, say -- only has to provide the denotation facts; the structure is read off here.
Section NaturalsDenotation.
  Context {C : Type} {eqC : EqRel C} {zC : Zero C} {oC : One C}
    {sC : HasSucc C} {aC : HasAdd C} {mC : HasMult C}.
  Context {N0 : Type} `{Naturals N0}.
  Variable denotes : C → N0 → Prop.
  Hypothesis denotes_total : ∀ c : C, ∃ n : N0, denotes c n.
  Hypothesis denotes_sur : ∀ n : N0, ∃ c : C, denotes c n.
  Hypothesis denotes_unique : ∀ (c : C) (n m : N0), denotes c n → denotes c m → n == m.
  Hypothesis denotes_faithful : ∀ (c d : C) (n m : N0), denotes c n → denotes d m → n == m → c == d.
  Hypothesis denotes_proper_l : ∀ (c d : C) (n : N0), c == d → denotes c n → denotes d n.
  Hypothesis denotes_proper_r : ∀ (c : C) (n m : N0), n == m → denotes c n → denotes c m.
  Hypothesis denotes_zero : denotes 0 0.
  Hypothesis denotes_one : denotes 1 1.
  Hypothesis denotes_succ : ∀ (c : C) (n : N0), denotes c n → denotes (succ c) (succ n).
  Hypothesis denotes_add : ∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (c + d) (a + b).
  Hypothesis denotes_mul : ∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (c × d) (a × b).

  Lemma denotation_equivalence : Equivalence (@equ C eqC).
  Proof.
    split.
    - intros c. destruct (denotes_total c) as [n Hn].
      exact (denotes_faithful c c n n Hn Hn (reflexivity n)).
    - intros c d Hcd. destruct (denotes_total c) as [n Hn].
      exact (denotes_faithful d c n n (denotes_proper_l c d n Hcd Hn) Hn (reflexivity n)).
    - intros c d e Hcd Hde. destruct (denotes_total c) as [n Hn].
      exact (denotes_faithful c e n n Hn (denotes_proper_l d e n Hde (denotes_proper_l c d n Hcd Hn)) (reflexivity n)).
  Qed.

Denotations of equal elements are equal numbers.
  Lemma denotes_inject : ∀ (c d : C) (n m : N0), c == d → denotes c n → denotes d m → n == m.
  Proof.
    intros c d n m Hcd Hn Hm. exact (denotes_unique d n m (denotes_proper_l c d n Hcd Hn) Hm).
  Qed.

A binary operation tracked by the denotation respects == and satisfies any law its image satisfies; the three helpers below say this once for the shapes the semiring needs.
  Lemma denotation_op_proper : ∀ (f : C → C → C) (g : N0 → N0 → N0),
    (∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (f c d) (g a b)) →
    Proper (equ ==> equ ==> equ) f.
  Proof.
    intros f g Hf c c' Hc d d' Hd.
    destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb].
    apply (denotes_faithful _ _ (g a b) (g a b)); [ exact (Hf c d a b Ha Hb) | | exact (reflexivity _) ].
    exact (Hf c' d' a b (denotes_proper_l c c' a Hc Ha) (denotes_proper_l d d' b Hd Hb)).
  Qed.

  Instance denotation_magma_add : Magma C add.
  Proof. split; [ exact denotation_equivalence | exact (denotation_op_proper add add denotes_add) ]. Qed.

  Instance denotation_magma_mul : Magma C mult.
  Proof. split; [ exact denotation_equivalence | exact (denotation_op_proper mult mult denotes_mul) ]. Qed.

  Theorem denotation_naturals : Naturals C 0 succ 1 add mult.
  Proof.
    assert (Hsucc_proper : Proper (equ ==> equ) (succ : C → C)).
    { intros c d Hcd. destruct (denotes_total c) as [n Hn].
      apply (denotes_faithful _ _ (succ n) (succ n)); [ exact (denotes_succ c n Hn) | | exact (reflexivity _) ].
      exact (denotes_succ d n (denotes_proper_l c d n Hcd Hn)). }
    split.
    - split.
      + exact denotation_equivalence.
      + exact Hsucc_proper.
      + intros c d Hs.
        destruct (denotes_total c) as [n Hn]. destruct (denotes_total d) as [m Hm].
        apply (denotes_faithful c d n m Hn Hm).
        apply (injective (f := (succ : N0 → N0))).
        exact (denotes_inject _ _ _ _ Hs (denotes_succ c n Hn) (denotes_succ d m Hm)).
      + intros c Hc. destruct (denotes_total c) as [n Hn].
        apply (Peano.succ_not_zero (N := N0) n).
        exact (denotes_inject _ _ _ _ Hc (denotes_succ c n Hn) denotes_zero).
      + intros P HP H0 HS c.
        destruct (denotes_total c) as [n Hn]. revert c Hn.
        apply (Peano.nat_induction (N := N0) (fun n ⇒ ∀ c : C, denotes c n → P c)).
        × intros k k' Hkk'. split; intros HQ c' Hc'.
          { exact (HQ c' (denotes_proper_r c' k' k (symmetry Hkk') Hc')). }
          { exact (HQ c' (denotes_proper_r c' k k' Hkk' Hc')). }
        × intros c' Hc'.
          apply (proj1 (HP 0 c' (denotes_faithful 0 c' 0 0 denotes_zero Hc' (reflexivity 0)))).
          exact H0.
        × intros m IH c' Hc'.
          destruct (denotes_sur m) as [d Hd].
          apply (proj1 (HP (succ d) c' (denotes_faithful (succ d) c' (succ m) (succ m) (denotes_succ d m Hd) Hc' (reflexivity _)))).
          exact (HS d (IH d Hd)).
    - split; [ split | ].
      + split; [ split | | ].
        × split; [ exact denotation_equivalence | exact (denotation_op_proper add add denotes_add) ].
        × intros c d e.
          destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb]. destruct (denotes_total e) as [k Hk].
          apply (denotes_faithful _ _ ((a + b) + k) (a + (b + k))).
          { exact (denotes_add _ _ _ _ (denotes_add c d a b Ha Hb) Hk). }
          { exact (denotes_add _ _ _ _ Ha (denotes_add d e b k Hb Hk)). }
          { exact (associative (op := (add : BinOp N0)) (action := (add : BinOp N0)) a b k). }
        × intros c. destruct (denotes_total c) as [a Ha].
          apply (denotes_faithful _ _ (0 + a) a); [ exact (denotes_add _ _ _ _ denotes_zero Ha) | exact Ha | ].
          exact (left_identity (action := add) a).
        × intros c. destruct (denotes_total c) as [a Ha].
          apply (denotes_faithful _ _ (a + 0) a); [ exact (denotes_add _ _ _ _ Ha denotes_zero) | exact Ha | ].
          exact (right_identity (action := add) a).
      + intros c d.
        destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb].
        apply (denotes_faithful _ _ (a + b) (b + a)).
        { exact (denotes_add c d a b Ha Hb). }
        { exact (denotes_add d c b a Hb Ha). }
        { exact (commutative (op := add) a b). }
      + split; [ split | | ].
        × split; [ exact denotation_equivalence | exact (denotation_op_proper mult mult denotes_mul) ].
        × intros c d e.
          destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb]. destruct (denotes_total e) as [k Hk].
          apply (denotes_faithful _ _ ((a × b) × k) (a × (b × k))).
          { exact (denotes_mul _ _ _ _ (denotes_mul c d a b Ha Hb) Hk). }
          { exact (denotes_mul _ _ _ _ Ha (denotes_mul d e b k Hb Hk)). }
          { exact (associative (op := (mult : BinOp N0)) (action := (mult : BinOp N0)) a b k). }
        × intros c. destruct (denotes_total c) as [a Ha].
          apply (denotes_faithful _ _ (1 × a) a); [ exact (denotes_mul _ _ _ _ denotes_one Ha) | exact Ha | ].
          exact (left_identity (action := mult) a).
        × intros c. destruct (denotes_total c) as [a Ha].
          apply (denotes_faithful _ _ (a × 1) a); [ exact (denotes_mul _ _ _ _ Ha denotes_one) | exact Ha | ].
          exact (right_identity (action := mult) a).
      + intros c d e.
        destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb]. destruct (denotes_total e) as [k Hk].
        apply (denotes_faithful _ _ (a × (b + k)) (a × b + a × k)).
        { exact (denotes_mul _ _ _ _ Ha (denotes_add d e b k Hb Hk)). }
        { exact (denotes_add _ _ _ _ (denotes_mul c d a b Ha Hb) (denotes_mul c e a k Ha Hk)). }
        { exact (left_distributive a b k). }
      + intros c d e.
        destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb]. destruct (denotes_total e) as [k Hk].
        apply (denotes_faithful _ _ ((a + b) × k) (a × k + b × k)).
        { exact (denotes_mul _ _ _ _ (denotes_add c d a b Ha Hb) Hk). }
        { exact (denotes_add _ _ _ _ (denotes_mul c e a k Ha Hk) (denotes_mul d e b k Hb Hk)). }
        { exact (right_distributive a b k). }
      + intros c. destruct (denotes_total c) as [a Ha].
        apply (denotes_faithful _ _ (0 × a) 0); [ exact (denotes_mul _ _ _ _ denotes_zero Ha) | exact denotes_zero | ].
        exact (left_absorbing a).
      + intros c. destruct (denotes_total c) as [a Ha].
        apply (denotes_faithful _ _ (a × 0) 0); [ exact (denotes_mul _ _ _ _ Ha denotes_zero) | exact denotes_zero | ].
        exact (right_absorbing a).
      + intros c d.
        destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb].
        apply (denotes_faithful _ _ (a × b) (b × a)).
        { exact (denotes_mul c d a b Ha Hb). }
        { exact (denotes_mul d c b a Hb Ha). }
        { exact (commutative (op := mult) a b). }
    - intros c. destruct (denotes_total c) as [a Ha].
      apply (denotes_faithful _ _ (succ a) (a + 1)).
      { exact (denotes_succ c a Ha). }
      { exact (denotes_add _ _ _ _ Ha denotes_one). }
      { exact (Naturals.succ_add (N := N0) a). }
  Qed.
End NaturalsDenotation.
