A setoid that denotes into a naturals model is a naturals model: if every element denotes some number, denotation is functional and faithful, every number is denoted, and zero, one, successor, addition and multiplication are tracked, then every law of the model transports along the denotation. An implementation -- the Church numerals, say -- only has to provide the denotation facts; the structure is read off here.
Section NaturalsDenotation.
  Context {C : Type} {eqC : EqRel C} {zC : Zero C} {oC : One C}
    {sC : HasSucc C} {aC : HasAdd C} {mC : HasMult C}.
  Context {N0 : Type} `{Naturals N0}.
  Variable denotes : C → N0 → Prop.
  Hypothesis denotes_total : ∀ c : C, ∃ n : N0, denotes c n.
  Hypothesis denotes_sur : ∀ n : N0, ∃ c : C, denotes c n.
  Hypothesis denotes_unique : ∀ (c : C) (n m : N0), denotes c n → denotes c m → n == m.
  Hypothesis denotes_faithful : ∀ (c d : C) (n m : N0), denotes c n → denotes d m → n == m → c == d.
  Hypothesis denotes_proper_l : ∀ (c d : C) (n : N0), c == d → denotes c n → denotes d n.
  Hypothesis denotes_proper_r : ∀ (c : C) (n m : N0), n == m → denotes c n → denotes c m.
  Hypothesis denotes_zero : denotes 0 0.
  Hypothesis denotes_one : denotes 1 1.
  Hypothesis denotes_succ : ∀ (c : C) (n : N0), denotes c n → denotes (succ c) (succ n).
  Hypothesis denotes_add : ∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (c + d) (a + b).
  Hypothesis denotes_mul : ∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (c × d) (a × b).

  Lemma denotation_equivalence : Equivalence (@equ C eqC).
  Proof.
    split.
    - intros c. destruct (denotes_total c) as [n Hn].
      exact (denotes_faithful c c n n Hn Hn (reflexivity n)).
    - intros c d Hcd. destruct (denotes_total c) as [n Hn].
      exact (denotes_faithful d c n n (denotes_proper_l c d n Hcd Hn) Hn (reflexivity n)).
    - intros c d e Hcd Hde. destruct (denotes_total c) as [n Hn].
      exact (denotes_faithful c e n n Hn (denotes_proper_l d e n Hde (denotes_proper_l c d n Hcd Hn)) (reflexivity n)).
  Qed.

Denotations of equal elements are equal numbers.
  Lemma denotes_inject : ∀ (c d : C) (n m : N0), c == d → denotes c n → denotes d m → n == m.
  Proof.
    intros c d n m Hcd Hn Hm. exact (denotes_unique d n m (denotes_proper_l c d n Hcd Hn) Hm).
  Qed.

An operation tracked by the denotation respects ==.
  Lemma denotation_op_proper : ∀ (f : C → C → C) (g : N0 → N0 → N0),
    (∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (f c d) (g a b)) →
    Proper (equ ==> equ ==> equ) f.
  Proof.
    intros f g Hf c c' Hc d d' Hd.
    destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb].
    apply (denotes_faithful _ _ (g a b) (g a b)); [ exact (Hf c d a b Ha Hb) | | exact (reflexivity _) ].
    exact (Hf c' d' a b (denotes_proper_l c c' a Hc Ha) (denotes_proper_l d d' b Hd Hb)).
  Qed.

  Lemma succ_proper : Proper (equ ==> equ) (succ : C → C).
  Proof.
    intros c d Hcd. destruct (denotes_total c) as [n Hn].
    apply (denotes_faithful _ _ (succ n) (succ n)); [ exact (denotes_succ c n Hn) | | exact (reflexivity _) ].
    exact (denotes_succ d n (denotes_proper_l c d n Hcd Hn)).
  Qed.

An equation between terms holds in C when it holds of their denotations: the denotation of a term is the term of the denotations, and the denotation is faithful. One transport per number of variables.
  Lemma transport1 : ∀ (f g : C → C) (f' g' : N0 → N0),
    (∀ (c : C) (a : N0), denotes c a → denotes (f c) (f' a)) →
    (∀ (c : C) (a : N0), denotes c a → denotes (g c) (g' a)) →
    (∀ a : N0, f' a == g' a) → ∀ c : C, f c == g c.
  Proof.
    intros f g f' g' Hf Hg Hlaw c. destruct (denotes_total c) as [a Ha].
    exact (denotes_faithful (f c) (g c) (f' a) (g' a) (Hf c a Ha) (Hg c a Ha) (Hlaw a)).
  Qed.

  Lemma transport2 : ∀ (f g : C → C → C) (f' g' : N0 → N0 → N0),
    (∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (f c d) (f' a b)) →
    (∀ (c d : C) (a b : N0), denotes c a → denotes d b → denotes (g c d) (g' a b)) →
    (∀ a b : N0, f' a b == g' a b) → ∀ c d : C, f c d == g c d.
  Proof.
    intros f g f' g' Hf Hg Hlaw c d.
    destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb].
    exact (denotes_faithful (f c d) (g c d) (f' a b) (g' a b) (Hf c d a b Ha Hb) (Hg c d a b Ha Hb) (Hlaw a b)).
  Qed.

  Lemma transport3 : ∀ (f g : C → C → C → C) (f' g' : N0 → N0 → N0 → N0),
    (∀ (c d e : C) (a b k : N0), denotes c a → denotes d b → denotes e k → denotes (f c d e) (f' a b k)) →
    (∀ (c d e : C) (a b k : N0), denotes c a → denotes d b → denotes e k → denotes (g c d e) (g' a b k)) →
    (∀ a b k : N0, f' a b k == g' a b k) → ∀ c d e : C, f c d e == g c d e.
  Proof.
    intros f g f' g' Hf Hg Hlaw c d e.
    destruct (denotes_total c) as [a Ha]. destruct (denotes_total d) as [b Hb]. destruct (denotes_total e) as [k Hk].
    exact (denotes_faithful (f c d e) (g c d e) (f' a b k) (g' a b k)
             (Hf c d e a b k Ha Hb Hk) (Hg c d e a b k Ha Hb Hk) (Hlaw a b k)).
  Qed.

The laws, each transported from the naturals.
  Lemma add_assoc : ∀ c d e : C, (c + d) + e == c + (d + e).
  Proof.
    apply (transport3 (fun c d e ⇒ (c + d) + e) (fun c d e ⇒ c + (d + e))
                      (fun a b k ⇒ (a + b) + k) (fun a b k ⇒ a + (b + k))).
    - intros c d e a b k Ha Hb Hk. exact (denotes_add _ _ _ _ (denotes_add c d a b Ha Hb) Hk).
    - intros c d e a b k Ha Hb Hk. exact (denotes_add _ _ _ _ Ha (denotes_add d e b k Hb Hk)).
    - intros a b k. exact (associative (op := (add : BinOp N0)) (action := (add : BinOp N0)) a b k).
  Qed.

  Lemma add_zero_l : ∀ c : C, 0 + c == c.
  Proof.
    apply (transport1 (fun c ⇒ 0 + c) (fun c ⇒ c) (fun a ⇒ 0 + a) (fun a ⇒ a)).
    - intros c a Ha. exact (denotes_add _ _ _ _ denotes_zero Ha).
    - intros c a Ha. exact Ha.
    - intros a. exact (left_identity (action := add) a).
  Qed.

  Lemma add_zero_r : ∀ c : C, c + 0 == c.
  Proof.
    apply (transport1 (fun c ⇒ c + 0) (fun c ⇒ c) (fun a ⇒ a + 0) (fun a ⇒ a)).
    - intros c a Ha. exact (denotes_add _ _ _ _ Ha denotes_zero).
    - intros c a Ha. exact Ha.
    - intros a. exact (right_identity (action := add) a).
  Qed.

  Lemma add_comm : ∀ c d : C, c + d == d + c.
  Proof.
    apply (transport2 (fun c d ⇒ c + d) (fun c d ⇒ d + c) (fun a b ⇒ a + b) (fun a b ⇒ b + a)).
    - intros c d a b Ha Hb. exact (denotes_add c d a b Ha Hb).
    - intros c d a b Ha Hb. exact (denotes_add d c b a Hb Ha).
    - intros a b. exact (commutative (op := add) a b).
  Qed.

  Lemma mul_assoc : ∀ c d e : C, (c × d) × e == c × (d × e).
  Proof.
    apply (transport3 (fun c d e ⇒ (c × d) × e) (fun c d e ⇒ c × (d × e))
                      (fun a b k ⇒ (a × b) × k) (fun a b k ⇒ a × (b × k))).
    - intros c d e a b k Ha Hb Hk. exact (denotes_mul _ _ _ _ (denotes_mul c d a b Ha Hb) Hk).
    - intros c d e a b k Ha Hb Hk. exact (denotes_mul _ _ _ _ Ha (denotes_mul d e b k Hb Hk)).
    - intros a b k. exact (associative (op := (mult : BinOp N0)) (action := (mult : BinOp N0)) a b k).
  Qed.

  Lemma mul_one_l : ∀ c : C, 1 × c == c.
  Proof.
    apply (transport1 (fun c ⇒ 1 × c) (fun c ⇒ c) (fun a ⇒ 1 × a) (fun a ⇒ a)).
    - intros c a Ha. exact (denotes_mul _ _ _ _ denotes_one Ha).
    - intros c a Ha. exact Ha.
    - intros a. exact (left_identity (action := mult) a).
  Qed.

  Lemma mul_one_r : ∀ c : C, c × 1 == c.
  Proof.
    apply (transport1 (fun c ⇒ c × 1) (fun c ⇒ c) (fun a ⇒ a × 1) (fun a ⇒ a)).
    - intros c a Ha. exact (denotes_mul _ _ _ _ Ha denotes_one).
    - intros c a Ha. exact Ha.
    - intros a. exact (right_identity (action := mult) a).
  Qed.

  Lemma mul_comm : ∀ c d : C, c × d == d × c.
  Proof.
    apply (transport2 (fun c d ⇒ c × d) (fun c d ⇒ d × c) (fun a b ⇒ a × b) (fun a b ⇒ b × a)).
    - intros c d a b Ha Hb. exact (denotes_mul c d a b Ha Hb).
    - intros c d a b Ha Hb. exact (denotes_mul d c b a Hb Ha).
    - intros a b. exact (commutative (op := mult) a b).
  Qed.

  Lemma mul_add_l : ∀ c d e : C, c × (d + e) == c × d + c × e.
  Proof.
    apply (transport3 (fun c d e ⇒ c × (d + e)) (fun c d e ⇒ c × d + c × e)
                      (fun a b k ⇒ a × (b + k)) (fun a b k ⇒ a × b + a × k)).
    - intros c d e a b k Ha Hb Hk. exact (denotes_mul _ _ _ _ Ha (denotes_add d e b k Hb Hk)).
    - intros c d e a b k Ha Hb Hk. exact (denotes_add _ _ _ _ (denotes_mul c d a b Ha Hb) (denotes_mul c e a k Ha Hk)).
    - intros a b k. exact (left_distributive a b k).
  Qed.

  Lemma mul_add_r : ∀ c d e : C, (c + d) × e == c × e + d × e.
  Proof.
    apply (transport3 (fun c d e ⇒ (c + d) × e) (fun c d e ⇒ c × e + d × e)
                      (fun a b k ⇒ (a + b) × k) (fun a b k ⇒ a × k + b × k)).
    - intros c d e a b k Ha Hb Hk. exact (denotes_mul _ _ _ _ (denotes_add c d a b Ha Hb) Hk).
    - intros c d e a b k Ha Hb Hk. exact (denotes_add _ _ _ _ (denotes_mul c e a k Ha Hk) (denotes_mul d e b k Hb Hk)).
    - intros a b k. exact (right_distributive a b k).
  Qed.

  Lemma mul_zero_l : ∀ c : C, 0 × c == 0.
  Proof.
    apply (transport1 (fun c ⇒ 0 × c) (fun _ ⇒ 0) (fun a ⇒ 0 × a) (fun _ ⇒ 0)).
    - intros c a Ha. exact (denotes_mul _ _ _ _ denotes_zero Ha).
    - intros c a Ha. exact denotes_zero.
    - intros a. exact (left_absorbing a).
  Qed.

  Lemma mul_zero_r : ∀ c : C, c × 0 == 0.
  Proof.
    apply (transport1 (fun c ⇒ c × 0) (fun _ ⇒ 0) (fun a ⇒ a × 0) (fun _ ⇒ 0)).
    - intros c a Ha. exact (denotes_mul _ _ _ _ Ha denotes_zero).
    - intros c a Ha. exact denotes_zero.
    - intros a. exact (right_absorbing a).
  Qed.

  Lemma succ_add_one : ∀ c : C, succ c == c + 1.
  Proof.
    apply (transport1 (fun c ⇒ succ c) (fun c ⇒ c + 1) (fun a ⇒ succ a) (fun a ⇒ a + 1)).
    - intros c a Ha. exact (denotes_succ c a Ha).
    - intros c a Ha. exact (denotes_add _ _ _ _ Ha denotes_one).
    - intros a. exact (Naturals.succ_add (N := N0) a).
  Qed.

  Lemma succ_injective : ∀ c d : C, succ c == succ d → c == d.
  Proof.
    intros c d Hs.
    destruct (denotes_total c) as [n Hn]. destruct (denotes_total d) as [m Hm].
    apply (denotes_faithful c d n m Hn Hm).
    apply (injective (f := (succ : N0 → N0))).
    exact (denotes_inject _ _ _ _ Hs (denotes_succ c n Hn) (denotes_succ d m Hm)).
  Qed.

  Lemma succ_not_zero : ∀ c : C, succ c =/= 0.
  Proof.
    intros c Hc. destruct (denotes_total c) as [n Hn].
    apply (Peano.succ_not_zero (N := N0) n).
    exact (denotes_inject _ _ _ _ Hc (denotes_succ c n Hn) denotes_zero).
  Qed.

  Lemma denotes_iff : ∀ (c : C) (n m : N0), n == m → (denotes c n ↔ denotes c m).
  Proof. intros c n m Hnm. split; [ exact (denotes_proper_r c n m Hnm) | exact (denotes_proper_r c m n (symmetry Hnm)) ]. Qed.

Induction transports along the denotation: a property of the elements denoting n is proved by induction on n.
  Lemma induction : ∀ P : C → Prop, Proper (equ ==> iff) P →
    P 0 → (∀ c : C, P c → P (succ c)) → ∀ c : C, P c.
  Proof.
    intros P HP H0 HS c.
    destruct (denotes_total c) as [n Hn]. revert c Hn.
    apply (Peano.nat_induction (N := N0) (fun n ⇒ ∀ c : C, denotes c n → P c)).
    - intros k k' Hkk'.
      exact (all_iff_morphism _ _ (fun c' ⇒ iff_iff_iff_impl_morphism _ _ (denotes_iff c' k k' Hkk') _ _ (iff_refl (P c')))).
    - intros c' Hc'.
      exact (proj1 (HP 0 c' (denotes_faithful 0 c' 0 0 denotes_zero Hc' (reflexivity 0))) H0).
    - intros m IH c' Hc'.
      destruct (denotes_sur m) as [d Hd].
      exact (proj1 (HP (succ d) c' (denotes_faithful (succ d) c' (succ m) (succ m) (denotes_succ d m Hd) Hc' (reflexivity _)))
               (HS d (IH d Hd))).
  Qed.

The structure, read off the laws.
  Instance denotation_magma_add : Magma C add :=
    { Magma.equivalence := denotation_equivalence; Magma.op_proper := denotation_op_proper add add denotes_add }.
  Instance denotation_magma_mul : Magma C mult :=
    { Magma.equivalence := denotation_equivalence; Magma.op_proper := denotation_op_proper mult mult denotes_mul }.
  Instance denotation_add_semigroup : Semigroup C add :=
    { Semigroup.magma := denotation_magma_add; Semigroup.is_associative := add_assoc }.
  Instance denotation_mul_semigroup : Semigroup C mult :=
    { Semigroup.magma := denotation_magma_mul; Semigroup.is_associative := mul_assoc }.
  Instance denotation_add_monoid : Monoid C 0 add :=
    { Monoid.semigroup := denotation_add_semigroup;
      Monoid.is_left_identity := add_zero_l; Monoid.is_right_identity := add_zero_r }.
  Instance denotation_mul_monoid : Monoid C 1 mult :=
    { Monoid.semigroup := denotation_mul_semigroup;
      Monoid.is_left_identity := mul_one_l; Monoid.is_right_identity := mul_one_r }.
  Instance denotation_semiring : Semiring C 0 add 1 mult :=
    { Semiring.add_monoid := denotation_add_monoid; Semiring.add_commutative := add_comm;
      Semiring.mul_monoid := denotation_mul_monoid;
      Semiring.is_left_distributive := mul_add_l; Semiring.is_right_distributive := mul_add_r;
      Semiring.mul_zero_left := mul_zero_l; Semiring.mul_zero_right := mul_zero_r }.
  Instance denotation_comm_semiring : CommSemiring C 0 add 1 mult :=
    { CommSemiring.semiring := denotation_semiring; CommSemiring.mul_commutative := mul_comm }.
  Instance denotation_peano : Peano C 0 succ :=
    { Peano.equivalence := denotation_equivalence; Peano.succ_proper := succ_proper;
      Peano.succ_injective := succ_injective; Peano.succ_not_zero := succ_not_zero;
      Peano.nat_induction := induction }.

  Instance denotation_naturals : Naturals C 0 succ 1 add mult :=
    { Naturals.peano := denotation_peano; Naturals.semiring := denotation_comm_semiring;
      Naturals.succ_add := succ_add_one }.
