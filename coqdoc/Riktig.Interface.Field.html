Library Riktig.Interface.Field

From Riktig.Interface Require Export Ring.

Module Field.
  Class Field (F : Type) {eq : EqRel F} {zero : Zero F} {add : HasAdd F}
              {neg : HasNeg F} {one : One F} {mult : HasMult F}
              {inv : HasInv F} := {
    ring :: Ring F;
    inv_proper :: Proper (equ ==> equ) inv;
    is_mul_commutative :: Commutative F mult;
    is_mul_left_inverse :: LeftInverseOn F (nonzero F) mult inv 1;

    nontrivial : (1 : F) =/= 0;
  }.
  Hint Mode Field ! - - - - - - - : typeclass_instances.
End Field.
Export Field.
