Library Riktig.Interface.Field

From Coq Require Export Morphisms.
From Riktig Require Export Ops Props Interface.Ring.

Module Field.
  Class Field (F : Type) {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F}
              {Fo : One F} {Fm : HasMult F} {Fi : HasInv F} := {
    equivalence : Equivalence (@equ F _);
    associative : ∀ x y w : F, (x + y) + w == x + (y + w);
    left_identity : ∀ x : F, 0 + x == x;
    right_identity : ∀ x : F, x + 0 == x;
    left_inverse : ∀ x : F, (- x) + x == 0;
    right_inverse : ∀ x : F, x + (- x) == 0;
    commutative : ∀ x y : F, x + y == y + x;
    mul_associative : ∀ x y w : F, (x × y) × w == x × (y × w);
    mul_left_identity : ∀ x : F, 1 × x == x;
    mul_right_identity : ∀ x : F, x × 1 == x;
    left_distributive : ∀ x y w : F, x × (y + w) == x × y + x × w;
    right_distributive : ∀ x y w : F, (x + y) × w == x × w + y × w;
    mul_commutative : ∀ x y : F, x × y == y × x;
    
    mul_left_inverse : ∀ x : F, x =/= 0 → inv x × x == 1;
    nontrivial : (1 : F) =/= 0;
    add_proper : ∀ x y : F, x == y → ∀ u v : F, u == v → (x + u) == (y + v);
    neg_proper : ∀ x y : F, x == y → (- x) == (- y);
    mul_proper : ∀ x y : F, x == y → ∀ u v : F, u == v → (x × u) == (y × v);
    inv_proper : ∀ x y : F, x == y → inv x == inv y
  }.
End Field.
Notation Field := Field.Field.

#[global] Instance Field_is_Ring F {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F}
  {Fo : One F} {Fm : HasMult F} {Fi : HasInv F} `{H : Field.Field F} : Ring F := {|
    Ring.equivalence := Field.equivalence;
    Ring.associative := Field.associative;
    Ring.left_identity := Field.left_identity;
    Ring.right_identity := Field.right_identity;
    Ring.left_inverse := Field.left_inverse;
    Ring.right_inverse := Field.right_inverse;
    Ring.commutative := Field.commutative;
    Ring.mul_associative := Field.mul_associative;
    Ring.mul_left_identity := Field.mul_left_identity;
    Ring.mul_right_identity := Field.mul_right_identity;
    Ring.left_distributive := Field.left_distributive;
    Ring.right_distributive := Field.right_distributive;
    Ring.add_proper := Field.add_proper;
    Ring.neg_proper := Field.neg_proper;
    Ring.mul_proper := Field.mul_proper
  |}.

#[global] Instance Field_inv_Proper F {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F}
  {Fo : One F} {Fm : HasMult F} {Fi : HasInv F} `{H : Field.Field F} : Proper (equ ==> equ) (@Ops.inv F _)
  := Field.inv_proper.

#[global] Instance Field_MulCommutative F {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F} {Fn : HasNeg F}
  {Fo : One F} {Fm : HasMult F} {Fi : HasInv F} `{H : Field.Field F} : Commutes F mult := Field.mul_commutative.
