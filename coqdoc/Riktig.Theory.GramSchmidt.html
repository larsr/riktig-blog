From Coq Require Export List.
From Riktig Require Export Interface.PosInner Theory.InnerProductSpace

Section GramSchmidtKit.
  Context {F V : Type} `{PosInner F V}.

  Fixpoint gram_schmidt_aux (done : list V) (v : V) : V :=
    match done with
    | nil ⇒ v
    | cons d done' ⇒ gram_schmidt_aux done' (v + - proj d v)
    end.

  Fixpoint gram_schmidt_step (acc vs : list V) : list V :=
    match vs with
    | nil ⇒ acc
    | cons v vs' ⇒ gram_schmidt_step (acc ++ (gram_schmidt_aux acc v :: nil)) vs'
    end.

  Definition gram_schmidt (vs : list V) : list V := gram_schmidt_step nil vs.

  Lemma gram_schmidt_aux_nil : ∀ v : V, gram_schmidt_aux nil v == v.
  Proof.
    intros v.
    reflexivity.
  Qed.

  Lemma gram_schmidt_aux_cons : ∀ (d : V) (done' : list V) (v : V),
    gram_schmidt_aux (d :: done') v == gram_schmidt_aux done' (v + - proj d v).
  Proof.
    intros d done' v.
    reflexivity.
  Qed.

  Definition PairwiseOrth (l : list V) : Prop :=
    ∀ d1 d2 : V, In d1 l → In d2 l → orthogonal d1 d2 ∨ d1 = d2.

  Lemma pairwise_orth_nil : PairwiseOrth (nil : list V).
  Proof.
    intros d1 d2 Hd1 Hd2.
    destruct Hd1.
  Qed.

  Lemma nonzero_nil : ∀ d : V, In d (nil : list V) → norm2 d =/= 0.
  Proof.
    intros d Hd.
    destruct Hd.
  Qed.

  Lemma in_singleton : ∀ x y : V, In x (y :: nil) → x = y.
  Proof.
    intros x y Hx.
    destruct Hx as [Heq | Hf].
    - symmetry.
      exact Heq.
    - destruct Hf.
  Qed.

  Lemma orthogonal_sub : ∀ u v w : V,
    orthogonal u w → orthogonal v w → orthogonal (u + - v) w.
  Proof.
    intros u v w Hu Hv.
    apply orthogonal_def.
    calc (⟨ u + - v , w ⟩)
      == (⟨ u , w ⟩ + ⟨ - v , w ⟩) { (inner_left_additive u (- v) w) }
      == (⟨ u , w ⟩ + - ⟨ v , w ⟩) { (inner_left_neg v w) }
      == (0 + - ⟨ v , w ⟩) { (orthogonal_eq u w Hu) }
      == (0 + - (0 : F)) { (orthogonal_eq v w Hv) }
      == ((0 : F) + 0) { neg_zero }
      == (0 : F) { right_identity }.
  Qed.

  Lemma gram_schmidt_aux_preserves_orth : ∀ (done : list V) (d0 : V),
    (∀ d : V, In d done → orthogonal d0 d ∨ d = d0) →
    ∀ w : V, orthogonal w d0 → orthogonal (gram_schmidt_aux done w) d0.
  Proof.
    intros done; induction done as [| d1 done' IH]; intros d0 Hdone w Hw.
    - exact Hw.
    - apply IH.
      + intros d Hd.
        apply Hdone.
        right.
        exact Hd.
      + destruct (Hdone d1 (or_introl eq_refl)) as [Hortho | Heq].
        × apply orthogonal_sub.
          -- exact Hw.
          -- exact (proj_orthogonal_to d0 d1 w Hortho).
        × apply orthogonal_def.
          calc (⟨ w + - proj d1 w , d0 ⟩)
            == (⟨ w + - proj d0 w , d0 ⟩) { Heq }
            == (⟨ w + - (0 : V) , d0 ⟩) { (proj_self_zero d0 w Hw) }
            == (⟨ w + (0 : V) , d0 ⟩) { neg_zero }
            == (⟨ w , d0 ⟩) { (right_identity w) }
            == (0 : F) { (orthogonal_eq w d0 Hw) }.
  Qed.

  Lemma gram_schmidt_aux_orth : ∀ done : list V,
    PairwiseOrth done →
    (∀ d : V, In d done → norm2 d =/= 0) →
    ∀ v d : V, In d done → orthogonal (gram_schmidt_aux done v) d.
  Proof.
    intros done; induction done as [| d0 done' IH]; intros Hpw Hnz v d Hd.
    - destruct Hd.
    - destruct Hd as [Heq | Hin].
      + rewrite <- Heq.
        apply (gram_schmidt_aux_preserves_orth done' d0).
        × intros d' Hd'.
          destruct (Hpw d0 d' (or_introl eq_refl) (or_intror Hd')) as [Ho | He].
          -- left. exact Ho.
          -- right. symmetry. exact He.
        × exact (proj_orthogonal v d0 (Hnz d0 (or_introl eq_refl))).
      + apply IH.
        × intros e1 e2 He1 He2.
          apply Hpw.
          -- right. exact He1.
          -- right. exact He2.
        × intros e He.
          apply Hnz.
          right.
          exact He.
        × exact Hin.
  Qed.

  Lemma gram_schmidt_step_extends : ∀ (vs acc : list V) (d : V),
    In d acc → In d (gram_schmidt_step acc vs).
  Proof.
    intros vs; induction vs as [| v vs' IH]; intros acc d Hd.
    - exact Hd.
    - apply IH.
      apply in_or_app.
      left.
      exact Hd.
  Qed.

  Lemma gram_schmidt_step_pairwise_extend : ∀ (acc : list V) (v : V),
    PairwiseOrth acc →
    (∀ d : V, In d acc → norm2 d =/= 0) →
    PairwiseOrth (acc ++ (gram_schmidt_aux acc v :: nil)).
  Proof.
    intros acc v Hpw Hnz d1 d2 Hd1 Hd2.
    destruct (in_app_or acc (gram_schmidt_aux acc v :: nil) d1 Hd1) as [Hd1a | Hd1w].
    - destruct (in_app_or acc (gram_schmidt_aux acc v :: nil) d2 Hd2) as [Hd2a | Hd2w].
      + exact (Hpw d1 d2 Hd1a Hd2a).
      + left.
        rewrite (in_singleton d2 (gram_schmidt_aux acc v) Hd2w).
        apply orthogonal_sym.
        exact (gram_schmidt_aux_orth acc Hpw Hnz v d1 Hd1a).
    - destruct (in_app_or acc (gram_schmidt_aux acc v :: nil) d2 Hd2) as [Hd2a | Hd2w].
      + left.
        rewrite (in_singleton d1 (gram_schmidt_aux acc v) Hd1w).
        exact (gram_schmidt_aux_orth acc Hpw Hnz v d2 Hd2a).
      + right.
        rewrite (in_singleton d1 (gram_schmidt_aux acc v) Hd1w),
          (in_singleton d2 (gram_schmidt_aux acc v) Hd2w).
        reflexivity.
  Qed.

  Lemma gram_schmidt_step_new_nonzero : ∀ (vs acc : list V) (v : V),
    (∀ d : V, In d (gram_schmidt_step acc (v :: vs)) → norm2 d =/= 0) →
    norm2 (gram_schmidt_aux acc v) =/= 0.
  Proof.
    intros vs acc v Hnzf.
    apply Hnzf.
    apply (gram_schmidt_step_extends vs (acc ++ (gram_schmidt_aux acc v :: nil))).
    apply in_or_app.
    right.
    left.
    reflexivity.
  Qed.

  Theorem gram_schmidt_step_pairwise : ∀ (vs acc : list V),
    PairwiseOrth acc →
    (∀ d : V, In d acc → norm2 d =/= 0) →
    (∀ d : V, In d (gram_schmidt_step acc vs) → norm2 d =/= 0) →
    PairwiseOrth (gram_schmidt_step acc vs).
  Proof.
    intros vs; induction vs as [| v vs' IH]; intros acc Hpw Hnz Hnzf.
    - exact Hpw.
    - apply IH.
      + exact (gram_schmidt_step_pairwise_extend acc v Hpw Hnz).
      + intros d Hd.
        destruct (in_app_or acc (gram_schmidt_aux acc v :: nil) d Hd) as [Hda | Hdw].
        × exact (Hnz d Hda).
        × rewrite (in_singleton d (gram_schmidt_aux acc v) Hdw).
          exact (gram_schmidt_step_new_nonzero vs' acc v Hnzf).
      + exact Hnzf.
  Qed.

  Theorem gram_schmidt_pairwise_orthogonal : ∀ vs : list V,
    (∀ d : V, In d (gram_schmidt vs) → norm2 d =/= 0) →
    ∀ d1 d2 : V,
      In d1 (gram_schmidt vs) → In d2 (gram_schmidt vs) → d1 ≠ d2 → orthogonal d1 d2.
  Proof.
    intros vs Hnz d1 d2 Hd1 Hd2 Hneq.
    destruct (gram_schmidt_step_pairwise vs nil pairwise_orth_nil nonzero_nil Hnz
                d1 d2 Hd1 Hd2) as [Ho | Heq].
    - exact Ho.
    - destruct (Hneq Heq).
  Qed.

  Lemma gram_schmidt_aux_combination : ∀ (done : list V) (v : V),
    ∃ coeffs : list F, gram_schmidt_aux done v == v + list_lincomb coeffs done.
  Proof.
    intros done; induction done as [| d done' IH]; intros v.
    - ∃ nil.
      rewrite gram_schmidt_aux_nil, list_lincomb_nil_l.
      symmetry.
      exact (right_identity _).
    - destruct (IH (v + - proj d v)) as [coeffs' Heq].
      ∃ ((- (⟨ v , d ⟩ × inv (norm2 d))) :: coeffs').
      calc (gram_schmidt_aux (d :: done') v)
        == (gram_schmidt_aux done' (v + - proj d v))
          { (gram_schmidt_aux_cons d done' v) }
        == ((v + - proj d v) + list_lincomb coeffs' done')
          { Heq }
        == ((v + - ((⟨ v , d ⟩ × inv (norm2 d)) *: d)) + list_lincomb coeffs' done')
          { (proj_eq v d) }
        == ((v + (- (⟨ v , d ⟩ × inv (norm2 d))) *: d) + list_lincomb coeffs' done')
          { (neg_scale (⟨ v , d ⟩ × inv (norm2 d)) d) }
        == (v + ((- (⟨ v , d ⟩ × inv (norm2 d))) *: d + list_lincomb coeffs' done'))
          { associative }
        == (v + list_lincomb ((- (⟨ v , d ⟩ × inv (norm2 d))) :: coeffs') (d :: done'))
          { (list_lincomb_cons (- (⟨ v , d ⟩ × inv (norm2 d))) coeffs' d done') }.
  Qed.

  Lemma gram_schmidt_step_new_combination : ∀ (vs acc : list V) (w : V),
    In w (gram_schmidt_step acc vs) →
    In w acc ∨
    ∃ (v : V) (pre : list V) (coeffs : list F),
      In v vs ∧ (∀ d : V, In d pre → In d (gram_schmidt_step acc vs)) ∧
      w == v + list_lincomb coeffs pre.
  Proof.
    intros vs; induction vs as [| v' vs' IH]; intros acc w Hw.
    - left. exact Hw.
    - destruct (IH (acc ++ (gram_schmidt_aux acc v' :: nil)) w Hw)
        as [Hacc' | [v [pre [coeffs [Hv [Hpre Heqw]]]]]].
      + destruct (in_app_or acc (gram_schmidt_aux acc v' :: nil) w Hacc')
          as [Hina | Hinw].
        × left. exact Hina.
        × right.
          destruct (gram_schmidt_aux_combination acc v') as [coeffs0 Heq0].
          ∃ v', acc, coeffs0.
          split.
          -- left. reflexivity.
          -- split.
             ++ intros d Hd.
                apply (gram_schmidt_step_extends vs' (acc ++ (gram_schmidt_aux acc v' :: nil)) d).
                apply in_or_app.
                left. exact Hd.
             ++ rewrite (in_singleton w (gram_schmidt_aux acc v') Hinw).
                exact Heq0.
      + right.
        ∃ v, pre, coeffs.
        split.
        × right. exact Hv.
        × split.
          -- exact Hpre.
          -- exact Heqw.
  Qed.

  Theorem gram_schmidt_element_combination : ∀ (vs : list V) (w : V),
    In w (gram_schmidt vs) →
    ∃ (v : V) (pre : list V) (coeffs : list F),
      In v vs ∧ (∀ d : V, In d pre → In d (gram_schmidt vs)) ∧
      w == v + list_lincomb coeffs pre.
  Proof.
    intros vs w Hw.
    destruct (gram_schmidt_step_new_combination vs nil w Hw) as [Hnil | Hex].
    - destruct Hnil.
    - exact Hex.
  Qed.
