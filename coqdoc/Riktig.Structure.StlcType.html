From Riktig.Structure Require Export Adt.

The simple types over base types named by b: the datatype of the signature const b + self × self, one level being a base type or an arrow between two holes.
Module StlcType.
  Notation shape den b := (shape_of den (styS (S := sig _) b)).

  Class StlcType (O : Type) `{Bicartesian O} {K : Type}
      (den : K → O) (b : K) (Ty : O)
      (build : HasBuild O (shape den b) Ty)
      (fold : HasCata O (shape den b) Ty) := {
    initial :: InitialAlgebra O (shape den b) Ty build fold;
  }.
End StlcType.
Export StlcType(StlcType).
#[export] Existing Instances StlcType.initial.
