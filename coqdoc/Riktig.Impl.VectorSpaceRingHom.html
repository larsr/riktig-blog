A ring with a ring homomorphism from a field is a vector space over that field: a scalar acts by multiplying with its image.
Section RingHomVS.
  Context {F R : Type} `{Field F} `{Ring R} (h : F → R) {Hh : RingHom F R h}.

  #[export] Instance RingHom_Scale : HasScale F R := fun a x ⇒ h a × x.

  #[export] Instance RingHom_ScaleLaws : ScaleLaws F R add RingHom_Scale.
  Proof.
    split.
    - intros a a' Ha x x' Hx. unfold scale, RingHom_Scale.
      apply op_proper; [ apply hom_proper; exact Ha | exact Hx ].
    - intros a x y. exact (left_distributive (op := mult) (h a) x y).
    - intros a b x. unfold scale, RingHom_Scale.
      calc (h (a + b) × x)
        == ((h a + h b) × x) { (additive (f := h) a b) }
        == (h a × x + h b × x) { (right_distributive (op := mult) (h a) (h b) x) }.
    - intros a b x. unfold scale, RingHom_Scale.
      calc (h (a × b) × x)
        == ((h a × h b) × x) { (additive (f := h) (opA := mult) a b) }
        == (h a × (h b × x)) { (associative (op := mult) (action := mult) (h a) (h b) x) }.
    - intros x. unfold scale, RingHom_Scale.
      calc (h 1 × x)
        == ((1 : R) × x) { hom_one }
        == x { (left_identity (action := mult) x) }.
  Qed.

  #[export] Instance RingHom_VectorSpace : VectorSpace F R 0 add neg RingHom_Scale :=
    {| VectorSpace.vectors := abgroup; VectorSpace.scaling := RingHom_ScaleLaws |}.
End RingHomVS.
