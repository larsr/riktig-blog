Section BilinearPartial.
  Context {F U V W : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context (b : BilinearMap F U V W).

  Lemma bilinear_map_appl_proper : ∀ u : U,
    Proper (equ ==> equ) (fun v : V ⇒ b u v).
  Proof.
    intros u v v' Hv.
    exact (bilinear_map_proper b u u (reflexivity u) v v' Hv).
  Qed.

  Definition bilinear_map_appl (u : U) : LinearMap F V W :=
    linear_map_from_laws
      (fun v : V ⇒ b u v)
      (bilinear_map_appl_proper u)
      (bilinear_right_additive b u)
      (fun (a : F) (v : V) ⇒ bilinear_right_scale b a u v).

  Lemma bilinear_map_appr_proper : ∀ v : V,
    Proper (equ ==> equ) (fun u : U ⇒ b u v).
  Proof.
    intros v u u' Hu.
    exact (bilinear_map_proper b u u' Hu v v (reflexivity v)).
  Qed.

  Definition bilinear_map_appr (v : V) : LinearMap F U W :=
    linear_map_from_laws
      (fun u : U ⇒ b u v)
      (bilinear_map_appr_proper v)
      (fun u u' : U ⇒ bilinear_left_additive b u u' v)
      (fun (a : F) (u : U) ⇒ bilinear_left_scale b a u v).

  Theorem bilinear_map_zero_l : ∀ v : V, b 0 v == 0.
  Proof.
    intros v.
    exact (linear_map_zero (bilinear_map_appr v)).
  Qed.

  Theorem bilinear_map_zero_r : ∀ u : U, b u 0 == 0.
  Proof.
    intros u.
    exact (linear_map_zero (bilinear_map_appl u)).
  Qed.

End BilinearPartial.

Lemma upd_at {V I : Type} {EV : EqRel V} {Ieq : EqDec I}
    (RV : Reflexive (@equ V EV)) :
  ∀ (vs : I → V) (i : I) (x : V),
    upd vs i x i == x.
Proof.
  intros vs i x.
  unfold upd; destruct (dec (i = i)) as [_ | Hne].
  - reflexivity.
  - destruct (Hne eq_refl).
Qed.

Lemma upd_off {V I : Type} {EV : EqRel V} {Ieq : EqDec I}
    (RV : Reflexive (@equ V EV)) :
  ∀ (vs : I → V) (i j : I) (x : V),
    i ≠ j → upd vs i x j == vs j.
Proof.
  intros vs i j x Hij.
  unfold upd; destruct (dec (i = j)) as [Heq | _].
  - destruct (Hij Heq).
  - reflexivity.
Qed.

Section ZeroBilinear.
  Context {F U V W : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Lemma bilinear_zero_map_proper :
    Proper (equ ==> equ ==> equ) (fun (_ : U) (_ : V) ⇒ (0 : W)).
  Proof.
    intros u u' _ v v' _.
    reflexivity.
  Qed.

  Lemma bilinear_zero_map_add_l : ∀ (u u' : U) (v : V), (0 : W) == 0 + 0.
  Proof.
    intros u u' v.
    rewrite (left_identity (0 : W)).
    reflexivity.
  Qed.

  Lemma bilinear_zero_map_add_r : ∀ (u : U) (v v' : V), (0 : W) == 0 + 0.
  Proof.
    intros u v v'.
    rewrite (left_identity (0 : W)).
    reflexivity.
  Qed.

  Lemma bilinear_zero_map_scale : ∀ (a : F) (u : U) (v : V), (0 : W) == a *: 0.
  Proof.
    intros a u v.
    rewrite (scale_zero a).
    reflexivity.
  Qed.

  Example bilinear_zero_map : BilinearMap F U V W :=
    bilinear_map_from_laws
      (fun (_ : U) (_ : V) ⇒ (0 : W))
      bilinear_zero_map_proper
      bilinear_zero_map_add_l
      bilinear_zero_map_scale
      bilinear_zero_map_add_r
      bilinear_zero_map_scale.

End ZeroBilinear.
