From Riktig.Structure Require Export Category.

A functor between categories sends objects to objects and arrows to arrows, respecting equality, identities and composition.
Module Functor.
  Class Functor (C D : Type) `{Category C} `{Category D} (F : C → D) (map : HasFmap C D F) := {
    fmap_proper :: ∀ x y : C, Proper (equ ==> equ) (@fmap C D _ _ F map x y);
    fmap_id : ∀ x : C, fmap (F := F) (cat_id (x := x)) == cat_id;
    fmap_comp : ∀ (x y z : C) (f : y ~> z) (g : x ~> y),
      fmap (F := F) (f ∘ g) == fmap (F := F) f ∘ fmap (F := F) g;
  }.
  Hint Mode Functor ! ! - - - - - - - - - - ! - : typeclass_instances.
End Functor.
Export Functor.
