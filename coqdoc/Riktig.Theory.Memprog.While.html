Local Notation interpZ :=
  (interp Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation EvalZ :=
  (Eval Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation interp_soundZ :=
  (interp_sound Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation eval_interpZ :=
  (eval_interp Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).


Lemma then_while_m {addr val b : Type}
    (c : MemT addr val bool) (body : MemT addr val unit) (k : MemT addr val b) :
  then_t (while_m c body) k = While c body k.
Proof. reflexivity. Qed.

Section WhileRules.
  Context (c : MemT Z Z bool) (body : MemT Z Z unit).
  Context (invariant exit_at : Machine Z Z → Prop).



  Definition while_invariant_step (measure : Machine Z Z → nat) : Prop :=
    ∀ m : Machine Z Z,
      invariant m →
      ∃ (taken : bool) (m1 : Machine Z Z),
        EvalZ c m (Some taken) m1 ∧
        (taken = false → exit_at m1) ∧
        (taken = true →
           ∃ m2 : Machine Z Z,
             EvalZ body m1 (Some tt) m2 ∧
             invariant m2 ∧
             (measure m2 < measure m)%nat).


  Theorem eval_while_measure (measure : Machine Z Z → nat) :
    while_invariant_step measure →
    ∀ m : Machine Z Z,
      invariant m →
      ∃ m1 : Machine Z Z,
        exit_at m1 ∧
        ∀ (b : Type) (k : MemT Z Z b) (r : option b) (m' : Machine Z Z),
          EvalZ k m1 r m' → EvalZ (While c body k) m r m'.
  Proof.
    intros Hstep m.
    induction m as [m IH]
      using (well_founded_ind (well_founded_ltof (Machine Z Z) measure)).
    intros Hinv.
    destruct (Hstep m Hinv) as (taken & m1 & Hc & Hfalse & Htrue).
    destruct taken.
    - destruct (Htrue eq_refl) as (m2 & Hbody & Hinv2 & Hdec).
      destruct (IH m2 Hdec Hinv2) as (mexit & Hexit & Hrun).
      ∃ mexit. split; [exact Hexit|].
      intros b k r m' Hk.
      eapply EvWhileTrue; [exact Hc | exact Hbody | exact (Hrun b k r m' Hk)].
    - ∃ m1. split; [exact (Hfalse eq_refl)|].
      intros b k r m' Hk. eapply EvWhileFalse; [exact Hc | exact Hk].
  Qed.

  Corollary eval_while_m_measure (measure : Machine Z Z → nat) :
    while_invariant_step measure →
    ∀ m : Machine Z Z,
      invariant m →
      ∃ m1 : Machine Z Z,
        exit_at m1 ∧ EvalZ (while_m c body) m (Some tt) m1.
  Proof.
    intros Hstep m Hinv.
    destruct (eval_while_measure measure Hstep m Hinv) as (m1 & Hexit & Hrun).
    ∃ m1. split; [exact Hexit|].
    apply (Hrun unit (ret_t tt)). apply eval_ret_t.
  Qed.



  Section PartialCorrectness.
    Hypothesis exit_at_false :
      ∀ m m1 : Machine Z Z,
        invariant m → EvalZ c m (Some false) m1 → exit_at m1.
    Hypothesis invariant_preserved :
      ∀ m m1 m2 : Machine Z Z,
        invariant m →
        EvalZ c m (Some true) m1 →
        EvalZ body m1 (Some tt) m2 →
        invariant m2.

    Lemma interp_while_invariant :
      ∀ (b : Type) (k : MemT Z Z b) (fuel : nat)
             (m : Machine Z Z) (r : option b) (m' : Machine Z Z),
        interpZ fuel (While c body k) m = (Some r, m') →
        invariant m →
        ∃ m1 : Machine Z Z, exit_at m1 ∧ EvalZ k m1 r m'.
    Proof.
      intros b k fuel. induction fuel as [|f IH]; intros m r m' Hrun Hinv.
      - cbn [interp] in Hrun. discriminate.
      - cbn [interp] in Hrun.
        destruct (interpZ f c m) as [rc mach1] eqn:Ec.
        destruct rc as [[[|]|]|]; try discriminate.
        +
          destruct (interpZ f body mach1) as [rb mach2] eqn:Eb.
          destruct rb as [[u|]|]; try discriminate.
          destruct u.
          refine (IH mach2 r m' Hrun _).
          exact (invariant_preserved m mach1 mach2 Hinv
                   (interp_soundZ f _ c m _ mach1 Ec)
                   (interp_soundZ f _ body mach1 _ mach2 Eb)).
        +
          ∃ mach1. split.
          × exact (exit_at_false m mach1 Hinv (interp_soundZ f _ c m _ mach1 Ec)).
          × exact (interp_soundZ f _ k mach1 r m' Hrun).
    Qed.

    Theorem while_eval_invariant :
      ∀ (b : Type) (k : MemT Z Z b)
             (m : Machine Z Z) (r : option b) (m' : Machine Z Z),
        EvalZ (While c body k) m r m' →
        invariant m →
        ∃ m1 : Machine Z Z, exit_at m1 ∧ EvalZ k m1 r m'.
    Proof.
      intros b k m r m' Hev Hinv.
      apply eval_interpZ in Hev as (fuel & Hfuel).
      exact (interp_while_invariant b k fuel m r m' Hfuel Hinv).
    Qed.

    Corollary while_m_eval_invariant :
      ∀ (m : Machine Z Z) (r : option unit) (m' : Machine Z Z),
        EvalZ (while_m c body) m r m' →
        invariant m →
        r = Some tt ∧ exit_at m'.
    Proof.
      intros m r m' Hev Hinv.
      destruct (while_eval_invariant unit (ret_t tt) m r m' Hev Hinv)
        as (m1 & Hexit & Hk).
      apply eval_ret in Hk. destruct Hk as [Hr Hm]. subst.
      exact (conj eq_refl Hexit).
    Qed.

  End PartialCorrectness.
End WhileRules.
