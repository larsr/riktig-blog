From Coq Require Import Arith.
From Riktig Require Export Structure.PeanoArith.

Coq's nat, with its recursion equations.
Instance Nat_EqRel : EqRel nat := eq.
Instance Nat_Zero : Zero nat := O.
Instance Nat_HasSucc : HasSucc nat := S.
Instance Nat_One : One nat := S O.
Instance Nat_HasAdd : HasAdd nat := Nat.add.
Instance Nat_HasMult : HasMult nat := Nat.mul.

Instance Nat_Peano : Peano nat Nat_Zero Nat_HasSucc :=
  {| Peano.equivalence := eq_equivalence;
     Peano.succ_proper := f_equal S;
     Peano.succ_injective := eq_add_S;
     Peano.succ_not_zero := Nat.neq_succ_0;
     Peano.nat_induction := fun P _ H0 HS n ⇒ nat_ind P H0 HS n |}.

Instance Nat_PeanoArith : PeanoArith nat Nat_Zero Nat_HasSucc Nat_One Nat_HasAdd Nat_HasMult :=
  {| PeanoArith.peano := Nat_Peano;
     PeanoArith.add_proper := fun x y Hx u v Hu ⇒ f_equal2 Nat.add Hx Hu;
     PeanoArith.mul_proper := fun x y Hx u v Hu ⇒ f_equal2 Nat.mul Hx Hu;
     PeanoArith.one_succ := eq_refl;
     PeanoArith.add_zero := Nat.add_0_r;
     PeanoArith.add_succ := Nat.add_succ_r;
     PeanoArith.mul_zero := Nat.mul_0_r;
     PeanoArith.mul_succ := Nat.mul_succ_r |}.
