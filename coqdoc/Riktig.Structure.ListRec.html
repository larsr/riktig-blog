Recursion on lists: from a value at the empty list and a step at each element, a function on lists, given by its two equations, which respects equality when the step does.
Module ListRec.
  Class ListRec (X L : Type) {eqX : EqRel X} {eqL : EqRel L} (emptyL : HasEmpty L)
      (prependL : HasPrepend X L) (rec : HasListRec X L) := {
    list_rec_proper : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (n : B) (c : X → B → B), Proper (equ ==> equ ==> equ) c →
      Proper (equ ==> equ) (list_rec B n c : L → B);
    list_rec_empty : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (n : B) (c : X → B → B),
      list_rec B n c (empty : L) == n;
    list_rec_prepend : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (n : B) (c : X → B → B) (x : X) (l : L),
      list_rec B n c (prepend x l) == c x (list_rec B n c l);
  }.
End ListRec.
