A determinant on V with respect to a basis induces a determinant on the table space over the same index, with respect to the unit tables: a family of tables is read as the family of the vectors with those coordinates. This is the determinant of a square matrix given by its columns.

Linear combinations of the basis

  Lemma lincomb_step_proper : ∀ t : T,
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : V) ⇒ acc + get t (elem k) *: basis (elem k)).
  Proof.
    intros t k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
    apply scale_proper; [ apply get_proper; [ reflexivity | apply elem_proper; exact Hk ] | apply basis_proper; apply elem_proper; exact Hk ].
  Qed.

  Lemma lincomb_proper : ∀ t t' : T, t == t' → lincomb t == lincomb t'.
  Proof.
    intros t t' Htt.
    apply (natrec_ext V _ _ 0 _ _ (lincomb_step_proper t) (lincomb_step_proper t')).
    intros k acc. apply op_proper; [ reflexivity | ].
    apply scale_proper; [ apply get_proper; [ exact Htt | reflexivity ] | reflexivity ].
  Qed.

  Lemma lincomb_add : ∀ t t' : T, lincomb (t + t') == lincomb t + lincomb t'.
  Proof.
    intros t t'.
    apply (nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) n
      == natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n
         + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper (t + t')) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper t) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper t') n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) 0)
        == (0 : V) { (natrec_zero V _ _ 0 _) }
        == ((0 : V) + 0) { (right_identity (action := add) (0 : V)) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) 0 + 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k))) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) 0
            + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) n
            + get (t + t') (elem n) *: basis (elem n))
             { (natrec_succ V _ _ 0 _ n) }
        == ((natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n)
            + (get t (elem n) + get t' (elem n)) *: basis (elem n))
             { (op_proper _ _ IH _ _ (scale_proper _ _ (get_add t t' (elem n)) _ _ (reflexivity _))) }
        == ((natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n)
            + (get t (elem n) *: basis (elem n) + get t' (elem n) *: basis (elem n)))
             { (right_distributive (op := scale) (get t (elem n)) (get t' (elem n)) (basis (elem n))) }
        == ((natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + get t (elem n) *: basis (elem n))
            + (natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n + get t' (elem n) *: basis (elem n)))
             { (add_add_add_comm _ _ _ _) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) (succ n)
            + (natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n + get t' (elem n) *: basis (elem n)))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) (succ n)
            + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) (succ n))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n) }.
  Qed.

  Lemma lincomb_smul : ∀ (a : F) (t : T), lincomb (a *: t) == a *: lincomb t.
  Proof.
    intros a t.
    apply (nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) n
      == a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper (a *: t)) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper t) n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) 0)
        == (0 : V) { (natrec_zero V _ _ 0 _) }
        == (a *: (0 : V)) { (smul_zero (V := V) a) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) n + get (a *: t) (elem n) *: basis (elem n))
             { (natrec_succ V _ _ 0 _ n) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + (a × get t (elem n)) *: basis (elem n))
             { (op_proper _ _ IH _ _ (scale_proper _ _ (get_smul a t (elem n)) _ _ (reflexivity _))) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + a *: (get t (elem n) *: basis (elem n)))
             { (associative (op := mult) (action := scale) a (get t (elem n)) (basis (elem n))) }
        == (a *: (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + get t (elem n) *: basis (elem n)))
             { (left_distributive (op := scale) a _ _) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) (succ n))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n) }.
  Qed.

The unit tables combine to the basis

  Lemma delta_fun_proper : ∀ j : I, Proper (equ ==> equ) (fun i : I ⇒ if equ_dec i j then (1 : F) else 0).
  Proof.
    intros j i i' Hi.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i' j) as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity (symmetry Hi) Hij).
    - destruct Hij. exact (transitivity Hi Hij').
  Qed.

  Lemma delta_proper : Proper (equ ==> equ) (fun j : I ⇒ delta j).
  Proof.
    intros j j' Hj. apply tabulate_proper. intros i.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i j') as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity Hij Hj).
    - destruct Hij. exact (transitivity Hij' (symmetry Hj)).
  Qed.

The partial sums: zero before the index of j, the basis vector from there on.
  Lemma lincomb_delta_partial : ∀ (j : I) (n : N), n ≤ card →
    (n ≤ index j → natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n == 0)
    ∧ (index j < n → natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n == basis j).
  Proof.
    intros j.
    apply (nat_induction (fun n ⇒ n ≤ card →
      (n ≤ index j → natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n == 0)
      ∧ (index j < n → natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n == basis j))).
    - intros n n' Hn.
      assert (He : natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n
                   == natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n')
        by exact (natrec_proper V _ _ 0 _ (lincomb_step_proper (delta j)) n n' Hn).
      split; intros Hall Hc.
      + assert (Hc' : n ≤ card) by (setoid_rewrite Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite <- He. apply HZ. setoid_rewrite Hn. exact Hle.
        × intros Hlt. setoid_rewrite <- He. apply HB.
          exact (proj2 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
      + assert (Hc' : n' ≤ card) by (setoid_rewrite <- Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite He. apply HZ. setoid_rewrite <- Hn. exact Hle.
        × intros Hlt. setoid_rewrite He. apply HB.
          exact (proj1 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
    - intros _. split.
      + intros _. exact (natrec_zero V _ _ 0 _).
      + intros Hlt. destruct (Theory.NatOrder.not_lt_zero (index j) Hlt).
    - intros n IH Hsn.
      assert (Hn : n ≤ card) by exact (Theory.NatOrder.le_trans n (succ n) card (Theory.NatOrder.le_succ_diag n) Hsn).
      destruct (IH Hn) as [HZ HB].
      assert (Hlt : n < card) by exact (Theory.NatOrder.lt_of_lt_of_le n (succ n) card (Theory.NatOrder.lt_succ_diag n) Hsn).
      pose proof (natrec_succ V _ _ 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n) as Hs.
      simpl in Hs.
      pose proof (@get_tabulate I F T _ _ _ getT tabT TT (fun i : I ⇒ if equ_dec i j then (1 : F) else 0) (elem n)) as Hg.
      cbv beta in Hg.
      destruct (equ_dec (elem n) j) as [Hej | Hej].
      + assert (Hnj : n == index j).
        { calc n == (index (I := I) (N := N) (elem (I := I) (N := N) n)) { (index_elem (I := I) (N := N) n Hlt) } == (index j) { (index_proper (elem n) j Hej) }. }
        split.
        × intros Hle. exfalso. apply (Theory.NatOrder.lt_irrefl n).
          apply (Theory.NatOrder.lt_of_lt_of_le n (succ n) n (Theory.NatOrder.lt_succ_diag n)).
          assert (Hback : index j ≤ n) by (setoid_rewrite Hnj; exact (Theory.NatOrder.le_refl (index j))).
          exact (Theory.NatOrder.le_trans (succ n) (index j) n Hle Hback).
        × intros _.
          assert (Hle0 : n ≤ index j) by (setoid_rewrite <- Hnj; exact (Theory.NatOrder.le_refl n)).
          calc (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) (succ n))
            == (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n + get (delta j) (elem n) *: basis (elem n)) { Hs }
            == (0 + (1 : F) *: basis (elem n))
                 { (op_proper _ _ (HZ Hle0) _ _ (scale_proper (V := V) _ _ Hg (basis (elem n)) (basis (elem n)) (reflexivity _))) }
            == ((1 : F) *: basis (elem n)) { (left_identity (action := add) ((1 : F) *: basis (elem n))) }
            == (basis (elem n)) { (left_identity (action := scale) (basis (elem n))) }
            == (basis j) { (basis_proper (elem n) j Hej) }.
      + assert (Heq : natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) (succ n)
                      == natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n).
        { calc (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) (succ n))
            == (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n + get (delta j) (elem n) *: basis (elem n)) { Hs }
            == (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n + (0 : F) *: basis (elem n))
                 { (op_proper _ _ (reflexivity _) _ _ (scale_proper (V := V) _ _ Hg (basis (elem n)) (basis (elem n)) (reflexivity _))) }
            == (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n + 0) { (zero_smul (basis (elem n))) }
            == (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n)
                 { (right_identity (action := add) (natrec V 0 (fun k acc ⇒ acc + get (delta j) (elem k) *: basis (elem k)) n)) }. }
        split.
        × intros Hle. setoid_rewrite Heq. apply HZ.
          exact (Theory.NatOrder.le_trans n (succ n) (index j) (Theory.NatOrder.le_succ_diag n) Hle).
        × intros Hlt'. setoid_rewrite Heq. apply HB.
          destruct (proj1 (Theory.NatOrder.lt_succ_iff (index j) n) Hlt') as [Hlt'' | Heq'']; [ exact Hlt'' | ].
          exfalso. apply Hej.
          calc (elem (I := I) (N := N) n)
            == (elem (I := I) (N := N) (index j)) { (elem_proper n (index j) (symmetry Heq'')) }
            == j { (elem_index j) }.
  Qed.

  Theorem lincomb_delta : ∀ j : I, lincomb (delta j) == basis j.
  Proof.
    intros j.
    exact (proj2 (lincomb_delta_partial j card (Theory.NatOrder.le_refl card)) (index_lt j)).
  Qed.

Families of tables as families of vectors

  Lemma cols_proper : ∀ M : I → T, Proper (equ ==> equ) M → Proper (equ ==> equ) (cols M).
  Proof. intros M HM j j' Hj. apply lincomb_proper. apply HM. exact Hj. Qed.

Replacing a table of the family replaces the vector of the family.
  Lemma cols_upd : ∀ (M : I → T) (j : I) (t : T) (k : I),
    cols (upd M j t) k == upd (cols M) j (lincomb t) k.
  Proof.
    intros M j t k. destruct (equ_dec k j) as [Hkj | Hkj].
    - calc (lincomb (upd M j t k)) == (lincomb t) { (lincomb_proper _ _ (upd_at M j t k Hkj)) }
        == (upd (cols M) j (lincomb t) k) { (symmetry (upd_at (cols M) j (lincomb t) k Hkj)) }.
    - calc (lincomb (upd M j t k)) == (lincomb (M k)) { (lincomb_proper _ _ (upd_off M j t k Hkj)) }
        == (upd (cols M) j (lincomb t) k) { (symmetry (upd_off (cols M) j (lincomb t) k Hkj)) }.
  Qed.

  Lemma det_cols_upd : ∀ (M : I → T) (j : I) (t : T), Proper (equ ==> equ) M →
    det (cols (upd M j t)) == det (upd (cols M) j (lincomb t)).
  Proof.
    intros M j t HM.
    exact (form_proper _ _ (cols_proper _ (upd_proper M j t HM)) (upd_proper (cols M) j _ (cols_proper M HM)) (cols_upd M j t)).
  Qed.

  Lemma det_upd_ext : ∀ (vs : I → V) (j : I) (u u' : V), Proper (equ ==> equ) vs → u == u' →
    det (upd vs j u) == det (upd vs j u').
  Proof.
    intros vs j u u' Hvs Hu.
    exact (form_proper _ _ (upd_proper vs j u Hvs) (upd_proper vs j u' Hvs) (upd_ext vs j u u' Hu)).
  Qed.

The laws of the determinant on families of tables

  Theorem cols_det_proper : ∀ M M' : I → T, Proper (equ ==> equ) M → Proper (equ ==> equ) M' →
    (∀ j : I, M j == M' j) → det (cols M) == det (cols M').
  Proof.
    intros M M' HM HM' Hmm.
    exact (form_proper _ _ (cols_proper M HM) (cols_proper M' HM') (fun j ⇒ lincomb_proper _ _ (Hmm j))).
  Qed.

  Theorem cols_det_additive : ∀ (M : I → T) (j : I), Proper (equ ==> equ) M →
    Additive T F add add (fun t ⇒ det (cols (upd M j t))).
  Proof.
    intros M j HM t t'.
    calc (det (cols (upd M j (t + t'))))
      == (det (upd (cols M) j (lincomb (t + t')))) { (det_cols_upd M j _ HM) }
      == (det (upd (cols M) j (lincomb t + lincomb t'))) { (det_upd_ext _ j _ _ (cols_proper M HM) (lincomb_add t t')) }
      == (det (upd (cols M) j (lincomb t)) + det (upd (cols M) j (lincomb t')))
           { (slot_additive (cols M) j (cols_proper M HM) (lincomb t) (lincomb t')) }
      == (det (cols (upd M j t)) + det (upd (cols M) j (lincomb t'))) { (symmetry (det_cols_upd M j t HM)) }
      == (det (cols (upd M j t)) + det (cols (upd M j t'))) { (symmetry (det_cols_upd M j t' HM)) }.
  Qed.

  Theorem cols_det_homogeneous : ∀ (M : I → T) (j : I), Proper (equ ==> equ) M →
    Homogeneous F T F mult (fun a ⇒ a) (fun t ⇒ det (cols (upd M j t))).
  Proof.
    intros M j HM a t.
    calc (det (cols (upd M j (a *: t))))
      == (det (upd (cols M) j (lincomb (a *: t)))) { (det_cols_upd M j _ HM) }
      == (det (upd (cols M) j (a *: lincomb t))) { (det_upd_ext _ j _ _ (cols_proper M HM) (lincomb_smul a t)) }
      == (a × det (upd (cols M) j (lincomb t))) { (slot_homogeneous (cols M) j (cols_proper M HM) a (lincomb t)) }
      == (a × det (cols (upd M j t))) { (symmetry (det_cols_upd M j t HM)) }.
  Qed.

  Theorem cols_det_alternating : ∀ (M : I → T) (i j : I), Proper (equ ==> equ) M →
    i =/= j → M i == M j → det (cols M) == 0.
  Proof.
    intros M i j HM Hij Hv.
    exact (alternating (cols M) i j (cols_proper M HM) Hij (lincomb_proper _ _ Hv)).
  Qed.

  Theorem cols_det_delta : det (cols (fun j : I ⇒ delta j)) == 1.
  Proof.
    calc (det (cols (fun j : I ⇒ delta j)))
      == (det basis) { (form_proper _ _ (cols_proper _ delta_proper) basis_proper lincomb_delta) }
      == (1 : F) { det_basis }.
  Qed.
