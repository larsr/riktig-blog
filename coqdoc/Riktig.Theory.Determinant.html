Section Pullback.
  Context {F V : Type} `{VV : VectorSpace F V} {I : Type}
    {eI : EqRel I} {IE : Equivalence (@equ I eI)} {Ieq : EquDec I}.
  Context {Vu : HasUpd I V} {VuL : Upd I V}.
  Context (f : LinearMap F V V) (omega : MultilinearForm I F V).

  Definition fstar_fun (vs : I → V) : F := omega (fun i ⇒ f (vs i)).

  #[export] Instance fstar_proper : Proper (@famEq V _ I ==> equ) fstar_fun.
  Proof.
    intros vs ws Hvw.
    apply (multilinear_form_proper omega).
    intros i.
    exact (linear_map_proper f (vs i) (ws i) (Hvw i)).
  Qed.

  Lemma fstar_upd_commutes : ∀ (vs : I → V) (i : I) (x : V)
      (j : I),
    f (upd vs i x j) == upd (fun k ⇒ f (vs k)) i (f x) j.
  Proof.
    intros vs i x j.
    destruct (dec (j == i)) as [Hji | Hji].
    - calc (f (upd vs i x j))
        == (f x)
             { (linear_map_proper f (upd vs i x j) x (upd_at vs i x j Hji)) }
        == (upd (fun k ⇒ f (vs k)) i (f x) j)
             { (symmetry (upd_at (fun k ⇒ f (vs k)) i (f x) j Hji)) }.
    - assert (Hij : ¬ (i == j)) by (intros Hc; exact (Hji (symmetry Hc))).
      calc (f (upd vs i x j))
        == (f (vs j))
             { (linear_map_proper f (upd vs i x j) (vs j)
                 (upd_off vs i j x Hij)) }
        == (upd (fun k ⇒ f (vs k)) i (f x) j)
             { (symmetry (upd_off (fun k ⇒ f (vs k)) i j (f x) Hij)) }.
  Qed.

  Lemma upd_val_proper : ∀ (ys : I → V) (i : I) (x y : V),
    x == y → famEq (upd ys i x) (upd ys i y).
  Proof.
    intros ys i x y Hxy j.
    destruct (dec (j == i)) as [Hji | Hji].
    - calc (upd ys i x j)
        == x { (upd_at ys i x j Hji) }
        == y { Hxy }
        == (upd ys i y j) { (symmetry (upd_at ys i y j Hji)) }.
    - assert (Hij : ¬ (i == j)) by (intros Hc; exact (Hji (symmetry Hc))).
      calc (upd ys i x j)
        == (ys j) { (upd_off ys i j x Hij) }
        == (upd ys i y j) { (symmetry (upd_off ys i j y Hij)) }.
  Qed.

  Lemma fstar_upd_eq : ∀ (vs : I → V) (i : I) (x : V),
    omega (upd (fun k ⇒ f (vs k)) i (f x)) == fstar_fun (upd vs i x).
  Proof.
    intros vs i x.
    unfold fstar_fun.
    apply (multilinear_form_proper omega).
    intros j.
    symmetry.
    exact (fstar_upd_commutes vs i x j).
  Qed.

  Lemma fstar_add : ∀ (vs : I → V) (i : I) (x y : V),
    fstar_fun (upd vs i (x + y))
      == fstar_fun (upd vs i x) + fstar_fun (upd vs i y).
  Proof.
    intros vs i x y.
    unfold fstar_fun.
    calc (omega (fun k ⇒ f (upd vs i (x + y) k)))
      == (omega (upd (fun k ⇒ f (vs k)) i (f (x + y))))
      == (omega (upd (fun k ⇒ f (vs k)) i (f x + f y)))
      == (omega (upd (fun k ⇒ f (vs k)) i (f x))
            + omega (upd (fun k ⇒ f (vs k)) i (f y)))
      == (fstar_fun (upd vs i x) + fstar_fun (upd vs i y)).
    - apply (multilinear_form_proper omega). intros j. exact (fstar_upd_commutes vs i (x + y) j).
    - apply (multilinear_form_proper omega).
      exact (upd_val_proper (fun k ⇒ f (vs k)) i (f (x + y)) (f x + f y)
               (linear_map_additive f x y)).
    - exact (multilinear_form_additive omega (fun k ⇒ f (vs k)) i (f x) (f y)).
    - now rewrite (fstar_upd_eq vs i x), (fstar_upd_eq vs i y).
  Qed.

  Lemma fstar_scale : ∀ (vs : I → V) (i : I) (a : F) (x : V),
    fstar_fun (upd vs i (a *: x)) == a × fstar_fun (upd vs i x).
  Proof.
    intros vs i a x.
    unfold fstar_fun.
    calc (omega (fun k ⇒ f (upd vs i (a *: x) k)))
      == (omega (upd (fun k ⇒ f (vs k)) i (f (a *: x))))
      == (omega (upd (fun k ⇒ f (vs k)) i (a *: f x)))
      == (a × omega (upd (fun k ⇒ f (vs k)) i (f x)))
      == (a × fstar_fun (upd vs i x)).
    - apply (multilinear_form_proper omega). intros j. exact (fstar_upd_commutes vs i (a *: x) j).
    - apply (multilinear_form_proper omega).
      exact (upd_val_proper (fun k ⇒ f (vs k)) i (f (a *: x)) (a *: f x)
               (linear_map_scale f a x)).
    - exact (multilinear_form_scale omega (fun k ⇒ f (vs k)) i a (f x)).
    - now rewrite (fstar_upd_eq vs i x).
  Qed.

  Definition fstar : MultilinearForm I F V :=
    multilinear_form_from_laws fstar_fun fstar_proper fstar_add fstar_scale.

  Theorem fstar_alternating : Alternating omega → Alternating fstar.
  Proof.
    intros Halt.
    apply alternating_from_law.
    intros vs i j Hij Heq.
    exact (alternating_repeat_zero Halt (fun k ⇒ f (vs k)) i j Hij
             (linear_map_proper f (vs i) (vs j) Heq)).
  Qed.

End Pullback.

Section CanonicalForm.
  Context {F V : Type} `{VV : VectorSpace F V} {p : nat}.
  Context {Vu : HasUpd (Fin.t (S p)) V} {VuL : Upd (Fin.t (S p)) V}.
  Context (B : Basis (S p) (Fin.t (S p)) F V).

  Definition mat_of_fam (vs : Fin.t (S p) → V) : Matrix (S p) (S p) F :=
    fun i j ⇒ Basis.coords B (vs j) i.

  Definition omega_B (vs : Fin.t (S p) → V) : F := detM (mat_of_fam vs).

  #[export] Instance omega_B_proper : Proper (@famEq V _ (Fin.t (S p)) ==> equ) omega_B.
  Proof.
    intros vs ws Hvw.
    apply detM_proper.
    intros i j.
    exact (linear_map_proper (Basis.coords B) (vs j) (ws j) (Hvw j) i).
  Qed.

  Lemma upd_F1_eq : ∀ (vs : Fin.t (S p) → V) (x : V),
    upd vs Fin.F1 x Fin.F1 == x.
  Proof.
    intros vs x.
    exact (upd_at vs Fin.F1 x Fin.F1 (reflexivity Fin.F1)).
  Qed.

  Lemma upd_F1_off : ∀ (vs : Fin.t (S p) → V) (x : V) (j' : Fin.t p),
    upd vs Fin.F1 x (Fin.FS j') == vs (Fin.FS j').
  Proof.
    intros vs x j'.
    assert (Hne : Fin.F1 ≠ Fin.FS j') by discriminate.
    exact (upd_off vs Fin.F1 (Fin.FS j') x Hne).
  Qed.

  Lemma mat_of_fam_upd_F1 : ∀ (vs : Fin.t (S p) → V) (x : V)
      (i j : Fin.t (S p)),
    mat_of_fam (upd vs Fin.F1 x) i j
      == upd_col1 (mat_of_fam vs) (fun i0 ⇒ Basis.coords B x i0) i j.
  Proof.
    intros vs x i j.
    apply (Fin.caseS' j
             (fun j0 ⇒ mat_of_fam (upd vs Fin.F1 x) i j0
               == upd_col1 (mat_of_fam vs) (fun i0 ⇒ Basis.coords B x i0) i j0)).
    - unfold mat_of_fam.
      transitivity (Basis.coords B x i).
      + exact (linear_map_proper (Basis.coords B) (upd vs Fin.F1 x Fin.F1) x
                 (upd_F1_eq vs x) i).
      + symmetry.
        exact (upd_col1_F1 (S p) p (mat_of_fam vs)
                 (fun i0 ⇒ Basis.coords B x i0) i).
    - intros j'.
      unfold mat_of_fam.
      transitivity (Basis.coords B (vs (Fin.FS j')) i).
      + exact (linear_map_proper (Basis.coords B) (upd vs Fin.F1 x (Fin.FS j'))
                 (vs (Fin.FS j')) (upd_F1_off vs x j') i).
      + symmetry.
        exact (upd_col1_FS (S p) p (mat_of_fam vs)
                 (fun i0 ⇒ Basis.coords B x i0) i j').
  Qed.

  Lemma mat_of_fam_upd_F1_sym : ∀ (vs : Fin.t (S p) → V) (x : V)
      (i j : Fin.t (S p)),
    upd_col1 (mat_of_fam vs) (fun i0 ⇒ Basis.coords B x i0) i j
      == mat_of_fam (upd vs Fin.F1 x) i j.
  Proof.
    intros vs x i j.
    symmetry.
    exact (mat_of_fam_upd_F1 vs x i j).
  Qed.

  Lemma upd_col1_ext : ∀ (A : Matrix (S p) (S p) F) (c c' : Fin.t (S p) → F),
    (∀ i, c i == c' i) →
    ∀ i j : Fin.t (S p), upd_col1 A c i j == upd_col1 A c' i j.
  Proof.
    intros A c c' Hc i j.
    apply (Fin.caseS' j (fun j0 ⇒ upd_col1 A c i j0 == upd_col1 A c' i j0)).
    - rewrite (upd_col1_F1 (S p) p A c i), (upd_col1_F1 (S p) p A c' i).
      exact (Hc i).
    - intros j'.
      rewrite (upd_col1_FS (S p) p A c i j'), (upd_col1_FS (S p) p A c' i j').
      reflexivity.
  Qed.

  Theorem omega_B_add_F1 : ∀ (vs : Fin.t (S p) → V) (x y : V),
    omega_B (upd vs Fin.F1 (x + y))
      == omega_B (upd vs Fin.F1 x) + omega_B (upd vs Fin.F1 y).
  Proof.
    intros vs x y.
    unfold omega_B.
    calc (detM (mat_of_fam (upd vs Fin.F1 (x + y))))
      == (detM (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B (x + y) i)))
      == (detM (upd_col1 (mat_of_fam vs)
            (fun i ⇒ Basis.coords B x i + Basis.coords B y i)))
      == (detM (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B x i))
            + detM (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B y i)))
      == (omega_B (upd vs Fin.F1 x) + omega_B (upd vs Fin.F1 y)).
    - exact (detM_proper (S p) (mat_of_fam (upd vs Fin.F1 (x + y)))
               (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B (x + y) i))
               (mat_of_fam_upd_F1 vs (x + y))).
    - exact (detM_proper (S p) _ _
               (upd_col1_ext (mat_of_fam vs) (fun i ⇒ Basis.coords B (x + y) i)
                  (fun i ⇒ Basis.coords B x i + Basis.coords B y i)
                  (fun i ⇒ linear_map_additive (Basis.coords B) x y i))).
    - exact (det_col1_add p (mat_of_fam vs)
               (fun i ⇒ Basis.coords B x i) (fun i ⇒ Basis.coords B y i)).
    - now rewrite (detM_proper (S p) _ _
                     (fun i j ⇒ mat_of_fam_upd_F1_sym vs x i j)),
                  (detM_proper (S p) _ _
                     (fun i j ⇒ mat_of_fam_upd_F1_sym vs y i j)).
  Qed.

  Theorem omega_B_scale_F1 : ∀ (vs : Fin.t (S p) → V) (a : F) (x : V),
    omega_B (upd vs Fin.F1 (a *: x)) == a × omega_B (upd vs Fin.F1 x).
  Proof.
    intros vs a x.
    unfold omega_B.
    calc (detM (mat_of_fam (upd vs Fin.F1 (a *: x))))
      == (detM (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B (a *: x) i)))
      == (detM (upd_col1 (mat_of_fam vs) (fun i ⇒ a × Basis.coords B x i)))
      == (a × detM (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B x i)))
      == (a × omega_B (upd vs Fin.F1 x)).
    - exact (detM_proper (S p) (mat_of_fam (upd vs Fin.F1 (a *: x)))
               (upd_col1 (mat_of_fam vs) (fun i ⇒ Basis.coords B (a *: x) i))
               (mat_of_fam_upd_F1 vs (a *: x))).
    - exact (detM_proper (S p) _ _
               (upd_col1_ext (mat_of_fam vs) (fun i ⇒ Basis.coords B (a *: x) i)
                  (fun i ⇒ a × Basis.coords B x i)
                  (fun i ⇒ linear_map_scale (Basis.coords B) a x i))).
    - exact (det_col1_scale p (mat_of_fam vs) a (fun i ⇒ Basis.coords B x i)).
    - now rewrite (detM_proper (S p) _ _
                     (fun i j ⇒ mat_of_fam_upd_F1_sym vs x i j)).
  Qed.


End CanonicalForm.

Section EndoDet.
  Context {F U : Type} `{VU : VectorSpace F U} {n : nat}.

  Definition det (B : Basis n (Fin.t n) F U) (f : LinearMap F U U) : F :=
    detM (mat_of B B f).

End EndoDet.

Section EndoDetConditional.
  Context {F U : Type} `{VU : VectorSpace F U} {n : nat}.
  Context (detM_mul : ∀ (m : nat) (A A' : Matrix m m F),
    detM (mat_mul A A') == detM A × detM A').

  Lemma change_of_basis_det_inv : ∀ B B' : Basis n (Fin.t n) F U,
    detM (change_of_basis B' B) × detM (change_of_basis B B') == 1.
  Proof.
    intros B B'.
    calc (detM (change_of_basis B' B) × detM (change_of_basis B B'))
      == (detM (mat_mul (change_of_basis B' B) (change_of_basis B B')))
      == (detM (@mat_id F _ _ n))
      == (1 : F).
    - symmetry. exact (detM_mul n (change_of_basis B' B) (change_of_basis B B')).
    - exact (detM_proper n (mat_mul (change_of_basis B' B) (change_of_basis B B')) mat_id
               (change_of_basis_inv B B')).
    - exact (det_id n).
  Qed.

  Theorem det_basis_independent : ∀ (B B' : Basis n (Fin.t n) F U) (f : LinearMap F U U),
    det B' f == det B f.
  Proof.
    intros B B' f.
    unfold det.
    calc (detM (mat_of B' B' f))
      == (detM (mat_mul (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B'))))
      == (detM (change_of_basis B' B) × detM (mat_mul (mat_of B B f) (change_of_basis B B')))
      == (detM (change_of_basis B' B) × (detM (mat_of B B f) × detM (change_of_basis B B')))
      == (detM (mat_of B B f) × (detM (change_of_basis B' B) × detM (change_of_basis B B')))
      == (detM (mat_of B B f) × 1)
      == (detM (mat_of B B f)).
    - exact (detM_proper n (mat_of B' B' f)
               (mat_mul (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B')))
               (mat_of_conj B B' f)).
    - exact (detM_mul n (change_of_basis B' B) (mat_mul (mat_of B B f) (change_of_basis B B'))).
    - now rewrite (detM_mul n (mat_of B B f) (change_of_basis B B')).
    - exact (mul_left_commutative (detM (change_of_basis B' B)) (detM (mat_of B B f))
               (detM (change_of_basis B B'))).
    - now rewrite (change_of_basis_det_inv B B').
    - exact (right_identity (detM (mat_of B B f))).
  Qed.

  Theorem det_comp : ∀ (B : Basis n (Fin.t n) F U) (f g : LinearMap F U U),
    det B (linear_map_comp f g) == det B f × det B g.
  Proof.
    intros B f g.
    unfold det.
    calc (detM (mat_of B B (linear_map_comp f g)))
      == (detM (mat_mul (mat_of B B f) (mat_of B B g)))
      == (detM (mat_of B B f) × detM (mat_of B B g)).
    - exact (detM_proper n (mat_of B B (linear_map_comp f g))
               (mat_mul (mat_of B B f) (mat_of B B g)) (mat_of_comp B B B f g)).
    - exact (detM_mul n (mat_of B B f) (mat_of B B g)).
  Qed.

End EndoDetConditional.

Section EndoDetId.
  Context {F U : Type} `{VU : VectorSpace F U} {n : nat}.

  Theorem det_id : ∀ B : Basis n (Fin.t n) F U,
    det B (linear_map_id (F:=F) (U:=U)) == 1.
  Proof.
    intros B.
    unfold det.
    calc (detM (mat_of B B (linear_map_id (F:=F) (U:=U))))
      == (detM (@mat_id F _ _ n))
      == (1 : F).
    - exact (detM_proper n (mat_of B B (linear_map_id (F:=F) (U:=U))) mat_id (mat_of_id B)).
    - exact (det_id n).
  Qed.

End EndoDetId.

Section SwapNegates.
  Context {F V : Type} `{VV : VectorSpace F V} {I : Type}
    {eI : EqRel I} {IE : Equivalence (@equ I eI)} {Ieq : EquDec I}.
  Context {Vu : HasUpd I V} {VuL : Upd I V}.
  Context (m : MultilinearForm I F V) (Halt : Alternating m).
  Context (swap : I → I → I → I) {Sw : Swap swap}.

  Definition fswap (vs : I → V) (i j : I) : I → V :=
    fun k ⇒ vs (swap i j k).


  Lemma fswap_at_i : ∀ (vs : I → V), Proper (equ ==> equ) vs →
    ∀ i j : I, fswap vs i j i == vs j.
  Proof.
    intros vs Pvs i j.
    unfold fswap.
    exact (Pvs (swap i j i) j (swap_at_i swap i j i (reflexivity i))).
  Qed.

  Lemma fswap_at_j : ∀ (vs : I → V), Proper (equ ==> equ) vs →
    ∀ i j : I, ¬ (i == j) → fswap vs i j j == vs i.
  Proof.
    intros vs Pvs i j Hij.
    unfold fswap.
    exact (Pvs (swap i j j) i (swap_at_j swap i j j Hij (reflexivity j))).
  Qed.

  Lemma upd_family_proper : ∀ (ws : I → V),
    Proper (equ ==> equ) ws →
    ∀ (i : I) (x : V), Proper (equ ==> equ) (upd ws i x).
  Proof.
    intros ws Pws i x k k' Hkk'.
    destruct (dec (k == i)) as [Hki | Hki].
    - calc (upd ws i x k)
        == x { (upd_at ws i x k Hki) }
        == (upd ws i x k') { (symmetry (upd_at ws i x k'
                               (transitivity (symmetry Hkk') Hki))) }.
    - assert (Hik : ¬ (i == k)) by (intros Hc; exact (Hki (symmetry Hc))).
      assert (Hik' : ¬ (i == k'))
        by (intros Hc; exact (Hik (transitivity Hc (symmetry Hkk')))).
      calc (upd ws i x k)
        == (ws k) { (upd_off ws i k x Hik) }
        == (ws k') { (Pws k k' Hkk') }
        == (upd ws i x k') { (symmetry (upd_off ws i k' x Hik')) }.
  Qed.

  Lemma u_idem : ∀ (ws : I → V), Proper (equ ==> equ) ws →
    ∀ (i : I) (x : V),
    x == ws i → ∀ k, upd ws i x k == ws k.
  Proof.
    intros ws Pws i x Hx k; destruct (dec (k == i)) as [Hki | Hki].
    - calc (upd ws i x k)
        == x { (upd_at ws i x k Hki) }
        == (ws i) { Hx }
        == (ws k) { (Pws i k (symmetry Hki)) }.
    - assert (Hik : ¬ (i == k)) by (intros Hc; exact (Hki (symmetry Hc))).
      exact (upd_off ws i k x Hik).
  Qed.

  Lemma upd_upd_commutes : ∀ (ws : I → V) (i j : I) (x y : V),
    ¬ (i == j) →
    ∀ k, upd (upd ws i x) j y k == upd (upd ws j y) i x k.
  Proof.
    intros ws i j x y Hij k.
    destruct (dec (k == j)) as [Hkj | Hkj];
      destruct (dec (k == i)) as [Hki | Hki].
    - destruct (Hij (transitivity (symmetry Hki) Hkj)).
    - assert (Hik : ¬ (i == k))
        by (intros Hc; exact (Hki (symmetry Hc))).
      calc (upd (upd ws i x) j y k)
        == y { (upd_at (upd ws i x) j y k Hkj) }
        == (upd ws j y k) { (symmetry (upd_at ws j y k Hkj)) }
        == (upd (upd ws j y) i x k)
             { (symmetry (upd_off (upd ws j y) i k x Hik)) }.
    - assert (Hjk : ¬ (j == k))
        by (intros Hc; exact (Hkj (symmetry Hc))).
      calc (upd (upd ws i x) j y k)
        == (upd ws i x k) { (upd_off (upd ws i x) j k y Hjk) }
        == x { (upd_at ws i x k Hki) }
        == (upd (upd ws j y) i x k)
             { (symmetry (upd_at (upd ws j y) i x k Hki)) }.
    - assert (Hjk : ¬ (j == k))
        by (intros Hc; exact (Hkj (symmetry Hc))).
      assert (Hik : ¬ (i == k))
        by (intros Hc; exact (Hki (symmetry Hc))).
      calc (upd (upd ws i x) j y k)
        == (upd ws i x k) { (upd_off (upd ws i x) j k y Hjk) }
        == (ws k) { (upd_off ws i k x Hik) }
        == (upd ws j y k) { (symmetry (upd_off ws j k y Hjk)) }
        == (upd (upd ws j y) i x k)
             { (symmetry (upd_off (upd ws j y) i k x Hik)) }.
  Qed.

  Theorem mf_swap_neg : ∀ (vs : I → V), Proper (equ ==> equ) vs →
    ∀ (i j : I),
    ¬ (i == j) → m (fswap vs i j) == - m vs.
  Proof.
    intros vs Pvs i j Hij.
    pose (a := vs i). pose (b := vs j).
    pose (VS' := upd vs i (a + b)).
    assert (PVS' : Proper (equ ==> equ) VS')
      by exact (upd_family_proper vs Pvs i (a + b)).
    assert (Hji : ¬ (j == i)) by (intro Heq; apply Hij; symmetry; exact Heq).
    assert (HW0 : m (upd VS' j a) + m (upd VS' j b) == 0).
    { assert (HWi : upd VS' j (a + b) i == a + b).
      { calc (upd VS' j (a + b) i)
          == (VS' i) { (upd_off VS' j i (a + b) Hji) }
          == (a + b) { (upd_at vs i (a + b) i (reflexivity i)) }. }
      assert (HWj : upd VS' j (a + b) j == a + b)
        by exact (upd_at VS' j (a + b) j (reflexivity j)).
      assert (HW0' : m (upd VS' j (a + b)) == 0).
      { apply (alternating_repeat_zero Halt (upd VS' j (a + b)) i j Hij).
        rewrite HWi, HWj. reflexivity. }
      rewrite <- HW0'.
      symmetry.
      exact (multilinear_form_additive m VS' j a b). }
    assert (HP2val : m (upd VS' j b) == m vs).
    { assert (HVS'j : b == VS' j)
        by (unfold VS'; rewrite (upd_off vs i j (a + b) Hij); reflexivity).
      assert (HP2 : ∀ k, upd VS' j b k == VS' k)
        by (intros k; exact (u_idem VS' PVS' j b HVS'j k)).
      assert (Hm_P2 : m (upd VS' j b) == m VS')
        by (apply (multilinear_form_proper m); exact HP2).
      assert (Hexp2 : m VS' == m (upd vs i a) + m (upd vs i b))
        by exact (multilinear_form_additive m vs i a b).
      assert (HVSia : ∀ k, upd vs i a k == vs k)
        by (intros k; exact (u_idem vs Pvs i a (reflexivity a) k)).
      assert (Hm1 : m (upd vs i a) == m vs)
        by (apply (multilinear_form_proper m); exact HVSia).
      assert (Hm2 : m (upd vs i b) == 0).
      { apply (alternating_repeat_zero Halt (upd vs i b) i j Hij).
        rewrite (upd_at vs i b i (reflexivity i)),
          (upd_off vs i j b Hij).
        reflexivity. }
      calc (m (upd VS' j b))
        == (m VS') { Hm_P2 }
        == (m (upd vs i a) + m (upd vs i b)) { Hexp2 }
        == (m vs + m (upd vs i b)) { Hm1 }
        == (m vs + 0) { Hm2 }
        == (m vs) { (right_identity (m vs)) }. }
    assert (HP1val : m (upd VS' j a) == m (fswap vs i j)).
    { assert (Hreorder : ∀ k, upd VS' j a k == upd (upd vs j a) i (a + b) k).
      { intros k. unfold VS'. exact (upd_upd_commutes vs i j (a + b) a Hij k). }
      assert (Hm_P1 : m (upd VS' j a) == m (upd (upd vs j a) i (a + b)))
        by (apply (multilinear_form_proper m); exact Hreorder).
      assert (Hexp3 : m (upd (upd vs j a) i (a + b))
                    == m (upd (upd vs j a) i a) + m (upd (upd vs j a) i b))
        by exact (multilinear_form_additive m (upd vs j a) i a b).
      assert (HT1 : m (upd (upd vs j a) i a) == 0).
      { apply (alternating_repeat_zero Halt (upd (upd vs j a) i a) i j Hij).
        calc (upd (upd vs j a) i a i)
          == a { (upd_at (upd vs j a) i a i (reflexivity i)) }
          == (upd vs j a j) { (upd_at vs j a j (reflexivity j)) }
          == (upd (upd vs j a) i a j) { (upd_off (upd vs j a) i j a Hij) }. }
      assert (HT2 : ∀ k, upd (upd vs j a) i b k == fswap vs i j k).
      { intros k.
        unfold fswap.
        destruct (dec (k == i)) as [Hki | Hki].
        - calc (upd (upd vs j a) i b k)
            == b { (upd_at (upd vs j a) i b k Hki) }
            == (vs (swap i j k))
                 { (symmetry (Pvs (swap i j k) j
                     (swap_at_i swap i j k Hki))) }.
        - assert (Hik : ¬ (i == k))
            by (intros Hc; exact (Hki (symmetry Hc))).
          destruct (dec (k == j)) as [Hkj | Hkj].
          + calc (upd (upd vs j a) i b k)
              == (upd vs j a k) { (upd_off (upd vs j a) i k b Hik) }
              == a { (upd_at vs j a k Hkj) }
              == (vs (swap i j k))
                   { (symmetry (Pvs (swap i j k) i
                       (swap_at_j swap i j k Hij Hkj))) }.
          + assert (Hjk : ¬ (j == k))
              by (intros Hc; exact (Hkj (symmetry Hc))).
            calc (upd (upd vs j a) i b k)
              == (upd vs j a k) { (upd_off (upd vs j a) i k b Hik) }
              == (vs k) { (upd_off vs j k a Hjk) }
              == (vs (swap i j k))
                   { (symmetry (Pvs (swap i j k) k
                       (swap_ne swap i j k Hki Hkj))) }. }
      assert (HT2m : m (upd (upd vs j a) i b) == m (fswap vs i j))
        by (apply (multilinear_form_proper m); exact HT2).
      calc (m (upd VS' j a))
        == (m (upd (upd vs j a) i (a + b))) { Hm_P1 }
        == (m (upd (upd vs j a) i a) + m (upd (upd vs j a) i b)) { Hexp3 }
        == (0 + m (upd (upd vs j a) i b)) { HT1 }
        == (0 + m (fswap vs i j)) { HT2m }
        == (m (fswap vs i j)) { (left_identity (m (fswap vs i j))) }. }
    assert (Hneg : m (upd VS' j a) == - m (upd VS' j b))
      by exact (neg_unique_l (m (upd VS' j b)) (m (upd VS' j a)) HW0).
    rewrite HP1val, HP2val in Hneg.
    exact Hneg.
  Qed.

End SwapNegates.

Theorem det_swap_neg : ∀ (n : nat) (I F V : Type)
    (eI : EqRel I) (IE : Equivalence (@equ I eI)) (Ieq : EquDec I)
    (IFin : FinIndex n I) `{FF : Field F}
    (eV : EqRel V) (zV : Zero V) (aV : HasAdd V) (nV : HasNeg V)
    (sV : HasScale F V) (VV : VectorSpace F V)
    (Vu : HasUpd I V) (VuL : Upd I V)
    (B : Basis n I F V) (D : Determinant n I F V B)
    (swap : I → I → I → I) {Sw : Swap swap}
    (vs : I → V), Proper (equ ==> equ) vs →
    ∀ (i j : I),
  ¬ (i == j) → D (fswap swap vs i j) == - D vs.
Proof.
  intros n I F V eI IE Ieq IFin Fe Fz Fa Fn Fo Fm Fi FF eV zV aV nV sV VV
    Vu VuL B D swap Sw vs Pvs i j Hij.
  exact (mf_swap_neg (determinant_multilinear_form D) (determinant_alternating D)
           swap vs Pvs i j Hij).
Qed.
