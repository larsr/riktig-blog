A determinant on V with respect to a basis induces a determinant on the table space over the same index, with respect to the unit tables: a family of tables is read as the family of the vectors with those coordinates. This is the determinant of a square matrix given by its columns.

Linear combinations of the basis

  Lemma lincomb_step_proper : ∀ t : T,
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : V) ⇒ acc + get t (elem k) *: basis (elem k)).
  Proof.
    intros t k k' Hk acc acc' Hacc. apply Magma.op_proper; [ exact Hacc | ].
    apply ScaleLaws.scale_proper; [ apply Table.get_proper; [ reflexivity | apply Finite.elem_proper; exact Hk ] | apply basis_proper; apply Finite.elem_proper; exact Hk ].
  Qed.

  Lemma lincomb_proper : ∀ t t' : T, t == t' → lincomb t == lincomb t'.
  Proof.
    intros t t' Htt.
    exact (natrec_ext V _ _ 0 _ _ (lincomb_step_proper t) (lincomb_step_proper t')
             (fun k acc ⇒ Magma.op_proper _ _ (reflexivity acc) _ _
                (ScaleLaws.scale_proper _ _ (Table.get_proper _ _ Htt _ _ (reflexivity (elem k))) _ _ (reflexivity (basis (elem k))))) card).
  Qed.

  Lemma lincomb_add : ∀ t t' : T, lincomb (t + t') == lincomb t + lincomb t'.
  Proof.
    intros t t'.
    apply (Peano.nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) n
      == natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n
         + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (NatRec.natrec_proper V _ _ 0 _ (lincomb_step_proper (t + t')) n n' Hn).
      setoid_rewrite (NatRec.natrec_proper V _ _ 0 _ (lincomb_step_proper t) n n' Hn).
      setoid_rewrite (NatRec.natrec_proper V _ _ 0 _ (lincomb_step_proper t') n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) 0)
        == (0 : V) { (NatRec.natrec_zero V _ _ 0 _) }
        == ((0 : V) + 0) { (right_identity (action := add) (0 : V)) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) 0 + 0)
             { (NatRec.natrec_zero V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k))) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) 0
            + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) 0)
             { (NatRec.natrec_zero V _ _ 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + get (t + t') (elem k) *: basis (elem k)) n
            + get (t + t') (elem n) *: basis (elem n))
             { (NatRec.natrec_succ V _ _ 0 _ n) }
        == ((natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n)
            + (get t (elem n) + get t' (elem n)) *: basis (elem n))
             { (Magma.op_proper _ _ IH _ _ (ScaleLaws.scale_proper _ _ (TableSpace.get_add t t' (elem n)) _ _ (reflexivity _))) }
        == ((natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n)
            + (get t (elem n) *: basis (elem n) + get t' (elem n) *: basis (elem n)))
             { (right_distributive (op := scale) (get t (elem n)) (get t' (elem n)) (basis (elem n))) }
        == ((natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + get t (elem n) *: basis (elem n))
            + (natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n + get t' (elem n) *: basis (elem n)))
             { (add_add_add_comm _ _ _ _) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) (succ n)
            + (natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n + get t' (elem n) *: basis (elem n)))
             { (NatRec.natrec_succ V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n) }
        == (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) (succ n)
            + natrec V 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) (succ n))
             { (NatRec.natrec_succ V _ _ 0 (fun k acc ⇒ acc + get t' (elem k) *: basis (elem k)) n) }.
  Qed.

  Lemma lincomb_smul : ∀ (a : F) (t : T), lincomb (a *: t) == a *: lincomb t.
  Proof.
    intros a t.
    apply (Peano.nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) n
      == a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (NatRec.natrec_proper V _ _ 0 _ (lincomb_step_proper (a *: t)) n n' Hn).
      setoid_rewrite (NatRec.natrec_proper V _ _ 0 _ (lincomb_step_proper t) n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) 0)
        == (0 : V) { (NatRec.natrec_zero V _ _ 0 _) }
        == (a *: (0 : V)) { (smul_zero (V := V) a) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) 0)
             { (NatRec.natrec_zero V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + get (a *: t) (elem k) *: basis (elem k)) n + get (a *: t) (elem n) *: basis (elem n))
             { (NatRec.natrec_succ V _ _ 0 _ n) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + (a × get t (elem n)) *: basis (elem n))
             { (Magma.op_proper _ _ IH _ _ (ScaleLaws.scale_proper _ _ (TableSpace.get_smul a t (elem n)) _ _ (reflexivity _))) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + a *: (get t (elem n) *: basis (elem n)))
             { (associative (op := mult) (action := scale) a (get t (elem n)) (basis (elem n))) }
        == (a *: (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n + get t (elem n) *: basis (elem n)))
             { (left_distributive (op := scale) a _ _) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) (succ n))
             { (NatRec.natrec_succ V _ _ 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n) }.
  Qed.

The unit tables combine to the basis

  Lemma delta_fun_proper : ∀ j : I, Proper (equ ==> equ) (fun i : I ⇒ if EquDec.equ_dec i j then (1 : F) else 0).
  Proof.
    intros j i i' Hi.
    exact (Theory.Decidable.if_dec_congr _ _ _ _ _ _ (equiv_iff Hi (reflexivity j)) (reflexivity 1) (reflexivity 0)).
  Qed.

  Lemma delta_proper : Proper (equ ==> equ) (fun j : I ⇒ delta j).
  Proof.
    intros j j' Hj. apply Table.tabulate_proper. intros i.
    exact (Theory.Decidable.if_dec_congr _ _ _ _ _ _ (equiv_iff (reflexivity i) Hj) (reflexivity 1) (reflexivity 0)).
  Qed.

The partial sums: zero before the index of j, the basis vector from there on. The unit table's entries: one at its own index, zero elsewhere.
  Lemma delta_same : ∀ i j : I, i == j → get (delta j) i == 1.
  Proof.
    intros i j Hij.
    calc (get (delta j) i)
      == (if EquDec.equ_dec i j then (1 : F) else 0) { (Table.get_tabulate _ i) }
      == (1 : F) { (Theory.Decidable.if_dec_l (i == j) 1 0 Hij) }.
  Qed.

  Lemma delta_other : ∀ i j : I, i =/= j → get (delta j) i == 0.
  Proof.
    intros i j Hij.
    calc (get (delta j) i)
      == (if EquDec.equ_dec i j then (1 : F) else 0) { (Table.get_tabulate _ i) }
      == (0 : F) { (Theory.Decidable.if_dec_r (i == j) 1 0 Hij) }.
  Qed.

The combination of the basis by the unit table at j, below n: zero up to the index of j, the basis vector j from there on.
  Local Notation "'lincomb_below' t n" := (natrec V 0 (fun k acc ⇒ acc + get t (elem k) *: basis (elem k)) n)
    (at level 10, t at level 9, n at level 9).

  Lemma lincomb_below_proper : ∀ (t : T) (n n' : N), n == n' → lincomb_below t n == lincomb_below t n'.
  Proof. intros t. exact (NatRec.natrec_proper V _ _ 0 _ (lincomb_step_proper t)). Qed.

  Lemma lincomb_delta_succ : ∀ (j : I) (n : N),
    lincomb_below (delta j) (succ n) == lincomb_below (delta j) n + get (delta j) (elem n) *: basis (elem n).
  Proof. intros j n. exact (NatRec.natrec_succ V _ _ 0 _ n). Qed.

  Lemma lincomb_delta_before : ∀ (j : I) (n : N), n ≤ index j → lincomb_below (delta j) n == 0.
  Proof.
    intros j.
    apply (Theory.NatOrder.nat_induction_le (index j) (fun n ⇒ lincomb_below (delta j) n == 0)).
    - intros n n' Hn. rewrite (lincomb_below_proper (delta j) n n' Hn). reflexivity.
    - exact (NatRec.natrec_zero V _ _ 0 _).
    - intros n Hn IH.
      calc (lincomb_below (delta j) (succ n))
        == (lincomb_below (delta j) n + get (delta j) (elem n) *: basis (elem n)) { (lincomb_delta_succ j n) }
        == ((0 : V) + get (delta j) (elem n) *: basis (elem n)) { IH }
        == ((0 : V) + (0 : F) *: basis (elem n))
             { (delta_other (elem n) j (fun Hej ⇒ proj2 Hn (Finite.index_eq_of_elem j n
                  (Theory.NatOrder.lt_trans n (index j) card Hn (Finite.index_lt j)) Hej))) }
        == ((0 : V) + 0) { (zero_smul (basis (elem n))) }
        == (0 : V) { (left_identity (action := add) (0 : V)) }.
  Qed.

  Lemma lincomb_delta_after : ∀ (j : I) (n : N), index j < n → n ≤ card →
    lincomb_below (delta j) n == basis j.
  Proof.
    intros j.
    apply (Theory.NatOrder.nat_induction_between (index j) card (fun n ⇒ lincomb_below (delta j) n == basis j)).
    - intros n n' Hn. rewrite (lincomb_below_proper (delta j) n n' Hn). reflexivity.
    - calc (lincomb_below (delta j) (succ (index j)))
        == (lincomb_below (delta j) (index j) + get (delta j) (elem (index j)) *: basis (elem (index j)))
             { (lincomb_delta_succ j (index j)) }
        == ((0 : V) + get (delta j) (elem (index j)) *: basis (elem (index j)))
             { (lincomb_delta_before j (index j) (Theory.NatOrder.le_refl (index j))) }
        == ((0 : V) + (1 : F) *: basis (elem (index j)))
             { (delta_same (elem (index j)) j (Finite.elem_index j)) }
        == ((1 : F) *: basis (elem (index j))) { (left_identity (action := add) ((1 : F) *: basis (elem (index j)))) }
        == (basis (elem (index j))) { (left_identity (action := scale) (basis (elem (index j)))) }
        == (basis j) { (basis_proper _ _ (Finite.elem_index j)) }.
    - intros n Hlt Hn IH.
      calc (lincomb_below (delta j) (succ n))
        == (lincomb_below (delta j) n + get (delta j) (elem n) *: basis (elem n)) { (lincomb_delta_succ j n) }
        == (basis j + get (delta j) (elem n) *: basis (elem n)) { IH }
        == (basis j + (0 : F) *: basis (elem n))
             { (delta_other (elem n) j (fun Hej ⇒ proj2 Hlt (symmetry (Finite.index_eq_of_elem j n Hn Hej)))) }
        == (basis j + 0) { (zero_smul (basis (elem n))) }
        == (basis j) { (right_identity (action := add) (basis j)) }.
  Qed.

  Theorem lincomb_delta : ∀ j : I, lincomb (delta j) == basis j.
  Proof.
    intros j.
    exact (lincomb_delta_after j card (Finite.index_lt j) (Theory.NatOrder.le_refl card)).
  Qed.

Families of tables as families of vectors

  Lemma cols_proper : ∀ M : I → T, Proper (equ ==> equ) M → Proper (equ ==> equ) (cols M).
  Proof. intros M HM j j' Hj. apply lincomb_proper. apply HM. exact Hj. Qed.

Replacing a table of the family replaces the vector of the family.
  Lemma cols_upd : ∀ (M : I → T) (j : I) (t : T) (k : I),
    cols (upd M j t) k == upd (cols M) j (lincomb t) k.
  Proof.
    intros M j t k. destruct (EquDec.equ_dec k j) as [Hkj | Hkj].
    - calc (lincomb (upd M j t k)) == (lincomb t) { (lincomb_proper _ _ (Upd.upd_at M j t k Hkj)) }
        == (upd (cols M) j (lincomb t) k) { (symmetry (Upd.upd_at (cols M) j (lincomb t) k Hkj)) }.
    - calc (lincomb (upd M j t k)) == (lincomb (M k)) { (lincomb_proper _ _ (Upd.upd_off M j t k Hkj)) }
        == (upd (cols M) j (lincomb t) k) { (symmetry (Upd.upd_off (cols M) j (lincomb t) k Hkj)) }.
  Qed.

  Lemma det_cols_upd : ∀ (M : I → T) (j : I) (t : T), Proper (equ ==> equ) M →
    det (cols (upd M j t)) == det (upd (cols M) j (lincomb t)).
  Proof.
    intros M j t HM.
    exact (Multilinear.form_proper _ _ (cols_proper _ (Upd.upd_proper M j t HM)) (Upd.upd_proper (cols M) j _ (cols_proper M HM)) (cols_upd M j t)).
  Qed.

  Lemma det_upd_ext : ∀ (vs : I → V) (j : I) (u u' : V), Proper (equ ==> equ) vs → u == u' →
    det (upd vs j u) == det (upd vs j u').
  Proof.
    intros vs j u u' Hvs Hu.
    exact (Multilinear.form_proper _ _ (Upd.upd_proper vs j u Hvs) (Upd.upd_proper vs j u' Hvs) (upd_ext vs j u u' Hu)).
  Qed.

The laws of the determinant on families of tables

  Theorem cols_det_proper : ∀ M M' : I → T, Proper (equ ==> equ) M → Proper (equ ==> equ) M' →
    (∀ j : I, M j == M' j) → det (cols M) == det (cols M').
  Proof.
    intros M M' HM HM' Hmm.
    exact (Multilinear.form_proper _ _ (cols_proper M HM) (cols_proper M' HM') (fun j ⇒ lincomb_proper _ _ (Hmm j))).
  Qed.

  Theorem cols_det_additive : ∀ (M : I → T) (j : I), Proper (equ ==> equ) M →
    Additive T F add add (fun t ⇒ det (cols (upd M j t))).
  Proof.
    intros M j HM t t'.
    calc (det (cols (upd M j (t + t'))))
      == (det (upd (cols M) j (lincomb (t + t')))) { (det_cols_upd M j _ HM) }
      == (det (upd (cols M) j (lincomb t + lincomb t'))) { (det_upd_ext _ j _ _ (cols_proper M HM) (lincomb_add t t')) }
      == (det (upd (cols M) j (lincomb t)) + det (upd (cols M) j (lincomb t')))
           { (Multilinear.slot_additive (cols M) j (cols_proper M HM) (lincomb t) (lincomb t')) }
      == (det (cols (upd M j t)) + det (upd (cols M) j (lincomb t'))) { (symmetry (det_cols_upd M j t HM)) }
      == (det (cols (upd M j t)) + det (cols (upd M j t'))) { (symmetry (det_cols_upd M j t' HM)) }.
  Qed.

  Theorem cols_det_homogeneous : ∀ (M : I → T) (j : I), Proper (equ ==> equ) M →
    Homogeneous F T F mult (fun a ⇒ a) (fun t ⇒ det (cols (upd M j t))).
  Proof.
    intros M j HM a t.
    calc (det (cols (upd M j (a *: t))))
      == (det (upd (cols M) j (lincomb (a *: t)))) { (det_cols_upd M j _ HM) }
      == (det (upd (cols M) j (a *: lincomb t))) { (det_upd_ext _ j _ _ (cols_proper M HM) (lincomb_smul a t)) }
      == (a × det (upd (cols M) j (lincomb t))) { (Multilinear.slot_homogeneous (cols M) j (cols_proper M HM) a (lincomb t)) }
      == (a × det (cols (upd M j t))) { (symmetry (det_cols_upd M j t HM)) }.
  Qed.

  Theorem cols_det_alternating : ∀ (M : I → T) (i j : I), Proper (equ ==> equ) M →
    i =/= j → M i == M j → det (cols M) == 0.
  Proof.
    intros M i j HM Hij Hv.
    exact (Alternating.alternating (cols M) i j (cols_proper M HM) Hij (lincomb_proper _ _ Hv)).
  Qed.

  Theorem cols_det_delta : det (cols (fun j : I ⇒ delta j)) == 1.
  Proof.
    calc (det (cols (fun j : I ⇒ delta j)))
      == (det basis) { (Multilinear.form_proper _ _ (cols_proper _ delta_proper) basis_proper lincomb_delta) }
      == (1 : F) { Determinant.det_basis }.
  Qed.
End DeterminantTheory.
