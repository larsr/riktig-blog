The determinant of a square matrix, through a determinant on families of vectors: the matrix is read as the family of its columns, each the linear combination of the basis with the column's entries. Linearity in a column, vanishing on two equal columns and the value 1 on the identity matrix all follow from the laws on families.

  Local Notation "'lincomb' u" := (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) card)

  Local Notation "'col' M j" := (natrec V 0 (fun k acc ⇒ acc + M (elem k) j *: basis (elem k)) card)

Linear combinations of the basis

  Lemma lincomb_step_proper : ∀ u : I → F, Proper (equ ==> equ) u →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : V) ⇒ acc + u (elem k) *: basis (elem k)).
  Proof.
    intros u Hu k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
    apply scale_proper; [ apply Hu; apply elem_proper; exact Hk | apply basis_proper; apply elem_proper; exact Hk ].
  Qed.

  Lemma lincomb_ext : ∀ u u' : I → F, Proper (equ ==> equ) u → Proper (equ ==> equ) u' →
    (∀ i : I, u i == u' i) → lincomb u == lincomb u'.
  Proof.
    intros u u' Hu Hu' Huu.
    apply (natrec_ext V _ _ 0 _ _ (lincomb_step_proper u Hu) (lincomb_step_proper u' Hu')).
    intros k acc. apply op_proper; [ reflexivity | ]. apply scale_proper; [ exact (Huu (elem k)) | reflexivity ].
  Qed.

  Lemma lincomb_add : ∀ u v : I → F, Proper (equ ==> equ) u → Proper (equ ==> equ) v →
    lincomb (fun i ⇒ u i + v i) == lincomb u + lincomb v.
  Proof.
    intros u v Hu Hv.
    assert (Huv : Proper (equ ==> equ) (fun i ⇒ u i + v i))
      by (intros i i' Hi; apply op_proper; [ apply Hu | apply Hv ]; exact Hi).
    apply (nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) n
      == natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n
         + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper _ Huv) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper u Hu) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper v Hv) n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) 0)
        == (0 : V) { (natrec_zero V _ _ 0 _) }
        == ((0 : V) + 0) { (right_identity (action := add) (0 : V)) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) 0 + 0) { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k))) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) 0
            + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) n
            + (u (elem n) + v (elem n)) *: basis (elem n))
             { (natrec_succ V _ _ 0 _ n) }
        == ((natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n)
            + (u (elem n) + v (elem n)) *: basis (elem n))
             { IH }
        == ((natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n)
            + (u (elem n) *: basis (elem n) + v (elem n) *: basis (elem n)))
             { (right_distributive (op := scale) (u (elem n)) (v (elem n)) (basis (elem n))) }
        == ((natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + u (elem n) *: basis (elem n))
            + (natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n + v (elem n) *: basis (elem n)))
             { (add_add_add_comm _ _ _ _) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) (succ n)
            + (natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n + v (elem n) *: basis (elem n)))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) (succ n)
            + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) (succ n))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n) }.
  Qed.

  Lemma lincomb_smul : ∀ (a : F) (u : I → F), Proper (equ ==> equ) u →
    lincomb (fun i ⇒ a × u i) == a *: lincomb u.
  Proof.
    intros a u Hu.
    assert (Hau : Proper (equ ==> equ) (fun i ⇒ a × u i))
      by (intros i i' Hi; apply op_proper; [ reflexivity | apply Hu; exact Hi ]).
    apply (nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) n
      == a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper _ Hau) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (lincomb_step_proper u Hu) n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) 0)
        == (0 : V) { (natrec_zero V _ _ 0 _) }
        == (a *: (0 : V)) { (smul_zero (V := V) a) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) n + (a × u (elem n)) *: basis (elem n))
             { (natrec_succ V _ _ 0 _ n) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + (a × u (elem n)) *: basis (elem n))
             { IH }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + a *: (u (elem n) *: basis (elem n)))
             { (associative (op := mult) (action := scale) a (u (elem n)) (basis (elem n))) }
        == (a *: (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + u (elem n) *: basis (elem n)))
             { (left_distributive (op := scale) a _ _) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) (succ n))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n) }.
  Qed.

The unit coordinate vectors combine to the basis

  Lemma delta_proper : ∀ j : I, Proper (equ ==> equ) (delta j).
  Proof.
    intros j i i' Hi.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i' j) as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity (symmetry Hi) Hij).
    - destruct Hij. exact (transitivity Hi Hij').
  Qed.

The partial sums: zero before the index of j, the basis vector from there on.
  Lemma lincomb_delta_partial : ∀ (j : I) (n : N), n ≤ card →
    (n ≤ index j → natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n == 0)
    ∧ (index j < n → natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n == basis j).
  Proof.
    intros j.
    apply (nat_induction (fun n ⇒ n ≤ card →
      (n ≤ index j → natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n == 0)
      ∧ (index j < n → natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n == basis j))).
    - intros n n' Hn.
      assert (He : natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n
                   == natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n')
        by exact (natrec_proper V _ _ 0 _ (lincomb_step_proper (delta j) (delta_proper j)) n n' Hn).
      split; intros Hall Hc.
      + assert (Hc' : n ≤ card) by (setoid_rewrite Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite <- He. apply HZ. setoid_rewrite Hn. exact Hle.
        × intros Hlt. setoid_rewrite <- He. apply HB.
          exact (proj2 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
      + assert (Hc' : n' ≤ card) by (setoid_rewrite <- Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite He. apply HZ. setoid_rewrite <- Hn. exact Hle.
        × intros Hlt. setoid_rewrite He. apply HB.
          exact (proj1 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
    - intros _. split.
      + intros _. exact (natrec_zero V _ _ 0 _).
      + intros Hlt. destruct (Theory.NatOrder.not_lt_zero (index j) Hlt).
    - intros n IH Hsn.
      assert (Hn : n ≤ card) by exact (Theory.NatOrder.le_trans n (succ n) card (Theory.NatOrder.le_succ_diag n) Hsn).
      destruct (IH Hn) as [HZ HB].
      assert (Hlt : n < card) by exact (Theory.NatOrder.lt_of_lt_of_le n (succ n) card (Theory.NatOrder.lt_succ_diag n) Hsn).
      pose proof (natrec_succ V _ _ 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n) as Hs.
      simpl in Hs.
      destruct (equ_dec (elem n) j) as [Hej | Hej].
      + assert (Hnj : n == index j).
        { calc n == (index (I := I) (N := N) (elem (I := I) (N := N) n)) { (index_elem (I := I) (N := N) n Hlt) } == (index j) { (index_proper (elem n) j Hej) }. }
        split.
        × intros Hle. exfalso. apply (Theory.NatOrder.lt_irrefl n).
          apply (Theory.NatOrder.lt_of_lt_of_le n (succ n) n (Theory.NatOrder.lt_succ_diag n)).
          assert (Hback : index j ≤ n) by (setoid_rewrite Hnj; exact (Theory.NatOrder.le_refl (index j))).
          exact (Theory.NatOrder.le_trans (succ n) (index j) n Hle Hback).
        × intros _.
          assert (Hle0 : n ≤ index j) by (setoid_rewrite <- Hnj; exact (Theory.NatOrder.le_refl n)).
          calc (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) (succ n))
            == (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n + (1 : F) *: basis (elem n)) { Hs }
            == (0 + (1 : F) *: basis (elem n)) { (HZ Hle0) }
            == ((1 : F) *: basis (elem n)) { (left_identity (action := add) ((1 : F) *: basis (elem n))) }
            == (basis (elem n)) { (left_identity (action := scale) (basis (elem n))) }
            == (basis j) { (basis_proper (elem n) j Hej) }.
      + assert (Heq : natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) (succ n)
                      == natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n).
        { calc (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) (succ n))
            == (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n + (0 : F) *: basis (elem n)) { Hs }
            == (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n + 0) { (zero_smul (basis (elem n))) }
            == (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n)
                 { (right_identity (action := add) (natrec V 0 (fun k acc ⇒ acc + delta j (elem k) *: basis (elem k)) n)) }. }
        split.
        × intros Hle. setoid_rewrite Heq. apply HZ.
          exact (Theory.NatOrder.le_trans n (succ n) (index j) (Theory.NatOrder.le_succ_diag n) Hle).
        × intros Hlt'. setoid_rewrite Heq. apply HB.
          destruct (proj1 (Theory.NatOrder.lt_succ_iff (index j) n) Hlt') as [Hlt'' | Heq'']; [ exact Hlt'' | ].
          exfalso. apply Hej.
          calc (elem (I := I) (N := N) n)
            == (elem (I := I) (N := N) (index j)) { (elem_proper n (index j) (symmetry Heq'')) }
            == j { (elem_index j) }.
  Qed.

  Theorem lincomb_delta : ∀ j : I, lincomb (delta j) == basis j.
  Proof.
    intros j.
    exact (proj2 (lincomb_delta_partial j card (Theory.NatOrder.le_refl card)) (index_lt j)).
  Qed.

The columns of a matrix

  Lemma col_fun_proper : ∀ (M : I → I → F) (j : I), proper_matrix M → Proper (equ ==> equ) (fun i ⇒ M i j).
  Proof. intros M j HM i i' Hi. apply HM; [ exact Hi | reflexivity ]. Qed.

  Lemma cols_proper : ∀ M : I → I → F, proper_matrix M → Proper (equ ==> equ) (cols M).
  Proof.
    intros M HM j j' Hj.
    apply (natrec_ext V _ _ 0 _ _ (lincomb_step_proper _ (col_fun_proper M j HM)) (lincomb_step_proper _ (col_fun_proper M j' HM))).
    intros k acc. apply op_proper; [ reflexivity | ]. apply scale_proper; [ apply HM; [ reflexivity | exact Hj ] | reflexivity ].
  Qed.

  Lemma col_ext : ∀ (M M' : I → I → F) (j j' : I), proper_matrix M → proper_matrix M' →
    (∀ i : I, M i j == M' i j') → col M j == col M' j'.
  Proof.
    intros M M' j j' HM HM' Hc.
    apply (natrec_ext V _ _ 0 _ _ (lincomb_step_proper _ (col_fun_proper M j HM)) (lincomb_step_proper _ (col_fun_proper M' j' HM'))).
    intros k acc. apply op_proper; [ reflexivity | ]. apply scale_proper; [ exact (Hc (elem k)) | reflexivity ].
  Qed.

  Lemma updm_proper : ∀ (M : I → I → F) (j : I) (u : I → F),
    proper_matrix M → Proper (equ ==> equ) u → proper_matrix (updm M j u).
  Proof.
    intros M j u HM Hu i i' Hi k k' Hk.
    destruct (equ_dec k j) as [Hkj | Hkj]; destruct (equ_dec k' j) as [Hkj' | Hkj'].
    - apply Hu. exact Hi.
    - destruct Hkj'. exact (transitivity (symmetry Hk) Hkj).
    - destruct Hkj. exact (transitivity Hk Hkj').
    - apply HM; assumption.
  Qed.

Replacing a column of the matrix replaces the slot of the family.
  Lemma cols_updm : ∀ (M : I → I → F) (j : I) (u : I → F) (k : I),
    col (updm M j u) k == upd (cols M) j (lincomb u) k.
  Proof. intros M j u k. simpl. destruct (equ_dec k j); reflexivity. Qed.

  Lemma det_cols_updm : ∀ (M : I → I → F) (j : I) (u : I → F), proper_matrix M → Proper (equ ==> equ) u →
    det (cols (updm M j u)) == det (upd (cols M) j (lincomb u)).
  Proof.
    intros M j u HM Hu.
    exact (det_proper _ _ (cols_proper _ (updm_proper M j u HM Hu)) (upd_proper _ j _ (cols_proper M HM)) (cols_updm M j u)).
  Qed.

  Lemma det_upd_ext : ∀ (vs : I → V) (j : I) (u u' : V), Proper (equ ==> equ) vs → u == u' →
    det (upd vs j u) == det (upd vs j u').
  Proof.
    intros vs j u u' Hvs Hu.
    exact (det_proper _ _ (upd_proper vs j u Hvs) (upd_proper vs j u' Hvs) (upd_ext vs j u u' Hu)).
  Qed.

The laws of the matrix determinant

  Theorem matdet_proper : ∀ M M' : I → I → F, proper_matrix M → proper_matrix M' →
    (∀ i j : I, M i j == M' i j) → det (cols M) == det (cols M').
  Proof.
    intros M M' HM HM' Hmm.
    exact (det_proper _ _ (cols_proper M HM) (cols_proper M' HM') (fun j ⇒ col_ext M M' j j HM HM' (fun i ⇒ Hmm i j))).
  Qed.

  Theorem matdet_col_add : ∀ (M : I → I → F) (j : I) (u v : I → F),
    proper_matrix M → Proper (equ ==> equ) u → Proper (equ ==> equ) v →
    det (cols (updm M j (fun i ⇒ u i + v i))) == det (cols (updm M j u)) + det (cols (updm M j v)).
  Proof.
    intros M j u v HM Hu Hv.
    assert (Huv : Proper (equ ==> equ) (fun i ⇒ u i + v i))
      by (intros i i' Hi; apply op_proper; [ apply Hu | apply Hv ]; exact Hi).
    calc (det (cols (updm M j (fun i ⇒ u i + v i))))
      == (det (upd (cols M) j (lincomb (fun i ⇒ u i + v i)))) { (det_cols_updm M j _ HM Huv) }
      == (det (upd (cols M) j (lincomb u + lincomb v))) { (det_upd_ext _ j _ _ (cols_proper M HM) (lincomb_add u v Hu Hv)) }
      == (det (upd (cols M) j (lincomb u)) + det (upd (cols M) j (lincomb v))) { (det_slot_add (cols M) j _ _ (cols_proper M HM)) }
      == (det (cols (updm M j u)) + det (upd (cols M) j (lincomb v))) { (symmetry (det_cols_updm M j u HM Hu)) }
      == (det (cols (updm M j u)) + det (cols (updm M j v))) { (symmetry (det_cols_updm M j v HM Hv)) }.
  Qed.

  Theorem matdet_col_smul : ∀ (M : I → I → F) (j : I) (a : F) (u : I → F),
    proper_matrix M → Proper (equ ==> equ) u →
    det (cols (updm M j (fun i ⇒ a × u i))) == a × det (cols (updm M j u)).
  Proof.
    intros M j a u HM Hu.
    assert (Hau : Proper (equ ==> equ) (fun i ⇒ a × u i))
      by (intros i i' Hi; apply op_proper; [ reflexivity | apply Hu; exact Hi ]).
    calc (det (cols (updm M j (fun i ⇒ a × u i))))
      == (det (upd (cols M) j (lincomb (fun i ⇒ a × u i)))) { (det_cols_updm M j _ HM Hau) }
      == (det (upd (cols M) j (a *: lincomb u))) { (det_upd_ext _ j _ _ (cols_proper M HM) (lincomb_smul a u Hu)) }
      == (a × det (upd (cols M) j (lincomb u))) { (det_slot_smul (cols M) j a _ (cols_proper M HM)) }
      == (a × det (cols (updm M j u))) { (symmetry (det_cols_updm M j u HM Hu)) }.
  Qed.

  Theorem matdet_alternating : ∀ (M : I → I → F) (j j' : I), proper_matrix M →
    j =/= j' → (∀ i : I, M i j == M i j') → det (cols M) == 0.
  Proof.
    intros M j j' HM Hjj' Hc.
    exact (det_alternating (cols M) j j' (cols_proper M HM) Hjj' (col_ext M M j j' HM HM Hc)).
  Qed.

  Lemma id_proper : proper_matrix (fun i j : I ⇒ if equ_dec i j then (1 : F) else 0).
  Proof.
    intros i i' Hi j j' Hj.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i' j') as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity (symmetry Hi) (transitivity Hij Hj)).
    - destruct Hij. exact (transitivity Hi (transitivity Hij' (symmetry Hj))).
  Qed.

  Theorem matdet_id : det (cols (fun i j : I ⇒ if equ_dec i j then (1 : F) else 0)) == 1.
  Proof.
    calc (det (cols (fun i j : I ⇒ if equ_dec i j then (1 : F) else 0)))
      == (det basis) { (det_proper _ _ (cols_proper _ id_proper) basis_proper lincomb_delta) }
      == (1 : F) { det_basis }.
  Qed.
