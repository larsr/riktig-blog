Library Riktig.Structure.LambdaMorphism

From Riktig Require Export Structure.Lambda.

A map between two models of the untyped lambda calculus that preserves the operations: the variables, the binder, application, substitution and its four substitution-carrier operations, and the value predicate.

Module LambdaMorphism.
  Class LambdaMorphism (T Sb U Sc : Type)
      {eqT : EqRel T} {varT : HasVar T} {lamT : HasLam T} {appT : HasApp T}
      {subT : HasSub T Sb} {idT : HasSubId Sb} {shiftT : HasSubShift Sb}
      {consT : HasSubCons T Sb} {upT : HasSubUp Sb} {valT : HasVal T}
      {modelT : LambdaModel T Sb}
      {eqU : EqRel U} {varU : HasVar U} {lamU : HasLam U} {appU : HasApp U}
      {subU : HasSub U Sc} {idU : HasSubId Sc} {shiftU : HasSubShift Sc}
      {consU : HasSubCons U Sc} {upU : HasSubUp Sc} {valU : HasVal U}
      {modelU : LambdaModel U Sc}
      (f : T → U) (g : Sb → Sc) := {
    morphism_proper :: Proper (equ ==> equ) f;
    morphism_var : ∀ n : nat, f (var n) == var n;
    morphism_lam : ∀ b : T, f (lam b) == lam (f b);
    morphism_app : ∀ a b : T, f (app a b) == app (f a) (f b);
    morphism_substitute : ∀ (t : T) (s : Sb), f (t.[s]) == (f t).[g s];
    morphism_sub_id : g sub_id = sub_id;
    morphism_sub_shift : g sub_shift = sub_shift;
    morphism_sub_cons : ∀ (u : T) (s : Sb), g (u .: s) = f u .: g s;
    morphism_sub_up : ∀ s : Sb, g (sub_up s) = sub_up (g s);
    
    morphism_val : ∀ t : T, val t → val (f t);
  }.
  Hint Mode LambdaMorphism ! - ! - - - - - - - - - - - - - - - - - - - - - - - ! -
    : typeclass_instances.
End LambdaMorphism.
