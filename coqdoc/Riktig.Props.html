Section ActionLaws.
  Context (A B : Type) {eqB : EqRel B} (action : A → B → B) (op : BinOp A).
  Local Infix "·" := action (at level 40, left associativity).
  Local Infix "⋆" := op (at level 40, left associativity).

  Class Associative := associative : ∀ (a b : A) (x : B), (a ⋆ b) · x == a · (b · x).
End ActionLaws.

Section GuardedInverseLaws.
  Context (T : Type) {eqT : EqRel T} (defined_on : Pred T)
          (op : BinOp T) (inverse : T → T) (unit : T).
  Local Infix "⋆" := op (at level 40, left associativity).

  Class LeftInverseOn :=
    left_inverse_on : ∀ a, defined_on a → inverse a ⋆ a == unit.
  Class RightInverseOn :=
    right_inverse_on : ∀ a, defined_on a → a ⋆ inverse a == unit.
End GuardedInverseLaws.

Instance left_inverse_defined_everywhere T {eqT : EqRel T}
    (defined_on : Pred T) (op : BinOp T) (inverse : T → T) (unit : T)
    `{!LeftInverse T op inverse unit}
  : LeftInverseOn T defined_on op inverse unit :=
  fun a _ ⇒ @left_inverse T eqT op inverse unit _ a.

Instance right_inverse_defined_everywhere T {eqT : EqRel T}
    (defined_on : Pred T) (op : BinOp T) (inverse : T → T) (unit : T)
    `{!RightInverse T op inverse unit}
  : RightInverseOn T defined_on op inverse unit :=
  fun a _ ⇒ @right_inverse T eqT op inverse unit _ a.

Definition nonzero (T : Type) {eqT : EqRel T} {zeroT : Zero T} : Pred T :=
  fun a ⇒ ¬ (a == 0).

Definition invertible (T : Type) {eqT : EqRel T} (op : BinOp T) (unit : T) : Pred T :=
  fun a ⇒ ∃ b : T, op a b == unit ∧ op b a == unit.

Section DistributiveLaws.
  Context (A B C : Type) (op : A → B → C).
  Local Infix "·" := op (at level 40, left associativity).

  Class LeftDistributive {eqB : EqRel B} {eqC : EqRel C}
        {addB : HasAdd B} {addC : HasAdd C} :=
    left_distributive : ∀ (a : A) (x y : B), a · (x + y) == a · x + a · y.

  Class RightDistributive {eqA : EqRel A} {eqC : EqRel C}
        {addA : HasAdd A} {addC : HasAdd C} :=
    right_distributive : ∀ (x y : A) (b : B), (x + y) · b == x · b + y · b.
End DistributiveLaws.

Section HomomorphismLaws.
  Context (A B : Type) {eqB : EqRel B}.

  Section CarryingAnOperation.
    Context (opA : BinOp A) (opB : BinOp B) (f : A → B).
    Local Infix "⋆" := opA (at level 40, left associativity).
    Local Infix "·" := opB (at level 40, left associativity).
    Class Additive := additive : ∀ x y : A, f (x ⋆ y) == f x · f y.
  End CarryingAnOperation.
End HomomorphismLaws.

Section HomogeneityLaw.
  Context (F A B : Type) {eqB : EqRel B} {scaleA : HasScale F A}
          (action : F → B → B) (weight : F → F) (f : A → B).
  Local Infix "·" := action (at level 40, left associativity).

  Class Homogeneous := homogeneous : ∀ (a : F) (x : A), f (a *: x) == weight a · f x.
End HomogeneityLaw.

Class Unique T {eqT : EqRel T} (P : Pred T) :=
  unique : ∀ a b, P a → P b → a == b.

Class Injective A B {eqA : EqRel A} {eqB : EqRel B} (f : A → B) :=
  injective : ∀ a b, f a == f b → a == b.
