Library Riktig.Lib.StringFacts

From Coq Require Import String.


Local Open Scope string_scope.

Lemma string_append_assoc : ∀ s1 s2 s3 : string,
  (s1 ++ s2) ++ s3 = s1 ++ (s2 ++ s3).
Proof.
  induction s1 as [| c s1 IH]; intros s2 s3; simpl.
  - reflexivity.
  - rewrite IH. reflexivity.
Qed.
