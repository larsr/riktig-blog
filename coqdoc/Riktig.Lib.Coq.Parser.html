Parsers with a list of successes: a parser reads a list of C and returns every way of reading something from its front, each with what is left. The combinators, the deterministic ones that commit to the first reading, and what it means for a parser to read a result.
Definition parser (C A : Type) : Type := list C → list (A × list C).

Section Combinators.
  Context {C : Type}.

  Definition p_ret {A : Type} (a : A) : parser C A := fun s ⇒ [(a, s)].
  Definition p_bind {A B : Type} (p : parser C A) (f : A → parser C B) : parser C B :=
    fun s ⇒ flat_map (fun r ⇒ f (fst r) (snd r)) (p s).
  Definition p_fail {A : Type} : parser C A := fun _ ⇒ [].
  Definition p_alt {A : Type} (p q : parser C A) : parser C A := fun s ⇒ p s ++ q s.
  Definition p_item : parser C C := fun s ⇒ match s with [] ⇒ [] | c :: s' ⇒ [(c, s')] end.

  Definition p_parses {A : Type} (p : parser C A) (s : list C) (a : A) (s' : list C) : Prop :=
    In (a, s') (p s).

Committing to the first reading, ordered choice, and greedy repetition -- bounded by the length of the input, as every repetition reads at least one element or stops.
  Definition p_first {A : Type} (p : parser C A) : parser C A := fun s ⇒ firstn 1 (p s).
  Definition p_or {A : Type} (p q : parser C A) : parser C A :=
    fun s ⇒ match p s with [] ⇒ q s | r ⇒ r end.
  Fixpoint p_many_n {A : Type} (n : nat) (p : parser C A) : parser C (list A) :=
    match n with
    | 0 ⇒ p_ret []
    | S n ⇒ fun s ⇒ match p s with
                      | [] ⇒ [([], s)]
                      | (a, s') :: _ ⇒ map (fun r ⇒ (a :: fst r, snd r)) (p_many_n n p s')
                      end
    end.
  Definition p_many {A : Type} (p : parser C A) : parser C (list A) := fun s ⇒ p_many_n (length s) p s.
  Definition p_many1 {A : Type} (p : parser C A) : parser C (list A) :=
    p_bind p (fun a ⇒ p_bind (p_many p) (fun l ⇒ p_ret (a :: l))).
  Definition p_sat (f : C → bool) : parser C C :=
    p_bind p_item (fun c ⇒ if f c then p_ret c else p_fail).

The meaning of each combinator

  Lemma p_parses_ret : ∀ (A : Type) (a : A) (s : list C) (a' : A) (s' : list C),
    p_parses (p_ret a) s a' s' ↔ a' = a ∧ s' = s.
  Proof.
    intros A a s a' s'. unfold p_parses, p_ret. simpl. split.
    - intros [H | []]. injection H as Ha Hs. subst. split; reflexivity.
    - intros [Ha Hs]. subst. left. reflexivity.
  Qed.

  Lemma p_parses_bind : ∀ (A B : Type) (p : parser C A) (f : A → parser C B) (s : list C) (b : B) (s'' : list C),
    p_parses (p_bind p f) s b s'' ↔ ∃ (a : A) (s' : list C), p_parses p s a s' ∧ p_parses (f a) s' b s''.
  Proof.
    intros A B p f s b s''. unfold p_parses, p_bind. rewrite in_flat_map. split.
    - intros [[a0 s0] [Hp Hf]]. ∃ a0, s0. split; assumption.
    - intros [a [s' [Hp Hf]]]. ∃ (a, s'). split; assumption.
  Qed.

  Lemma p_parses_fail : ∀ (A : Type) (s : list C) (a : A) (s' : list C), ¬ p_parses p_fail s a s'.
  Proof. intros A s a s' H. destruct H. Qed.

  Lemma p_parses_alt : ∀ (A : Type) (p q : parser C A) (s : list C) (a : A) (s' : list C),
    p_parses (p_alt p q) s a s' ↔ p_parses p s a s' ∨ p_parses q s a s'.
  Proof. intros A p q s a s'. unfold p_parses, p_alt. apply in_app_iff. Qed.

  Lemma p_parses_item : ∀ (s : list C) (c : C) (s' : list C), p_parses p_item s c s' ↔ s = c :: s'.
  Proof.
    intros s c s'. unfold p_parses, p_item. destruct s as [| c0 s0]; simpl.
    - split; [ intros [] | discriminate ].
    - split.
      + intros [H | []]. injection H as Hc Hs. subst. reflexivity.
      + intros H. injection H as Hc Hs. subst. left. reflexivity.
  Qed.
End Combinators.
