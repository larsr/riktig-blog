Parsers with a list of successes over a stream S: a parser reads a stream and returns every way of reading something from its front, each with what is left. The stream is read through next, which yields an element and the rest, or nothing. The combinators, the deterministic ones that commit to the first reading, and what it means for a parser to read a result.
Definition parser (St A : Type) : Type := St → list (A × St).

Class HasNext (C St : Type) := next : St → option (C × St).

Section Combinators.
  Context {St : Type}.

  Definition p_ret {A : Type} (a : A) : parser St A := fun s ⇒ [(a, s)].
  Definition p_bind {A B : Type} (p : parser St A) (f : A → parser St B) : parser St B :=
    fun s ⇒ flat_map (fun r ⇒ f (fst r) (snd r)) (p s).
  Definition p_fail {A : Type} : parser St A := fun _ ⇒ [].
  Definition p_alt {A : Type} (p q : parser St A) : parser St A := fun s ⇒ p s ++ q s.
  Definition p_item {C : Type} {NX : HasNext C St} : parser St C :=
    fun s ⇒ match next s with Some (c, s') ⇒ [(c, s')] | None ⇒ [] end.

p reads a from s leaving s' when one of its readings is a up to the relation R on results, with exactly s' left.
  Definition p_parses {A : Type} (R : A → A → Prop) (p : parser St A) (s : St) (a : A) (s' : St) : Prop :=
    ∃ a0 : A, In (a0, s') (p s) ∧ R a0 a.

Committing to the first reading, ordered choice, and the step of a greedy repetition -- the loop over it lives in Programs, with the fuel abstract.
  Definition p_first {A : Type} (p : parser St A) : parser St A := fun s ⇒ firstn 1 (p s).
  Definition p_or {A : Type} (p q : parser St A) : parser St A :=
    fun s ⇒ match p s with [] ⇒ q s | r ⇒ r end.
  Definition p_many_step {A : Type} (p : parser St A) (rec : parser St (list A)) : parser St (list A) :=
    fun s ⇒ match p s with
             | nil ⇒ (nil, s) :: nil
             | (a, s') :: _ ⇒ map (fun r ⇒ (a :: fst r, snd r)) (rec s')
             end.

  Definition p_sat {C : Type} {NX : HasNext C St} (f : C → bool) : parser St C :=
    p_bind p_item (fun c ⇒ if f c then p_ret c else p_fail).

The meaning of each combinator

  Section Meaning.
    Context {A : Type} (R : A → A → Prop) {HR : Equivalence R}.

    Lemma p_parses_ret : ∀ (a : A) (s : St) (a' : A) (s' : St),
      p_parses R (p_ret a) s a' s' ↔ R a' a ∧ s' = s.
    Proof.
      intros a s a' s'. unfold p_parses, p_ret. simpl. split.
      - intros [a0 [[H | []] Ha]]. injection H as Ha0 Hs. subst. split; [ symmetry; exact Ha | reflexivity ].
      - intros [Ha Hs]. subst. ∃ a. split; [ left; reflexivity | symmetry; exact Ha ].
    Qed.

    Lemma p_parses_bind : ∀ {B : Type} (RB : B → B → Prop) (p : parser St A) (f : A → parser St B),
      (∀ a a' : A, R a a' → ∀ (s : St) (b : B) (s' : St), p_parses RB (f a) s b s' ↔ p_parses RB (f a') s b s') →
      ∀ (s : St) (b : B) (s'' : St),
      p_parses RB (p_bind p f) s b s'' ↔ ∃ (a : A) (s' : St), p_parses R p s a s' ∧ p_parses RB (f a) s' b s''.
    Proof.
      intros B RB p f Hf s b s''. unfold p_parses at 1, p_bind. split.
      - intros [b0 [Hin Hb]]. rewrite in_flat_map in Hin. destruct Hin as [[a0 s0] [Hp Hfa]]. ∃ a0, s0. split.
        + ∃ a0. split; [ exact Hp | reflexivity ].
        + ∃ b0. split; assumption.
      - intros [a [s' [[a0 [Hp Ha]] Hfa]]]. apply (Hf a0 a Ha) in Hfa. destruct Hfa as [b0 [Hin Hb]].
        ∃ b0. split; [ | exact Hb ]. rewrite in_flat_map. ∃ (a0, s'). split; assumption.
    Qed.

    Lemma p_parses_fail : ∀ (s : St) (a : A) (s' : St), ¬ p_parses R p_fail s a s'.
    Proof. intros s a s' [a0 [H _]]. destruct H. Qed.

    Lemma p_parses_alt : ∀ (p q : parser St A) (s : St) (a : A) (s' : St),
      p_parses R (p_alt p q) s a s' ↔ p_parses R p s a s' ∨ p_parses R q s a s'.
    Proof.
      intros p q s a s'. unfold p_parses, p_alt. split.
      - intros [a0 [Hin Ha]]. apply in_app_iff in Hin. destruct Hin as [Hin | Hin]; [ left | right ]; ∃ a0; split; assumption.
      - intros [[a0 [Hin Ha]] | [a0 [Hin Ha]]]; ∃ a0; (split; [ apply in_app_iff; first [ left; exact Hin | right; exact Hin ] | exact Ha ]).
    Qed.
  End Meaning.

  Lemma p_parses_item : ∀ {C : Type} {NX : HasNext C St} (s : St) (c : C) (s' : St),
    p_parses eq p_item s c s' ↔ next s = Some (c, s').
  Proof.
    intros C NX s c s'. unfold p_parses, p_item. destruct (next s) as [[c0 s0] |]; simpl.
    - split.
      + intros [a0 [[H | []] Ha]]. injection H as Hc Hs. subst. reflexivity.
      + intros H. injection H as Hc Hs. subst. ∃ c. split; [ left; reflexivity | reflexivity ].
    - split; [ intros [a0 [[] _]] | discriminate ].
  Qed.
End Combinators.
