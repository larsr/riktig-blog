Library Riktig.Lib.Coq.Vector

From Coq Require Import Arith.
From Coq.Vectors Require Import Fin Vector.

Tabulating a function on Fin.t n as a vector of length n, and that reading the result back at an index gives the function's value there.
Fixpoint vec_tabulate {A : Type} (n : nat) : (Fin.t n → A) → Vector.t A n :=
  match n with
  | 0 ⇒ fun _ ⇒ Vector.nil A
  | S m ⇒ fun f ⇒ Vector.cons A (f Fin.F1) m (vec_tabulate m (fun i ⇒ f (Fin.FS i)))
  end.

Lemma nth_vec_tabulate : ∀ (A : Type) (n : nat) (f : Fin.t n → A) (i : Fin.t n),
  Vector.nth (vec_tabulate n f) i = f i.
Proof.
  intros A n f i. revert f.
  induction i as [n | n i IH]; intros f; simpl; [ reflexivity | apply IH ].
Qed.
