From Coq Require Import Ascii String.
From Riktig.Structure Require Export Streaming.
From Riktig.Impl Require Export EquDecAscii.
From Riktig.Lib Require Export Coq.String.

Coq strings as streaming sequences of characters: a string yields its head and leaves its tail, and how much is left is its length. Everything is compared by Coq's equality.
Instance String_HasStep : HasStep ascii string := fun s c s' ⇒ s = String c s'.
Instance String_HasNext : HasNext ascii string :=
  fun s ⇒ match s with EmptyString ⇒ None | String c s' ⇒ Some (c, s') end.
Instance String_HasSize : HasSize string nat := String.length.

Instance String_Sequence : Sequence ascii string String_HasStep (eqC := eq).
Proof.
  split.
  - exact eq_equivalence.
  - exact eq_equivalence.
  - intros s s0 Hs c c0 Hc t t0 Ht. unfold equ, String_EqRel in ×. subst. reflexivity.
  - intros s c c' s' s'' H1 H2. unfold step, String_HasStep in ×. subst.
    injection H2 as Hc Hs. split; assumption.
Qed.

Instance String_Streaming : Streaming ascii string String_HasNext.
Proof.
  split.
  intros s c s'. unfold next, String_HasNext, step, String_HasStep.
  destruct s as [| c0 s0].
  - split; [ discriminate | intros [c1 [H _]]; discriminate H ].
  - split.
    + intros H. injection H as Hc Hs. subst. ∃ c. split; reflexivity.
    + intros [c1 [H Hc]]. injection H as Hc1 Hs. change (c1 = c) in Hc. subst. reflexivity.
Qed.
