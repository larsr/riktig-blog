Library Riktig.Interface.Group

From Coq Require Export Morphisms.
From Riktig Require Export Ops Props Interface.Monoid.

Module Group.
  Class Group (G : Type) {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G} := {
    equivalence : Equivalence (@equ G _);
    associative : ∀ x y w : G, (x + y) + w == x + (y + w);
    left_identity : ∀ x : G, 0 + x == x;
    right_identity : ∀ x : G, x + 0 == x;
    left_inverse : ∀ x : G, (- x) + x == 0;
    right_inverse : ∀ x : G, x + (- x) == 0;
    add_proper : ∀ x y : G, x == y → ∀ u v : G, u == v → (x + u) == (y + v);
    neg_proper : ∀ x y : G, x == y → (- x) == (- y)
  }.
End Group.
Notation Group := Group.Group.

#[global] Instance Group_is_Monoid G {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G}
  `{H : Group.Group G} : Monoid G := {|
    Monoid.equivalence := Group.equivalence;
    Monoid.associative := Group.associative;
    Monoid.left_identity := Group.left_identity;
    Monoid.right_identity := Group.right_identity;
    Monoid.add_proper := Group.add_proper
  |}.

#[global] Instance Group_neg_Proper G {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G}
  `{H : Group.Group G} : Proper (equ ==> equ) (@Ops.neg G _)
  := Group.neg_proper.

#[global] Instance Group_LeftInverse G {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G}
  `{H : Group.Group G} : Invl G add neg zero := Group.left_inverse.
#[global] Instance Group_RightInverse G {Ge : EqRel G} {Gz : Zero G} {Ga : HasAdd G} {Gn : HasNeg G}
  `{H : Group.Group G} : Invr G add neg zero := Group.right_inverse.
