Library Riktig.Interface.Group

From Riktig.Interface Require Export Monoid.

Module Group.
  Class Group (G : Type) {eq : EqRel G} {zero : Zero G} {add : HasAdd G}
              {neg : HasNeg G} := {
    monoid :: Monoid G;
    neg_proper :: Proper (equ ==> equ) neg;
    is_left_inverse :: LeftInverse G add neg 0;
    is_right_inverse :: RightInverse G add neg 0;
  }.
  Hint Mode Group ! - - - - : typeclass_instances.
  Hint Rewrite @left_inverse @right_inverse : algebra_db.
End Group.
Export Group.
