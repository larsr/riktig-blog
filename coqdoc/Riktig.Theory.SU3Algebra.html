A table of structure constants makes a Lie algebra of coordinate vectors. The bracket is the double sum the table weights, and all three Lie laws come from two facts about the table alone: that it changes sign when its first two indices are swapped, and that the Jacobi sum vanishes on the standard basis. Nothing in this file knows how many generators there are or what the numbers say — su(3)'s own table, its nine numbers and its eight generators, is Impl/SU3.v, which checks the second fact and instantiates the theorem at the end.

  Lemma bracket_table_F : ∀ (x y : Fin.t n → F) (c : Fin.t n),
    [ x , y ] c == bigopF (fun a ⇒ bigopF (fun b ⇒ x a × y b × f a b c)).
  Proof.
    intros x y c.
    calc ([ x , y ] c)
      == (bigopI (fun a ⇒ bigopI (fun b ⇒ x a × y b × f a b c)))
           { (bracket_table f x y c) }
      == (bigopF (fun a ⇒ bigopI (fun b ⇒ x a × y b × f a b c)))
           { (symmetry (bigopF_bigopI n
                (fun a ⇒ bigopI (fun b ⇒ x a × y b × f a b c)))) }
      == (bigopF (fun a ⇒ bigopF (fun b ⇒ x a × y b × f a b c)))
           { (bigopF_ext n _ _ (fun a ⇒
                symmetry (bigopF_bigopI n
                  (fun b ⇒ x a × y b × f a b c)))) }.
  Qed.

  Ltac bigopF_ext_ring :=
    first
      [ apply bigopF_ext; intros ?; bigopF_ext_ring
      | rewrite ?pw_add_at, ?pw_scale_at, ?scale_is_mult; ring ].

  Lemma bracket_add_l : ∀ (x1 x2 y : Fin.t n → F) (c : Fin.t n),
    [ x1 + x2 , y ] c == [ x1 , y ] c + [ x2 , y ] c.
  Proof.
    intros x1 x2 y c.
    rewrite (bracket_table_F (x1 + x2) y c), (bracket_table_F x1 y c),
      (bracket_table_F x2 y c).
    transitivity (bigopF (fun a : Fin.t n ⇒
                            bigopF (fun b ⇒ x1 a × y b × f a b c)
                            + bigopF (fun b ⇒ x2 a × y b × f a b c))).
    - apply bigopF_ext. intros a.
      rewrite <- (bigopF_add n (fun b ⇒ x1 a × y b × f a b c)
                    (fun b ⇒ x2 a × y b × f a b c)).
      bigopF_ext_ring.
    - exact (bigopF_add n (fun a ⇒ bigopF (fun b ⇒ x1 a × y b × f a b c))
               (fun a ⇒ bigopF (fun b ⇒ x2 a × y b × f a b c))).
  Qed.

  Lemma bracket_add_r : ∀ (x y1 y2 : Fin.t n → F) (c : Fin.t n),
    [ x , y1 + y2 ] c == [ x , y1 ] c + [ x , y2 ] c.
  Proof.
    intros x y1 y2 c.
    rewrite (bracket_table_F x (y1 + y2) c), (bracket_table_F x y1 c),
      (bracket_table_F x y2 c).
    transitivity (bigopF (fun a : Fin.t n ⇒
                            bigopF (fun b ⇒ x a × y1 b × f a b c)
                            + bigopF (fun b ⇒ x a × y2 b × f a b c))).
    - apply bigopF_ext. intros a.
      rewrite <- (bigopF_add n (fun b ⇒ x a × y1 b × f a b c)
                    (fun b ⇒ x a × y2 b × f a b c)).
      bigopF_ext_ring.
    - exact (bigopF_add n (fun a ⇒ bigopF (fun b ⇒ x a × y1 b × f a b c))
               (fun a ⇒ bigopF (fun b ⇒ x a × y2 b × f a b c))).
  Qed.

  Lemma bracket_scale_l : ∀ (a : F) (x y : Fin.t n → F) (c : Fin.t n),
    [ a *: x , y ] c == a × [ x , y ] c.
  Proof.
    intros a x y c.
    rewrite (bracket_table_F (a *: x) y c), (bracket_table_F x y c).
    transitivity (bigopF (fun p : Fin.t n ⇒
                            a × bigopF (fun q ⇒ x p × y q × f p q c))).
    - apply bigopF_ext. intros p.
      rewrite <- (bigopF_mul_l n a (fun q ⇒ x p × y q × f p q c)).
      bigopF_ext_ring.
    - exact (bigopF_mul_l n a (fun p ⇒ bigopF (fun q ⇒ x p × y q × f p q c))).
  Qed.

  Lemma bracket_scale_r : ∀ (a : F) (x y : Fin.t n → F) (c : Fin.t n),
    [ x , a *: y ] c == a × [ x , y ] c.
  Proof.
    intros a x y c.
    rewrite (bracket_table_F x (a *: y) c), (bracket_table_F x y c).
    transitivity (bigopF (fun p : Fin.t n ⇒
                            a × bigopF (fun q ⇒ x p × y q × f p q c))).
    - apply bigopF_ext. intros p.
      rewrite <- (bigopF_mul_l n a (fun q ⇒ x p × y q × f p q c)).
      bigopF_ext_ring.
    - exact (bigopF_mul_l n a (fun p ⇒ bigopF (fun q ⇒ x p × y q × f p q c))).
  Qed.

  Lemma bracket_proper :
    Proper (equ ==> equ ==> equ) (bracket (V := Fin.t n → F)).
  Proof.
    intros x x' Hx y y' Hy c.
    rewrite (bracket_table_F x y c), (bracket_table_F x' y' c).
    apply bigopF_ext. intros a.
    apply bigopF_ext. intros b.
    exact (op_proper (x a × y b) (x' a × y' b)
             (op_proper (x a) (x' a) (Hx a) (y b) (y' b) (Hy b))
             (f a b c) (f a b c) (reflexivity (f a b c))).
  Qed.

  Definition structure_bracket_bilinear :
    Bilinear F (Fin.t n → F) (Fin.t n → F) (Fin.t n → F) bracket :=
    {| Bilinear.proper := bracket_proper;
       Bilinear.linear_l := fun v ⇒
         {| Linear.proper := fun x x' Hx ⇒
              bracket_proper x x' Hx v v (reflexivity v);
            Linear.is_additive := fun x x' ⇒ bracket_add_l x x' v;
            Linear.is_homogeneous := fun a x ⇒ bracket_scale_l a x v |};
       Bilinear.linear_r := fun u ⇒
         {| Linear.proper := fun y y' Hy ⇒
              bracket_proper u u (reflexivity u) y y' Hy;
            Linear.is_additive := fun y y' ⇒ bracket_add_r u y y';
            Linear.is_homogeneous := fun a y ⇒ bracket_scale_r a u y |} |}.

  #[local] Existing Instance structure_bracket_bilinear.

  Lemma mul_f_antisym_term : ∀ (x y : Fin.t n → F) (a b c : Fin.t n),
    x a × y b × f b a c == - (x a × y b × f a b c).
  Proof.
    intros x y a b c.
    rewrite (table_antisym f b a c).
    exact (mul_neg_r (x a × y b) (f a b c)).
  Qed.

  Lemma bracket_antisym : ∀ (x y : Fin.t n → F) (c : Fin.t n),
    [ y , x ] c == - [ x , y ] c.
  Proof.
    intros x y c.
    rewrite (bracket_table_F y x c), (bracket_table_F x y c).
    calc (bigopF (fun a ⇒ bigopF (fun b ⇒ y a × x b × f a b c)))
      == (bigopF (fun a ⇒ bigopF (fun b ⇒ x b × y a × f a b c)))
      == (bigopF (fun a ⇒ bigopF (fun b ⇒ x a × y b × f b a c)))
      == (bigopF (fun a ⇒ bigopF (fun b ⇒ - (x a × y b × f a b c))))
      == (bigopF (fun a ⇒ - bigopF (fun b ⇒ x a × y b × f a b c)))
      == (- bigopF (fun a ⇒ bigopF (fun b ⇒ x a × y b × f a b c))).
    - bigopF_ext_ring.
    - apply bigopF_swap.
    - exact (bigopF_ext n _ _
               (fun a ⇒ bigopF_ext n _ _ (fun b ⇒ mul_f_antisym_term x y a b c))).
    - exact (bigopF_ext n _ _
               (fun a ⇒ bigopF_neg n (fun b ⇒ x a × y b × f a b c))).
    - apply bigopF_neg.
  Qed.

  Lemma bracket_self_zero : ∀ (x : Fin.t n → F) (c : Fin.t n),
    [ x , x ] c == 0.
  Proof.
    intros x c.
    apply (field_self_neg_zero two_neq_0).
    exact (bracket_antisym x x c).
  Qed.

  Definition structure_bracket_alternates : Alternates (Fin.t n → F) bracket :=
    bracket_self_zero.

  Lemma e_mul_delta_term : ∀ (K : Fin.t n → F) (j b : Fin.t n),
    e j b × K b == K b × e b j.
  Proof.
    intros K j b.
    rewrite (mul_commutative (e j b) (K b)).
    rewrite (e_sym n j b).
    reflexivity.
  Qed.

  Lemma e_mul_delta : ∀ (K : Fin.t n → F) (j : Fin.t n),
    bigopF (fun b ⇒ e j b × K b) == K j.
  Proof.
    intros K j.
    rewrite (bigopF_ext n _ _ (fun b ⇒ e_mul_delta_term K j b)).
    exact (lincomb_std_delta n K j).
  Qed.

  Lemma bracket_inner_collapse : ∀ i j d a : Fin.t n,
    bigopF (fun b ⇒ e i a × e j b × f a b d) == e i a × f a j d.
  Proof.
    intros i j d a.
    calc (bigopF (fun b ⇒ e i a × e j b × f a b d))
      == (bigopF (fun b ⇒ e i a × (e j b × f a b d)))
        { (bigopF_ext n _ _ (fun b ⇒ associative (e i a) (e j b) (f a b d))) }
      == (e i a × bigopF (fun b ⇒ e j b × f a b d))
        { bigopF_mul_l }
      == (e i a × f a j d)
        { e_mul_delta }.
  Qed.

The bracket of two basis generators is a single table lookup.
  Lemma bracket_e_e : ∀ i j d : Fin.t n, [ e i , e j ] d == f i j d.
  Proof.
    intros i j d.
    rewrite (bracket_table_F (e i) (e j) d).
    rewrite (bigopF_ext n _ _ (fun a ⇒ bracket_inner_collapse i j d a)).
    exact (e_mul_delta (fun a ⇒ f a j d) i).
  Qed.

  Lemma bracket_e_l_inner : ∀ (i : Fin.t n) (y : Fin.t n → F) (c a : Fin.t n),
    bigopF (fun b ⇒ e i a × y b × f a b c) == e i a × bigopF (fun b ⇒ y b × f a b c).
  Proof.
    intros i y c a.
    rewrite (bigopF_ext n _ _ (fun b ⇒ associative (e i a) (y b) (f a b c))).
    exact (bigopF_mul_l n (e i a) (fun b ⇒ y b × f a b c)).
  Qed.

  Lemma bracket_e_l : ∀ (i : Fin.t n) (y : Fin.t n → F) (c : Fin.t n),
    [ e i , y ] c == bigopF (fun b ⇒ y b × f i b c).
  Proof.
    intros i y c.
    rewrite (bracket_table_F (e i) y c).
    rewrite (bigopF_ext n _ _ (fun a ⇒ bracket_e_l_inner i y c a)).
    exact (e_mul_delta (fun a ⇒ bigopF (fun b ⇒ y b × f a b c)) i).
  Qed.

  Lemma jacobi_sum_at : ∀ (x y z : Fin.t n → F) (d : Fin.t n),
    ([ x , [ y , z ] ] + [ y , [ z , x ] ] + [ z , [ x , y ] ]) d
      == [ x , [ y , z ] ] d + [ y , [ z , x ] ] d + [ z , [ x , y ] ] d.
  Proof. intros x y z d. reflexivity. Qed.

  Lemma jacobi_term : ∀ i j k d : Fin.t n,
    [ e i , [ e j , e k ] ] d == bigopF (fun b ⇒ f j k b × f i b d).
  Proof.
    intros i j k d.
    rewrite (bracket_e_l i [ e j , e k ] d).
    exact (bigopF_ext n _ _ (fun b ⇒ op_proper ([ e j , e k ] b) (f j k b)
             (bracket_e_e j k b) (f i b d) (f i b d) (reflexivity (f i b d)))).
  Qed.

The Jacobi sum on three generators, in the table alone: three contractions of the table with itself. This is the goal a concrete table has to discharge.
  Lemma jacobi_basis_expand : ∀ i j k d : Fin.t n,
    ([ e i , [ e j , e k ] ] + [ e j , [ e k , e i ] ] + [ e k , [ e i , e j ] ]) d
      == bigopF (fun b ⇒ f j k b × f i b d) + bigopF (fun b ⇒ f k i b × f j b d)
         + bigopF (fun b ⇒ f i j b × f k b d).
  Proof.
    intros i j k d.
    calc (([ e i , [ e j , e k ] ] + [ e j , [ e k , e i ] ]
             + [ e k , [ e i , e j ] ]) d)
      == ([ e i , [ e j , e k ] ] d + [ e j , [ e k , e i ] ] d
            + [ e k , [ e i , e j ] ] d)
        { (jacobi_sum_at (e i) (e j) (e k) d) }
      == (bigopF (fun b ⇒ f j k b × f i b d) + [ e j , [ e k , e i ] ] d
            + [ e k , [ e i , e j ] ] d)
        { jacobi_term }
      == (bigopF (fun b ⇒ f j k b × f i b d) + bigopF (fun b ⇒ f k i b × f j b d)
            + [ e k , [ e i , e j ] ] d)
        { jacobi_term }
      == (bigopF (fun b ⇒ f j k b × f i b d) + bigopF (fun b ⇒ f k i b × f j b d)
            + bigopF (fun b ⇒ f i j b × f k b d))
        { jacobi_term }.
  Qed.

The theorem the table buys: checking the Jacobi sum on the generators is enough. Bilinearity and antisymmetry are already the table's, and Theory/LieAlgebra.v's basis builder carries the remaining law from the generators to every element.
  Definition lie_algebra_of_table
      (jacobi_on_basis : ∀ i j k : Fin.t n,
         [ e i , [ e j , e k ] ] + [ e j , [ e k , e i ] ] + [ e k , [ e i , e j ] ]
           == 0)
      : LieAlgebra F (Fin.t n → F) :=
    lie_algebra_from_basis (FinVec_Basis n) jacobi_on_basis.
