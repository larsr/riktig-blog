The ratio test, with ratio a half: any sequence that respects equality and, past some index, shrinks by at least half at every step, has Cauchy partial sums -- the tail is under a geometric majorant. This is the exponential argument with the term structure abstracted away.
Section RatioTest.
  Context {R Q N : Type} `{ConstructiveReals R Q N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}
    {of_nat : N → Q} {AQ : Archimedean Q N of_nat} {LDQ : LtDec Q}.

  Local Notation "'psum' f n" := (natrec _ 0 (fun k acc ⇒ acc + f k) n) (at level 10, f at level 9, n at level 9).
  Local Notation "'is_cauchy' u" :=
    (∀ eps : Q, 0 < eps →
      {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps})
    (at level 10, u at level 9).
  Local Notation half := (inv ((1 : Q) + 1)).
  Local Notation "'pow' r n" := (natrec Q 1 (fun _ acc ⇒ acc × r) n) (at level 10, r at level 9, n at level 9).

  Variable t : N → R.
  Hypothesis Ht : Proper (equ ==> equ) t.
  Variable m : N.
  Hypothesis Hratio : ∀ k : N, m ≤ k → abs (t (succ k)) ≼ cast half × abs (t k).

  Lemma ratio_tail : ∀ j : N, abs (t (m + j)) ≼ abs (t m) × cast (pow half j).
  Proof.
    apply (nat_induction (fun j ⇒ abs (t (m + j)) ≼ abs (t m) × cast (pow half j))).
    - intros j j' Hj. split; intros Hle.
      + apply (le_equ_l (abs (t (m + j))));
          [ apply ConstructiveReals.abs_proper; apply Ht; exact (op_proper m m (reflexivity m) j j' Hj) | ].
        apply (le_equ_r _ (abs (t m) × cast (pow half j)));
          [ apply op_proper; [ reflexivity | apply hom_proper; exact (pow_proper half j j' Hj) ] | exact Hle ].
      + apply (le_equ_l (abs (t (m + j'))));
          [ apply ConstructiveReals.abs_proper; apply Ht; exact (op_proper m m (reflexivity m) j' j (symmetry Hj)) | ].
        apply (le_equ_r _ (abs (t m) × cast (pow half j')));
          [ apply op_proper; [ reflexivity | apply hom_proper; exact (pow_proper half j' j (symmetry Hj)) ] | exact Hle ].
    - apply (le_equ_l (abs (t m)));
        [ apply ConstructiveReals.abs_proper; apply Ht; exact (symmetry (Theory.Naturals.add_zero m)) | ].
      apply le_of_equ.
      calc (abs (t m))
        == (abs (t m) × 1) { (right_identity (action := mult) (abs (t m))) }
        == (abs (t m) × cast 1) { hom_one }
        == (abs (t m) × cast (pow half (0 : N))) { (pow_zero half) }.
    - intros j IH.
      apply (le_equ_l (abs (t (succ (m + j))))).
      { apply ConstructiveReals.abs_proper. apply Ht. exact (symmetry (Theory.Naturals.add_succ m j)). }
      apply (Theory.ConstructiveReals.le_trans _ (cast half × abs (t (m + j)))).
      { exact (Hratio (m + j) (Theory.NatOrder.le_add_r m j)). }
      apply (le_equ_r _ (cast half × (abs (t m) × cast (pow half j)))).
      { calc (cast half × (abs (t m) × cast (pow half j)))
          == ((abs (t m) × cast (pow half j)) × cast half) { (commutative (op := mult) (cast half) (abs (t m) × cast (pow half j))) }
          == (abs (t m) × (cast (pow half j) × cast half)) { (associative (op := mult) (action := mult) (abs (t m)) (cast (pow half j)) (cast half)) }
          == (abs (t m) × cast (pow half j × half)) { (additive (f := cast) (opA := mult) (pow half j) half) }
          == (abs (t m) × cast (pow half (succ j))) { (pow_succ half j) }. }
      apply ConstructiveReals.mul_le_mono_l.
      + exact (cast_nonneg _ (Theory.GeometricSeries.half_nonneg (F := Q))).
      + exact IH.
  Qed.

  Lemma ratio_majorant : ∀ C : Q, abs (t m) ≼ cast C →
    ∀ j : N, abs (t (m + j)) ≼ cast (C × pow half j).
  Proof.
    intros C HC j.
    apply (Theory.ConstructiveReals.le_trans _ (abs (t m) × cast (pow half j))); [ exact (ratio_tail j) | ].
    apply (le_equ_r _ (cast C × cast (pow half j)));
      [ symmetry; exact (additive (f := cast) (opA := mult) C (pow half j)) | ].
    apply Theory.ConstructiveReals.mul_le_mono_r; [ | exact HC ].
    exact (cast_nonneg _ (pow_nonneg half (Theory.GeometricSeries.half_nonneg (F := Q)) j)).
  Qed.

  Theorem ratio_summable : is_cauchy (fun n ⇒ psum (fun j ⇒ t j) n).
  Proof.
    destruct (ConstructiveReals.archimedean (abs (t m))) as [C HC].
    assert (HC0 : 0 ≤ C).
    { apply (Theory.Order.le_of_lt 0 C). apply (lt_cast 0 C).
      apply (lt_equ_l 0 (cast 0)); [ symmetry; exact (hom_zero cast) | ].
      exact (le_lt_trans 0 (abs (t m)) (cast C) (Theory.ConstructiveReals.abs_nonneg _) HC). }
    assert (Hshift : is_cauchy (fun c ⇒ psum (fun j ⇒ t (m + j)) c)).
    { apply (cauchy_psum_of_abs_le (fun j ⇒ t (m + j)) (fun j ⇒ C × pow half j)).
      - intros j j' Hj. apply Ht. exact (op_proper m m (reflexivity m) j j' Hj).
      - intros j j' Hj. apply op_proper; [ reflexivity | exact (pow_proper half j j' Hj) ].
      - intros j. exact (ratio_majorant C (Theory.ConstructiveReals.le_of_lt _ _ HC) j).
      - intros eps Heps.
        destruct (majorant_tail_small (N := N) (of_nat := of_nat) C eps HC0 Heps) as [n Hn]. ∃ n.
        intros i c Hi.
        apply (transitivity (y := psum (fun k ⇒ C × pow half (i + k)) c)); [ | exact (Hn i c Hi) ].
        apply le_of_eq. apply psum_ext.
        + intros k k' Hk. apply op_proper; [ reflexivity | exact (pow_proper half (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk)) ].
        + intros k k' Hk. apply op_proper; [ reflexivity | exact (pow_proper half (i + k) (i + k') (op_proper i i (reflexivity i) k k' Hk)) ].
        + intros k. reflexivity. }
    apply (cauchy_of_shift (fun n ⇒ psum (fun j ⇒ t j) n) m (psum_proper _ Ht)).
    apply (Theory.ConstructiveRealsLimits.cauchy_proper (fun c ⇒ psum (fun j ⇒ t j) m + psum (fun j ⇒ t (m + j)) c)).
    - intros c. symmetry. exact (psum_shift (fun j ⇒ t j) Ht m c).
    - exact (Theory.ConstructiveRealsLimits.cauchy_add _ _ (cauchy_of_converges _ _ (converges_const (psum (fun j ⇒ t j) m))) Hshift).
  Defined.
End RatioTest.
