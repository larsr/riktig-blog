Section AccumulatorSum.
  Context {M : Type} `{MM : Monoid M}.

bigop_go a f l adds f over l onto a running total a.
  Fixpoint bigop_go {J : Type} (a : M) (f : J → M) (l : list J) : M :=
    match l with
    | nil ⇒ a
    | j :: l' ⇒ bigop_go (a + f j) f l'
    end.

bigop_acc f l starts that walk at zero.
  Definition bigop_acc {J : Type} (f : J → M) (l : list J) : M :=
    bigop_go 0 f l.

  Lemma bigop_go_shift : ∀ {J : Type} (f : J → M) (l : list J) (a : M),
    bigop_go a f l == a + bigop_go 0 f l.
  Proof.
    intros J f l; induction l as [| j l IH]; intros a.
    - calc (bigop_go a f nil)
        == a { (reflexivity a) }
        == (a + (0 : M)) { (symmetry (right_identity a)) }
        == (a + bigop_go 0 f nil) { (reflexivity (a + (0 : M))) }.
    - calc (bigop_go a f (j :: l))
        == (bigop_go (a + f j) f l) { (reflexivity (bigop_go (a + f j) f l)) }
        == ((a + f j) + bigop_go 0 f l) { (IH (a + f j)) }
        == (a + (f j + bigop_go 0 f l)) { (symmetry (associative a (f j) (bigop_go 0 f l))) }
        == (a + ((0 : M) + f j + bigop_go 0 f l))
             { (op_proper a a (reflexivity a) _ _
                 (op_proper _ _ (symmetry (left_identity (f j)))
                            _ _ (reflexivity (bigop_go 0 f l)))) }
        == (a + bigop_go ((0 : M) + f j) f l)
             { (op_proper a a (reflexivity a) _ _
                 (symmetry (IH ((0 : M) + f j)))) }
        == (a + bigop_go 0 f (j :: l))
             { (reflexivity (a + bigop_go ((0 : M) + f j) f l)) }.
  Qed.

The accumulator sum obeys the two equations that say what a big sum is, so every law proved from them holds of it.
  #[export] Instance BigOp_acc : BigOp M (@bigop_acc).
  Proof.
    split.
    - intros J f. exact (reflexivity (0 : M)).
    - intros J f j l.
      calc (bigop_acc f (j :: l))
        == (bigop_go ((0 : M) + f j) f l)
             { (reflexivity (bigop_go ((0 : M) + f j) f l)) }
        == (((0 : M) + f j) + bigop_go 0 f l)
             { (bigop_go_shift f l ((0 : M) + f j)) }
        == (f j + bigop_go 0 f l)
             { (op_proper _ _ (left_identity (f j)) _ _
                 (reflexivity (bigop_go 0 f l))) }
        == (f j + bigop_acc f l)
             { (reflexivity (f j + bigop_go 0 f l)) }.
  Qed.

End AccumulatorSum.

Example acc_append (M : Type) `{Monoid M} (J : Type) (f : J → M)
    (l1 l2 : list J) :
  bigop_acc f (l1 ++ l2) == bigop_acc f l1 + bigop_acc f l2 :=
  bigop_list_append (bigop_list := @bigop_acc M _ _) f l1 l2.

Example acc_computes_on_Z :
  bigop_acc (fun i : Z ⇒ i) (1 :: 2 :: 3 :: 4 :: nil)%Z = 10%Z.
Proof. reflexivity. Qed.
