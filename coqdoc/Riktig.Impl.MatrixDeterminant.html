The determinant of a matrix as the determinant of the family of its columns, each read as a vector through the basis.
Section MatrixDeterminantOfDeterminant.
  Context {F I V N : Type} `{VectorSpace F V} `{Finite I N} {EI : EquDec I}
    {natrecN : HasNatRec N} {NR : NatRec N natrecN}
    {basis : I → V} {basis_proper : Proper (equ ==> equ) basis}
    {DT : HasDet F I V} {D : Determinant F I V basis DT}.

  #[export] Instance Columns_MatDet : HasMatDet F I :=
    fun M ⇒ det (fun j ⇒ natrec V 0 (fun k acc ⇒ acc + M (elem k) j *: basis (elem k)) card).

  #[export] Instance Columns_MatrixDeterminant : MatrixDeterminant F I Columns_MatDet.
  Proof.
    split; unfold matdet, Columns_MatDet.
    - exact matdet_proper.
    - exact matdet_col_add.
    - exact matdet_col_smul.
    - exact matdet_alternating.
    - exact matdet_id.
  Qed.
End MatrixDeterminantOfDeterminant.
