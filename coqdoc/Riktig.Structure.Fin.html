A finite index type is n distinguishable indices: equality up to == is decidable and one list, duplicate-free up to ==, enumerates every element up to ==.
Module FinIndex.
  Class FinIndex (n : nat) (I : Type) {eI : EqRel I} {Ieq : EquDec I} := {
    enum_equivalence :: Equivalence (@equ I eI);
    enum : list I;
    enum_NoDup : NoDupA equ enum;
    enum_complete : ∀ i : I, InA equ i enum;
    enum_length : length enum = n;
  }.
  Hint Mode FinIndex - ! - - : typeclass_instances.
End FinIndex.
Export FinIndex.
