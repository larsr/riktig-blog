A finite index type is n distinguishable indices: equality is decidable and one duplicate-free list of length n enumerates every element.
Module FinIndex.
  Class FinIndex (n : nat) (I : Type) {eqdec : EqDec I} := {
    enum : list I;
    enum_NoDup : NoDup enum;
    enum_complete : ∀ i : I, In i enum;
    enum_length : length enum = n;
  }.
  Hint Mode FinIndex - ! - : typeclass_instances.
End FinIndex.
