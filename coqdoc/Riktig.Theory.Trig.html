Section TrigTheory.
  Context {R : Type} `{Trig R}.

  Theorem sin_sub : ∀ a b : R, sin (a + - b) == sin a × cos b + - (cos a × sin b).
  Proof.
    intros a b.
    calc (sin (a + - b))
      == (sin a × cos (- b) + cos a × sin (- b)) { sin_add }
      == (sin a × cos b + cos a × sin (- b)) { cos_neg }
      == (sin a × cos b + cos a × - sin b) { sin_neg }
      == (sin a × cos b + - (cos a × sin b)) { mul_neg }.
  Qed.

  Theorem cos_sub : ∀ a b : R, cos (a + - b) == cos a × cos b + sin a × sin b.
  Proof.
    intros a b.
    calc (cos (a + - b))
      == (cos a × cos (- b) + - (sin a × sin (- b))) { cos_add }
      == (cos a × cos b + - (sin a × sin (- b))) { cos_neg }
      == (cos a × cos b + - (sin a × - sin b)) { sin_neg }
      == (cos a × cos b + - - (sin a × sin b)) { mul_neg }
      == (cos a × cos b + sin a × sin b) { neg_neg }.
  Qed.

The Pythagorean identity is the cosine of a + - a.
  Theorem cos_sq_add_sin_sq : ∀ a : R, cos a × cos a + sin a × sin a == 1.
  Proof.
    intros a.
    calc (cos a × cos a + sin a × sin a)
      == (cos (a + - a)) { cos_sub }
      == (cos (0 : R)) { (right_inverse (op := add) a) }
      == (1 : R) { cos_zero }.
  Qed.

  Theorem sin_sq_add_cos_sq : ∀ a : R, sin a × sin a + cos a × cos a == 1.
  Proof.
    intros a.
    calc (sin a × sin a + cos a × cos a)
      == (cos a × cos a + sin a × sin a) { (commutative (op := add) (sin a × sin a) (cos a × cos a)) }
      == (1 : R) { cos_sq_add_sin_sq }.
  Qed.

  Theorem sin_sq : ∀ a : R, sin a × sin a == 1 + - (cos a × cos a).
  Proof. intros a. exact (eq_sub_of_add_eq' _ _ _ (cos_sq_add_sin_sq a)). Qed.

  Theorem cos_sq' : ∀ a : R, cos a × cos a == 1 + - (sin a × sin a).
  Proof. intros a. exact (eq_sub_of_add_eq _ _ _ (cos_sq_add_sin_sq a)). Qed.

  Theorem sin_two_mul : ∀ a : R, sin (a + a) == (1 + 1) × (sin a × cos a).
  Proof.
    intros a.
    calc (sin (a + a))
      == (sin a × cos a + cos a × sin a) { sin_add }
      == (sin a × cos a + sin a × cos a) { (commutative (op := mult) (cos a) (sin a)) }
      == ((1 + 1) × (sin a × cos a)) { two_mul }.
  Qed.

  Theorem cos_two_mul : ∀ a : R, cos (a + a) == cos a × cos a + - (sin a × sin a).
  Proof. intros a. exact (cos_add a a). Qed.

  Theorem cos_two_mul' : ∀ a : R, cos (a + a) == (1 + 1) × (cos a × cos a) + - (1).
  Proof.
    intros a.
    calc (cos (a + a))
      == (cos a × cos a + - (sin a × sin a)) { cos_add }
      == (cos a × cos a + - (1 + - (cos a × cos a))) { sin_sq }
      == (cos a × cos a + (- (1) + - - (cos a × cos a))) { neg_add }
      == (cos a × cos a + (- (1) + cos a × cos a)) { neg_neg }
      == (cos a × cos a + (cos a × cos a + - (1))) { (commutative (op := add) (- (1)) (cos a × cos a)) }
      == ((cos a × cos a + cos a × cos a) + - (1)) { associative }
      == ((1 + 1) × (cos a × cos a) + - (1)) { two_mul }.
  Qed.
End TrigTheory.
