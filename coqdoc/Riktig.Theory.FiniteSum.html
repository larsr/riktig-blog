The sum of a table over a finite index, by recursion on the count: the entries at the numbers below n, added in that order. The four laws of a sum hold of it at n = card; the fourth because the sum is 0 until the distinguished index is passed and a after.
Section FiniteSum.
  Context {I N A T : Type} `{Finite I N} {natrec : HasNatRec N} `{!NatRec N natrec}
          `{Monoid A} `{!Commutative A A add}
          {eqT : EqRel T} {get : HasGet T I A} {tabulate : HasTabulate T I A}
          `{!Table I A T get tabulate}.

  Local Notation sum_below t n := (natrec A 0 (fun k acc ⇒ acc + get t (elem k)) n).

  Theorem step_proper : ∀ t : T,
    Proper (equ ==> equ ==> equ) (fun k acc ⇒ acc + get t (elem k)).
  Proof. intros t k k' Hk acc acc' Hacc. rewrite Hacc, Hk. reflexivity. Qed.

  Theorem sum_below_n_proper : ∀ (t : T) (n n' : N), n == n' → sum_below t n == sum_below t n'.
  Proof. intros t. exact (NatRec.natrec_proper A _ _ 0 _ (step_proper t)). Qed.

  Theorem sum_below_zero : ∀ t : T, sum_below t 0 == 0.
  Proof. intros t. exact (NatRec.natrec_zero A _ _ 0 _). Qed.

  Theorem sum_below_succ : ∀ (t : T) (n : N),
    sum_below t (succ n) == sum_below t n + get t (elem n).
  Proof. intros t n. exact (NatRec.natrec_succ A _ _ 0 _ n). Qed.

  Theorem sum_below_t_proper : ∀ (s t : T) (n : N),
    s == t → sum_below s n == sum_below t n.
  Proof.
    intros s t n Hst. revert n.
    apply (Peano.nat_induction (fun n ⇒ sum_below s n == sum_below t n)).
    - intros n n' Hn. rewrite (sum_below_n_proper s n n' Hn), (sum_below_n_proper t n n' Hn). reflexivity.
    - calc (sum_below s 0)
        == (0 : A) { sum_below_zero }
        == (sum_below t 0) { sum_below_zero }.
    - intros n IH.
      calc (sum_below s (succ n))
        == (sum_below s n + get s (elem n)) { sum_below_succ }
        == (sum_below t n + get s (elem n)) { IH }
        == (sum_below t n + get t (elem n)) { Hst }
        == (sum_below t (succ n)) { sum_below_succ }.
  Qed.

  Theorem sum_below_zero_table : ∀ n : N,
    sum_below (tabulate (fun _ ⇒ 0) : T) n == 0.
  Proof.
    apply (Peano.nat_induction (fun n ⇒ sum_below (tabulate (fun _ ⇒ 0) : T) n == 0)).
    - intros n n' Hn. rewrite (sum_below_n_proper _ n n' Hn). reflexivity.
    - exact (sum_below_zero _).
    - intros n IH.
      calc (sum_below (tabulate (fun _ ⇒ 0) : T) (succ n))
        == (sum_below (tabulate (fun _ ⇒ 0) : T) n + get (tabulate (fun _ ⇒ 0) : T) (elem n))
             { sum_below_succ }
        == ((0 : A) + get (tabulate (fun _ ⇒ 0) : T) (elem n)) { IH }
        == ((0 : A) + 0) { (Table.get_tabulate (fun _ ⇒ (0 : A)) (elem n)) }
        == (0 : A) { left_identity }.
  Qed.

  Theorem sum_below_add : ∀ (s t : T) (n : N),
    sum_below (tabulate (fun i ⇒ get s i + get t i) : T) n == sum_below s n + sum_below t n.
  Proof.
    intros s t.
    apply (Peano.nat_induction (fun n ⇒
      sum_below (tabulate (fun i ⇒ get s i + get t i) : T) n == sum_below s n + sum_below t n)).
    - intros n n' Hn.
      rewrite (sum_below_n_proper _ n n' Hn), (sum_below_n_proper s n n' Hn), (sum_below_n_proper t n n' Hn).
      reflexivity.
    - calc (sum_below (tabulate (fun i ⇒ get s i + get t i) : T) 0)
        == (0 : A) { sum_below_zero }
        == ((0 : A) + 0) { left_identity }
        == (sum_below s 0 + 0) { (sum_below_zero s) }
        == (sum_below s 0 + sum_below t 0) { (sum_below_zero t) }.
    - intros n IH.
      calc (sum_below (tabulate (fun i ⇒ get s i + get t i) : T) (succ n))
        == (sum_below (tabulate (fun i ⇒ get s i + get t i) : T) n
              + get (tabulate (fun i ⇒ get s i + get t i) : T) (elem n)) { sum_below_succ }
        == ((sum_below s n + sum_below t n)
              + get (tabulate (fun i ⇒ get s i + get t i) : T) (elem n)) { IH }
        == ((sum_below s n + sum_below t n) + (get s (elem n) + get t (elem n)))
             { (Table.get_tabulate (fun i ⇒ get s i + get t i) (elem n)) }
        == ((sum_below s n + get s (elem n)) + (sum_below t n + get t (elem n)))
             { add_add_add_comm }
        == (sum_below s (succ n) + (sum_below t n + get t (elem n))) { (sum_below_succ s n) }
        == (sum_below s (succ n) + sum_below t (succ n)) { (sum_below_succ t n) }.
  Qed.

A table with a at i and 0 elsewhere: below i's index the sum is still empty, and past it the sum is a.
  Section Single.
    Variables (t : T) (i : I) (a : A).
    Hypothesis Hi : get t i == a.
    Hypothesis Hj : ∀ j : I, j =/= i → get t j == 0.

    Lemma entry_at_index : ∀ n : N, n == index i → get t (elem n) == a.
    Proof.
      intros n Hn.
      calc (get t (elem n))
        == (get t i) { (Finite.elem_eq_of_index i n Hn) }
        == a { Hi }.
    Qed.

    Lemma entry_off_index : ∀ n : N, n < card → n =/= index i → get t (elem n) == 0.
    Proof.
      intros n Hn Hne. apply Hj. intros Hin. apply Hne.
      exact (Finite.index_eq_of_elem i n Hn Hin).
    Qed.

    Theorem sum_below_before : ∀ n : N, n ≤ index i → sum_below t n == 0.
    Proof.
      apply (nat_induction_le (index i) (fun n ⇒ sum_below t n == 0)).
      - intros n n' Hn. rewrite (sum_below_n_proper t n n' Hn). reflexivity.
      - exact (sum_below_zero t).
      - intros n Hn IH.
        calc (sum_below t (succ n))
          == (sum_below t n + get t (elem n)) { sum_below_succ }
          == ((0 : A) + get t (elem n)) { IH }
          == ((0 : A) + 0)
               { (entry_off_index n (lt_trans n (index i) card Hn (Finite.index_lt i)) (proj2 Hn)) }
          == (0 : A) { left_identity }.
    Qed.

    Theorem sum_below_after : ∀ n : N, index i < n → n ≤ card → sum_below t n == a.
    Proof.
      apply (nat_induction_between (index i) card (fun n ⇒ sum_below t n == a)).
      - intros n n' Hn. rewrite (sum_below_n_proper t n n' Hn). reflexivity.
      - calc (sum_below t (succ (index i)))
          == (sum_below t (index i) + get t (elem (index i))) { sum_below_succ }
          == ((0 : A) + get t (elem (index i))) { (sum_below_before (index i) (le_refl (index i))) }
          == ((0 : A) + a) { (entry_at_index (index i) (reflexivity (index i))) }
          == a { left_identity }.
      - intros n Hlt Hn IH.
        calc (sum_below t (succ n))
          == (sum_below t n + get t (elem n)) { sum_below_succ }
          == (a + get t (elem n)) { IH }
          == (a + 0) { (entry_off_index n Hn (fun Hni ⇒ proj2 Hlt (symmetry Hni))) }
          == a { right_identity }.
    Qed.
  End Single.

The four laws, at the count.
  Theorem sum_card_proper : Proper (equ ==> equ) (fun t : T ⇒ sum_below t card).
  Proof. intros s t Hst. exact (sum_below_t_proper s t card Hst). Qed.

  Theorem sum_card_zero : sum_below (tabulate (fun _ ⇒ 0) : T) card == 0.
  Proof. exact (sum_below_zero_table card). Qed.

  Theorem sum_card_add : ∀ s t : T,
    sum_below (tabulate (fun i ⇒ get s i + get t i) : T) card == sum_below s card + sum_below t card.
  Proof. intros s t. exact (sum_below_add s t card). Qed.

  Theorem sum_card_single : ∀ (t : T) (i : I) (a : A),
    get t i == a → (∀ j : I, j =/= i → get t j == 0) → sum_below t card == a.
  Proof.
    intros t i a Hi Hj.
    exact (sum_below_after t i a Hi Hj card (Finite.index_lt i) (le_refl card)).
  Qed.

End FiniteSum.
