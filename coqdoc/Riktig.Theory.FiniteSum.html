The sum of a table over a finite index, by recursion on the count: the entries at the numbers below n, added in that order. The four laws of a sum hold of it at n = card; the fourth by an invariant on n: the sum is a once the distinguished index has been passed and 0 before.
Section FiniteSum.
  Context {I N A T : Type} `{Finite I N} {natrec : HasNatRec N} `{!NatRec N natrec}
          `{Monoid A} `{!Commutative A A add}
          {eqT : EqRel T} {get : HasGet T I A} {tabulate : HasTabulate T I A}
          `{!Table I A T get tabulate}.

  Local Notation sum_below t n := (natrec A 0 (fun k acc ⇒ acc + get t (elem k)) n).

  Theorem step_proper : ∀ t : T,
    Proper (equ ==> equ ==> equ) (fun k acc ⇒ acc + get t (elem k)).
  Proof. intros t k k' Hk acc acc' Hacc. rewrite Hacc, Hk. reflexivity. Qed.

  Theorem sum_below_n_proper : ∀ t : T, Proper (equ ==> equ) (fun n ⇒ sum_below t n).
  Proof. intros t. exact (natrec_proper A _ _ 0 _ (step_proper t)). Qed.

  Theorem sum_below_zero : ∀ t : T, sum_below t 0 == 0.
  Proof. intros t. exact (natrec_zero A _ _ 0 _). Qed.

  Theorem sum_below_succ : ∀ (t : T) (n : N),
    sum_below t (succ n) == sum_below t n + get t (elem n).
  Proof. intros t n. exact (natrec_succ A _ _ 0 _ n). Qed.

  Theorem sum_below_t_proper : ∀ (s t : T) (n : N),
    s == t → sum_below s n == sum_below t n.
  Proof.
    intros s t.
    apply (nat_induction (fun n ⇒ s == t → sum_below s n == sum_below t n)).
    - intros n n' Hn. split; intros Hst Hst';
        [ rewrite <- (sum_below_n_proper s n n' Hn), <- (sum_below_n_proper t n n' Hn)
        | rewrite (sum_below_n_proper s n n' Hn), (sum_below_n_proper t n n' Hn) ];
        exact (Hst Hst').
    - intros _.
      calc (sum_below s 0)
        == (0 : A) { sum_below_zero }
        == (sum_below t 0) { sum_below_zero }.
    - intros n IH Hst.
      calc (sum_below s (succ n))
        == (sum_below s n + get s (elem n)) { sum_below_succ }
        == (sum_below t n + get s (elem n)) { (IH Hst) }
        == (sum_below t n + get t (elem n)) { Hst }
        == (sum_below t (succ n)) { sum_below_succ }.
  Qed.

  Theorem sum_below_zero_table : ∀ n : N,
    sum_below (tabulate (fun _ ⇒ 0) : T) n == 0.
  Proof.
    apply (nat_induction (fun n ⇒ sum_below (tabulate (fun _ ⇒ 0) : T) n == 0)).
    - intros n n' Hn. rewrite (sum_below_n_proper _ n n' Hn). reflexivity.
    - exact (sum_below_zero _).
    - intros n IH.
      calc (sum_below (tabulate (fun _ ⇒ 0) : T) (succ n))
        == (sum_below (tabulate (fun _ ⇒ 0) : T) n + get (tabulate (fun _ ⇒ 0) : T) (elem n))
             { sum_below_succ }
        == ((0 : A) + get (tabulate (fun _ ⇒ 0) : T) (elem n)) { IH }
        == ((0 : A) + 0) { (get_tabulate (fun _ ⇒ (0 : A)) (elem n)) }
        == (0 : A) { left_identity }.
  Qed.

  Theorem sum_below_add : ∀ (s t : T) (n : N),
    sum_below (tabulate (fun i ⇒ get s i + get t i) : T) n == sum_below s n + sum_below t n.
  Proof.
    intros s t.
    apply (nat_induction (fun n ⇒
      sum_below (tabulate (fun i ⇒ get s i + get t i) : T) n == sum_below s n + sum_below t n)).
    - intros n n' Hn.
      rewrite (sum_below_n_proper _ n n' Hn), (sum_below_n_proper s n n' Hn),
              (sum_below_n_proper t n n' Hn). reflexivity.
    - calc (sum_below (tabulate (fun i ⇒ get s i + get t i) : T) 0)
        == (0 : A) { sum_below_zero }
        == ((0 : A) + 0) { left_identity }
        == (sum_below s 0 + 0) { (sum_below_zero s) }
        == (sum_below s 0 + sum_below t 0) { (sum_below_zero t) }.
    - intros n IH.
      calc (sum_below (tabulate (fun i ⇒ get s i + get t i) : T) (succ n))
        == (sum_below (tabulate (fun i ⇒ get s i + get t i) : T) n
              + get (tabulate (fun i ⇒ get s i + get t i) : T) (elem n)) { sum_below_succ }
        == ((sum_below s n + sum_below t n)
              + get (tabulate (fun i ⇒ get s i + get t i) : T) (elem n)) { IH }
        == ((sum_below s n + sum_below t n) + (get s (elem n) + get t (elem n)))
             { (get_tabulate (fun i ⇒ get s i + get t i) (elem n)) }
        == ((sum_below s n + get s (elem n)) + (sum_below t n + get t (elem n)))
             { add_add_add_comm }
        == (sum_below s (succ n) + (sum_below t n + get t (elem n))) { (sum_below_succ s n) }
        == (sum_below s (succ n) + sum_below t (succ n)) { (sum_below_succ t n) }.
  Qed.

  Theorem sum_below_single : ∀ (t : T) (i : I) (a : A),
    get t i == a → (∀ j : I, j =/= i → get t j == 0) →
    ∀ n : N, n ≤ card →
      (index i < n ∧ sum_below t n == a) ∨ (¬ index i < n ∧ sum_below t n == 0).
  Proof.
    intros t i a Hi Hj.
    apply (nat_induction (fun n ⇒ n ≤ card →
      (index i < n ∧ sum_below t n == a) ∨ (¬ index i < n ∧ sum_below t n == 0))).
    - assert (Hmono : ∀ n n' : N, n == n' →
        (n ≤ card → (index i < n ∧ sum_below t n == a) ∨ (¬ index i < n ∧ sum_below t n == 0)) →
        (n' ≤ card → (index i < n' ∧ sum_below t n' == a) ∨ (¬ index i < n' ∧ sum_below t n' == 0))).
      { intros n n' Hn Hp Hle'.
        assert (Hle : n ≤ card) by exact (proj2 (le_proper n n' Hn card card (reflexivity card)) Hle').
        destruct (Hp Hle) as [[Hlt Hs] | [Hnlt Hs]].
        - left. split.
          + exact (proj1 (lt_proper (index i) (index i) (reflexivity _) n n' Hn) Hlt).
          + rewrite <- (sum_below_n_proper t n n' Hn). exact Hs.
        - right. split.
          + intros Hlt'. apply Hnlt.
            exact (proj2 (lt_proper (index i) (index i) (reflexivity _) n n' Hn) Hlt').
          + rewrite <- (sum_below_n_proper t n n' Hn). exact Hs. }
      intros n n' Hn. split; [ exact (Hmono n n' Hn) | exact (Hmono n' n (symmetry Hn)) ].
    - intros _. right. split; [ exact (not_lt_zero (index i)) | exact (sum_below_zero t) ].
    - intros n IH Hsn.
      assert (Hn : n < card) by exact (lt_of_lt_of_le n (succ n) card (lt_succ_diag n) Hsn).
      destruct (IH (le_of_lt n card Hn)) as [[Hlt Hsum] | [Hnlt Hsum]].
      + left. split.
        × apply lt_succ_iff. left. exact Hlt.
        × destruct (eq_or_ne_index (elem n) i) as [Heq | Hne].
          -- exfalso.
             assert (Hin : index i == n).
             { calc (index i)
                 == (index (elem n : I)) { Heq }
                 == n { (index_elem n Hn) }. }
             exact (lt_irrefl n (proj1 (lt_proper (index i) n Hin n n (reflexivity n)) Hlt)).
          -- calc (sum_below t (succ n))
               == (sum_below t n + get t (elem n)) { sum_below_succ }
               == (a + get t (elem n)) { Hsum }
               == (a + 0) { (Hj (elem n) Hne) }
               == a { right_identity }.
      + destruct (eq_or_ne_index (elem n) i) as [Heq | Hne].
        × left. split.
          -- apply lt_succ_iff. right.
             calc (index i)
               == (index (elem n : I)) { Heq }
               == n { (index_elem n Hn) }.
          -- calc (sum_below t (succ n))
               == (sum_below t n + get t (elem n)) { sum_below_succ }
               == ((0 : A) + get t (elem n)) { Hsum }
               == ((0 : A) + get t i) { Heq }
               == ((0 : A) + a) { Hi }
               == a { left_identity }.
        × right. split.
          -- intros Hlt. destruct (proj1 (lt_succ_iff (index i) n) Hlt) as [Hlt' | Heq'].
             ++ exact (Hnlt Hlt').
             ++ apply Hne. symmetry.
                calc i
                  == (elem (index i) : I) { elem_index }
                  == (elem n : I) { Heq' }.
          -- calc (sum_below t (succ n))
               == (sum_below t n + get t (elem n)) { sum_below_succ }
               == ((0 : A) + get t (elem n)) { Hsum }
               == ((0 : A) + 0) { (Hj (elem n) Hne) }
               == (0 : A) { left_identity }.
  Qed.

The four laws, at the count.
  Theorem sum_card_proper : Proper (equ ==> equ) (fun t : T ⇒ sum_below t card).
  Proof. intros s t Hst. exact (sum_below_t_proper s t card Hst). Qed.

  Theorem sum_card_zero : sum_below (tabulate (fun _ ⇒ 0) : T) card == 0.
  Proof. exact (sum_below_zero_table card). Qed.

  Theorem sum_card_add : ∀ s t : T,
    sum_below (tabulate (fun i ⇒ get s i + get t i) : T) card == sum_below s card + sum_below t card.
  Proof. intros s t. exact (sum_below_add s t card). Qed.

  Theorem sum_card_single : ∀ (t : T) (i : I) (a : A),
    get t i == a → (∀ j : I, j =/= i → get t j == 0) → sum_below t card == a.
  Proof.
    intros t i a Hi Hj.
    destruct (sum_below_single t i a Hi Hj card (le_refl card)) as [[_ Hsum] | [Hnlt _]].
    - exact Hsum.
    - destruct (Hnlt (index_lt i)).
  Qed.

End FiniteSum.
