From Riktig.Structure Require Export Peano Semiring.

The natural numbers: a commutative semiring on the Peano numbers in which the successor adds one.
Module Naturals.
  Class Naturals (N : Type) {eq : EqRel N} (zero : Zero N) (succ : HasSucc N)
      (one : One N) (add : HasAdd N) (mult : HasMult N) := {
    peano :: Peano N 0 succ;
    semiring :: CommSemiring N 0 add 1 mult;
    succ_add : ∀ n : N, succ n == n + 1;
  }.
  Hint Mode Naturals ! - - - - - - : typeclass_instances.
End Naturals.
Export Naturals.
