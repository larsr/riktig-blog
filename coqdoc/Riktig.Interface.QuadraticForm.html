Library Riktig.Interface.QuadraticForm

From Riktig.Interface Require Export VectorSpace Linear.

Definition polar {F V : Type} `{VectorSpace F V} (Q : V → F) (u v : V) : F :=
  Q (u + v) + - Q u + - Q v.

Module QuadraticForm.
  Class QuadraticForm (F V : Type) `{VectorSpace F V} (Q : V → F) := {
    quad_proper :: Proper (equ ==> equ) Q;
    quad_scale :: Homogeneous F V F mult (fun a ⇒ a × a) Q;
    polar_linear :: ∀ v : V, Linear F V F (fun u ⇒ polar Q u v);
  }.
  Hint Mode QuadraticForm - ! - - - - - - - - - - - - - - ! : typeclass_instances.
End QuadraticForm.
Export QuadraticForm.
