From Riktig.Structure Require Export Products.

Exponentials: an object of arrows with an evaluation, through which every arrow out of a product factors uniquely by currying.
Module Exponentials.
  Class Exponentials (O : Type) `{Products O} (exponential : HasExp O)
      (evaluation : HasEval O) (currying : HasCurry O) := {
    curry_proper :: ∀ x a b : O, Proper (equ ==> equ) (@curry O _ _ _ currying x a b);
    eval_curry : ∀ (x a b : O) (h : x ⨯ a ~> b),
      eval ∘ lift (curry h ∘ pi1) pi2 == h;
    curry_unique : ∀ (x a b : O) (h : x ⨯ a ~> b) (k : x ~> a ⟹ b),
      eval ∘ lift (k ∘ pi1) pi2 == h → k == curry h;
  }.
  Hint Mode Exponentials ! - - - - - - - - - - - - - : typeclass_instances.
End Exponentials.
Export Exponentials.
