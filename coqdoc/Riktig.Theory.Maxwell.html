Library Riktig.Theory.Maxwell

From Coq Require Import Ring.
From Coq.Vectors Require Fin.
From Riktig Require Import Impl.Clifford.

Section Maxwell.
  Context {F : Type} `{FF : Field F}.
  Add Ring maxwell_ring : (ring_theory_of_Ring (R := F) commutative).

  #[local] Instance Spacetime_Mult : HasMult (Cl F 4) := cl_mul sig13.

  Definition fb (b0 b1 b2 b3 : bool) : Blade 4 :=
    bcons b0 (bcons b1 (bcons b2 (bcons b3 bnil))).

  Lemma blade4_case : ∀ D : Blade 4,
    D = fb (bhead D) (bhead (btail D)) (bhead (btail (btail D)))
           (bhead (btail (btail (btail D)))).
  Proof.
    intros D.
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    exact (blade_0 (btail (btail (btail (btail D))))).
  Qed.

  Lemma cl4_ext : ∀ x y : Cl F 4,
    x (fb false false false false) == y (fb false false false false) →
    x (fb true false false false) == y (fb true false false false) →
    x (fb false true false false) == y (fb false true false false) →
    x (fb false false true false) == y (fb false false true false) →
    x (fb false false false true) == y (fb false false false true) →
    x (fb true true false false) == y (fb true true false false) →
    x (fb true false true false) == y (fb true false true false) →
    x (fb true false false true) == y (fb true false false true) →
    x (fb false true true false) == y (fb false true true false) →
    x (fb false true false true) == y (fb false true false true) →
    x (fb false false true true) == y (fb false false true true) →
    x (fb true true true false) == y (fb true true true false) →
    x (fb true true false true) == y (fb true true false true) →
    x (fb true false true true) == y (fb true false true true) →
    x (fb false true true true) == y (fb false true true true) →
    x (fb true true true true) == y (fb true true true true) →
    x == y.
  Proof.
    intros x y H0 H1 H2 H3 H4 H5 H6 H7 H8 H9 H10 H11 H12 H13 H14 H15 D.
    rewrite (blade4_case D).
    destruct (bhead D), (bhead (btail D)), (bhead (btail (btail D))),
             (bhead (btail (btail (btail D))));
      assumption.
  Qed.

  Context (E1 E2 E3 B1 B2 B3 : F).
  Context (d0 d1 d2 d3 : F).

  Definition F_field : Cl F 4 :=
    fun D ⇒
      match bhead D, bhead (btail D), bhead (btail (btail D)),
            bhead (btail (btail (btail D))) with
      | true, true, false, false ⇒ E1
      | true, false, true, false ⇒ E2
      | true, false, false, true ⇒ E3
      | false, true, true, false ⇒ - B3
      | false, true, false, true ⇒ B2
      | false, false, true, true ⇒ - B1
      | _, _, _, _ ⇒ 0
      end.

  Definition spacetime_vector (v0 v1 v2 v3 : F) : Fin.t 4 → F :=
    fun i ⇒ match i with
             | Fin.F1 ⇒ v0
             | Fin.FS Fin.F1 ⇒ v1
             | Fin.FS (Fin.FS Fin.F1) ⇒ v2
             | Fin.FS (Fin.FS (Fin.FS _)) ⇒ v3
             end.

  Definition Nabla : Cl F 4 := ga_vec (spacetime_vector d0 d1 d2 d3).

  Ltac blade_reduce :=
    unfold Nabla, F_field, spacetime_vector;
    repeat change (@mult (Cl F 4) Spacetime_Mult) with Spacetime_Mult;
    unfold Spacetime_Mult, cl_mul, sig13, bsign, parity_sign, fb, bxor,
      bdiff, band;
    simpl.
  Ltac by_blade := blade_reduce; ring.

  Theorem F_nabla_grade_decomposition :
    (F_field × Nabla) (fb false false false false) == 0 ∧
    (F_field × Nabla) (fb true true false false) == 0 ∧
    (F_field × Nabla) (fb true false true false) == 0 ∧
    (F_field × Nabla) (fb true false false true) == 0 ∧
    (F_field × Nabla) (fb false true true false) == 0 ∧
    (F_field × Nabla) (fb false true false true) == 0 ∧
    (F_field × Nabla) (fb false false true true) == 0 ∧
    (F_field × Nabla) (fb true true true true) == 0.
  Proof. repeat split; by_blade. Qed.

  Theorem F_nabla_0 :
    (F_field × Nabla) (fb true false false false)
    == d1 × E1 + (d2 × E2 + d3 × E3).
  Proof. by_blade. Qed.

  Theorem F_nabla_1 :
    (F_field × Nabla) (fb false true false false)
    == d0 × E1 + (- (d2 × B3) + d3 × B2).
  Proof. by_blade. Qed.

  Theorem F_nabla_2 :
    (F_field × Nabla) (fb false false true false)
    == d0 × E2 + (d1 × B3 + - (d3 × B1)).
  Proof. by_blade. Qed.

  Theorem F_nabla_3 :
    (F_field × Nabla) (fb false false false true)
    == d0 × E3 + (- (d1 × B2) + d2 × B1).
  Proof. by_blade. Qed.

  Theorem F_nabla_123 :
    (F_field × Nabla) (fb false true true true)
    == - (d1 × B1) + (- (d2 × B2) + - (d3 × B3)).
  Proof. by_blade. Qed.

  Theorem F_nabla_012 :
    (F_field × Nabla) (fb true true true false)
    == - (d0 × B3) + (- (d1 × E2) + d2 × E1).
  Proof. by_blade. Qed.

  Theorem F_nabla_013 :
    (F_field × Nabla) (fb true true false true)
    == d0 × B2 + (- (d1 × E3) + d3 × E1).
  Proof. by_blade. Qed.

  Theorem F_nabla_023 :
    (F_field × Nabla) (fb true false true true)
    == - (d0 × B1) + (- (d2 × E3) + d3 × E2).
  Proof. by_blade. Qed.

  Definition J : Cl F 4 :=
    ga_vec (spacetime_vector
      (d1 × E1 + (d2 × E2 + d3 × E3))
      (d0 × E1 + (- (d2 × B3) + d3 × B2))
      (d0 × E2 + (d1 × B3 + - (d3 × B1)))
      (d0 × E3 + (- (d1 × B2) + d2 × B1))).

  Theorem F_nabla_eq_J :
    (F_field × Nabla) (fb false true true true) == 0 →
    (F_field × Nabla) (fb true true true false) == 0 →
    (F_field × Nabla) (fb true true false true) == 0 →
    (F_field × Nabla) (fb true false true true) == 0 →
    F_field × Nabla == J.
  Proof.
    intros H123 H012 H013 H023.
    destruct F_nabla_grade_decomposition
      as (Hs & H01 & H02 & H03 & H12 & H13 & H23 & H0123).
    apply cl4_ext.
    - exact Hs.
    - exact F_nabla_0.
    - exact F_nabla_1.
    - exact F_nabla_2.
    - exact F_nabla_3.
    - exact H01.
    - exact H02.
    - exact H03.
    - exact H12.
    - exact H13.
    - exact H23.
    - exact H012.
    - exact H013.
    - exact H023.
    - exact H123.
    - exact H0123.
  Qed.

End Maxwell.
