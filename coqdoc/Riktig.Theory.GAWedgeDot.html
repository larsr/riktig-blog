Section FieldHalving.
  Context {F : Type} `{OrderedField F}.

  Lemma half_one_double : ∀ x : F, half 1 × (x + x) == x.
  Proof.
    intros x.
    calc (half 1 × (x + x))
      == (half 1 × x + half 1 × x) { left_distributive }
      == ((half 1 + half 1) × x) { right_distributive }
      == (1 × x) { half_sum }
      == x { left_identity }.
  Qed.

End FieldHalving.

Section ScaleHalving.
  Context {F M : Type} `{OrderedField F}.
  Context `{MG : AbGroup M} {MS : HasScale F M} {ML : ScaleLaws F M}.

  Lemma sc_half_double : ∀ x : M, half 1 *: (x + x) == x.
  Proof.
    intros x.
    calc (half 1 *: (x + x))
      == (half 1 *: x + half 1 *: x) { left_distributive }
      == ((half 1 + half 1) *: x) { right_distributive }
      == (1 *: x) { half_sum }
      == x { left_identity }.
  Qed.

End ScaleHalving.


Section CleanDecomposition.
  Context {F V A : Type} `{OrderedField F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {iV : HasInner F V}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {oA : One A} {mA : HasMult A}
    {sA : HasScale F A}.
  Context {VS : VectorSpace F V} {IP : InnerProductSpace F V}
          {GA : GeometricAlgebra F V A (Q := norm2)}.

  Lemma ga_symmetric_part_inner : ∀ u v : V,
    ga_vec u × ga_vec v + ga_vec v × ga_vec u == (⟪ u , v ⟫ + ⟪ u , v ⟫) *: (1 : A).
  Proof.
    intros u v.
    rewrite (ga_symmetric_part u v).
    rewrite (polar_norm2 u v).
    reflexivity.
  Qed.

  Theorem ga_dot_from_product : ∀ u v : V,
    ⟪ u , v ⟫ *: (1 : A)
    == half 1 *: (ga_vec u × ga_vec v + ga_vec v × ga_vec u).
  Proof.
    intros u v.
    calc ((⟪ u , v ⟫ *: (1 : A)) : A)
      == (((half 1 × (⟪ u , v ⟫ + ⟪ u , v ⟫)) *: (1 : A)) : A)
        { half_one_double }
      == ((half 1 *: ((⟪ u , v ⟫ + ⟪ u , v ⟫) *: (1 : A))) : A)
        { associative }
      == ((half 1 *: (ga_vec u × ga_vec v + ga_vec v × ga_vec u)) : A)
        { ga_symmetric_part_inner }.
  Qed.

  Corollary ga_symmetric_part_clean : ∀ u v : V,
    half 1 *: (ga_vec u × ga_vec v + ga_vec v × ga_vec u)
    == ⟪ u , v ⟫ *: (1 : A).
  Proof.
    intros u v.
    symmetry.
    exact (ga_dot_from_product u v).
  Qed.

  Theorem ga_product_decompose : ∀ u v : V,
    ga_vec u × ga_vec v
    == ⟪ u , v ⟫ *: (1 : A) + half 1 *: wedge (ga_vec u) (ga_vec v).
  Proof.
    intros u v.
    calc ((ga_vec u × ga_vec v) : A)
      == ((half 1 *: (ga_vec u × ga_vec v + ga_vec u × ga_vec v)) : A)
        { sc_half_double }
      ==[hard] ((half 1 *: ((ga_vec u × ga_vec v + ga_vec v × ga_vec u)
            + (ga_vec u × ga_vec v + - (ga_vec v × ga_vec u)))) : A)
      == ((half 1 *: (ga_vec u × ga_vec v + ga_vec v × ga_vec u)
            + half 1 *: (ga_vec u × ga_vec v + - (ga_vec v × ga_vec u))) : A)
        { left_distributive }
      == ((⟪ u , v ⟫ *: (1 : A) + half 1 *: wedge (ga_vec u) (ga_vec v)) : A)
        { ga_symmetric_part_clean }.
    [hard]: calcrw ((ga_vec u × ga_vec v + ga_vec u × ga_vec v) : A)
      == (((ga_vec u × ga_vec v + ga_vec u × ga_vec v) + 0) : A)
        { right_identity }
      == (((ga_vec u × ga_vec v + ga_vec u × ga_vec v)
            + (ga_vec v × ga_vec u + - (ga_vec v × ga_vec u))) : A)
        { right_inverse }
      == (((ga_vec u × ga_vec v + ga_vec v × ga_vec u)
            + (ga_vec u × ga_vec v + - (ga_vec v × ga_vec u))) : A)
        { add_interchange }.
  Qed.

  Corollary ga_antisymmetric_part_clean : ∀ u v : V,
    half 1 *: wedge (ga_vec u) (ga_vec v)
    == ga_vec u × ga_vec v + - (⟪ u , v ⟫ *: (1 : A)).
  Proof.
    intros u v.
    calc ((half 1 *: wedge (ga_vec u) (ga_vec v)) : A)
      == ((0 + half 1 *: wedge (ga_vec u) (ga_vec v)) : A)
        { left_identity }
      == (((- (⟪ u , v ⟫ *: (1 : A)) + ⟪ u , v ⟫ *: (1 : A))
            + half 1 *: wedge (ga_vec u) (ga_vec v)) : A)
        { left_inverse }
      == ((- (⟪ u , v ⟫ *: (1 : A))
            + (⟪ u , v ⟫ *: (1 : A) + half 1 *: wedge (ga_vec u) (ga_vec v))) : A)
        { associative }
      == ((- (⟪ u , v ⟫ *: (1 : A)) + ga_vec u × ga_vec v) : A)
        { ga_product_decompose }
      == ((ga_vec u × ga_vec v + - (⟪ u , v ⟫ *: (1 : A))) : A)
        { commutative }.
  Qed.

End CleanDecomposition.



Section WedgeScaling.
  Context {F A : Type} `{Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A}
    {oA : One A} {mA : HasMult A} {sA : HasScale F A}.
  Context {RA : Ring A} {SL : ScaleLaws F A}.
  Context {mulL : ∀ y : A, Linear F A A (fun x ⇒ x × y)}
          {mulR : ∀ x : A, Linear F A A (fun y ⇒ x × y)}.

  Theorem wedge_scale_l : ∀ (a : F) (x y : A),
    wedge (a *: x) y == a *: wedge x y.
  Proof.
    intros a x y.
    unfold wedge.
    calc (((a *: x) × y + - (y × (a *: x))) : A)
      == ((a *: (x × y) + - (y × (a *: x))) : A)
        { homogeneous[fun z ⇒ z × y] }
      == ((a *: (x × y) + - (a *: (y × x))) : A)
        { homogeneous[fun z ⇒ y × z] }
      == ((a *: (x × y) + a *: (- (y × x))) : A)
        { scale_neg }
      == ((a *: (x × y + - (y × x))) : A)
        { left_distributive }.
  Qed.

  Theorem wedge_scale_r : ∀ (a : F) (x y : A),
    wedge x (a *: y) == a *: wedge x y.
  Proof.
    intros a x y.
    unfold wedge.
    calc ((x × (a *: y) + - ((a *: y) × x)) : A)
      == ((a *: (x × y) + - ((a *: y) × x)) : A)
        { homogeneous[fun z ⇒ x × z] }
      == ((a *: (x × y) + - (a *: (y × x))) : A)
        { homogeneous[fun z ⇒ z × x] }
      == ((a *: (x × y) + a *: (- (y × x))) : A)
        { scale_neg }
      == ((a *: (x × y + - (y × x))) : A)
        { left_distributive }.
  Qed.

  #[export] Instance wedge_l_homogeneous (y : A) :
    Homogeneous F A A scale (fun a ⇒ a) (fun x ⇒ wedge x y) :=
    fun a x ⇒ wedge_scale_l a x y.

  #[export] Instance wedge_r_homogeneous (x : A) :
    Homogeneous F A A scale (fun a ⇒ a) (wedge x) :=
    fun a y ⇒ wedge_scale_r a x y.

End WedgeScaling.

Class WedgeDotPresentation (F V A : Type) `{InnerProductSpace F V}
    {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A} {Ao : One A} {Am : HasMult A} := {
  wd_ring :: Ring A;
  wd_scale :: HasScale F A;
  wd_scale_laws :: ScaleLaws F A;
  wd_scale_mul_l : ∀ (a : F) (x y : A), (a *: x) × y == a *: (x × y);
  wd_scale_mul_r : ∀ (a : F) (x y : A), x × (a *: y) == a *: (x × y);
  wd_vec : V → A;
  wd_vec_proper :: Proper (equ ==> equ) wd_vec;
  wd_vec_add : ∀ u v : V, wd_vec (u + v) == wd_vec u + wd_vec v;
  wd_vec_scale : ∀ (a : F) (u : V), wd_vec (a *: u) == a *: wd_vec u;
  wd_w : V → V → A;
  wd_w_proper :: Proper (equ ==> equ ==> equ) wd_w;
  wd_w_add_l : ∀ u u' v : V, wd_w (u + u') v == wd_w u v + wd_w u' v;
  wd_w_scale_l : ∀ (a : F) (u v : V), wd_w (a *: u) v == a *: wd_w u v;
  wd_w_add_r : ∀ u v v' : V, wd_w u (v + v') == wd_w u v + wd_w u v';
  wd_w_scale_r : ∀ (a : F) (u v : V), wd_w u (a *: v) == a *: wd_w u v;
  wd_w_alt : ∀ u : V, wd_w u u == 0;
  wd_decomp : ∀ u v : V,
                     wd_vec u × wd_vec v == ⟪ u , v ⟫ *: 1 + wd_w u v
}.

Section WedgeDotTheory.
  Context {F V A : Type} `{WedgeDotPresentation F V A}.

  Theorem wd_w_anti : ∀ u v : V, (wd_w u v : A) == - wd_w v u.
  Proof.
    intros u v.
    apply neg_unique_l.
    calc ((wd_w u v + wd_w v u) : A)
      ==[pad] (((wd_w u u + wd_w u v) + (wd_w v u + wd_w v v)) : A)
      ==[gather] ((wd_w (u + v) (u + v)) : A)
      ==[alt] (0 : A).
    [pad]: now rewrite (wd_w_alt u), (wd_w_alt v), left_identity, right_identity.
    [gather]: now rewrite (wd_w_add_l u v (u + v)), (wd_w_add_r u u v),
                (wd_w_add_r v u v).
    [alt]: exact (wd_w_alt (u + v)).
  Qed.

  Theorem wedgedot_to_contract : ∀ u : V,
    wd_vec u × wd_vec u == norm2 u *: (1 : A).
  Proof.
    intros u.
    calc ((wd_vec u × wd_vec u) : A)
      == ((⟪ u , u ⟫ *: (1 : A) + wd_w u u) : A) { wd_decomp }
      == ((⟪ u , u ⟫ *: (1 : A) + 0) : A) { wd_w_alt }
      == ((norm2 u *: (1 : A)) : A) { right_identity }.
  Qed.

  Definition wedgedot_to_ga : GeometricAlgebra F V A (Q := norm2).
  Proof.
    unshelve econstructor.
    - exact wd_vec.
    - exact wd_ring.
    - exact wd_scale_laws.
    - intros y. unshelve econstructor.
      + intros x x' Hx. now rewrite Hx.
      + intros x x'. exact (right_distributive x x' y).
      + intros a x. exact (wd_scale_mul_l a x y).
    - intros x. unshelve econstructor.
      + intros y y' Hy. now rewrite Hy.
      + intros y y'. exact (left_distributive x y y').
      + intros a y. exact (wd_scale_mul_r a x y).
    - unshelve econstructor.
      + exact wd_vec_proper.
      + exact wd_vec_add.
      + exact wd_vec_scale.
    - exact wedgedot_to_contract.
  Defined.

End WedgeDotTheory.


Section ContractToWedgeDot.
  Context {F V A : Type} `{OrderedField F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {iV : HasInner F V}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {oA : One A} {mA : HasMult A}
    {sA : HasScale F A}.
  Context {VS : VectorSpace F V} {IP : InnerProductSpace F V}
          {GA : GeometricAlgebra F V A (Q := norm2)}.

  Definition half_wedge (u v : V) : A := half 1 *: wedge (ga_vec u) (ga_vec v).

  #[export] Instance half_wedge_proper :
    Proper (equ ==> equ ==> equ) half_wedge.
  Proof.
    intros u u' Hu v v' Hv.
    unfold half_wedge.
    now rewrite Hu, Hv.
  Qed.

  Lemma half_wedge_add_l : ∀ u u' v : V,
    half_wedge (u + u') v == half_wedge u v + half_wedge u' v.
  Proof.
    intros u u' v.
    unfold half_wedge.
    calc ((half 1 *: wedge (ga_vec (u + u')) (ga_vec v)) : A)
      == ((half 1 *: wedge (ga_vec u + ga_vec u') (ga_vec v)) : A)
        { additive[ga_vec] }
      == ((half 1 *: (wedge (ga_vec u) (ga_vec v) + wedge (ga_vec u') (ga_vec v))) : A)
        { additive[fun z ⇒ wedge z (ga_vec v)] }
      == ((half 1 *: wedge (ga_vec u) (ga_vec v)
            + half 1 *: wedge (ga_vec u') (ga_vec v)) : A)
        { left_distributive }.
  Qed.

  Lemma half_wedge_scale_l : ∀ (a : F) (u v : V),
    half_wedge (a *: u) v == a *: half_wedge u v.
  Proof.
    intros a u v.
    unfold half_wedge.
    calc ((half 1 *: wedge (ga_vec (a *: u)) (ga_vec v)) : A)
      == ((half 1 *: wedge (a *: ga_vec u) (ga_vec v)) : A)
        { homogeneous[ga_vec] }
      == ((half 1 *: (a *: wedge (ga_vec u) (ga_vec v))) : A)
        { homogeneous[fun z ⇒ wedge z (ga_vec v)] }
      == ((a *: (half 1 *: wedge (ga_vec u) (ga_vec v))) : A)
        { scale_swap }.
  Qed.

  Lemma half_wedge_add_r : ∀ u v v' : V,
    half_wedge u (v + v') == half_wedge u v + half_wedge u v'.
  Proof.
    intros u v v'.
    unfold half_wedge.
    calc ((half 1 *: wedge (ga_vec u) (ga_vec (v + v'))) : A)
      == ((half 1 *: wedge (ga_vec u) (ga_vec v + ga_vec v')) : A)
        { additive[ga_vec] }
      == ((half 1 *: (wedge (ga_vec u) (ga_vec v) + wedge (ga_vec u) (ga_vec v'))) : A)
        { additive[wedge (ga_vec u)] }
      == ((half 1 *: wedge (ga_vec u) (ga_vec v)
            + half 1 *: wedge (ga_vec u) (ga_vec v')) : A)
        { left_distributive }.
  Qed.

  Lemma half_wedge_scale_r : ∀ (a : F) (u v : V),
    half_wedge u (a *: v) == a *: half_wedge u v.
  Proof.
    intros a u v.
    unfold half_wedge.
    calc ((half 1 *: wedge (ga_vec u) (ga_vec (a *: v))) : A)
      == ((half 1 *: wedge (ga_vec u) (a *: ga_vec v)) : A)
        { homogeneous[ga_vec] }
      == ((half 1 *: (a *: wedge (ga_vec u) (ga_vec v))) : A)
        { homogeneous[wedge (ga_vec u)] }
      == ((a *: (half 1 *: wedge (ga_vec u) (ga_vec v))) : A)
        { scale_swap }.
  Qed.

  Lemma half_wedge_alt : ∀ u : V, half_wedge u u == 0.
  Proof.
    intros u.
    unfold half_wedge.
    calc ((half 1 *: wedge (ga_vec u) (ga_vec u)) : A)
      == ((half 1 *: (0 : A)) : A) { wedge_self_zero }
      == (0 : A) { scale_zero }.
  Qed.

  Lemma half_wedge_decomp : ∀ u v : V,
    ga_vec u × ga_vec v == ⟪ u , v ⟫ *: (1 : A) + half_wedge u v.
  Proof.
    intros u v.
    exact (ga_product_decompose u v).
  Qed.

  Definition contract_to_wedgedot : WedgeDotPresentation F V A :=
    {| wd_ring := GeometricAlgebra.ring;
       wd_scale := sA;
       wd_scale_laws := GeometricAlgebra.scaling;
       wd_scale_mul_l := (fun a x y ⇒ homogeneous[fun z ⇒ z × y] a x);
       wd_scale_mul_r := (fun a x y ⇒ homogeneous[fun z ⇒ x × z] a y);
       wd_vec := ga_vec;
       wd_vec_proper := (Linear.proper (f := ga_vec));
       wd_vec_add := additive[ga_vec];
       wd_vec_scale := homogeneous[ga_vec];
       wd_w := half_wedge;
       wd_w_proper := half_wedge_proper;
       wd_w_add_l := half_wedge_add_l;
       wd_w_scale_l := half_wedge_scale_l;
       wd_w_add_r := half_wedge_add_r;
       wd_w_scale_r := half_wedge_scale_r;
       wd_w_alt := half_wedge_alt;
       wd_decomp := half_wedge_decomp |}.

End ContractToWedgeDot.
