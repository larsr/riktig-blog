Library Riktig.Theory.MatVec

From Coq Require Export Morphisms.
From Coq.Vectors Require Fin.
From Riktig Require Export Theory.BigOp.
From Riktig Require Import Impl.BigopFold.
From Riktig Require Export Theory.LinearMap Theory.FieldVS Impl.Matrix.

Section MatVecOps.
  Context {F : Type} `{FF : Field F}.

  Definition mat_vec {m n : nat} (A : Matrix m n F) (v : Fin.t n → F)
      : Fin.t m → F :=
    fun i ⇒ bigopI (fun j ⇒ A i j × v j).

  Lemma mat_vec_proper : ∀ (m n : nat),
    Proper (equ ==> equ ==> equ) (@mat_vec m n).
  Proof.
    intros m n A A' HA v v' Hv i.
    exact (bigopI_ext (fun j ⇒ A i j × v j) (fun j ⇒ A' i j × v' j)
             (fun j ⇒ op_proper (A i j) (A' i j) (HA i j)
                         (v j) (v' j) (Hv j))).
  Qed.

  Lemma mat_vec_proper_v : ∀ (m n : nat) (A : Matrix m n F),
    Proper (equ ==> equ) (mat_vec A).
  Proof.
    intros m n A v v' Hv i.
    exact (bigopI_ext (fun j ⇒ A i j × v j) (fun j ⇒ A i j × v' j)
             (fun j ⇒ op_proper (A i j) (A i j) (reflexivity (A i j))
                         (v j) (v' j) (Hv j))).
  Qed.

  Lemma mat_vec_add : ∀ (m n : nat) (A : Matrix m n F)
      (v w : Fin.t n → F),
    mat_vec A (v + w) == mat_vec A v + mat_vec A w.
  Proof.
    intros m n A v w i.
    unfold mat_vec.
    calc (bigopI (fun j ⇒ A i j × (v + w) j))
      == (bigopI (fun j ⇒ A i j × v j + A i j × w j))
      == ((mat_vec A v + mat_vec A w) i).
    - apply bigopI_ext. intros j.
      exact (left_distributive (A i j) (v j) (w j)).
    - exact (bigopI_add (fun j ⇒ A i j × v j) (fun j ⇒ A i j × w j)).
  Qed.

  Lemma mat_vec_scale : ∀ (m n : nat) (A : Matrix m n F) (a : F)
      (v : Fin.t n → F),
    mat_vec A (a *: v) == a *: mat_vec A v.
  Proof.
    intros m n A a v i.
    unfold mat_vec.
    calc (bigopI (fun j ⇒ A i j × (a *: v) j))
      == (bigopI (fun j ⇒ a × (A i j × v j)))
      == ((a *: mat_vec A v) i).
    - apply bigopI_ext. intros j.
      exact (mul_left_commutative (A i j) a (v j)).
    - exact (bigopI_mul_l a (fun j ⇒ A i j × v j)).
  Qed.

  Definition mat_vec_lm {m n : nat} (A : Matrix m n F) :
      LinearMap F (Fin.t n → F) (Fin.t m → F) :=
    linear_map_from_laws
      (mat_vec A)
      (mat_vec_proper_v m n A)
      (mat_vec_add m n A)
      (mat_vec_scale m n A).

  Theorem mat_vec_mul : ∀ (m n p : nat) (A : Matrix m n F)
      (B : Matrix n p F) (v : Fin.t p → F),
    mat_vec (mat_mul A B) v == mat_vec A (mat_vec B v).
  Proof.
    intros m n p A B v i.
    exact (mat_mul_assoc A B (fun (k : Fin.t p) (_ : Fin.t 1) ⇒ v k) i Fin.F1).
  Qed.

  Theorem mat_vec_lm_mul : ∀ (m n p : nat) (A : Matrix m n F)
      (B : Matrix n p F),
    mat_vec_lm (mat_mul A B) == linear_map_comp (mat_vec_lm A) (mat_vec_lm B).
  Proof.
    intros m n p A B v.
    exact (mat_vec_mul m n p A B v).
  Qed.

  Theorem mat_vec_id : ∀ (n : nat) (v : Fin.t n → F),
    mat_vec mat_id v == v.
  Proof.
    intros n v i.
    exact (mat_mul_left_identity (fun (k : Fin.t n) (_ : Fin.t 1) ⇒ v k)
             (leibniz_proper_first (fun (k : Fin.t n) (_ : Fin.t 1) ⇒ v k)) i Fin.F1).
  Qed.

  Definition mat_transpose {m n : nat} (A : Matrix m n F) : Matrix n m F :=
    fun i j ⇒ A j i.

  Lemma mat_transpose_proper : ∀ (m n : nat),
    Proper (equ ==> equ) (@mat_transpose m n).
  Proof.
    intros m n A A' HA i j.
    exact (HA j i).
  Qed.

End MatVecOps.
