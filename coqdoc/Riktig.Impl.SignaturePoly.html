Library Riktig.Impl.SignaturePoly

From Riktig.Structure Require Export Signature.
From Riktig.Lib Require Export Coq.Poly.

The stock signatures: the syntax tree, folded by its recursion. The equations are the fold's own, and its uniqueness is the tree's induction.
Instance Poly_Zero {K : Type} : Zero (poly K) := PZero.
Instance Poly_One {K : Type} : One (poly K) := POne.
Instance Poly_Self {K : Type} : HasSelf (poly K) := PSelf.
Instance Poly_Const {K : Type} : HasConst K (poly K) := @PConst K.
Instance Poly_Add {K : Type} : HasAdd (poly K) := @PPlus K.
Instance Poly_Mult {K : Type} : HasMult (poly K) := @PTimes K.
Instance Poly_Rec {K : Type} : HasPolyRec K (poly K) :=
  fun B z o s c p t q ⇒ poly_fold z o s c p t q.

Instance Poly_Signature {K : Type} : Signature K (poly K) _ _ _ _ _ _ _ :=
  {| Signature.rec_zero := fun B z o s c p t ⇒ eq_refl;
     Signature.rec_one := fun B z o s c p t ⇒ eq_refl;
     Signature.rec_self := fun B z o s c p t ⇒ eq_refl;
     Signature.rec_const := fun B z o s c p t k ⇒ eq_refl;
     Signature.rec_plus := fun B z o s c p t a b ⇒ eq_refl;
     Signature.rec_times := fun B z o s c p t a b ⇒ eq_refl;
     Signature.rec_unique := fun B z o s c p t u ⇒ poly_fold_unique K B z o s c p t u |}.
