Section FibonacciMatrix.
  Context {F : Type} `{FF : Field F}.

  Add Ring fib_matrix_ring : (ring_theory_of_Ring (R := F) commutative).

  Fixpoint fib (n : nat) : F :=
    match n with
    | O ⇒ 0
    | S m ⇒ match m with
             | O ⇒ 1
             | S k ⇒ fib m + fib k
             end
    end.

  Definition fib_matrix : Matrix 2 2 F := mat2 1 1 1 0.

Multiplying by the Fibonacci matrix shifts a column pair one step along the recurrence. This is the only matrix product written out in the file; mat_mul on Fin.t 2 is a two-term sum, so cbn exposes it and the field closes it.
  Lemma fib_matrix_mul : ∀ a b c d : F,
    fib_matrix × mat2 a b c d == mat2 (a + c) (b + d) a b.
  Proof.
    intros a b c d i k; revert k; revert i.
    refine (fin2_case _ _ _); [ | intros i1 ]; refine (fin2_case _ _ _);
      intros; cbn; ring.
  Qed.

  Lemma fib_matrix_is_first_power :
    fib_matrix == mat2 (fib 2) (fib 1) (fib 1) (fib 0).
  Proof.
    intros i k; revert k; revert i.
    refine (fin2_case _ _ _); [ | intros i1 ]; refine (fin2_case _ _ _);
      intros; cbn; ring.
  Qed.

The powers of the Fibonacci matrix are the Fibonacci numbers.
  Theorem fib_matrix_power : ∀ n : nat,
    pow_nat fib_matrix (S n)
      == mat2 (fib (S (S n))) (fib (S n)) (fib (S n)) (fib n).
  Proof.
    intros n; induction n as [| n IH].
    - calc (pow_nat fib_matrix 1 : Matrix 2 2 F)
        == (fib_matrix : Matrix 2 2 F)
             { (right_identity fib_matrix
                  : fib_matrix × (1 : Matrix 2 2 F) == fib_matrix) }
        == (mat2 (fib 2) (fib 1) (fib 1) (fib 0)) { fib_matrix_is_first_power }.
    - calc (pow_nat fib_matrix (S (S n)) : Matrix 2 2 F)
        == (fib_matrix
              × mat2 (fib (S (S n))) (fib (S n)) (fib (S n)) (fib n))
             { (op_proper fib_matrix fib_matrix (reflexivity fib_matrix)
                  _ _ IH) }
        == (mat2 (fib (S (S n)) + fib (S n)) (fib (S n) + fib n)
                 (fib (S (S n))) (fib (S n)))
             { (fib_matrix_mul (fib (S (S n))) (fib (S n))
                  (fib (S n)) (fib n)) }.
  Qed.

What the program returns: the corner the caller reads is the Fibonacci number of the exponent.
  Corollary fib_by_squaring : ∀ p : positive,
    pow_positive fib_matrix p Fin.F1 (Fin.FS Fin.F1) == fib (Pos.to_nat p).
  Proof.
    intros p.
    destruct (Pos2Nat.is_succ p) as [n Hn].
    calc (pow_positive fib_matrix p Fin.F1 (Fin.FS Fin.F1) : F)
      == (pow_nat fib_matrix (Pos.to_nat p) Fin.F1 (Fin.FS Fin.F1))
           { (pow_positive_is_pow_nat fib_matrix p Fin.F1 (Fin.FS Fin.F1)) }
      == (pow_nat fib_matrix (S n) Fin.F1 (Fin.FS Fin.F1))
           { (pow_exponent pow_nat fib_matrix _ _ Hn Fin.F1 (Fin.FS Fin.F1)) }
      == (fib (S n))
           { (fib_matrix_power n Fin.F1 (Fin.FS Fin.F1)) }
      == (fib (Pos.to_nat p))
           { (f_equal fib (eq_sym Hn)) }.
  Qed.

End FibonacciMatrix.


Definition fibQ_matrix : Matrix 2 2 Q := fib_matrix.

Example fib_matrix_squared :
  pow_positive fibQ_matrix 2 == mat2 (2#1)%Q (1#1)%Q (1#1)%Q (1#1)%Q.
Proof.
  intros i k; revert k; revert i.
  refine (fin2_case _ _ _); [ | intros i1 ]; refine (fin2_case _ _ _);
    intros; vm_compute; reflexivity.
Qed.

Example fib_matrix_to_the_thirtieth :
  pow_positive fibQ_matrix 30
    == mat2 (1346269#1)%Q (832040#1)%Q (832040#1)%Q (514229#1)%Q.
Proof.
  intros i k; revert k; revert i.
  refine (fin2_case _ _ _); [ | intros i1 ]; refine (fin2_case _ _ _);
    intros; vm_compute; reflexivity.
Qed.

The thirtieth Fibonacci number, by squaring matrices. The number is not checked against a second computation: fib_by_squaring says the corner IS fib 30, and the literal above is what the corner computed to.
Corollary fib_thirty : fib 30 == (832040#1)%Q.
Proof.
  calc (fib 30 : Q)
    == (pow_positive fibQ_matrix 30 Fin.F1 (Fin.FS Fin.F1))
         { (symmetry (fib_by_squaring 30)) }
    == (832040#1)%Q
         { (fib_matrix_to_the_thirtieth Fin.F1 (Fin.FS Fin.F1)) }.
Qed.


Definition fib_matrix_ix : Ix 1 → Ix 1 → Q :=
  ix2 (1#1)%Q (1#1)%Q (1#1)%Q (0#1)%Q.

Example fib_square_divided :
  mat_mul_rec 1 fib_matrix_ix fib_matrix_ix
    == ix2 (2#1)%Q (1#1)%Q (1#1)%Q (1#1)%Q.
Proof.
  intros i j.
  destruct i as [[] | []]; destruct j as [[] | []]; vm_compute; reflexivity.
Qed.

Example fib_square_contracted :
  mat_mul fib_matrix_ix fib_matrix_ix
    == ix2 (2#1)%Q (1#1)%Q (1#1)%Q (1#1)%Q.
Proof.
  intros i j.
  calc (mat_mul fib_matrix_ix fib_matrix_ix i j)
    == (mat_mul_rec 1 fib_matrix_ix fib_matrix_ix i j)
         { (mat_mul_rec_correct 1 (Ix 1) (Ix 1)
              fib_matrix_ix fib_matrix_ix i j) }
    == (ix2 (2#1)%Q (1#1)%Q (1#1)%Q (1#1)%Q i j)
         { (fib_square_divided i j) }.
Qed.
