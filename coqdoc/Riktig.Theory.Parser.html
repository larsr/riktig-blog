Derived combinators, as notation: reading a function of what is read, reading two things in sequence, keeping one of two readings, and reading one given element.
Notation "f <$> p" := (p >>= fun a ⇒ ret (f a)) (at level 61, left associativity).
Notation "p ⊗ q" := (p >>= fun a ⇒ q >>= fun b ⇒ ret (a, b)) (at level 40, left associativity).
Notation "p *> q" := (p >>= fun _ ⇒ q) (at level 40, left associativity).
Notation "p <* q" := (p >>= fun a ⇒ q >>= fun _ ⇒ ret a) (at level 40, left associativity).
Notation "'sym' c" := (item >>= fun c' ⇒ if equ_dec c' c then ret c' else fail) (at level 10, c at level 9).

What holds of every parser: equality of parsers is an equivalence, return and bind make a monad, failure and choice a commutative idempotent monoid, bind distributes over choice and is absorbed by failure, and the derived combinators read what one expects. The functions fed to bind respect equality throughout.

  Context {C S : Type} {P : Type → Type} `{Parser C S P}.
  Local Instance elements_equivalence : Equivalence (@equ C eqC) := @elements C S eqC eqS stepping Seq.
  Local Instance elements_setoid : Setoid C := {| setoid_equivalence := elements_equivalence |}.

  Lemma equ_of_parses : ∀ `{Setoid A} (p q : P A),
    (∀ (s s' : S) (a : A), parses p s a s' ↔ parses q s a s') → p == q.
  Proof. intros A eqA EqvA p q Hpq. exact (proj2 (parser_equ p q) Hpq). Qed.

  Lemma parses_of_equ : ∀ `{Setoid A} (p q : P A), p == q →
    ∀ (s s' : S) (a : A), parses p s a s' ↔ parses q s a s'.
  Proof. intros A eqA EqvA p q Hpq. exact (proj1 (parser_equ p q) Hpq). Qed.

  Lemma parses_rest : ∀ `{Setoid A} (p : P A) (s : S) (a : A) (s1 s2 : S),
    s1 == s2 → parses p s a s1 → parses p s a s2.
  Proof. intros A eqA EqvA p s a s1 s2 Hs Hp. exact (proj1 (parses_proper p s s (reflexivity s) a a (reflexivity a) s1 s2 Hs) Hp). Qed.

  Lemma parses_input : ∀ `{Setoid A} (p : P A) (s1 s2 : S) (a : A) (s' : S),
    s1 == s2 → parses p s1 a s' → parses p s2 a s'.
  Proof. intros A eqA EqvA p s1 s2 a s' Hs Hp. exact (proj1 (parses_proper p s1 s2 Hs a a (reflexivity a) s' s' (reflexivity s')) Hp). Qed.

  Lemma parses_result : ∀ `{Setoid A} (p : P A) (s : S) (a a' : A) (s' : S),
    a == a' → parses p s a s' → parses p s a' s'.
  Proof. intros A eqA EqvA p s a a' s' Ha Hp. exact (proj1 (parses_proper p s s (reflexivity s) a a' Ha s' s' (reflexivity s')) Hp). Qed.

  #[export] Instance parser_equivalence : ∀ `{Setoid A}, Equivalence (@equ (P A) (eqP A eqA)).
  Proof.
    intros A eqA EqvA. split.
    - intros p. apply equ_of_parses. intros s s' a. reflexivity.
    - intros p q Hpq. apply equ_of_parses. intros s s' a. symmetry. exact (parses_of_equ p q Hpq s s' a).
    - intros p q r Hpq Hqr. apply equ_of_parses. intros s s' a.
      transitivity (parses q s a s'); [ exact (parses_of_equ p q Hpq s s' a) | exact (parses_of_equ q r Hqr s s' a) ].
  Qed.

Return and bind respect equality

  Theorem ret_proper : ∀ `{Setoid A}, Proper (equ ==> equ) (ret : A → P A).
  Proof.
    intros A eqA EqvA a a' Ha. apply equ_of_parses. intros s s' b. rewrite !parses_ret. split.
    - intros [Hb Hs]. split; [ transitivity a; assumption | exact Hs ].
    - intros [Hb Hs]. split; [ transitivity a'; [ exact Hb | symmetry; exact Ha ] | exact Hs ].
  Qed.

  Theorem bind_proper_l : ∀ `{Setoid A} `{Setoid B}
      (p p' : P A) (f : A → P B), Proper (equ ==> equ) f → p == p' → p >>= f == p' >>= f.
  Proof.
    intros A eqA EqvA B eqB EqvB p p' f Hf Hp. apply equ_of_parses. intros s s'' b.
    rewrite !(parses_bind _ _ _ _ _ Hf). split; intros [a [s' [Hr Hfa]]]; ∃ a, s'; split; try assumption.
    - exact (proj1 (parses_of_equ p p' Hp s s' a) Hr).
    - exact (proj2 (parses_of_equ p p' Hp s s' a) Hr).
  Qed.

  Theorem bind_proper_r : ∀ `{Setoid A} `{Setoid B}
      (p : P A) (f g : A → P B), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
      (∀ a : A, f a == g a) → p >>= f == p >>= g.
  Proof.
    intros A eqA EqvA B eqB EqvB p f g Hf Hg Hfg. apply equ_of_parses. intros s s'' b.
    rewrite (parses_bind _ _ _ _ _ Hf), (parses_bind _ _ _ _ _ Hg).
    split; intros [a [s' [Hp Hr]]]; ∃ a, s'; split; try assumption.
    - exact (proj1 (parses_of_equ _ _ (Hfg a) s' s'' b) Hr).
    - exact (proj2 (parses_of_equ _ _ (Hfg a) s' s'' b) Hr).
  Qed.

  Lemma bind_fun_proper : ∀ `{Setoid A} `{Setoid B} `{Setoid D}
      (f : A → P B) (g : B → P D), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
      Proper (equ ==> equ) (fun a ⇒ f a >>= g).
  Proof. intros A eqA EqvA B eqB EqvB D eqD EqvD f g Hf Hg a a' Ha. apply bind_proper_l; [ exact Hg | apply Hf; exact Ha ]. Qed.

The monad laws

  Theorem bind_ret_l : ∀ `{Setoid A} `{Setoid B} (a : A) (f : A → P B),
    Proper (equ ==> equ) f → (ret a : P A) >>= f == f a.
  Proof.
    intros A eqA EqvA B eqB EqvB a f Hf. apply equ_of_parses. intros s s'' b. rewrite (parses_bind _ _ _ _ _ Hf). split.
    - intros [a' [s' [Hr Hfa]]]. apply parses_ret in Hr. destruct Hr as [Ha Hs].
      apply (parses_input (f a) s' s b s'' Hs). exact (proj1 (parses_of_equ _ _ (Hf a' a Ha) s' s'' b) Hfa).
    - intros Hfa. ∃ a, s. split; [ apply parses_ret; split; reflexivity | exact Hfa ].
  Qed.

  Theorem bind_ret_r : ∀ `{Setoid A} (p : P A), p >>= ret == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s'' a. rewrite (parses_bind _ _ _ _ _ ret_proper). split.
    - intros [a' [s' [Hp Hr]]]. apply parses_ret in Hr. destruct Hr as [Ha Hs].
      apply (parses_rest p s a s' s'' (symmetry Hs)). apply (parses_result p s a' a s' (symmetry Ha)). exact Hp.
    - intros Hp. ∃ a, s''. split; [ exact Hp | apply parses_ret; split; reflexivity ].
  Qed.

  Theorem bind_assoc : ∀ `{Setoid A} `{Setoid B} `{Setoid D}
      (p : P A) (f : A → P B) (g : B → P D), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    (p >>= f) >>= g == p >>= (fun a ⇒ f a >>= g).
  Proof.
    intros A eqA EqvA B eqB EqvB D eqD EqvD p f g Hf Hg. apply equ_of_parses. intros s s3 d. split.
    - intros Hx. apply (parses_bind _ _ _ _ _ Hg) in Hx. destruct Hx as [b [s2 [Hpf Hgb]]].
      apply (parses_bind _ _ _ _ _ Hf) in Hpf. destruct Hpf as [a [s1 [Hp Hfa]]].
      apply (parses_bind _ _ _ _ _ (bind_fun_proper f g Hf Hg)). ∃ a, s1. split; [ exact Hp | ].
      apply (parses_bind _ _ _ _ _ Hg). ∃ b, s2. split; assumption.
    - intros Hx. apply (parses_bind _ _ _ _ _ (bind_fun_proper f g Hf Hg)) in Hx. destruct Hx as [a [s1 [Hp Hfg]]].
      apply (parses_bind _ _ _ _ _ Hg) in Hfg. destruct Hfg as [b [s2 [Hfa Hgb]]].
      apply (parses_bind _ _ _ _ _ Hg). ∃ b, s2. split; [ | exact Hgb ].
      apply (parses_bind _ _ _ _ _ Hf). ∃ a, s1. split; assumption.
  Qed.

Choice and failure

  Theorem alt_assoc : ∀ `{Setoid A} (p q r : P A), (p <|> q) <|> r == p <|> (q <|> r).
  Proof.
    intros A eqA EqvA p q r. apply equ_of_parses. intros s s' a. split.
    - intros Hx. apply parses_alt in Hx. apply parses_alt. destruct Hx as [Hx | Hr].
      + apply parses_alt in Hx. destruct Hx as [Hp | Hq]; [ left; exact Hp | right; apply parses_alt; left; exact Hq ].
      + right. apply parses_alt. right. exact Hr.
    - intros Hx. apply parses_alt in Hx. apply parses_alt. destruct Hx as [Hp | Hx].
      + left. apply parses_alt. left. exact Hp.
      + apply parses_alt in Hx. destruct Hx as [Hq | Hr]; [ left; apply parses_alt; right; exact Hq | right; exact Hr ].
  Qed.

  Theorem alt_comm : ∀ `{Setoid A} (p q : P A), p <|> q == q <|> p.
  Proof.
    intros A eqA EqvA p q. apply equ_of_parses. intros s s' a.
    split; intros Hx; apply parses_alt in Hx; apply parses_alt; tauto.
  Qed.

  Theorem alt_idem : ∀ `{Setoid A} (p : P A), p <|> p == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s' a.
    split; [ intros Hx; apply parses_alt in Hx; tauto | intros Hp; apply parses_alt; left; exact Hp ].
  Qed.

  Theorem alt_fail_l : ∀ `{Setoid A} (p : P A), (fail : P A) <|> p == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s' a. split.
    - intros Hx. apply parses_alt in Hx. destruct Hx as [Hf | Hp]; [ destruct (parses_fail s s' a Hf) | exact Hp ].
    - intros Hp. apply parses_alt. right. exact Hp.
  Qed.

  Theorem alt_fail_r : ∀ `{Setoid A} (p : P A), p <|> fail == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s' a. split.
    - intros Hx. apply parses_alt in Hx. destruct Hx as [Hp | Hf]; [ exact Hp | destruct (parses_fail s s' a Hf) ].
    - intros Hp. apply parses_alt. left. exact Hp.
  Qed.

  Theorem bind_fail_l : ∀ `{Setoid A} `{Setoid B} (f : A → P B),
    Proper (equ ==> equ) f → (fail : P A) >>= f == fail.
  Proof.
    intros A eqA EqvA B eqB EqvB f Hf. apply equ_of_parses. intros s s'' b. rewrite (parses_bind _ _ _ _ _ Hf). split.
    - intros [a [s' [Hfl _]]]. destruct (parses_fail s s' a Hfl).
    - intros Hfl. destruct (parses_fail s s'' b Hfl).
  Qed.

  Theorem bind_fail_r : ∀ `{Setoid A} `{Setoid B} (p : P A),
    p >>= (fun _ ⇒ fail) == (fail : P B).
  Proof.
    intros A eqA EqvA B eqB EqvB p. apply equ_of_parses. intros s s'' b.
    assert (Hc : Proper (equ ==> equ) (fun _ : A ⇒ (fail : P B))) by (intros x y _; reflexivity).
    rewrite (parses_bind _ _ _ _ _ Hc). split.
    - intros [a [s' [_ Hfl]]]. destruct (parses_fail s' s'' b Hfl).
    - intros Hfl. destruct (parses_fail s s'' b Hfl).
  Qed.

  Theorem bind_alt_l : ∀ `{Setoid A} `{Setoid B} (p q : P A) (f : A → P B),
    Proper (equ ==> equ) f → (p <|> q) >>= f == (p >>= f) <|> (q >>= f).
  Proof.
    intros A eqA EqvA B eqB EqvB p q f Hf. apply equ_of_parses. intros s s'' b. split.
    - intros Hx. apply (parses_bind _ _ _ _ _ Hf) in Hx. destruct Hx as [a [s' [Hpq Hfa]]]. apply parses_alt in Hpq.
      apply parses_alt. destruct Hpq as [Hp | Hq]; [ left | right ]; apply (parses_bind _ _ _ _ _ Hf); ∃ a, s'; split; assumption.
    - intros Hx. apply parses_alt in Hx. apply (parses_bind _ _ _ _ _ Hf).
      destruct Hx as [Hx | Hx]; apply (parses_bind _ _ _ _ _ Hf) in Hx; destruct Hx as [a [s' [Hpq Hfa]]]; ∃ a, s';
        (split; [ apply parses_alt; first [ left; exact Hpq | right; exact Hpq ] | exact Hfa ]).
  Qed.

  Theorem bind_alt_r : ∀ `{Setoid A} `{Setoid B} (p : P A) (f g : A → P B),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    p >>= (fun a ⇒ f a <|> g a) == (p >>= f) <|> (p >>= g).
  Proof.
    intros A eqA EqvA B eqB EqvB p f g Hf Hg.
    assert (Hfg : Proper (equ ==> equ) (fun a ⇒ f a <|> g a)).
    { intros a a' Ha. apply equ_of_parses. intros s s' b.
      split; intros Hx; apply parses_alt in Hx; apply parses_alt; destruct Hx as [Hx | Hx].
      - left. exact (proj1 (parses_of_equ _ _ (Hf a a' Ha) s s' b) Hx).
      - right. exact (proj1 (parses_of_equ _ _ (Hg a a' Ha) s s' b) Hx).
      - left. exact (proj2 (parses_of_equ _ _ (Hf a a' Ha) s s' b) Hx).
      - right. exact (proj2 (parses_of_equ _ _ (Hg a a' Ha) s s' b) Hx). }
    apply equ_of_parses. intros s s'' b. split.
    - intros Hx. apply (parses_bind _ _ _ _ _ Hfg) in Hx. destruct Hx as [a [s' [Hp Hx]]]. apply parses_alt in Hx.
      apply parses_alt. destruct Hx as [Hfa | Hga]; [ left; apply (parses_bind _ _ _ _ _ Hf) | right; apply (parses_bind _ _ _ _ _ Hg) ];
        ∃ a, s'; split; assumption.
    - intros Hx. apply parses_alt in Hx. apply (parses_bind _ _ _ _ _ Hfg).
      destruct Hx as [Hx | Hx]; [ apply (parses_bind _ _ _ _ _ Hf) in Hx | apply (parses_bind _ _ _ _ _ Hg) in Hx ];
        destruct Hx as [a [s' [Hp Hr]]]; ∃ a, s'; (split; [ exact Hp | apply parses_alt; first [ left; exact Hr | right; exact Hr ] ]).
  Qed.

The derived combinators

  Lemma fmap_fun_proper : ∀ `{Setoid A} `{Setoid B} (f : A → B),
    Proper (equ ==> equ) f → Proper (equ ==> equ) (fun a ⇒ (ret (f a) : P B)).
  Proof. intros A eqA EqvA B eqB EqvB f Hf a a' Ha. apply ret_proper. apply Hf. exact Ha. Qed.

  Theorem parses_fmap : ∀ `{Setoid A} `{Setoid B} (f : A → B) (p : P A),
    Proper (equ ==> equ) f → ∀ (s s' : S) (b : B),
    parses (f <$> p) s b s' ↔ ∃ a : A, parses p s a s' ∧ b == f a.
  Proof.
    intros A eqA EqvA B eqB EqvB f p Hf s s' b. rewrite (parses_bind _ _ _ _ _ (fmap_fun_proper f Hf)). split.
    - intros [a [s1 [Hp Hr]]]. apply parses_ret in Hr. destruct Hr as [Hb Hs]. ∃ a.
      split; [ exact (parses_rest p s a s1 s' (symmetry Hs) Hp) | exact Hb ].
    - intros [a [Hp Hb]]. ∃ a, s'. split; [ exact Hp | apply parses_ret; split; [ exact Hb | reflexivity ] ].
  Qed.

  Theorem fmap_id : ∀ `{Setoid A} (p : P A), (fun a ⇒ a) <$> p == p.
  Proof. intros A eqA EqvA p. exact (bind_ret_r p). Qed.

  Theorem fmap_comp : ∀ `{Setoid A} `{Setoid B} `{Setoid D}
      (f : A → B) (g : B → D) (p : P A), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    g <$> (f <$> p) == (fun a ⇒ g (f a)) <$> p.
  Proof.
    intros A eqA EqvA B eqB EqvB D eqD EqvD f g p Hf Hg.
    assert (Hgf : Proper (equ ==> equ) (fun a ⇒ g (f a))) by (intros a a' Ha; apply Hg; apply Hf; exact Ha).
    apply equ_of_parses. intros s s' d. rewrite (parses_fmap g _ Hg), (parses_fmap _ p Hgf). split.
    - intros [b [Hb Hd]]. apply (parses_fmap f p Hf) in Hb. destruct Hb as [a [Hp Hb]]. ∃ a.
      split; [ exact Hp | transitivity (g b); [ exact Hd | apply Hg; exact Hb ] ].
    - intros [a [Hp Hd]]. ∃ (f a). split; [ apply (parses_fmap f p Hf); ∃ a; split; [ exact Hp | reflexivity ] | exact Hd ].
  Qed.

  Theorem parses_seq_r : ∀ `{Setoid A} `{Setoid B} (p : P A) (q : P B) (s s'' : S) (b : B),
    parses (p *> q) s b s'' ↔ ∃ (a : A) (s' : S), parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A eqA EqvA B eqB EqvB p q s s'' b.
    assert (Hc : Proper (equ ==> equ) (fun _ : A ⇒ q)) by (intros x y _; reflexivity).
    exact (parses_bind p (fun _ ⇒ q) s s'' b Hc).
  Qed.

  Theorem parses_seq_l : ∀ `{Setoid A} `{Setoid B} (p : P A) (q : P B) (s s'' : S) (a : A),
    parses (p <* q) s a s'' ↔ ∃ (s' : S) (b : B), parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A eqA EqvA B eqB EqvB p q s s'' a.
    assert (Hc : Proper (equ ==> equ) (fun a : A ⇒ q >>= fun _ ⇒ (ret a : P A))).
    { intros x y Hxy. apply bind_proper_r; [ intros u v _; reflexivity | intros u v _; reflexivity | intros _; apply ret_proper; exact Hxy ]. }
    assert (Hq : ∀ a0 : A, Proper (equ ==> equ) (fun _ : B ⇒ (ret a0 : P A))) by (intros a0 x y _; reflexivity).
    rewrite (parses_bind _ _ _ _ _ Hc). split.
    - intros [a' [s' [Hp Hr]]]. apply (parses_bind _ _ _ _ _ (Hq a')) in Hr. destruct Hr as [b [s2 [Hqb Hr]]].
      apply parses_ret in Hr. destruct Hr as [Ha Hs]. ∃ s', b.
      split; [ exact (parses_result p s a' a s' (symmetry Ha) Hp) | exact (parses_rest q s' b s2 s'' (symmetry Hs) Hqb) ].
    - intros [s' [b [Hp Hqb]]]. ∃ a, s'. split; [ exact Hp | ]. apply (parses_bind _ _ _ _ _ (Hq a)). ∃ b, s''.
      split; [ exact Hqb | apply parses_ret; split; reflexivity ].
  Qed.

Pairs are compared componentwise, whatever setoid they carry.
  Theorem parses_pair : ∀ `{Setoid A} `{Setoid B} `{Setoid (A × B)},
    (∀ (a a' : A) (b b' : B), (a, b) == (a', b') ↔ a == a' ∧ b == b') →
    ∀ (p : P A) (q : P B) (s s'' : S) (a : A) (b : B),
    parses (p ⊗ q) s (a, b) s'' ↔ ∃ s' : S, parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A eqA EqvA B eqB EqvB eqAB EqvAB Hpair p q s s'' a b.
    assert (Hin : ∀ a0 : A, Proper (equ ==> equ) (fun b0 : B ⇒ (ret (a0, b0) : P (A × B)))).
    { intros a0 x y Hxy. apply ret_proper. apply Hpair. split; [ reflexivity | exact Hxy ]. }
    assert (Hout : Proper (equ ==> equ) (fun a0 : A ⇒ q >>= fun b0 ⇒ (ret (a0, b0) : P (A × B)))).
    { intros x y Hxy. apply bind_proper_r; [ exact (Hin x) | exact (Hin y) | intros b0; apply ret_proper; apply Hpair; split; [ exact Hxy | reflexivity ] ]. }
    rewrite (parses_bind _ _ _ _ _ Hout). split.
    - intros [a' [s' [Hp Hq]]]. apply (parses_bind _ _ _ _ _ (Hin a')) in Hq. destruct Hq as [b' [s2 [Hq Hr]]].
      apply parses_ret in Hr. destruct Hr as [Hab Hs]. apply Hpair in Hab. destruct Hab as [Ha Hb].
      ∃ s'. split; [ exact (parses_result p s a' a s' (symmetry Ha) Hp) | exact (parses_rest q s' b s2 s'' (symmetry Hs) (parses_result q s' b' b s2 (symmetry Hb) Hq)) ].
    - intros [s' [Hp Hq]]. ∃ a, s'. split; [ exact Hp | ]. apply (parses_bind _ _ _ _ _ (Hin a)). ∃ b, s''.
      split; [ exact Hq | apply parses_ret; split; reflexivity ].
  Qed.

  Section Symbols.
    Context {EC : @EquDec C eqC}.

    Theorem parses_sym : ∀ (c : C) (s s' : S) (c' : C),
      parses (sym c : P C) s c' s' ↔ c' == c ∧ step s c' s'.
    Proof.
      intros c s s' c'.
      assert (Hk : Proper (equ ==> equ) (fun c0 : C ⇒ if equ_dec c0 c then (ret c0 : P C) else fail)).
      { intros x y Hxy. destruct (equ_dec x c) as [Hx | Hx]; destruct (equ_dec y c) as [Hy | Hy].
        - exact (ret_proper x y Hxy).
        - destruct Hy. transitivity x; [ symmetry; exact Hxy | exact Hx ].
        - destruct Hx. transitivity y; [ exact Hxy | exact Hy ].
        - reflexivity. }
      rewrite (parses_bind _ _ _ _ _ Hk). split.
      - intros [c0 [s0 [Hi Hr]]]. apply parses_item in Hi. destruct (equ_dec c0 c) as [Hc | Hc].
        + apply parses_ret in Hr. destruct Hr as [Hc' Hs]. split; [ transitivity c0; assumption | ].
          exact (proj1 (step_proper s s (reflexivity s) c0 c' (symmetry Hc') s0 s' (symmetry Hs)) Hi).
        + destruct (parses_fail s0 s' c' Hr).
      - intros [Hc Hs]. ∃ c', s'. split; [ apply parses_item; exact Hs | ].
        destruct (equ_dec c' c) as [_ | Hn]; [ apply parses_ret; split; reflexivity | destruct (Hn Hc) ].
    Qed.
  End Symbols.
