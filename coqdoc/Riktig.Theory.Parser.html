Derived combinators, as notation: reading a function of what is read, reading two things in sequence, keeping one of two readings, and reading one given element.
Notation "f <$> p" := (p >>= fun a ⇒ ret (f a)) (at level 61, left associativity).
Notation "p ⊗ q" := (p >>= fun a ⇒ q >>= fun b ⇒ ret (a, b)) (at level 40, left associativity).
Notation "p *> q" := (p >>= fun _ ⇒ q) (at level 40, left associativity).
Notation "p <* q" := (p >>= fun a ⇒ q >>= fun _ ⇒ ret a) (at level 40, left associativity).
Notation "'sym' c" := (item >>= fun c' ⇒ if equ_dec c' c then ret c' else fail) (at level 10, c at level 9).

What holds of every parser: equality of parsers is an equivalence, return and bind make a monad, failure and choice a commutative idempotent monoid, bind distributes over choice and is absorbed by failure, and the derived combinators have the readings one expects.
Section ParserTheory.
  Context {C S : Type} {P : Type → Type} `{Parser C S P}.
  Local Instance elements_equivalence : Equivalence (@equ C eqC) := @elements C S eqC eqS empty_ cons_ Seq.

  Lemma equ_of_parses : ∀ (A : Type) (p q : P A),
    (∀ (s : S) (a : A) (s' : S), parses p s a s' ↔ parses q s a s') → p == q.
  Proof. intros A p q Hpq. exact (proj2 (parser_equ A p q) Hpq). Qed.

  Lemma parses_of_equ : ∀ (A : Type) (p q : P A), p == q →
    ∀ (s : S) (a : A) (s' : S), parses p s a s' ↔ parses q s a s'.
  Proof. intros A p q Hpq. exact (proj1 (parser_equ A p q) Hpq). Qed.

  Lemma parses_rest : ∀ (A : Type) (p : P A) (s : S) (a : A) (s1 s2 : S),
    s1 == s2 → parses p s a s1 → parses p s a s2.
  Proof. intros A p s a s1 s2 Hs Hp. exact (proj1 (parses_proper A p s s (reflexivity s) a a eq_refl s1 s2 Hs) Hp). Qed.

  Lemma parses_input : ∀ (A : Type) (p : P A) (s1 s2 : S) (a : A) (s' : S),
    s1 == s2 → parses p s1 a s' → parses p s2 a s'.
  Proof. intros A p s1 s2 a s' Hs Hp. exact (proj1 (parses_proper A p s1 s2 Hs a a eq_refl s' s' (reflexivity s')) Hp). Qed.

  #[export] Instance parser_equivalence : ∀ A : Type, Equivalence (@equ (P A) (eqP A)).
  Proof.
    intros A. split.
    - intros p. apply equ_of_parses. intros s a s'. reflexivity.
    - intros p q Hpq. apply equ_of_parses. intros s a s'. symmetry. exact (parses_of_equ A p q Hpq s a s').
    - intros p q r Hpq Hqr. apply equ_of_parses. intros s a s'.
      transitivity (parses q s a s'); [ exact (parses_of_equ A p q Hpq s a s') | exact (parses_of_equ A q r Hqr s a s') ].
  Qed.

The monad laws

  Theorem bind_ret_l : ∀ (A B : Type) (a : A) (f : A → P B), (ret a : P A) >>= f == f a.
  Proof.
    intros A B a f. apply equ_of_parses. intros s b s''. rewrite parses_bind. split.
    - intros [a' [s' [Hr Hf]]]. apply parses_ret in Hr. destruct Hr as [Ha Hs]. subst a'.
      rewrite Hs in Hf. exact Hf.
    - intros Hf. ∃ a, s. split; [ apply parses_ret; split; reflexivity | exact Hf ].
  Qed.

  Theorem bind_ret_r : ∀ (A : Type) (p : P A), p >>= ret == p.
  Proof.
    intros A p. apply equ_of_parses. intros s a s''. rewrite parses_bind. split.
    - intros [a' [s' [Hp Hr]]]. apply parses_ret in Hr. destruct Hr as [Ha Hs]. subst a'.
      rewrite Hs. exact Hp.
    - intros Hp. ∃ a, s''. split; [ exact Hp | apply parses_ret; split; reflexivity ].
  Qed.

  Theorem bind_assoc : ∀ (A B D : Type) (p : P A) (f : A → P B) (g : B → P D),
    (p >>= f) >>= g == p >>= (fun a ⇒ f a >>= g).
  Proof.
    intros A B D p f g. apply equ_of_parses. intros s d s3. split.
    - intros Hx. apply parses_bind in Hx. destruct Hx as [b [s2 [Hpf Hg]]].
      apply parses_bind in Hpf. destruct Hpf as [a [s1 [Hp Hf]]].
      apply parses_bind. ∃ a, s1. split; [ exact Hp | ].
      apply parses_bind. ∃ b, s2. split; assumption.
    - intros Hx. apply parses_bind in Hx. destruct Hx as [a [s1 [Hp Hfg]]].
      apply parses_bind in Hfg. destruct Hfg as [b [s2 [Hf Hg]]].
      apply parses_bind. ∃ b, s2. split; [ | exact Hg ].
      apply parses_bind. ∃ a, s1. split; assumption.
  Qed.

Choice and failure

  Theorem alt_assoc : ∀ (A : Type) (p q r : P A), (p <|> q) <|> r == p <|> (q <|> r).
  Proof.
    intros A p q r. apply equ_of_parses. intros s a s'. split.
    - intros Hx. apply parses_alt in Hx. apply parses_alt. destruct Hx as [Hx | Hr].
      + apply parses_alt in Hx. destruct Hx as [Hp | Hq]; [ left; exact Hp | right; apply parses_alt; left; exact Hq ].
      + right. apply parses_alt. right. exact Hr.
    - intros Hx. apply parses_alt in Hx. apply parses_alt. destruct Hx as [Hp | Hx].
      + left. apply parses_alt. left. exact Hp.
      + apply parses_alt in Hx. destruct Hx as [Hq | Hr]; [ left; apply parses_alt; right; exact Hq | right; exact Hr ].
  Qed.

  Theorem alt_comm : ∀ (A : Type) (p q : P A), p <|> q == q <|> p.
  Proof.
    intros A p q. apply equ_of_parses. intros s a s'.
    split; intros Hx; apply parses_alt in Hx; apply parses_alt; tauto.
  Qed.

  Theorem alt_idem : ∀ (A : Type) (p : P A), p <|> p == p.
  Proof.
    intros A p. apply equ_of_parses. intros s a s'.
    split; [ intros Hx; apply parses_alt in Hx; tauto | intros Hp; apply parses_alt; left; exact Hp ].
  Qed.

  Theorem alt_fail_l : ∀ (A : Type) (p : P A), (fail : P A) <|> p == p.
  Proof.
    intros A p. apply equ_of_parses. intros s a s'. split.
    - intros Hx. apply parses_alt in Hx. destruct Hx as [Hf | Hp]; [ destruct (parses_fail A s a s' Hf) | exact Hp ].
    - intros Hp. apply parses_alt. right. exact Hp.
  Qed.

  Theorem alt_fail_r : ∀ (A : Type) (p : P A), p <|> fail == p.
  Proof.
    intros A p. apply equ_of_parses. intros s a s'. split.
    - intros Hx. apply parses_alt in Hx. destruct Hx as [Hp | Hf]; [ exact Hp | destruct (parses_fail A s a s' Hf) ].
    - intros Hp. apply parses_alt. left. exact Hp.
  Qed.

  Theorem bind_fail_l : ∀ (A B : Type) (f : A → P B), (fail : P A) >>= f == fail.
  Proof.
    intros A B f. apply equ_of_parses. intros s b s''. rewrite parses_bind. split.
    - intros [a [s' [Hf _]]]. destruct (parses_fail A s a s' Hf).
    - intros Hf. destruct (parses_fail B s b s'' Hf).
  Qed.

  Theorem bind_fail_r : ∀ (A B : Type) (p : P A), p >>= (fun _ ⇒ fail) == (fail : P B).
  Proof.
    intros A B p. apply equ_of_parses. intros s b s''. rewrite parses_bind. split.
    - intros [a [s' [_ Hf]]]. destruct (parses_fail B s' b s'' Hf).
    - intros Hf. destruct (parses_fail B s b s'' Hf).
  Qed.

  Theorem bind_alt_l : ∀ (A B : Type) (p q : P A) (f : A → P B),
    (p <|> q) >>= f == (p >>= f) <|> (q >>= f).
  Proof.
    intros A B p q f. apply equ_of_parses. intros s b s''. split.
    - intros Hx. apply parses_bind in Hx. destruct Hx as [a [s' [Hpq Hf]]]. apply parses_alt in Hpq.
      apply parses_alt. destruct Hpq as [Hp | Hq]; [ left | right ]; apply parses_bind; ∃ a, s'; split; assumption.
    - intros Hx. apply parses_alt in Hx. apply parses_bind.
      destruct Hx as [Hx | Hx]; apply parses_bind in Hx; destruct Hx as [a [s' [Hpq Hf]]]; ∃ a, s';
        (split; [ apply parses_alt; first [ left; exact Hpq | right; exact Hpq ] | exact Hf ]).
  Qed.

  Theorem bind_alt_r : ∀ (A B : Type) (p : P A) (f g : A → P B),
    p >>= (fun a ⇒ f a <|> g a) == (p >>= f) <|> (p >>= g).
  Proof.
    intros A B p f g. apply equ_of_parses. intros s b s''. split.
    - intros Hx. apply parses_bind in Hx. destruct Hx as [a [s' [Hp Hfg]]]. apply parses_alt in Hfg.
      apply parses_alt. destruct Hfg as [Hf | Hg]; [ left | right ]; apply parses_bind; ∃ a, s'; split; assumption.
    - intros Hx. apply parses_alt in Hx. apply parses_bind.
      destruct Hx as [Hx | Hx]; apply parses_bind in Hx; destruct Hx as [a [s' [Hp Hfg]]]; ∃ a, s';
        (split; [ exact Hp | apply parses_alt; first [ left; exact Hfg | right; exact Hfg ] ]).
  Qed.

The derived combinators

  Theorem parses_fmap : ∀ (A B : Type) (f : A → B) (p : P A) (s : S) (b : B) (s' : S),
    parses (f <$> p) s b s' ↔ ∃ a : A, parses p s a s' ∧ b = f a.
  Proof.
    intros A B f p s b s'. rewrite parses_bind. split.
    - intros [a [s1 [Hp Hr]]]. apply parses_ret in Hr. destruct Hr as [Hb Hs]. ∃ a. rewrite <- Hs in Hp. split; assumption.
    - intros [a [Hp Hb]]. ∃ a, s'. split; [ exact Hp | apply parses_ret; split; [ exact Hb | reflexivity ] ].
  Qed.

  Theorem fmap_id : ∀ (A : Type) (p : P A), (fun a ⇒ a) <$> p == p.
  Proof. intros A p. exact (bind_ret_r A p). Qed.

  Theorem fmap_comp : ∀ (A B D : Type) (f : A → B) (g : B → D) (p : P A),
    g <$> (f <$> p) == (fun a ⇒ g (f a)) <$> p.
  Proof.
    intros A B D f g p. apply equ_of_parses. intros s d s'. rewrite !parses_fmap. split.
    - intros [b [Hb Hd]]. apply parses_fmap in Hb. destruct Hb as [a [Hp Hb]]. ∃ a. subst. split; [ exact Hp | reflexivity ].
    - intros [a [Hp Hd]]. ∃ (f a). split; [ apply parses_fmap; ∃ a; split; [ exact Hp | reflexivity ] | exact Hd ].
  Qed.

  Theorem parses_pair : ∀ (A B : Type) (p : P A) (q : P B) (s : S) (a : A) (b : B) (s'' : S),
    parses (p ⊗ q) s (a, b) s'' ↔ ∃ s' : S, parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A B p q s a b s''. rewrite parses_bind. split.
    - intros [a' [s' [Hp Hq]]]. apply parses_bind in Hq. destruct Hq as [b' [s2 [Hq Hr]]].
      apply parses_ret in Hr. destruct Hr as [Hab Hs]. injection Hab as Ha Hb. subst a' b'.
      ∃ s'. split; [ exact Hp | exact (parses_rest B q s' b s2 s'' (symmetry Hs) Hq) ].
    - intros [s' [Hp Hq]]. ∃ a, s'. split; [ exact Hp | ]. apply parses_bind. ∃ b, s''.
      split; [ exact Hq | apply parses_ret; split; reflexivity ].
  Qed.

  Theorem pair_unit_l : ∀ (A : Type) (p : P A), snd <$> ((ret tt : P unit) ⊗ p) == p.
  Proof.
    intros A p. apply equ_of_parses. intros s a s'. rewrite parses_fmap. split.
    - intros [[u a'] [Hpair Ha]]. apply parses_pair in Hpair. destruct Hpair as [s1 [Hr Hp]].
      apply parses_ret in Hr. destruct Hr as [_ Hs]. simpl in Ha. subst a'. exact (parses_input A p s1 s a s' Hs Hp).
    - intros Hp. ∃ (tt, a). split; [ | reflexivity ]. apply parses_pair. ∃ s.
      split; [ apply parses_ret; split; reflexivity | exact Hp ].
  Qed.

  Theorem pair_unit_r : ∀ (A : Type) (p : P A), fst <$> (p ⊗ (ret tt : P unit)) == p.
  Proof.
    intros A p. apply equ_of_parses. intros s a s'. rewrite parses_fmap. split.
    - intros [[a' u] [Hpair Ha]]. apply parses_pair in Hpair. destruct Hpair as [s1 [Hp Hr]].
      apply parses_ret in Hr. destruct Hr as [_ Hs]. simpl in Ha. subst a'. exact (parses_rest A p s a s1 s' (symmetry Hs) Hp).
    - intros Hp. ∃ (a, tt). split; [ | reflexivity ]. apply parses_pair. ∃ s'.
      split; [ exact Hp | apply parses_ret; split; reflexivity ].
  Qed.

  Theorem pair_assoc : ∀ (A B D : Type) (p : P A) (q : P B) (r : P D),
    (fun x ⇒ (fst (fst x), (snd (fst x), snd x))) <$> ((p ⊗ q) ⊗ r) == p ⊗ (q ⊗ r).
  Proof.
    intros A B D p q r. apply equ_of_parses. intros s [a [b d]] s3. rewrite parses_fmap, parses_pair. split.
    - intros [[[a' b'] d'] [Hx Hx']]. simpl in Hx'. injection Hx' as Ha Hb Hd. subst a' b' d'.
      apply parses_pair in Hx. destruct Hx as [s2 [Hpq Hr]]. apply parses_pair in Hpq. destruct Hpq as [s1 [Hp Hq]].
      ∃ s1. split; [ exact Hp | ]. apply parses_pair. ∃ s2. split; assumption.
    - intros [s1 [Hp Hqr]]. apply parses_pair in Hqr. destruct Hqr as [s2 [Hq Hr]].
      ∃ ((a, b), d). split; [ | reflexivity ]. apply parses_pair. ∃ s2. split; [ | exact Hr ].
      apply parses_pair. ∃ s1. split; assumption.
  Qed.

  Theorem parses_seq_r : ∀ (A B : Type) (p : P A) (q : P B) (s : S) (b : B) (s'' : S),
    parses (p *> q) s b s'' ↔ ∃ (a : A) (s' : S), parses p s a s' ∧ parses q s' b s''.
  Proof. intros. exact (parses_bind A B p (fun _ ⇒ q) s b s''). Qed.

  Theorem parses_seq_l : ∀ (A B : Type) (p : P A) (q : P B) (s : S) (a : A) (s'' : S),
    parses (p <* q) s a s'' ↔ ∃ (s' : S) (b : B), parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A B p q s a s''. rewrite parses_bind. split.
    - intros [a' [s' [Hp Hq]]]. apply parses_bind in Hq. destruct Hq as [b [s2 [Hq Hr]]].
      apply parses_ret in Hr. destruct Hr as [Ha Hs]. subst a'. ∃ s', b. split; [ exact Hp | exact (parses_rest B q s' b s2 s'' (symmetry Hs) Hq) ].
    - intros [s' [b [Hp Hq]]]. ∃ a, s'. split; [ exact Hp | ]. apply parses_bind. ∃ b, s''.
      split; [ exact Hq | apply parses_ret; split; reflexivity ].
  Qed.

  Theorem item_empty : ∀ (c : C) (s' : S), ¬ parses (item : P C) empty c s'.
  Proof. intros c s' Hi. apply parses_item in Hi. exact (cons_not_empty c s' (symmetry Hi)). Qed.

  Section Symbols.
    Context {EC : @EquDec C eqC}.

    Theorem parses_sym : ∀ (c : C) (s : S) (c' : C) (s' : S),
      parses (sym c : P C) s c' s' ↔ c' == c ∧ s == cons c' s'.
    Proof.
      intros c s c' s'. rewrite parses_bind. split.
      - intros [c0 [s0 [Hi Hk]]]. apply parses_item in Hi. destruct (equ_dec c0 c) as [Hc | Hc].
        + apply parses_ret in Hk. destruct Hk as [Hc' Hs]. subst c0. split; [ exact Hc | ].
          transitivity (cons c' s0); [ exact Hi | apply cons_proper; [ reflexivity | symmetry; exact Hs ] ].
        + destruct (parses_fail C s0 c' s' Hk).
      - intros [Hc Hs]. ∃ c', s'. split; [ apply parses_item; exact Hs | ].
        destruct (equ_dec c' c) as [_ | Hn]; [ apply parses_ret; split; reflexivity | destruct (Hn Hc) ].
    Qed.
  End Symbols.
