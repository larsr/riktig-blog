Derived combinators, as notation: reading a function of what is read, reading two things in sequence, keeping one of two readings, and reading one given element.
Notation "f <$> p" := (p >>= fun a ⇒ ret (f a)) (at level 61, left associativity).
Notation "p ⊗ q" := (p >>= fun a ⇒ q >>= fun b ⇒ ret (a, b)) (at level 40, left associativity).
Notation "p *> q" := (p >>= fun _ ⇒ q) (at level 40, left associativity).
Notation "p <* q" := (p >>= fun a ⇒ q >>= fun _ ⇒ ret a) (at level 40, left associativity).
Notation "'sym' c" := (item >>= fun c' ⇒ if EquDec.equ_dec c' c then ret c' else fail) (at level 10, c at level 9).

What holds of every parser: equality of parsers is an equivalence, return and bind make a monad, failure and choice a commutative idempotent monoid, bind distributes over choice and is absorbed by failure, and the derived combinators read what one expects. The functions fed to bind respect equality throughout.

  Context {C S : Type} {P : Type → Type} `{Parser C S P}.
  Local Instance elements_equivalence : Equivalence (@equ C eqC) := @Sequence.elements C S eqC eqS stepping Seq.
  Local Instance elements_setoid : Setoid C := {| setoid_equivalence := elements_equivalence |}.

  Lemma equ_of_parses : ∀ `{Setoid A} (p q : P A),
    (∀ (s s' : S) (a : A), parses p s a s' ↔ parses q s a s') → p == q.
  Proof. intros A eqA EqvA p q Hpq. exact (proj2 (Parser.parser_equ p q) Hpq). Qed.

  Lemma parses_of_equ : ∀ `{Setoid A} (p q : P A), p == q →
    ∀ (s s' : S) (a : A), parses p s a s' ↔ parses q s a s'.
  Proof. intros A eqA EqvA p q Hpq. exact (proj1 (Parser.parser_equ p q) Hpq). Qed.

  Lemma parses_rest : ∀ `{Setoid A} (p : P A) (s : S) (a : A) (s1 s2 : S),
    s1 == s2 → parses p s a s1 → parses p s a s2.
  Proof. intros A eqA EqvA p s a s1 s2 Hs Hp. exact (proj1 (Parser.parses_proper p s s (reflexivity s) a a (reflexivity a) s1 s2 Hs) Hp). Qed.

  Lemma parses_input : ∀ `{Setoid A} (p : P A) (s1 s2 : S) (a : A) (s' : S),
    s1 == s2 → parses p s1 a s' → parses p s2 a s'.
  Proof. intros A eqA EqvA p s1 s2 a s' Hs Hp. exact (proj1 (Parser.parses_proper p s1 s2 Hs a a (reflexivity a) s' s' (reflexivity s')) Hp). Qed.

  Lemma parses_result : ∀ `{Setoid A} (p : P A) (s : S) (a a' : A) (s' : S),
    a == a' → parses p s a s' → parses p s a' s'.
  Proof. intros A eqA EqvA p s a a' s' Ha Hp. exact (proj1 (Parser.parses_proper p s s (reflexivity s) a a' Ha s' s' (reflexivity s')) Hp). Qed.

Meaning respects the result and the rest, and, through a function that respects equality, the read element it is applied to.
  Lemma parses_proper2 : ∀ `{Setoid A} (p : P A) (s : S),
    Proper (equ ==> equ ==> iff) (fun (a : A) (s' : S) ⇒ parses p s a s').
  Proof. intros A eqA EqvA p s a a' Ha s' s'' Hs. exact (Parser.parses_proper p s s (reflexivity s) a a' Ha s' s'' Hs). Qed.

  Lemma parses_fun_proper : ∀ `{Setoid A} `{Setoid B} (f : A → P B) (b : B) (s'' : S),
    Proper (equ ==> equ) f → Proper (equ ==> equ ==> iff) (fun (a : A) (s' : S) ⇒ parses (f a) s' b s'').
  Proof.
    intros A eqA EqvA B eqB EqvB f b s'' Hf a a' Ha s' s2 Hs.
    exact (iff_trans (parses_of_equ _ _ (Hf a a' Ha) s' s'' b)
                     (Parser.parses_proper (f a') s' s2 Hs b b (reflexivity b) s'' s'' (reflexivity s''))).
  Qed.

  Lemma parses_fail_iff : ∀ `{Setoid A} (s s' : S) (a : A), parses (fail : P A) s a s' ↔ False.
  Proof. intros A eqA EqvA s s' a. split; [ exact (Parser.parses_fail s s' a) | intros [] ]. Qed.

  #[export] Instance parser_equivalence : ∀ `{Setoid A}, Equivalence (@equ (P A) (eqP A eqA)).
  Proof.
    intros A eqA EqvA. split.
    - intros p. exact (equ_of_parses p p (fun s s' a ⇒ iff_refl _)).
    - intros p q Hpq. exact (equ_of_parses q p (fun s s' a ⇒ iff_sym (parses_of_equ p q Hpq s s' a))).
    - intros p q r Hpq Hqr.
      exact (equ_of_parses p r (fun s s' a ⇒ iff_trans (parses_of_equ p q Hpq s s' a) (parses_of_equ q r Hqr s s' a))).
  Qed.

Return and bind respect equality

  Theorem ret_proper : ∀ `{Setoid A}, Proper (equ ==> equ) (ret : A → P A).
  Proof.
    intros A eqA EqvA a a' Ha. apply equ_of_parses. intros s s' b.
    calc (parses (ret a : P A) s b s')
      ↔ (b == a ∧ s' == s) { (Parser.parses_ret a b s s') }
      ↔ (b == a' ∧ s' == s) { (and_iff_morphism _ _ (equiv_iff (reflexivity b) Ha) _ _ (iff_refl _)) }
      ↔ (parses (ret a' : P A) s b s') { (Parser.parses_ret a' b s s') }.
  Qed.

  Theorem bind_proper_l : ∀ `{Setoid A} `{Setoid B}
      (p p' : P A) (f : A → P B), Proper (equ ==> equ) f → p == p' → p >>= f == p' >>= f.
  Proof.
    intros A eqA EqvA B eqB EqvB p p' f Hf Hp. apply equ_of_parses. intros s s'' b.
    calc (parses (p >>= f : P B) s b s'')
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ parses (f a) s' b s'') { (Parser.parses_bind p f s s'' b Hf) }
      ↔ (∃ (a : A) (s' : S), parses p' s a s' ∧ parses (f a) s' b s'') { (ex2_and_l _ _ _ (fun a s' ⇒ parses_of_equ p p' Hp s s' a)) }
      ↔ (parses (p' >>= f : P B) s b s'') { (Parser.parses_bind p' f s s'' b Hf) }.
  Qed.

  Theorem bind_proper_r : ∀ `{Setoid A} `{Setoid B}
      (p : P A) (f g : A → P B), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
      (∀ a : A, f a == g a) → p >>= f == p >>= g.
  Proof.
    intros A eqA EqvA B eqB EqvB p f g Hf Hg Hfg. apply equ_of_parses. intros s s'' b.
    calc (parses (p >>= f : P B) s b s'')
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ parses (f a) s' b s'') { (Parser.parses_bind p f s s'' b Hf) }
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ parses (g a) s' b s'') { (ex2_and_r _ _ _ (fun a s' ⇒ parses_of_equ _ _ (Hfg a) s' s'' b)) }
      ↔ (parses (p >>= g : P B) s b s'') { (Parser.parses_bind p g s s'' b Hg) }.
  Qed.

  Lemma bind_fun_proper : ∀ `{Setoid A} `{Setoid B} `{Setoid D}
      (f : A → P B) (g : B → P D), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
      Proper (equ ==> equ) (fun a ⇒ f a >>= g).
  Proof. intros A eqA EqvA B eqB EqvB D eqD EqvD f g Hf Hg a a' Ha. apply bind_proper_l; [ exact Hg | apply Hf; exact Ha ]. Qed.

  Lemma const_fun_proper : ∀ `{Setoid A} `{Setoid B} (q : P B), Proper (equ ==> equ) (fun _ : A ⇒ q).
  Proof. intros A eqA EqvA B eqB EqvB q x y _. reflexivity. Qed.

The monad laws

  Theorem bind_ret_l : ∀ `{Setoid A} `{Setoid B} (a : A) (f : A → P B),
    Proper (equ ==> equ) f → (ret a : P A) >>= f == f a.
  Proof.
    intros A eqA EqvA B eqB EqvB a f Hf. apply equ_of_parses. intros s s'' b.
    calc (parses ((ret a : P A) >>= f) s b s'')
      ↔ (∃ (a' : A) (s' : S), parses (ret a : P A) s a' s' ∧ parses (f a') s' b s'') { (Parser.parses_bind (ret a) f s s'' b Hf) }
      ↔ (∃ (a' : A) (s' : S), (a' == a ∧ s' == s) ∧ parses (f a') s' b s'') { (ex2_and_l _ _ _ (fun a' s' ⇒ Parser.parses_ret a a' s s')) }
      ↔ (parses (f a) s b s'') { (ex2_equ_l a s _ (parses_fun_proper f b s'' Hf)) }.
  Qed.

  Theorem bind_ret_r : ∀ `{Setoid A} (p : P A), p >>= ret == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s'' a.
    calc (parses (p >>= ret) s a s'')
      ↔ (∃ (a' : A) (s' : S), parses p s a' s' ∧ parses (ret a' : P A) s' a s'') { (Parser.parses_bind p ret s s'' a ret_proper) }
      ↔ (∃ (a' : A) (s' : S), parses p s a' s' ∧ (a == a' ∧ s'' == s')) { (ex2_and_r _ _ _ (fun a' s' ⇒ Parser.parses_ret a' a s' s'')) }
      ↔ (parses p s a s'') { (ex2_equ_r a s'' _ (parses_proper2 p s)) }.
  Qed.

  Theorem bind_assoc : ∀ `{Setoid A} `{Setoid B} `{Setoid D}
      (p : P A) (f : A → P B) (g : B → P D), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    (p >>= f) >>= g == p >>= (fun a ⇒ f a >>= g).
  Proof.
    intros A eqA EqvA B eqB EqvB D eqD EqvD p f g Hf Hg. apply equ_of_parses. intros s s3 d.
    calc (parses ((p >>= f) >>= g) s d s3)
      ↔ (∃ (b : B) (s2 : S), parses (p >>= f) s b s2 ∧ parses (g b) s2 d s3)
           { (Parser.parses_bind (p >>= f) g s s3 d Hg) }
      ↔ (∃ (b : B) (s2 : S), (∃ (a : A) (s1 : S), parses p s a s1 ∧ parses (f a) s1 b s2) ∧ parses (g b) s2 d s3)
           { (ex2_and_l _ _ _ (fun b s2 ⇒ Parser.parses_bind p f s s2 b Hf)) }
      ↔ (∃ (a : A) (s1 : S), parses p s a s1 ∧ (∃ (b : B) (s2 : S), parses (f a) s1 b s2 ∧ parses (g b) s2 d s3))
           { (ex2_assoc _ _ _) }
      ↔ (∃ (a : A) (s1 : S), parses p s a s1 ∧ parses (f a >>= g) s1 d s3)
           { (ex2_and_r _ _ _ (fun a s1 ⇒ Parser.parses_bind (f a) g s1 s3 d Hg)) }
      ↔ (parses (p >>= (fun a ⇒ f a >>= g)) s d s3)
           { (Parser.parses_bind p (fun a ⇒ f a >>= g) s s3 d (bind_fun_proper f g Hf Hg)) }.
  Qed.

Choice and failure

  Theorem alt_assoc : ∀ `{Setoid A} (p q r : P A), (p <|> q) <|> r == p <|> (q <|> r).
  Proof.
    intros A eqA EqvA p q r. apply equ_of_parses. intros s s' a.
    calc (parses ((p <|> q) <|> r) s a s')
      ↔ (parses (p <|> q) s a s' ∨ parses r s a s') { (Parser.parses_alt (p <|> q) r s s' a) }
      ↔ ((parses p s a s' ∨ parses q s a s') ∨ parses r s a s') { (Parser.parses_alt p q s s' a) }
      ↔ (parses p s a s' ∨ (parses q s a s' ∨ parses r s a s')) { (or_assoc _ _ _) }
      ↔ (parses p s a s' ∨ parses (q <|> r) s a s') { (Parser.parses_alt q r s s' a) }
      ↔ (parses (p <|> (q <|> r)) s a s') { (Parser.parses_alt p (q <|> r) s s' a) }.
  Qed.

  Theorem alt_comm : ∀ `{Setoid A} (p q : P A), p <|> q == q <|> p.
  Proof.
    intros A eqA EqvA p q. apply equ_of_parses. intros s s' a.
    calc (parses (p <|> q) s a s')
      ↔ (parses p s a s' ∨ parses q s a s') { (Parser.parses_alt p q s s' a) }
      ↔ (parses q s a s' ∨ parses p s a s') { (or_comm _ _) }
      ↔ (parses (q <|> p) s a s') { (Parser.parses_alt q p s s' a) }.
  Qed.

  Theorem alt_idem : ∀ `{Setoid A} (p : P A), p <|> p == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s' a.
    calc (parses (p <|> p) s a s')
      ↔ (parses p s a s' ∨ parses p s a s') { (Parser.parses_alt p p s s' a) }
      ↔ (parses p s a s') { (or_same _) }.
  Qed.

  Theorem alt_fail_l : ∀ `{Setoid A} (p : P A), (fail : P A) <|> p == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s' a.
    calc (parses ((fail : P A) <|> p) s a s')
      ↔ (parses (fail : P A) s a s' ∨ parses p s a s') { (Parser.parses_alt fail p s s' a) }
      ↔ (False ∨ parses p s a s') { (or_iff_morphism _ _ (parses_fail_iff s s' a) _ _ (iff_refl _)) }
      ↔ (parses p s a s') { (or_false_l _) }.
  Qed.

  Theorem alt_fail_r : ∀ `{Setoid A} (p : P A), p <|> fail == p.
  Proof.
    intros A eqA EqvA p. apply equ_of_parses. intros s s' a.
    calc (parses (p <|> (fail : P A)) s a s')
      ↔ (parses p s a s' ∨ parses (fail : P A) s a s') { (Parser.parses_alt p fail s s' a) }
      ↔ (parses p s a s' ∨ False) { (or_iff_morphism _ _ (iff_refl _) _ _ (parses_fail_iff s s' a)) }
      ↔ (parses p s a s') { (or_false_r _) }.
  Qed.

  Theorem bind_fail_l : ∀ `{Setoid A} `{Setoid B} (f : A → P B),
    Proper (equ ==> equ) f → (fail : P A) >>= f == fail.
  Proof.
    intros A eqA EqvA B eqB EqvB f Hf. apply equ_of_parses. intros s s'' b.
    calc (parses ((fail : P A) >>= f) s b s'')
      ↔ (∃ (a : A) (s' : S), parses (fail : P A) s a s' ∧ parses (f a) s' b s'') { (Parser.parses_bind fail f s s'' b Hf) }
      ↔ (∃ (a : A) (s' : S), False ∧ parses (f a) s' b s'') { (ex2_and_l _ _ _ (fun a s' ⇒ parses_fail_iff s s' a)) }
      ↔ False { (ex2_false_l _) }
      ↔ (parses (fail : P B) s b s'') { (parses_fail_iff s s'' b) }.
  Qed.

  Theorem bind_fail_r : ∀ `{Setoid A} `{Setoid B} (p : P A),
    p >>= (fun _ ⇒ fail) == (fail : P B).
  Proof.
    intros A eqA EqvA B eqB EqvB p. apply equ_of_parses. intros s s'' b.
    calc (parses (p >>= (fun _ ⇒ fail) : P B) s b s'')
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ parses (fail : P B) s' b s'') { (Parser.parses_bind p (fun _ ⇒ fail) s s'' b (const_fun_proper fail)) }
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ False) { (ex2_and_r _ _ _ (fun a s' ⇒ parses_fail_iff s' s'' b)) }
      ↔ False { (ex2_false_r _) }
      ↔ (parses (fail : P B) s b s'') { (parses_fail_iff s s'' b) }.
  Qed.

  Theorem bind_alt_l : ∀ `{Setoid A} `{Setoid B} (p q : P A) (f : A → P B),
    Proper (equ ==> equ) f → (p <|> q) >>= f == (p >>= f) <|> (q >>= f).
  Proof.
    intros A eqA EqvA B eqB EqvB p q f Hf. apply equ_of_parses. intros s s'' b.
    calc (parses ((p <|> q) >>= f) s b s'')
      ↔ (∃ (a : A) (s' : S), parses (p <|> q) s a s' ∧ parses (f a) s' b s'')
           { (Parser.parses_bind (p <|> q) f s s'' b Hf) }
      ↔ (∃ (a : A) (s' : S), (parses p s a s' ∨ parses q s a s') ∧ parses (f a) s' b s'')
           { (ex2_and_l _ _ _ (fun a s' ⇒ Parser.parses_alt p q s s' a)) }
      ↔ ((∃ (a : A) (s' : S), parses p s a s' ∧ parses (f a) s' b s'')
           ∨ (∃ (a : A) (s' : S), parses q s a s' ∧ parses (f a) s' b s''))
           { (ex2_or_l _ _ _) }
      ↔ (parses (p >>= f) s b s'' ∨ parses (q >>= f) s b s'')
           { (or_iff_morphism _ _ (Parser.parses_bind p f s s'' b Hf) _ _ (Parser.parses_bind q f s s'' b Hf)) }
      ↔ (parses ((p >>= f) <|> (q >>= f)) s b s'')
           { (Parser.parses_alt (p >>= f) (q >>= f) s s'' b) }.
  Qed.

  Lemma alt_fun_proper : ∀ `{Setoid A} `{Setoid B} (f g : A → P B),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g → Proper (equ ==> equ) (fun a ⇒ f a <|> g a).
  Proof.
    intros A eqA EqvA B eqB EqvB f g Hf Hg a a' Ha. apply equ_of_parses. intros s s' b.
    calc (parses (f a <|> g a) s b s')
      ↔ (parses (f a) s b s' ∨ parses (g a) s b s') { (Parser.parses_alt (f a) (g a) s s' b) }
      ↔ (parses (f a') s b s' ∨ parses (g a') s b s') { (or_iff_morphism _ _ (parses_of_equ _ _ (Hf a a' Ha) s s' b) _ _ (parses_of_equ _ _ (Hg a a' Ha) s s' b)) }
      ↔ (parses (f a' <|> g a') s b s') { (Parser.parses_alt (f a') (g a') s s' b) }.
  Qed.

  Theorem bind_alt_r : ∀ `{Setoid A} `{Setoid B} (p : P A) (f g : A → P B),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    p >>= (fun a ⇒ f a <|> g a) == (p >>= f) <|> (p >>= g).
  Proof.
    intros A eqA EqvA B eqB EqvB p f g Hf Hg. apply equ_of_parses. intros s s'' b.
    calc (parses (p >>= (fun a ⇒ f a <|> g a)) s b s'')
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ parses (f a <|> g a) s' b s'')
           { (Parser.parses_bind p (fun a ⇒ f a <|> g a) s s'' b (alt_fun_proper f g Hf Hg)) }
      ↔ (∃ (a : A) (s' : S), parses p s a s' ∧ (parses (f a) s' b s'' ∨ parses (g a) s' b s''))
           { (ex2_and_r _ _ _ (fun a s' ⇒ Parser.parses_alt (f a) (g a) s' s'' b)) }
      ↔ ((∃ (a : A) (s' : S), parses p s a s' ∧ parses (f a) s' b s'')
           ∨ (∃ (a : A) (s' : S), parses p s a s' ∧ parses (g a) s' b s''))
           { (ex2_or_r _ _ _) }
      ↔ (parses (p >>= f) s b s'' ∨ parses (p >>= g) s b s'')
           { (or_iff_morphism _ _ (Parser.parses_bind p f s s'' b Hf) _ _ (Parser.parses_bind p g s s'' b Hg)) }
      ↔ (parses ((p >>= f) <|> (p >>= g)) s b s'')
           { (Parser.parses_alt (p >>= f) (p >>= g) s s'' b) }.
  Qed.

The derived combinators

  Lemma fmap_fun_proper : ∀ `{Setoid A} `{Setoid B} (f : A → B),
    Proper (equ ==> equ) f → Proper (equ ==> equ) (fun a ⇒ (ret (f a) : P B)).
  Proof. intros A eqA EqvA B eqB EqvB f Hf a a' Ha. apply ret_proper. apply Hf. exact Ha. Qed.

  Theorem parses_fmap : ∀ `{Setoid A} `{Setoid B} (f : A → B) (p : P A),
    Proper (equ ==> equ) f → ∀ (s s' : S) (b : B),
    parses (f <$> p) s b s' ↔ ∃ a : A, parses p s a s' ∧ b == f a.
  Proof.
    intros A eqA EqvA B eqB EqvB f p Hf s s' b.
    calc (parses (f <$> p) s b s')
      ↔ (∃ (a : A) (s1 : S), parses p s a s1 ∧ parses (ret (f a) : P B) s1 b s')
           { (Parser.parses_bind p (fun a ⇒ ret (f a)) s s' b (fmap_fun_proper f Hf)) }
      ↔ (∃ (a : A) (s1 : S), parses p s a s1 ∧ (b == f a ∧ s' == s1))
           { (ex2_and_r _ _ _ (fun a s1 ⇒ Parser.parses_ret (f a) b s1 s')) }
      ↔ (∃ a : A, parses p s a s' ∧ b == f a)
           { _ }.
    split.
    - intros [a [s1 [Hp [Hb Hs]]]]. ∃ a. split; [ exact (parses_rest p s a s1 s' (symmetry Hs) Hp) | exact Hb ].
    - intros [a [Hp Hb]]. ∃ a, s'. split; [ exact Hp | split; [ exact Hb | reflexivity ] ].
  Qed.

  Theorem fmap_id : ∀ `{Setoid A} (p : P A), (fun a ⇒ a) <$> p == p.
  Proof. intros A eqA EqvA p. exact (bind_ret_r p). Qed.

  Lemma comp_proper : ∀ `{Setoid A} `{Setoid B} `{Setoid D} (f : A → B) (g : B → D),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g → Proper (equ ==> equ) (fun a ⇒ g (f a)).
  Proof. intros A eqA EqvA B eqB EqvB D eqD EqvD f g Hf Hg a a' Ha. exact (Hg _ _ (Hf a a' Ha)). Qed.

  Theorem fmap_comp : ∀ `{Setoid A} `{Setoid B} `{Setoid D}
      (f : A → B) (g : B → D) (p : P A), Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    g <$> (f <$> p) == (fun a ⇒ g (f a)) <$> p.
  Proof.
    intros A eqA EqvA B eqB EqvB D eqD EqvD f g p Hf Hg. apply equ_of_parses. intros s s' d.
    calc (parses (g <$> (f <$> p)) s d s')
      ↔ (∃ b : B, parses (f <$> p) s b s' ∧ d == g b)
           { (parses_fmap g (f <$> p) Hg s s' d) }
      ↔ (∃ b : B, (∃ a : A, parses p s a s' ∧ b == f a) ∧ d == g b)
           { (ex_iff_morphism _ _ (fun b ⇒ and_iff_morphism _ _ (parses_fmap f p Hf s s' b) _ _ (iff_refl _))) }
      ↔ (∃ a : A, parses p s a s' ∧ d == g (f a))
           { _ }
      ↔ (parses ((fun a ⇒ g (f a)) <$> p) s d s')
           { (parses_fmap _ p (comp_proper f g Hf Hg) s s' d) }.
    split.
    - intros [b [[a [Hp Hb]] Hd]]. ∃ a. split; [ exact Hp | ].
      calc d == (g b) { Hd } == (g (f a)) { (Hg _ _ Hb) }.
    - intros [a [Hp Hd]]. ∃ (f a). split; [ ∃ a; split; [ exact Hp | reflexivity ] | exact Hd ].
  Qed.

  Theorem parses_seq_r : ∀ `{Setoid A} `{Setoid B} (p : P A) (q : P B) (s s'' : S) (b : B),
    parses (p *> q) s b s'' ↔ ∃ (a : A) (s' : S), parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A eqA EqvA B eqB EqvB p q s s'' b.
    exact (Parser.parses_bind p (fun _ ⇒ q) s s'' b (const_fun_proper q)).
  Qed.

  Lemma seq_l_fun_proper : ∀ `{Setoid A} `{Setoid B} (q : P B),
    Proper (equ ==> equ) (fun a : A ⇒ q >>= fun _ ⇒ (ret a : P A)).
  Proof.
    intros A eqA EqvA B eqB EqvB q x y Hxy.
    apply bind_proper_r; [ exact (const_fun_proper _) | exact (const_fun_proper _) | intros _; exact (ret_proper x y Hxy) ].
  Qed.

  Theorem parses_seq_l : ∀ `{Setoid A} `{Setoid B} (p : P A) (q : P B) (s s'' : S) (a : A),
    parses (p <* q) s a s'' ↔ ∃ (s' : S) (b : B), parses p s a s' ∧ parses q s' b s''.
  Proof.
    intros A eqA EqvA B eqB EqvB p q s s'' a.
    calc (parses (p <* q) s a s'')
      ↔ (∃ (a' : A) (s' : S), parses p s a' s' ∧ parses (q >>= fun _ ⇒ (ret a' : P A)) s' a s'')
           { (Parser.parses_bind p _ s s'' a (seq_l_fun_proper q)) }
      ↔ (∃ (a' : A) (s' : S), parses p s a' s' ∧ (∃ (b : B) (s2 : S), parses q s' b s2 ∧ parses (ret a' : P A) s2 a s''))
           { (ex2_and_r _ _ _ (fun a' s' ⇒ Parser.parses_bind q (fun _ ⇒ ret a') s' s'' a (const_fun_proper _))) }
      ↔ (∃ (a' : A) (s' : S), parses p s a' s' ∧ (∃ (b : B) (s2 : S), parses q s' b s2 ∧ (a == a' ∧ s'' == s2)))
           { (ex2_and_r _ _ _ (fun a' s' ⇒ ex2_and_r _ _ _ (fun b s2 ⇒ Parser.parses_ret a' a s2 s''))) }
      ↔ (∃ (s' : S) (b : B), parses p s a s' ∧ parses q s' b s'')
           { _ }.
    split.
    - intros [a' [s' [Hp [b [s2 [Hq [Ha Hs]]]]]]]. ∃ s', b.
      split; [ exact (parses_result p s a' a s' (symmetry Ha) Hp) | exact (parses_rest q s' b s2 s'' (symmetry Hs) Hq) ].
    - intros [s' [b [Hp Hq]]]. ∃ a, s'. split; [ exact Hp | ∃ b, s''; split; [ exact Hq | split; reflexivity ] ].
  Qed.

Pairs are compared componentwise, whatever setoid they carry.
  Section Pairs.
    Context `{Setoid A} `{Setoid B} `{Setoid (A × B)}
      (Hpair : ∀ (a a' : A) (b b' : B), (a, b) == (a', b') ↔ a == a' ∧ b == b').

    Lemma pair_in_proper : ∀ a0 : A, Proper (equ ==> equ) (fun b0 : B ⇒ (ret (a0, b0) : P (A × B))).
    Proof. intros a0 x y Hxy. apply ret_proper. apply Hpair. split; [ reflexivity | exact Hxy ]. Qed.

    Lemma pair_out_proper : ∀ q : P B, Proper (equ ==> equ) (fun a0 : A ⇒ q >>= fun b0 ⇒ (ret (a0, b0) : P (A × B))).
    Proof.
      intros q x y Hxy.
      apply bind_proper_r; [ exact (pair_in_proper x) | exact (pair_in_proper y) | ].
      intros b0. apply ret_proper. apply Hpair. split; [ exact Hxy | reflexivity ].
    Qed.

    Theorem parses_pair : ∀ (p : P A) (q : P B) (s s'' : S) (a : A) (b : B),
      parses (p ⊗ q) s (a, b) s'' ↔ ∃ s' : S, parses p s a s' ∧ parses q s' b s''.
    Proof.
      intros p q s s'' a b.
      calc (parses (p ⊗ q) s (a, b) s'')
        ↔ (∃ (a' : A) (s' : S), parses p s a' s' ∧ parses (q >>= fun b0 ⇒ (ret (a', b0) : P (A × B))) s' (a, b) s'')
             { (Parser.parses_bind p _ s s'' (a, b) (pair_out_proper q)) }
        ↔ (∃ (a' : A) (s' : S), parses p s a' s'
               ∧ (∃ (b' : B) (s2 : S), parses q s' b' s2 ∧ parses (ret (a', b') : P (A × B)) s2 (a, b) s''))
             { (ex2_and_r _ _ _ (fun a' s' ⇒ Parser.parses_bind q _ s' s'' (a, b) (pair_in_proper a'))) }
        ↔ (∃ (a' : A) (s' : S), parses p s a' s'
               ∧ (∃ (b' : B) (s2 : S), parses q s' b' s2 ∧ ((a, b) == (a', b') ∧ s'' == s2)))
             { (ex2_and_r _ _ _ (fun a' s' ⇒ ex2_and_r _ _ _ (fun b' s2 ⇒ Parser.parses_ret (a', b') (a, b) s2 s''))) }
        ↔ (∃ s' : S, parses p s a s' ∧ parses q s' b s'')
             { _ }.
      split.
      - intros [a' [s' [Hp [b' [s2 [Hq [Hab Hs]]]]]]]. destruct (proj1 (Hpair a a' b b') Hab) as [Ha Hb].
        ∃ s'. split; [ exact (parses_result p s a' a s' (symmetry Ha) Hp) | ].
        exact (parses_rest q s' b s2 s'' (symmetry Hs) (parses_result q s' b' b s2 (symmetry Hb) Hq)).
      - intros [s' [Hp Hq]]. ∃ a, s'. split; [ exact Hp | ].
        ∃ b, s''. split; [ exact Hq | split; [ apply Hpair; split; reflexivity | reflexivity ] ].
    Qed.
  End Pairs.

  Section Symbols.
    Context {EC : @EquDec C eqC}.

    Lemma sym_fun_proper : ∀ c : C,
      Proper (equ ==> equ) (fun c0 : C ⇒ if EquDec.equ_dec c0 c then (ret c0 : P C) else fail).
    Proof.
      intros c x y Hxy.
      exact (Theory.Decidable.if_dec_congr _ _ _ _ _ _ (equiv_iff Hxy (reflexivity c)) (ret_proper x y Hxy) (reflexivity fail)).
    Qed.

    Theorem parses_sym : ∀ (c : C) (s s' : S) (c' : C),
      parses (sym c : P C) s c' s' ↔ c' == c ∧ step s c' s'.
    Proof.
      intros c s s' c'.
      calc (parses (sym c : P C) s c' s')
        ↔ (∃ (c0 : C) (s0 : S), parses (item : P C) s c0 s0
               ∧ parses (if EquDec.equ_dec c0 c then (ret c0 : P C) else fail) s0 c' s')
             { (Parser.parses_bind item _ s s' c' (sym_fun_proper c)) }
        ↔ (∃ (c0 : C) (s0 : S), step s c0 s0
               ∧ parses (if EquDec.equ_dec c0 c then (ret c0 : P C) else fail) s0 c' s')
             { (ex2_and_l _ _ _ (fun c0 s0 ⇒ Parser.parses_item s s0 c0)) }
        ↔ (c' == c ∧ step s c' s')
             { _ }.
      split.
      - intros [c0 [s0 [Hst Hr]]]. destruct (EquDec.equ_dec c0 c) as [Hc | Hc].
        + destruct (proj1 (Parser.parses_ret c0 c' s0 s') Hr) as [Hc' Hs].
          split; [ exact (transitivity Hc' Hc) | ].
          exact (proj1 (Sequence.step_proper s s (reflexivity s) c0 c' (symmetry Hc') s0 s' (symmetry Hs)) Hst).
        + destruct (Parser.parses_fail s0 s' c' Hr).
      - intros [Hc Hs]. ∃ c', s'. split; [ exact Hs | ].
        destruct (EquDec.equ_dec c' c) as [_ | Hn]; [ | destruct (Hn Hc) ].
        exact (proj2 (Parser.parses_ret c' c' s' s') (conj (reflexivity c') (reflexivity s'))).
    Qed.

    Theorem parses_sym_not : ∀ (c : C) (s s' : S) (c' : C),
      ¬ (c' == c ∧ step s c' s') → ¬ parses (sym c : P C) s c' s'.
    Proof.
      intros c s s' c' Hn Hp. exact (Hn (proj1 (parses_sym c s s' c') Hp)).
    Qed.
  End Symbols.
