From Riktig.Theory Require Export InitialAlgebra Coproducts Signature.
From Riktig.Lib Require Import Calc.

Section NaturalNumbersTheory.
  Context {O : Type} `{Bicartesian O} {N : O}
    {build : HasBuild O (NaturalNumbers.shape O) N} {fold : HasCata O (NaturalNumbers.shape O) N}
    {NN : NaturalNumbers O N build fold}.

  Local Notation unbuild :=
    (fold (NaturalNumbers.shape O N) (fmap (F := NaturalNumbers.shape O) build)).

Taking a number apart after building it exposes the level.
  Theorem unbuild_zero :
    unbuild ∘ (build ∘ inj1) == (inj1 : term ~> NaturalNumbers.shape O N).
  Proof.
    calc (unbuild ∘ (build ∘ inj1))
      == (unbuild ∘ build ∘ inj1) { Category.comp_assoc }
      == (cat_id (x := NaturalNumbers.shape O N) ∘ inj1) { unbuild_build }
      == (inj1 : term ~> NaturalNumbers.shape O N) { left_identity }.
  Qed.

  Theorem unbuild_succ :
    unbuild ∘ (build ∘ inj2) == (inj2 : N ~> NaturalNumbers.shape O N).
  Proof.
    calc (unbuild ∘ (build ∘ inj2))
      == (unbuild ∘ build ∘ inj2) { Category.comp_assoc }
      == (cat_id (x := NaturalNumbers.shape O N) ∘ inj2) { unbuild_build }
      == (inj2 : N ~> NaturalNumbers.shape O N) { left_identity }.
  Qed.

The successor is a split mono: descending with zero on the left retracts it.
  Theorem pred_succ :
    desc (build ∘ inj1) cat_id ∘ unbuild ∘ (build ∘ inj2) == cat_id (x := N).
  Proof.
    calc (desc (build ∘ inj1) cat_id ∘ unbuild ∘ (build ∘ inj2))
      == (desc (build ∘ inj1) cat_id ∘ (unbuild ∘ (build ∘ inj2))) { Category.comp_assoc }
      == (desc (build ∘ inj1) cat_id ∘ inj2) { unbuild_succ }
      == (cat_id (x := N)) { Coproducts.desc_inj2 }.
  Qed.
End NaturalNumbersTheory.

Induction, from the one theorem for every signature: nothing number-specific remains to prove.
Section NaturalNumbersInduction.
  Context {N : Type}
    {nb : HasBuild Type (NaturalNumbers.shape Type) N} {nf : HasCata Type (NaturalNumbers.shape Type) N}
    `{!NaturalNumbers Type N nb nf}.

  Theorem naturals_induction : ∀ P : N → Prop,
    P (nb (inl tt)) → (∀ n : N, P n → P (nb (inr n))) → ∀ n : N, P n.
  Proof.
    intros P Hz Hs.
    apply (datatype_induction (fun A : Type ⇒ A) (natS (S := sig Type)) (T := N)
      (tb := nb) (tf := nf) (IA := NaturalNumbers.initial)).
    intros s [s' Es]. destruct s' as [[] | [n pn]]; cbn in Es; subst; cbn.
    - exact Hz.
    - exact (Hs n pn).
  Qed.
End NaturalNumbersInduction.
