Ascii characters as the lambda grammar's lexicon: blank is space, tab or a line break, names are made of letters and the underscore. The classes are booleans underneath, so deciding them computes.
Definition ascii_spaceb (c : ascii) : bool :=
  let n := nat_of_ascii c in
  Nat.eqb n 32 || Nat.eqb n 10 || Nat.eqb n 13 || Nat.eqb n 9.
Definition ascii_digitb (c : ascii) : bool :=
  let n := nat_of_ascii c in Nat.leb 48 n && Nat.leb n 57.
Definition ascii_alphab (c : ascii) : bool :=
  let n := nat_of_ascii c in
  (Nat.leb 97 n && Nat.leb n 122) || (Nat.leb 65 n && Nat.leb n 90) || Nat.eqb n 95.

Definition ascii_space (c : ascii) : Prop := ascii_spaceb c = true.
Definition ascii_alpha (c : ascii) : Prop := ascii_alphab c = true.
Definition ascii_digit (c : ascii) : Prop := ascii_digitb c = true.
Definition ascii_digit_value (c : ascii) : nat := nat_of_ascii c - 48.

Instance Ascii_Lexicon
  : Lexicon ascii nat ascii_space ascii_alpha ascii_digit ascii_digit_value
      "092"%char "."%char "("%char ")"%char "="%char ";"%char ":"%char "-"%char ">"%char.
Proof.
  split.
  - exact eq_equivalence.
  - intros a b Hab. change (a = b) in Hab. subst. reflexivity.
  - intros a b Hab. change (a = b) in Hab. subst. reflexivity.
  - intros a b Hab. change (a = b) in Hab. subst. reflexivity.
  - intros a b Hab. change (a = b) in Hab. subst. reflexivity.
  - intro c. unfold ascii_space.
    destruct (ascii_spaceb c); [ left; reflexivity | right; discriminate ].
  - intro c. unfold ascii_alpha.
    destruct (ascii_alphab c); [ left; reflexivity | right; discriminate ].
  - intro c. unfold ascii_digit.
    destruct (ascii_digitb c); [ left; reflexivity | right; discriminate ].
Defined.

Names -- the identifiers the lexicon's characters spell -- carry the list carrier's own equality; only its decision is new.
Instance Name_EquDec : EquDec name (eqA := eq) :=
  fun x y ⇒ List.list_eq_dec Ascii.ascii_dec x y.
