Module GeometricAlgebra.
  Class GeometricAlgebra (F V A : Type) `{InnerProduct.InnerProduct F V}
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
      {Ao : One A} {Am : HasMult A} {As : HasScale F A} := {
    equivalence : Equivalence (@equ A _);
    associative : ∀ x y w : A, (x + y) + w == x + (y + w);
    left_identity : ∀ x : A, 0 + x == x;
    right_identity : ∀ x : A, x + 0 == x;
    left_inverse : ∀ x : A, (- x) + x == 0;
    right_inverse : ∀ x : A, x + (- x) == 0;
    commutative : ∀ x y : A, x + y == y + x;
    mul_associative : ∀ x y w : A, (x × y) × w == x × (y × w);
    mul_left_identity : ∀ x : A, 1 × x == x;
    mul_right_identity : ∀ x : A, x × 1 == x;
    left_distributive : ∀ x y w : A, x × (y + w) == x × y + x × w;
    right_distributive : ∀ x y w : A, (x + y) × w == x × w + y × w;
    add_proper : ∀ x y : A, x == y → ∀ u v : A, u == v → (x + u) == (y + v);
    neg_proper : ∀ x y : A, x == y → (- x) == (- y);
    mul_proper : ∀ x y : A, x == y → ∀ u v : A, u == v → (x × u) == (y × v);
    scale_proper : ∀ a b : F, a == b → ∀ x y : A, x == y → a *: x == b *: y;
    scale_vector_add : ∀ (a : F) (x y : A), a *: (x + y) == a *: x + a *: y;
    scale_scalar_add : ∀ (a b : F) (x : A), (a + b) *: x == a *: x + b *: x;
    scale_assoc : ∀ (a b : F) (x : A), (a × b) *: x == a *: (b *: x);
    scale_identity : ∀ x : A, 1 *: x == x;
    scale_mul_left : ∀ (a : F) (x y : A), (a *: x) × y == a *: (x × y);
    scale_mul_right : ∀ (a : F) (x y : A), x × (a *: y) == a *: (x × y);
    vec : V → A;
    vec_proper : ∀ u v : V, u == v → vec u == vec v;
    vec_additive : ∀ u v : V, vec (u + v) == vec u + vec v;
    vec_scale : ∀ (a : F) (u : V), vec (a *: u) == a *: vec u;
    contraction : ∀ u : V, vec u × vec u == norm2 u *: 1
  }.
End GeometricAlgebra.
