From Riktig.Structure Require Export Field.

A scalar action of F: scaling distributes over vector addition and over scalar addition, successive scalings compose by field multiplication, and 1 acts as the identity.
Module ScaleLaws.
  Class ScaleLaws (F V : Type) `{Field F}
      {Ve : EqRel V} (add : HasAdd V) (scale : HasScale F V) := {
    scale_proper :: Proper (equ ==> equ ==> equ) scale;
    is_left_distributive :: LeftDistributive F V V scale;
    is_right_distributive :: RightDistributive F V V scale;
    is_associative :: Associative F V scale mult;
    is_left_identity :: LeftIdentity F V scale 1;
  }.
  Hint Mode ScaleLaws - ! - - - - - - - - - - - : typeclass_instances.
End ScaleLaws.
Export ScaleLaws.

A vector space over F is an abelian group of vectors with a scalar action of F.
Module VectorSpace.
  Class VectorSpace (F V : Type) `{Field F}
      {Ve : EqRel V} (Vz : Zero V) (Va : HasAdd V) (Vn : HasNeg V)
      (Vs : HasScale F V) := {
    vectors :: AbGroup V 0 add neg;
    scaling :: ScaleLaws F V add scale;
  }.
  Hint Mode VectorSpace - ! - - - - - - - - - - - - - : typeclass_instances.
End VectorSpace.
Export VectorSpace.
