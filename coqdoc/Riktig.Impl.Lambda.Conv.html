Inductive conv : tm → tm → Prop :=
  | conv_step : ∀ t t', step t t' → conv t t'
  | conv_abs : ∀ b b', conv b b' → conv (Abs b) (Abs b')
  | conv_appl : ∀ f f' a, conv f f' → conv (App f a) (App f' a)
  | conv_appr : ∀ f a a', conv a a' → conv (App f a) (App f a')
  | conv_refl : ∀ t, conv t t
  | conv_sym : ∀ t u, conv t u → conv u t
  | conv_trans : ∀ t u v, conv t u → conv u v → conv t v.

Instance Tm_EqRel : EqRel tm := conv.

Instance Tm_Equivalence : Equivalence conv.
Proof.
  split.
  - exact conv_refl.
  - exact conv_sym.
  - exact conv_trans.
Qed.

Instance Abs_Proper : Proper (equ ==> equ) Abs.
Proof.
  intros b b' Hb. apply conv_abs. exact Hb.
Qed.

Instance App_Proper : Proper (equ ==> equ ==> equ) App.
Proof.
  intros f f' Hf a a' Ha.
  eapply conv_trans.
  - apply conv_appl. exact Hf.
  - apply conv_appr. exact Ha.
Qed.


Lemma conv_of_star : ∀ t t', star t t' → t == t'.
Proof.
  intros t t' Hs. induction Hs as [t0 | t0 u v Hstep _ IH].
  - apply conv_refl.
  - eapply conv_trans.
    + apply conv_step. exact Hstep.
    + exact IH.
Qed.

Lemma conv_of_eval : ∀ n t, t == eval n t.
Proof.
  intros n t. apply conv_of_star. apply eval_sound.
Qed.

Lemma conv_by_eval : ∀ n m s t, eval n s = eval m t → s == t.
Proof.
  intros n m s t Heq.
  calc s
    == (eval n s) { (conv_of_eval n s) }
    == (eval m t) { Heq }
    == t { (conv_of_eval m t) }.
Qed.

Lemma conv_beta : ∀ b v, value v → App (Abs b) v == subst 0 v b.
Proof.
  intros b v Hv. apply conv_step. apply step_beta. exact Hv.
Qed.


Example conv_id_id : App idt idt == idt.
Proof. apply (conv_by_eval 10 10). reflexivity. Qed.

Example conv_k : App (App k idt) (Abs (Abs (Var 0))) == idt.
Proof. apply (conv_by_eval 10 10). reflexivity. Qed.


From Riktig Require Import Theory.Lambda.

Lemma conv_ren : ∀ t t' r, conv t t' → conv (ren r t) (ren r t').
Proof.
  intros t t' r Hc. revert r.
  induction Hc as [t0 t1 Hs | b b' Hc IH | f f' a Hc IH | f a a' Hc IH
                   | t0 | t0 u Hc IH | t0 u v Hc1 IH1 Hc2 IH2]; intros r; simpl.
  - apply conv_step. apply step_ren. exact Hs.
  - apply conv_abs. apply IH.
  - apply conv_appl. apply IH.
  - apply conv_appr. apply IH.
  - apply conv_refl.
  - apply conv_sym. apply IH.
  - eapply conv_trans.
    + apply IH1.
    + apply IH2.
Qed.

Definition tmsub : Type := nat → tm.

Instance Tm_HasVar : HasVar tm := Var.
Instance Tm_HasLam : HasLam tm := Abs.
Instance Tm_HasApp : HasApp tm := App.
Instance Tm_HasSub : HasSub tm tmsub := fun t s ⇒ psub s t.
Instance Tmsub_HasSubId : HasSubId tmsub := Var.
Instance Tmsub_HasSubShift : HasSubShift tmsub := fun n ⇒ Var (S n).
Instance Tmsub_HasSubCons : HasSubCons tm tmsub := scons.
Instance Tmsub_HasSubUp : HasSubUp tmsub := pup.
Instance Tm_HasVal : HasVal tm := value.

Instance Tm_LambdaModel : LambdaModel tm tmsub.
Proof.
  constructor.
  - exact Tm_Equivalence.
  - exact Abs_Proper.
  - exact App_Proper.
  -
    intros t t' Ht.
    unfold substitute, Tm_HasSub, sub_shift, Tmsub_HasSubShift.
    rewrite !(ren_as_psub _ S). apply conv_ren. exact Ht.
  -
    intros t. unfold substitute, Tm_HasSub, sub_id, Tmsub_HasSubId.
    rewrite psub_var. reflexivity.
  - intros u s. reflexivity.
  - intros f a s. reflexivity.
  - intros b s. reflexivity.
  - intros s. reflexivity.
  -
    intros n s.
    unfold substitute, Tm_HasSub, sub_shift, Tmsub_HasSubShift, var, Tm_HasVar.
    rewrite (ren_as_psub _ S). reflexivity.
  - intros n. reflexivity.
  -
    intros t u s.
    unfold substitute, Tm_HasSub, sub_shift, Tmsub_HasSubShift,
      sub_cons, Tmsub_HasSubCons.
    rewrite (ren_as_psub _ S), psub_scons_ren_shift. reflexivity.
  -
    intros t s.
    unfold substitute, Tm_HasSub, sub_shift, Tmsub_HasSubShift,
      sub_up, Tmsub_HasSubUp.
    rewrite !(ren_as_psub _ S), psub_ren, ren_psub. reflexivity.
  -
    intros b u Hu.
    unfold app, Tm_HasApp, lam, Tm_HasLam, substitute, Tm_HasSub,
      sub_cons, Tmsub_HasSubCons, sub_id, Tmsub_HasSubId.
    rewrite <- subst_zero_as_psub. apply conv_beta. exact Hu.
  - intros b. constructor.
Qed.


Example tm_church_if_true : ∀ g h : tm, value g → value h →
  church_if church_true g h == g.
Proof.
  intros g h Hg Hh. exact (church_if_true g h Hg Hh).
Qed.

Example tm_church_fst_pair : ∀ v w : tm, value v → value w →
  app (app (app church_pair v) w) church_true == v.
Proof.
  intros v w Hv Hw. exact (church_fst_pair v w Hv Hw).
Qed.
