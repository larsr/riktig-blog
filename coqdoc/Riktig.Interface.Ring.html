From Coq Require Export Morphisms.

Module Ring.
  Class Ring (R : Type) {Re : EqRel R} {Rz : Zero R} {Ra : HasAdd R} {Rn : HasNeg R}
             {Ro : One R} {Rm : HasMult R} := {
    equivalence : Equivalence (@equ R _);
    associative : ∀ x y w : R, (x + y) + w == x + (y + w);
    left_identity : ∀ x : R, 0 + x == x;
    right_identity : ∀ x : R, x + 0 == x;
    left_inverse : ∀ x : R, (- x) + x == 0;
    right_inverse : ∀ x : R, x + (- x) == 0;
    commutative : ∀ x y : R, x + y == y + x;
    mul_associative : ∀ x y w : R, (x × y) × w == x × (y × w);
    mul_left_identity : ∀ x : R, 1 × x == x;
    mul_right_identity : ∀ x : R, x × 1 == x;
    left_distributive : ∀ x y w : R, x × (y + w) == x × y + x × w;
    right_distributive : ∀ x y w : R, (x + y) × w == x × w + y × w;
    add_proper : ∀ x y : R, x == y → ∀ u v : R, u == v → (x + u) == (y + v);
    neg_proper : ∀ x y : R, x == y → (- x) == (- y);
    mul_proper : ∀ x y : R, x == y → ∀ u v : R, u == v → (x × u) == (y × v)
  }.
End Ring.
Notation Ring := Ring.Ring.
