Reading lambda terms and definitions, over structures alone: any characters a Lexicon describes, any parser structure over them, any list structure for the names and for the binders in scope, any term representation with the three constructors, any dictionary for the definitions. Syntax: \x y. body abstracts, juxtaposition applies, parentheses group, a number is a Church numeral, and a definition is name = term ;. A name in a term is the innermost binder of that name, or else the definition it was given earlier. The fuel bounds every loop.

Characters and tokens.
  Definition csym (c : C) : P C :=
    p_sat (fun d ⇒ c == d) (fun d ⇒ EquDec.equ_dec c d).
  Definition spaces (fuel : N) : P unit :=
    p_fold fuel (p_sat space Lexicon.space_dec) tt (fun _ _ ⇒ tt).
  Definition token {A : Type} (fuel : N) (p : P A) : P A :=
    p >>= (fun a ⇒ spaces fuel >>= (fun _ ⇒ ret a)).

A name, built as it is read.
  Definition ident (fuel : N) : P K :=
    token fuel
      (p_sat alpha Lexicon.alpha_dec >>= (fun c ⇒
       p_fold fuel (p_sat alpha Lexicon.alpha_dec) (empty : K) prepend >>= (fun k ⇒
       ret (prepend c k)))).

Numbers, as Church numerals.
  Definition ten : N := succ (succ (succ (succ (succ (succ (succ (succ (succ (succ 0))))))))).
  Definition number (fuel : N) : P N :=
    token fuel
      (p_sat digit Lexicon.digit_dec >>= (fun d0 ⇒
       p_foldl fuel (p_sat digit Lexicon.digit_dec)
         (digit_value d0) (fun acc d ⇒ acc × ten + digit_value d))).
  Definition church (n : N) : T :=
    lam (lam (natrec T (var 0) (fun _ acc ⇒ app (var 1) acc) n)).

The innermost binder holding a name: the names are compared by their decided equality.
  Definition debruijn (x : K) : E → option N :=
    list_rec (option N) None
      (fun k acc ⇒ if EquDec.equ_dec k x then Some 0
                    else match acc with Some n ⇒ Some (succ n) | None ⇒ None end).

The grammar, carrying the dictionary of earlier definitions and the binders in scope; natrec is the loop.
  Definition EnvParser : Type := D → E → P T.

  Definition term_step (fuel : N) (rec : EnvParser × EnvParser) : EnvParser :=
    fun d env ⇒
      (token fuel (csym backslash) >>= (fun _ ⇒
       ident fuel >>= (fun k0 ⇒
       p_foldl fuel (ident fuel)
         ((fun e : E ⇒ prepend k0 e), (fun t : T ⇒ lam t))
         (fun (ew : (E → E) × (T → T)) k ⇒
            ((fun e ⇒ prepend k (fst ew e)), (fun t ⇒ lam (snd ew t)))) >>= (fun ew ⇒
       token fuel (csym dot_c) >>= (fun _ ⇒
       fst rec d (fst ew env) >>= (fun b ⇒
       ret (snd ew b)))))))
      <|>
      (snd rec d env >>= (fun a0 ⇒
       p_foldl fuel (snd rec d env) a0 (fun f x ⇒ app f x))).

  Definition atom_step (fuel : N) (rec : EnvParser × EnvParser) : EnvParser :=
    fun d env ⇒
      (ident fuel >>= (fun x ⇒
         match debruijn x env with
         | Some n ⇒ ret (var n : T)
         | None ⇒ match find d x with
                   | Some t ⇒ ret t
                   | None ⇒ fail
                   end
         end))
      <|>
      ((number fuel >>= (fun n ⇒ ret (church n)))
       <|>
       (token fuel (csym lparen) >>= (fun _ ⇒
        fst rec d env >>= (fun t ⇒
        token fuel (csym rparen) >>= (fun _ ⇒ ret t))))).

  Definition term_p (fuel : N) : EnvParser :=
    fst (natrec (EnvParser × EnvParser)
           ((fun _ _ ⇒ fail), (fun _ _ ⇒ fail))
           (fun _ rec ⇒ (term_step fuel rec, atom_step fuel rec)) fuel).

One definition, read against what came before, and all of them: each is bound in the dictionary the next is read against.
  Definition def_p (fuel : N) (d : D) : P (K × T) :=
    ident fuel >>= (fun x ⇒
    token fuel (csym eqsign) >>= (fun _ ⇒
    term_p fuel d (empty : E) >>= (fun t ⇒
    token fuel (csym semi) >>= (fun _ ⇒ ret (x, t))))).

  Definition defs_loop (fuel : N) : D → P D :=
    natrec (D → P D)
      (fun d ⇒ ret d)
      (fun _ rec d ⇒
        (def_p fuel d >>= (fun kt ⇒ rec (insert (fst kt) (snd kt) d)))
        <|> (ret d))
      fuel.

  Definition defs_p (fuel : N) : P D :=
    spaces fuel >>= (fun _ ⇒ defs_loop fuel (empty : D)).
