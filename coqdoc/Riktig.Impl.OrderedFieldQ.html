     OrderedField.le_total := Qle_total;
     OrderedField.le_stable := Qle_stable;
     OrderedField.add_le_mono_l := Qplus_le_mono_l;
     OrderedField.mul_nonneg := Qmult_le_0_compat;

     OrderedField.abs_neg := Qabs_opp |}.
