Library Riktig.Impl.NamedTermRecNamed

From Riktig.Structure Require Export NamedTermRec.
From Riktig.Lib Require Export Coq.Named.

The inductive named terms are named terms with recursion: the constructors are the operations, and the recursion computes.
Section NamedTermRecNamed.
  Context {K N : Type} {eqK : EqRel K} {eqN : EqRel N}.

  #[export] Instance Named_EqRel : EqRel (named K N) := eq.
  #[export] Instance Named_HasNamedVar : HasNamedVar K (named K N) := NVar.
  #[export] Instance Named_HasNum : HasNum N (named K N) := NNum.
  #[export] Instance Named_HasNamedLam : HasNamedLam K (named K N) := NLam.
  #[export] Instance Named_HasApp : HasApp (named K N) := NApp.
  #[export] Instance Named_HasNamedTermRec : HasNamedTermRec K N (named K N) := @named_fold K N.

  #[export] Instance Named_NamedTermRec :
    NamedTermRec K N (named K N) Named_HasNamedVar Named_HasNum Named_HasNamedLam Named_HasApp Named_HasNamedTermRec := {|
    NamedTermRec.ntermrec_proper := fun B eB EB v c l a _ _ _ _ t t' E ⇒
      eq_ind t (fun u ⇒ named_fold B v c l a t == named_fold B v c l a u) (reflexivity _) t' E;
    NamedTermRec.ntermrec_var := fun B eB EB v c l a x ⇒ reflexivity _;
    NamedTermRec.ntermrec_num := fun B eB EB v c l a n ⇒ reflexivity _;
    NamedTermRec.ntermrec_lam := fun B eB EB v c l a x b ⇒ reflexivity _;
    NamedTermRec.ntermrec_app := fun B eB EB v c l a f u ⇒ reflexivity _;
  |}.
End NamedTermRecNamed.
