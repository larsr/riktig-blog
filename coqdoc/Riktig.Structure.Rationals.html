From Riktig.Structure Require Export Field Integers RingHom.

The rationals over a ring of integers: a field with an injective ring homomorphism from the integers, in which every element is a ratio of two of them.
Module Rationals.
  Class Rationals (Q Z : Type) `{Field Q} `{Integers Z} (cast : Z → Q) := {
    cast_hom :: RingHom Z Q cast;
    cast_injective :: Injective Z Q cast;
    is_fraction : ∀ q : Q, ∃ a b : Z,
      cast b =/= 0 ∧ q × cast b == cast a;
  }.
  Hint Mode Rationals ! - - - - - - - - - - - - - - - - - : typeclass_instances.
End Rationals.
Export Rationals.
