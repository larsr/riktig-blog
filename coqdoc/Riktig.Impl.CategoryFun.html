Coq's types and functions: a cartesian closed category with an initial object and coproducts, arrows equal when they agree at every argument.
Instance Fun_HasHom : HasHom Type := fun X Y ⇒ X → Y.
Instance Fun_EqRel : ∀ X Y : Type, EqRel (X ~> Y) := fn_equ.
Instance Fun_HasId : HasId Type := fun X (x : X) ⇒ x.
Instance Fun_HasComp : HasComp Type := fun X Y Z g f x ⇒ g (f x).

Instance Fun_Category : Category Type Fun_HasId Fun_HasComp :=
  {| Category.hom_equivalence := fun X Y ⇒
       Build_Equivalence _ (fn_equ_refl X Y) (fn_equ_sym X Y) (fn_equ_trans X Y);
     Category.comp_proper := fun X Y Z g g' Hg f f' Hf ⇒ fn_comp_ext X Y Z g g' f f' Hg Hf;
     Category.is_left_identity := fun X Y f x ⇒ eq_refl;
     Category.is_right_identity := fun X Y f x ⇒ eq_refl;
     Category.comp_assoc := fun W X Y Z f g h w ⇒ eq_refl |}.

Instance Fun_HasTerm : HasTerm Type := unit.
Instance Fun_HasToTerm : HasToTerm Type := fun X (_ : X) ⇒ tt.
Instance Fun_Terminal : Terminal Type Fun_HasTerm Fun_HasToTerm :=
  {| Terminal.to_term_unique := unit_unique |}.

Instance Fun_HasInit : HasInit Type := Empty_set.
Instance Fun_HasFromInit : HasFromInit Type := fun X (e : Empty_set) ⇒ match e with end.
Instance Fun_Initial : Initial Type Fun_HasInit Fun_HasFromInit :=
  {| Initial.from_init_unique := empty_unique |}.

Instance Fun_HasProd : HasProd Type := prod.
Instance Fun_HasPi1 : HasPi1 Type := fun A B ⇒ @fst A B.
Instance Fun_HasPi2 : HasPi2 Type := fun A B ⇒ @snd A B.
Instance Fun_HasLift : HasLift Type := fun X A B f g x ⇒ (f x, g x).
Instance Fun_Products : Products Type Fun_HasProd Fun_HasPi1 Fun_HasPi2 Fun_HasLift :=
  {| Products.lift_proper := fun X A B f f' Hf g g' Hg ⇒ pair_ext X A B f f' g g' Hf Hg;
     Products.pi1_lift := fun X A B f g x ⇒ eq_refl;
     Products.pi2_lift := fun X A B f g x ⇒ eq_refl;
     Products.lift_unique := pair_unique |}.

Instance Fun_HasCoprod : HasCoprod Type := sum.
Instance Fun_HasInj1 : HasInj1 Type := fun A B ⇒ @inl A B.
Instance Fun_HasInj2 : HasInj2 Type := fun A B ⇒ @inr A B.
Instance Fun_HasDesc : HasDesc Type := case_fn.
Instance Fun_Coproducts : Coproducts Type Fun_HasCoprod Fun_HasInj1 Fun_HasInj2 Fun_HasDesc :=
  {| Coproducts.desc_proper := fun X A B f f' Hf g g' Hg ⇒ case_ext X A B f f' g g' Hf Hg;
     Coproducts.desc_inj1 := fun X A B f g a ⇒ eq_refl;
     Coproducts.desc_inj2 := fun X A B f g b ⇒ eq_refl;
     Coproducts.desc_unique := case_unique |}.

Instance Fun_HasExp : HasExp Type := fun A B ⇒ A → B.
Instance Fun_HasEval : HasEval Type := fun A B (p : (A → B) × A) ⇒ fst p (snd p).
Instance Fun_HasCurry : HasCurry Type := fun X A B (h : X × A → B) x a ⇒ h (x, a).
Instance Fun_Exponentials : Exponentials Type Fun_HasExp Fun_HasEval Fun_HasCurry.
Proof.
  split.
  - exact (fun X A B h h' Hh ⇒ curry_ext X A B h h' Hh).
  - exact eval_curry_fn.
  - exact curry_unique_fn.
Qed.

Instance Fun_CartesianClosed : CartesianClosed Type Fun_HasTerm Fun_HasToTerm
    Fun_HasProd Fun_HasPi1 Fun_HasPi2 Fun_HasLift Fun_HasExp Fun_HasEval Fun_HasCurry :=
  {| CartesianClosed.has_terminal := Fun_Terminal;
     CartesianClosed.has_products := Fun_Products;
     CartesianClosed.has_exponentials := Fun_Exponentials |}.
