The conjugate of a complex number.
Notation "'conj' z" := (re z *: 1 + - (im z *: imaginary)) (at level 10, z at level 9).

The product, in parts: the textbook formula.
  Theorem mul_parts : ∀ z w : C,
    z × w == (re z × re w + - (im z × im w)) *: (1 : C) + (re z × im w + im z × re w) *: (imaginary : C).
  Proof.
    intros z w.
    calc (z × w)
      == ((re z *: (1 : C) + im z *: (imaginary : C)) × w) { (re_im z) }
      == ((re z *: (1 : C) + im z *: (imaginary : C)) × (re w *: (1 : C) + im w *: (imaginary : C))) { (re_im w) }
      == ((re z × re w + - (im z × im w)) *: (1 : C) + (re z × im w + im z × re w) *: (imaginary : C))
           { (combination_mul imaginary i_sq (re z) (im z) (re w) (im w)) }.
  Qed.

  Lemma re_combination_parts : ∀ a b : F,
    re (a *: (1 : C) + b *: (imaginary : C)) == a × re (1 : C) + b × re (imaginary : C).
  Proof.
    intros a b.
    calc (re (a *: (1 : C) + b *: (imaginary : C)))
      == (re (a *: (1 : C)) + re (b *: (imaginary : C))) { (additive (f := re) (a *: (1 : C)) (b *: (imaginary : C))) }
      == (a × re (1 : C) + re (b *: (imaginary : C))) { (homogeneous (f := re) a (1 : C)) }
      == (a × re (1 : C) + b × re (imaginary : C)) { (homogeneous (f := re) b (imaginary : C)) }.
  Qed.

  Lemma re_combination : ∀ a b : F, re (a *: (1 : C) + b *: (imaginary : C)) == a.
  Proof.
    intros a b.
    calc (re (a *: (1 : C) + b *: (imaginary : C)))
      == (a × re (1 : C) + b × re (imaginary : C)) { re_combination_parts }
      == (a × 1 + b × re (imaginary : C)) { re_one }
      == (a × 1 + b × 0) { re_i }
      == (a + b × 0) { (right_identity (action := mult) a) }
      == (a + 0) { (mul_zero b) }
      == a { (right_identity (action := add) a) }.
  Qed.

  Lemma im_combination_parts : ∀ a b : F,
    im (a *: (1 : C) + b *: (imaginary : C)) == a × im (1 : C) + b × im (imaginary : C).
  Proof.
    intros a b.
    calc (im (a *: (1 : C) + b *: (imaginary : C)))
      == (im (a *: (1 : C)) + im (b *: (imaginary : C))) { (additive (f := im) (a *: (1 : C)) (b *: (imaginary : C))) }
      == (a × im (1 : C) + im (b *: (imaginary : C))) { (homogeneous (f := im) a (1 : C)) }
      == (a × im (1 : C) + b × im (imaginary : C)) { (homogeneous (f := im) b (imaginary : C)) }.
  Qed.

  Lemma im_combination : ∀ a b : F, im (a *: (1 : C) + b *: (imaginary : C)) == b.
  Proof.
    intros a b.
    calc (im (a *: (1 : C) + b *: (imaginary : C)))
      == (a × im (1 : C) + b × im (imaginary : C)) { im_combination_parts }
      == (a × 0 + b × im (imaginary : C)) { im_one }
      == (a × 0 + b × 1) { im_i }
      == (0 + b × 1) { (mul_zero a) }
      == (0 + b) { (right_identity (action := mult) b) }
      == b { (left_identity (action := add) b) }.
  Qed.

  Theorem re_mul : ∀ z w : C, re (z × w) == re z × re w + - (im z × im w).
  Proof.
    intros z w.
    calc (re (z × w))
      == (re ((re z × re w + - (im z × im w)) *: (1 : C) + (re z × im w + im z × re w) *: (imaginary : C)))
           { (mul_parts z w) }
      == (re z × re w + - (im z × im w)) { re_combination }.
  Qed.

  Theorem im_mul : ∀ z w : C, im (z × w) == re z × im w + im z × re w.
  Proof.
    intros z w.
    calc (im (z × w))
      == (im ((re z × re w + - (im z × im w)) *: (1 : C) + (re z × im w + im z × re w) *: (imaginary : C)))
           { (mul_parts z w) }
      == (re z × im w + im z × re w) { im_combination }.
  Qed.

  Lemma mul_conj_parts : ∀ z : C,
    z × conj z == (re z × re z + - (im z × - im z)) *: (1 : C) + (re z × - im z + im z × re z) *: (imaginary : C).
  Proof.
    intros z.
    calc (z × conj z)
      == (z × (re z *: (1 : C) + (- im z) *: (imaginary : C))) { (neg_smul (im z) (imaginary : C)) }
      == ((re z *: (1 : C) + im z *: (imaginary : C)) × (re z *: (1 : C) + (- im z) *: (imaginary : C))) { (re_im z) }
      == ((re z × re z + - (im z × - im z)) *: (1 : C) + (re z × - im z + im z × re z) *: (imaginary : C))
           { (combination_mul imaginary i_sq (re z) (im z) (re z) (- im z)) }.
  Qed.

  Lemma conj_real_part : ∀ z : C, re z × re z + - (im z × - im z) == re z × re z + im z × im z.
  Proof.
    intros z.
    calc (re z × re z + - (im z × - im z))
      == (re z × re z + - - (im z × im z)) { (mul_neg (im z) (im z)) }
      == (re z × re z + im z × im z) { (neg_neg (im z × im z)) }.
  Qed.

  Lemma conj_imaginary_part : ∀ z : C, re z × - im z + im z × re z == 0.
  Proof.
    intros z.
    calc (re z × - im z + im z × re z)
      == (- (re z × im z) + im z × re z) { (mul_neg (re z) (im z)) }
      == (- (im z × re z) + im z × re z) { (commutative (op := mult) (re z) (im z)) }
      == (0 : F) { (left_inverse (op := add) (im z × re z)) }.
  Qed.

A number times its conjugate is the square of its modulus.
  Theorem mul_conj : ∀ z : C, z × conj z == (re z × re z + im z × im z) *: 1.
  Proof.
    intros z.
    calc (z × conj z)
      == ((re z × re z + - (im z × - im z)) *: (1 : C) + (re z × - im z + im z × re z) *: (imaginary : C))
           { mul_conj_parts }
      == ((re z × re z + im z × im z) *: (1 : C) + (re z × - im z + im z × re z) *: (imaginary : C))
           { (conj_real_part z) }
      == ((re z × re z + im z × im z) *: (1 : C) + (0 : F) *: (imaginary : C))
           { (conj_imaginary_part z) }
      == ((re z × re z + im z × im z) *: (1 : C) + 0)
           { (zero_smul (imaginary : C)) }
      == ((re z × re z + im z × im z) *: (1 : C))
           { (right_identity (action := add) ((re z × re z + im z × im z) *: (1 : C))) }.
  Qed.
