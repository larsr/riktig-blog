Library Riktig.Impl.FiniteFin

From Riktig.Structure Require Export Finite.
From Riktig.Impl Require Export NatOrderNat.
From Riktig.Lib Require Import Coq.Fin.

Coq's Fin.t (S n) is a finite index of count S n, numbered by nat. The empty Fin.t 0 is not one: numbering it would need an element to send a number to.
Instance Fin_EqRel n : EqRel (Fin.t (S n)) := eq.

Instance Fin_HasCard n : HasCard (Fin.t (S n)) nat := S n.
Instance Fin_HasIndex n : HasIndex (Fin.t (S n)) nat := fin_index.
Instance Fin_HasElem n : HasElem (Fin.t (S n)) nat := fin_elem.

Instance Fin_Finite n :
  Finite (Fin.t (S n)) nat (Fin_HasCard n) (Fin_HasIndex n) (Fin_HasElem n) (eqI := eq) :=
  {| Finite.equivalence := eq_equivalence;
     Finite.index_proper := f_equal fin_index;
     Finite.elem_proper := f_equal fin_elem;
     Finite.index_lt := fin_index_lt n;
     Finite.elem_index := fin_elem_index n;
     Finite.index_elem := fin_index_elem n |}.
