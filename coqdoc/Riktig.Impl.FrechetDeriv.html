Library Riktig.Impl.FrechetDeriv

From Coq Require Import Morphisms Setoid.
From Riktig Require Export Theory.DerivSpec Theory.Derivative
  Theory.BoundedLinear.

Record NormObj (F : Type) {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F}
    {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F} {Fi : HasInv F}
    {FF : Field F} {Fl : HasLe F} {Fab : HasAbs F} {OF : OrderedField F} := {
  no_space : VectObj F;
  no_norm : HasNorm F (vo_car no_space);
  no_normed : NormedSpace F (vo_car no_space) (Vnorm := no_norm)
}.

Arguments no_space {F Fe Fz Fa Fn Fo Fm Fi FF Fl Fab OF}.
Arguments no_norm {F Fe Fz Fa Fn Fo Fm Fi FF Fl Fab OF}.
Arguments no_normed {F Fe Fz Fa Fn Fo Fm Fi FF Fl Fab OF}.

#[global] Existing Instance no_norm.
#[global] Existing Instance no_normed.

Section FrechetInstance.
  Context {F : Type} `{OF : OrderedField F}.

  Definition frechet_admissible (X Y : NormObj F)
    (L : LinearMap F (vo_car (no_space X)) (vo_car (no_space Y))) : Prop :=
    linear_map_bounded L.

  Definition frechet_differential (X Y : NormObj F)
    (f : vo_car (no_space X) → vo_car (no_space Y))
    (a : vo_car (no_space X))
    (L : LinearMap F (vo_car (no_space X)) (vo_car (no_space Y))) : Prop :=
    is_deriv f a L.

  #[export] Instance Frechet_AdmissibleArrows :
    AdmissibleArrows F (NormObj F) no_space frechet_admissible := {|
    admissible_zero := fun X Y ⇒ bounded_const0;
    admissible_identity := fun X ⇒ bounded_id;
    admissible_sum := fun X Y ⇒ bounded_add;
    admissible_scaled := fun X Y ⇒ bounded_scale;
    admissible_negated := fun X Y ⇒ bounded_neg;
    admissible_composite := fun X Y Z ⇒ bounded_comp
  |}.

  #[export] Instance Frechet_DerivSpec : DerivSpec F (NormObj F) no_space := {|
    admissible := frechet_admissible;
    deriv_arrows := Frechet_AdmissibleArrows;
    differential := frechet_differential;
    deriv_of_linear := fun X Y ⇒ deriv_lm;
    deriv_of_constant := fun X Y ⇒ deriv_const;
    deriv_of_sum := fun X Y ⇒ deriv_add;
    deriv_of_scaled := fun X Y ⇒ deriv_scale;
    deriv_of_composite := fun X Y Z ⇒ deriv_comp;
    deriv_respects_arrow := fun X Y ⇒ is_deriv_L_equ;
    deriv_respects_function := fun X Y ⇒ is_deriv_fun_equ;
    differential_unique := fun X Y ⇒ deriv_unique
  |}.

End FrechetInstance.

Definition normObj (F : Type) {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F}
    {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F} {Fi : HasInv F}
    {FF : Field F} {Fl : HasLe F} {Fab : HasAbs F} {OF : OrderedField F}
    (T : Type) {e : EqRel T} {z : Zero T} {ad : HasAdd T} {ng : HasNeg T}
    {sc : HasScale F T} {VS : VectorSpace F T} {nm : HasNorm F T}
    {NS : NormedSpace F T} : NormObj F :=
  {| no_space := vectObj F T; no_norm := nm; no_normed := NS |}.
