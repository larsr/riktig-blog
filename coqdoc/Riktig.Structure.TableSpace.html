Tables of scalars as a vector space: reading an entry respects the zero, the sum and the scaling of tables.
Module TableSpace.
  Class TableSpace (F I T : Type) `{VectorSpace F T} {eqI : EqRel I}
      {get : HasGet T I F} {tabulate : HasTabulate T I F} {TT : Table I F T get tabulate} := {
    get_zero : ∀ i : I, get 0 i == 0;
    get_add : ∀ (s t : T) (i : I), get (s + t) i == get s i + get t i;
    get_smul : ∀ (a : F) (t : T) (i : I), get (a *: t) i == a × get t i;
  }.
  Hint Mode TableSpace - - ! - - - - - - - - - - - - - - - - - - : typeclass_instances.
End TableSpace.
