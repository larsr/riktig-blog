From Coq Require Export Morphisms.
From Riktig.Structure Require Export Naturals.

The simply typed lambda calculus: the same calculus as the untyped one -- reduction is a congruent preorder contracting a redex and passing under substitution, conversion is joinability, confluence makes it an equivalence -- with the abstraction carrying the type it binds, which reduction and substitution leave untouched.
Module StlcCalculus.
  Class StlcCalculus (N Ty T Sb : Type) `{Naturals N} {eqTy : EqRel Ty} {eqT : EqRel T}
      (var_ : HasVar N T) (tylam_ : HasTyLam Ty T) (app_ : HasApp T) (subst_ : HasSubst T Sb)
      (id_ : HasSubId Sb) (shift_ : HasSubShift Sb) (cons_ : HasSubCons T Sb) (up_ : HasSubUp Sb)
      (reduction : HasReduces T) := {
    
    equ_join : ∀ t u : T, t == u ↔ (∃ w : T, t -->* w ∧ u -->* w);
    
    reduces_refl : ∀ t : T, t -->* t;
    reduces_trans : ∀ t u v : T, t -->* u → u -->* v → t -->* v;
    reduces_tylam : ∀ (A : Ty) (b b' : T), b -->* b' → tylam A b -->* tylam A b';
    reduces_app : ∀ f f' a a' : T, f -->* f' → a -->* a' → app f a -->* app f' a';
    reduces_subst : ∀ (t t' : T) (s : Sb), t -->* t' → t.[s] -->* t'.[s];
    
    tylam_proper : ∀ (A A' : Ty) (b : T), A == A' → tylam A b == tylam A' b;
    
    beta : ∀ (A : Ty) (b u : T), app (tylam A b) u -->* b.[u .: sub_id];
    
    var_injective : ∀ n m : N, (var n : T) == var m → n == m;
    
    subst_id : ∀ t : T, t.[sub_id] == t;
    var_zero_cons : ∀ (u : T) (s : Sb), (var 0 : T).[u .: s] == u;
    var_succ_cons : ∀ (n : N) (u : T) (s : Sb), (var (succ n) : T).[u .: s] == (var n : T).[s];
    var_shift : ∀ n : N, (var n : T).[sub_shift] == var (succ n);
    var_zero_up : ∀ s : Sb, (var 0 : T).[sub_up s] == var 0;
    var_succ_up : ∀ (n : N) (s : Sb), (var (succ n) : T).[sub_up s] == ((var n : T).[s]).[sub_shift];
    app_subst : ∀ (f a : T) (s : Sb), (app f a).[s] == app (f.[s]) (a.[s]);
    tylam_subst : ∀ (A : Ty) (b : T) (s : Sb), (tylam A b).[s] == tylam A (b.[sub_up s]);
    shift_cons : ∀ (t u : T) (s : Sb), (t.[sub_shift]).[u .: s] == t.[s];
    shift_up : ∀ (t : T) (s : Sb), (t.[sub_shift]).[sub_up s] == (t.[s]).[sub_shift];
    
    confluence : ∀ t u v : T, t -->* u → t -->* v → ∃ w : T, u -->* w ∧ v -->* w;
  }.
End StlcCalculus.
