Section Derivative.
  Context {F U V : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.

  Definition is_deriv (f : U → V) (a : U) (L : LinearMap F U V) : Prop :=
    linear_map_bounded L ∧
    ∀ eps : F, 0 < eps →
      ∃ delta : F, 0 < delta ∧
        ∀ h : U, ‖ h ‖ < delta →
          ‖ f (a + h) + - (f a + L h) ‖ ≤ eps × ‖ h ‖.

  Lemma deriv_increment_bound : ∀ (f : U → V) (a h : U)
      (L : LinearMap F U V) (e C : F),
    0 ≤ C → (∀ u : U, ‖ L u ‖ ≤ C × ‖ u ‖) → e ≤ 1 →
    ‖ f (a + h) + - (f a + L h) ‖ ≤ e × ‖ h ‖ →
    ‖ f (a + h) + - f a ‖ ≤ (1 + C) × ‖ h ‖.
  Proof.
    intros f a h L e C HC Hb He Hr.
    step == (‖ (f (a + h) + - (f a + L h)) + L h ‖)
      { (sub_add_cancel_r (f (a + h)) (f a) (L h)) }.
    step ≤ (‖ f (a + h) + - (f a + L h) ‖ + ‖ L h ‖)
      { triangle }.
    step ≤ (e × ‖ h ‖ + C × ‖ h ‖)
      { (add_le_mono _ _ _ _ Hr (Hb h)) }.
    step == ((e + C) × ‖ h ‖)
      { right_distributive }.
    apply mul_le_mono_r; [ apply nonneg | apply add_le_mono_r; exact He ].
  Qed.

  Theorem deriv_const : ∀ (v : V) (a : U),
    is_deriv (fun _ : U ⇒ v) a linear_map_const0.
  Proof.
    intros v a.
    split.
    - exact bounded_const0.
    - intros eps Heps.
      ∃ 1.
      split.
      + exact lt_0_1.
      + intros h Hh.
        step == (0 : F) { (const_remainder_zero v) }.
        apply mul_nonneg_closed.
        × apply lt_le. exact Heps.
        × apply nonneg.
  Qed.

  Theorem deriv_lm : ∀ (g : LinearMap F U V) (a : U),
    linear_map_bounded g → is_deriv (LinearMap.map g) a g.
  Proof.
    intros g a Hb.
    split.
    - exact Hb.
    - intros eps Heps.
      ∃ 1.
      split.
      + exact lt_0_1.
      + intros h Hh.
        step == (0 : F) { (lm_remainder_zero g a h) }.
        apply mul_nonneg_closed.
        × apply lt_le. exact Heps.
        × apply nonneg.
  Qed.

  Theorem deriv_add : ∀ (f g : U → V) (a : U)
      (Lf Lg : LinearMap F U V),
    is_deriv f a Lf → is_deriv g a Lg →
    is_deriv (fun u : U ⇒ f u + g u) a (Lf + Lg).
  Proof.
    intros f g a Lf Lg [Hbf HDf] [Hbg HDg].
    split.
    - exact (bounded_add Lf Lg Hbf Hbg).
    - intros eps Heps.
      destruct (HDf (half eps) (half_pos eps Heps)) as [d1 [Hd1 HR1]].
      destruct (HDg (half eps) (half_pos eps Heps)) as [d2 [Hd2 HR2]].
      destruct (pos_min_exists d1 d2 Hd1 Hd2) as [d [Hd [Hdd1 Hdd2]]].
      ∃ d.
      split.
      + exact Hd.
      + intros h Hh.
        step ≤ (‖ f (a + h) + - (f a + Lf h) ‖ + ‖ g (a + h) + - (g a + Lg h) ‖)
          { (add_remainder_split (f (a + h)) (g (a + h)) (f a) (g a)
               (Lf h) (Lg h)) }.
        apply add_half_mul_le.
        × apply HR1. exact (lt_le_trans (‖ h ‖) d d1 Hh Hdd1).
        × apply HR2. exact (lt_le_trans (‖ h ‖) d d2 Hh Hdd2).
  Qed.

  Theorem deriv_scale : ∀ (c : F) (f : U → V) (a : U)
      (Lf : LinearMap F U V),
    is_deriv f a Lf → is_deriv (fun u : U ⇒ c *: f u) a (c *: Lf).
  Proof.
    intros c f a Lf [Hbf HDf].
    split.
    - exact (bounded_scale c Lf Hbf).
    - intros eps Heps.
      destruct (HDf (eps × inv (1 + abs c))
          (pad_div_pos eps (abs c) Heps (abs_nonneg c)))
        as [d [Hd HR]].
      ∃ d.
      split.
      + exact Hd.
      + intros h Hh.
        step == (abs c × ‖ f (a + h) + - (f a + Lf h) ‖)
          { (scale_remainder_norm c (f (a + h)) (f a) (Lf h)) }.
        step ≤ (abs c × (eps × inv (1 + abs c) × ‖ h ‖))
          { (mul_le_mono_l _ _ _ (abs_nonneg c) (HR h Hh)) }.
        exact (mul_pad_div_mul_le eps (abs c) (‖ h ‖)
          (lt_le 0 eps Heps) (abs_nonneg c) (nonneg h)).
  Qed.

  Theorem deriv_neg : ∀ (f : U → V) (a : U) (Lf : LinearMap F U V),
    is_deriv f a Lf → is_deriv (fun u : U ⇒ - f u) a (- Lf).
  Proof.
    intros f a Lf [Hbf HDf].
    split.
    - exact (bounded_neg Lf Hbf).
    - intros eps Heps.
      destruct (HDf eps Heps) as [d [Hd HR]].
      ∃ d.
      split.
      + exact Hd.
      + intros h Hh.
        step == (‖ f (a + h) + - (f a + Lf h) ‖)
          { (neg_remainder_norm (f (a + h)) (f a) (Lf h)) }.
        exact (HR h Hh).
  Qed.

  Lemma deriv_diff_bound : ∀ (f : U → V) (a : U)
      (L1 L2 : LinearMap F U V) (eps : F) (u : U),
    is_deriv f a L1 → is_deriv f a L2 → 0 < eps →
    ‖ L1 u + - L2 u ‖ ≤ eps × ‖ u ‖.
  Proof.
    intros f a L1 L2 eps u [_ HD1] [_ HD2] Heps.
    destruct (HD1 (half eps) (half_pos eps Heps)) as [d1 [Hd1 HR1]].
    destruct (HD2 (half eps) (half_pos eps Heps)) as [d2 [Hd2 HR2]].
    destruct (pos_min_exists d1 d2 Hd1 Hd2) as [d [Hd [Hdd1 Hdd2]]].
    destruct (small_scale_exists d (‖ u ‖) Hd (nonneg u))
      as [t [Ht Htu]].
    assert (Htnorm : ‖ t *: u ‖ == t × ‖ u ‖)
      by exact (norm_scale_nonneg t u (lt_le 0 t Ht)).
    assert (Htu1 : ‖ t *: u ‖ < d1)
      by (rewrite Htnorm; exact (lt_le_trans (t × ‖ u ‖) d d1 Htu Hdd1)).
    assert (Htu2 : ‖ t *: u ‖ < d2)
      by (rewrite Htnorm; exact (lt_le_trans (t × ‖ u ‖) d d2 Htu Hdd2)).
    apply (mul_le_cancel_l (‖ L1 u + - L2 u ‖) (eps × ‖ u ‖) t Ht).
    step == (‖ t *: (L1 u + - L2 u) ‖)
      { (norm_scale_nonneg t (L1 u + - L2 u) (lt_le 0 t Ht)) }.
    step == (‖ t *: L1 u + - (t *: L2 u) ‖)
      { scale_sub }.
    step == (‖ L1 (t *: u) + - (t *: L2 u) ‖)
      { linear_map_scale }.
    step == (‖ L1 (t *: u) + - L2 (t *: u) ‖)
      { linear_map_scale }.
    step == (‖ (f (a + t *: u) + - (f a + L2 (t *: u)))
               + - (f (a + t *: u) + - (f a + L1 (t *: u))) ‖)
      { sub_common_cancel }.
    step ≤ (‖ f (a + t *: u) + - (f a + L2 (t *: u)) ‖
             + ‖ f (a + t *: u) + - (f a + L1 (t *: u)) ‖)
      { norm_sub_le_add }.
    step ≤ (eps × ‖ t *: u ‖)
      { (add_half_mul_le _ _ eps (‖ t *: u ‖)
           (HR2 (t *: u) Htu2) (HR1 (t *: u) Htu1)) }.
    step == (eps × (t × ‖ u ‖))
      { Htnorm }.
    apply eq_le. exact (mul_left_commutative eps t (‖ u ‖)).
  Qed.

  Theorem deriv_unique : ∀ (f : U → V) (a : U)
      (L1 L2 : LinearMap F U V),
    is_deriv f a L1 → is_deriv f a L2 → L1 == L2.
  Proof.
    intros f a L1 L2 HD1 HD2 u.
    apply sub_eq_zero.
    apply definite.
    apply nonneg_below_all_pos_is_zero.
    - apply nonneg.
    - intros eps Heps.
      step ≤ ((eps × inv (1 + ‖ u ‖)) × ‖ u ‖)
        { (deriv_diff_bound f a L1 L2 (eps × inv (1 + ‖ u ‖)) u HD1 HD2
             (pad_div_pos eps (‖ u ‖) Heps (nonneg u))) }.
      step == (‖ u ‖ × (eps × inv (1 + ‖ u ‖)))
        { (commutative (eps × inv (1 + ‖ u ‖)) (‖ u ‖)) }.
      apply mul_pad_div_le.
      + apply lt_le. exact Heps.
      + apply nonneg.
  Qed.


  Lemma is_deriv_L_equ : ∀ (f : U → V) (a : U)
      (L1 L2 : LinearMap F U V),
    L1 == L2 → is_deriv f a L1 → is_deriv f a L2.
  Proof.
    intros f a L1 L2 HL [[C [HC Hbnd]] HD].
    split.
    - ∃ C.
      split.
      + exact HC.
      + intros u. rewrite <- (HL u). exact (Hbnd u).
    - intros eps Heps.
      destruct (HD eps Heps) as [delta [Hd HR]].
      ∃ delta.
      split.
      + exact Hd.
      + intros h Hh. rewrite <- (HL h). exact (HR h Hh).
  Qed.

  Lemma is_deriv_fun_equ : ∀ (f g : U → V) (a : U)
      (L : LinearMap F U V),
    (∀ x : U, f x == g x) → is_deriv f a L → is_deriv g a L.
  Proof.
    intros f g a L Hfg [Hb HD].
    split.
    - exact Hb.
    - intros eps Heps.
      destruct (HD eps Heps) as [d [Hd HR]].
      ∃ d.
      split.
      + exact Hd.
      + intros h Hh.
        rewrite <- (Hfg (a + h)), <- (Hfg a).
        exact (HR h Hh).
  Qed.

  Lemma is_deriv_of_local_eq : ∀ (f g : U → V),
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    ∀ (a : U) (L : LinearMap F U V) (d : F),
    0 < d → (∀ h : U, ‖ h ‖ < d → f (a + h) == g (a + h)) →
    is_deriv g a L → is_deriv f a L.
  Proof.
    intros f g Pf Pg a L d Hd Hfg [Hb HD].
    assert (H0d : ‖ (0 : U) ‖ < d).
    { rewrite norm_zero. exact Hd. }
    
    assert (Hfa0 : f a == g a).
    { calc (f a)
        == (f (a + 0)) { (Pf (a + 0) a (right_identity a)) }
        == (g (a + 0)) { (Hfg 0 H0d) }
        == (g a) { (Pg (a + 0) a (right_identity a)) }. }
    split.
    - exact Hb.
    - intros eps Heps.
      destruct (HD eps Heps) as [d2 [Hd2 HR]].
      destruct (pos_min_exists d d2 Hd Hd2) as [d0 [Hd0 [Hdd Hdd2]]].
      ∃ d0.
      split.
      + exact Hd0.
      + intros h Hh.
        rewrite (Hfg h (lt_le_trans (‖ h ‖) d0 d Hh Hdd)), Hfa0.
        exact (HR h (lt_le_trans (‖ h ‖) d0 d2 Hh Hdd2)).
  Qed.

  Theorem deriv_continuous : ∀ (f : U → V) (a : U)
      (L : LinearMap F U V),
    Proper (equ ==> equ) f → is_deriv f a L → continuous_at f a.
  Proof.
    intros f a L Pf [[C [HC Hb]] HD].
    intros eps Heps.
    destruct (HD 1 lt_0_1) as [d0 [Hd0 HR]].
    destruct (pos_min_exists d0 (eps × inv (1 + (1 + C))) Hd0
        (pad_div_pos eps (1 + C) Heps
           (lt_le 0 (1 + C) (one_plus_nonneg_pos C HC))))
      as [delta [Hdelta [Hdd0 Hdd2]]].
    ∃ delta.
    split.
    - exact Hdelta.
    - intros u Hu.
      assert (Hinc : ‖ f (a + (u + - a)) + - f a ‖ ≤ (1 + C) × ‖ u + - a ‖).
      { apply (deriv_increment_bound f a (u + - a) L 1 C HC Hb).
        - reflexivity.
        - apply HR. exact (lt_le_trans (‖ u + - a ‖) delta d0 Hu Hdd0). }
      
      rewrite <- (add_sub_cancel a u).
      apply (le_lt_trans _ ((1 + C) × ‖ u + - a ‖) eps).
      + exact Hinc.
      + apply mul_lt_of_lt_pad_div.
        × apply lt_le. apply one_plus_nonneg_pos. exact HC.
        × apply nonneg.
        × exact (lt_le_trans (‖ u + - a ‖) delta
            (eps × inv (1 + (1 + C))) Hu Hdd2).
  Qed.

End Derivative.

Section DerivativeComp.
  Context {F U V W : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VSW : VectorSpace F W} {nmW : HasNorm F W} {NSW : NormedSpace F W}.

  Theorem deriv_comp : ∀ (f : U → V) (g : V → W) (a : U)
      (Lf : LinearMap F U V) (Lg : LinearMap F V W),
    Proper (equ ==> equ) g →
    is_deriv f a Lf → is_deriv g (f a) Lg →
    is_deriv (fun u : U ⇒ g (f u)) a (linear_map_comp Lg Lf).
  Proof.
    intros f g a Lf Lg Pg [Hbf HDf] [Hbg HDg].
    split.
    - exact (bounded_comp Lg Lf Hbg Hbf).
    - destruct Hbf as [Cf [HCf HbF]].
      destruct Hbg as [Cg [HCg HbG]].
      intros eps Heps.
      destruct (pos_min_exists 1 (half eps × inv (1 + Cg)) lt_0_1
          (pad_div_pos (half eps) Cg (half_pos eps Heps) HCg))
        as [e1 [He1 [He1_1 He1_g]]].
      destruct (HDf e1 He1) as [df [Hdf HRf]].
      destruct (HDg (half eps × inv (1 + (1 + Cf)))
          (pad_div_pos (half eps) (1 + Cf) (half_pos eps Heps)
             (lt_le 0 (1 + Cf) (one_plus_nonneg_pos Cf HCf))))
        as [dg [Hdg HRg]].
      destruct (pos_min_exists df (dg × inv (1 + Cf)) Hdf
          (pad_div_pos dg Cf Hdg HCf))
        as [d [Hd [Hd_f Hd_g]]].
      ∃ d.
      split.
      + exact Hd.
      + intros h Hh.
        assert (Hrem : ‖ f (a + h) + - (f a + Lf h) ‖ ≤ e1 × ‖ h ‖)
          by exact (HRf h (lt_le_trans (‖ h ‖) d df Hh Hd_f)).
        assert (Hinc : ‖ f (a + h) + - f a ‖ ≤ (1 + Cf) × ‖ h ‖)
          by exact (deriv_increment_bound f a h Lf e1 Cf HCf HbF He1_1 Hrem).
        assert (Hincdg : ‖ f (a + h) + - f a ‖ < dg).
        { apply (le_lt_trans _ ((1 + Cf) × ‖ h ‖) dg).
          - exact Hinc.
          - apply (lt_le_trans _ ((1 + Cf) × (dg × inv (1 + Cf))) dg).
            + apply mul_lt_mono_l.
              × apply one_plus_nonneg_pos. exact HCf.
              × exact (lt_le_trans (‖ h ‖) d (dg × inv (1 + Cf)) Hh Hd_g).
            + apply eq_le.
              apply field_mul_div_cancel.
              apply pos_neq_0.
              apply one_plus_nonneg_pos.
              exact HCf. }
        assert (Htol : 0 ≤ half eps × inv (1 + (1 + Cf)))
          by exact (lt_le 0 (half eps × inv (1 + (1 + Cf)))
                      (pad_div_pos (half eps) (1 + Cf) (half_pos eps Heps)
                         (lt_le 0 (1 + Cf) (one_plus_nonneg_pos Cf HCf)))).
        step ≤ (‖ g (f a + (f (a + h) + - f a))
                   + - (g (f a) + Lg (f (a + h) + - f a)) ‖
                 + ‖ Lg (f (a + h) + - (f a + Lf h)) ‖)
          { (comp_remainder_split g Lg Pg (f a) (f (a + h)) (Lf h)) }.
        apply add_half_mul_le.
        × step ≤ ((half eps × inv (1 + (1 + Cf))) × ‖ f (a + h) + - f a ‖)
            { (HRg (f (a + h) + - f a) Hincdg) }.
          step ≤ ((half eps × inv (1 + (1 + Cf))) × ((1 + Cf) × ‖ h ‖))
            { (mul_le_mono_l _ _ _ Htol Hinc) }.
          exact (pad_div_mul_le (half eps) (1 + Cf) (‖ h ‖)
            (lt_le 0 (half eps) (half_pos eps Heps))
            (lt_le 0 (1 + Cf) (one_plus_nonneg_pos Cf HCf))
            (nonneg h)).
        × step ≤ (Cg × ‖ f (a + h) + - (f a + Lf h) ‖)
            { (HbG (f (a + h) + - (f a + Lf h))) }.
          step ≤ (Cg × (e1 × ‖ h ‖))
            { (mul_le_mono_l _ _ _ HCg Hrem) }.
          step == ((Cg × e1) × ‖ h ‖)
            { associative }.
          apply mul_le_mono_r.
          -- apply nonneg.
          --
             step ≤ (Cg × (half eps × inv (1 + Cg)))
               { (mul_le_mono_l _ _ _ HCg He1_g) }.
             apply mul_pad_div_le.
             ++ apply lt_le. apply half_pos. exact Heps.
             ++ exact HCg.
  Qed.

End DerivativeComp.

Section Partials.
  Context {F U V : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.

  Definition partial_at {n : nat} {I : Type} {eI : EqRel I}
      {Ieq : EquDec I} {IFin : FinIndex n I} {Vl : HasLincomb I F U}
      (L : LinearMap F U V) (B : Basis n I F U)
      (i : I) : V :=
    L (Basis.vec B i).

  Lemma partial_at_welldef : ∀ {n : nat} {I : Type} {eI : EqRel I}
      {Ieq : EquDec I} {IFin : FinIndex n I} {Vl : HasLincomb I F U}
      (f : U → V) (a : U)
      (L1 L2 : LinearMap F U V) (B : Basis n I F U) (i : I),
    is_deriv f a L1 → is_deriv f a L2 →
    partial_at L1 B i == partial_at L2 B i.
  Proof.
    intros n I eI Ieq IFin Vl f a L1 L2 B i HD1 HD2.
    exact (deriv_unique f a L1 L2 HD1 HD2 (Basis.vec B i)).
  Qed.

End Partials.
