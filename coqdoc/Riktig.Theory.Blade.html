Every law of a blade carrier, proved once from membership. Each is three lines: faithfulness, the membership equations, and the bit law of the same name — so a second representation of blades writes its Blade instance and inherits all of this.
Section BladeTheory.
  Context {I K B : Type} `{Boolean K}
    {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B}
    {oB : One B} {mB : HasMult B} {BL : Blade I K B}.

  Lemma member_at_empty : ∀ i : I, member (0 : B) i == (0 : K).
  Proof. exact member_empty. Qed.

  Lemma member_at_full : ∀ i : I, member (1 : B) i == (1 : K).
  Proof. exact member_full. Qed.

  Lemma member_at_add : ∀ (x y : B) (i : I),
    member (x + y) i == member x i + member y i.
  Proof. exact member_symmetric_difference. Qed.

  Lemma member_at_mult : ∀ (x y : B) (i : I),
    member (x × y) i == member x i × member y i.
  Proof. exact member_intersection. Qed.

  Lemma member_at_neg : ∀ (x : B) (i : I),
    member (- x) i == - member x i.
  Proof. exact member_complement. Qed.

  Ltac blade_pointwise :=
    apply member_faithful; intros i;
    repeat first [ rewrite member_at_add | rewrite member_at_mult
                 | rewrite member_at_neg | rewrite member_at_empty
                 | rewrite member_at_full ].

  #[export] Instance blade_add_proper :
    Proper (equ ==> equ ==> equ) (@add B _).
  Proof.
    intros x x' Hx y y' Hy.
    apply member_faithful; intros i.
    calc (member (x + y) i)
      == (member x i + member y i) { (member_at_add x y i) }
      == (member x' i + member y' i)
           { (op_proper _ _ (member_proper x x' Hx i i eq_refl)
                        _ _ (member_proper y y' Hy i i eq_refl)) }
      == (member (x' + y') i) { (member_at_add x' y' i) }.
  Qed.

  #[export] Instance blade_mult_proper :
    Proper (equ ==> equ ==> equ) (@mult B _).
  Proof.
    intros x x' Hx y y' Hy.
    apply member_faithful; intros i.
    calc (member (x × y) i)
      == (member x i × member y i) { (member_at_mult x y i) }
      == (member x' i × member y' i)
           { (op_proper _ _ (member_proper x x' Hx i i eq_refl)
                        _ _ (member_proper y y' Hy i i eq_refl)) }
      == (member (x' × y') i) { (member_at_mult x' y' i) }.
  Qed.

  #[export] Instance blade_neg_proper : Proper (equ ==> equ) (@neg B _).
  Proof.
    intros x x' Hx.
    apply member_faithful; intros i.
    calc (member (- x) i)
      == (- member x i) { (member_at_neg x i) }
      == (- member x' i) { (neg_proper _ _ (member_proper x x' Hx i i eq_refl)) }
      == (member (- x') i) { (member_at_neg x' i) }.
  Qed.

  #[export] Instance blade_add_associative :
    Associative B B add add := ltac:(intros x y z; blade_pointwise; apply associative).
  #[export] Instance blade_add_left_identity :
    LeftIdentity B B add 0 := ltac:(intros x; blade_pointwise; apply left_identity).
  #[export] Instance blade_add_right_identity :
    RightIdentity B B add 0 := ltac:(intros x; blade_pointwise; apply right_identity).
  #[export] Instance blade_add_commutative :
    Commutative B B add := ltac:(intros x y; blade_pointwise; apply commutative).
  #[export] Instance blade_add_left_inverse :
    LeftInverse B add neg 0 := ltac:(intros x; blade_pointwise; apply left_inverse).
  #[export] Instance blade_add_right_inverse :
    RightInverse B add neg 0 := ltac:(intros x; blade_pointwise; apply right_inverse).
  #[export] Instance blade_mult_associative :
    Associative B B mult mult := ltac:(intros x y z; blade_pointwise; apply associative).
  #[export] Instance blade_mult_left_identity :
    LeftIdentity B B mult 1 := ltac:(intros x; blade_pointwise; apply left_identity).
  #[export] Instance blade_mult_right_identity :
    RightIdentity B B mult 1 := ltac:(intros x; blade_pointwise; apply right_identity).
  #[export] Instance blade_left_distributive :
    LeftDistributive B B B mult :=
    ltac:(intros x y z; blade_pointwise;
          apply (left_distributive (member x i) (member y i) (member z i))).
  #[export] Instance blade_right_distributive :
    RightDistributive B B B mult :=
    ltac:(intros x y z; blade_pointwise;
          apply (right_distributive (member x i) (member y i) (member z i))).
  #[export] Instance blade_mult_idempotent :
    Idempotent B mult := ltac:(intros x; blade_pointwise; apply idempotent).

  #[export] Instance Blade_Magma : Magma B add := {}.
  #[export] Instance Blade_Semigroup : Semigroup B add := {}.
  #[export] Instance Blade_Monoid : Monoid B := {}.
  #[export] Instance Blade_Group : Group B := {}.
  #[export] Instance Blade_AbGroup : AbGroup B := {}.
  #[export] Instance Blade_MulMagma : Magma B mult := {}.
  #[export] Instance Blade_MulSemigroup : Semigroup B mult := {}.
  #[export] Instance Blade_MulMonoid : @Monoid B _ oB mB := {}.
  #[export] Instance Blade_Ring : Ring B := {}.

The law Impl/Blade.v chose a vector for: shifting by a blade twice returns the blade. It is characteristic two at the blade carrier, and reindexing a sum by symmetric difference is an involution because of it.
  Theorem blade_shift : ∀ x y : B, x + (x + y) == y.
  Proof. exact add_shift. Qed.
