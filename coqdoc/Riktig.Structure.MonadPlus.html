Generalizable All Variables.

A monad with failure and choice: choice respects equality and, with failure, is a monoid; bind is absorbed by failure on the left and distributes over choice on the left.
Module MonadPlus.
  Class MonadPlus (P : Type → Type) {eqP : ∀ (A : Type) (eqA : EqRel A), EqRel (P A)}
      (return_ : HasReturn P) (bind_ : HasBind P) (fail_ : HasFail P) (alt_ : HasAlt P) := {
    monad :: Monad P return_ bind_;
    alt_proper :: ∀ `{Setoid A}, Proper (equ ==> equ ==> equ) (alt : P A → P A → P A);
    alt_assoc : ∀ `{Setoid A} (p q r : P A), (p <|> q) <|> r == p <|> (q <|> r);
    alt_fail_l : ∀ `{Setoid A} (p : P A), (fail : P A) <|> p == p;
    alt_fail_r : ∀ `{Setoid A} (p : P A), p <|> fail == p;
    bind_fail_l : ∀ `{Setoid A} `{Setoid B} (f : A → P B),
      Proper (equ ==> equ) f → (fail : P A) >>= f == fail;
    bind_alt_l : ∀ `{Setoid A} `{Setoid B} (p q : P A) (f : A → P B),
      Proper (equ ==> equ) f → (p <|> q) >>= f == (p >>= f) <|> (q >>= f);
  }.
End MonadPlus.
