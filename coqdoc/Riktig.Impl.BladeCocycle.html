Lemma bzip_map2 : ∀ (f : bool → bool → bool) (n : nat)
    (v w : Vector.t bool n), bzip f v w = Vector.map2 f v w.
Proof.
  intros f n; induction n as [| m IH]; intros v w.
  - rewrite (blade_0 (bzip f v w)).
    apply (Vector.case0 (fun u ⇒ Vector.nil bool = Vector.map2 f v u)).
    revert v.
    apply (Vector.case0
      (fun u ⇒ Vector.nil bool = Vector.map2 f u (Vector.nil bool))).
    reflexivity.
  - rewrite (blade_S m v), (blade_S m w), bzip_cons, IH. reflexivity.
Qed.

Lemma bxor_is_add : ∀ (n : nat) (v w : Vector.t bool n), bxor v w = v + w.
Proof. intros n v w. exact (bzip_map2 xorb n v w). Qed.

Lemma band_is_mult : ∀ (n : nat) (v w : Vector.t bool n), band v w = v × w.
Proof. intros n v w. exact (bzip_map2 andb n v w). Qed.

Lemma bempty_is_zero : ∀ n : nat, @bempty n = (0 : Vector.t bool n).
Proof.
  induction n as [| m IH]; [ reflexivity |].
  change (@bempty (S m)) with (bcons false (@bempty m)).
  rewrite IH. reflexivity.
Qed.

Lemma band_bempty_l : ∀ (n : nat) (v : Vector.t bool n),
  band bempty v = bempty.
Proof.
  intros n v.
  rewrite <- (bdiff_bxor_self n bempty v), (bxor_bempty_l n v).
  exact (bdiff_bempty_l n v).
Qed.

Lemma band_bempty_r : ∀ (n : nat) (v : Vector.t bool n),
  band v bempty = bempty.
Proof.
  intros n v.
  rewrite <- (bdiff_bxor_self n v bempty), (bxor_bempty_r n v).
  exact (bdiff_self n v).
Qed.

Lemma band_bxor_self : ∀ (n : nat) (v d : Vector.t bool n),
  band v (bxor v d) = bdiff v d.
Proof.
  intros n v d.
  rewrite <- (bdiff_bxor_self n v (bxor v d)), (bxor_involutive n v d).
  reflexivity.
Qed.

Section BladeCocycle.
  Context {F : Type} `{Field F}.
  Add Ring blade_cocycle_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma bsign_cocycle : ∀ (n : nat) (u v w : Vector.t bool n),
    (bsign u v : F) × bsign (bxor u v) w == bsign v w × bsign u (bxor v w).
  Proof.
    intros n u v w. unfold bsign.
    rewrite <- !parity_sign_xor, (bpar_cocycle n u v w). reflexivity.
  Qed.

  Lemma sigprod_cocycle : ∀ (n : nat) (s : Signature F n)
      (u v w : Vector.t bool n),
    sigprod s (band u v) × sigprod s (band (bxor u v) w)
    == sigprod s (band v w) × sigprod s (band u (bxor v w)).
  Proof.
    induction n as [| m IH]; intros s u v w.
    - reflexivity.
    - rewrite !sigprod_S, !bhead_band, !bhead_bxor, !btail_band, !btail_bxor.
      apply pair_mul.
      + destruct (bhead u), (bhead v), (bhead w); simpl; ring.
      + exact (IH (Vector.tl s) (btail u) (btail v) (btail w)).
  Qed.

  Definition blade_cocycle {n : nat} (s : Signature F n)
      (v w : Vector.t bool n) : F := bsign v w × sigprod s (band v w).

  Lemma blade_cocycle_left_normal : ∀ (n : nat) (s : Signature F n)
      (v : Vector.t bool n), blade_cocycle s 0 v == 1.
  Proof.
    intros n s v. unfold blade_cocycle.
    rewrite <- (bempty_is_zero n), (band_bempty_l n v),
            (bsign_bempty_l n v), (sigprod_bempty n s).
    ring.
  Qed.

  Lemma blade_cocycle_right_normal : ∀ (n : nat) (s : Signature F n)
      (v : Vector.t bool n), blade_cocycle s v 0 == 1.
  Proof.
    intros n s v. unfold blade_cocycle.
    rewrite <- (bempty_is_zero n), (band_bempty_r n v),
            (bsign_bempty_r n v), (sigprod_bempty n s).
    ring.
  Qed.

  Lemma blade_cocycle_equation : ∀ (n : nat) (s : Signature F n)
      (u v w : Vector.t bool n),
    blade_cocycle s u v × blade_cocycle s (u + v) w
    == blade_cocycle s v w × blade_cocycle s u (v + w).
  Proof.
    intros n s u v w. unfold blade_cocycle.
    rewrite <- !(bxor_is_add n).
    apply pair_mul; [ exact (bsign_cocycle n u v w)
                    | exact (sigprod_cocycle n s u v w) ].
  Qed.

  #[export] Instance blade_cocycle_proper (n : nat) (s : Signature F n) :
    Proper (equ ==> equ ==> equ) (blade_cocycle s).
  Proof.
    intros u u' Hu v v' Hv.
    rewrite (vector_bool_equ_eq n u u' Hu), (vector_bool_equ_eq n v v' Hv).
    reflexivity.
  Qed.

  #[export] Instance Blade_Cocycle (n : nat) (s : Signature F n) :
    Cocycle F (Vector.t bool n) (blade_cocycle s) :=
    {| Cocycle.cocycle_proper := blade_cocycle_proper n s;
       Cocycle.cocycle_equation := blade_cocycle_equation n s;
       Cocycle.cocycle_left_normal := blade_cocycle_left_normal n s;
       Cocycle.cocycle_right_normal := blade_cocycle_right_normal n s |}.

End BladeCocycle.

Section BladeSumIsEnumerationFold.
  Context {F : Type} `{Field F}.
  Add Ring blade_bridge_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma bigop_list_blade_enum_S : ∀ (m : nat) (f : Vector.t bool (S m) → F),
    bigop_list f (blade_enum (S m))
    == bigop_list (fun v ⇒ f (bcons false v)) (blade_enum m)
       + bigop_list (fun v ⇒ f (bcons true v)) (blade_enum m).
  Proof.
    intros m f. rewrite (blade_enum_S m).
    calc (bigop_list f (List.map (Vector.cons bool false m) (blade_enum m)
                        ++ List.map (Vector.cons bool true m) (blade_enum m)))
      == (bigop_list f (List.map (Vector.cons bool false m) (blade_enum m))
          + bigop_list f (List.map (Vector.cons bool true m) (blade_enum m)))
           { bigop_list_append }
      == (bigop_list (fun v ⇒ f (bcons false v)) (blade_enum m)
          + bigop_list (fun v ⇒ f (bcons true v)) (blade_enum m))
           { (op_proper
                _ _ (bigop_list_map (Vector.cons bool false m) f (blade_enum m))
                _ _ (bigop_list_map (Vector.cons bool true m) f (blade_enum m))) }.
  Qed.

  Lemma bsum_bigop_list : ∀ (n : nat) (f : Vector.t bool n → F),
    bsum f == bigop_list f (blade_enum n).
  Proof.
    induction n as [| m IH]; intros f.
    - calc (bsum f)
        == (f (Vector.nil bool) + (0 : F))
             { (right_identity (f (Vector.nil bool))) }
        == (f (Vector.nil bool) + bigop_list f nil)
             { (bigop_list_nil f) }
        == (bigop_list f (blade_enum 0))
             { (bigop_list_cons f (Vector.nil bool) nil) }.
    - calc (bsum f)
        == (bsum (fun v ⇒ f (bcons false v)) + bsum (fun v ⇒ f (bcons true v)))
             { (bsum_S m f) }
        == (bigop_list (fun v ⇒ f (bcons false v)) (blade_enum m)
            + bigop_list (fun v ⇒ f (bcons true v)) (blade_enum m))
             { (op_proper _ _ (IH (fun v ⇒ f (bcons false v)))
                          _ _ (IH (fun v ⇒ f (bcons true v)))) }
        == (bigop_list f (blade_enum (S m)))
             { (bigop_list_blade_enum_S m f) }.
  Qed.

  Lemma bsum_is_bigopI : ∀ (n : nat) (f : Vector.t bool n → F),
    bsum f == bigopI f.
  Proof. intros n f. exact (bsum_bigop_list n f). Qed.

  Lemma blade_cocycle_at_partner : ∀ (n : nat) (s : Signature F n)
      (v d : Vector.t bool n),
    blade_cocycle s v (v + d) == bsign v (bxor v d) × sigprod s (bdiff v d).
  Proof.
    intros n s v d. unfold blade_cocycle.
    rewrite <- (bxor_is_add n v d), (band_bxor_self n v d). reflexivity.
  Qed.

  Theorem twist_is_cl_mul : ∀ (n : nat) (s : Signature F n) (x y : Cl F n),
    twist (blade_cocycle s) x y == cl_mul s x y.
  Proof.
    intros n s x y d. unfold twist, cl_mul.
    transitivity (bsum (fun v ⇒ blade_cocycle s v (v + d) × (x v × y (v + d)))).
    - symmetry.
      exact (bsum_is_bigopI n
        (fun v ⇒ blade_cocycle s v (v + d) × (x v × y (v + d)))).
    - apply bsum_ext. intros v.
      rewrite (blade_cocycle_at_partner n s v d), (bxor_is_add n v d).
      reflexivity.
  Qed.

  Corollary cl_mul_assoc_by_cocycle : ∀ (n : nat) (s : Signature F n)
      (x y z : Cl F n),
    cl_mul s (cl_mul s x y) z == cl_mul s x (cl_mul s y z).
  Proof.
    intros n s x y z.
    calc (cl_mul s (cl_mul s x y) z)
      == (twist (blade_cocycle s) (cl_mul s x y) z)
           { (twist_is_cl_mul n s (cl_mul s x y) z) }
      == (twist (blade_cocycle s) (twist (blade_cocycle s) x y) z)
           { (twist_proper (blade_cocycle s)
                _ _ (symmetry (twist_is_cl_mul n s x y))
                _ _ (reflexivity z)) }
      == (twist (blade_cocycle s) x (twist (blade_cocycle s) y z))
           { (twist_assoc (blade_cocycle s) x y z
                (vector_blade_families_proper n y)
                (vector_blade_families_proper n z)) }
      == (twist (blade_cocycle s) x (cl_mul s y z))
           { (twist_proper (blade_cocycle s)
                _ _ (reflexivity x)
                _ _ (twist_is_cl_mul n s y z)) }
      == (cl_mul s x (cl_mul s y z))
           { (twist_is_cl_mul n s x (cl_mul s y z)) }.
  Qed.

End BladeSumIsEnumerationFold.


Definition blade_family {F : Type} {eF : EqRel F} {zF : Zero F} {oF : One F}
    {n : nat} (v : Vector.t bool n) : Vector.t bool n → F :=
  fun w ⇒ delta w v.

Section BladeCocycleComputes.

  Definition euclidean_signature : Signature Q 3 := Vector.const (1 : Q) 3.
  Definition minkowski_signature : Signature Q 3 :=
    Vector.cons Q (-1) 2 (Vector.const (1 : Q) 2).

  Notation e1 := (generator Fin.F1).
  Notation e2 := (generator (Fin.FS Fin.F1)).
  Notation e12 := (generator Fin.F1 + generator (Fin.FS Fin.F1)).

  Example cocycle_e1_e2 : blade_cocycle euclidean_signature e1 e2 = (1 : Q).
  Proof. vm_compute. reflexivity. Qed.

  Example cocycle_e2_e1 : blade_cocycle euclidean_signature e2 e1 = (-1)%Q.
  Proof. vm_compute. reflexivity. Qed.

  Example e1_times_e2 :
    twist (blade_cocycle euclidean_signature)
      (blade_family e1) (blade_family e2) e12 = (1 : Q).
  Proof. vm_compute. reflexivity. Qed.

  Example e2_times_e1 :
    twist (blade_cocycle euclidean_signature)
      (blade_family e2) (blade_family e1) e12 = (-1)%Q.
  Proof. vm_compute. reflexivity. Qed.

  Example e1_times_e2_everywhere :
    List.map (twist (blade_cocycle euclidean_signature)
                (blade_family e1) (blade_family e2)) (blade_enum 3)
    = (0 :: 0 :: 0 :: 0 :: 0 :: 0 :: 1 :: 0 :: nil)%Q.
  Proof. vm_compute. reflexivity. Qed.

  Example e1_squared_euclidean :
    twist (blade_cocycle euclidean_signature)
      (blade_family e1) (blade_family e1) 0 = (1 : Q).
  Proof. vm_compute. reflexivity. Qed.

  Example e1_squared_minkowski :
    twist (blade_cocycle minkowski_signature)
      (blade_family e1) (blade_family e1) 0 = (-1)%Q.
  Proof. vm_compute. reflexivity. Qed.

  Example e12_squared_euclidean :
    twist (blade_cocycle euclidean_signature)
      (blade_family e12) (blade_family e12) 0 = (-1)%Q.
  Proof. vm_compute. reflexivity. Qed.

End BladeCocycleComputes.
