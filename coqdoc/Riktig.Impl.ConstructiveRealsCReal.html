Coq's constructive Cauchy reals are constructive reals over Coq's Q and nat: every operation computes, every strict inequality carries a computed index, and an element is approximated by its sequence at the precision's denominator.
Instance CReal_EqRel : EqRel CReal := CRealEq.
Instance CReal_Zero : Zero CReal := inject_Q 0%Q.
Instance CReal_One : One CReal := inject_Q 1%Q.
Instance CReal_HasAdd : HasAdd CReal := CReal_plus.
Instance CReal_HasMult : HasMult CReal := CReal_mult.
Instance CReal_HasNeg : HasNeg CReal := CReal_opp.
Instance CReal_HasAbs : HasAbs CReal := CReal_abs.
Instance CReal_HasLt : HasLt CReal := CRealLt.
Instance CReal_HasInvApart : HasInvApart CReal := CReal_inv.
Instance CReal_HasApprox : HasApprox CReal Q := CReal_approx.

Instance CReal_add_Magma : Magma CReal CReal_HasAdd :=
  {| Magma.equivalence := CRealEq_rel; Magma.op_proper := CReal_plus_morph_Proper |}.
Instance CReal_add_Semigroup : Semigroup CReal CReal_HasAdd :=
  {| Semigroup.magma := CReal_add_Magma; Semigroup.is_associative := CReal_plus_assoc |}.
Instance CReal_add_Monoid : Monoid CReal CReal_Zero CReal_HasAdd :=
  {| Monoid.semigroup := CReal_add_Semigroup;
     Monoid.is_left_identity := CReal_plus_0_l; Monoid.is_right_identity := CReal_plus_0_r |}.
Instance CReal_Group : Group CReal CReal_Zero CReal_HasAdd CReal_HasNeg :=
  {| Group.monoid := CReal_add_Monoid; Group.neg_proper := CReal_opp_morph_Proper;
     Group.is_left_inverse := CReal_plus_opp_l; Group.is_right_inverse := CReal_plus_opp_r |}.
Instance CReal_AbGroup : AbGroup CReal CReal_Zero CReal_HasAdd CReal_HasNeg :=
  {| AbGroup.group := CReal_Group; AbGroup.is_commutative := CReal_plus_comm |}.

Instance CReal_mul_Magma : Magma CReal CReal_HasMult :=
  {| Magma.equivalence := CRealEq_rel; Magma.op_proper := CReal_mult_morph_Proper |}.
Instance CReal_mul_Semigroup : Semigroup CReal CReal_HasMult :=
  {| Semigroup.magma := CReal_mul_Magma; Semigroup.is_associative := CReal_mult_assoc |}.
Instance CReal_mul_Monoid : Monoid CReal CReal_One CReal_HasMult :=
  {| Monoid.semigroup := CReal_mul_Semigroup;
     Monoid.is_left_identity := CReal_mult_1_l; Monoid.is_right_identity := CReal_mult_1_r |}.

Instance CReal_Ring : Ring CReal CReal_Zero CReal_HasAdd CReal_HasNeg CReal_One CReal_HasMult :=
  {| Ring.abgroup := CReal_AbGroup; Ring.mul_monoid := CReal_mul_Monoid;
     Ring.is_left_distributive := CReal_mult_plus_distr_l;
     Ring.is_right_distributive := fun x y b ⇒ CReal_mult_plus_distr_r b x y |}.
Instance CReal_mul_Commutative : Commutative CReal CReal CReal_HasMult := CReal_mult_comm.

Instance CReal_cast_RingHom : RingHom Q CReal inject_Q :=
  {| RingHom.hom_proper := inject_Q_proper;
     RingHom.hom_additive := inject_Q_plus;
     RingHom.hom_multiplicative := inject_Q_mult;
     RingHom.hom_one := inject_Q_one |}.

Instance CReal_ConstructiveReals :
