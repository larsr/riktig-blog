from_nat n is the image of n under the unique ring map nat → F.
  Fixpoint from_nat (n : nat) : F :=
    match n with
    | O ⇒ 0
    | S m ⇒ 1 + from_nat m
    end.

  Context (from_nat_neq_0 : ∀ n : nat, from_nat (S n) =/= 0).

exp_term x n is the n-th term of the exponential series, x^n / n!, written as the recursion that defines it.
  Fixpoint exp_term (x : A) (n : nat) : A :=
    match n with
    | O ⇒ 1
    | S m ⇒ inv (from_nat (S m)) *: (x × exp_term x m)
    end.

cos_term t k is (-1)^k t^(2k) / (2k)! and sin_term t k is (-1)^k t^(2k+1) / (2k+1)!, each written as its two-step recursion — one step of the exponential series squares t and divides by the two factors the index passes.
  Fixpoint cos_term (t : F) (k : nat) : F :=
    match k with
    | O ⇒ 1
    | S j ⇒ - (inv (from_nat (S (2 × j))) × inv (from_nat (S (S (2 × j))))
                  × (t × t) × cos_term t j)
    end.

  Fixpoint sin_term (t : F) (k : nat) : F :=
    match k with
    | O ⇒ t
    | S j ⇒ - (inv (from_nat (S (S (2 × j)))) × inv (from_nat (S (S (S (2 × j)))))
                  × (t × t) × sin_term t j)
    end.

  Definition cexp_sum (x : A) : A := ssum (exp_term x).
  Definition cos_sum (t : F) : F := ssum (cos_term t).
  Definition sin_sum (t : F) : F := ssum (sin_term t).

  Context (exp_summable : ∀ x : A, summable (exp_term x))
          (cos_summable : ∀ t : F, summable (cos_term t))
          (sin_summable : ∀ t : F, summable (sin_term t)).


  Lemma from_nat_add : ∀ m n : nat,
    from_nat (m + n)%nat == from_nat m + from_nat n.
  Proof.
    intros m; induction m as [| m IH]; intros n.
    - calc (from_nat n)
        == ((0 : F) + from_nat n) { left_identity }.
    - calc ((1 : F) + from_nat (m + n)%nat)
        == ((1 : F) + (from_nat m + from_nat n)) { (IH n) }
        == (((1 : F) + from_nat m) + from_nat n) { associative }.
  Qed.

  Lemma from_nat_one : from_nat 1 == 1.
  Proof.
    calc ((1 : F) + from_nat O)
      == ((1 : F) + 0) { tac: reflexivity }
      == (1 : F) { right_identity }.
  Qed.

  Lemma from_nat_split : ∀ (j n : nat), (j ≤ n)%nat →
    from_nat j + from_nat (n - j)%nat == from_nat n.
  Proof.
    intros j n Hj.
    calc (from_nat j + from_nat (n - j)%nat)
      == (from_nat (j + (n - j))%nat) { (from_nat_add j (n - j)%nat) }
      == (from_nat n) { tac: (rewrite Nat.add_comm,
                                                (Nat.sub_add j n Hj);
                                              reflexivity) }.
  Qed.

  Lemma exp_term_scale : ∀ (x : A) (m : nat),
    from_nat (S m) *: exp_term x (S m) == x × exp_term x m.
  Proof.
    intros x m.
    calc (from_nat (S m) *: (inv (from_nat (S m)) *: (x × exp_term x m)))
      == ((from_nat (S m) × inv (from_nat (S m))) *: (x × exp_term x m))
           { (associative (B := A) (from_nat (S m)) (inv (from_nat (S m)))) }
      == ((1 : F) *: (x × exp_term x m))
           { (field_right_inverse (from_nat (S m)) (from_nat_neq_0 m)) }
      == (x × exp_term x m)
           { left_identity }.
  Qed.

  Lemma exp_term_proper : ∀ (n : nat) (x y : A),
    x == y → exp_term x n == exp_term y n.
  Proof.
    intros n; induction n as [| m IH]; intros x y Hxy.
    - reflexivity.
    - exact (scale_proper _ _ (reflexivity _) _ _
               ((op_proper (op := mult)) _ _ Hxy _ _ (IH x y Hxy))).
  Qed.

  Lemma exp_term_succ : ∀ (x : A) (m : nat),
    exp_term x (S m) == inv (from_nat (S m)) *: (x × exp_term x m).
  Proof. reflexivity. Qed.

  Lemma exp_term_start : ∀ x : A, exp_term x O == 1.
  Proof. reflexivity. Qed.

  Lemma exp_term_zero : ∀ m : nat, exp_term (0 : A) (S m) == 0.
  Proof.
    intros m.
    calc (exp_term (0 : A) (S m))
      == (inv (from_nat (S m)) *: ((0 : A) × exp_term 0 m))
           { (exp_term_succ 0 m) }
      == (inv (from_nat (S m)) *: (0 : A)) { mul_0_l }
      == (0 : A) { scale_zero }.
  Qed.

  Lemma scale_mul_left : ∀ (a : F) (u v : A), (a *: u) × v == a *: (u × v).
  Proof. intros a u v. exact (homogeneous[fun x ⇒ x × v] a u). Qed.

  Lemma scale_mul_right : ∀ (a : F) (u v : A), u × (a *: v) == a *: (u × v).
  Proof. intros a u v. exact (homogeneous[fun y ⇒ u × y] a v). Qed.

  Lemma scale_mul_scale : ∀ (a b : F) (u v : A),
    (a *: u) × (b *: v) == (a × b) *: (u × v).
  Proof.
    intros a b u v.
    calc ((a *: u) × (b *: v))
      == (a *: (u × (b *: v))) { (scale_mul_left a u (b *: v)) }
      == (a *: (b *: (u × v))) { (scale_mul_right b u v) }
      == ((a × b) *: (u × v)) { (associative (B := A) a b) }.
  Qed.

  Lemma exp_term_commutes : ∀ (x y : A), x × y == y × x →
    ∀ m : nat, y × exp_term x m == exp_term x m × y.
  Proof.
    intros x y Hxy m; induction m as [| m IH].
    - calc (y × (1 : A))
        == y { right_identity }
        == ((1 : A) × y) { left_identity }.
    - calc (y × (inv (from_nat (S m)) *: (x × exp_term x m)))
        == (inv (from_nat (S m)) *: (y × (x × exp_term x m)))
             { (scale_mul_right (inv (from_nat (S m))) y (x × exp_term x m)) }
        == (inv (from_nat (S m)) *: ((y × x) × exp_term x m))
             { (associative (B := A) y x) }
        == (inv (from_nat (S m)) *: ((x × y) × exp_term x m))
             { Hxy }
        == (inv (from_nat (S m)) *: (x × (y × exp_term x m)))
             { (associative (B := A) x y) }
        == (inv (from_nat (S m)) *: (x × (exp_term x m × y)))
             { IH }
        == (inv (from_nat (S m)) *: ((x × exp_term x m) × y))
             { (associative (B := A) x (exp_term x m)) }
        == ((inv (from_nat (S m)) *: (x × exp_term x m)) × y)
             { (scale_mul_left (inv (from_nat (S m))) (x × exp_term x m) y) }.
  Qed.


  Lemma binomial_term_left : ∀ (x y : A) (n k : nat),
    x × (exp_term x k × exp_term y (n - k)%nat)
      == from_nat (S k) *: (exp_term x (S k) × exp_term y (S n - S k)%nat).
  Proof.
    intros x y n k.
    calc (x × (exp_term x k × exp_term y (n - k)%nat))
      == ((x × exp_term x k) × exp_term y (n - k)%nat)
           { (associative (B := A) x (exp_term x k)) }
      == ((from_nat (S k) *: exp_term x (S k)) × exp_term y (n - k)%nat)
           { (exp_term_scale x k) }
      == (from_nat (S k) *: (exp_term x (S k) × exp_term y (n - k)%nat))
           { (scale_mul_left (from_nat (S k)) (exp_term x (S k))
                (exp_term y (n - k)%nat)) }.
  Qed.

  Lemma binomial_term_right : ∀ (x y : A) (n k : nat),
    x × y == y × x → (k ≤ n)%nat →
    y × (exp_term x k × exp_term y (n - k)%nat)
      == from_nat (S n - k)%nat *: (exp_term x k × exp_term y (S n - k)%nat).
  Proof.
    intros x y n k Hxy Hk.
    rewrite (Nat.sub_succ_l k n Hk).
    calc (y × (exp_term x k × exp_term y (n - k)%nat))
      == ((y × exp_term x k) × exp_term y (n - k)%nat)
           { (associative (B := A) y (exp_term x k)) }
      == ((exp_term x k × y) × exp_term y (n - k)%nat)
           { (exp_term_commutes x y Hxy k) }
      == (exp_term x k × (y × exp_term y (n - k)%nat))
           { (associative (B := A) (exp_term x k) y) }
      == (exp_term x k
            × (from_nat (S (n - k)%nat) *: exp_term y (S (n - k)%nat)))
           { (exp_term_scale y (n - k)%nat) }
      == (from_nat (S (n - k)%nat)
            *: (exp_term x k × exp_term y (S (n - k)%nat)))
           { (scale_mul_right (from_nat (S (n - k)%nat)) (exp_term x k)
                (exp_term y (S (n - k)%nat))) }.
  Qed.

  Lemma binomial_term_sum : ∀ (j n : nat) (v : A), (j ≤ n)%nat →
    from_nat j *: v + from_nat (n - j)%nat *: v == from_nat n *: v.
  Proof.
    intros j n v Hj.
    calc (from_nat j *: v + from_nat (n - j)%nat *: v)
      == ((from_nat j + from_nat (n - j)%nat) *: v) { right_distributive }
      == (from_nat n *: v) { (from_nat_split j n Hj) }.
  Qed.

  Lemma binomial_left : ∀ (x y : A) (n : nat),
    x × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)
      == bigop (S (S n))
           (fun j ⇒ from_nat j *: (exp_term x j × exp_term y (S n - j)%nat)).
  Proof.
    intros x y n.
    calc (x × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat))
      == (bigop (S n) (fun k ⇒ x × (exp_term x k × exp_term y (n - k)%nat)))
           { bigop_mul_l }
      == (bigop (S n)
            (fun k ⇒ from_nat (S k)
                        *: (exp_term x (S k) × exp_term y (S n - S k)%nat)))
           { (bigop_ext (S n) _ _ (binomial_term_left x y n)) }
      == ((0 : A)
            + bigop (S n)
                (fun k ⇒ from_nat (S k)
                            *: (exp_term x (S k) × exp_term y (S n - S k)%nat)))
           { left_identity }
      == (from_nat O *: (exp_term x O × exp_term y (S n - O)%nat)
            + bigop (S n)
                (fun k ⇒ from_nat (S k)
                            *: (exp_term x (S k) × exp_term y (S n - S k)%nat)))
           { (zero_scale (exp_term x O × exp_term y (S n - O)%nat)) }
      == (bigop (S (S n))
            (fun j ⇒ from_nat j *: (exp_term x j × exp_term y (S n - j)%nat)))
           { bigop_S }.
  Qed.

  Lemma binomial_last_term : ∀ (x y : A) (n : nat),
    from_nat (S n - S n)%nat
      *: (exp_term x (S n) × exp_term y (S n - S n)%nat) == 0.
  Proof.
    intros x y n.
    rewrite Nat.sub_diag.
    exact (zero_scale (exp_term x (S n) × exp_term y O)).
  Qed.

  Lemma binomial_right : ∀ (x y : A) (n : nat), x × y == y × x →
    y × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)
      == bigop (S (S n))
           (fun j ⇒ from_nat (S n - j)%nat
                       *: (exp_term x j × exp_term y (S n - j)%nat)).
  Proof.
    intros x y n Hxy.
    calc (y × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat))
      == (bigop (S n) (fun k ⇒ y × (exp_term x k × exp_term y (n - k)%nat)))
           { bigop_mul_l }
      == (bigop (S n)
            (fun k ⇒ from_nat (S n - k)%nat
                        *: (exp_term x k × exp_term y (S n - k)%nat)))
           { (bigop_ext_lt (S n) _ _
                (fun k Hk ⇒ binomial_term_right x y n k Hxy (le_S_n k n Hk))) }
      == (bigop (S n)
            (fun j ⇒ from_nat (S n - j)%nat
                        *: (exp_term x j × exp_term y (S n - j)%nat))
            + (0 : A))
           { right_identity }
      == (bigop (S n)
            (fun j ⇒ from_nat (S n - j)%nat
                        *: (exp_term x j × exp_term y (S n - j)%nat))
            + from_nat (S n - S n)%nat
                *: (exp_term x (S n) × exp_term y (S n - S n)%nat))
           { (binomial_last_term x y n) }
      == (bigop (S (S n))
            (fun j ⇒ from_nat (S n - j)%nat
                        *: (exp_term x j × exp_term y (S n - j)%nat)))
           { (bigop_snoc (S n)
                (fun j ⇒ from_nat (S n - j)%nat
                            *: (exp_term x j × exp_term y (S n - j)%nat))) }.
  Qed.

  Lemma dexp_binomial : ∀ (x y : A), x × y == y × x → ∀ n : nat,
    bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)
      == exp_term (x + y) n.
  Proof.
    intros x y Hxy n; induction n as [| n IH].
    - calc (bigop 1 (fun k ⇒ exp_term x k × exp_term y (O - k)%nat))
        == (exp_term x O × exp_term y O
              + bigop O (fun k ⇒ exp_term x (S k) × exp_term y (O - S k)%nat))
             { bigop_S }
        == ((1 : A) × (1 : A) + (0 : A)) { bigop_0 }
        == ((1 : A) × (1 : A)) { right_identity }
        == (1 : A) { left_identity }.
    - assert (Hsum : (x + y)
                × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)
              == from_nat (S n)
                   *: bigop (S (S n))
                        (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat)).
      { calc ((x + y)
                × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat))
          == (x × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)
                + y × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat))
               { right_distributive }
          == (bigop (S (S n))
                (fun j ⇒ from_nat j *: (exp_term x j × exp_term y (S n - j)%nat))
                + y × bigop (S n) (fun k ⇒ exp_term x k × exp_term y (n - k)%nat))
               { (binomial_left x y n) }
          == (bigop (S (S n))
                (fun j ⇒ from_nat j *: (exp_term x j × exp_term y (S n - j)%nat))
                + bigop (S (S n))
                    (fun j ⇒ from_nat (S n - j)%nat
                                *: (exp_term x j × exp_term y (S n - j)%nat)))
               { (binomial_right x y n Hxy) }
          == (bigop (S (S n))
                (fun j ⇒ from_nat j *: (exp_term x j × exp_term y (S n - j)%nat)
                    + from_nat (S n - j)%nat
                        *: (exp_term x j × exp_term y (S n - j)%nat)))
               { (bigop_add (S (S n))
                    (fun j ⇒ from_nat j
                                *: (exp_term x j × exp_term y (S n - j)%nat))
                    (fun j ⇒ from_nat (S n - j)%nat
                                *: (exp_term x j × exp_term y (S n - j)%nat))) }
          == (bigop (S (S n))
                (fun j ⇒ from_nat (S n)
                            *: (exp_term x j × exp_term y (S n - j)%nat)))
               { (bigop_ext_lt (S (S n)) _ _
                    (fun j Hj ⇒ binomial_term_sum j (S n)
                       (exp_term x j × exp_term y (S n - j)%nat)
                       (le_S_n j (S n) Hj))) }
          == (from_nat (S n)
                *: bigop (S (S n))
                     (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat))
               { (bigop_scale (S (S n)) (from_nat (S n))
                    (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat)) }. }
      calc (bigop (S (S n)) (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat))
        == ((1 : F) *: bigop (S (S n))
              (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat))
             { left_identity }
        == ((inv (from_nat (S n)) × from_nat (S n))
              *: bigop (S (S n))
                   (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat))
             { ((left_inverse_on (defined_on := nonzero _))
                  (from_nat (S n)) (from_nat_neq_0 n)) }
        == (inv (from_nat (S n))
              *: (from_nat (S n)
                    *: bigop (S (S n))
                         (fun j ⇒ exp_term x j × exp_term y (S n - j)%nat)))
             { (associative (B := A) (inv (from_nat (S n))) (from_nat (S n))) }
        == (inv (from_nat (S n))
              *: ((x + y)
                    × bigop (S n)
                        (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)))
             { Hsum }
        == (inv (from_nat (S n)) *: ((x + y) × exp_term (x + y) n))
             { IH }.
  Qed.


  Lemma exp_term_two_step : ∀ (x : A) (m : nat),
    exp_term x (S (S m))
      == (inv (from_nat (S m)) × inv (from_nat (S (S m))))
           *: ((x × x) × exp_term x m).
  Proof.
    intros x m.
    calc (exp_term x (S (S m)))
      == (inv (from_nat (S (S m))) *: (x × exp_term x (S m)))
           { (exp_term_succ x (S m)) }
      == (inv (from_nat (S (S m)))
            *: (x × (inv (from_nat (S m)) *: (x × exp_term x m))))
           { (exp_term_succ x m) }
      == (inv (from_nat (S (S m)))
            *: (inv (from_nat (S m)) *: (x × (x × exp_term x m))))
           { (scale_mul_right (inv (from_nat (S m))) x (x × exp_term x m)) }
      == ((inv (from_nat (S (S m))) × inv (from_nat (S m)))
            *: (x × (x × exp_term x m)))
           { (associative (B := A) (inv (from_nat (S (S m))))
                (inv (from_nat (S m)))) }
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))))
            *: (x × (x × exp_term x m)))
           { (mul_commutative (inv (from_nat (S (S m))))
                (inv (from_nat (S m)))) }
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))))
            *: ((x × x) × exp_term x m))
           { (associative (B := A) x x (exp_term x m)) }.
  Qed.

  Lemma square_of_scale : ∀ (t : F) (e : A), e × e == - (1) →
    (t *: e) × (t *: e) == - ((t × t) *: (1 : A)).
  Proof.
    intros t e He.
    calc ((t *: e) × (t *: e))
      == ((t × t) *: (e × e)) { (scale_mul_scale t t e e) }
      == ((t × t) *: (- (1 : A))) { He }
      == (- ((t × t) *: (1 : A))) { scale_neg }.
  Qed.

  Lemma neg_scale_one_mul : ∀ (b a : F) (v : A),
    (- (b *: (1 : A))) × (a *: v) == (- (b × a)) *: v.
  Proof.
    intros b a v.
    calc ((- (b *: (1 : A))) × (a *: v))
      == (- ((b *: (1 : A)) × (a *: v))) { mul_neg_l }
      == (- ((b × a) *: ((1 : A) × v))) { (scale_mul_scale b a 1 v) }
      == (- ((b × a) *: v)) { (mul_left_identity v) }
      == ((- (b × a)) *: v) { neg_scale }.
  Qed.

  Lemma coefficient_normalize : ∀ c b a : F,
    c × (- (b × a)) == - (c × b × a).
  Proof. intros c b a. ring. Qed.

  Lemma exp_term_line_step : ∀ (t : F) (e : A) (a : F) (v : A) (m : nat),
    e × e == - (1) →
    exp_term (t *: e) m == a *: v →
    exp_term (t *: e) (S (S m))
      == (- (inv (from_nat (S m)) × inv (from_nat (S (S m))) × (t × t) × a)) *: v.
  Proof.
    intros t e a v m He Hm.
    calc (exp_term (t *: e) (S (S m)))
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))))
            *: (((t *: e) × (t *: e)) × exp_term (t *: e) m))
           { (exp_term_two_step (t *: e) m) }
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))))
            *: ((- ((t × t) *: (1 : A))) × exp_term (t *: e) m))
           { (square_of_scale t e He) }
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))))
            *: ((- ((t × t) *: (1 : A))) × (a *: v)))
           { Hm }
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))))
            *: ((- ((t × t) × a)) *: v))
           { (neg_scale_one_mul (t × t) a v) }
      == ((inv (from_nat (S m)) × inv (from_nat (S (S m))) × (- ((t × t) × a)))
            *: v)
           { (associative (B := A)
                (inv (from_nat (S m)) × inv (from_nat (S (S m))))
                (- ((t × t) × a))) }
      == ((- (inv (from_nat (S m)) × inv (from_nat (S (S m))) × (t × t) × a))
            *: v)
           { (coefficient_normalize
                (inv (from_nat (S m)) × inv (from_nat (S (S m)))) (t × t) a) }.
  Qed.

  Lemma euler_even : ∀ (t : F) (e : A), e × e == - (1) →
    ∀ k : nat, exp_term (t *: e) (2 × k)%nat == cos_term t k *: (1 : A).
  Proof.
    intros t e He k; induction k as [| j IH].
    - exact (symmetry (left_identity (action := scale) (1 : A))).
    - rewrite (two_mul_succ j).
      exact (exp_term_line_step t e (cos_term t j) 1 (2 × j)%nat He IH).
  Qed.

  Lemma euler_odd : ∀ (t : F) (e : A), e × e == - (1) →
    ∀ k : nat, exp_term (t *: e) (S (2 × k)%nat) == sin_term t k *: e.
  Proof.
    intros t e He k; induction k as [| j IH].
    - calc (exp_term (t *: e) (S (2 × O)%nat))
        == (inv (from_nat (S O)) *: ((t *: e) × exp_term (t *: e) O))
             { (exp_term_succ (t *: e) O) }
        == (inv (from_nat (S O)) *: ((t *: e) × (1 : A)))
             { (exp_term_start (t *: e)) }
        == (inv (from_nat (S O)) *: (t *: e))
             { (mul_right_identity (t *: e)) }
        == (inv (1 : F) *: (t *: e))
             { from_nat_one }
        == ((1 : F) *: (t *: e))
             { inv_one }
        == (t *: e)
             { (left_identity (action := scale) (t *: e)) }.
    - rewrite (two_mul_succ j).
      exact (exp_term_line_step t e (sin_term t j) e (S (2 × j)%nat) He IH).
  Qed.


  Lemma cos_term_neg : ∀ (t : F) (k : nat),
    cos_term (- t) k == cos_term t k.
  Proof.
    intros t k; induction k as [| j IH].
    - reflexivity.
    - calc (- (inv (from_nat (S (2 × j))) × inv (from_nat (S (S (2 × j))))
                 × ((- t) × (- t)) × cos_term (- t) j))
        == (- (inv (from_nat (S (2 × j))) × inv (from_nat (S (S (2 × j))))
                 × ((- t) × (- t)) × cos_term t j))
             { IH }
        == (- (inv (from_nat (S (2 × j))) × inv (from_nat (S (S (2 × j))))
                 × (t × t) × cos_term t j))
             { (mul_neg_neg t t) }.
  Qed.

  Lemma sin_term_neg : ∀ (t : F) (k : nat),
    sin_term (- t) k == - sin_term t k.
  Proof.
    intros t k; induction k as [| j IH].
    - reflexivity.
    - calc (- (inv (from_nat (S (S (2 × j)))) × inv (from_nat (S (S (S (2 × j)))))
                 × ((- t) × (- t)) × sin_term (- t) j))
        == (- (inv (from_nat (S (S (2 × j)))) × inv (from_nat (S (S (S (2 × j)))))
                 × ((- t) × (- t)) × (- sin_term t j)))
             { IH }
        == (- (inv (from_nat (S (S (2 × j)))) × inv (from_nat (S (S (S (2 × j)))))
                 × (t × t) × (- sin_term t j)))
             { (mul_neg_neg t t) }
        == (- (- (inv (from_nat (S (S (2 × j)))) × inv (from_nat (S (S (S (2 × j)))))
                    × (t × t) × sin_term t j)))
             { (mul_neg_r (inv (from_nat (S (S (2 × j))))
                             × inv (from_nat (S (S (S (2 × j))))) × (t × t))
                          (sin_term t j)) }.
  Qed.

  Lemma cos_term_proper : ∀ (k : nat) (s t : F),
    s == t → cos_term s k == cos_term t k.

    intros k; induction k as [| j IH]; intros s t Hst.
    - reflexivity.
    - exact (neg_proper _ _
               ((op_proper (op := mult)) _ _
                  ((op_proper (op := mult)) _ _ (reflexivity _) _ _
                     ((op_proper (op := mult)) _ _ Hst _ _ Hst))
                  _ _ (IH s t Hst))).
  Qed.

  Qed.

  Theorem series_cexp_add : ∀ x y : A, x × y == y × x →
    cexp_sum (x + y) == cexp_sum x × cexp_sum y.
  Proof.
    intros x y Hxy.
    calc (ssum (exp_term (x + y)))
      == (ssum (fun n ⇒ bigop (S n)
                  (fun k ⇒ exp_term x k × exp_term y (n - k)%nat)))
           { (ssum_proper _ _
                (fun n ⇒ symmetry (dexp_binomial x y Hxy n))) }
      == (ssum (exp_term x) × ssum (exp_term y))
           { (sum_cauchy (exp_term x) (exp_term y)
                (exp_summable x) (exp_summable y)) }.
  Qed.

  Theorem series_euler : ∀ (t : F) (e : A), e × e == - (1) →
    cexp_sum (t *: e) == cos_sum t *: (1 : A) + sin_sum t *: e.
  Proof.
    intros t e He.
    calc (ssum (exp_term (t *: e)))
      == (ssum (fun k ⇒ exp_term (t *: e) (2 × k)%nat)
            + ssum (fun k ⇒ exp_term (t *: e) (S (2 × k)%nat)))
           { (sum_parity (exp_term (t *: e)) (exp_summable (t *: e))) }
      == (ssum (fun k ⇒ cos_term t k *: (1 : A))
            + ssum (fun k ⇒ exp_term (t *: e) (S (2 × k)%nat)))
           { (ssum_proper _ _ (euler_even t e He)) }
      == (ssum (fun k ⇒ cos_term t k *: (1 : A))
            + ssum (fun k ⇒ sin_term t k *: e))
           { (ssum_proper _ _ (euler_odd t e He)) }
      == (ssum (cos_term t) *: (1 : A) + ssum (fun k ⇒ sin_term t k *: e))
           { (sum_scale_r (cos_term t) (1 : A) (cos_summable t)) }
      == (ssum (cos_term t) *: (1 : A) + ssum (sin_term t) *: e)
           { (sum_scale_r (sin_term t) e (sin_summable t)) }.
  Qed.

  Theorem series_cexp_zero : cexp_sum (0 : A) == 1.

      == (bigop (S O) (exp_term (0 : A)))
           { (sum_supported (exp_term (0 : A)) (S O) exp_term_vanishes_from_one) }
      == (exp_term (0 : A) O + bigop O (fun i ⇒ exp_term (0 : A) (S i)))
           { bigop_S }
      == ((1 : A) + (0 : A))
           { bigop_0 }
      == (1 : A)
           { right_identity }.
  Qed.

  Lemma scale_one_scale_one : ∀ a c : F,
    (a *: (1 : A)) × (c *: (1 : A)) == (a × c) *: (1 : A).
  Proof.
    intros a c.
    calc ((a *: (1 : A)) × (c *: (1 : A)))
      == ((a × c) *: ((1 : A) × (1 : A))) { (scale_mul_scale a c 1 1) }
      == ((a × c) *: (1 : A)) { (mul_left_identity (1 : A)) }.
  Qed.

  Lemma scale_one_scale_imaginary : ∀ a d : F,

      == ((a × d) *: ((1 : A) × imaginary)) { (scale_mul_scale a d 1 imaginary) }

  Lemma scale_imaginary_scale_one : ∀ b c : F,

      == ((b × c) *: (imaginary × (1 : A))) { (scale_mul_scale b c imaginary 1) }
      == ((b × c) *: imaginary) { (mul_right_identity imaginary) }.
  Qed.

  Lemma scale_imaginary_scale_imaginary : ∀ b d : F,

      == (- ((b × d) *: (1 : A))) { scale_neg }
      == ((- (b × d)) *: (1 : A)) { neg_scale }.
  Qed.

  Lemma line_mul : ∀ a b c d : F,
    (a *: (1 : A) + b *: imaginary) × (c *: (1 : A) + d *: imaginary)
      == (a × c + - (b × d)) *: (1 : A) + (a × d + b × c) *: imaginary.
  Proof.
    intros a b c d.
    assert (Hexpand :
      (a *: (1 : A) + b *: imaginary) × (c *: (1 : A) + d *: imaginary)
        == ((a × c) *: (1 : A) + (b × c) *: imaginary)
             + ((a × d) *: imaginary + (- (b × d)) *: (1 : A))).
    { calc ((a *: (1 : A) + b *: imaginary) × (c *: (1 : A) + d *: imaginary))
        == ((a *: (1 : A)) × (c *: (1 : A)) + (b *: imaginary) × (c *: (1 : A))
              + ((a *: (1 : A)) × (d *: imaginary)
                   + (b *: imaginary) × (d *: imaginary)))
             { mul_add_expand }
        == ((a × c) *: (1 : A) + (b *: imaginary) × (c *: (1 : A))
              + ((a *: (1 : A)) × (d *: imaginary)
                   + (b *: imaginary) × (d *: imaginary)))
             { (scale_one_scale_one a c) }
        == ((a × c) *: (1 : A) + (b × c) *: imaginary
              + ((a *: (1 : A)) × (d *: imaginary)
                   + (b *: imaginary) × (d *: imaginary)))
             { (scale_imaginary_scale_one b c) }
        == ((a × c) *: (1 : A) + (b × c) *: imaginary
              + ((a × d) *: imaginary + (b *: imaginary) × (d *: imaginary)))
             { (scale_one_scale_imaginary a d) }
        == ((a × c) *: (1 : A) + (b × c) *: imaginary
              + ((a × d) *: imaginary + (- (b × d)) *: (1 : A)))
             { (scale_imaginary_scale_imaginary b d) }. }
    calc ((a *: (1 : A) + b *: imaginary) × (c *: (1 : A) + d *: imaginary))
      == ((a × c) *: (1 : A) + (b × c) *: imaginary
            + ((a × d) *: imaginary + (- (b × d)) *: (1 : A)))
           { Hexpand }
      == ((a × c) *: (1 : A) + (b × c) *: imaginary
            + ((- (b × d)) *: (1 : A) + (a × d) *: imaginary))
           { (commutative ((a × d) *: imaginary)) }
      == ((a × c) *: (1 : A) + (- (b × d)) *: (1 : A)
            + ((b × c) *: imaginary + (a × d) *: imaginary))
           { add_interchange }
      == ((a × c + - (b × d)) *: (1 : A)
            + ((b × c) *: imaginary + (a × d) *: imaginary))
           { right_distributive }
      == ((a × c + - (b × d)) *: (1 : A) + (b × c + a × d) *: imaginary)
           { right_distributive }
      == ((a × c + - (b × d)) *: (1 : A) + (a × d + b × c) *: imaginary)
           { (add_commutative (b × c) (a × d)) }.
  Qed.

  Theorem series_pythagoras : ∀ t : F,
    cos_sum t × cos_sum t + sin_sum t × sin_sum t == 1.
  Proof.
    intros t.
    assert (Hcommutes :
      (t *: imaginary) × ((- t) *: imaginary)
        == ((- t) *: imaginary) × (t *: imaginary)).
    { calc ((t *: imaginary) × ((- t) *: imaginary))
        == ((t × - t) *: (imaginary × imaginary))
             { (scale_mul_scale t (- t) imaginary imaginary) }
        == (((- t) × t) *: (imaginary × imaginary))
             { (mul_commutative t (- t)) }
        == (((- t) *: imaginary) × (t *: imaginary))
             { (scale_mul_scale (- t) t imaginary imaginary) }. }
    assert (Hproduct :
      cexp_sum (t *: imaginary) × cexp_sum ((- t) *: imaginary) == (1 : A)).
    { calc (cexp_sum (t *: imaginary) × cexp_sum ((- t) *: imaginary))
        == (cexp_sum (t *: imaginary + (- t) *: imaginary))
             { (series_cexp_add (t *: imaginary) ((- t) *: imaginary) Hcommutes) }
        == (cexp_sum ((t + - t) *: imaginary))
             { right_distributive }
        == (cexp_sum ((0 : F) *: imaginary))
             { (right_inverse t) }
        == (cexp_sum (0 : A))
             { zero_scale }
        == (1 : A)
             { series_cexp_zero }. }
    assert (Hnegative :
      cexp_sum ((- t) *: imaginary)
        == cos_sum t *: (1 : A) + (- sin_sum t) *: imaginary).
    { calc (cexp_sum ((- t) *: imaginary))
        == (cos_sum (- t) *: (1 : A) + sin_sum (- t) *: imaginary)
             { (series_euler (- t) imaginary imaginary_square) }
        == (cos_sum t *: (1 : A) + sin_sum (- t) *: imaginary)
             { (series_cos_even t) }
        == (cos_sum t *: (1 : A) + (- sin_sum t) *: imaginary)
             { (series_sin_odd t) }. }
    assert (Hpad :
      (cos_sum t × cos_sum t + sin_sum t × sin_sum t) *: (1 : A)
        == (cos_sum t × cos_sum t + - (sin_sum t × - sin_sum t)) *: (1 : A)
             + (cos_sum t × - sin_sum t + sin_sum t × cos_sum t) *: imaginary).
    { calc ((cos_sum t × cos_sum t + sin_sum t × sin_sum t) *: (1 : A))
        == ((cos_sum t × cos_sum t + sin_sum t × sin_sum t) *: (1 : A) + (0 : A))
             { right_identity }
        == ((cos_sum t × cos_sum t + - (sin_sum t × - sin_sum t)) *: (1 : A)
              + (0 : A))
             { (pythagoras_diagonal (cos_sum t) (sin_sum t)) }
        == ((cos_sum t × cos_sum t + - (sin_sum t × - sin_sum t)) *: (1 : A)
              + (0 : F) *: imaginary)
             { zero_scale }
        == ((cos_sum t × cos_sum t + - (sin_sum t × - sin_sum t)) *: (1 : A)
              + (cos_sum t × - sin_sum t + sin_sum t × cos_sum t) *: imaginary)
             { (pythagoras_cross (cos_sum t) (sin_sum t)) }. }
    apply scale_one_injective.
    calc ((cos_sum t × cos_sum t + sin_sum t × sin_sum t) *: (1 : A))
      == ((cos_sum t × cos_sum t + - (sin_sum t × - sin_sum t)) *: (1 : A)
            + (cos_sum t × - sin_sum t + sin_sum t × cos_sum t) *: imaginary)
           { Hpad }
      == ((cos_sum t *: (1 : A) + sin_sum t *: imaginary)
            × (cos_sum t *: (1 : A) + (- sin_sum t) *: imaginary))
           { (line_mul (cos_sum t) (sin_sum t) (cos_sum t) (- sin_sum t)) }
      == (cexp_sum (t *: imaginary)
            × (cos_sum t *: (1 : A) + (- sin_sum t) *: imaginary))
           { (series_euler t imaginary imaginary_square) }
      == (cexp_sum (t *: imaginary) × cexp_sum ((- t) *: imaginary))
           { Hnegative }
      == (1 : A)
           { Hproduct }
      == ((1 : F) *: (1 : A))
           { (left_identity (action := scale) (1 : A)) }.
  Qed.
