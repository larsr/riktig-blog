Import ListNotations.

The Church numerals are naturals: the carrier is the terms that are convertible to a numeral, compared up to conversion; zero and one are the numerals, successor, addition and multiplication the programs. Every law is read off the corresponding law of Coq's naturals through the denotation.
Definition ChurchNat : Type := { t : term | ∃ n : nat, joinable t (church n) }.

Instance ChurchNat_EqRel : EqRel ChurchNat := fun c d ⇒ joinable (proj1_sig c) (proj1_sig d).
Instance ChurchNat_Zero : Zero ChurchNat := exist _ (church 0) (is_numeral 0).
Instance ChurchNat_One : One ChurchNat := exist _ (church 1) (is_numeral 1).

Lemma succ_wit : ∀ t : term, (∃ n : nat, joinable t (church n)) →
  ∃ n : nat, joinable (App SUCC t) (church n).
Proof.
  intros t [n Hn]. ∃ (S n).
  apply (joinable_trans _ (App SUCC (church n))).
  - apply joinable_app; [ apply joinable_refl | exact Hn ].
  - exact (succ_join n).
Qed.

Lemma add_wit : ∀ t u : term,
  (∃ n : nat, joinable t (church n)) → (∃ n : nat, joinable u (church n)) →
  ∃ n : nat, joinable (App (App PLUS t) u) (church n).
Proof.
  intros t u [a Ha] [b Hb]. ∃ ((a + b)%nat).
  apply (joinable_trans _ (App (App PLUS (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Ha ] | exact Hb ].
  - exact (plus_join a b).
Qed.

Lemma mul_wit : ∀ t u : term,
  (∃ n : nat, joinable t (church n)) → (∃ n : nat, joinable u (church n)) →
  ∃ n : nat, joinable (App (App MULT t) u) (church n).
Proof.
  intros t u [a Ha] [b Hb]. ∃ ((a × b)%nat).
  apply (joinable_trans _ (App (App MULT (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Ha ] | exact Hb ].
  - exact (mult_join a b).
Qed.

Instance ChurchNat_HasSucc : HasSucc ChurchNat :=
  fun c ⇒ exist _ (App SUCC (proj1_sig c)) (succ_wit _ (proj2_sig c)).
Instance ChurchNat_HasAdd : HasAdd ChurchNat :=
  fun c d ⇒ exist _ (App (App PLUS (proj1_sig c)) (proj1_sig d)) (add_wit _ _ (proj2_sig c) (proj2_sig d)).
Instance ChurchNat_HasMult : HasMult ChurchNat :=
  fun c d ⇒ exist _ (App (App MULT (proj1_sig c)) (proj1_sig d)) (mul_wit _ _ (proj2_sig c) (proj2_sig d)).

Denotation

Definition denotes (c : ChurchNat) (n : nat) : Prop := joinable (proj1_sig c) (church n).

Lemma denotes_ex : ∀ c : ChurchNat, ∃ n : nat, denotes c n.
Proof. intros c. exact (proj2_sig c). Qed.

Lemma denotes_unique : ∀ (c : ChurchNat) (n m : nat), denotes c n → denotes c m → n = m.
Proof. intros c n m Hn Hm. exact (church_injective n m (joinable_trans _ _ _ (joinable_sym _ _ Hn) Hm)). Qed.

Lemma equ_of_denotes : ∀ (c d : ChurchNat) (n : nat), denotes c n → denotes d n → c == d.
Proof. intros c d n Hc Hd. exact (joinable_trans _ _ _ Hc (joinable_sym _ _ Hd)). Qed.

Lemma denotes_proper : ∀ (c d : ChurchNat) (n : nat), c == d → denotes c n → denotes d n.
Proof. intros c d n Hcd Hc. exact (joinable_trans _ _ _ (joinable_sym _ _ Hcd) Hc). Qed.

Lemma denotes_zero : denotes 0 0.
Proof. exact (joinable_refl _). Qed.

Lemma denotes_one : denotes 1 1.
Proof. exact (joinable_refl _). Qed.

Lemma denotes_succ : ∀ (c : ChurchNat) (n : nat), denotes c n → denotes (succ c) (S n).
Proof.
  intros c n Hc. unfold denotes. simpl.
  apply (joinable_trans _ (App SUCC (church n))).
  - apply joinable_app; [ apply joinable_refl | exact Hc ].
  - exact (succ_join n).
Qed.

Lemma denotes_add : ∀ (c d : ChurchNat) (a b : nat), denotes c a → denotes d b → denotes (c + d) ((a + b)%nat).
Proof.
  intros c d a b Hc Hd. unfold denotes. simpl.
  apply (joinable_trans _ (App (App PLUS (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Hc ] | exact Hd ].
  - exact (plus_join a b).
Qed.

Lemma denotes_mul : ∀ (c d : ChurchNat) (a b : nat), denotes c a → denotes d b → denotes (c × d) ((a × b)%nat).
Proof.
  intros c d a b Hc Hd. unfold denotes. simpl.
  apply (joinable_trans _ (App (App MULT (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Hc ] | exact Hd ].
  - exact (mult_join a b).
Qed.

The structure

Instance ChurchNat_Equivalence : Equivalence (@equ ChurchNat ChurchNat_EqRel).
Proof.
  split.
  - intros c. apply joinable_refl.
  - intros c d H. exact (joinable_sym _ _ H).
  - intros c d e H1 H2. exact (joinable_trans _ _ _ H1 H2).
Qed.

Instance ChurchNat_add_Magma : Magma ChurchNat add.
Proof.
  split; [ exact ChurchNat_Equivalence | ].
  intros c c' Hc d d' Hd.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb].
  apply (equ_of_denotes _ _ ((a + b)%nat)).
  - exact (denotes_add c d a b Ha Hb).
  - exact (denotes_add c' d' a b (denotes_proper _ _ _ Hc Ha) (denotes_proper _ _ _ Hd Hb)).
Qed.

Instance ChurchNat_add_Semigroup : Semigroup ChurchNat add.
Proof.
  split; [ exact ChurchNat_add_Magma | ].
  intros c d e.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb]. destruct (denotes_ex e) as [k Hk].
  apply (equ_of_denotes _ _ ((a + b + k)%nat)).
  - exact (denotes_add _ _ _ _ (denotes_add c d a b Ha Hb) Hk).
  - rewrite <- (Nat.add_assoc a b k). exact (denotes_add _ _ _ _ Ha (denotes_add d e b k Hb Hk)).
Qed.

Instance ChurchNat_add_Monoid : Monoid ChurchNat 0 add.
Proof.
  split; [ exact ChurchNat_add_Semigroup | | ].
  - intros c. destruct (denotes_ex c) as [a Ha].
    apply (equ_of_denotes _ _ a); [ | exact Ha ].
    exact (denotes_add _ _ _ _ denotes_zero Ha).
  - intros c. destruct (denotes_ex c) as [a Ha].
    apply (equ_of_denotes _ _ a); [ | exact Ha ].
    rewrite <- (Nat.add_0_r a) at 1. exact (denotes_add _ _ _ _ Ha denotes_zero).
Qed.

Instance ChurchNat_add_Commutative : Commutative ChurchNat ChurchNat add.
Proof.
  intros c d.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb].
  apply (equ_of_denotes _ _ ((a + b)%nat)).
  - exact (denotes_add c d a b Ha Hb).
  - rewrite (Nat.add_comm a b). exact (denotes_add d c b a Hb Ha).
Qed.

Instance ChurchNat_mul_Magma : Magma ChurchNat mult.
Proof.
  split; [ exact ChurchNat_Equivalence | ].
  intros c c' Hc d d' Hd.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb].
  apply (equ_of_denotes _ _ ((a × b)%nat)).
  - exact (denotes_mul c d a b Ha Hb).
  - exact (denotes_mul c' d' a b (denotes_proper _ _ _ Hc Ha) (denotes_proper _ _ _ Hd Hb)).
Qed.

Instance ChurchNat_mul_Semigroup : Semigroup ChurchNat mult.
Proof.
  split; [ exact ChurchNat_mul_Magma | ].
  intros c d e.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb]. destruct (denotes_ex e) as [k Hk].
  apply (equ_of_denotes _ _ ((a × b × k)%nat)).
  - exact (denotes_mul _ _ _ _ (denotes_mul c d a b Ha Hb) Hk).
  - rewrite <- (Nat.mul_assoc a b k). exact (denotes_mul _ _ _ _ Ha (denotes_mul d e b k Hb Hk)).
Qed.

Instance ChurchNat_mul_Monoid : Monoid ChurchNat 1 mult.
Proof.
  split; [ exact ChurchNat_mul_Semigroup | | ].
  - intros c. destruct (denotes_ex c) as [a Ha].
    apply (equ_of_denotes _ _ a); [ | exact Ha ].
    rewrite <- (Nat.mul_1_l a) at 1. exact (denotes_mul _ _ _ _ denotes_one Ha).
  - intros c. destruct (denotes_ex c) as [a Ha].
    apply (equ_of_denotes _ _ a); [ | exact Ha ].
    rewrite <- (Nat.mul_1_r a) at 1. exact (denotes_mul _ _ _ _ Ha denotes_one).
Qed.

Instance ChurchNat_mul_Commutative : Commutative ChurchNat ChurchNat mult.
Proof.
  intros c d.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb].
  apply (equ_of_denotes _ _ ((a × b)%nat)).
  - exact (denotes_mul c d a b Ha Hb).
  - rewrite (Nat.mul_comm a b). exact (denotes_mul d c b a Hb Ha).
Qed.

Instance ChurchNat_LeftDistributive : LeftDistributive ChurchNat ChurchNat ChurchNat mult.
Proof.
  intros c d e.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb]. destruct (denotes_ex e) as [k Hk].
  apply (equ_of_denotes _ _ ((a × (b + k))%nat)).
  - exact (denotes_mul _ _ _ _ Ha (denotes_add d e b k Hb Hk)).
  - rewrite (Nat.mul_add_distr_l a b k).
    exact (denotes_add _ _ _ _ (denotes_mul c d a b Ha Hb) (denotes_mul c e a k Ha Hk)).
Qed.

Instance ChurchNat_RightDistributive : RightDistributive ChurchNat ChurchNat ChurchNat mult.
Proof.
  intros c d e.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb]. destruct (denotes_ex e) as [k Hk].
  apply (equ_of_denotes _ _ (((a + b) × k)%nat)).
  - exact (denotes_mul _ _ _ _ (denotes_add c d a b Ha Hb) Hk).
  - rewrite (Nat.mul_add_distr_r a b k).
    exact (denotes_add _ _ _ _ (denotes_mul c e a k Ha Hk) (denotes_mul d e b k Hb Hk)).
Qed.

Instance ChurchNat_LeftAbsorbing : LeftAbsorbing ChurchNat ChurchNat mult 0 0.
Proof.
  intros c. destruct (denotes_ex c) as [a Ha].
  apply (equ_of_denotes _ _ 0); [ | exact denotes_zero ].
  exact (denotes_mul _ _ _ _ denotes_zero Ha).
Qed.

Instance ChurchNat_RightAbsorbing : RightAbsorbing ChurchNat ChurchNat mult 0 0.
Proof.
  intros c. destruct (denotes_ex c) as [a Ha].
  apply (equ_of_denotes _ _ 0); [ | exact denotes_zero ].
  rewrite <- (Nat.mul_0_r a). exact (denotes_mul _ _ _ _ Ha denotes_zero).
Qed.

Instance ChurchNat_Semiring : Semiring ChurchNat 0 add 1 mult := {}.
Instance ChurchNat_CommSemiring : CommSemiring ChurchNat 0 add 1 mult := {}.

Instance ChurchNat_succ_Injective : Injective ChurchNat ChurchNat succ.
Proof.
  intros c d H.
  destruct (denotes_ex c) as [a Ha]. destruct (denotes_ex d) as [b Hb].
  assert (Hs : S a = S b).
  { apply (denotes_unique (succ c)); [ exact (denotes_succ c a Ha) | ].
    exact (denotes_proper _ _ _ (symmetry H) (denotes_succ d b Hb)). }
  injection Hs as Hs. subst.
  exact (equ_of_denotes c d b Ha Hb).
Qed.

Instance ChurchNat_Peano : Peano ChurchNat 0 succ.
Proof.
  split.
  - exact ChurchNat_Equivalence.
  - intros c d H. destruct (denotes_ex c) as [a Ha].
    apply (equ_of_denotes _ _ (S a)); [ exact (denotes_succ c a Ha) | ].
    exact (denotes_succ d a (denotes_proper _ _ _ H Ha)).
  - exact ChurchNat_succ_Injective.
  - intros c H. destruct (denotes_ex c) as [a Ha].
    assert (Hz : S a = 0%nat).
    { apply (denotes_unique (succ c)); [ exact (denotes_succ c a Ha) | ].
      exact (denotes_proper _ _ _ (symmetry H) denotes_zero). }
    discriminate Hz.
  - intros P HP H0 HS c.
    destruct (denotes_ex c) as [n Hn]. revert c Hn.
    induction n as [| n IH]; intros c Hn.
    + exact (proj1 (HP _ _ (equ_of_denotes _ _ (0%nat) denotes_zero Hn)) H0).
    + set (c' := exist (fun t ⇒ ∃ k : nat, joinable t (church k)) (church n) (is_numeral n) : ChurchNat).
      assert (Hc' : denotes c' n) by apply joinable_refl.
      apply (proj1 (HP (succ c') c (equ_of_denotes _ _ (S n) (denotes_succ c' n Hc') Hn))).
      exact (HS c' (IH c' Hc')).
Qed.

Instance ChurchNat_Naturals : Naturals ChurchNat 0 succ 1 add mult.
Proof.
  split; [ exact ChurchNat_Peano | exact ChurchNat_CommSemiring | ].
  intros c. destruct (denotes_ex c) as [a Ha].
  apply (equ_of_denotes _ _ (S a)); [ exact (denotes_succ c a Ha) | ].
  rewrite <- (Nat.add_1_r a). exact (denotes_add _ _ _ _ Ha denotes_one).
Qed.
