The Church numerals are naturals: the carrier is the terms convertible to a numeral, compared up to conversion; zero and one are the numerals, successor, addition and multiplication the programs. Denotation into Coq's naturals provides every law through denotation_naturals; every cited fact is Lib.Coq.Church's.
Instance ChurchNat_EqRel : EqRel ChurchNat := fun c d ⇒ joinable (proj1_sig c) (proj1_sig d).
Instance ChurchNat_Zero : Zero ChurchNat := exist _ ZERO (is_numeral 0).
Instance ChurchNat_One : One ChurchNat := exist _ (church 1) (is_numeral 1).
Instance ChurchNat_HasSucc : HasSucc ChurchNat :=
  fun c ⇒ exist _ (App SUCC (proj1_sig c)) (succ_wit _ (proj2_sig c)).
Instance ChurchNat_HasAdd : HasAdd ChurchNat :=
  fun c d ⇒ exist _ (App (App PLUS (proj1_sig c)) (proj1_sig d)) (add_wit _ _ (proj2_sig c) (proj2_sig d)).
Instance ChurchNat_HasMult : HasMult ChurchNat :=
  fun c d ⇒ exist _ (App (App MULT (proj1_sig c)) (proj1_sig d)) (mul_wit _ _ (proj2_sig c) (proj2_sig d)).

Instance ChurchNat_Naturals : Naturals ChurchNat 0 succ 1 add mult :=
  denotation_naturals (N0 := nat) denotes
    denotes_total denotes_sur
    (fun c n m Hn Hm ⇒ denotes_unique c n m Hn Hm)
    (fun c d n m Hc Hd Hnm ⇒ denotes_faithful c d n m Hc Hd Hnm)
    (fun c d n Hcd Hc ⇒ denotes_join c d n Hcd Hc)
    (fun c n m Hnm Hc ⇒ eq_ind n (denotes c) Hc m Hnm)
    (joinable_refl _) (joinable_refl _)
    (fun c n Hc ⇒ denotes_succ (proj1_sig c) (proj2_sig c) n Hc)
    (fun c d a b Hc Hd ⇒ denotes_add (proj1_sig c) (proj1_sig d) a b Hc Hd)
    (fun c d a b Hc Hd ⇒ denotes_mul (proj1_sig c) (proj1_sig d) a b Hc Hd).
