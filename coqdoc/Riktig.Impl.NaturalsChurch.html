The Church numerals are naturals: the carrier is the terms convertible to a numeral, compared up to conversion; zero and one are the numerals, successor, addition and multiplication the programs. Denotation into Coq's naturals provides every law through denotation_naturals.
Definition ChurchNat : Type := { t : term | ∃ n : nat, joinable t (church n) }.

Instance ChurchNat_EqRel : EqRel ChurchNat := fun c d ⇒ joinable (proj1_sig c) (proj1_sig d).
Instance ChurchNat_Zero : Zero ChurchNat := exist _ (church 0) (is_numeral 0).
Instance ChurchNat_One : One ChurchNat := exist _ (church 1) (is_numeral 1).

Lemma succ_wit : ∀ t : term, (∃ n : nat, joinable t (church n)) →
  ∃ n : nat, joinable (App SUCC t) (church n).
Proof.
  intros t [n Hn]. ∃ (S n).
  apply (joinable_trans _ (App SUCC (church n))).
  - apply joinable_app; [ apply joinable_refl | exact Hn ].
  - exact (succ_join n).
Qed.

Lemma add_wit : ∀ t u : term,
  (∃ n : nat, joinable t (church n)) → (∃ n : nat, joinable u (church n)) →
  ∃ n : nat, joinable (App (App PLUS t) u) (church n).
Proof.
  intros t u [a Ha] [b Hb]. ∃ ((a + b)%nat).
  apply (joinable_trans _ (App (App PLUS (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Ha ] | exact Hb ].
  - exact (plus_join a b).
Qed.

Lemma mul_wit : ∀ t u : term,
  (∃ n : nat, joinable t (church n)) → (∃ n : nat, joinable u (church n)) →
  ∃ n : nat, joinable (App (App MULT t) u) (church n).
Proof.
  intros t u [a Ha] [b Hb]. ∃ ((a × b)%nat).
  apply (joinable_trans _ (App (App MULT (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Ha ] | exact Hb ].
  - exact (mult_join a b).
Qed.

Instance ChurchNat_HasSucc : HasSucc ChurchNat :=
  fun c ⇒ exist _ (App SUCC (proj1_sig c)) (succ_wit _ (proj2_sig c)).
Instance ChurchNat_HasAdd : HasAdd ChurchNat :=
  fun c d ⇒ exist _ (App (App PLUS (proj1_sig c)) (proj1_sig d)) (add_wit _ _ (proj2_sig c) (proj2_sig d)).
Instance ChurchNat_HasMult : HasMult ChurchNat :=
  fun c d ⇒ exist _ (App (App MULT (proj1_sig c)) (proj1_sig d)) (mul_wit _ _ (proj2_sig c) (proj2_sig d)).

The denotation facts

Definition denotes (c : ChurchNat) (n : nat) : Prop := joinable (proj1_sig c) (church n).

Lemma denotes_total : ∀ c : ChurchNat, ∃ n : nat, denotes c n.
Proof. intros c. exact (proj2_sig c). Qed.

Lemma denotes_sur : ∀ n : nat, ∃ c : ChurchNat, denotes c n.
Proof. intros n. ∃ (exist _ (church n) (is_numeral n)). apply joinable_refl. Qed.

Lemma denotes_unique : ∀ (c : ChurchNat) (n m : nat), denotes c n → denotes c m → n == m.
Proof. intros c n m Hn Hm. exact (church_injective n m (joinable_trans _ _ _ (joinable_sym _ _ Hn) Hm)). Qed.

Lemma denotes_faithful : ∀ (c d : ChurchNat) (n m : nat), denotes c n → denotes d m → n == m → c == d.
Proof.
  intros c d n m Hc Hd Hnm. assert (Heq : n = m) by exact Hnm. subst m.
  exact (joinable_trans _ _ _ Hc (joinable_sym _ _ Hd)).
Qed.

Lemma denotes_proper_l : ∀ (c d : ChurchNat) (n : nat), c == d → denotes c n → denotes d n.
Proof. intros c d n Hcd Hc. exact (joinable_trans _ _ _ (joinable_sym _ _ Hcd) Hc). Qed.

Lemma denotes_proper_r : ∀ (c : ChurchNat) (n m : nat), n == m → denotes c n → denotes c m.
Proof. intros c n m Hnm Hc. assert (Heq : n = m) by exact Hnm. subst m. exact Hc. Qed.

Lemma denotes_zero : denotes 0 0.
Proof. exact (joinable_refl _). Qed.

Lemma denotes_one : denotes 1 1.
Proof. exact (joinable_refl _). Qed.

Lemma denotes_succ : ∀ (c : ChurchNat) (n : nat), denotes c n → denotes (succ c) (S n).
Proof.
  intros c n Hc. unfold denotes. simpl.
  apply (joinable_trans _ (App SUCC (church n))).
  - apply joinable_app; [ apply joinable_refl | exact Hc ].
  - exact (succ_join n).
Qed.

Lemma denotes_add : ∀ (c d : ChurchNat) (a b : nat), denotes c a → denotes d b → denotes (c + d) ((a + b)%nat).
Proof.
  intros c d a b Hc Hd. unfold denotes. simpl.
  apply (joinable_trans _ (App (App PLUS (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Hc ] | exact Hd ].
  - exact (plus_join a b).
Qed.

Lemma denotes_mul : ∀ (c d : ChurchNat) (a b : nat), denotes c a → denotes d b → denotes (c × d) ((a × b)%nat).
Proof.
  intros c d a b Hc Hd. unfold denotes. simpl.
  apply (joinable_trans _ (App (App MULT (church a)) (church b))).
  - apply joinable_app; [ apply joinable_app; [ apply joinable_refl | exact Hc ] | exact Hd ].
  - exact (mult_join a b).
Qed.

The structure, read off the denotation

Instance ChurchNat_Naturals : Naturals ChurchNat 0 succ 1 add mult :=
  denotation_naturals (N0 := nat) denotes
    denotes_total denotes_sur denotes_unique denotes_faithful
    denotes_proper_l denotes_proper_r denotes_zero denotes_one
    denotes_succ denotes_add denotes_mul.
