Lemma cons_eq_head : ∀ {A : Type} (a1 a2 : A) (l1 l2 : list A),
  a1 :: l1 = a2 :: l2 → a1 = a2.
Proof.
  intros A a1 a2 l1 l2 H.
  exact (f_equal (fun l ⇒ match l with [] ⇒ a1 | a :: _ ⇒ a end) H).
Qed.

Lemma cons_eq_tail : ∀ {A : Type} (a1 a2 : A) (l1 l2 : list A),
  a1 :: l1 = a2 :: l2 → l1 = l2.
Proof.
  intros A a1 a2 l1 l2 H.
  exact (f_equal (fun l ⇒ match l with [] ⇒ l1 | _ :: t ⇒ t end) H).
Qed.

Lemma map_roundtrip : ∀ {A B : Type} (enc : A → B) (dec : B → A),
  (∀ a : A, dec (enc a) = a) →
  ∀ l : list A, map dec (map enc l) = l.
Proof.
  intros A B enc dec Hround l.
  induction l as [| a l IH]; simpl.
  - reflexivity.
  - rewrite Hround, IH. reflexivity.
Qed.

Lemma perm_transpose : ∀ {A : Type} (a b c : list A) (x y : A),
  Permutation (a ++ x :: b ++ y :: c) (a ++ y :: b ++ x :: c).
Proof.
  intros A a b c x y.
  apply Permutation_app_head.
  calc (x :: b ++ y :: c)
    == (x :: y :: b ++ c)
    == (y :: x :: b ++ c)
    == (y :: b ++ x :: c).
  - apply perm_skip, Permutation_sym, Permutation_middle.
  - apply perm_swap.
  - apply perm_skip, Permutation_middle.
Qed.

Lemma NoDup_map_injective : ∀ {J K : Type} (h : J → K) (l : list J),
  (∀ x y : J, h x = h y → x = y) →
  NoDup l → NoDup (map h l).
Proof.
  intros J K h l Hinj; induction l as [| a l IH]; intros Hnd.
  - constructor.
  - inversion Hnd as [| x xs Hnotin Hnd' Hxs]; subst.
    rewrite (map_cons h a l).
    constructor.
    + intros Hin.
      apply in_map_iff in Hin.
      destruct Hin as [x [Hx Hinx]].
      apply Hnotin.
      rewrite <- (Hinj x a Hx).
      exact Hinx.
    + exact (IH Hnd').
Qed.

Lemma map_add_seq : ∀ (len s a : nat),
  map (Nat.add s) (seq a len) = seq (a + s) len.
Proof.
  induction len as [| len IH]; intros s a.
  - reflexivity.
  - simpl. rewrite (IH s (S a)), (Nat.add_comm s a). reflexivity.
Qed.

Lemma seq_split_at : ∀ (n j : nat), (j < n)%nat →
  seq 0 n = seq 0 j ++ j :: seq (S j) (n - S j).
Proof.
  intros n j Hj.
  replace n with (j + (n - j))%nat at 1 by lia.
  rewrite (seq_app j (n - j) 0).
  replace (n - j)%nat with (S (n - S j)) by lia.
  reflexivity.
Qed.

Lemma flat_map_cons : ∀ {J K : Type} (h : J → list K) (a : J) (l : list J),
  flat_map h (a :: l) = h a ++ flat_map h l.
Proof.
  intros J K h a l.
  reflexivity.
Qed.

Lemma NoDup_flat_map : ∀ {J K : Type} (h : J → list K) (l : list J),
  (∀ j : J, In j l → NoDup (h j)) →
  (∀ (j1 j2 : J) (x : K),
     In j1 l → In j2 l → In x (h j1) → In x (h j2) → j1 = j2) →
  NoDup l → NoDup (flat_map h l).
Proof.
  intros J K h l; induction l as [| a l IH]; intros Hblocks Hdisjoint Hnd.
  - constructor.
  - inversion Hnd as [| x xs Hnotin Hnd' Hxs]; subst.
    rewrite (flat_map_cons h a l).
    apply NoDup_app.
    + exact (Hblocks a (or_introl eq_refl)).
    + apply IH.
      × intros j Hj. exact (Hblocks j (or_intror Hj)).
      × intros j1 j2 x Hj1 Hj2. exact (Hdisjoint j1 j2 x (or_intror Hj1) (or_intror Hj2)).
      × exact Hnd'.
    + intros x Hxa Hxl.
      apply in_flat_map in Hxl.
      destruct Hxl as [j [Hj Hxj]].
      apply Hnotin.
      rewrite (Hdisjoint a j x (or_introl eq_refl) (or_intror Hj) Hxa Hxj).
      exact Hj.
Qed.

Lemma length_flat_map_constant : ∀ {J K : Type} (h : J → list K)
    (c : nat) (l : list J),
  (∀ j : J, In j l → length (h j) = c) →
  length (flat_map h l) = (length l × c)%nat.
Proof.
  intros J K h c l; induction l as [| a l IH]; intros Hlen.
  - reflexivity.
  - rewrite (flat_map_cons h a l),
      (length_app (h a) (flat_map h l)), (Hlen a (or_introl eq_refl)),
      (IH (fun j Hj ⇒ Hlen j (or_intror Hj))).
    reflexivity.
Qed.

Section RemoveFacts.
  Context {J : Type}.
  Context (eq_dec : ∀ x y : J, {x = y} + {x ≠ y}).

  Lemma remove_cons_other : ∀ (p a : J) (l : list J),
    p ≠ a → remove eq_dec p (a :: l) = a :: remove eq_dec p l.
  Proof.
    intros p a l Hne.
    simpl.
    destruct (eq_dec p a) as [Heq | _].
    - destruct (Hne Heq).
    - reflexivity.
  Qed.

  Lemma remove_preserves_NoDup : ∀ (p : J) (l : list J),
    NoDup l → NoDup (remove eq_dec p l).
  Proof.
    intros p l; induction l as [| a l IH]; intros Hnd.
    - constructor.
    - inversion Hnd as [| x xs Hnotin Hnd' Hxs]; subst.
      destruct (eq_dec p a) as [Heq | Hne].
      + rewrite <- Heq, (remove_cons eq_dec p l).
        exact (IH Hnd').
      + rewrite (remove_cons_other p a l Hne).
        constructor.
        × intros Hin.
          exact (Hnotin (proj1 (in_remove eq_dec l a p Hin))).
        × exact (IH Hnd').
  Qed.

End RemoveFacts.


Lemma PermutationA_length : ∀ {A : Type} {eqA : relation A}
    (l l' : list A),
  PermutationA eqA l l' → length l = length l'.
Proof.
  intros A eqA l l' Hp;
    induction Hp as [| x y l1 l2 _ _ IH | | l1 l2 l3 _ IH1 _ IH2].
  - reflexivity.
  - simpl. rewrite IH. reflexivity.
  - reflexivity.
  - rewrite IH1. exact IH2.
Qed.

Lemma InA_map : ∀ {A : Type} {eqA : relation A} (h : A → A)
    (Ph : Proper (eqA ==> eqA) h) (l : list A) (x : A),
  InA eqA x l → InA eqA (h x) (map h l).
Proof.
  intros A eqA h Ph l x Hin; induction Hin as [y l Hxy | y l _ IH].
  - exact (InA_cons_hd (map h l) (Ph x y Hxy)).
  - exact (InA_cons_tl (h y) IH).
Qed.

Lemma NoDupA_map_injectiveA : ∀ {A : Type} {eqA : relation A}
    (h : A → A) (l : list A),
  (∀ x y : A, eqA (h x) (h y) → eqA x y) →
  NoDupA eqA l → NoDupA eqA (map h l).
Proof.
  intros A eqA h l Hinj Hnd; induction Hnd as [| a l Hnotin _ IH].
  - constructor.
  - constructor.
    + intros Hin.
      apply Hnotin.
      clear Hnotin IH.
      induction l as [| b l IHl]; inversion Hin; subst.
      × exact (InA_cons_hd l (Hinj a b ltac:(assumption))).
      × exact (InA_cons_tl b (IHl ltac:(assumption))).
    + exact IH.
Qed.

Lemma removeA_notin : ∀ {A : Type} {eqA : relation A}
    (dA : ∀ x y : A, {eqA x y} + {¬ eqA x y}) (p : A) (l : list A),
  ¬ InA eqA p l → removeA dA p l = l.
Proof.
  intros A eqA dA p l; induction l as [| a l IH]; intros Hnotin.
  - reflexivity.
  - simpl. destruct (dA p a) as [Heq | Hne].
    + destruct (Hnotin (InA_cons_hd l Heq)).
    + rewrite (IH (fun Hin ⇒ Hnotin (InA_cons_tl a Hin))). reflexivity.
Qed.


Lemma InA_eq_of_In : ∀ (A : Type) (x : A) (l : list A),
  In x l → InA eq x l.
Proof.
  intros A x l Hin.
  exact (In_InA eq_equivalence Hin).
Qed.

Lemma NoDupA_eq_of_NoDup : ∀ (A : Type) (l : list A),
  NoDup l → NoDupA eq l.
Proof.
  intros A l Hnd; induction Hnd as [| a l Hnotin _ IH].
  - constructor.
  - constructor.
    + intros Hin.
      apply Hnotin.
      destruct (proj1 (InA_alt eq a l) Hin) as [y [Heq Hy]].
      rewrite Heq. exact Hy.
    + exact IH.
Qed.


Lemma In_of_InA_discrete : ∀ (A : Type) (R : relation A) (x : A) (l : list A),
  (∀ a b : A, R a b → a = b) → InA R x l → In x l.
Proof.
  intros A R x l Hdiscrete Hin; induction Hin as [y l Hxy | y l _ IH].
  - left. exact (eq_sym (Hdiscrete x y Hxy)).
  - right. exact IH.
Qed.

Lemma NoDupA_of_NoDup_discrete : ∀ (A : Type) (R : relation A) (l : list A),
  (∀ a b : A, R a b → a = b) → NoDup l → NoDupA R l.
Proof.
  intros A R l Hdiscrete Hnd; induction Hnd as [| a l Hnotin _ IH].
  - constructor.
  - constructor; [| exact IH ].
    intros Hin. exact (Hnotin (In_of_InA_discrete A R a l Hdiscrete Hin)).
Qed.

Lemma InA_of_In_reflexive : ∀ (A : Type) (R : relation A) (x : A) (l : list A),
  Reflexive R → In x l → InA R x l.
Proof.
  intros A R x l Hrefl; induction l as [| a t IH]; intros Hin.
  - destruct Hin.
  - destruct Hin as [Hax | Hin].
    + left. rewrite Hax. exact (Hrefl x).
    + right. exact (IH Hin).
Qed.

Lemma NoDup_of_NoDupA : ∀ (A : Type) (R : relation A) (l : list A),
  Reflexive R → NoDupA R l → NoDup l.
Proof.
  intros A R l Hrefl Hnd; induction Hnd as [| a l HnotinA _ IH].
  - constructor.
  - constructor; [| exact IH ].
    intros Hin. exact (HnotinA (InA_of_In_reflexive A R a l Hrefl Hin)).
Qed.

Lemma two_in_split : ∀ {A : Type} (l : list A) (x y : A),
  x ≠ y → In x l → In y l →
  ∃ (l1 : list A) (a : A) (l2 : list A) (b : A) (l3 : list A),
    l = l1 ++ a :: l2 ++ b :: l3
    ∧ ((a = x ∧ b = y) ∨ (a = y ∧ b = x)).
Proof.
  intros A l x y Hxy; induction l as [| h t IH]; intros Hx Hy.
  - destruct Hx.
  - destruct Hx as [Hhx | Hx]; destruct Hy as [Hhy | Hy].
    + destruct (Hxy (eq_trans (eq_sym Hhx) Hhy)).
    + destruct (in_split y t Hy) as [l2 [l3 Ht]].
      ∃ [], h, l2, y, l3.
      split; [ rewrite Ht; reflexivity | left; split; [ exact Hhx | reflexivity ] ].
    + destruct (in_split x t Hx) as [l2 [l3 Ht]].
      ∃ [], h, l2, x, l3.
      split; [ rewrite Ht; reflexivity | right; split; [ exact Hhy | reflexivity ] ].
    + destruct (IH Hx Hy) as [l1 [a [l2 [b [l3 [Ht Hab]]]]]].
      ∃ (h :: l1), a, l2, b, l3.
      split; [ rewrite Ht; reflexivity | exact Hab ].
Qed.


Fixpoint upd_nth {A : Type} (i : nat) (x : A) (l : list A) : list A :=
  match i, l with
  | _, [] ⇒ []
  | O, _ :: t ⇒ x :: t
  | S i', h :: t ⇒ h :: upd_nth i' x t
  end.

Lemma upd_nth_length {A : Type} : ∀ (l : list A) (i : nat) (x : A),
  length (upd_nth i x l) = length l.
Proof.
  induction l as [|h t IH]; intros [|i] x; cbn; try reflexivity.
  now rewrite IH.
Qed.

Lemma nth_error_upd_nth_eq {A : Type} : ∀ (l : list A) (i : nat) (x : A),
  (i < length l)%nat → nth_error (upd_nth i x l) i = Some x.
Proof.
  induction l as [|h t IH]; intros [|i] x H; cbn in *; try lia.
  - reflexivity.
  - apply IH; lia.
Qed.

Lemma nth_error_upd_nth_neq {A : Type} : ∀ (l : list A) (i j : nat) (x : A),
  j ≠ i → nth_error (upd_nth i x l) j = nth_error l j.
Proof.
  induction l as [|h t IH]; intros [|i] [|j] x N; cbn; try reflexivity.
  - congruence.
  - apply IH; congruence.
Qed.


Lemma firstn_S_nth_error {A : Type} : ∀ (l : list A) (t : nat) (x : A),
  nth_error l t = Some x → firstn (S t) l = firstn t l ++ [x].
Proof.
  induction l as [|h tl IH]; intros [|t] x H; cbn in H; try discriminate.
  - injection H as <-. reflexivity.
  - rewrite !firstn_cons, (IH t x H). reflexivity.
Qed.


Lemma perm_by_count {A : Type} (eq_dec : ∀ x y : A, {x = y} + {x ≠ y})
    (l1 l2 : list A) :
  forallb (fun x ⇒ Nat.eqb (count_occ eq_dec l1 x) (count_occ eq_dec l2 x))
          (l1 ++ l2) = true →
  Permutation l1 l2.
Proof.
  intros H. apply (Permutation_count_occ eq_dec). intros x.
  destruct (in_dec eq_dec x (l1 ++ l2)) as [I|NI].
  - apply Nat.eqb_eq. exact (proj1 (forallb_forall _ (l1 ++ l2)) H x I).
  - assert (N1 : ¬ In x l1) by (intro; apply NI, in_or_app; auto).
    assert (N2 : ¬ In x l2) by (intro; apply NI, in_or_app; auto).
    apply (count_occ_not_In eq_dec) in N1, N2.
    now rewrite N1, N2.
Qed.
