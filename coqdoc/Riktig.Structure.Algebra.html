From Riktig.Structure Require Export Ring VectorSpace.

An algebra over a field is a ring that is also a vector space over the field, a scalar on either factor of a product being a scalar on the product.
Module Algebra.
  Class Algebra (F A : Type) `{Field F}
      {Ae : EqRel A} (Az : Zero A) (Aa : HasAdd A) (An : HasNeg A)
      (Ao : One A) (Am : HasMult A) (As : HasScale F A) := {
    vectors :: VectorSpace F A 0 add neg scale;
    ring :: Ring A 0 add neg 1 mult;
    smul_mul_assoc : ∀ (a : F) (x y : A), (a *: x) × y == a *: (x × y);
    mul_smul_comm : ∀ (a : F) (x y : A), x × (a *: y) == a *: (x × y);
  }.
  Hint Mode Algebra - ! - - - - - - - - - - - - - - - : typeclass_instances.
End Algebra.
