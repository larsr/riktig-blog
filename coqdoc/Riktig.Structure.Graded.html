Library Riktig.Structure.Graded

From Riktig.Structure Require Export Algebra Linear Naturals NatRec.

A grading of an algebra, with grades in the naturals N up to a top grade: taking the grade-k part is linear, respects the grade, is idempotent, kills every other grade, and the parts up to the top grade add up to the element.
Module Graded.
  Class Graded (N F A : Type) `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}
      `{Algebra F A} (grading : HasGrade N A) (top_grade : N) := {
    grade_linear :: ∀ k : N, Linear F A A (grade_part k);
    grade_index_proper : ∀ (k k' : N) (x : A), k == k' → ⟨x⟩_k == ⟨x⟩_k';
    grade_idempotent : ∀ (k : N) (x : A), ⟨⟨x⟩_k⟩_k == ⟨x⟩_k;
    grade_orthogonal : ∀ (k l : N) (x : A), k =/= l → ⟨⟨x⟩_l⟩_k == 0;
    grade_complete : ∀ x : A, natrec A 0 (fun k acc ⇒ acc + ⟨x⟩_k) (succ top_grade) == x;
  }.
  Hint Mode Graded - ! ! - - - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Graded.
