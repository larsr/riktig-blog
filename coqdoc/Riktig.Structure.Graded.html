The grade projection: grade_part k a is the grade-k part of a, written ⟨ a ⟩_ k.
Class HasGrade (A : Type) := grade_part : nat → A → A.
Notation "⟨ a ⟩_ k" := (grade_part k a)
  (at level 0, k at level 0, format "⟨ a ⟩_ k").
Hint Mode HasGrade ! : typeclass_instances.
Typeclasses Transparent HasGrade.

A grading of A with top grade n: each grade projection is linear, projections are idempotent and mutually orthogonal, and the bounded sum of the grade parts of any element returns it.
A parity grading of a ring: an even part and an odd part, each closed under zero and addition, whose products obey the Z/2 rule — even times even is even, even times odd is odd, odd times odd is even again.
Module ParityGraded.
  Class ParityGraded (A : Type)
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A}
      {Ao : One A} {Am : HasMult A} (even odd : A → Prop) := {
    parity_even_zero : even 0;
    parity_odd_zero : odd 0;
    parity_even_one : even 1;
    parity_even_add : ∀ x y : A, even x → even y → even (x + y);
    parity_odd_add : ∀ x y : A, odd x → odd y → odd (x + y);

    even_mul_even : ∀ x y : A, even x → even y → even (x × y);
    even_mul_odd : ∀ x y : A, even x → odd y → odd (x × y);
    odd_mul_even : ∀ x y : A, odd x → even y → odd (x × y);
    odd_mul_odd : ∀ x y : A, odd x → odd y → even (x × y);
  }.
  Hint Mode ParityGraded ! - - - - - - - : typeclass_instances.
End ParityGraded.
Export ParityGraded.

Module Graded.
  Class Graded (n : nat) (F A : Type)
      {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {As : HasScale F A}
      {AM : Monoid A} {Ag : HasGrade A}
      (bigop_list : ∀ J : Type, (J → A) → list J → A)
      {BL : BigOp A bigop_list} := {
    grade_linear :: ∀ k : nat, Linear F A A (grade_part k);
    grade_idempotent : ∀ (k : nat) (a : A), ⟨ ⟨ a ⟩_ k ⟩_ k == ⟨ a ⟩_ k;
    grade_orthogonal : ∀ (k l : nat) (a : A),
                         k ≠ l → ⟨ ⟨ a ⟩_ l ⟩_ k == 0;
    grade_complete : ∀ a : A,
      bigop_list _ (fun k ⇒ ⟨ a ⟩_ k) (seq 0 (S n)) == a;
  }.
  Hint Mode Graded - - ! - - - - - - - - : typeclass_instances.
End Graded.
Export Graded.
