From Riktig Require Export Structure.AbGroup Theory.Group.

Section AbGroupTheory.
  Context {G : Type} `{AbGroup G}.

  Theorem add_commutative : ∀ a b : G, a + b == b + a.
  Proof.
    intros a b. exact (commutative a b).
  Qed.

  Theorem add_swap_r : ∀ a b c : G, (a + b) + c == (a + c) + b.
  Proof.
    intros a b c.
    calc ((a + b) + c)
      == (a + (b + c)) { associative }
      == (a + (c + b)) { (commutative b) }
      == ((a + c) + b) { associative }.
  Qed.

  Theorem add_swap_l : ∀ a b c : G, a + (b + c) == b + (a + c).
  Proof. exact add_swap_front. Qed.

  Theorem add_cycle : ∀ a b c : G, (a + b) + c == (b + c) + a.
  Proof.
    intros a b c.
    calc ((a + b) + c)
      == (a + (b + c)) { associative }
      == ((b + c) + a) { commutative }.
  Qed.

  Theorem add_interchange : ∀ a b c d : G,
    (a + b) + (c + d) == (a + c) + (b + d).
  Proof.
    intros a b c d.
    calc ((a + b) + (c + d))
      == (a + (b + (c + d))) { associative }
      == (a + (c + (b + d))) { (add_swap_l b) }
      == ((a + c) + (b + d)) { associative }.
  Qed.

  Theorem add_interchange_triple : ∀ a1 a2 b1 b2 c1 c2 : G,
    ((a1 + a2) + (b1 + b2)) + (c1 + c2)
      == ((a1 + b1) + c1) + ((a2 + b2) + c2).
  Proof.
    intros a1 a2 b1 b2 c1 c2.
    calc (((a1 + a2) + (b1 + b2)) + (c1 + c2))
      == (((a1 + b1) + (a2 + b2)) + (c1 + c2)) { (add_interchange a1) }
      == (((a1 + b1) + c1) + ((a2 + b2) + c2)) { add_interchange }.
  Qed.

  Theorem add_rotate : ∀ a b c d : G,
    (a + b) + (c + d) == (a + d) + (b + c).
  Proof.
    intros a b c d.
    calc ((a + b) + (c + d))
      == ((a + b) + (d + c)) { (commutative c) }
      == ((a + d) + (b + c)) { add_interchange }.
  Qed.

  Theorem neg_add : ∀ a b : G, - (a + b) == - a + - b.
  Proof.
    intros a b.
    calc (- (a + b))
      == (- b + - a) { neg_add_flip }
      == (- a + - b) { commutative }.
  Qed.

  #[export] Instance neg_additive : Additive G G add add neg := neg_add.

  Theorem add_sub_cancel : ∀ a b : G, a + (b + - a) == b.
  Proof.
    intros a b.
    calc (a + (b + - a))
      == (a + (- a + b)) { (commutative b) }
      == ((a + - a) + b) { associative }
      == (0 + b) { right_inverse }
      == b { left_identity }.
  Qed.

  Theorem add_sub_pair_cancel : ∀ a b c : G,
    (a + b) + ((c + - a) + - b) == c.
  Proof.
    intros a b c.
    calc ((a + b) + ((c + - a) + - b))
      == ((a + b) + (c + (- a + - b))) { (associative c) }
      == ((a + b) + (c + - (a + b))) { additive[neg] }
      == c { add_sub_cancel }.
  Qed.

  Theorem sub_pair : ∀ a b c d : G,
    (a + b) + - (c + d) == (a + - c) + (b + - d).
  Proof.
    intros a b c d.
    calc ((a + b) + - (c + d))
      == ((a + b) + (- c + - d)) { additive[neg] }
      == ((a + - c) + (b + - d)) { add_interchange }.
  Qed.

  Theorem sub_add_cancel_r : ∀ p q x : G, (p + - (q + x)) + x == p + - q.
  Proof.
    intros p q x.
    calc ((p + - (q + x)) + x)
      == ((p + (- q + - x)) + x) { additive[neg] }
      == (((p + - q) + - x) + x) { associative }
      == ((p + - q) + (- x + x)) { associative }
      == ((p + - q) + 0) { left_inverse }
      == (p + - q) { right_identity }.
  Qed.

  Theorem neg_add_associative : ∀ a b c : G, a + - (b + c) == (a + - b) + - c.
  Proof.
    intros a b c.
    calc (a + - (b + c))
      == (a + (- b + - c)) { additive[neg] }
      == ((a + - b) + - c) { associative }.
  Qed.

  Theorem sub_common_cancel : ∀ p q x y : G,
    (p + - (q + x)) + - (p + - (q + y)) == y + - x.
  Proof.
    intros p q x y.
    calc ((p + - (q + x)) + - (p + - (q + y)))
      ==[negp] ((p + - (q + x)) + (- p + (q + y)))
      ==[interL] (- (q + x) + (q + y))
      ==[interR] (- x + y)
      ==[swap] (y + - x).
    [negp]: now rewrite (additive[neg] p (- (q + y))), (neg_involutive (q + y)).
    [interL]: now rewrite (add_interchange p (- (q + x)) (- p) (q + y)), right_inverse, left_identity.
    [interR]: now rewrite (additive[neg] q x), (add_interchange (- q) (- x) q y), left_inverse, left_identity.
    [swap]: exact (commutative _ _).
  Qed.

  Theorem add_neg_cancel_l : ∀ a b : G, (a + b) + - a == b.
  Proof.
    intros a b.
    calc ((a + b) + - a)
      == ((b + a) + - a) { (commutative a) }
      == (b + (a + - a)) { associative }
      == (b + 0) { right_inverse }
      == b { right_identity }.
  Qed.

  Theorem sub_mid_cancel : ∀ p q x y : G,
    (p + - (q + x)) + (x + - y) == p + - (q + y).
  Proof.
    intros p q x y.
    calc ((p + - (q + x)) + (x + - y))
      == (p + (- (q + x) + (x + - y))) { associative }
      == (p + ((- q + - x) + (x + - y))) { additive[neg] }
      == (p + (- q + - y)) { sub_add_mid_cancel }
      == (p + - (q + y)) { additive[neg] }.
  Qed.

  Theorem add_transpose : ∀ a b c : G, a + b == c → b == c + - a.
  Proof.
    intros a b c Heq.
    apply (eq_sub_of_add_eq b c a).
    calc (b + a)
      == (a + b) { commutative }
      == (c : G) { Heq }.
  Qed.

  Theorem add_double_split : ∀ a b : G, a + a == (a + b) + (a + - b).
  Proof.
    intros a b.
    calc (a + a)
      == ((a + a) + (0 : G)) { right_identity }
      == ((a + a) + (b + - b)) { right_inverse }
      == ((a + b) + (a + - b)) { add_interchange }.
  Qed.

  Theorem four_term_cancel : ∀ a b c d : G,
    (a + b) + (c + d) + - (a + (b + c)) == d.
  Proof.
    intros a b c d.
    calc (((a + b) + (c + d)) + - (a + (b + c)))
      == ((a + (b + (c + d))) + - (a + (b + c))) { (associative a) }
      == ((a + ((b + c) + d)) + - (a + (b + c))) { (associative b) }
      == (((a + (b + c)) + d) + - (a + (b + c))) { associative }
      == d { add_neg_cancel_l }.
  Qed.

End AbGroupTheory.
