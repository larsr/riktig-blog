Section AbGroupTheory.
  Context {G : Type} `{AbGroup G}.

  Theorem neg_add : ∀ a b : G, - (a + b) == - a + - b.
  Proof.
    intros a b.
    calc (- (a + b))
      == (- b + - a) { neg_add_rev }
      == (- a + - b) { commutative }.
  Qed.

  Theorem add_sub_cancel_left : ∀ a b : G, (a + b) + - a == b.
  Proof.
    intros a b.
    calc ((a + b) + - a)
      == ((b + a) + - a) { (commutative a b) }
      == b { add_neg_cancel_right }.
  Qed.

  Theorem add_add_neg_cancel : ∀ a b : G, a + (b + - a) == b.
  Proof.
    intros a b.
    calc (a + (b + - a))
      == (a + (- a + b)) { (commutative b (- a)) }
      == b { add_neg_cancel_left }.
  Qed.

  Theorem sub_sub : ∀ a b c : G, a + - (b + c) == (a + - b) + - c.
  Proof.
    intros a b c.
    calc (a + - (b + c))
      == (a + (- b + - c)) { neg_add }
      == ((a + - b) + - c) { associative }.
  Qed.

  Theorem add_sub_add_comm : ∀ a b c d : G,
    (a + b) + - (c + d) == (a + - c) + (b + - d).
  Proof.
    intros a b c d.
    calc ((a + b) + - (c + d))
      == ((a + b) + (- c + - d)) { neg_add }
      == ((a + - c) + (b + - d)) { add_add_add_comm }.
  Qed.

Moving the left summand across an equation.
  Theorem eq_sub_of_add_eq' : ∀ a b c : G, a + b == c → b == c + - a.
  Proof.
    intros a b c Heq.
    apply (eq_sub_of_add_eq b c a).
    calc (b + a)
      == (a + b) { commutative }
      == c { Heq }.
  Qed.

End AbGroupTheory.
