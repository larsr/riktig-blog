Local Open Scope Z_scope.

Local Notation EvalZ :=
  (Eval Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).



Theorem sum_loop_eval :
  ∀ (base acc idx : Z) (len : nat) (l : list Z) (m : Machine Z Z),
    acc ≠ idx →
    (acc < base ∨ base + Z.of_nat len ≤ acc) →
    (idx < base ∨ base + Z.of_nat len ≤ idx) →
    mem_read_cells len base m = Some l →
    store_lookup Z.eqb acc (mach_store m) = Some (CellVal 0) →
    store_lookup Z.eqb idx (mach_store m) = Some (CellVal 0) →
    ∃ m' : Machine Z Z,
      EvalZ (while_m (lt_ Z.ltb (read_m idx) (ret_t (Z.of_nat len)))
                     (sum_l1_body base acc idx))
            m (Some tt) m' ∧
      store_lookup Z.eqb acc (mach_store m')
        = Some (CellVal (fold_left Z.add l 0)) ∧
      mem_read_cells len base m' = Some l.
Proof.
  intros base acc idx len l m Hne Haccout Hidxout Hview Hacc0 Hidx0.
  assert (Hstep : while_invariant_step
      (lt_ Z.ltb (read_m idx) (ret_t (Z.of_nat len)))
      (sum_l1_body base acc idx)
      (fun mm : Machine Z Z ⇒
         ∃ t : nat, (t ≤ len)%nat ∧
           store_lookup Z.eqb idx (mach_store mm)
             = Some (CellVal (Z.of_nat t)) ∧
           store_lookup Z.eqb acc (mach_store mm)
             = Some (CellVal (fold_left Z.add (firstn t l) 0)) ∧
           mem_read_cells len base mm = Some l)
      (fun mm : Machine Z Z ⇒
         store_lookup Z.eqb acc (mach_store mm)
           = Some (CellVal (fold_left Z.add l 0)) ∧
         mem_read_cells len base mm = Some l)
      (fun mm : Machine Z Z ⇒
         match store_lookup Z.eqb idx (mach_store mm) with
         | Some (CellVal v) ⇒ Z.to_nat (Z.of_nat len - v)
         | _ ⇒ 0%nat
         end)).
  { intros mm (t & Hle & Hidxm & Haccm & Hviewm).
    ∃ (Z.of_nat t <? Z.of_nat len), mm.
    split; [| split].
    -
      eapply eval_lt_; [eapply eval_read_m; exact Hidxm | apply eval_ret_t].
    -
      intros Hfalse.
      apply Z.ltb_ge in Hfalse.
      assert (Ht : t = len) by lia. subst t.
      rewrite firstn_all2 in Haccm
        by (rewrite (mem_read_cells_length len base mm l Hviewm); lia).
      exact (conj Haccm Hviewm).
    -
      intros Htrue.
      apply Z.ltb_lt in Htrue.
      assert (Hlt : (t < len)%nat) by lia.
      destruct (mem_read_cells_nth_error len t base mm l Hviewm Hlt)
        as (v & Hnth & Hlk).
      ∃ (mach_write idx (CellVal (Z.of_nat t + 1))
                (mach_write acc
                   (CellVal (fold_left Z.add (firstn t l) 0 + v)) mm)).
      split; [| split].
      + unfold sum_l1_body.
        eapply eval_bind_t; [eapply eval_read_m; exact Hidxm|].
        eapply eval_bind_t;
          [eapply eval_read_arr; [apply eval_ret_t | exact Hlk]|].
        eapply eval_bind_t; [eapply eval_read_m; exact Haccm|].
        eapply eval_then_t.
        × eapply eval_write_m. exact (store_lookup_member Z.eqb _ _ _ Haccm).
        × eapply eval_write_m. rewrite mach_write_store.
          apply store_member_insert_monotone.
          exact (store_lookup_member Z.eqb _ _ _ Hidxm).
      + ∃ (S t).
        split; [lia|]. split; [| split].
        × rewrite !mach_write_store.
          rewrite (store_lookup_insert_eq Z.eqb Z.eqb_eq).
          replace (Z.of_nat (S t)) with (Z.of_nat t + 1) by lia.
          reflexivity.
        × rewrite !mach_write_store.
          rewrite (store_lookup_insert_neq Z.eqb Z.eqb_eq) by exact Hne.
          rewrite (store_lookup_insert_eq Z.eqb Z.eqb_eq).
          rewrite (firstn_S_nth_error l t v Hnth), fold_left_app.
          reflexivity.
        × rewrite (mem_read_cells_insert_outside len base idx
                     (CellVal (Z.of_nat t + 1)) _ Hidxout).
          rewrite (mem_read_cells_insert_outside len base acc
                     (CellVal (fold_left Z.add (firstn t l) 0 + v)) mm Haccout).
          exact Hviewm.
      + cbn beta. rewrite !mach_write_store.
        rewrite (store_lookup_insert_eq Z.eqb Z.eqb_eq), Hidxm.
        lia. }
  destruct (eval_while_m_measure _ _ _ _ _ Hstep m
              ltac:(∃ 0%nat;
                    cbn [Z.of_nat firstn fold_left];
                    repeat split;
                    [lia | exact Hidx0 | exact Hacc0 | exact Hview]))
    as (m' & (Hacc' & Hview') & Hev).
  ∃ m'. exact (conj Hev (conj Hacc' Hview')).
Qed.



Lemma sum_l1_setup :
  ∀ (len : nat) (base : Z) (l : list Z) (m : Machine Z Z),
    mem_read_cells len base m = Some l →
    Z.leb (mach_sp m) (mach_hp m) = true →
    Z.leb (mach_sp m + 1) (mach_hp m) = true →
    (mach_sp m + 1 < base ∨ base + Z.of_nat len ≤ mach_sp m) →
    ∃ m4 : Machine Z Z,
      (∀ (a : Type) (k : Z → Z → MemT Z Z a)
              (r : option a) (m' : Machine Z Z),
         EvalZ (k (mach_sp m) (mach_sp m + 1)) m4 r m' →
         EvalZ (bind_t (@stack_alloc_m Z Z) (fun acc ⇒
                bind_t (@stack_alloc_m Z Z) (fun idx ⇒
                then_t (write_m acc 0)
                       (then_t (write_m idx 0) (k acc idx))))) m r m') ∧
      mem_read_cells len base m4 = Some l ∧
      store_lookup Z.eqb (mach_sp m) (mach_store m4) = Some (CellVal 0) ∧
      store_lookup Z.eqb (mach_sp m + 1) (mach_store m4) = Some (CellVal 0).
Proof.
  intros len base l m Hview Hsp1 Hsp2 Hout.
  ∃ (mach_write (mach_sp m + 1) (CellVal 0)
            (mach_write (mach_sp m) (CellVal 0)
               (MkMachine
                  (store_insert (mach_sp m + 1) (CellVal 0)
                     (store_insert (mach_sp m) (CellVal 0) (mach_store m)))
                  (mach_sp m + 1 + 1) (mach_hp m)))).
  split; [| split; [| split]].
  - intros a k r m' Hk.
    eapply eval_bind_t; [apply eval_stack_alloc_m; exact Hsp1|].
    eapply eval_bind_t;
      [apply eval_stack_alloc_m; cbn [mach_sp mach_hp]; exact Hsp2|].
    eapply eval_then_t.
    + apply eval_write_m. cbn [mach_store].
      apply store_member_insert_monotone.
      apply (store_member_insert_eq Z.eqb Z.eqb_eq).
    + eapply eval_then_t; [| exact Hk].
      apply eval_write_m. cbn [mach_write mach_store].
      apply store_member_insert_monotone.
      apply (store_member_insert_eq Z.eqb Z.eqb_eq).
  -
    transitivity (mem_read_cells len base m); [| exact Hview].
    apply mem_read_cells_ext. intros ad Had.
    cbn [mach_write mach_store].
    rewrite !(store_lookup_insert_neq Z.eqb Z.eqb_eq) by lia.
    reflexivity.
  - cbn [mach_write mach_store].
    rewrite (store_lookup_insert_neq Z.eqb Z.eqb_eq) by lia.
    apply (store_lookup_insert_eq Z.eqb Z.eqb_eq).
  - cbn [mach_write mach_store].
    apply (store_lookup_insert_eq Z.eqb Z.eqb_eq).
Qed.


Theorem sum_l1_correct :
  ∀ (len : nat) (base : Z) (l : list Z) (m : Machine Z Z),
    mem_read_cells len base m = Some l →
    Z.leb (mach_sp m) (mach_hp m) = true →
    Z.leb (mach_sp m + 1) (mach_hp m) = true →
    (mach_sp m + 1 < base ∨ base + Z.of_nat len ≤ mach_sp m) →
    ∃ m' : Machine Z Z,
      EvalZ (sum_l1 base (Z.of_nat len)) m (Some (fold_left Z.add l 0)) m' ∧
      mem_read_cells len base m' = Some l.
Proof.
  intros len base l m Hview Hsp1 Hsp2 Hout.
  destruct (sum_l1_setup len base l m Hview Hsp1 Hsp2 Hout)
    as (m4 & Hsetup & Hview4 & Hacc4 & Hidx4).
  destruct (sum_loop_eval base (mach_sp m) (mach_sp m + 1) len l m4
              ltac:(lia) ltac:(lia) ltac:(lia) Hview4 Hacc4 Hidx4)
    as (m5 & Hloop & Hacc5 & Hview5).
  ∃ m5. split; [| exact Hview5].
  apply (Hsetup Z (fun acc idx ⇒
           then_t (while_m (lt_ Z.ltb (read_m idx) (ret_t (Z.of_nat len)))
                           (sum_l1_body base acc idx))
                  (read_m acc))).
  cbn beta.
  eapply eval_then_t; [exact Hloop | eapply eval_read_m; exact Hacc5].
Qed.
