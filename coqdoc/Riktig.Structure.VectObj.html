From Coq Require Export Morphisms.
From Riktig Require Export Structure.VectorSpace Structure.Category
  Theory.LinearMap.

Record VectObj (F : Type) {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F}
    {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F} {Fi : HasInv F}
    {FF : Field F} := {
  vo_car : Type;
  vo_eq : EqRel vo_car;
  vo_zero : Zero vo_car;
  vo_add : HasAdd vo_car;
  vo_neg : HasNeg vo_car;
  vo_scale : HasScale F vo_car;
  vo_vs : VectorSpace F vo_car (Ve:=vo_eq) (Vz:=vo_zero) (Va:=vo_add) (Vn:=vo_neg) (Vs:=vo_scale)
}.

Arguments vo_car {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_eq {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_zero {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_add {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_neg {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_scale {F Fe Fz Fa Fn Fo Fm Fi FF}.
Arguments vo_vs {F Fe Fz Fa Fn Fo Fm Fi FF}.

#[global] Existing Instance vo_eq.
#[global] Existing Instance vo_zero.
#[global] Existing Instance vo_add.
#[global] Existing Instance vo_neg.
#[global] Existing Instance vo_scale.
#[global] Existing Instance vo_vs.

Definition vectObj (F : Type) `{FF : Field F} (T : Type)
  {e : EqRel T} {z : Zero T} {a : HasAdd T} {n : HasNeg T} {sc : HasScale F T}
  {VS : VectorSpace F T} : VectObj F :=
  {| vo_car := T; vo_eq := e; vo_zero := z; vo_add := a; vo_neg := n;
     vo_scale := sc; vo_vs := VS |}.
