Section ConstantAssignment.
  Context {F A : Type} `{Field F}
          {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {An : HasNeg A}
          {Ao : One A} {Am : HasMult A} {As : HasScale F A}
          {AR : Ring A} {AVS : VectorSpace F A}.
  Add Ring constant_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition constant_cexp : A → A := fun _ ⇒ 1.
  Definition constant_cos : F → F := fun _ ⇒ 1.
  Definition constant_sin : F → F := fun _ ⇒ 0.

  Lemma constant_cexp_add : ∀ x y : A,
    x × y == y × x →
    constant_cexp (x + y) == constant_cexp x × constant_cexp y.
  Proof. intros x y _. symmetry. exact (left_identity (1 : A)). Qed.

  Lemma constant_pythagoras : ∀ t : F,
    constant_cos t × constant_cos t + constant_sin t × constant_sin t == 1.
  Proof. intros t. unfold constant_cos, constant_sin. ring. Qed.

  Lemma constant_euler : ∀ (t : F) (e : A),
    e × e == - (1) →
    constant_cexp (t *: e) == constant_cos t *: 1 + constant_sin t *: e.
  Proof.
    intros t e _.
    symmetry.
    calc (((1 : F) *: (1 : A) + (0 : F) *: e) : A)
      == (((1 : A) + (0 : F) *: e) : A) { left_identity }
      == (((1 : A) + (0 : A)) : A) { zero_scale }
      == ((1 : A)) { right_identity }.
  Qed.

  Lemma constant_cos_even : ∀ t : F, constant_cos (- t) == constant_cos t.
  Proof. intros t. reflexivity. Qed.

  Lemma constant_sin_odd : ∀ t : F,
    constant_sin (- t) == - constant_sin t.
  Proof. intros t. symmetry. exact neg_zero. Qed.

  Lemma constant_sin_vanishes : ∀ t : F, constant_sin t == 0.
  Proof. intros t. reflexivity. Qed.

End ConstantAssignment.


Section CircleCharacteristicTwo.
  Context {F : Type} `{Field F} {FD : EquDec F} (two : (1 : F) + 1 == 0).
  Add Ring circle_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition circle_cos (t : F) : F := 1 + t.
  Definition circle_sin (t : F) : F := t.
  Definition circle_cexp (z : C F) : C F := mkC (1 + im z) (im z).

  Lemma circle_cos_proper : Proper (equ ==> equ) circle_cos.
  Proof. intros s t Hst. exact (op_proper 1 1 (reflexivity 1) s t Hst). Qed.

  Lemma circle_sin_proper : Proper (equ ==> equ) circle_sin.
  Proof. intros s t Hst. exact Hst. Qed.

  Lemma circle_cexp_proper : Proper (equ ==> equ) circle_cexp.
  Proof.
    intros z w [_ Him].
    split; simpl;
      [ exact (op_proper 1 1 (reflexivity 1) (im z) (im w) Him) | exact Him ].
  Qed.

  Lemma circle_pythagoras : ∀ t : F,
    circle_cos t × circle_cos t + circle_sin t × circle_sin t == 1.
  Proof.
    intros t. unfold circle_cos, circle_sin.
    calc ((1 + t) × (1 + t) + t × t)
      == ((1 × 1 + t × t) + t × t) { (square_of_sum two) }
      == [regroup] (1 + (t × t + t × t))
      == ((1 : F) + 0) { (double_vanishes two (t × t)) }
      == (1 : F) { right_identity }.
    [regroup]: ring.
  Qed.

  Lemma circle_cos_even : ∀ t : F, circle_cos (- t) == circle_cos t.
  Proof.
    intros t. unfold circle_cos.
    exact (op_proper 1 1 (reflexivity 1) (- t) t (neg_is_self two t)).
  Qed.

  Lemma circle_sin_odd : ∀ t : F, circle_sin (- t) == - circle_sin t.
  Proof. intros t. reflexivity. Qed.

  Lemma rotation_re : ∀ y z : F,
    1 + (y + z) == (1 + y) × (1 + z) + - (y × z).
  Proof. intros y z. ring. Qed.

  Lemma rotation_im : ∀ y z : F,
    y + z == (1 + y) × z + y × (1 + z).
  Proof.
    intros y z.
    calc (y + z)
      == ((y + z) + 0) { right_identity }
      == ((y + z) + (y × z + y × z)) { (double_vanishes two (y × z)) }
      == [regroup] ((1 + y) × z + y × (1 + z)).
    [regroup]: ring.
  Qed.

  Lemma circle_cexp_add : ∀ z w : C F,
    z × w == w × z → circle_cexp (z + w) == circle_cexp z × circle_cexp w.
  Proof.
    intros z w _.
    split.
    - exact (rotation_re (im z) (im w)).
    - exact (rotation_im (im z) (im w)).
  Qed.

  Lemma unit_sum_line : ∀ a b : F, a × a + b × b == 1 → b == 1 + a.
  Proof.
    intros a b Hab.
    assert (Hsquare : (1 + (a + b)) × (1 + (a + b)) == 0).
    { calc ((1 + (a + b)) × (1 + (a + b)))
        == (1 × 1 + (a + b) × (a + b)) { (square_of_sum two) }
        == (1 × 1 + (a × a + b × b)) { (square_of_sum two) }
        == ((1 : F) × 1 + 1) { Hab }
        == ((1 : F) + 1) { left_identity }
        == (0 : F) { two }. }
    apply (add_zero_swap two).
    calc ((1 + a) + b)
      == (1 + (a + b)) { associative }
      == (0 : F) { (square_zero (1 + (a + b)) Hsquare) }.
  Qed.

  Lemma line_unit_sum : ∀ a b : F, b == 1 + a → a × a + b × b == 1.
  Proof.
    intros a b Hb.
    assert (Hsquare : b × b == 1 + a × a).
    { calc (b × b)
        == ((1 + a) × (1 + a)) { Hb }
        == (1 × 1 + a × a) { (square_of_sum two) }
        == [collapse] (1 + a × a).
      [collapse]: ring. }
    calc (a × a + b × b)
      == (a × a + (1 + a × a)) { Hsquare }
      == [regroup] (1 + (a × a + a × a))
      == ((1 : F) + 0) { (double_vanishes two (a × a)) }
      == (1 : F) { right_identity }.
    [regroup]: ring.
  Qed.

  Lemma line_flip : ∀ a b : F, b == 1 + a → a == 1 + b.
  Proof.
    intros a b Hb.
    apply (unit_sum_line b a).
    calc (b × b + a × a)
      == (a × a + b × b) { commutative }
      == (1 : F) { (line_unit_sum a b Hb) }.
  Qed.

  Lemma euler_re : ∀ a b t : F,
    b == 1 + a → 1 + t × b == (1 + t) × 1 + t × a.
  Proof.
    intros a b t Hb.
    calc (1 + t × b)
      == (1 + t × (1 + a)) { Hb }
      == [expand] ((1 + t) × 1 + t × a).
    [expand]: ring.
  Qed.

  Lemma euler_im : ∀ a t : F, t × a == (1 + t) × 0 + t × a.
  Proof. intros a t. ring. Qed.

  Lemma root_of_minus_one_line : ∀ e : C F,
    e × e == - (1) → im e == 1 + re e.
  Proof.
    intros e [Hre _].
    apply (unit_sum_line (re e) (im e)).
    calc (re e × re e + im e × im e)
      == (re e × re e + - (im e × im e)) { (neg_is_self two (im e × im e)) }
      == (- (1) : F) { Hre }
      == (1 : F) { (neg_is_self two 1) }.
  Qed.

  Lemma circle_euler : ∀ (t : F) (e : C F),
    e × e == - (1) →
    circle_cexp (t *: e) == circle_cos t *: 1 + circle_sin t *: e.
  Proof.
    intros t e He.
    split.
    - exact (euler_re (re e) (im e) t (root_of_minus_one_line e He)).
    - exact (euler_im (im e) t).
  Qed.

  Lemma circle_sin_nontrivial : ∃ t : F, circle_sin t =/= 0.
  Proof. ∃ 1. exact nontrivial. Qed.

  Definition circular_characteristic_two
    : Circular F (C F)
        (Ax := circle_cexp) (Fc := circle_cos) (Fs := circle_sin) :=
    {| Circular.ring := C_Ring;
       Circular.scaling := C_ScaleLaws;
       
       Circular.mul_linear_l := GeometricAlgebra.mul_linear_l;
       Circular.mul_linear_r := GeometricAlgebra.mul_linear_r;
       Circular.cexp_proper := circle_cexp_proper;
       Circular.cos_proper := circle_cos_proper;
       Circular.sin_proper := circle_sin_proper;
       Circular.cexp_add := circle_cexp_add;
       Circular.pythagoras := circle_pythagoras;
       Circular.euler := circle_euler;
       Circular.cos_even := circle_cos_even;
       Circular.sin_odd := circle_sin_odd;
       Circular.sin_nontrivial := circle_sin_nontrivial |}.


  Definition on_circle (z : C F) : Prop := re z × re z + im z × im z == 1.

  Lemma circle_i_on_circle : on_circle circle_i.
  Proof. unfold on_circle, circle_i. simpl. ring. Qed.

  Lemma cexp_on_circle : ∀ z : C F, on_circle (circle_cexp z).
  Proof. intros z. exact (circle_pythagoras (im z)). Qed.

  Lemma cexp_fixes_circle : ∀ z : C F, on_circle z → circle_cexp z == z.
  Proof.
    intros z Hz.
    split; [ | reflexivity ].
    symmetry.
    exact (line_flip (re z) (im z) (unit_sum_line (re z) (im z) Hz)).
  Qed.

  Lemma cexp_reaches_circle : ∀ z : C F,
    on_circle z → circle_cexp (mkC 0 (im z)) == z.
  Proof. intros z Hz. exact (cexp_fixes_circle z Hz). Qed.

  Lemma cexp_angle_add : ∀ s t : F,
    circle_cexp (mkC 0 (s + t))
    == circle_cexp (mkC 0 s) × circle_cexp (mkC 0 t).
  Proof.
    intros s t.
    exact (circle_cexp_add (mkC 0 s) (mkC 0 t)
             (commutative (mkC 0 s) (mkC 0 t))).
  Qed.

  Lemma cexp_angle_injective : ∀ s t : F,
    circle_cexp (mkC 0 s) == circle_cexp (mkC 0 t) → s == t.
  Proof. intros s t [_ Him]. exact Him. Qed.

  Lemma euler_at_circle_i : ∀ t : F,
    circle_cexp (t *: circle_i) == mkC (circle_cos t) (circle_sin t).
  Proof.
    intros t.
    calc (circle_cexp (t *: circle_i))
      == ((circle_cos t *: (1 : C F) + circle_sin t *: circle_i) : C F)
         { (circle_euler t circle_i circle_i_square) }
      == (mkC (circle_cos t) (circle_sin t) : C F)
         { scale_one_scale_i }.
  Qed.

End CircleCharacteristicTwo.

Instance Bool_Sin : HasSin bool := circle_sin.

Instance Bool_Circular : Circular bool (C bool) :=
  circular_characteristic_two bool_characteristic_two.

Example bool_conjugate_unique : ∀ f : C bool → C bool,

  f circle_i == - circle_i →
  ∀ z : C bool, f z == conjugate z.
Proof.
  exact (plane_conjugate_unique circle_i conjugate
           C_conjugate_scale conjugate_circle_i).
Qed.
