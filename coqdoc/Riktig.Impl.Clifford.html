Section BladeSum.
  Context {F : Type} `{Field F}.
  Add Ring blade_sum_ring : (ring_theory_of_Ring (R := F) commutative).

  Fixpoint bsum {n : nat} : (Blade n → F) → F :=
    match n with
    | 0 ⇒ fun f ⇒ f bnil
    | S m ⇒ fun f ⇒
        bsum (fun A ⇒ f (bcons false A)) + bsum (fun A ⇒ f (bcons true A))
    end.

  Lemma bsum_S : ∀ n (f : Blade (S n) → F),
    bsum f = bsum (fun A ⇒ f (bcons false A)) + bsum (fun A ⇒ f (bcons true A)).
  Proof. reflexivity. Qed.

  Lemma bsum_ext : ∀ n (f g : Blade n → F),
    (∀ A, f A == g A) → bsum f == bsum g.
  Proof.
    induction n as [| m IH]; intros f g Hfg.
    - exact (Hfg _).
    - rewrite !bsum_S.
      rewrite (IH (fun A ⇒ f (bcons false A)) (fun A ⇒ g (bcons false A)))
        by (intro A; apply Hfg).
      rewrite (IH (fun A ⇒ f (bcons true A)) (fun A ⇒ g (bcons true A)))
        by (intro A; apply Hfg).
      reflexivity.
  Qed.

  Lemma bsum_zero : ∀ n, bsum (fun _ : Blade n ⇒ (0 : F)) == 0.
  Proof.
    induction n as [| m IH].
    - reflexivity.
    - rewrite bsum_S, !IH. ring.
  Qed.

  Lemma bsum_add : ∀ n (f g : Blade n → F),
    bsum (fun A ⇒ f A + g A) == bsum f + bsum g.
  Proof.
    induction n as [| m IH]; intros f g.
    - reflexivity.
    - rewrite !bsum_S.
      rewrite (IH (fun A ⇒ f (bcons false A)) (fun A ⇒ g (bcons false A))).
      rewrite (IH (fun A ⇒ f (bcons true A)) (fun A ⇒ g (bcons true A))).
      ring.
  Qed.

  Lemma bsum_mul_l : ∀ n (c : F) (f : Blade n → F),
    bsum (fun A ⇒ c × f A) == c × bsum f.
  Proof.
    induction n as [| m IH]; intros c f.
    - reflexivity.
    - rewrite !bsum_S.
      rewrite (IH c (fun A ⇒ f (bcons false A))).
      rewrite (IH c (fun A ⇒ f (bcons true A))).
      ring.
  Qed.

  Lemma bsum_neg : ∀ n (f : Blade n → F),
    bsum (fun A ⇒ - f A) == - bsum f.
  Proof.
    induction n as [| m IH]; intros f.
    - reflexivity.
    - rewrite !bsum_S.
      rewrite (IH (fun A ⇒ f (bcons false A))).
      rewrite (IH (fun A ⇒ f (bcons true A))).
      ring.
  Qed.

  Lemma bsum_mul_r : ∀ n (c : F) (f : Blade n → F),
    bsum (fun A ⇒ f A × c) == bsum f × c.
  Proof.
    intros n c f.
    rewrite (bsum_ext n (fun A ⇒ f A × c) (fun A ⇒ c × f A))
      by (intro A; ring).
    rewrite bsum_mul_l. ring.
  Qed.

  Lemma bsum_swap : ∀ m n (f : Blade m → Blade n → F),
    bsum (fun A ⇒ bsum (fun B ⇒ f A B)) == bsum (fun B ⇒ bsum (fun A ⇒ f A B)).
  Proof.
    induction m as [| k IH]; intros n f.
    - reflexivity.
    - rewrite bsum_S.
      rewrite (IH n (fun A B ⇒ f (bcons false A) B)).
      rewrite (IH n (fun A B ⇒ f (bcons true A) B)).
      rewrite <- bsum_add.
      apply bsum_ext. intros B.
      rewrite (bsum_S k (fun A ⇒ f A B)). reflexivity.
  Qed.

  Lemma bxor_cons_split : ∀ n (D : Blade (S n)) (b : bool) (A : Blade n),
    bxor D (bcons b A) = bcons (xorb (bhead D) b) (bxor (btail D) A).
  Proof.
    intros n D b A.
    rewrite (blade_S n D). reflexivity.
  Qed.

  Lemma bsum_reindex : ∀ n (D : Blade n) (f : Blade n → F),
    bsum (fun A ⇒ f (bxor D A)) == bsum f.
  Proof.
    induction n as [| m IH]; intros D f.
    - simpl. rewrite (blade_0 (bxor D bnil)). reflexivity.
    - rewrite !bsum_S.
      rewrite (bsum_ext m (fun A ⇒ f (bxor D (bcons false A)))
                 (fun A ⇒ f (bcons (bhead D) (bxor (btail D) A))))
        by (intro A; rewrite bxor_cons_split, xorb_false_r; reflexivity).
      rewrite (bsum_ext m (fun A ⇒ f (bxor D (bcons true A)))
                 (fun A ⇒ f (bcons (negb (bhead D)) (bxor (btail D) A))))
        by (intro A; rewrite bxor_cons_split, xorb_true_r; reflexivity).
      destruct (bhead D); simpl negb;
        rewrite (IH (btail D) (fun A ⇒ f (bcons false A))),
                (IH (btail D) (fun A ⇒ f (bcons true A)));
        ring.
  Qed.

End BladeSum.

Definition Cl (F : Type) (n : nat) : Type := Blade n → F.

Typeclasses Transparent Cl.

Definition Signature (F : Type) (n : nat) : Type := Vector.t F n.

Section CliffordProduct.
  Context {F : Type} `{Field F}.
  Add Ring clifford_ring : (ring_theory_of_Ring (R := F) commutative).

  Fixpoint sigprod {n : nat} : Signature F n → Blade n → F :=
    match n with
    | 0 ⇒ fun _ _ ⇒ 1
    | S m ⇒ fun s A ⇒
        (if bhead A then Vector.hd s else 1) × sigprod (Vector.tl s) (btail A)
    end.

  Lemma pair_mul : ∀ a b p q c d r t : F,
    a × b == c × d → p × q == r × t → (a × p) × (b × q) == (c × r) × (d × t).
  Proof.
    intros a b p q c d r t Hab Hpq.
    transitivity ((a × b) × (p × q)); [ ring |].
    rewrite Hab, Hpq. ring.
  Qed.

  Lemma sigprod_S : ∀ m (s : Signature F (S m)) (A : Blade (S m)),
    sigprod s A = (if bhead A then Vector.hd s else 1) × sigprod (Vector.tl s) (btail A).
  Proof. reflexivity. Qed.

  Definition parity_sign (b : bool) : F := if b then - (1) else 1.

  Lemma parity_sign_true : parity_sign true = - (1).
  Proof. reflexivity. Qed.

  Lemma parity_sign_false : parity_sign false = 1.
  Proof. reflexivity. Qed.

  Definition bsign {n : nat} (A B : Blade n) : F := parity_sign (bpar A B).

  Lemma parity_sign_xor : ∀ p q : bool,
    parity_sign (xorb p q) == parity_sign p × parity_sign q.
  Proof. intros [] []; unfold parity_sign; simpl; ring. Qed.

  Definition cl_mul {n : nat} (s : Signature F n) (x y : Cl F n) : Cl F n :=
    fun D ⇒ bsum (fun A ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × y (bxor A D))).

  Fixpoint bis_nil {n : nat} : Blade n → bool :=
    match n with
    | 0 ⇒ fun _ ⇒ true
    | S m ⇒ fun A ⇒ andb (negb (bhead A)) (bis_nil (btail A))
    end.

  Definition cl_one {n : nat} : Cl F n :=
    fun A ⇒ if bis_nil A then 1 else 0.

  Lemma cl_one_cons_true : ∀ {m} (A : Blade m),
    cl_one (bcons true A) = 0.
  Proof. reflexivity. Qed.

  Lemma cl_one_cons_false : ∀ {m} (A : Blade m),
    cl_one (bcons false A) = cl_one A.
  Proof. reflexivity. Qed.

  Lemma cl_scale_at : ∀ n (c : F) (x : Cl F n) (D : Blade n),
    (c *: x) D == c × x D.
  Proof. reflexivity. Qed.

  Lemma scale_one_at : ∀ n (c : F) (D : Blade n),
    (c *: cl_one) D == c × cl_one D.
  Proof. intros n c D. exact (cl_scale_at n c cl_one D). Qed.

  Lemma sigprod_bempty : ∀ n (s : Signature F n), sigprod s bempty == 1.
  Proof.
    induction n as [| m IH]; intros s.
    - reflexivity.
    - rewrite (sigprod_S m s bempty), bhead_bempty, btail_bempty,
              (IH (Vector.tl s)).
      simpl. ring.
  Qed.

  Lemma bsign_bempty_l : ∀ n (A : Blade n), bsign bempty A == 1.
  Proof. intros n A. unfold bsign. rewrite bpar_bempty_l. reflexivity. Qed.

  Lemma bsign_bempty_r : ∀ n (A : Blade n), bsign A bempty == 1.
  Proof. intros n A. unfold bsign. rewrite bpar_bempty_r. reflexivity. Qed.

  Lemma bis_nil_cons : ∀ n (b : bool) (A : Blade n),
    bis_nil (bcons b A) = andb (negb b) (bis_nil A).
  Proof. reflexivity. Qed.

  Lemma bis_nil_true : ∀ n (A : Blade n),
    bis_nil A = true → A = bempty.
  Proof.
    induction n as [| m IH]; intros A Hnil.
    - apply blade_0.
    - rewrite (blade_S m A) in Hnil |- ×.
      rewrite bis_nil_cons in Hnil.
      apply andb_true_iff in Hnil. destruct Hnil as [Hh Ht].
      rewrite (IH (btail A) Ht).
      destruct (bhead A); [ discriminate | reflexivity ].
  Qed.

  Lemma sigprod_cons : ∀ m (s : Signature F (S m)) (a : bool) (A : Blade m),
    sigprod s (bcons a A) = (if a then Vector.hd s else 1) × sigprod (Vector.tl s) A.
  Proof. reflexivity. Qed.

  Lemma bsum_delta : ∀ n (f : Blade n → F),
    bsum (fun A ⇒ (if bis_nil A then 1 else 0) × f A) == f bempty.
  Proof.
    induction n as [| m IH]; intros f.
    - simpl. ring.
    - rewrite bsum_S.
      rewrite (bsum_ext m
                 (fun A ⇒ (if bis_nil (bcons false A) then 1 else 0)
                           × f (bcons false A))
                 (fun A ⇒ (if bis_nil A then 1 else 0) × f (bcons false A)))
        by (intro A; rewrite bis_nil_cons; destruct (bis_nil A); simpl; ring).
      rewrite (IH (fun A ⇒ f (bcons false A))).
      rewrite (bsum_ext m
                 (fun A ⇒ (if bis_nil (bcons true A) then 1 else 0)
                           × f (bcons true A))
                 (fun _ ⇒ 0))
        by (intro A; rewrite bis_nil_cons; simpl; ring).
      rewrite bsum_zero.
      transitivity (f bempty + 0); [ reflexivity | ring ].
  Qed.

  Lemma bsum_delta_at : ∀ n (B : Blade n) (f : Blade n → F),
    bsum (fun A ⇒ (if bis_nil (bxor B A) then 1 else 0) × f A) == f B.
  Proof.
    intros n B f.
    rewrite <- (bsum_reindex n B (fun A ⇒
      (if bis_nil (bxor B A) then 1 else 0) × f A)).
    transitivity (bsum (fun A ⇒ (if bis_nil A then 1 else 0) × f (bxor B A))).
    { apply bsum_ext. intros A.
      transitivity
        ((if bis_nil (bxor B (bxor B A)) then 1 else 0) × f (bxor B A));
        [ reflexivity |].
      rewrite (bxor_involutive n B A). reflexivity. }
    transitivity (f (bxor B bempty)).
    { apply (bsum_delta n (fun A ⇒ f (bxor B A))). }
    rewrite (bxor_bempty_r n B). reflexivity.
  Qed.

  Lemma bpar_shuffle : ∀ n (B C D : Blade n),
    xorb (bpar (bxor B C) (bxor (bxor B C) D)) (bpar B C)
    = xorb (bpar B (bxor B D)) (bpar C (bxor C (bxor B D))).
  Proof.
    intros n B C D.
    rewrite !bpar_bxor_l, !bpar_bxor_r.
    destruct (bpar B B), (bpar B C), (bpar B D),
             (bpar C B), (bpar C C), (bpar C D);
      reflexivity.
  Qed.

  Lemma sigprod_shuffle : ∀ n (s : Signature F n) (B C D : Blade n),
    sigprod s (bdiff (bxor B C) D) × sigprod s (band B C)
    == sigprod s (bdiff B D) × sigprod s (bdiff C (bxor B D)).
  Proof.
    induction n as [| m IH]; intros s B C D.
    - reflexivity.
    - rewrite !sigprod_S, !bhead_bdiff, !bhead_band, !bhead_bxor,
              !btail_bdiff, !btail_band, !btail_bxor.
      apply pair_mul.
      + destruct (bhead B), (bhead C), (bhead D); simpl; ring.
      + exact (IH (Vector.tl s) (btail B) (btail C) (btail D)).
  Qed.

  Lemma cl_coeff : ∀ n (s : Signature F n) (B C D : Blade n),
    (bsign (bxor B C) (bxor (bxor B C) D) × sigprod s (bdiff (bxor B C) D))
    × (bsign B C × sigprod s (band B C))
    == (bsign B (bxor B D) × sigprod s (bdiff B D))
    × (bsign C (bxor C (bxor B D)) × sigprod s (bdiff C (bxor B D))).
  Proof.
    intros n s B C D.
    apply pair_mul.
    - unfold bsign.
      rewrite <- !parity_sign_xor, (bpar_shuffle n B C D). reflexivity.
    - exact (sigprod_shuffle n s B C D).
  Qed.

  Lemma cl_mul_dist_l : ∀ n (s : Signature F n) (x y z : Cl F n),
    cl_mul s x (y + z) == cl_mul s x y + cl_mul s x z.
  Proof.
    intros n s x y z D. unfold cl_mul.
    transitivity
      (bsum (fun A ⇒ bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × y (bxor A D)))
       + bsum (fun A ⇒ bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × z (bxor A D)))).
    - rewrite <- bsum_add. apply bsum_ext. intros A.
      rewrite (pw_add_at y z (bxor A D)). ring.
    - reflexivity.
  Qed.

  Lemma cl_mul_dist_r : ∀ n (s : Signature F n) (x y z : Cl F n),
    cl_mul s (x + y) z == cl_mul s x z + cl_mul s y z.
  Proof.
    intros n s x y z D. unfold cl_mul.
    transitivity
      (bsum (fun A ⇒ bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × z (bxor A D)))
       + bsum (fun A ⇒ bsign A (bxor A D) × sigprod s (bdiff A D) × (y A × z (bxor A D)))).
    - rewrite <- bsum_add. apply bsum_ext. intros A.
      rewrite (pw_add_at x y A). ring.
    - reflexivity.
  Qed.

  Lemma cl_one_l : ∀ n (s : Signature F n) (x : Cl F n),
    cl_mul s cl_one x == x.
  Proof.
    intros n s x D. unfold cl_mul.
    transitivity (bsum (fun A ⇒ (if bis_nil A then 1 else 0)
      × (bsign A (bxor A D) × sigprod s (bdiff A D) × x (bxor A D)))).
    { apply bsum_ext. intros A. unfold cl_one.
      destruct (bis_nil A); ring. }
    transitivity
      (bsign bempty (bxor bempty D) × sigprod s (bdiff bempty D)
       × x (bxor bempty D)).
    { apply (bsum_delta n (fun A ⇒
        bsign A (bxor A D) × sigprod s (bdiff A D) × x (bxor A D))). }
    rewrite (bxor_bempty_l n D), (bdiff_bempty_l n D).
    rewrite (bsign_bempty_l n D), (sigprod_bempty n s).
    ring.
  Qed.

  Lemma cl_one_r : ∀ n (s : Signature F n) (x : Cl F n),
    cl_mul s x cl_one == x.
  Proof.
    intros n s x D. unfold cl_mul.
    rewrite <- (bsum_reindex n D (fun A ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × cl_one (bxor A D)))).
    transitivity (bsum (fun A ⇒ (if bis_nil A then 1 else 0)
      × (bsign (bxor D A) A × sigprod s (bdiff (bxor D A) D) × x (bxor D A)))).
    { apply bsum_ext. intros A.
      transitivity
        (bsign (bxor D A) (bxor (bxor D A) D) × sigprod s (bdiff (bxor D A) D)
         × (x (bxor D A) × cl_one (bxor (bxor D A) D)));
        [ reflexivity |].
      rewrite (bxor_comm n (bxor D A) D), (bxor_involutive n D A).
      unfold cl_one. destruct (bis_nil A); ring. }
    transitivity
      (bsign (bxor D bempty) bempty × sigprod s (bdiff (bxor D bempty) D)
       × x (bxor D bempty)).
    { apply (bsum_delta n (fun A ⇒
        bsign (bxor D A) A × sigprod s (bdiff (bxor D A) D) × x (bxor D A))). }
    rewrite (bxor_bempty_r n D), (bdiff_self n D).
    rewrite (bsign_bempty_r n D), (sigprod_bempty n s).
    ring.
  Qed.

  Lemma cl_mul_scale_l : ∀ n (s : Signature F n) (a : F) (x y : Cl F n),
    cl_mul s (a *: x) y == a *: cl_mul s x y.
  Proof.
    intros n s a x y D. unfold cl_mul.
    transitivity (a × bsum (fun A ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × y (bxor A D)))).
    2: reflexivity.
    rewrite <- (bsum_mul_l n a (fun A ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × y (bxor A D)))).
    apply bsum_ext. intros A.
    rewrite (pw_scale_at a x A), (scale_is_mult a (x A)). ring.
  Qed.

  Lemma cl_mul_scale_r : ∀ n (s : Signature F n) (a : F) (x y : Cl F n),
    cl_mul s x (a *: y) == a *: cl_mul s x y.
  Proof.
    intros n s a x y D. unfold cl_mul.
    transitivity (a × bsum (fun A ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × y (bxor A D)))).
    2: reflexivity.
    rewrite <- (bsum_mul_l n a (fun A ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D) × (x A × y (bxor A D)))).
    apply bsum_ext. intros A.
    rewrite (pw_scale_at a y (bxor A D)), (scale_is_mult a (y (bxor A D))). ring.
  Qed.

  Fixpoint ga_vec {n : nat} : (Fin.t n → F) → Cl F n :=
    match n with
    | 0 ⇒ fun _ _ ⇒ 0
    | S m ⇒ fun v D ⇒
        if bhead D
        then if bis_nil (btail D) then v Fin.F1 else 0
        else ga_vec (fun i ⇒ v (Fin.FS i)) (btail D)
    end.

  Lemma ga_vec_cons_true : ∀ m (v : Fin.t (S m) → F) (A : Blade m),
    ga_vec v (bcons true A) = if bis_nil A then v Fin.F1 else 0.
  Proof. reflexivity. Qed.

  Lemma ga_vec_cons_false : ∀ m (v : Fin.t (S m) → F) (A : Blade m),
    ga_vec v (bcons false A) = ga_vec (fun i ⇒ v (Fin.FS i)) A.
  Proof. reflexivity. Qed.

  Lemma ga_vec_parity : ∀ n (v : Fin.t n → F) (A : Blade n),
    bparity A = false → ga_vec v A == 0.
  Proof.
    induction n as [| m IH]; intros v A Hpar.
    - reflexivity.
    - rewrite bparity_S in Hpar.
      rewrite (blade_S m A).
      destruct (bhead A) eqn:Hh.
      + rewrite ga_vec_cons_true.
        destruct (bis_nil (btail A)) eqn:Hnil; [| reflexivity].
        rewrite (bis_nil_true m (btail A) Hnil), bparity_bempty in Hpar.
        discriminate.
      + rewrite ga_vec_cons_false. exact (IH _ (btail A) Hpar).
  Qed.

  Fixpoint qform {n : nat} : Signature F n → (Fin.t n → F) → F :=
    match n with
    | 0 ⇒ fun _ _ ⇒ 0
    | S m ⇒ fun s v ⇒
        Vector.hd s × (v Fin.F1 × v Fin.F1)
        + qform (Vector.tl s) (fun i ⇒ v (Fin.FS i))
    end.

  Lemma qform_S : ∀ m (s : Signature F (S m)) (v : Fin.t (S m) → F),
    qform s v = Vector.hd s × (v Fin.F1 × v Fin.F1)
                + qform (Vector.tl s) (fun i ⇒ v (Fin.FS i)).
  Proof. reflexivity. Qed.

  Lemma qform_nil : ∀ (s : Signature F 0) (v : Fin.t 0 → F),
    qform s v = 0.
  Proof. reflexivity. Qed.

  Lemma ga_vec_nil : ∀ (v : Fin.t 0 → F) (D : Blade 0),
    ga_vec v D = 0.
  Proof. reflexivity. Qed.

  Lemma tail_add : ∀ m (u w : Fin.t (S m) → F),
    (fun i ⇒ (u + w) (Fin.FS i))
    = (fun i ⇒ u (Fin.FS i)) + (fun i ⇒ w (Fin.FS i)).
  Proof. reflexivity. Qed.

  Lemma tail_scale : ∀ m (a : F) (u : Fin.t (S m) → F),
    (fun i ⇒ (a *: u) (Fin.FS i)) = a *: (fun i ⇒ u (Fin.FS i)).
  Proof. reflexivity. Qed.

End CliffordProduct.

Section CliffordAssoc.
  Context {F : Type} `{Field F}.
  Add Ring clifford_assoc_ring : (ring_theory_of_Ring (R := F) commutative).

  Lemma cl_mul_assoc : ∀ n (s : Signature F n) (x y z : Cl F n),
    cl_mul s (cl_mul s x y) z == cl_mul s x (cl_mul s y z).
  Proof.
    intros n s x y z D. unfold cl_mul.
    transitivity (bsum (fun A ⇒ bsum (fun B ⇒
      bsign A (bxor A D) × sigprod s (bdiff A D)
      × (bsign B (bxor B A) × sigprod s (bdiff B A) × (x B × y (bxor B A)))
      × z (bxor A D)))).
    { apply bsum_ext. intros A.
      rewrite <- bsum_mul_r, <- bsum_mul_l. apply bsum_ext. intros B. ring. }
    
    transitivity (bsum (fun B ⇒ bsum (fun C ⇒
      bsign B (bxor B D) × sigprod s (bdiff B D)

Section CliffordQuadraticForm.
  Context {F : Type} `{Field F}.
  Add Ring clifford_qf_ring : (ring_theory_of_Ring (R := F) commutative).

  Fixpoint pform {n : nat}
    : Signature F n → (Fin.t n → F) → (Fin.t n → F) → F :=
    match n with
    | 0 ⇒ fun _ _ _ ⇒ 0
    | S m ⇒ fun s u w ⇒
        Vector.hd s × (u Fin.F1 × w Fin.F1 + u Fin.F1 × w Fin.F1)
        + pform (Vector.tl s) (fun i ⇒ u (Fin.FS i)) (fun i ⇒ w (Fin.FS i))
    end.

  Lemma pform_S : ∀ m (s : Signature F (S m)) (u w : Fin.t (S m) → F),
    pform s u w
    = Vector.hd s × (u Fin.F1 × w Fin.F1 + u Fin.F1 × w Fin.F1)
      + pform (Vector.tl s) (fun i ⇒ u (Fin.FS i)) (fun i ⇒ w (Fin.FS i)).
  Proof. reflexivity. Qed.

  Lemma pform_nil : ∀ (s : Signature F 0) (u w : Fin.t 0 → F),
    pform s u w = 0.
  Proof. reflexivity. Qed.

  Lemma qform_polar : ∀ n (s : Signature F n) (u w : Fin.t n → F),
    polar (qform s) u w == pform s u w.
  Proof.
    induction n as [| m IH]; intros s u w; unfold polar.
    - rewrite (qform_nil s (u + w)), (qform_nil s u), (qform_nil s w),
              (pform_nil s u w).
      ring.
    - rewrite (qform_S m s (u + w)), (qform_S m s u), (qform_S m s w).
      rewrite (pw_add_at u w Fin.F1), (tail_add m u w).
      rewrite (pform_S m s u w).
      pose proof (IH (Vector.tl s)
                     (fun i ⇒ u (Fin.FS i)) (fun i ⇒ w (Fin.FS i))) as HIH.
      unfold polar in HIH.
      rewrite <- HIH.
      ring.
  Qed.

  Lemma pform_proper_l : ∀ n (s : Signature F n) (u u' w : Fin.t n → F),
    (∀ i, u i == u' i) → pform s u w == pform s u' w.
  Proof.
    induction n as [| m IH]; intros s u u' w Hu.
    - reflexivity.
    - rewrite (pform_S m s u w), (pform_S m s u' w).
      rewrite (Hu Fin.F1).
      rewrite (IH (Vector.tl s) (fun i ⇒ u (Fin.FS i))
                  (fun i ⇒ u' (Fin.FS i)) (fun i ⇒ w (Fin.FS i))
                  (fun i ⇒ Hu (Fin.FS i))).
      reflexivity.
  Qed.

  Lemma pform_add_l : ∀ n (s : Signature F n) (u u' w : Fin.t n → F),
    pform s (u + u') w == pform s u w + pform s u' w.
  Proof.
    induction n as [| m IH]; intros s u u' w.
    - rewrite (pform_nil s (u + u') w), (pform_nil s u w),
              (pform_nil s u' w).
      ring.
    - rewrite (pform_S m s (u + u') w), (pform_S m s u w), (pform_S m s u' w).
      rewrite (pw_add_at u u' Fin.F1), (tail_add m u u').
      rewrite (IH (Vector.tl s) (fun i ⇒ u (Fin.FS i))
                  (fun i ⇒ u' (Fin.FS i)) (fun i ⇒ w (Fin.FS i))).
      ring.
  Qed.

  Lemma pform_scale_l : ∀ n (s : Signature F n) (a : F) (u w : Fin.t n → F),
    pform s (a *: u) w == a × pform s u w.
  Proof.
    induction n as [| m IH]; intros s a u w.
    - rewrite (pform_nil s (a *: u) w), (pform_nil s u w).
      ring.
    - rewrite (pform_S m s (a *: u) w), (pform_S m s u w).
      rewrite (pw_scale_at a u Fin.F1), (scale_is_mult a (u Fin.F1)),
              (tail_scale m a u).
      rewrite (IH (Vector.tl s) a (fun i ⇒ u (Fin.FS i))
                  (fun i ⇒ w (Fin.FS i))).
      ring.
  Qed.

  Lemma qform_proper : ∀ n (s : Signature F n) (u u' : Fin.t n → F),
    (∀ i, u i == u' i) → qform s u == qform s u'.
  Proof.
    induction n as [| m IH]; intros s u u' Hu.
    - reflexivity.
    - rewrite (qform_S m s u), (qform_S m s u').
      rewrite (Hu Fin.F1).
      rewrite (IH (Vector.tl s) (fun i ⇒ u (Fin.FS i))
                  (fun i ⇒ u' (Fin.FS i)) (fun i ⇒ Hu (Fin.FS i))).
      reflexivity.
  Qed.

  Lemma qform_scale : ∀ n (s : Signature F n) (a : F) (u : Fin.t n → F),
    qform s (a *: u) == a × a × qform s u.
  Proof.
    induction n as [| m IH]; intros s a u.
    - rewrite (qform_nil s (a *: u)), (qform_nil s u).
      ring.
    - rewrite (qform_S m s (a *: u)), (qform_S m s u).
      rewrite (pw_scale_at a u Fin.F1), (scale_is_mult a (u Fin.F1)),
              (tail_scale m a u).
      rewrite (IH (Vector.tl s) a (fun i ⇒ u (Fin.FS i))).
      ring.
  Qed.

  Lemma ga_vec_ext : ∀ n (u u' : Fin.t n → F),
    (∀ i, u i == u' i) → ∀ D, ga_vec u D == ga_vec u' D.
  Proof.
    induction n as [| m IH]; intros u u' Hu D.
    - reflexivity.
    - rewrite (blade_S m D). destruct (bhead D).
      + rewrite (ga_vec_cons_true m u (btail D)),
                (ga_vec_cons_true m u' (btail D)).
        destruct (bis_nil (btail D)); [ exact (Hu Fin.F1) | reflexivity ].
      + rewrite (ga_vec_cons_false m u (btail D)),
                (ga_vec_cons_false m u' (btail D)).
        exact (IH (fun i ⇒ u (Fin.FS i)) (fun i ⇒ u' (Fin.FS i))
                  (fun i ⇒ Hu (Fin.FS i)) (btail D)).
  Qed.

  Lemma ga_vec_add : ∀ n (u w : Fin.t n → F) (D : Blade n),
    ga_vec (u + w) D == ga_vec u D + ga_vec w D.
  Proof.
    induction n as [| m IH]; intros u w D.
    - rewrite (ga_vec_nil (u + w) D), (ga_vec_nil u D), (ga_vec_nil w D).
      ring.
    - rewrite (blade_S m D). destruct (bhead D).
      + rewrite (ga_vec_cons_true m (u + w) (btail D)),
                (ga_vec_cons_true m u (btail D)),
                (ga_vec_cons_true m w (btail D)).
        destruct (bis_nil (btail D)); [ reflexivity | ring ].
      + rewrite (ga_vec_cons_false m (u + w) (btail D)),
                (ga_vec_cons_false m u (btail D)),
                (ga_vec_cons_false m w (btail D)).
        rewrite (tail_add m u w).
        exact (IH (fun i ⇒ u (Fin.FS i)) (fun i ⇒ w (Fin.FS i)) (btail D)).
  Qed.

  Lemma polar_qform_proper : ∀ n (s : Signature F n) (w u u' : Fin.t n → F),
    (∀ i, u i == u' i) → polar (qform s) u w == polar (qform s) u' w.
  Proof.
    intros n s w u u' Hu.
    rewrite (qform_polar n s u w), (qform_polar n s u' w).
    exact (pform_proper_l n s u u' w Hu).
  Qed.

  Lemma polar_qform_add : ∀ n (s : Signature F n) (w u u' : Fin.t n → F),
    polar (qform s) (u + u') w == polar (qform s) u w + polar (qform s) u' w.
  Proof.
    intros n s w u u'.
    rewrite (qform_polar n s (u + u') w), (qform_polar n s u w),
            (qform_polar n s u' w).
    exact (pform_add_l n s u u' w).
  Qed.

  Lemma polar_qform_scale : ∀ n (s : Signature F n) (w : Fin.t n → F)
      (a : F) (u : Fin.t n → F),
    polar (qform s) (a *: u) w == a × polar (qform s) u w.
  Proof.
    intros n s w a u.
    rewrite (qform_polar n s (a *: u) w), (qform_polar n s u w).
    exact (pform_scale_l n s a u w).
  Qed.

  Lemma ga_vec_scale : ∀ n (a : F) (u : Fin.t n → F) (D : Blade n),
    ga_vec (a *: u) D == a × ga_vec u D.
  Proof.
    induction n as [| m IH]; intros a u D.
    - rewrite (ga_vec_nil (a *: u) D), (ga_vec_nil u D).
      ring.
    - rewrite (blade_S m D). destruct (bhead D).
      + rewrite (ga_vec_cons_true m (a *: u) (btail D)),
                (ga_vec_cons_true m u (btail D)).
        destruct (bis_nil (btail D)); [ reflexivity | ring ].
      + rewrite (ga_vec_cons_false m (a *: u) (btail D)),
                (ga_vec_cons_false m u (btail D)).
        rewrite (tail_scale m a u).
        exact (IH a (fun i ⇒ u (Fin.FS i)) (btail D)).
  Qed.

End CliffordQuadraticForm.

Section CliffordRing.
  Context {F : Type} `{Field F} {n : nat} (s : Signature F n).

  #[local] Instance Cl_One : One (Cl F n) := cl_one.
  #[local] Instance Cl_Mult : HasMult (Cl F n) := cl_mul s.

  #[local] Instance cl_mul_proper :
    Proper (equ ==> equ ==> equ) (@mult (Cl F n) _).
  Proof.
    intros x x' Hx y y' Hy D.
    unfold mult, Cl_Mult, cl_mul. apply bsum_ext. intros A.
    rewrite (Hx A), (Hy (bxor A D)). reflexivity.
  Qed.

  #[local] Instance cl_mul_associative :
    Associative (Cl F n) (Cl F n) mult mult.
  Proof. exact (cl_mul_assoc n s). Qed.

  #[local] Instance cl_mul_left_identity :
    LeftIdentity (Cl F n) (Cl F n) mult 1.
  Proof. exact (cl_one_l n s). Qed.

  #[local] Instance cl_mul_right_identity :
    RightIdentity (Cl F n) (Cl F n) mult 1.
  Proof. exact (cl_one_r n s). Qed.

  #[local] Instance cl_left_distributive :
    LeftDistributive (Cl F n) (Cl F n) (Cl F n) mult.
  Proof. exact (cl_mul_dist_l n s). Qed.

  #[local] Instance cl_right_distributive :
    RightDistributive (Cl F n) (Cl F n) (Cl F n) mult.
  Proof. exact (cl_mul_dist_r n s). Qed.

  #[local] Instance Cl_MulMagma : Magma (Cl F n) mult := {}.
  #[local] Instance Cl_MulSemigroup : Semigroup (Cl F n) mult := {}.
  #[local] Instance Cl_MulMonoid : @Monoid (Cl F n) _ Cl_One Cl_Mult := {}.
  #[local] Instance Cl_Ring : Ring (Cl F n) := {}.

  #[local] Instance Cl_QuadProper : Proper (equ ==> equ) (qform s).
  Proof. intros u u' Hu. exact (qform_proper n s u u' Hu). Qed.

  #[local] Instance Cl_QuadScale :
    Homogeneous F (Fin.t n → F) F mult (fun a ⇒ a × a) (qform s).
  Proof. intros a u. exact (qform_scale n s a u). Qed.

  #[local] Instance Cl_PolarLinear (w : Fin.t n → F) :
    Linear F (Fin.t n → F) F (fun u ⇒ polar (qform s) u w).
  Proof.
    constructor.
    - intros u u' Hu. exact (polar_qform_proper n s w u u' Hu).
    - intros u u'. exact (polar_qform_add n s w u u').
    - intros a u. exact (polar_qform_scale n s w a u).
  Qed.

  #[local] Instance Cl_QuadraticForm :
    QuadraticForm F (Fin.t n → F) (qform s) := {}.

  #[local] Instance Cl_VecLinear : Linear F (Fin.t n → F) (Cl F n) ga_vec.
  Proof.
    constructor.
    - intros u u' Hu D. exact (ga_vec_ext n u u' Hu D).
    - intros u w D. exact (ga_vec_add n u w D).
    - intros a u D. exact (ga_vec_scale n a u D).
  Qed.

  Definition Cl_GA :
    GeometricAlgebra F (Fin.t n → F) (Cl F n) (Q := qform s) :=
    {| GeometricAlgebra.ring := Cl_Ring;
       GeometricAlgebra.scaling := PW_ScaleLaws;
       GeometricAlgebra.mul_linear_l := fun y ⇒
         {| Linear.proper :=
              fun x x' Hx ⇒ cl_mul_proper x x' Hx y y (reflexivity y);
            Linear.is_additive := fun x x' ⇒ cl_mul_dist_r n s x x' y;
            Linear.is_homogeneous := fun a x ⇒ cl_mul_scale_l n s a x y |};
       GeometricAlgebra.mul_linear_r := fun x ⇒
         {| Linear.proper :=
              fun y y' Hy ⇒ cl_mul_proper x x (reflexivity x) y y' Hy;
            Linear.is_additive := cl_mul_dist_l n s x;
            Linear.is_homogeneous := fun a y ⇒ cl_mul_scale_r n s a x y |};
       GeometricAlgebra.ga_vec := ga_vec;
       GeometricAlgebra.vec_linear := Cl_VecLinear;
       GeometricAlgebra.contraction := gv_square n s |}.

End CliffordRing.

Definition coord {F : Type} {zF : Zero F} {oF : One F}
    {n : nat} (i : Fin.t n) : Fin.t n → F :=
  fun j ⇒ if dec (i = j) then 1 else 0.

Section TwoGenerators.
  Context {F : Type} `{FF : Field F}.
  Add Ring cl_two_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition sig2 : Signature F 2 :=
    Vector.cons F 1 1 (Vector.cons F 1 0 (Vector.nil F)).

  #[local] Instance Two_One : One (Cl F 2) := cl_one.
  #[local] Instance Two_Mult : HasMult (Cl F 2) := cl_mul sig2.
  #[local] Instance Two_Ring : Ring (Cl F 2) := Cl_Ring sig2.
  #[local] Instance Two_QF :
    QuadraticForm F (Fin.t 2 → F) (qform sig2) := Cl_QuadraticForm sig2.
  #[local] Instance Two_GA :
    GeometricAlgebra F (Fin.t 2 → F) (Cl F 2) (Q := qform sig2) :=
    Cl_GA sig2.

  Local Notation e1 := (ga_vec (coord Fin.F1)).
  Local Notation e2 := (ga_vec (coord (Fin.FS Fin.F1))).

  Lemma qform_e1 : qform sig2 (coord (F := F) Fin.F1) == 1.
  Proof. unfold sig2, coord. simpl. ring. Qed.

  Lemma qform_e2 : qform sig2 (coord (F := F) (Fin.FS Fin.F1)) == 1.
  Proof. unfold sig2, coord. simpl. ring. Qed.

  Lemma polar_e1_e2 :
    polar (qform sig2) (coord Fin.F1) (coord (Fin.FS Fin.F1)) == 0.
  Proof.
    rewrite (qform_polar 2 sig2 (coord Fin.F1) (coord (Fin.FS Fin.F1))).
    unfold sig2, coord. simpl. ring.
  Qed.

  Example e1_squared : e1 × e1 == 1.
  Proof.
    calc (e1 × e1)
      == ((qform sig2 (coord Fin.F1) *: 1) : Cl F 2)
         { contraction }
      == (((1 : F) *: 1) : Cl F 2) { qform_e1 }
      == (1 : Cl F 2) { left_identity }.
  Qed.

  Example e2_squared : e2 × e2 == 1.
  Proof.
    calc (e2 × e2)
      == ((qform sig2 (coord (Fin.FS Fin.F1)) *: 1) : Cl F 2)
         { contraction }
      == (((1 : F) *: 1) : Cl F 2) { qform_e2 }
      == (1 : Cl F 2) { left_identity }.
  Qed.

  Example e1_e2_anticommute : e1 × e2 == - (e2 × e1).
  Proof.
    calc (e1 × e2)
      == ((e1 × e2 + 0) : Cl F 2)
         { right_identity }
      == ((e1 × e2 + (e2 × e1 + - (e2 × e1))) : Cl F 2)
         { right_inverse }
      == (((e1 × e2 + e2 × e1) + - (e2 × e1)) : Cl F 2)
         { associative }
      == (((polar (qform sig2) (coord Fin.F1) (coord (Fin.FS Fin.F1)) *: 1)
          + - (e2 × e1)) : Cl F 2)
         { polarization }
      == ((((0 : F) *: 1) + - (e2 × e1)) : Cl F 2)
         { polar_e1_e2 }
      == ((0 + - (e2 × e1)) : Cl F 2)
         { zero_scale }
      == ((- (e2 × e1)) : Cl F 2)
         { left_identity }.
  Qed.

  Example e2_e1_anticommute : e2 × e1 == - (e1 × e2).
  Proof.
    rewrite e1_e2_anticommute. symmetry.
    exact (neg_involutive (e2 × e1)).
  Qed.

  Example e12_squared : (e1 × e2) × (e1 × e2) == - (1).
  Proof.
    transitivity ((e1 × (- e1)) : Cl F 2).
    - calc ((e1 × e2) × (e1 × e2))
        == ((e1 × (e2 × (e1 × e2))) : Cl F 2)
           { associative }
        == ((e1 × ((e2 × e1) × e2)) : Cl F 2)
           { associative }
        == ((e1 × ((- (e1 × e2)) × e2)) : Cl F 2)
           { e2_e1_anticommute }
        == ((e1 × (- ((e1 × e2) × e2))) : Cl F 2)
           { mul_neg_l }
        == ((e1 × (- (e1 × (e2 × e2)))) : Cl F 2)
           { associative }
        == ((e1 × (- (e1 × (1 : Cl F 2)))) : Cl F 2)
           { e2_squared }
        == ((e1 × (- e1)) : Cl F 2)
           { right_identity }.
    - calc ((e1 × (- e1)) : Cl F 2)
        == ((- (e1 × e1)) : Cl F 2)
           { mul_neg_r }
        == ((- (1)) : Cl F 2)
           { e1_squared }.
  Qed.

End TwoGenerators.

Fixpoint bcount {n : nat} : Blade n → nat :=
  match n with
  | O ⇒ fun _ ⇒ O
  | S m ⇒ fun A ⇒
      if bhead A then S (bcount (btail A)) else bcount (btail A)
  end.

Fixpoint rev_parity (k : nat) : bool :=
  match k with
  | 0 ⇒ false
  | S m ⇒ xorb (Nat.odd m) (rev_parity m)
  end.

Section GradeStructure.
  Context {F : Type} `{FF : Field F}.
  Add Ring cl_grade_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition cl_reverse {n : nat} (x : Cl F n) : Cl F n :=
    fun D ⇒ parity_sign (rev_parity (bcount D)) × x D.

  Definition cl_even {n : nat} (x : Cl F n) : Prop :=
    ∀ D : Blade n, bparity D = true → x D == 0.

  Definition cl_grade {n : nat} (k : nat) (x : Cl F n) : Cl F n :=
    fun D ⇒ if Nat.eqb (bcount D) k then x D else 0.

  Lemma cl_reverse_at : ∀ n (x : Cl F n) (D : Blade n),
    cl_reverse x D == parity_sign (rev_parity (bcount D)) × x D.
  Proof. reflexivity. Qed.

  Lemma cl_even_add : ∀ n (x y : Cl F n),
    cl_even x → cl_even y → cl_even (x + y).
  Proof.
    intros n x y Hx Hy D HD.
    rewrite (pw_add_at x y D), (Hx D HD), (Hy D HD). ring.
  Qed.

  Lemma cl_even_neg : ∀ n (x : Cl F n),
    cl_even x → cl_even (- x).
  Proof.
    intros n x Hx D HD.
    transitivity (- x D); [ reflexivity |].
    rewrite (Hx D HD). ring.
  Qed.

  Lemma cl_even_zero : ∀ n, cl_even (0 : Cl F n).
  Proof. intros n D HD. reflexivity. Qed.

  Lemma cl_even_scale : ∀ n (a : F) (x : Cl F n),
    cl_even x → cl_even (a *: x).
  Proof.
    intros n a x Hx D HD.
    rewrite (cl_scale_at n a x D), (Hx D HD). apply mul_0_r.
  Qed.

  Lemma cl_even_reverse : ∀ n (x : Cl F n),
    cl_even x → cl_even (cl_reverse x).
  Proof.
    intros n x Hx D HD.
    rewrite (cl_reverse_at n x D), (Hx D HD). apply mul_0_r.
  Qed.

  #[export] Instance cl_reverse_proper (n : nat) :
    Proper (equ ==> equ) (cl_reverse (n := n)).
  Proof.
    intros x y Hxy D.
    rewrite (cl_reverse_at n x D), (cl_reverse_at n y D), (Hxy D).
    reflexivity.
  Qed.

  #[export] Instance cl_even_proper (n : nat) :
    Proper (equ ==> iff) (cl_even (n := n)).
  Proof.
    intros x y Hxy.
    split; intros Hev D HD; [ rewrite <- (Hxy D) | rewrite (Hxy D) ];
      exact (Hev D HD).
  Qed.

End GradeStructure.
