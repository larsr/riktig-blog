Fixpoint skip {n : nat} (i : Fin.t (S n)) {struct n}
    : Fin.t n → Fin.t (S n) :=
  match n as n0 return Fin.t (S n0) → Fin.t n0 → Fin.t (S n0) with
  | O ⇒ fun _ j ⇒ Fin.case0 (fun _ ⇒ Fin.t 1) j
  | S m ⇒ fun i0 ⇒
      Fin.caseS' i0 (fun _ ⇒ Fin.t (S m) → Fin.t (S (S m)))
        Fin.FS
        (fun (i' : Fin.t (S m)) (j : Fin.t (S m)) ⇒
           Fin.caseS' j (fun _ ⇒ Fin.t (S (S m)))
             Fin.F1
             (fun j' : Fin.t m ⇒ Fin.FS (skip i' j')))
  end i.


Lemma skip_F1 : ∀ (n : nat) (j : Fin.t n),
  skip Fin.F1 j = Fin.FS j.
Proof.
  intros n j.
  destruct j as [m | m j'].
  - reflexivity.
  - reflexivity.
Qed.

Lemma skip_FS_F1 : ∀ (n : nat) (i : Fin.t (S n)),
  skip (Fin.FS i) Fin.F1 = Fin.F1.
Proof.
  intros n i.
  reflexivity.
Qed.

Lemma skip_FS_FS : ∀ (n : nat) (i : Fin.t (S n)) (j : Fin.t n),
  skip (Fin.FS i) (Fin.FS j) = Fin.FS (skip i j).
Proof.
  intros n i j.
  reflexivity.
Qed.

Lemma skip_ne : ∀ (n : nat) (i : Fin.t (S n)) (s : Fin.t n),
  skip i s ≠ i.
Proof.
  intros n; induction n as [| n IH]; intros i s.
  - exact (Fin.case0 (fun s0 ⇒ skip i s0 ≠ i) s).
  - revert s.
    apply (Fin.caseS' i (fun i0 ⇒ ∀ s : Fin.t (S n), skip i0 s ≠ i0)).
    + intros s Heq.
      rewrite (skip_F1 (S n) s) in Heq.
      exact (fin_FS_neq_F1 (S n) s Heq).
    + intros i' s.
      apply (Fin.caseS' s (fun s0 ⇒ skip (Fin.FS i') s0 ≠ Fin.FS i')).
      × intros Heq.
        rewrite (skip_FS_F1 n i') in Heq.
        exact (fin_F1_neq_FS (S n) i' Heq).
      × intros s' Heq.
        rewrite (skip_FS_FS n i' s') in Heq.
        exact (IH i' s' (Fin.FS_inj (skip i' s') i' Heq)).
Qed.

Lemma skip_inj : ∀ (n : nat) (i : Fin.t (S n)) (s1 s2 : Fin.t n),
  skip i s1 = skip i s2 → s1 = s2.
Proof.
  intros n; induction n as [| n IH]; intros i s1 s2 Heq.
  - exact (Fin.case0 (fun s ⇒ s = s2) s1).
  - revert s1 s2 Heq.
    apply (Fin.caseS' i (fun i0 ⇒ ∀ s1 s2 : Fin.t (S n),
      skip i0 s1 = skip i0 s2 → s1 = s2)).
    + intros s1 s2 Heq.
      rewrite (skip_F1 (S n) s1), (skip_F1 (S n) s2) in Heq.
      exact (Fin.FS_inj s1 s2 Heq).
    + intros i' s1 s2 Heq.
      apply (Fin.caseS' s1 (fun s10 ⇒ skip (Fin.FS i') s10 = skip (Fin.FS i') s2 →
        s10 = s2)).
      × apply (Fin.caseS' s2 (fun s20 ⇒ skip (Fin.FS i') Fin.F1 = skip (Fin.FS i') s20 →
          Fin.F1 = s20)).
        -- intros _. reflexivity.
        -- intros s2' Heq2.
           rewrite (skip_FS_F1 n i'), (skip_FS_FS n i' s2') in Heq2.
           destruct (fin_F1_neq_FS (S n) (skip i' s2') Heq2).
      × intros s1'.
        apply (Fin.caseS' s2 (fun s20 ⇒ skip (Fin.FS i') (Fin.FS s1') = skip (Fin.FS i') s20 →
          Fin.FS s1' = s20)).
        -- intros Heq2.
           rewrite (skip_FS_F1 n i'), (skip_FS_FS n i' s1') in Heq2.
           destruct (fin_FS_neq_F1 (S n) (skip i' s1') Heq2).
        -- intros s2' Heq2.
           rewrite (skip_FS_FS n i' s1'), (skip_FS_FS n i' s2') in Heq2.
           exact (f_equal Fin.FS (IH i' s1' s2' (Fin.FS_inj (skip i' s1') (skip i' s2') Heq2))).
      × exact Heq.
Qed.


Fixpoint unskip {n : nat} (r i : Fin.t (S n)) {struct n} : option (Fin.t n) :=
  match n as n0 return Fin.t (S n0) → Fin.t (S n0) → option (Fin.t n0) with
  | O ⇒ fun _ _ ⇒ None
  | S m ⇒ fun r0 i0 ⇒
      Fin.caseS' r0 (fun _ ⇒ option (Fin.t (S m)))
        (Fin.caseS' i0 (fun _ ⇒ option (Fin.t (S m)))
           None
           (fun i' ⇒ Some i'))
        (fun r' : Fin.t (S m) ⇒
           Fin.caseS' i0 (fun _ ⇒ option (Fin.t (S m)))
             (Some Fin.F1)
             (fun i' : Fin.t (S m) ⇒
                match unskip r' i' with
                | Some s' ⇒ Some (Fin.FS s')
                | None ⇒ None
                end))
  end r i.


Lemma unskip_F1_FS : ∀ (n : nat) (i' : Fin.t n),
  unskip Fin.F1 (Fin.FS i') = Some i'.
Proof.
  intros n; destruct n as [| m]; intros i'.
  - exact (Fin.case0 (fun i'0 ⇒ unskip Fin.F1 (Fin.FS i'0) = Some i'0) i').
  - reflexivity.
Qed.

Lemma unskip_FS_F1 : ∀ (n : nat) (r' : Fin.t (S n)),
  unskip (Fin.FS r') Fin.F1 = Some Fin.F1.
Proof.
  intros n r'. reflexivity.
Qed.

Lemma unskip_FS_FS : ∀ (n : nat) (r' i' : Fin.t (S n)),
  unskip (Fin.FS r') (Fin.FS i') =
    match unskip r' i' with
    | Some s' ⇒ Some (Fin.FS s')
    | None ⇒ None
    end.
Proof.
  intros n r' i'. reflexivity.
Qed.

Lemma unskip_F1_F1 : ∀ (n : nat), unskip (n := S n) Fin.F1 Fin.F1 = None.
Proof.
  intros n. reflexivity.
Qed.

Lemma unskip_some : ∀ (n : nat) (r i : Fin.t (S n)),
  r ≠ i → ∃ s : Fin.t n, unskip r i = Some s ∧ skip r s = i.
Proof.
  intros n; induction n as [| n IH]; intros r i Hri.
  - revert i Hri.
    apply (Fin.caseS' r (fun r0 ⇒ ∀ i0 : Fin.t 1, r0 ≠ i0 →
      ∃ s : Fin.t 0, unskip r0 i0 = Some s ∧ skip r0 s = i0)).
    + intros i0.
      apply (Fin.caseS' i0 (fun i1 ⇒ Fin.F1 ≠ i1 →
        ∃ s : Fin.t 0, unskip Fin.F1 i1 = Some s ∧ skip Fin.F1 s = i1)).
      × intros Hc. destruct (Hc eq_refl).
      × intros i' _.
        exact (Fin.case0 (fun i'0 ⇒ ∃ s : Fin.t 0,
          unskip Fin.F1 (Fin.FS i'0) = Some s
          ∧ skip Fin.F1 s = Fin.FS i'0) i').
    + intros r' i0 _.
      exact (Fin.case0 (fun r'0 ⇒ ∃ s : Fin.t 0,
        unskip (Fin.FS r'0) i0 = Some s ∧ skip (Fin.FS r'0) s = i0) r').
  - revert i Hri.
    apply (Fin.caseS' r (fun r0 ⇒ ∀ i0 : Fin.t (S (S n)), r0 ≠ i0 →
      ∃ s : Fin.t (S n), unskip r0 i0 = Some s ∧ skip r0 s = i0)).
    + intros i0.
      apply (Fin.caseS' i0 (fun i1 ⇒ Fin.F1 ≠ i1 →
        ∃ s : Fin.t (S n), unskip Fin.F1 i1 = Some s
          ∧ skip Fin.F1 s = i1)).
      × intros Hc. destruct (Hc eq_refl).
      × intros i' _.
        ∃ i'. split.
        -- exact (unskip_F1_FS (S n) i').
        -- exact (skip_F1 (S n) i').
    + intros r' i0.
      apply (Fin.caseS' i0 (fun i1 ⇒ Fin.FS r' ≠ i1 →
        ∃ s : Fin.t (S n), unskip (Fin.FS r') i1 = Some s
          ∧ skip (Fin.FS r') s = i1)).
      × intros _.
        ∃ Fin.F1. split.
        -- exact (unskip_FS_F1 n r').
        -- exact (skip_FS_F1 n r').
      × intros i' Hne.
        assert (Hne' : r' ≠ i').
        { intros Heq. apply Hne. exact (f_equal Fin.FS Heq). }
        destruct (IH r' i' Hne') as [s' [Hus Hsk]].
        ∃ (Fin.FS s'). split.
        -- rewrite (unskip_FS_FS n r' i'), Hus. reflexivity.
        -- rewrite (skip_FS_FS n r' s'), Hsk. reflexivity.
Qed.

Lemma unskip_skip : ∀ (n : nat) (r : Fin.t (S n)) (s : Fin.t n),
  unskip r (skip r s) = Some s.
Proof.
  intros n r s.
  destruct (unskip_some n r (skip r s) (fun Heq ⇒ skip_ne n r s (eq_sym Heq)))
    as [s' [Hu Hsk]].
  rewrite Hu.
  rewrite (skip_inj n r s' s Hsk).
  reflexivity.
Qed.


Lemma unskip_correct : ∀ (n : nat) (r i : Fin.t (S n)) (s : Fin.t n),
  unskip r i = Some s → skip r s = i.
Proof.
  intros n; induction n as [| n IH]; intros r i s Hu.
  - simpl in Hu. discriminate Hu.
  - revert i Hu.
    apply (Fin.caseS' r (fun r0 ⇒ ∀ i0 : Fin.t (S (S n)),
      unskip r0 i0 = Some s → skip r0 s = i0)).
    + intros i0.
      apply (Fin.caseS' i0 (fun i1 ⇒
        unskip Fin.F1 i1 = Some s → skip Fin.F1 s = i1)).
      × intros Hu0. rewrite (unskip_F1_F1 n) in Hu0. discriminate Hu0.
      × intros i' Hu0.
        rewrite (unskip_F1_FS (S n) i') in Hu0.
        injection Hu0 as Hu0.
        rewrite <- Hu0.
        exact (skip_F1 (S n) i').
    + intros r' i0.
      apply (Fin.caseS' i0 (fun i1 ⇒
        unskip (Fin.FS r') i1 = Some s → skip (Fin.FS r') s = i1)).
      × intros Hu0.
        rewrite (unskip_FS_F1 n r') in Hu0.
        injection Hu0 as Hu0.
        rewrite <- Hu0.
        exact (skip_FS_F1 n r').
      × intros i' Hu0.
        rewrite (unskip_FS_FS n r' i') in Hu0.
        destruct (unskip r' i') as [s' |] eqn:Hu'.
        -- injection Hu0 as Hu0.
           rewrite <- Hu0, (skip_FS_FS n r' s'), (IH r' i' s' Hu').
           reflexivity.
        -- discriminate Hu0.
Qed.


Fixpoint fin_weaken {n : nat} (k : Fin.t n) {struct k} : Fin.t (S n) :=
  match k with
  | Fin.F1 ⇒ Fin.F1
  | Fin.FS k' ⇒ Fin.FS (fin_weaken k')
  end.

Lemma fin_weaken_F1 : ∀ n : nat, @fin_weaken (S n) Fin.F1 = Fin.F1.
Proof. intros n. reflexivity. Qed.

Lemma fin_weaken_FS : ∀ (n : nat) (k' : Fin.t n),
  fin_weaken (Fin.FS k') = Fin.FS (fin_weaken k').
Proof. intros n k'. reflexivity. Qed.

Lemma skip_fweaken_self : ∀ (n : nat) (k : Fin.t n),
  skip (fin_weaken k) k = Fin.FS k.
Proof.
  intros n k; induction k as [n0 | n0 k' IH].
  - exact (skip_F1 (S n0) Fin.F1).
  - calc (skip (fin_weaken (Fin.FS k')) (Fin.FS k'))
      == (skip (Fin.FS (fin_weaken k')) (Fin.FS k')) { (fin_weaken_FS n0 k') }
      == (Fin.FS (skip (fin_weaken k') k')) { (skip_FS_FS n0 (fin_weaken k') k') }
      == (Fin.FS (Fin.FS k')) { IH }.
Qed.

Lemma skip_FS_self : ∀ (n : nat) (k : Fin.t n),
  skip (Fin.FS k) k = fin_weaken k.
Proof.
  intros n k; induction k as [n0 | n0 k' IH].
  - exact (skip_FS_F1 n0 Fin.F1).
  - calc (skip (Fin.FS (Fin.FS k')) (Fin.FS k'))
      == (Fin.FS (skip (Fin.FS k') k')) { (skip_FS_FS n0 (Fin.FS k') k') }
      == (Fin.FS (fin_weaken k')) { IH }
      == (fin_weaken (Fin.FS k')) { (fin_weaken_FS n0 k') }.
Qed.

Lemma fin_weaken_neq_FS : ∀ (n : nat) (k : Fin.t n), fin_weaken k ≠ Fin.FS k.
Proof.
  intros n k; induction k as [n0 | n0 k' IH].
  - exact (fin_F1_neq_FS (S n0) Fin.F1).
  - rewrite (fin_weaken_FS n0 k').
    intros Heq.
    exact (IH (Fin.FS_inj (fin_weaken k') (Fin.FS k') Heq)).
Qed.

Lemma skip_fweaken_eq_skip_FS : ∀ (n : nat) (k y : Fin.t n),
  y ≠ k → skip (fin_weaken k) y = skip (Fin.FS k) y.
Proof.
  intros n k; induction k as [n0 | n0 k' IH]; intros y Hne.
  - revert Hne.
    apply (Fin.caseS' y (fun y0 ⇒ y0 ≠ Fin.F1 →
      skip (fin_weaken (n := S n0) Fin.F1) y0 = skip (Fin.FS (n := S n0) Fin.F1) y0)).
    + intros Hc. destruct (Hc eq_refl).
    + intros y' _.
      calc (skip (fin_weaken (n := S n0) Fin.F1) (Fin.FS y'))
        == (skip Fin.F1 (Fin.FS y')) { (fin_weaken_F1 n0) }
        == (Fin.FS (Fin.FS y')) { (skip_F1 (S n0) (Fin.FS y')) }
        == (Fin.FS (skip Fin.F1 y')) { (skip_F1 n0 y') }
        == (skip (Fin.FS (n := S n0) Fin.F1) (Fin.FS y'))
             { (skip_FS_FS n0 Fin.F1 y') }.
  - revert Hne.
    apply (Fin.caseS' y (fun y0 ⇒ y0 ≠ Fin.FS k' →
      skip (fin_weaken (n := S n0) (Fin.FS k')) y0 = skip (Fin.FS (n := S n0) (Fin.FS k')) y0)).
    + intros _.
      calc (skip (fin_weaken (n := S n0) (Fin.FS k')) (Fin.F1 (n := n0)))
        == (skip (Fin.FS (fin_weaken k')) (Fin.F1 (n := n0)))
             { (fin_weaken_FS n0 k') }
        == (Fin.F1 (n := S n0)) { (skip_FS_F1 n0 (fin_weaken k')) }
        == (skip (Fin.FS (n := S n0) (Fin.FS k')) (Fin.F1 (n := n0)))
             { (skip_FS_F1 n0 (Fin.FS k')) }.
    + intros y' Hne'.
      assert (Hne'' : y' ≠ k').
      { intros Heq. apply Hne'. exact (f_equal Fin.FS Heq). }
      calc (skip (fin_weaken (n := S n0) (Fin.FS k')) (Fin.FS y'))
        == (skip (Fin.FS (fin_weaken k')) (Fin.FS y')) { (fin_weaken_FS n0 k') }
        == (Fin.FS (skip (fin_weaken k') y'))
             { (skip_FS_FS n0 (fin_weaken k') y') }
        == (Fin.FS (skip (Fin.FS k') y')) { (IH y' Hne'') }
        == (skip (Fin.FS (n := S n0) (Fin.FS k')) (Fin.FS y'))
             { (skip_FS_FS n0 (Fin.FS k') y') }.
Qed.

Lemma unskip_fweaken_FS : ∀ (n : nat) (k : Fin.t n),
  unskip (fin_weaken k) (Fin.FS k) = Some k.
Proof.
  intros n k; induction k as [n0 | n0 k' IH].
  - exact (unskip_F1_FS (S n0) Fin.F1).
  - calc (unskip (fin_weaken (Fin.FS k')) (Fin.FS (Fin.FS k')))
      == (unskip (Fin.FS (fin_weaken k')) (Fin.FS (Fin.FS k')))
           { (fin_weaken_FS n0 k') }
      == (match unskip (fin_weaken k') (Fin.FS k') with
          | Some s' ⇒ Some (Fin.FS s')
          | None ⇒ None
          end)
           { (unskip_FS_FS n0 (fin_weaken k') (Fin.FS k')) }
      == (Some (Fin.FS k')) { IH }.
Qed.

Lemma unskip_FS_fweaken : ∀ (n : nat) (k : Fin.t n),
  unskip (Fin.FS k) (fin_weaken k) = Some k.
Proof.
  intros n k; induction k as [n0 | n0 k' IH].
  - exact (unskip_FS_F1 n0 Fin.F1).
  - calc (unskip (Fin.FS (Fin.FS k')) (fin_weaken (Fin.FS k')))
      == (unskip (Fin.FS (Fin.FS k')) (Fin.FS (fin_weaken k')))
           { (fin_weaken_FS n0 k') }
      == (match unskip (Fin.FS k') (fin_weaken k') with
          | Some s' ⇒ Some (Fin.FS s')
          | None ⇒ None
          end)
           { (unskip_FS_FS n0 (Fin.FS k') (fin_weaken k')) }
      == (Some (Fin.FS k')) { IH }.
Qed.

Lemma unskip_adjacent_shape : ∀ (n0 : nat) (k : Fin.t (S n0)) (c : Fin.t (S (S n0))),
  c ≠ fin_weaken k → c ≠ Fin.FS k →
  ∃ k' : Fin.t n0,
    unskip c (fin_weaken k) = Some (fin_weaken k') ∧ unskip c (Fin.FS k) = Some (Fin.FS k').
Proof.
  intros n0; induction n0 as [| n0 IHn]; intros k c Hci Hcj.
  - revert c Hci Hcj.
    apply (Fin.caseS' k (fun k0 ⇒ ∀ c0 : Fin.t 2,
      c0 ≠ fin_weaken k0 → c0 ≠ Fin.FS k0 →
      ∃ w : Fin.t 0, unskip c0 (fin_weaken k0) = Some (fin_weaken w)
        ∧ unskip c0 (Fin.FS k0) = Some (Fin.FS w))).
    + intros c0.
      apply (Fin.caseS' c0 (fun c1 ⇒
        c1 ≠ fin_weaken (n := 1) Fin.F1 → c1 ≠ Fin.FS Fin.F1 →
        ∃ w : Fin.t 0, unskip c1 (fin_weaken (n := 1) Fin.F1) = Some (fin_weaken w)
          ∧ unskip c1 (Fin.FS Fin.F1) = Some (Fin.FS w))).
      × intros Hc _. rewrite fin_weaken_F1 in Hc. destruct (Hc eq_refl).
      × intros c1.
        apply (Fin.caseS' c1 (fun c2 ⇒
          Fin.FS c2 ≠ fin_weaken (n := 1) Fin.F1 → Fin.FS c2 ≠ Fin.FS Fin.F1 →
          ∃ w : Fin.t 0, unskip (Fin.FS c2) (fin_weaken (n := 1) Fin.F1) = Some (fin_weaken w)
            ∧ unskip (Fin.FS c2) (Fin.FS Fin.F1) = Some (Fin.FS w))).
        -- intros _ Hc. destruct (Hc eq_refl).
        -- intros c3.
           exact (Fin.case0 (fun c4 ⇒
             Fin.FS (Fin.FS c4) ≠ fin_weaken (n := 1) Fin.F1 → Fin.FS (Fin.FS c4) ≠ Fin.FS Fin.F1 →
             ∃ w : Fin.t 0, unskip (Fin.FS (Fin.FS c4)) (fin_weaken (n := 1) Fin.F1) = Some (fin_weaken w)
               ∧ unskip (Fin.FS (Fin.FS c4)) (Fin.FS Fin.F1) = Some (Fin.FS w)) c3).
    + intros k0. exact (Fin.case0 (fun k1 ⇒ ∀ c0 : Fin.t 2,
        c0 ≠ fin_weaken (Fin.FS k1) → c0 ≠ Fin.FS (Fin.FS k1) →
        ∃ w : Fin.t 0, unskip c0 (fin_weaken (Fin.FS k1)) = Some (fin_weaken w)
          ∧ unskip c0 (Fin.FS (Fin.FS k1)) = Some (Fin.FS w)) k0).
  - revert c Hci Hcj.
    apply (Fin.caseS' k (fun k0 ⇒ ∀ c0 : Fin.t (S (S (S n0))),
      c0 ≠ fin_weaken k0 → c0 ≠ Fin.FS k0 →
      ∃ w : Fin.t (S n0), unskip c0 (fin_weaken k0) = Some (fin_weaken w)
        ∧ unskip c0 (Fin.FS k0) = Some (Fin.FS w))).
    + intros c0.
      apply (Fin.caseS' c0 (fun c1 ⇒
        c1 ≠ fin_weaken Fin.F1 → c1 ≠ Fin.FS Fin.F1 →
        ∃ w : Fin.t (S n0), unskip c1 (fin_weaken Fin.F1) = Some (fin_weaken w)
          ∧ unskip c1 (Fin.FS Fin.F1) = Some (Fin.FS w))).
      × intros Hc _. rewrite fin_weaken_F1 in Hc. destruct (Hc eq_refl).
      × intros c1 _ Hcj.
        revert Hcj.
        apply (Fin.caseS' c1 (fun c2 ⇒
          Fin.FS c2 ≠ Fin.FS Fin.F1 →
          ∃ w : Fin.t (S n0), unskip (Fin.FS c2) (fin_weaken Fin.F1) = Some (fin_weaken w)
            ∧ unskip (Fin.FS c2) (Fin.FS Fin.F1) = Some (Fin.FS w))).
        -- intros Hc. destruct (Hc eq_refl).
        -- intros c2 _.
           ∃ Fin.F1.
           rewrite fin_weaken_F1.
           split.
           ++ exact (unskip_FS_F1 (S n0) (Fin.FS c2)).
           ++ rewrite (unskip_FS_FS (S n0) (Fin.FS c2) Fin.F1),
                (unskip_FS_F1 n0 c2).
              reflexivity.
    + intros k'.
      intros c0.
      apply (Fin.caseS' c0 (fun c1 ⇒
        c1 ≠ fin_weaken (Fin.FS k') → c1 ≠ Fin.FS (Fin.FS k') →
        ∃ w : Fin.t (S n0), unskip c1 (fin_weaken (Fin.FS k')) = Some (fin_weaken w)
          ∧ unskip c1 (Fin.FS (Fin.FS k')) = Some (Fin.FS w))).
      × intros _ _.
        ∃ k'.
        rewrite fin_weaken_FS.
        split.
        -- exact (unskip_F1_FS (S (S n0)) (fin_weaken k')).
        -- exact (unskip_F1_FS (S (S n0)) (Fin.FS k')).
      × intros c1 Hci Hcj.
        assert (Hci' : c1 ≠ fin_weaken k').
        { intros Heq. apply Hci. rewrite fin_weaken_FS. exact (f_equal Fin.FS Heq). }
        assert (Hcj' : c1 ≠ Fin.FS k').
        { intros Heq. apply Hcj. exact (f_equal Fin.FS Heq). }
        destruct (IHn k' c1 Hci' Hcj') as [w [Hu1 Hu2]].
        ∃ (Fin.FS w).
        rewrite fin_weaken_FS.
        split.
        -- rewrite (unskip_FS_FS (S n0) c1 (fin_weaken k')), Hu1, fin_weaken_FS.
           reflexivity.
        -- rewrite (unskip_FS_FS (S n0) c1 (Fin.FS k')), Hu2.
           reflexivity.
Qed.


Lemma unskip_fweaken_eq_unskip_FS : ∀ (n : nat) (k : Fin.t n) (x : Fin.t (S n)),
  x ≠ fin_weaken k → x ≠ Fin.FS k →
  unskip (fin_weaken k) x = unskip (Fin.FS k) x.
Proof.
  intros n k x Hx1 Hx2.
  destruct (unskip_some n (fin_weaken k) x (fun Heq ⇒ Hx1 (eq_sym Heq)))
    as [s [Hu1 Hsk1]].
  destruct (unskip_some n (Fin.FS k) x (fun Heq ⇒ Hx2 (eq_sym Heq)))
    as [t [Hu2 Hsk2]].
  assert (Hsne : s ≠ k).
  { intros Heq. apply Hx2. rewrite <- Hsk1, Heq. exact (skip_fweaken_self n k). }
  assert (Hst : s = t).
  { apply (skip_inj n (Fin.FS k) s t).
    calc (skip (Fin.FS k) s)
      == (skip (fin_weaken k) s) { (skip_fweaken_eq_skip_FS n k s Hsne) }
      == x { Hsk1 }
      == (skip (Fin.FS k) t) { Hsk2 }. }
  rewrite Hu1, Hu2, Hst.
  reflexivity.
Qed.


Definition fin_restrict {m : nat} (f : Fin.t (S m) → Fin.t (S m))
    (y : Fin.t m) : Fin.t m :=
  match unskip (f Fin.F1) (f (Fin.FS y)) with
  | Some t ⇒ t
  | None ⇒ y
  end.

Lemma fin_restrict_skip : ∀ (m : nat) (f : Fin.t (S m) → Fin.t (S m)),
  (∀ x y : Fin.t (S m), f x = f y → x = y) →
  ∀ y : Fin.t m, skip (f Fin.F1) (fin_restrict f y) = f (Fin.FS y).
Proof.
  intros m f Hinj y.
  assert (Hne : f Fin.F1 ≠ f (Fin.FS y)).
  { intros Heq. exact (fin_F1_neq_FS m y (Hinj Fin.F1 (Fin.FS y) Heq)). }
  destruct (unskip_some m (f Fin.F1) (f (Fin.FS y)) Hne) as [t [Hu Hsk]].
  unfold fin_restrict.
  rewrite Hu.
  exact Hsk.
Qed.

Lemma fin_restrict_inj : ∀ (m : nat) (f : Fin.t (S m) → Fin.t (S m)),
  (∀ x y : Fin.t (S m), f x = f y → x = y) →
  ∀ y1 y2 : Fin.t m, fin_restrict f y1 = fin_restrict f y2 → y1 = y2.
Proof.
  intros m f Hinj y1 y2 Heq.
  apply (Fin.FS_inj y1 y2).
  apply (Hinj (Fin.FS y1) (Fin.FS y2)).
  rewrite <- (fin_restrict_skip m f Hinj y1), <- (fin_restrict_skip m f Hinj y2),
    Heq.
  reflexivity.
Qed.

Lemma fin_restrict_ext : ∀ (m : nat) (f g : Fin.t (S m) → Fin.t (S m)),
  (∀ x : Fin.t (S m), f x = g x) →
  ∀ y : Fin.t m, fin_restrict f y = fin_restrict g y.
Proof.
  intros m f g Hfg y.
  unfold fin_restrict.
  rewrite (Hfg Fin.F1), (Hfg (Fin.FS y)).
  reflexivity.
Qed.


Lemma fin_restrict_swap_adjacent_at_lower : ∀ (n : nat) (k : Fin.t n)
    (g : Fin.t (S n) → Fin.t (S n)),
  (∀ x y : Fin.t (S n), g x = g y → x = y) →
  g Fin.F1 = fin_weaken k →
  ∀ y : Fin.t n,
  fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x)) y
    = fin_restrict g y.
Proof.
  intros n k g Hinj Hc y.
  unfold fin_restrict.
  rewrite Hc, (swap_fin_at_i (S n) (fin_weaken k) (Fin.FS k)).
  assert (Hne1 : g (Fin.FS y) ≠ fin_weaken k).
  { intros Heq. apply (fin_FS_neq_F1 n y). apply Hinj. rewrite Heq, Hc. reflexivity. }
  destruct (Fin.eq_dec (g (Fin.FS y)) (Fin.FS k)) as [Heq2 | Hne2].
  - rewrite Heq2,
      (swap_fin_at_j (S n) (fin_weaken k) (Fin.FS k) (fin_weaken_neq_FS n k)),
      (unskip_FS_fweaken n k), (unskip_fweaken_FS n k).
    reflexivity.
  - rewrite (swap_fin_ne (S n) (fin_weaken k) (Fin.FS k) (g (Fin.FS y)) Hne1 Hne2),
      (unskip_fweaken_eq_unskip_FS n k (g (Fin.FS y)) Hne1 Hne2).
    reflexivity.
Qed.

Lemma fin_restrict_swap_adjacent_at_upper : ∀ (n : nat) (k : Fin.t n)
    (g : Fin.t (S n) → Fin.t (S n)),
  (∀ x y : Fin.t (S n), g x = g y → x = y) →
  g Fin.F1 = Fin.FS k →
  ∀ y : Fin.t n,
  fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x)) y
    = fin_restrict g y.
Proof.
  intros n k g Hinj Hc y.
  unfold fin_restrict.
  rewrite Hc,
    (swap_fin_at_j (S n) (fin_weaken k) (Fin.FS k) (fin_weaken_neq_FS n k)).
  assert (Hne2 : g (Fin.FS y) ≠ Fin.FS k).
  { intros Heq. apply (fin_FS_neq_F1 n y). apply Hinj. rewrite Heq, Hc. reflexivity. }
  destruct (Fin.eq_dec (g (Fin.FS y)) (fin_weaken k)) as [Heq1 | Hne1].
  - rewrite Heq1, (swap_fin_at_i (S n) (fin_weaken k) (Fin.FS k)),
      (unskip_fweaken_FS n k), (unskip_FS_fweaken n k).
    reflexivity.
  - rewrite (swap_fin_ne (S n) (fin_weaken k) (Fin.FS k) (g (Fin.FS y)) Hne1 Hne2),
      (unskip_fweaken_eq_unskip_FS n k (g (Fin.FS y)) Hne1 Hne2).
    reflexivity.
Qed.

Lemma fin_restrict_swap_adjacent_missed : ∀ (n0 : nat) (k : Fin.t (S n0))
    (k' : Fin.t n0) (g : Fin.t (S (S n0)) → Fin.t (S (S n0))),
  (∀ x y : Fin.t (S (S n0)), g x = g y → x = y) →
  g Fin.F1 ≠ fin_weaken k → g Fin.F1 ≠ Fin.FS k →
  unskip (g Fin.F1) (fin_weaken k) = Some (fin_weaken k') →
  unskip (g Fin.F1) (Fin.FS k) = Some (Fin.FS k') →
  ∀ y : Fin.t (S n0),
  fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x)) y
    = swap_fin (fin_weaken k') (Fin.FS k') (fin_restrict g y).
Proof.
  intros n0 k k' g Hinj Hc1 Hc2 Hu1 Hu2 y.
  unfold fin_restrict.
  rewrite (swap_fin_ne (S (S n0)) (fin_weaken k) (Fin.FS k) (g Fin.F1) Hc1 Hc2).
  destruct (Fin.eq_dec (g (Fin.FS y)) (fin_weaken k)) as [Heq1 | Hne1].
  - rewrite Heq1, (swap_fin_at_i (S (S n0)) (fin_weaken k) (Fin.FS k)), Hu2, Hu1,
      (swap_fin_at_i (S n0) (fin_weaken k') (Fin.FS k')).
    reflexivity.
  - destruct (Fin.eq_dec (g (Fin.FS y)) (Fin.FS k)) as [Heq2 | Hne2].
    + rewrite Heq2,
        (swap_fin_at_j (S (S n0)) (fin_weaken k) (Fin.FS k) (fin_weaken_neq_FS (S n0) k)),
        Hu1, Hu2,
        (swap_fin_at_j (S n0) (fin_weaken k') (Fin.FS k') (fin_weaken_neq_FS n0 k')).
      reflexivity.
    + rewrite (swap_fin_ne (S (S n0)) (fin_weaken k) (Fin.FS k) (g (Fin.FS y)) Hne1 Hne2).
      destruct (unskip (g Fin.F1) (g (Fin.FS y))) as [t |] eqn:Hut.
      × assert (Hsk : skip (g Fin.F1) t = g (Fin.FS y)).
        { exact (unskip_correct (S n0) (g Fin.F1) (g (Fin.FS y)) t Hut). }
        assert (Ht1 : t ≠ fin_weaken k').
        { intros Heq. apply Hne1. rewrite <- Hsk, Heq.
          exact (unskip_correct (S n0) (g Fin.F1) (fin_weaken k) (fin_weaken k') Hu1). }
        assert (Ht2 : t ≠ Fin.FS k').
        { intros Heq. apply Hne2. rewrite <- Hsk, Heq.
          exact (unskip_correct (S n0) (g Fin.F1) (Fin.FS k) (Fin.FS k') Hu2). }
        rewrite (swap_fin_ne (S n0) (fin_weaken k') (Fin.FS k') t Ht1 Ht2).
        reflexivity.
      × assert (Hne : g Fin.F1 ≠ g (Fin.FS y)).
        { intros Heq. exact (fin_F1_neq_FS (S n0) y (Hinj Fin.F1 (Fin.FS y) Heq)). }
        destruct (unskip_some (S n0) (g Fin.F1) (g (Fin.FS y)) Hne) as [s [Hus _]].
        rewrite Hus in Hut.
        discriminate Hut.
Qed.

Lemma fin_to_nat_fweaken : ∀ (n : nat) (k : Fin.t n), fin_to_nat (fin_weaken k) = fin_to_nat k.
Proof.
  intros n; induction n as [| n IH]; intros k.
  - exact (Fin.case0 (fun k0 ⇒ fin_to_nat (fin_weaken k0) = fin_to_nat k0) k).
  - apply (Fin.caseS' k (fun k0 ⇒ fin_to_nat (fin_weaken k0) = fin_to_nat k0)).
    + reflexivity.
    + intros k'.
      calc (fin_to_nat (fin_weaken (Fin.FS k')))
        == (fin_to_nat (Fin.FS (fin_weaken k'))) { (fin_weaken_FS n k') }
        == (S (fin_to_nat (fin_weaken k'))) { (fin_to_nat_FS (S n) (fin_weaken k')) }
        == (S (fin_to_nat k')) { (IH k') }
        == (fin_to_nat (Fin.FS k')) { (fin_to_nat_FS n k') }.
Qed.

Lemma fin_weaken_surj : ∀ (n0 : nat) (i : Fin.t (S n0)),
  (fin_to_nat i < n0)%nat → ∃ k : Fin.t n0, i = fin_weaken k.
Proof.
  intros n0; induction n0 as [| m IH]; intros i Hlt.
  - exfalso. lia.
  - revert Hlt.
    apply (Fin.caseS' i (fun i0 ⇒ (fin_to_nat i0 < S m)%nat → ∃ k : Fin.t (S m), i0 = fin_weaken k)).
    + intros _. ∃ Fin.F1. symmetry. exact (fin_weaken_F1 m).
    + intros i' Hlt.
      assert (Hlt' : (fin_to_nat i' < m)%nat).
      { rewrite (fin_to_nat_FS (S m) i') in Hlt. lia. }
      destruct (IH i' Hlt') as [k' Hk'].
      ∃ (Fin.FS k').
      rewrite Hk'. symmetry. exact (fin_weaken_FS m k').
Qed.
