From Riktig.Structure Require Export BinaryArith BinRec.
From Riktig.Theory Require Export BinRec.
From Riktig.Programs Require Export BinaryNumber.

Any binary numbers with recursion have their arithmetic: the programs of Programs/BinaryNumber, whose equations and properness are Theory.BinRec's.
Section BinaryArithBinRec.
  Context {N : Type} {eqN : EqRel N} {zeroN : Zero N} {bit0N : HasBit0 N} {bit1N : HasBit1 N}
    {BN : BinaryNumber N zeroN bit0N bit1N} {binrecN : HasBinRec N} {BR : BinRec N binrecN}.

  #[export] Instance BinRec_HasSucc : HasSucc N := bin_succ.
  #[export] Instance BinRec_One : One N := bit1 0.
  #[export] Instance BinRec_HasAdd : HasAdd N := bin_add.
  #[export] Instance BinRec_HasMult : HasMult N := bin_mult.

  #[export] Instance BinRec_BinaryArith :
    BinaryArith N zeroN bit0N bit1N BinRec_HasSucc BinRec_One BinRec_HasAdd BinRec_HasMult := {|
    BinaryArith.binary := BN;
    BinaryArith.succ_proper := bin_succ_proper;
    BinaryArith.add_proper := bin_add_proper;
    BinaryArith.mul_proper := bin_mult_proper;
    BinaryArith.one_bit1 := reflexivity (bit1 0);
    BinaryArith.succ_zero := bin_succ_zero;
    BinaryArith.succ_bit0 := bin_succ_bit0;
    BinaryArith.succ_bit1 := bin_succ_bit1;
    BinaryArith.add_zero_l := bin_add_zero_l;
    BinaryArith.add_bit0_zero := bin_add_bit0_zero;
    BinaryArith.add_bit0_bit0 := bin_add_bit0_bit0;
    BinaryArith.add_bit0_bit1 := bin_add_bit0_bit1;
    BinaryArith.add_bit1_zero := bin_add_bit1_zero;
    BinaryArith.add_bit1_bit0 := bin_add_bit1_bit0;
    BinaryArith.add_bit1_bit1 := bin_add_bit1_bit1;
    BinaryArith.mul_zero_l := bin_mult_zero_l;
    BinaryArith.mul_bit0 := bin_mult_bit0;
    BinaryArith.mul_bit1 := bin_mult_bit1;
  |}.
End BinaryArithBinRec.
