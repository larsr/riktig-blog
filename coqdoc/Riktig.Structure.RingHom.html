Library Riktig.Structure.RingHom

A ring homomorphism is a map that respects one, addition and multiplication.
From Coq Require Import Morphisms.
From Riktig Require Export Ops Props.

Module RingHom.
  Class RingHom (A B : Type)
      {eA : EqRel A} {oA : One A} {aA : HasAdd A} {mA : HasMult A}
      {eB : EqRel B} {oB : One B} {aB : HasAdd B} {mB : HasMult B}
      (h : A → B) := {
    hom_proper :: Proper (equ ==> equ) h;
    hom_additive :: Additive A B add add h;
    hom_multiplicative :: Additive A B mult mult h;
    hom_one : h 1 == (1 : B);
  }.
  Hint Mode RingHom ! ! - - - - - - - - ! : typeclass_instances.
End RingHom.
Export RingHom.
