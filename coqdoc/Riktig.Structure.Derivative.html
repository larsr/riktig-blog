Library Riktig.Structure.Derivative

From Riktig.Structure Require Export VectorSpace Linear.

Differentiation between two vector spaces over a field: a linear map is the derivative of a function at a point. The derivative respects equality, is linear and unique, constants have derivative zero, a linear map is its own derivative everywhere, and derivatives add and scale with the functions.
Module Derivative.
  Class Derivative (F U V : Type) `{VectorSpace F U}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
      {VSV : VectorSpace F V zV aV nV sV}
      (differential : HasDifferential U V) := {
    differential_proper : ∀ (f f' : U → V) (a a' : U) (L L' : U → V),
      (∀ x : U, f x == f' x) → a == a' → (∀ h : U, L h == L' h) →
      differential f a L → differential f' a' L';
    differential_linear : ∀ (f : U → V) (a : U) (L : U → V),
      differential f a L → Linear F U V L;
    differential_unique : ∀ (f : U → V) (a : U) (L L' : U → V),
      differential f a L → differential f a L' → ∀ h : U, L h == L' h;
    differential_const : ∀ (v : V) (a : U), differential (fun _ ⇒ v) a (fun _ ⇒ 0);
    differential_of_linear : ∀ f : U → V, Linear F U V f → ∀ a : U, differential f a f;
    differential_add : ∀ (f g : U → V) (a : U) (Lf Lg : U → V),
      differential f a Lf → differential g a Lg →
      differential (fun x ⇒ f x + g x) a (fun h ⇒ Lf h + Lg h);
    differential_smul : ∀ (c : F) (f : U → V) (a : U) (Lf : U → V),
      differential f a Lf → differential (fun x ⇒ c *: f x) a (fun h ⇒ c *: Lf h);
  }.
  Hint Mode Derivative - ! ! - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Derivative.
