Section ScaleAdditive.
  Context {F V : Type} `{Field F}.
  Context {Ve : EqRel V} {Va : HasAdd V} {Vs : HasScale F V} {SL : ScaleLaws F V}.

  #[export] Instance scale_l_additive (a : F) : Additive V V add add (scale a) :=
    left_distributive a.

  #[export] Instance scale_r_additive (u : V) :
    Additive F V add add (fun a : F ⇒ a *: u) :=
    fun a b ⇒ right_distributive a b u.

End ScaleAdditive.

Section VectorSpaceTheory.
  Context {F V : Type} `{Field F}.
  Context {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V}
          {Vs : HasScale F V} {VG : AbGroup V} {SL : ScaleLaws F V}.

  #[export] Instance scale_l_proper : ∀ a : F, Proper (equ ==> equ) (@scale F V _ a).
  Proof.
    intros a u v Huv.
    rewrite Huv.
    reflexivity.
  Qed.

  #[export] Instance scale_r_proper : ∀ u : V, Proper (equ ==> equ) (fun a : F ⇒ a *: u).
  Proof.
    intros u a b Hab.
    rewrite Hab.
    reflexivity.
  Qed.

  Theorem scale_zero : ∀ a : F, a *: (0 : V) == 0.
  Proof.
    intros a.
    exact (additive_zero (scale a)).
  Qed.

  Theorem zero_scale : ∀ u : V, (0 : F) *: u == 0.
  Proof.
    intros u.
    exact (additive_zero (fun a : F ⇒ a *: u)).
  Qed.

  Theorem neg_scale : ∀ (a : F) (u : V), (- a) *: u == - (a *: u).
  Proof.
    intros a u.
    exact (additive_negation (fun x : F ⇒ x *: u) a).
  Qed.

  Theorem scale_neg : ∀ (a : F) (u : V), a *: (- u) == - (a *: u).
  Proof.
    intros a u.
    exact (additive_negation (scale a) u).
  Qed.

  Theorem scale_sub : ∀ (a : F) (u v : V),
    a *: (u + - v) == a *: u + - (a *: v).
  Proof.
    intros a u v.
    calc (a *: (u + - v))
      == (a *: u + a *: (- v)) { left_distributive }
      == (a *: u + - (a *: v)) { scale_neg }.
  Qed.

  Theorem sub_scale : ∀ (a b : F) (u : V),
    (a + - b) *: u == a *: u + - (b *: u).
  Proof.
    intros a b u.
    calc ((a + - b) *: u)
      == (a *: u + (- b) *: u) { right_distributive }
      == (a *: u + - (b *: u)) { neg_scale }.
  Qed.

  Theorem scale_swap : ∀ (a b : F) (u : V),
    a *: (b *: u) == b *: (a *: u).
  Proof.
    intros a b u.
    calc (a *: (b *: u))
      == ((a × b) *: u) { associative }
      == ((b × a) *: u) { commutative }
      == (b *: (a *: u)) { associative }.
  Qed.

  Theorem neg1_scale : ∀ u : V, (- (1 : F)) *: u == - u.
  Proof.
    intros u.
    calc ((- (1 : F)) *: u)
      == (- (1 *: u)) { neg_scale }
      == (- u) { left_identity }.
  Qed.

  Theorem scale_flip : ∀ (a b : F) (u : V),
    a *: (b *: u) == (b × a) *: u.
  Proof.
    intros a b u.
    calc (a *: (b *: u))
      == ((a × b) *: u) { associative }
      == ((b × a) *: u) { commutative }.
  Qed.

End VectorSpaceTheory.
