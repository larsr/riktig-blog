Section VectorSpaceTheory.
  Context {F V : Type} `{VectorSpace F V}.

  Theorem smul_zero : ∀ a : F, a *: (0 : V) == 0.
  Proof.
    intros a.
    apply (add_left_cancel (a *: (0 : V))).
    calc (a *: (0 : V) + a *: (0 : V))
      == (a *: ((0 : V) + 0)) { left_distributive }
      == (a *: (0 : V)) { (right_identity (action := add) (0 : V)) }
      == (a *: (0 : V) + 0) { right_identity }.
  Qed.

  Theorem zero_smul : ∀ u : V, (0 : F) *: u == 0.
  Proof.
    intros u.
    apply (add_left_cancel ((0 : F) *: u)).
    calc ((0 : F) *: u + (0 : F) *: u)
      == (((0 : F) + 0) *: u) { right_distributive }
      == ((0 : F) *: u) { (right_identity (action := add) (0 : F)) }
      == ((0 : F) *: u + 0) { right_identity }.
  Qed.

  Theorem neg_smul : ∀ (a : F) (u : V), - a *: u == - (a *: u).
  Proof.
    intros a u.
    apply eq_neg_of_add_eq_zero_left.
    calc (- a *: u + a *: u)
      == ((- a + a) *: u) { right_distributive }
      == ((0 : F) *: u) { (left_inverse (op := add) a) }
      == (0 : V) { zero_smul }.
  Qed.

  Theorem smul_neg : ∀ (a : F) (u : V), a *: - u == - (a *: u).
  Proof.
    intros a u.
    apply eq_neg_of_add_eq_zero_left.
    calc (a *: - u + a *: u)
      == (a *: (- u + u)) { left_distributive }
      == (a *: (0 : V)) { (left_inverse (op := add) u) }
      == (0 : V) { smul_zero }.
  Qed.

  Theorem smul_sub : ∀ (a : F) (u v : V),
    a *: (u + - v) == a *: u + - (a *: v).
  Proof.
    intros a u v.
    calc (a *: (u + - v))
      == (a *: u + a *: - v) { left_distributive }
      == (a *: u + - (a *: v)) { smul_neg }.
  Qed.

  Theorem sub_smul : ∀ (a b : F) (u : V),
    (a + - b) *: u == a *: u + - (b *: u).
  Proof.
    intros a b u.
    calc ((a + - b) *: u)
      == (a *: u + - b *: u) { right_distributive }
      == (a *: u + - (b *: u)) { neg_smul }.
  Qed.

  Theorem smul_comm : ∀ (a b : F) (u : V), a *: (b *: u) == b *: (a *: u).
  Proof.
    intros a b u.
    calc (a *: (b *: u))
      == ((a × b) *: u) { associative }
      == ((b × a) *: u) { (commutative (op := mult) a b) }
      == (b *: (a *: u)) { associative }.
  Qed.

  Theorem neg_one_smul : ∀ u : V, - (1 : F) *: u == - u.
  Proof.
    intros u.
    calc (- (1 : F) *: u)
      == (- ((1 : F) *: u)) { neg_smul }
      == (- u) { left_identity }.
  Qed.

Doubling by scaling and doubling by adding agree, in every characteristic.
  Theorem two_smul : ∀ u : V, ((1 : F) + 1) *: u == u + u.
  Proof.
    intros u.
    calc (((1 : F) + 1) *: u)
      == ((1 : F) *: u + (1 : F) *: u) { right_distributive }
      == (u + u) { (left_identity (action := scale) u) }.
  Qed.

End VectorSpaceTheory.
