From Coq Require Import NArith.
From Riktig Require Export Structure.NatRec Impl.NaturalsN.

Recursion over the binary naturals: the loop combinator for programs whose fuel should not be a unary numeral.
Instance N_HasNatRec : HasNatRec N := fun A a f n ⇒ N.peano_rect (fun _ ⇒ A) a (fun k acc ⇒ f k acc) n.

Instance N_NatRec : NatRec N N_HasNatRec.
Proof.
  split.
  - intros A eA EA a f Hf n n' Hn. unfold equ, N_EqRel in Hn. subst. reflexivity.
  - intros A eA EA a f. reflexivity.
  - intros A eA EA a f n. unfold natrec, N_HasNatRec, succ, N_HasSucc.
    rewrite N.peano_rect_succ. reflexivity.
Qed.
