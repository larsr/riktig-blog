Record VSIso (F U V : Type) `{FF : Field.Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {VV : VectorSpace F V} := {
  vsi_to : LinearMap F U V;
  vsi_from : LinearMap F V U;
  vsi_fromto : ∀ u : U, vsi_from (vsi_to u) == u;
  vsi_tofrom : ∀ v : V, vsi_to (vsi_from v) == v
}.

Arguments vsi_to {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.
Arguments vsi_from {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.
Arguments vsi_fromto {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.
Arguments vsi_tofrom {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.


Section Unitor.
  Context {F A : Type} `{FF : Field.Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.

  Let VAU : VectorSpace F (A × unit) := Prod_VectorSpace.

  #[global] Instance unitor_to_proper : Proper (equ ==> equ) (fun p : A × unit ⇒ fst p).
  Proof.
    intros p q [H1 _].
    exact H1.
  Qed.

  Lemma unitor_to_add : ∀ p q : A × unit, fst (p + q) == fst p + fst q.
  Proof.
    intros p q.
    reflexivity.
  Qed.

  Lemma unitor_to_scale : ∀ (a : F) (p : A × unit),
    fst (a *: p) == a *: fst p.
  Proof.
    intros a p.
    reflexivity.
  Qed.

  Definition unitor_to : LinearMap F (A × unit) A :=
    {| lm_fun := (fun p : A × unit ⇒ fst p);
       linear_map_proper := unitor_to_proper;
       linear_map_additive := unitor_to_add;
       linear_map_scale := unitor_to_scale |}.

  #[global] Instance unitor_from_proper : Proper (equ ==> equ) (fun a : A ⇒ (a, tt)).
  Proof.
    intros a b Hab.
    split.
    - exact Hab.
    - exact I.
  Qed.

  Lemma unitor_from_add : ∀ a b : A, (a + b, tt) == (a, tt) + (b, tt).
  Proof.
    intros a b.
    split.
    - reflexivity.
    - exact I.
  Qed.

  Lemma unitor_from_scale : ∀ (a : F) (u : A),
    (a *: u, tt) == a *: (u, tt).
  Proof.
    intros a u.
    split.
    - reflexivity.
    - exact I.
  Qed.

  Definition unitor_from : LinearMap F A (A × unit) :=
    {| lm_fun := (fun a : A ⇒ (a, tt));
       linear_map_proper := unitor_from_proper;
       linear_map_additive := unitor_from_add;
       linear_map_scale := unitor_from_scale |}.

  Lemma unitor_fromto : ∀ p : A × unit, (fst p, tt) == p.
  Proof.
    intros p.
    split.
    - reflexivity.
    - exact I.
  Qed.

  Lemma unitor_tofrom : ∀ a : A, fst (a, tt) == a.
  Proof.
    intros a.
    reflexivity.
  Qed.

  Definition vsiso_unitor : VSIso F (A × unit) A :=
    {| vsi_to := unitor_to; vsi_from := unitor_from;
       vsi_fromto := unitor_fromto; vsi_tofrom := unitor_tofrom |}.

End Unitor.


Section Swap.
  Context {F A B : Type} `{FF : Field.Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VBA : VectorSpace F (B × A) := Prod_VectorSpace.

  #[global] Instance swap_proper : Proper (equ ==> equ) (fun p : A × B ⇒ (snd p, fst p)).
  Proof.
    intros p q [H1 H2].
    split.
    - exact H2.
    - exact H1.
  Qed.

  Lemma swap_add : ∀ p q : A × B,
    (snd (p + q), fst (p + q)) == (snd p, fst p) + (snd q, fst q).
  Proof.
    intros p q.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Lemma swap_scale : ∀ (a : F) (p : A × B),
    (snd (a *: p), fst (a *: p)) == a *: (snd p, fst p).
  Proof.
    intros a p.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Definition swap_lm : LinearMap F (A × B) (B × A) :=
    {| lm_fun := (fun p : A × B ⇒ (snd p, fst p));
       linear_map_proper := swap_proper;
       linear_map_additive := swap_add;
       linear_map_scale := swap_scale |}.

  Lemma prod_eta : ∀ p : A × B, (fst p, snd p) == p.
  Proof.
    intros p.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

End Swap.

Section Braiding.
  Context {F A B : Type} `{FF : Field.Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VBA : VectorSpace F (B × A) := Prod_VectorSpace.

  Definition vsiso_braiding : VSIso F (A × B) (B × A) :=
    {| vsi_to := swap_lm; vsi_from := swap_lm;
       vsi_fromto := prod_eta; vsi_tofrom := prod_eta |}.

End Braiding.


Section Associator.
  Context {F A B C : Type} `{FF : Field.Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.
  Context {eC : EqRel C} {zC : Zero C} {aC : HasAdd C} {nC : HasNeg C} {sC : HasScale F C}.
  Context {VC : VectorSpace F C}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VBC : VectorSpace F (B × C) := Prod_VectorSpace.
  Let VAB_C : VectorSpace F ((A × B) × C) := Prod_VectorSpace.
  Let VA_BC : VectorSpace F (A × (B × C)) := Prod_VectorSpace.

  #[global] Instance assoc_to_proper : Proper (equ ==> equ)
    (fun p : (A × B) × C ⇒ (fst (fst p), (snd (fst p), snd p))).
  Proof.
    intros p q [[H1 H2] H3].
    split.
    - exact H1.
    - split.
      + exact H2.
      + exact H3.
  Qed.

  Lemma assoc_to_add : ∀ p q : (A × B) × C,
    (fst (fst (p + q)), (snd (fst (p + q)), snd (p + q)))
      == (fst (fst p), (snd (fst p), snd p))
         + (fst (fst q), (snd (fst q), snd q)).
  Proof.
    intros p q.
    split.
    - reflexivity.
    - split.
      + reflexivity.
      + reflexivity.
  Qed.

  Lemma assoc_to_scale : ∀ (a : F) (p : (A × B) × C),
    (fst (fst (a *: p)), (snd (fst (a *: p)), snd (a *: p)))
      == a *: (fst (fst p), (snd (fst p), snd p)).
  Proof.
    intros a p.
    split.
    - reflexivity.
    - split.
      + reflexivity.
      + reflexivity.
  Qed.

  Definition assoc_to : LinearMap F ((A × B) × C) (A × (B × C)) :=
    {| lm_fun := (fun p : (A × B) × C ⇒ (fst (fst p), (snd (fst p), snd p)));
       linear_map_proper := assoc_to_proper;
       linear_map_additive := assoc_to_add;
       linear_map_scale := assoc_to_scale |}.

  #[global] Instance assoc_from_proper : Proper (equ ==> equ)
    (fun q : A × (B × C) ⇒ ((fst q, fst (snd q)), snd (snd q))).
  Proof.
    intros p q [H1 [H2 H3]].
    split.
    - split.
      + exact H1.
      + exact H2.
    - exact H3.
  Qed.

  Lemma assoc_from_add : ∀ p q : A × (B × C),
    ((fst (p + q), fst (snd (p + q))), snd (snd (p + q)))
      == ((fst p, fst (snd p)), snd (snd p))
         + ((fst q, fst (snd q)), snd (snd q)).
  Proof.
    intros p q.
    split.
    - split.
      + reflexivity.
      + reflexivity.
    - reflexivity.
  Qed.

  Lemma assoc_from_scale : ∀ (a : F) (q : A × (B × C)),
    ((fst (a *: q), fst (snd (a *: q))), snd (snd (a *: q)))
      == a *: ((fst q, fst (snd q)), snd (snd q)).
  Proof.
    intros a q.
    split.
    - split.
      + reflexivity.
      + reflexivity.
    - reflexivity.
  Qed.

  Definition assoc_from : LinearMap F (A × (B × C)) ((A × B) × C) :=
    {| lm_fun := (fun q : A × (B × C) ⇒ ((fst q, fst (snd q)), snd (snd q)));
       linear_map_proper := assoc_from_proper;
       linear_map_additive := assoc_from_add;
       linear_map_scale := assoc_from_scale |}.

  Lemma assoc_fromto : ∀ p : (A × B) × C,
    ((fst (fst p), snd (fst p)), snd p) == p.
  Proof.
    intros p.
    split.
    - split.
      + reflexivity.
      + reflexivity.
    - reflexivity.
  Qed.

  Lemma assoc_tofrom : ∀ q : A × (B × C),
    (fst q, (fst (snd q), snd (snd q))) == q.
  Proof.
    intros q.
    split.
    - reflexivity.
    - split.
      + reflexivity.
      + reflexivity.
  Qed.

  Definition vsiso_associator :
    VSIso F ((A × B) × C) (A × (B × C)) :=
    {| vsi_to := assoc_to; vsi_from := assoc_from;
       vsi_fromto := assoc_fromto; vsi_tofrom := assoc_tofrom |}.

End Associator.


Section SoA.
  Context {F X A B : Type} `{FF : Field.Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VXAB : VectorSpace F (X → A × B) := PW_VectorSpace.
  Let VXA : VectorSpace F (X → A) := PW_VectorSpace.
  Let VXB : VectorSpace F (X → B) := PW_VectorSpace.
  Let VCols : VectorSpace F ((X → A) × (X → B)) :=
    Prod_VectorSpace.

  #[global] Instance soa_split_proper : Proper (equ ==> equ)
    (fun f : X → A × B ⇒ (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x))).
  Proof.
    intros f g Hfg.
    split.
    - intros x.
      destruct (Hfg x) as [H1 _].
      exact H1.
    - intros x.
      destruct (Hfg x) as [_ H2].
      exact H2.
  Qed.

  Lemma soa_split_add : ∀ f g : X → A × B,
    (fun x : X ⇒ fst ((f + g) x), fun x : X ⇒ snd ((f + g) x))
      == (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x))
         + (fun x : X ⇒ fst (g x), fun x : X ⇒ snd (g x)).
  Proof.
    intros f g.
    split.
    - intros x.
      reflexivity.
    - intros x.
      reflexivity.
  Qed.

  Lemma soa_split_scale : ∀ (a : F) (f : X → A × B),
    (fun x : X ⇒ fst ((a *: f) x), fun x : X ⇒ snd ((a *: f) x))
      == a *: (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x)).
  Proof.
    intros a f.
    split.
    - intros x.
      reflexivity.
    - intros x.
      reflexivity.
  Qed.

  Definition soa_split : LinearMap F (X → A × B) ((X → A) × (X → B)) :=
    {| lm_fun := (fun f : X → A × B ⇒ (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x)));
       linear_map_proper := soa_split_proper;
       linear_map_additive := soa_split_add;
       linear_map_scale := soa_split_scale |}.

  #[global] Instance soa_merge_proper : Proper (equ ==> equ)
    (fun (p : (X → A) × (X → B)) (x : X) ⇒ (fst p x, snd p x)).
  Proof.
    intros p q [H1 H2] x.
    split.
    - exact (H1 x).
    - exact (H2 x).
  Qed.

  Lemma soa_merge_add : ∀ p q : (X → A) × (X → B),
    (fun x : X ⇒ (fst (p + q) x, snd (p + q) x))
      == (fun x : X ⇒ (fst p x, snd p x)) + (fun x : X ⇒ (fst q x, snd q x)).
  Proof.
    intros p q x.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Lemma soa_merge_scale : ∀ (a : F) (p : (X → A) × (X → B)),
    (fun x : X ⇒ (fst (a *: p) x, snd (a *: p) x))
      == a *: (fun x : X ⇒ (fst p x, snd p x)).
  Proof.
    intros a p x.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Definition soa_merge : LinearMap F ((X → A) × (X → B)) (X → A × B) :=
    {| lm_fun := (fun (p : (X → A) × (X → B)) (x : X) ⇒ (fst p x, snd p x));
       linear_map_proper := soa_merge_proper;
       linear_map_additive := soa_merge_add;
       linear_map_scale := soa_merge_scale |}.

  Lemma soa_fromto : ∀ f : X → A × B,
    (fun x : X ⇒ (fst (f x), snd (f x))) == f.
  Proof.
    intros f x.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Lemma soa_tofrom : ∀ p : (X → A) × (X → B),
    ((fun x : X ⇒ fst p x), (fun x : X ⇒ snd p x)) == p.
  Proof.
    intros p.
    split.
    - intros x.
      reflexivity.
    - intros x.
      reflexivity.
  Qed.

  Definition vsiso_soa :
    VSIso F (X → A × B) ((X → A) × (X → B)) :=
    {| vsi_to := soa_split; vsi_from := soa_merge;
       vsi_fromto := soa_fromto; vsi_tofrom := soa_tofrom |}.

End SoA.


Section Currying.
  Context {F U V W : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Local Notation HomVW := (LinearMap F V W).
  Local Notation Bilin := (BilinearMap F U V W).

  Let HVW : VectorSpace F HomVW := Hom_VectorSpace.
  Let HomUVW : VectorSpace F (LinearMap F U HomVW) := Hom_VectorSpace.
  Let VBilin : VectorSpace F Bilin := Bilin_VectorSpace.

  #[global] Instance curry_lm_proper : Proper (equ ==> equ) (curry (F:=F) (U:=U) (V:=V) (W:=W)).
  Proof.
    intros b b' Hbb' u v.
    exact (Hbb' u v).
  Qed.

  Lemma curry_lm_add : ∀ b b' : Bilin, curry (b + b') == curry b + curry b'.
  Proof.
    intros b b' u v.
    reflexivity.
  Qed.

  Lemma curry_lm_scale : ∀ (a : F) (b : Bilin), curry (a *: b) == a *: curry b.
  Proof.
    intros a b u v.
    reflexivity.
  Qed.

  Definition curry_lm : LinearMap F Bilin (LinearMap F U HomVW) :=
    {| lm_fun := curry;
       linear_map_proper := curry_lm_proper;
       linear_map_additive := curry_lm_add;
       linear_map_scale := curry_lm_scale |}.

  #[global] Instance uncurry_lm_proper : Proper (equ ==> equ) (uncurry (F:=F) (U:=U) (V:=V) (W:=W)).
  Proof.
    intros g g' Hgg' u v.
    exact (Hgg' u v).
  Qed.

  Lemma uncurry_lm_add : ∀ g g' : LinearMap F U HomVW,
    uncurry (g + g') == uncurry g + uncurry g'.
  Proof.
    intros g g' u v.
    reflexivity.
  Qed.

  Lemma uncurry_lm_scale : ∀ (a : F) (g : LinearMap F U HomVW),
    uncurry (a *: g) == a *: uncurry g.
  Proof.
    intros a g u v.
    reflexivity.
  Qed.

  Definition uncurry_lm : LinearMap F (LinearMap F U HomVW) Bilin :=
    {| lm_fun := uncurry;
       linear_map_proper := uncurry_lm_proper;
       linear_map_additive := uncurry_lm_add;
       linear_map_scale := uncurry_lm_scale |}.

  Definition vsiso_curry : VSIso F Bilin (LinearMap F U HomVW) :=
    {| vsi_to := curry_lm; vsi_from := uncurry_lm;
       vsi_fromto := uncurry_curry; vsi_tofrom := curry_uncurry |}.

End Currying.

Example vsiso_curry_check (F U V W : Type) `{FF : Field.Field F}
  {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {HU : VectorSpace F U}
  {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {HV : VectorSpace F V}
  {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W} {HW : VectorSpace F W} :=
  vsiso_curry (F:=F) (U:=U) (V:=V) (W:=W).


Section Sym.
  Context {F U V : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Definition vsiso_sym (i : VSIso F U V) : VSIso F V U :=
    {| vsi_to := vsi_from i; vsi_from := vsi_to i;
       vsi_fromto := vsi_tofrom i; vsi_tofrom := vsi_fromto i |}.

End Sym.

Section Trans.
  Context {F U V W : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Lemma vsiso_trans_fromto : ∀ (i1 : VSIso F U V) (i2 : VSIso F V W) (u : U),
    vsi_from i1 (vsi_from i2 (vsi_to i2 (vsi_to i1 u))) == u.
  Proof.
    intros i1 i2 u.
    calc (vsi_from i1 (vsi_from i2 (vsi_to i2 (vsi_to i1 u))))
      == (vsi_from i1 (vsi_to i1 u))
           { (linear_map_proper (vsi_from i1) _ _ (vsi_fromto i2 (vsi_to i1 u))) }
      == u { (vsi_fromto i1 u) }.
  Qed.

  Lemma vsiso_trans_tofrom : ∀ (i1 : VSIso F U V) (i2 : VSIso F V W) (w : W),
    vsi_to i2 (vsi_to i1 (vsi_from i1 (vsi_from i2 w))) == w.
  Proof.
    intros i1 i2 w.
    calc (vsi_to i2 (vsi_to i1 (vsi_from i1 (vsi_from i2 w))))
      == (vsi_to i2 (vsi_from i2 w))
           { (linear_map_proper (vsi_to i2) _ _ (vsi_tofrom i1 (vsi_from i2 w))) }
      == w { (vsi_tofrom i2 w) }.
  Qed.

  Definition vsiso_trans (i1 : VSIso F U V) (i2 : VSIso F V W) : VSIso F U W :=
    {| vsi_to := linear_map_comp (vsi_to i2) (vsi_to i1);
       vsi_from := linear_map_comp (vsi_from i1) (vsi_from i2);
       vsi_fromto := vsiso_trans_fromto i1 i2;
       vsi_tofrom := vsiso_trans_tofrom i1 i2 |}.

End Trans.

      {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {VU : VectorSpace F U}
      {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {VV : VectorSpace F V}

      {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {VU : VectorSpace F U}

      (i : Iso (vectObj F U) (vectObj F V)) :
      VSIso F U V :=
    {| vsi_to := iso_to i;
       vsi_from := iso_from i;
