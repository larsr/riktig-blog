Section VSIsoLaws.
  Context {F U V : Type} {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F}
    {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {VV : VectorSpace F V}.

  Definition vsiso_from_to (i : VSIso F U V)
    : ∀ u : U, VSIso.from i (VSIso.to i u) == u :=
    Inverse.from_to (f := VSIso.to i) (g := VSIso.from i).

  Definition vsiso_to_from (i : VSIso F U V)
    : ∀ v : V, VSIso.to i (VSIso.from i v) == v :=
    Inverse.to_from (f := VSIso.to i) (g := VSIso.from i).

  Definition vsiso_from_laws (to : LinearMap F U V) (from : LinearMap F V U)
      (fromto : ∀ u : U, from (to u) == u)
      (tofrom : ∀ v : V, to (from v) == v)
    : VSIso F U V :=
    {| VSIso.to := to;
       VSIso.from := from;
       VSIso.inverse :=
         {| Inverse.from_to := fromto;
            Inverse.to_from := tofrom |} |}.
End VSIsoLaws.

Section InverseLinearity.
  Context {F U V : Type} {Fe : EqRel F} {Fz : Zero F} {Fa : HasAdd F}
    {Fn : HasNeg F} {Fo : One F} {Fm : HasMult F} {Fi : HasInv F} {FF : Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {VV : VectorSpace F V}.

  Context (f : LinearMap F U V) (g : V → U)
    (gp : Proper (equ ==> equ) g)
    (fromto : ∀ u : U, g (f u) == u)
    (tofrom : ∀ v : V, f (g v) == v).

  Lemma linear_inverse_additive : ∀ v v' : V,
    g (v + v') == g v + g v'.
  Proof.
    intros v v'.
    calc (g (v + v'))
      == (g (f (g v) + f (g v')))
           { (gp _ _ (op_proper (f (g v)) v (tofrom v)
                        (f (g v')) v' (tofrom v'))) }
      == (g (f (g v + g v')))
           { (gp _ _ (additive[LinearMap.map f] (g v) (g v'))) }
      == (g v + g v') { fromto }.
  Qed.

  Lemma linear_inverse_homogeneous : ∀ (a : F) (v : V),
    g (a *: v) == a *: g v.
  Proof.
    intros a v.
    calc (g (a *: v))
      == (g (a *: f (g v)))
           { (gp _ _ (scale_proper a a (reflexivity a)
                        (f (g v)) v (tofrom v))) }
      == (g (f (a *: g v)))
           { (gp _ _ (homogeneous[LinearMap.map f] a (g v))) }
      == (a *: g v) { fromto }.
  Qed.

  Theorem linear_inverse_linear : Linear F V U g.
  Proof.
    exact {| Linear.proper := gp;
             Linear.is_additive := linear_inverse_additive;
             Linear.is_homogeneous := linear_inverse_homogeneous |}.
  Qed.

  Definition vsiso_from_linear_inverse : VSIso F U V :=
    vsiso_from_laws f
      (linear_map_from_laws g gp
         linear_inverse_additive linear_inverse_homogeneous)
      fromto tofrom.

End InverseLinearity.

Example vsiso_from_linear_inverse_check (F U : Type) `{FF : Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {HU : VectorSpace F U} :
  VSIso F U U :=
  vsiso_from_linear_inverse linear_map_id (fun u : U ⇒ u)
    (fun u u' Huu' ⇒ Huu')
    (fun u : U ⇒ reflexivity u) (fun u : U ⇒ reflexivity u).

Section Unitor.
  Context {F A : Type} `{VA : VectorSpace F A}.

  Let VAU : VectorSpace F (A × unit) := Prod_VectorSpace.

  #[export] Instance unitor_to_proper : Proper (equ ==> equ) (fun p : A × unit ⇒ fst p).
  Proof.
    intros p q [H1 _].
    exact H1.
  Qed.

  Lemma unitor_to_add : ∀ p q : A × unit, fst (p + q) == fst p + fst q.
  Proof.
    intros p q.
    reflexivity.
  Qed.

  Lemma unitor_to_scale : ∀ (a : F) (p : A × unit),
    fst (a *: p) == a *: fst p.
  Proof.
    intros a p.
    reflexivity.
  Qed.

  Definition unitor_to : LinearMap F (A × unit) A :=
    linear_map_from_laws
      (fun p : A × unit ⇒ fst p)
      unitor_to_proper
      unitor_to_add
      unitor_to_scale.

  #[export] Instance unitor_from_proper : Proper (equ ==> equ) (fun a : A ⇒ (a, tt)).
  Proof.
    intros a b Hab.
    split.
    - exact Hab.
    - exact I.
  Qed.

  Lemma unitor_from_add : ∀ a b : A, (a + b, tt) == (a, tt) + (b, tt).
  Proof.
    intros a b.
    split.
    - reflexivity.
    - exact I.
  Qed.

  Lemma unitor_from_scale : ∀ (a : F) (u : A),
    (a *: u, tt) == a *: (u, tt).
  Proof.
    intros a u.
    split.
    - reflexivity.
    - exact I.
  Qed.

  Definition unitor_from : LinearMap F A (A × unit) :=
    linear_map_from_laws
      (fun a : A ⇒ (a, tt))
      unitor_from_proper
      unitor_from_add
      unitor_from_scale.

  Lemma unitor_fromto : ∀ p : A × unit, (fst p, tt) == p.
  Proof.
    intros p.
    split.
    - reflexivity.
    - exact I.
  Qed.

  Lemma unitor_tofrom : ∀ a : A, fst (a, tt) == a.
  Proof.
    intros a.
    reflexivity.
  Qed.

  Definition vsiso_unitor : VSIso F (A × unit) A :=
    vsiso_from_laws unitor_to unitor_from unitor_fromto unitor_tofrom.

End Unitor.


Section Swap.
  Context {F A B : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VBA : VectorSpace F (B × A) := Prod_VectorSpace.

  #[export] Instance swap_proper : Proper (equ ==> equ) (fun p : A × B ⇒ (snd p, fst p)).
  Proof.
    intros p q [H1 H2].
    split.
    - exact H2.
    - exact H1.
  Qed.

  Lemma swap_add : ∀ p q : A × B,
    (snd (p + q), fst (p + q)) == (snd p, fst p) + (snd q, fst q).
  Proof.
    intros p q.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Lemma swap_scale : ∀ (a : F) (p : A × B),
    (snd (a *: p), fst (a *: p)) == a *: (snd p, fst p).
  Proof.
    intros a p.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Definition swap_lm : LinearMap F (A × B) (B × A) :=
    linear_map_from_laws
      (fun p : A × B ⇒ (snd p, fst p))
      swap_proper
      swap_add
      swap_scale.

  Lemma prod_eta : ∀ p : A × B, (fst p, snd p) == p.
  Proof.
    intros p.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

End Swap.

Section Associator.
  Context {F A B C : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.
  Context {eC : EqRel C} {zC : Zero C} {aC : HasAdd C} {nC : HasNeg C} {sC : HasScale F C}.
  Context {VC : VectorSpace F C}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VBC : VectorSpace F (B × C) := Prod_VectorSpace.
  Let VAB_C : VectorSpace F ((A × B) × C) := Prod_VectorSpace.
  Let VA_BC : VectorSpace F (A × (B × C)) := Prod_VectorSpace.

  #[export] Instance assoc_to_proper : Proper (equ ==> equ)
    (fun p : (A × B) × C ⇒ (fst (fst p), (snd (fst p), snd p))).
  Proof.
    intros p q [[H1 H2] H3].
    split.
    - exact H1.
    - split.
      + exact H2.
      + exact H3.
  Qed.

  Lemma assoc_to_add : ∀ p q : (A × B) × C,
    (fst (fst (p + q)), (snd (fst (p + q)), snd (p + q)))
      == (fst (fst p), (snd (fst p), snd p))
         + (fst (fst q), (snd (fst q), snd q)).
  Proof.
    intros p q.
    split.
    - reflexivity.
    - split.
      + reflexivity.
      + reflexivity.
  Qed.

  Lemma assoc_to_scale : ∀ (a : F) (p : (A × B) × C),
    (fst (fst (a *: p)), (snd (fst (a *: p)), snd (a *: p)))
      == a *: (fst (fst p), (snd (fst p), snd p)).
  Proof.
    intros a p.
    split.
    - reflexivity.
    - split.
      + reflexivity.
      + reflexivity.
  Qed.

  Definition assoc_to : LinearMap F ((A × B) × C) (A × (B × C)) :=
    linear_map_from_laws
      (fun p : (A × B) × C ⇒ (fst (fst p), (snd (fst p), snd p)))
      assoc_to_proper
      assoc_to_add
      assoc_to_scale.

  #[export] Instance assoc_from_proper : Proper (equ ==> equ)
    (fun q : A × (B × C) ⇒ ((fst q, fst (snd q)), snd (snd q))).
  Proof.
    intros p q [H1 [H2 H3]].
    split.
    - split.
      + exact H1.
      + exact H2.
    - exact H3.
  Qed.

  Lemma assoc_from_add : ∀ p q : A × (B × C),
    ((fst (p + q), fst (snd (p + q))), snd (snd (p + q)))
      == ((fst p, fst (snd p)), snd (snd p))
         + ((fst q, fst (snd q)), snd (snd q)).
  Proof.
    intros p q.
    split.
    - split.
      + reflexivity.
      + reflexivity.
    - reflexivity.
  Qed.

  Lemma assoc_from_scale : ∀ (a : F) (q : A × (B × C)),
    ((fst (a *: q), fst (snd (a *: q))), snd (snd (a *: q)))
      == a *: ((fst q, fst (snd q)), snd (snd q)).
  Proof.
    intros a q.
    split.
    - split.
      + reflexivity.
      + reflexivity.
    - reflexivity.
  Qed.

  Definition assoc_from : LinearMap F (A × (B × C)) ((A × B) × C) :=
    linear_map_from_laws
      (fun q : A × (B × C) ⇒ ((fst q, fst (snd q)), snd (snd q)))
      assoc_from_proper
      assoc_from_add
      assoc_from_scale.

  Lemma assoc_fromto : ∀ p : (A × B) × C,
    ((fst (fst p), snd (fst p)), snd p) == p.
  Proof.
    intros p.
    split.
    - split.
      + reflexivity.
      + reflexivity.
    - reflexivity.
  Qed.

  Lemma assoc_tofrom : ∀ q : A × (B × C),
    (fst q, (fst (snd q), snd (snd q))) == q.
  Proof.
    intros q.
    split.
    - reflexivity.
    - split.
      + reflexivity.
      + reflexivity.
  Qed.

  Definition vsiso_associator :
    VSIso F ((A × B) × C) (A × (B × C)) :=
    vsiso_from_laws assoc_to assoc_from assoc_fromto assoc_tofrom.

End Associator.


Section SoA.
  Context {F X A B : Type} `{FF : Field F}.
  Context {eA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A} {sA : HasScale F A}.
  Context {VA : VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B}.

  Let VAB : VectorSpace F (A × B) := Prod_VectorSpace.
  Let VXAB : VectorSpace F (X → A × B) := PW_VectorSpace.
  Let VXA : VectorSpace F (X → A) := PW_VectorSpace.
  Let VXB : VectorSpace F (X → B) := PW_VectorSpace.
  Let VCols : VectorSpace F ((X → A) × (X → B)) :=
    Prod_VectorSpace.

  #[export] Instance soa_split_proper : Proper (equ ==> equ)
    (fun f : X → A × B ⇒ (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x))).
  Proof.
    intros f g Hfg.
    split.
    - intros x.
      destruct (Hfg x) as [H1 _].
      exact H1.
    - intros x.
      destruct (Hfg x) as [_ H2].
      exact H2.
  Qed.

  Lemma soa_split_add : ∀ f g : X → A × B,
    (fun x : X ⇒ fst ((f + g) x), fun x : X ⇒ snd ((f + g) x))
      == (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x))
         + (fun x : X ⇒ fst (g x), fun x : X ⇒ snd (g x)).
  Proof.
    intros f g.
    split.
    - intros x.
      reflexivity.
    - intros x.
      reflexivity.
  Qed.

  Lemma soa_split_scale : ∀ (a : F) (f : X → A × B),
    (fun x : X ⇒ fst ((a *: f) x), fun x : X ⇒ snd ((a *: f) x))
      == a *: (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x)).
  Proof.
    intros a f.
    split.
    - intros x.
      reflexivity.
    - intros x.
      reflexivity.
  Qed.

  Definition soa_split : LinearMap F (X → A × B) ((X → A) × (X → B)) :=
    linear_map_from_laws
      (fun f : X → A × B ⇒ (fun x : X ⇒ fst (f x), fun x : X ⇒ snd (f x)))
      soa_split_proper
      soa_split_add
      soa_split_scale.

  #[export] Instance soa_merge_proper : Proper (equ ==> equ)
    (fun (p : (X → A) × (X → B)) (x : X) ⇒ (fst p x, snd p x)).
  Proof.
    intros p q [H1 H2] x.
    split.
    - exact (H1 x).
    - exact (H2 x).
  Qed.

  Lemma soa_merge_add : ∀ p q : (X → A) × (X → B),
    (fun x : X ⇒ (fst (p + q) x, snd (p + q) x))
      == (fun x : X ⇒ (fst p x, snd p x)) + (fun x : X ⇒ (fst q x, snd q x)).
  Proof.
    intros p q x.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Lemma soa_merge_scale : ∀ (a : F) (p : (X → A) × (X → B)),
    (fun x : X ⇒ (fst (a *: p) x, snd (a *: p) x))
      == a *: (fun x : X ⇒ (fst p x, snd p x)).
  Proof.
    intros a p x.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Definition soa_merge : LinearMap F ((X → A) × (X → B)) (X → A × B) :=
    linear_map_from_laws
      (fun (p : (X → A) × (X → B)) (x : X) ⇒ (fst p x, snd p x))
      soa_merge_proper
      soa_merge_add
      soa_merge_scale.

  Lemma soa_fromto : ∀ f : X → A × B,
    (fun x : X ⇒ (fst (f x), snd (f x))) == f.
  Proof.
    intros f x.
    split.
    - reflexivity.
    - reflexivity.
  Qed.

  Lemma soa_tofrom : ∀ p : (X → A) × (X → B),
    ((fun x : X ⇒ fst p x), (fun x : X ⇒ snd p x)) == p.
  Proof.
    intros p.
    split.
    - intros x.
      reflexivity.
    - intros x.
      reflexivity.
  Qed.

  Definition vsiso_soa :
    VSIso F (X → A × B) ((X → A) × (X → B)) :=
    vsiso_from_laws soa_split soa_merge soa_fromto soa_tofrom.

End SoA.

Section Currying.
  Context {F U V W : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Local Notation HomVW := (LinearMap F V W).
  Local Notation Bilin := (BilinearMap F U V W).

  Let HVW : VectorSpace F HomVW := Hom_VectorSpace.
  Let HomUVW : VectorSpace F (LinearMap F U HomVW) := Hom_VectorSpace.
  Let VBilin : VectorSpace F Bilin := Bilin_VectorSpace.

  #[export] Instance curry_lm_proper : Proper (equ ==> equ) (curry (F:=F) (U:=U) (V:=V) (W:=W)).
  Proof.
    intros b b' Hbb' u v.
    exact (Hbb' u v).
  Qed.

  Lemma curry_lm_add : ∀ b b' : Bilin, curry (b + b') == curry b + curry b'.
  Proof.
    intros b b' u v.
    reflexivity.
  Qed.

  Lemma curry_lm_scale : ∀ (a : F) (b : Bilin), curry (a *: b) == a *: curry b.
  Proof.
    intros a b u v.
    reflexivity.
  Qed.

  Definition curry_lm : LinearMap F Bilin (LinearMap F U HomVW) :=
    linear_map_from_laws
      curry
      curry_lm_proper
      curry_lm_add
      curry_lm_scale.

  #[export] Instance uncurry_lm_proper : Proper (equ ==> equ) (uncurry (F:=F) (U:=U) (V:=V) (W:=W)).
  Proof.
    intros g g' Hgg' u v.
    exact (Hgg' u v).
  Qed.

  Lemma uncurry_lm_add : ∀ g g' : LinearMap F U HomVW,
    uncurry (g + g') == uncurry g + uncurry g'.
  Proof.
    intros g g' u v.
    reflexivity.
  Qed.

  Lemma uncurry_lm_scale : ∀ (a : F) (g : LinearMap F U HomVW),
    uncurry (a *: g) == a *: uncurry g.
  Proof.
    intros a g u v.
    reflexivity.
  Qed.

  Definition uncurry_lm : LinearMap F (LinearMap F U HomVW) Bilin :=
    linear_map_from_laws
      uncurry
      uncurry_lm_proper
      uncurry_lm_add
      uncurry_lm_scale.

  Definition vsiso_curry : VSIso F Bilin (LinearMap F U HomVW) :=
    vsiso_from_laws curry_lm uncurry_lm uncurry_curry curry_uncurry.

End Currying.

Example vsiso_curry_check (F U V W : Type) `{FF : Field F}
  {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U} {HU : VectorSpace F U}
  {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V} {HV : VectorSpace F V}
  {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W} {HW : VectorSpace F W} :=

Section Sym.

  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.

Section Trans.
  Context {F U V W : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Lemma vsiso_trans_fromto : ∀ (i1 : VSIso F U V) (i2 : VSIso F V W) (u : U),
    VSIso.from i1 (VSIso.from i2 (VSIso.to i2 (VSIso.to i1 u))) == u.
  Proof.
    intros i1 i2 u.
    calc (VSIso.from i1 (VSIso.from i2 (VSIso.to i2 (VSIso.to i1 u))))
      == (VSIso.from i1 (VSIso.to i1 u))
           { (linear_map_proper (VSIso.from i1) _ _ (vsiso_from_to i2 (VSIso.to i1 u))) }
      == u { vsiso_from_to }.
  Qed.

  Lemma vsiso_trans_tofrom : ∀ (i1 : VSIso F U V) (i2 : VSIso F V W) (w : W),
    VSIso.to i2 (VSIso.to i1 (VSIso.from i1 (VSIso.from i2 w))) == w.
  Proof.
    intros i1 i2 w.
    calc (VSIso.to i2 (VSIso.to i1 (VSIso.from i1 (VSIso.from i2 w))))
      == (VSIso.to i2 (VSIso.from i2 w))
           { (linear_map_proper (VSIso.to i2) _ _ (vsiso_to_from i1 (VSIso.from i2 w))) }
      == w { vsiso_to_from }.
  Qed.

  Definition vsiso_trans (i1 : VSIso F U V) (i2 : VSIso F V W) : VSIso F U W :=
    vsiso_from_laws
      (linear_map_comp (VSIso.to i2) (VSIso.to i1))
      (linear_map_comp (VSIso.from i1) (VSIso.from i2))
      (vsiso_trans_fromto i1 i2)
      (vsiso_trans_tofrom i1 i2).

End Trans.
