Section NormedSpaceTheory.
  Context {F V : Type} `{NormedSpace F V}.

  Theorem norm_zero : ‖ (0 : V) ‖ == 0.
  Proof.
    calc (‖ (0 : V) ‖)
      == (‖ (0 : F) *: (0 : V) ‖) { zero_smul }
      == (abs (0 : F) × ‖ (0 : V) ‖) { norm_smul }
      == ((0 : F) × ‖ (0 : V) ‖) { abs_zero }
      == (0 : F) { zero_mul }.
  Qed.

  Theorem norm_neg : ∀ v : V, ‖ - v ‖ == ‖ v ‖.
  Proof.
    intros v.
    calc (‖ - v ‖)
      == (‖ - (1 : F) *: v ‖) { neg_one_smul }
      == (abs (- (1 : F)) × ‖ v ‖) { norm_smul }
      == (abs (1 : F) × ‖ v ‖) { abs_neg }
      == ((1 : F) × ‖ v ‖) { (abs_of_nonneg (1 : F) zero_le_one) }
      == (‖ v ‖) { left_identity }.
  Qed.

  Theorem norm_sub_rev : ∀ u v : V, ‖ u + - v ‖ == ‖ v + - u ‖.
  Proof.
    intros u v.
    calc (‖ u + - v ‖)
      == (‖ - (u + - v) ‖) { norm_neg }
      == (‖ - u + - - v ‖) { neg_add }
      == (‖ - u + v ‖) { neg_neg }
      == (‖ v + - u ‖) { (commutative (op := add) (- u) v) }.
  Qed.

The reverse triangle inequality.
  Theorem norm_sub_norm_le : ∀ u v : V, ‖ u ‖ + - ‖ v ‖ ≤ ‖ u + - v ‖.
  Proof.
    intros u v.
    pose proof (norm_add_le (u + - v) v) as Hsum.
    setoid_rewrite (neg_add_cancel_right u v) in Hsum.
    pose proof (add_le_mono_r _ _ (- ‖ v ‖) Hsum) as Hsub.
    setoid_rewrite (add_neg_cancel_right (‖ u + - v ‖) (‖ v ‖)) in Hsub.
    exact Hsub.
  Qed.
End NormedSpaceTheory.
