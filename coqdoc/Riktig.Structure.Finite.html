From Riktig.Structure Require Export NatOrder.

A finite index type: its elements are numbered below a count, and numbering an element and taking the element at a number cancel.
Module Finite.
  Class Finite (I N : Type) `{NatOrder N} {eqI : EqRel I}
      (card : HasCard I N) (index : HasIndex I N) (elem : HasElem I N) := {
    equivalence :: Equivalence eqI;
    index_proper :: Proper (equ ==> equ) index;
    elem_proper :: Proper (equ ==> equ) elem;
    index_lt : ∀ i : I, index i < card;
    elem_index : ∀ i : I, elem (index i) == i;
    index_elem : ∀ k : N, k < card → index (elem k) == k;
  }.
  Hint Mode Finite ! - - - - - - - - - - - - - - : typeclass_instances.
End Finite.
Export Finite.
