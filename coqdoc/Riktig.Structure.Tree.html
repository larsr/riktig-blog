From Riktig.Structure Require Export Adt.

The binary trees with leaves named by x: the datatype of the signature const x + self × self, one level being a leaf or two holes.
Module Tree.
  Notation shape den x := (shape_of den (treeS (S := sig _) x)).

  Class Tree (O : Type) `{Bicartesian O} {K : Type}
      (den : K → O) (x : K) (T : O)
      (build : HasBuild O (shape den x) T)
      (fold : HasCata O (shape den x) T) := {
    initial :: InitialAlgebra O (shape den x) T build fold;
  }.

  Notation tip := (build (F := shape _ _) ∘ inj1).
  Notation fork := (build (F := shape _ _) ∘ inj2).
  Notation fold := (cata (F := shape _ _)).
  Notation map := (fmap (F := shape _ _)).
End Tree.
