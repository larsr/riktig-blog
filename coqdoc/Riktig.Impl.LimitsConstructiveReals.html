The constructive reals as a space over the rationals, with limits by modulus: convergent when every positive rational precision has an index from which on the terms are that close to the limit, Cauchy when the terms are that close to each other; and complete, the limit being what the structure computes from the Cauchy evidence.
Section ConstructiveRealsLimits.
  Context {R Q N : Type} `{ConstructiveReals R Q N}.

  #[export] Instance CR_Scale : HasScale Q R := RingHom_Scale cast.
  #[export] Instance CR_VectorSpace : VectorSpace Q R 0 add neg CR_Scale := RingHom_VectorSpace cast.

  #[export] Instance CR_Tendsto : HasTendsto N R :=
    fun u l ⇒ ∀ eps : Q, 0 < eps → {n : N & ∀ i : N, n ≤ i → abs (u i + - l) ≼ cast eps}.
  #[export] Instance CR_Cauchy : HasCauchy N R :=
    fun u ⇒ ∀ eps : Q, 0 < eps →
      {n : N & ∀ i j : N, n ≤ i → n ≤ j → abs (u i + - u j) ≼ cast eps}.

  #[export] Instance CR_Limits : Limits Q N R CR_Tendsto CR_Cauchy.
  Proof.
    split.
    - exact converges_proper.
    - exact converges_unique.
    - exact converges_const.
    - exact converges_add.
    - exact converges_neg.
    - exact converges_smul.
    - exact converges_tail.
    - exact cauchy_proper.
    - exact cauchy_of_converges.
    - exact cauchy_add.
    - exact cauchy_neg.
    - exact cauchy_smul.
    - exact cauchy_tail.
  Defined.

  #[export] Instance CR_Lim : HasLim N R := fun u Hu ⇒ projT1 (complete u Hu).

  #[export] Instance CR_Complete : Complete Q N R CR_Lim.
  Proof. split. exact (fun u Hu ⇒ projT2 (complete u Hu)). Defined.
End ConstructiveRealsLimits.
