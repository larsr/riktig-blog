In a graded geometric algebra the grade-0 part of a product of two vectors is their inner product, and the grade-2 part is their wedge: the product of two vectors is u ⋅ v + u ∧ v, grade by grade.
Section GradedGATheory.
  Context {N F V A : Type} `{GradedGA N F V A}.

  Theorem scalar_grade_zero : ∀ a : F, ⟨a *: (1 : A)⟩_0 == a *: 1.
  Proof.
    intros a.
    calc (⟨a *: (1 : A)⟩_0) == (a *: ⟨(1 : A)⟩_0) { (grade_smul 0 a (1 : A)) } == (a *: (1 : A)) { one_grade_zero }.
  Qed.

  Theorem reverse_grade_zero : ∀ x : A, reverse ⟨x⟩_0 == ⟨x⟩_0.
  Proof.
    intros x.
    calc (reverse ⟨x⟩_0)
      == (reverse (scalar_part x *: (1 : A))) { (reverse_proper _ _ (grade_zero_scalar x)) }
      == (scalar_part x *: reverse (1 : A)) { (homogeneous (f := reverse) (scalar_part x) (1 : A)) }
      == (scalar_part x *: (1 : A)) { (scale_proper _ _ (reflexivity _) _ _ reverse_one) }
      == (⟨x⟩_0) { (symmetry (grade_zero_scalar x)) }.
  Qed.

  Theorem grade_zero_reverse : ∀ x : A, ⟨reverse x⟩_0 == ⟨x⟩_0.
  Proof.
    intros x.
    calc (⟨reverse x⟩_0) == (reverse ⟨x⟩_0) { (symmetry (reverse_grade 0 x)) } == (⟨x⟩_0) { (reverse_grade_zero x) }.
  Qed.

  Theorem reverse_vec_mul : ∀ u v : V, reverse (vec u × vec v) == vec v × vec u.
  Proof.
    intros u v.
    calc (reverse (vec u × vec v))
      == (reverse (vec v) × reverse (vec u)) { (reverse_anti (vec u) (vec v)) }
      == (vec v × vec u) { (op_proper _ _ (reverse_vec v) _ _ (reverse_vec u)) }.
  Qed.

Pure-grade components: a scaled grade-k part stays pure.

  Lemma grade_pure_smul : ∀ (k : N) (c : F) (y : A), c *: ⟨y⟩_k == ⟨c *: ⟨y⟩_k⟩_k.
  Proof.
    intros k c y.
    calc (c *: ⟨y⟩_k)
      == (c *: ⟨⟨y⟩_k⟩_k) { (scale_proper _ _ (reflexivity c) _ _ (symmetry (grade_idempotent k y))) }
      == (⟨c *: ⟨y⟩_k⟩_k) { (symmetry (grade_smul k c ⟨y⟩_k)) }.
  Qed.

  Lemma grade_pure_orth : ∀ (k l : N) (c : F) (y : A), k =/= l → ⟨c *: ⟨y⟩_l⟩_k == 0.
  Proof.
    intros k l c y Hkl.
    calc (⟨c *: ⟨y⟩_l⟩_k)
      == (c *: ⟨⟨y⟩_l⟩_k) { (grade_smul k c ⟨y⟩_l) }
      == (c *: (0 : A)) { (scale_proper _ _ (reflexivity c) _ _ (grade_orthogonal k l y Hkl)) }
      == (0 : A) { (smul_zero c) }.
  Qed.

  Local Notation "'low' k b" := (natrec A 0 (fun (j : N) (_ : A) ⇒ ⟨b⟩_j) k) (at level 10, k at level 9, b at level 9).

A vector times a grade-k part splits into the part one below and the part one above -- uniformly in k, the one below read through low.
  Lemma vec_mul_grade_low : ∀ (u : V) (y : A) (k : N),
    vec u × ⟨y⟩_k == low k (vec u × ⟨y⟩_k) + ⟨vec u × ⟨y⟩_k⟩_(succ k).
  Proof.
    intros u y k. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - assert (Hy : ⟨y⟩_k == ⟨y⟩_0) by exact (grade_index_proper k 0 y Hk).
      assert (HX : vec u × ⟨y⟩_k == vec u × ⟨y⟩_0) by exact (op_proper _ _ (reflexivity (vec u)) _ _ Hy).
      assert (Hlow : low k (vec u × ⟨y⟩_k) == 0).
      { apply (transitivity (low_index_proper k 0 _ Hk)). exact (low_zero _). }
      assert (Hup : ⟨vec u × ⟨y⟩_k⟩_(succ k) == ⟨vec u × ⟨y⟩_0⟩_(succ 0)).
      { apply (transitivity (grade_index_proper (succ k) (succ 0) _ (succ_proper _ _ Hk))).
        exact (grade_proper (succ 0) _ _ HX). }
      calc (vec u × ⟨y⟩_k)
        == (vec u × ⟨y⟩_0) { HX }
        == (⟨vec u × ⟨y⟩_0⟩_(succ 0)) { (vec_mul_grade_zero u y) }
        == (0 + ⟨vec u × ⟨y⟩_0⟩_(succ 0)) { (symmetry (left_identity (action := add) _)) }
        == (low k (vec u × ⟨y⟩_k) + ⟨vec u × ⟨y⟩_0⟩_(succ 0)) { (op_proper _ _ (symmetry Hlow) _ _ (reflexivity _)) }
        == (low k (vec u × ⟨y⟩_k) + ⟨vec u × ⟨y⟩_k⟩_(succ k)) { (op_proper _ _ (reflexivity _) _ _ (symmetry Hup)) }.
    - assert (Hy : ⟨y⟩_k == ⟨y⟩_(succ m)) by exact (grade_index_proper k (succ m) y Hk).
      assert (HX : vec u × ⟨y⟩_k == vec u × ⟨y⟩_(succ m)) by exact (op_proper _ _ (reflexivity (vec u)) _ _ Hy).
      assert (Hlow : low k (vec u × ⟨y⟩_k) == ⟨vec u × ⟨y⟩_(succ m)⟩_m).
      { apply (transitivity (low_index_proper k (succ m) _ Hk)).
        apply (transitivity (low_succ m _)). exact (grade_proper m _ _ HX). }
      assert (Hup : ⟨vec u × ⟨y⟩_k⟩_(succ k) == ⟨vec u × ⟨y⟩_(succ m)⟩_(succ (succ m))).
      { apply (transitivity (grade_index_proper (succ k) (succ (succ m)) _ (succ_proper _ _ Hk))).
        exact (grade_proper (succ (succ m)) _ _ HX). }
      calc (vec u × ⟨y⟩_k)
        == (vec u × ⟨y⟩_(succ m)) { HX }
        == (⟨vec u × ⟨y⟩_(succ m)⟩_m + ⟨vec u × ⟨y⟩_(succ m)⟩_(succ (succ m))) { (vec_mul_grade_succ u y m) }
        == (low k (vec u × ⟨y⟩_k) + ⟨vec u × ⟨y⟩_(succ m)⟩_(succ (succ m))) { (op_proper _ _ (symmetry Hlow) _ _ (reflexivity _)) }
        == (low k (vec u × ⟨y⟩_k) + ⟨vec u × ⟨y⟩_k⟩_(succ k)) { (op_proper _ _ (reflexivity _) _ _ (symmetry Hup)) }.
  Qed.

Nothing pure one grade above k survives low k.
  Lemma low_of_pure_above : ∀ (k : N) (c : F) (y : A), low k (c *: ⟨y⟩_(succ k)) == 0.
  Proof.
    intros k c y. destruct (zero_or_succ k) as [Hk | [m Hk]].
    - apply (transitivity (low_index_proper k 0 _ Hk)). exact (low_zero _).
    - apply (transitivity (low_index_proper k (succ m) _ Hk)).
      apply (transitivity (low_succ m _)).
      apply (transitivity (grade_proper m _ _ (scale_proper _ _ (reflexivity c) _ _ (grade_index_proper (succ k) (succ (succ m)) y (succ_proper _ _ Hk))))).
      apply grade_pure_orth. intros Hm. exact (ne_succ_succ m Hm).
  Qed.

  Context (two_ne_zero : (1 + 1 : F) =/= 0).

  Theorem grade_zero_vec_mul : ∀ u v : V, ⟨vec u × vec v⟩_0 == ⟪ u , v ⟫ *: (1 : A).
  Proof.
    intros u v.
    assert (Hrev : ⟨vec v × vec u⟩_0 == ⟨vec u × vec v⟩_0).
    { calc (⟨vec v × vec u⟩_0)
        == (⟨reverse (vec u × vec v)⟩_0) { (grade_proper 0 _ _ (symmetry (reverse_vec_mul u v))) }
        == (⟨vec u × vec v⟩_0) { (grade_zero_reverse (vec u × vec v)) }. }
    calc (⟨vec u × vec v⟩_0)
      == (inv (1 + 1) *: (⟨vec u × vec v⟩_0 + ⟨vec u × vec v⟩_0)) { (symmetry (half_double two_ne_zero _)) }
      == (inv (1 + 1) *: (⟨vec u × vec v⟩_0 + ⟨vec v × vec u⟩_0)) { (scale_proper _ _ (reflexivity _) _ _ (op_proper _ _ (reflexivity _) _ _ (symmetry Hrev))) }
      == (inv (1 + 1) *: ⟨vec u × vec v + vec v × vec u⟩_0) { (scale_proper _ _ (reflexivity _) _ _ (symmetry (grade_add 0 _ _))) }
      == (⟨u ⋅ v⟩_0) { (symmetry (grade_smul 0 _ _)) }
      == (⟨⟪ u , v ⟫ *: (1 : A)⟩_0) { (grade_proper 0 _ _ (dot_eq_inner two_ne_zero u v)) }
      == (⟪ u , v ⟫ *: (1 : A)) { (scalar_grade_zero _) }.
  Qed.

  Theorem grade_two_vec_mul : ∀ u v : V, ⟨vec u × vec v⟩_(succ (succ 0)) == u ∧ v.
  Proof.
    intros u v.
    assert (Hs : vec u × vec v == ⟨vec u × vec v⟩_0 + ⟨vec u × vec v⟩_(succ (succ 0))).
    { pose proof (vec_mul_grade_succ u (vec v) 0) as Hg.
      assert (Hv : vec u × ⟨vec v⟩_(succ 0) == vec u × vec v) by exact (op_proper _ _ (reflexivity _) _ _ (vec_grade_one v)).
      calc (vec u × vec v)
        == (vec u × ⟨vec v⟩_(succ 0)) { (symmetry Hv) }
        == (⟨vec u × ⟨vec v⟩_(succ 0)⟩_0 + ⟨vec u × ⟨vec v⟩_(succ 0)⟩_(succ (succ 0))) { Hg }
        == (⟨vec u × vec v⟩_0 + ⟨vec u × vec v⟩_(succ (succ 0)))
             { (op_proper _ _ (grade_proper 0 _ _ Hv) _ _ (grade_proper (succ (succ 0)) _ _ Hv)) }. }
    calc (⟨vec u × vec v⟩_(succ (succ 0)))
      == (vec u × vec v + - ⟨vec u × vec v⟩_0)
           { (eq_sub_of_add_eq _ _ _ (transitivity (commutative (op := add) _ _) (symmetry Hs))) }
      == (vec u × vec v + - (⟪ u , v ⟫ *: (1 : A))) { (op_proper _ _ (reflexivity _) _ _ (neg_proper _ _ (grade_zero_vec_mul u v))) }
      == (u ∧ v) { (symmetry (wedge_eq two_ne_zero u v)) }.
  Qed.
