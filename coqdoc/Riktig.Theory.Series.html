Series: the partial sums of a sequence by recursion on the index, a sequence being summable when its partial sums are Cauchy, and its sum being their limit. The algebra of sums follows from the algebra of limits.
Section PartialSums.
  Context {N V : Type} `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {AG : AbGroup V 0 add neg}.

  Local Notation psum f n := (natrec V 0 (fun k acc ⇒ acc + f k) n).

  Lemma step_proper : ∀ f : N → V, Proper (equ ==> equ) f →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : V) ⇒ acc + f k).
  Proof. intros f Hf k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | apply Hf; exact Hk ]. Qed.

  Theorem psum_proper : ∀ f : N → V, Proper (equ ==> equ) f → Proper (equ ==> equ) (fun n ⇒ psum f n).
  Proof. intros f Hf. exact (natrec_proper V _ _ 0 _ (step_proper f Hf)). Qed.

  Theorem psum_zero : ∀ f : N → V, psum f 0 == 0.
  Proof. intros f. exact (natrec_zero V _ _ 0 _). Qed.

  Theorem psum_succ : ∀ (f : N → V) (n : N), psum f (succ n) == psum f n + f n.
  Proof. intros f n. exact (natrec_succ V _ _ 0 _ n). Qed.

  Theorem psum_ext : ∀ f g : N → V, Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    (∀ k : N, f k == g k) → ∀ n : N, psum f n == psum g n.
  Proof.
    intros f g Hf Hg Hfg.
    apply (nat_induction (fun n ⇒ psum f n == psum g n)).
    - intros n n' Hn. setoid_rewrite (psum_proper f Hf n n' Hn). setoid_rewrite (psum_proper g Hg n n' Hn). reflexivity.
    - calc (psum f 0) == (0 : V) { (psum_zero f) } == (psum g 0) { (psum_zero g) }.
    - intros n IH.
      calc (psum f (succ n))
        == (psum f n + f n) { (psum_succ f n) }
        == (psum g n + f n) { IH }
        == (psum g n + g n) { (Hfg n) }
        == (psum g (succ n)) { (psum_succ g n) }.
  Qed.

  Theorem psum_add : ∀ f g : N → V, Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    ∀ n : N, psum (fun k ⇒ f k + g k) n == psum f n + psum g n.
  Proof.
    intros f g Hf Hg.
    apply (nat_induction (fun n ⇒ psum (fun k ⇒ f k + g k) n == psum f n + psum g n)).
    - intros n n' Hn.
      assert (Hfg : Proper (equ ==> equ) (fun k ⇒ f k + g k))
        by (intros k k' Hk; apply op_proper; [ apply Hf | apply Hg ]; exact Hk).
      split; intros Heq.
      + setoid_rewrite <- (psum_proper _ Hfg n n' Hn). setoid_rewrite <- (psum_proper f Hf n n' Hn).
        setoid_rewrite <- (psum_proper g Hg n n' Hn). exact Heq.
      + setoid_rewrite (psum_proper _ Hfg n n' Hn). setoid_rewrite (psum_proper f Hf n n' Hn).
        setoid_rewrite (psum_proper g Hg n n' Hn). exact Heq.
    - calc (psum (fun k ⇒ f k + g k) 0)
        == (0 : V) { (psum_zero (fun k ⇒ f k + g k)) }
        == ((0 : V) + 0) { (right_identity (action := add) (0 : V)) }
        == (psum f 0 + 0) { (psum_zero f) }
        == (psum f 0 + psum g 0) { (psum_zero g) }.
    - intros n IH.
      calc (psum (fun k ⇒ f k + g k) (succ n))
        == (psum (fun k ⇒ f k + g k) n + (f n + g n)) { (psum_succ (fun k ⇒ f k + g k) n) }
        == ((psum f n + psum g n) + (f n + g n)) { IH }
        == ((psum f n + f n) + (psum g n + g n)) { (add_add_add_comm (psum f n) (psum g n) (f n) (g n)) }
        == (psum f (succ n) + (psum g n + g n)) { (psum_succ f n) }
        == (psum f (succ n) + psum g (succ n)) { (psum_succ g n) }.
  Qed.

  Theorem psum_neg : ∀ f : N → V, Proper (equ ==> equ) f →
    ∀ n : N, psum (fun k ⇒ - f k) n == - psum f n.
  Proof.
    intros f Hf.
    apply (nat_induction (fun n ⇒ psum (fun k ⇒ - f k) n == - psum f n)).
    - intros n n' Hn.
      assert (Hnf : Proper (equ ==> equ) (fun k ⇒ - f k))
        by (intros k k' Hk; apply neg_proper; apply Hf; exact Hk).
      split; intros Heq.
      + setoid_rewrite <- (psum_proper _ Hnf n n' Hn). setoid_rewrite <- (psum_proper f Hf n n' Hn). exact Heq.
      + setoid_rewrite (psum_proper _ Hnf n n' Hn). setoid_rewrite (psum_proper f Hf n n' Hn). exact Heq.
    - calc (psum (fun k ⇒ - f k) 0)
        == (0 : V) { (psum_zero (fun k ⇒ - f k)) }
        == (- (0 : V)) { neg_zero }
        == (- psum f 0) { (psum_zero f) }.
    - intros n IH.
      calc (psum (fun k ⇒ - f k) (succ n))
        == (psum (fun k ⇒ - f k) n + - f n) { (psum_succ (fun k ⇒ - f k) n) }
        == (- psum f n + - f n) { IH }
        == (- (psum f n + f n)) { (neg_add (psum f n) (f n)) }
        == (- psum f (succ n)) { (psum_succ f n) }.
  Qed.

  Theorem psum_shift : ∀ f : N → V, Proper (equ ==> equ) f → ∀ i c : N,
    psum f (i + c) == psum f i + psum (fun k ⇒ f (i + k)) c.
  Proof.
    intros f Hf i.
    assert (Hg : Proper (equ ==> equ) (fun k ⇒ f (i + k)))
      by (intros k k' Hk; apply Hf; apply op_proper; [ reflexivity | exact Hk ]).
    apply (nat_induction (fun c ⇒ psum f (i + c) == psum f i + psum (fun k ⇒ f (i + k)) c)).
    - intros c c' Hc. split; intros Heq.
      + setoid_rewrite <- (psum_proper f Hf (i + c) (i + c') (op_proper i i (reflexivity i) c c' Hc)).
        setoid_rewrite <- (psum_proper _ Hg c c' Hc). exact Heq.
      + setoid_rewrite (psum_proper f Hf (i + c) (i + c') (op_proper i i (reflexivity i) c c' Hc)).
        setoid_rewrite (psum_proper _ Hg c c' Hc). exact Heq.
    - calc (psum f (i + 0))
        == (psum f i) { (psum_proper f Hf (i + 0) i (Theory.Naturals.add_zero i)) }
        == (psum f i + 0) { (right_identity (action := add) (psum f i)) }
        == (psum f i + psum (fun k ⇒ f (i + k)) 0) { (psum_zero (fun k ⇒ f (i + k))) }.
    - intros c IH.
      calc (psum f (i + succ c))
        == (psum f (succ (i + c))) { (psum_proper f Hf (i + succ c) (succ (i + c)) (Theory.Naturals.add_succ i c)) }
        == (psum f (i + c) + f (i + c)) { (psum_succ f (i + c)) }
        == ((psum f i + psum (fun k ⇒ f (i + k)) c) + f (i + c)) { IH }
        == (psum f i + (psum (fun k ⇒ f (i + k)) c + f (i + c))) { associative }
        == (psum f i + psum (fun k ⇒ f (i + k)) (succ c)) { (psum_succ (fun k ⇒ f (i + k)) c) }.
  Qed.

Peeling the first term instead of the last.
  Theorem psum_succ_l : ∀ f : N → V, Proper (equ ==> equ) f →
    ∀ n : N, psum f (succ n) == f 0 + psum (fun k ⇒ f (succ k)) n.
  Proof.
    intros f Hf n.
    assert (Hs : Proper (equ ==> equ) (fun k : N ⇒ f (1 + k)))
      by (intros k k' Hk; apply Hf; apply op_proper; [ reflexivity | exact Hk ]).
    calc (psum f (succ n))
      == (psum f (1 + n)) { (psum_proper f Hf (succ n) (1 + n) (symmetry (Theory.Naturals.add_one_l n))) }
      == (psum f 1 + psum (fun k ⇒ f (1 + k)) n) { (psum_shift f Hf 1 n) }
      == ((psum f 0 + f 0) + psum (fun k ⇒ f (1 + k)) n)
           { (op_proper _ _ (transitivity (psum_proper f Hf 1 (succ 0) Theory.Naturals.one_succ) (psum_succ f 0)) _ _ (reflexivity _)) }
      == (((0 : V) + f 0) + psum (fun k ⇒ f (1 + k)) n)
           { (op_proper _ _ (op_proper _ _ (psum_zero f) _ _ (reflexivity (f 0))) _ _ (reflexivity _)) }
      == (f 0 + psum (fun k ⇒ f (1 + k)) n) { (op_proper _ _ (left_identity (action := add) (f 0)) _ _ (reflexivity _)) }
      == (f 0 + psum (fun k ⇒ f (succ k)) n)
           { (op_proper _ _ (reflexivity (f 0)) _ _ (psum_ext _ _ Hs (fun k k' Hk ⇒ Hf _ _ (succ_proper k k' Hk)) (fun k ⇒ Hf _ _ (Theory.Naturals.add_one_l k)) n)) }.
  Qed.

End PartialSums.

Section PartialSumsScale.
  Context {F N V : Type} `{VectorSpace F V} `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Local Notation psum f n := (natrec V 0 (fun k acc ⇒ acc + f k) n).

  Theorem psum_smul : ∀ (a : F) (f : N → V), Proper (equ ==> equ) f →
    ∀ n : N, psum (fun k ⇒ a *: f k) n == a *: psum f n.
  Proof.
    intros a f Hf.
    apply (nat_induction (fun n ⇒ psum (fun k ⇒ a *: f k) n == a *: psum f n)).
    - intros n n' Hn.
      assert (Haf : Proper (equ ==> equ) (fun k ⇒ a *: f k))
        by (intros k k' Hk; apply scale_proper; [ reflexivity | apply Hf; exact Hk ]).
      split; intros Heq.
      + setoid_rewrite <- (psum_proper _ Haf n n' Hn). setoid_rewrite <- (psum_proper f Hf n n' Hn). exact Heq.
      + setoid_rewrite (psum_proper _ Haf n n' Hn). setoid_rewrite (psum_proper f Hf n n' Hn). exact Heq.
    - calc (psum (fun k ⇒ a *: f k) 0)
        == (0 : V) { (psum_zero (fun k ⇒ a *: f k)) }
        == (a *: (0 : V)) { (smul_zero (V := V) a) }
        == (a *: psum f 0) { (psum_zero f) }.
    - intros n IH.
      calc (psum (fun k ⇒ a *: f k) (succ n))
        == (psum (fun k ⇒ a *: f k) n + a *: f n) { (psum_succ (fun k ⇒ a *: f k) n) }
        == (a *: psum f n + a *: f n) { IH }
        == (a *: (psum f n + f n)) { (left_distributive (op := scale) a (psum f n) (f n)) }
        == (a *: psum f (succ n)) { (psum_succ f n) }.
  Qed.

End PartialSumsScale.

Section SeriesTheory.
  Context {F N V : Type} `{Limits F N V} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

Summable sequences: Cauchy partial sums.
  Local Notation summable f := (cauchy (fun n ⇒ psum f n)).

  Theorem summable_add : ∀ f g : N → V, Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    summable f → summable g → summable (fun k ⇒ f k + g k).
  Proof.
    intros f g Hf Hg Hsf Hsg.
    exact (cauchy_proper (fun n ⇒ psum f n + psum g n) _ (fun n ⇒ symmetry (psum_add f g Hf Hg n))
             (cauchy_add _ _ Hsf Hsg)).
  Defined.

  Theorem summable_smul : ∀ (a : F) (f : N → V), Proper (equ ==> equ) f →
    summable f → summable (fun k ⇒ a *: f k).
  Proof.
    intros a f Hf Hsf.
    exact (cauchy_proper (fun n ⇒ a *: psum f n) _ (fun n ⇒ symmetry (psum_smul a f Hf n))
             (cauchy_smul a _ Hsf)).
  Defined.

  Theorem summable_neg : ∀ f : N → V, Proper (equ ==> equ) f →
    summable f → summable (fun k ⇒ - f k).
  Proof.
    intros f Hf Hsf.
    exact (cauchy_proper (fun n ⇒ - psum f n) _ (fun n ⇒ symmetry (psum_neg f Hf n))
             (cauchy_neg _ Hsf)).
  Defined.
End SeriesTheory.

Section SeriesSum.
  Context {F N V : Type} `{Complete F N V} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Local Notation psum f n := (natrec V 0 (fun k acc ⇒ acc + f k) n).
  Local Notation ssum f H := (lim (fun n ⇒ psum f n) H).

  Theorem ssum_add : ∀ (f g : N → V) (Hf : Proper (equ ==> equ) f) (Hg : Proper (equ ==> equ) g)
      (Hsf : cauchy (fun n ⇒ psum f n)) (Hsg : cauchy (fun n ⇒ psum g n))
      (Hs : cauchy (fun n ⇒ psum (fun k ⇒ f k + g k) n)),
    ssum (fun k ⇒ f k + g k) Hs == ssum f Hsf + ssum g Hsg.
  Proof.
    intros f g Hf Hg Hsf Hsg Hs.
    calc (ssum (fun k ⇒ f k + g k) Hs)
      == (lim (fun n ⇒ psum f n + psum g n) (cauchy_add _ _ Hsf Hsg)) { (lim_proper _ _ Hs _ (psum_add f g Hf Hg)) }
      == (ssum f Hsf + ssum g Hsg) { (lim_add _ _ Hsf Hsg _) }.
  Qed.

  Theorem ssum_smul : ∀ (a : F) (f : N → V) (Hf : Proper (equ ==> equ) f)
      (Hsf : cauchy (fun n ⇒ psum f n)) (Hs : cauchy (fun n ⇒ psum (fun k ⇒ a *: f k) n)),
    ssum (fun k ⇒ a *: f k) Hs == a *: ssum f Hsf.
  Proof.
    intros a f Hf Hsf Hs.
    calc (ssum (fun k ⇒ a *: f k) Hs)
      == (lim (fun n ⇒ a *: psum f n) (cauchy_smul a _ Hsf)) { (lim_proper _ _ Hs _ (psum_smul a f Hf)) }
      == (a *: ssum f Hsf) { (lim_smul a _ Hsf _) }.
  Qed.
End SeriesSum.
