From Riktig Require Export Theory.Determinant.

The determinant of a family of tables -- a square matrix given by its columns -- through a determinant on the vectors with those coordinates; the unit tables are its basis.
Section DeterminantTable.
  Context {F I V N T : Type} `{VectorSpace F V} `{Finite I N} {EI : EquDec I}
    {natrecN : HasNatRec N} {NR : NatRec N natrecN}
    {updV : HasUpd I V} {UV : Upd I V updV}
    {basis : I → V} {basis_proper : Proper (equ ==> equ) basis}
    {DT : HasDet F I V} {D : Determinant F I V basis DT}
    {eqT : EqRel T} {getT : HasGet T I F} {tabT : HasTabulate T I F} {TT : Table I F T getT tabT}
    {zT : Zero T} {aT : HasAdd T} {nT : HasNeg T} {sT : HasScale F T} {VT : VectorSpace F T zT aT nT sT}
    {TS : TableSpace F I T} {updT : HasUpd I T} {UT : Upd I T updT}.

  #[export] Instance Columns_Det : HasDet F I T :=
    fun M ⇒ det (fun j ⇒ natrec V 0 (fun k acc ⇒ acc + get (M j) (elem k) *: basis (elem k)) card).

  #[export] Instance Columns_Multilinear : Multilinear F I T Columns_Det.
  Proof.
    split; unfold det, Columns_Det.
    - exact cols_det_proper.
    - exact cols_det_additive.
    - exact cols_det_homogeneous.
  Qed.

  #[export] Instance Columns_Alternating : Alternating F I T Columns_Det.
  Proof. unfold det, Columns_Det. exact cols_det_alternating. Qed.

  #[export] Instance Columns_Determinant :
    Determinant F I T (fun j : I ⇒ tabulate (fun i : I ⇒ if equ_dec i j then (1 : F) else 0)) Columns_Det.
  Proof.
    split; [ exact Columns_Multilinear | exact Columns_Alternating | ].
    unfold det, Columns_Det.
    exact (cols_det_delta (F := F) (I := I) (V := V) (N := N) (T := T) (basis := basis) (DT := DT)).
  Qed.
End DeterminantTable.
