proj v u is the component of u along v.
Class HasProj (V : Type) := proj : V → V → V.
Hint Mode HasProj ! : typeclass_instances.
Typeclasses Transparent HasProj.

Module Proj.
  Class Proj (F V : Type) `{InnerProductSpace F V} {Vp : HasProj V} :=
    is_proj_def : ∀ u v : V,
      proj v u == (⟪ u , v ⟫ × inv (norm2 v)) *: v.
  Hint Mode Proj - ! - - - - - - - - - - - - - - - - - : typeclass_instances.
End Proj.
