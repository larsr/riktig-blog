The same reader, from a primitive array through a cursor -- the parser walks the index, it does not follow links -- ending in the same run of Euclid's algorithm.

Definition prelude : string := " true = \a b. a ; false = \a b. b ; succ = \n f x. f (n f x) ; add = \m n f x. m f (n f x) ; mul = \m n f. m (n f) ; pred = \n f x. n (\g h. h (g f)) (\u. x) (\u. u) ; sub = \m n. n pred m ; iszero = \n. n (\x. false) true ; leq = \m n. iszero (sub m n) ; fix = \f. (\x. f (\v. x x v)) (\x. f (\v. x x v)) ; mod = fix (\mod a b. (leq b a) (\u. mod (sub a b) b) (\u. a) 0) ; gcd = fix (\gcd a b. (iszero b) (\u. a) (\u. gcd b (mod a b)) 0) ; ".

Definition G_array : list (name × term) :=
  parse_first (defs_p (P := parser (ArrayCursor ascii)) (E := list name) 64)
    nil (array_of_string prelude, 0%nat).

Definition run_array (s : string) : option nat :=
  parse_first
    (term_p (P := parser (ArrayCursor ascii)) (T := term) 64 G_array (nil : list name)
       >>= fun t ⇒ ret (eval_numeral_names G_array 10000 t))
    None (array_of_string s, 0%nat).

Eval vm_compute in run_array "gcd 12 34".

Example gcd_12_34_array : run_array "gcd 12 34" = Some 2.
Proof. vm_compute. reflexivity. Qed.
