The same grammar read from a primitive array through a cursor -- the parser walks the index, it does not follow links.
Definition array_of_string (s : string) : array ascii :=
  let cs := list_ascii_of_string s in
  tabulate_array (length cs) " "%char (fun k ⇒ nth k cs " "%char).

Definition G_array : ctx :=
  match read_defs array_size (array_of_string prelude, O) with
  | (G, _) :: _ ⇒ G
  | [] ⇒ []
  end.

Eval vm_compute in map fst G_array.

Definition run_array (s : string) : option nat :=
  match read_term array_size (array_of_string s, O) with
  | (t, _) :: _ ⇒ eval_numeral G_array 10000 t
  | [] ⇒ None
  end.

Eval vm_compute in run_array "gcd 12 34".

Example gcd_12_34_array : run_array "gcd 12 34" = Some 2.
Proof. vm_compute. reflexivity. Qed.

The array parsers are parsers in the structure's sense too.
Instance ArrayParser : Parser ascii (ArrayCursor ascii) (parser (ArrayCursor ascii))
    Stream_HasReturn Stream_HasBind Stream_HasFail Stream_HasAlt Stream_HasItem Stream_HasParses :=
  Stream_Parser array_next_step.

Example ident_idem_array : ident array_size <|> ident array_size == ident (St := ArrayCursor ascii) array_size.
Proof. apply alt_idem. Qed.
