Local Notation EvalZ :=
  (Eval Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).


Lemma eval_at_ (base : Z) (mi : MemT Z Z Z) (m m1 : Machine Z Z) (i : Z) :
  EvalZ mi m (Some i) m1 → EvalZ (at_ base mi) m (Some (base + i)) m1.
Proof.
  intros H. unfold at_.
  eapply eval_bind_t; [exact H | apply eval_offset_m].
Qed.

Lemma eval_read_arr (base : Z) (mi : MemT Z Z Z) (m m1 : Machine Z Z) (i v : Z) :
  EvalZ mi m (Some i) m1 →
  store_lookup Z.eqb (base + i) (mach_store m1) = Some (CellVal v) →
  EvalZ (read_arr base mi) m (Some v) m1.
Proof.
  intros H Hlk. unfold read_arr.
  eapply eval_bind_t; [exact (eval_at_ base mi m m1 i H)|].
  eapply eval_read_m; exact Hlk.
Qed.

Lemma eval_plus_ (mx my : MemT Z Z Z) (m m1 m2 : Machine Z Z) (x y : Z) :
  EvalZ mx m (Some x) m1 →
  EvalZ my m1 (Some y) m2 →
  EvalZ (plus_ Z.add mx my) m (Some (x + y)) m2.
Proof.
  intros Hx Hy. unfold plus_.
  eapply eval_bind_t; [exact Hx|].
  eapply eval_bind_t; [exact Hy|].
  apply eval_ret_t.
Qed.

Lemma eval_if_m (a : Type) (cond : MemT Z Z bool) (thn els : MemT Z Z a)
    (m m1 m2 : Machine Z Z) (taken : bool) (r : option a) :
  EvalZ cond m (Some taken) m1 →
  EvalZ (if taken then thn else els) m1 r m2 →
  EvalZ (if_m cond thn els) m r m2.
Proof.
  intros Hc Hb. unfold if_m.
  eapply eval_bind_t; [exact Hc | exact Hb].
Qed.

Lemma eval_when_m (cond : MemT Z Z bool) (b : MemT Z Z unit)
    (m m1 m2 : Machine Z Z) (taken : bool) (r : option unit) :
  EvalZ cond m (Some taken) m1 →
  EvalZ (if taken then b else ret_t tt) m1 r m2 →
  EvalZ (when_m cond b) m r m2.
Proof.
  intros Hc Hb. unfold when_m.
  exact (eval_if_m unit cond b (ret_t tt) m m1 m2 taken r Hc Hb).
Qed.
