From Coq Require Import ZArith QArith Qabs Lia.

The integer square root, read as an approximate root on Q: the root of x to within 1 # s is the integer root of x scaled by s^2, over s. Integer arithmetic throughout.

Open Scope Z_scope.

Lemma sqrt_div_lower : ∀ a b s : Z, 0 ≤ a → 0 < b → 0 < s →
  let q := a × s × s / b in let k := Z.sqrt q in k × k × b ≤ a × s × s.
Proof.
  intros a b s Ha Hb Hs. simpl.
  set (q := a × s × s / b). set (k := Z.sqrt q).
  assert (Hq : 0 ≤ q) by (apply Z.div_pos; [nia | exact Hb]).
  assert (Hk : k × k ≤ q) by (apply (Z.sqrt_spec q Hq)).
  assert (Hqb : b × (a × s × s / b) ≤ a × s × s) by (apply Z.mul_div_le; lia).
  assert (Hqb2 : q × b ≤ a × s × s) by (unfold q; lia).
  nia.
Qed.

Lemma sqrt_div_upper : ∀ a b s : Z, 0 ≤ a → 0 < b → 0 < s →
  let q := a × s × s / b in let k := Z.sqrt q in
  a × s × s < (k + 1) × (k + 1) × b.
Proof.
  intros a b s Ha Hb Hs. simpl.
  set (q := a × s × s / b). set (k := Z.sqrt q).
  assert (Hq : 0 ≤ q) by (apply Z.div_pos; [nia | exact Hb]).
  assert (Hk : q < Z.succ k × Z.succ k) by (apply (Z.sqrt_spec q Hq)).
  assert (Hmod : a × s × s = b × (a × s × s / b) + (a × s × s) mod b)
    by (apply Z.div_mod; lia).
  assert (Hmod2 : a × s × s = q × b + (a × s × s) mod b) by (unfold q; lia).
  assert (Hr : 0 ≤ (a × s × s) mod b < b) by (apply Z.mod_pos_bound; exact Hb).
  unfold Z.succ in Hk.
  nia.
Qed.

Close Scope Z_scope.
Open Scope Q_scope.

Definition qsqrt_at (s : positive) (x : Q) : Q :=
  Z.sqrt (Qnum x × Z.pos s × Z.pos s / Z.pos (Qden x)) # s.

Lemma Q_unit_den_le : ∀ eps : Q, 0 < eps → (1 # Qden eps) ≤ eps.
Proof.
  intros [a b] Heps.
  unfold Qlt, Qnum, Qden in Heps. simpl in Heps.
  unfold Qle, Qnum, Qden. simpl.
  nia.
Qed.

Lemma qsqrt_at_nonneg : ∀ (s : positive) (x : Q), 0 ≤ qsqrt_at s x.
Proof.
  intros s [a b]. unfold qsqrt_at, Qle, Qnum, Qden. simpl.
  assert (H := Z.sqrt_nonneg (a × Z.pos s × Z.pos s / Z.pos b)).
  lia.
Qed.

Lemma qsqrt_at_le : ∀ (s : positive) (x : Q),
  0 ≤ x → qsqrt_at s x × qsqrt_at s x ≤ x.
Proof.
  intros s [a b] Hx.
  assert (Ha : (0 ≤ a)%Z)
    by (unfold Qle, Qnum, Qden in Hx; simpl in Hx; lia).
  assert (H := sqrt_div_lower a (Z.pos b) (Z.pos s) Ha
                 (Pos2Z.is_pos b) (Pos2Z.is_pos s)).
  simpl in H.
  unfold qsqrt_at, Qmult, Qle, Qnum, Qden. simpl.
  nia.
Qed.

Lemma qsqrt_at_close : ∀ (s : positive) (x : Q),
  0 ≤ x → x ≤ (qsqrt_at s x + (1 # s)) × (qsqrt_at s x + (1 # s)).
Proof.
  intros s [a b] Hx.
  assert (Ha : (0 ≤ a)%Z)
    by (unfold Qle, Qnum, Qden in Hx; simpl in Hx; lia).
  assert (H := sqrt_div_upper a (Z.pos b) (Z.pos s) Ha
                 (Pos2Z.is_pos b) (Pos2Z.is_pos s)).
  simpl in H.
  unfold qsqrt_at, Qplus, Qmult, Qle, Qnum, Qden. simpl.
  nia.
Qed.

The three laws of an approximate root, at the root to within 1 # Qden eps, which is within eps.
Definition qsqrt (x eps : Q) : Q := qsqrt_at (Qden eps) x.

Lemma qsqrt_nonneg : ∀ x eps : Q, 0 ≤ qsqrt x eps.
Proof. intros x eps. exact (qsqrt_at_nonneg (Qden eps) x). Qed.

Lemma qsqrt_le : ∀ x eps : Q, 0 ≤ x → (0 ≤ eps ∧ ¬ 0 == eps) →
  qsqrt x eps × qsqrt x eps ≤ x.
Proof. intros x eps Hx _. exact (qsqrt_at_le (Qden eps) x Hx). Qed.

Lemma qsqrt_close : ∀ x eps : Q, 0 ≤ x → (0 ≤ eps ∧ ¬ 0 == eps) →
  x ≤ (qsqrt x eps + eps) × (qsqrt x eps + eps).
Proof.
  intros x eps Hx [Heps Hne].
  assert (Hlt : 0 < eps).
  { destruct (proj1 (Qle_lteq 0 eps) Heps) as [H | H]; [ exact H | destruct (Hne H) ]. }
  unfold qsqrt.
  set (r := qsqrt_at (Qden eps) x). set (u := (1 # Qden eps)).
  assert (Hr : 0 ≤ r) by exact (qsqrt_at_nonneg (Qden eps) x).
  assert (Hu : 0 ≤ u) by (unfold u, Qle; simpl; lia).
  assert (Hunit : u ≤ eps) by exact (Q_unit_den_le eps Hlt).
  assert (Hsum : r + u ≤ r + eps) by exact (proj2 (Qplus_le_r u eps r) Hunit).
  assert (Hsum0 : 0 ≤ r + u).
  { apply (Qle_trans _ (r + 0)); [ rewrite Qplus_0_r; exact Hr | ].
    exact (proj2 (Qplus_le_r 0 u r) Hu). }
  assert (Hre : 0 ≤ r + eps) by exact (Qle_trans _ _ _ Hsum0 Hsum).
  apply (Qle_trans _ _ _ (qsqrt_at_close (Qden eps) x Hx)).
  apply (Qle_trans _ ((r + eps) × (r + u))).
  - exact (Qmult_le_compat_r _ _ _ Hsum Hsum0).
  - rewrite (Qmult_comm (r + eps) (r + u)).
    exact (Qmult_le_compat_r _ _ _ Hsum Hre).
Qed.
