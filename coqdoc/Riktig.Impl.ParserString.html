Library Riktig.Impl.ParserString

From Coq Require Import Ascii String.
From Riktig Require Export Impl.ParserStream Impl.EquDecAscii.

Coq strings as sequences of characters, and the list-of-successes parsers over them: parsers run on string literals directly.
Instance String_HasStep : HasStep ascii string := fun s c s' ⇒ s = String c s'.
Instance String_HasNext : HasNext ascii string :=
  fun s ⇒ match s with EmptyString ⇒ None | String c s' ⇒ Some (c, s') end.

Instance String_Sequence : Sequence ascii string String_HasStep (eqC := eq).
Proof.
  split.
  - exact eq_equivalence.
  - exact eq_equivalence.
  - intros s s0 Hs c c0 Hc t t0 Ht. unfold equ, String_EqRel in ×. subst. reflexivity.
  - intros s c c' s' s'' H1 H2. unfold step, String_HasStep in ×. subst. injection H2 as Hc Hs. split; assumption.
Qed.

Lemma string_next_step : ∀ (s : string) (c : ascii) (s' : string),
  step s c s' ↔ ∃ c0 : ascii, next s = Some (c0, s') ∧ c0 = c.
Proof.
  intros s c s'. unfold next, String_HasNext, step, String_HasStep. destruct s as [| c0 s0].
  - split; [ discriminate | intros [c1 [H _]]; discriminate H ].
  - split.
    + intros H. injection H as Hc Hs. subst. ∃ c. split; reflexivity.
    + intros [c1 [H Hc]]. injection H as Hc1 Hs. subst. reflexivity.
Qed.

Instance StringParser : Parser ascii string (parser string)
    Stream_HasReturn Stream_HasBind Stream_HasFail Stream_HasAlt Stream_HasItem Stream_HasParses :=
  Stream_Parser string_next_step.
