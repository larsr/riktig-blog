Library Riktig.Theory.PlaneRotor

From Coq Require Import Morphisms Setoid Ring Bool.
From Coq.Vectors Require Fin.
From Riktig Require Export Impl.Clifford Impl.Complex.
From Riktig Require Import Lib.Calc.

Section PlaneRotor.
  Context {F : Type} `{FF : Field F}.
  Add Ring plane_rotor_ring : (ring_theory_of_Ring (R := F) commutative).

  #[local] Instance Plane_One : One (Cl F 2) := cl_one.
  #[local] Instance Plane_Mult : HasMult (Cl F 2) := cl_mul sig2.

  Definition pb (b1 b2 : bool) : Blade 2 := bcons b1 (bcons b2 bnil).

  Lemma blade2_case : ∀ D : Blade 2,
    D = pb (bhead D) (bhead (btail D)).
  Proof.
    intros D.
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    exact (blade_0 (btail (btail D))).
  Qed.

  Definition mk2 (c a b p : F) : Cl F 2 :=
    fun D ⇒
      if bhead D
      then if bhead (btail D) then p else a
      else if bhead (btail D) then b else c.

  Definition rotor (c s : F) : Cl F 2 := mk2 c 0 0 s.

  Definition plane_vector (a b : F) : Fin.t 2 → F :=
    fun j ⇒ match j with Fin.F1 ⇒ a | Fin.FS _ ⇒ b end.

  Definition sandwich (R x : Cl F 2) : Cl F 2 := R × x × cl_reverse R.

  Definition even_to_complex (x : Cl F 2) : C F :=
    mkC (x (pb false false)) (x (pb true true)).

  Definition complex_to_even (z : C F) : Cl F 2 := mk2 (re z) 0 0 (im z).

  Lemma cl2_ext : ∀ x y : Cl F 2,
    x (pb false false) == y (pb false false) →
    x (pb true false) == y (pb true false) →
    x (pb false true) == y (pb false true) →
    x (pb true true) == y (pb true true) →
    x == y.
  Proof.
    intros x y H00 H10 H01 H11 D.
    rewrite (blade2_case D).
    destruct (bhead D), (bhead (btail D)); assumption.
  Qed.

  Ltac blade_reduce :=
    unfold sandwich, rotor, complex_to_even, even_to_complex;
    repeat change (@mult (Cl F 2) Plane_Mult) with Plane_Mult;
    repeat change (@one (Cl F 2) Plane_One) with Plane_One;
    unfold Plane_Mult, Plane_One, cl_mul, cl_one, cl_reverse, bsign,
      parity_sign, mk2, pb, bxor, bdiff, band;
    simpl.
  Ltac by_blade := blade_reduce; ring.


  Lemma reverse_mk2 : ∀ c a b p : F,
    cl_reverse (mk2 c a b p) == mk2 c a b (- p).
  Proof. intros c a b p. apply cl2_ext; by_blade. Qed.

  Lemma reverse_anti2 : ∀ x y : Cl F 2,
    cl_reverse (x × y) == cl_reverse y × cl_reverse x.
  Proof. intros x y. apply cl2_ext; by_blade. Qed.


  Lemma rotor_even : ∀ c s : F, cl_even (rotor c s).
  Proof.
    intros c s D HD.
    rewrite (blade2_case D) in ×.
    destruct (bhead D), (bhead (btail D));
      solve [ discriminate HD | reflexivity ].
  Qed.

  Lemma rotor_mul : ∀ c1 s1 c2 s2 : F,
    rotor c1 s1 × rotor c2 s2
    == rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2).
  Proof. intros. apply cl2_ext; by_blade. Qed.

  Lemma rotor_reverse : ∀ c s : F,
    cl_reverse (rotor c s) == rotor c (- s).
  Proof. intros c s. exact (reverse_mk2 c 0 0 s). Qed.

  Lemma rotor_reverse_mul : ∀ c s : F,
    rotor c s × cl_reverse (rotor c s) == rotor (c×c + s×s) 0.
  Proof. intros. apply cl2_ext; by_blade. Qed.

  Lemma rotor_one : rotor 1 0 == (1 : Cl F 2).
  Proof. apply cl2_ext; by_blade. Qed.

  Lemma scale_rotor : ∀ c a b : F,
    c *: rotor a b == rotor (c × a) (c × b).
  Proof.
    intros c a b.
    apply cl2_ext; rewrite (cl_scale_at 2 c (rotor a b)); by_blade.
  Qed.

  #[export] Instance rotor_proper : Proper (equ ==> equ ==> equ) rotor.
  Proof.
    intros c c' Hc s s' Hs D.
    rewrite (blade2_case D).
    unfold rotor, mk2, pb.
    destruct (bhead D), (bhead (btail D));
      solve [ exact Hc | exact Hs | reflexivity ].
  Qed.

  Lemma rotor_unit : ∀ c s : F,
    c×c + s×s == 1 → rotor c s × cl_reverse (rotor c s) == 1.
  Proof.
    intros c s Hcs.
    calc ((rotor c s × cl_reverse (rotor c s)) : Cl F 2)
      == (rotor (c×c + s×s) 0) { rotor_reverse_mul }
      == (rotor 1 0) { Hcs }
      == (1 : Cl F 2) { rotor_one }.
  Qed.

  Lemma rotor_unit_necessary : ∀ c s : F,
    rotor c s × cl_reverse (rotor c s) == 1 → c×c + s×s == 1.
  Proof.
    intros c s Hru.
    calc (c×c + s×s)
      == ((rotor c s × cl_reverse (rotor c s)) (pb false false))
         { rotor_reverse_mul }
      == ((1 : Cl F 2) (pb false false))
         { Hru }
      == (1 : F)
         { reflexivity }.
  Qed.

  Lemma rotor_unit_iff : ∀ c s : F,
    rotor c s × cl_reverse (rotor c s) == 1 ↔ c×c + s×s == 1.
  Proof.
    intros c s.
    split; [ exact (rotor_unit_necessary c s) | exact (rotor_unit c s) ].
  Qed.


  Lemma vec_plane : ∀ v1 v2 : F,
    ga_vec (plane_vector v1 v2) == mk2 0 v1 v2 0.
  Proof. intros v1 v2. apply cl2_ext; by_blade. Qed.

  Lemma sandwich_rotor_vec : ∀ c s v1 v2 : F,
    sandwich (rotor c s) (ga_vec (plane_vector v1 v2))
    == ga_vec (plane_vector
         ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
         (- ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2)).
  Proof. intros. apply cl2_ext; by_blade. Qed.

  Lemma sandwich_compose : ∀ A B x : Cl F 2,
    sandwich A (sandwich B x) == sandwich (A × B) x.
  Proof. intros A B x. apply cl2_ext; by_blade. Qed.

  Lemma qform_plane : ∀ v1 v2 : F,
    qform sig2 (plane_vector v1 v2) == v1×v1 + v2×v2.
  Proof. intros v1 v2. unfold sig2, plane_vector. simpl. ring. Qed.

  Lemma rotation_preserves_qform : ∀ c s v1 v2 : F,
    c×c + s×s == 1 →
    qform sig2 (plane_vector
        ((c×c + - (s×s)) × v1 + (c×s + c×s) × v2)
        (- ((c×s + c×s) × v1) + (c×c + - (s×s)) × v2))
    == qform sig2 (plane_vector v1 v2).
  Proof.
    intros c s v1 v2 Hcs.
    rewrite !qform_plane.
    transitivity ((c×c + s×s) × ((c×c + s×s) × (v1×v1 + v2×v2)));
      [ ring |].
    rewrite Hcs. ring.
  Qed.


  Lemma even_round_trip : ∀ x : Cl F 2,
    cl_even x → complex_to_even (even_to_complex x) == x.
  Proof.
    intros x Hx.
    apply cl2_ext; unfold complex_to_even, even_to_complex, mk2, pb; simpl;
      solve [ reflexivity
            | symmetry; apply Hx; reflexivity ].
  Qed.

  Lemma complex_round_trip : ∀ z : C F,
    even_to_complex (complex_to_even z) == z.
  Proof. intros z. split; reflexivity. Qed.

  Lemma even_is_rotor : ∀ x : Cl F 2,
    cl_even x → ∃ a b : F, x == rotor a b.
  Proof.
    intros x Hx.
    ∃ (x (pb false false)), (x (pb true true)).
    symmetry. exact (even_round_trip x Hx).
  Qed.

  Lemma even_to_complex_rotor : ∀ a b : F,
    even_to_complex (rotor a b) == mkC a b.
  Proof. intros a b. split; reflexivity. Qed.

  Lemma complex_to_even_is_even : ∀ z : C F,
    cl_even (complex_to_even z).
  Proof.
    intros z D HD.
    rewrite (blade2_case D) in ×.
    destruct (bhead D), (bhead (btail D));
      solve [ discriminate HD | reflexivity ].
  Qed.

  Lemma even_to_complex_add : ∀ x y : Cl F 2,
    even_to_complex (x + y) == even_to_complex x + even_to_complex y.
  Proof. intros x y. split; reflexivity. Qed.

  Lemma even_to_complex_one : even_to_complex (1 : Cl F 2) == 1.
  Proof. split; reflexivity. Qed.

  Lemma even_to_complex_mul : ∀ x y : Cl F 2,
    cl_even x → cl_even y →
    even_to_complex (x × y) == even_to_complex x × even_to_complex y.
  Proof.
    intros x y Hx Hy.
    pose proof (Hx (pb true false) eq_refl) as Hx1.
    pose proof (Hx (pb false true) eq_refl) as Hx2.
    pose proof (Hy (pb true false) eq_refl) as Hy1.
    pose proof (Hy (pb false true) eq_refl) as Hy2.
    unfold pb in Hx1, Hx2, Hy1, Hy2.
    split; blade_reduce; rewrite Hx1, Hx2, Hy1, Hy2; ring.
  Qed.

  #[export] Instance even_to_complex_proper :
    Proper (equ ==> equ) even_to_complex.
  Proof.
    intros x y Hxy.
    split; [ exact (Hxy (pb false false)) | exact (Hxy (pb true true)) ].
  Qed.

  #[export] Instance complex_to_even_proper :
    Proper (equ ==> equ) complex_to_even.
  Proof.
    intros z w [Hre Him].
    apply cl2_ext; unfold complex_to_even, mk2, pb; simpl;
      solve [ exact Hre | exact Him | reflexivity ].
  Qed.

  Theorem rotor_composition_is_u1 : ∀ c1 s1 c2 s2 : F,
    even_to_complex (rotor c1 s1) × even_to_complex (rotor c2 s2)
    == even_to_complex (rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)).
  Proof.
    intros c1 s1 c2 s2.
    calc (even_to_complex (rotor c1 s1) × even_to_complex (rotor c2 s2))
      == (even_to_complex (rotor c1 s1 × rotor c2 s2))
         { (even_to_complex_mul (rotor c1 s1) (rotor c2 s2)
              (rotor_even c1 s1) (rotor_even c2 s2)) }
      == (even_to_complex (rotor (c1×c2 + - (s1×s2)) (c1×s2 + s1×c2)))
         { rotor_mul }.
  Qed.

End PlaneRotor.
