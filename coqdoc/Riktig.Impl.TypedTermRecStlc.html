From Riktig.Structure Require Export TypedTermRec.
From Riktig.Theory Require Import InitialAlgebra Stlc.
From Riktig.Impl Require Export CategoryFun SignaturePoly StlcOps.

Any typed-term datatype at Coq's types recurses: the recursion is the fold, and the equations are the fold's own, read at one level.
Section S.
  Context {N Ty T : Type} {eqN : EqRel N} {eqTy : EqRel Ty}
    {tb : HasBuild Type (Stlc.shape (fun T0 : Type ⇒ T0) N Ty) T}
    {tf : HasCata Type (Stlc.shape (fun T0 : Type ⇒ T0) N Ty) T}
    `{!Stlc Type (fun T0 : Type ⇒ T0) N Ty T tb tf}.

  #[export] Instance Stlc_HasTypedTermRec : HasTypedTermRec N Ty T :=
    fun B v l a ⇒ tf B (fun s ⇒ match s with
      | inl (inl n) ⇒ v n
      | inl (inr (A, b)) ⇒ l A b
      | inr (f, x) ⇒ a f x
      end).

  #[export] Instance Stlc_TypedTermRec :
      TypedTermRec N Ty T (eqT := eq)
        Stlc_HasVar Stlc_HasTyLam Stlc_HasApp Stlc_HasTypedTermRec.
  Proof.
    split.
    - intros B eB EB v l a _ _ _ t t' Ht. rewrite Ht. reflexivity.
    - intros B eB EB v l a n.
      assert (E : stermrec B v l a (var n : T) = v n).
      { exact (InitialAlgebra.cata_build (t := T) B _ (inl (inl n))). }
      rewrite E. reflexivity.
    - intros B eB EB v l a A b.
      assert (E : stermrec B v l a (tylam A b : T) = l A (stermrec B v l a b)).
      { exact (InitialAlgebra.cata_build (t := T) B _ (inl (inr (A, b)))). }
      rewrite E. reflexivity.
    - intros B eB EB v l a f u.
      assert (E : stermrec B v l a (app f u : T) = a (stermrec B v l a f) (stermrec B v l a u)).
      { exact (InitialAlgebra.cata_build (t := T) B _ (inr (f, u))). }
      rewrite E. reflexivity.
  Qed.
End S.
