Local Notation interpZ :=

Lemma swap_cells_ret {addr val : Type} (base : addr) (i j : val) :
  swap_cells base (ret_t i) (ret_t j) =
  Offset base i (fun ai ⇒
  Offset base j (fun aj ⇒
  ReadVal ai (fun vi ⇒
  ReadVal aj (fun vj ⇒
  WriteVal ai vj (WriteVal aj vi (Ret (Some tt))))))).
Proof. reflexivity. Qed.



Theorem swap_cells_eval :
  ∀ (base : Z) (mi mj : MemT Z Z Z) (m m1 m2 : Machine Z Z) (i j vi vj : Z),
    EvalZ mi m (Some i) m1 →
    EvalZ mj m1 (Some j) m2 →
    store_lookup Z.eqb (base + i) (mach_store m2) = Some (CellVal vi) →
    store_lookup Z.eqb (base + j) (mach_store m2) = Some (CellVal vj) →
    EvalZ (swap_cells base mi mj) m (Some tt)
      (mach_write (base + j) (CellVal vi) (mach_write (base + i) (CellVal vj) m2)).
Proof.
  intros base mi mj m m1 m2 i j vi vj Hi Hj Hli Hlj.
  unfold swap_cells.
  eapply eval_bind_t; [exact (eval_at_ base mi m m1 i Hi)|].
  eapply eval_bind_t; [exact (eval_at_ base mj m1 m2 j Hj)|].
  eapply eval_bind_t; [eapply eval_read_m; exact Hli|].
  eapply eval_bind_t; [eapply eval_read_m; exact Hlj|].
  eapply eval_then_t.
  - eapply eval_write_m. exact (store_lookup_member Z.eqb _ _ _ Hli).
  - eapply eval_write_m.
    exact (store_member_insert_monotone Z.eqb (base + i) (base + j) (CellVal vj)
             (mach_store m2) (store_lookup_member Z.eqb _ _ _ Hlj)).
Qed.



Theorem swap_cells_view :
  ∀ (n i j : nat) (base : Z) (m : Machine Z Z) (l : list Z) (vi vj : Z),
    mem_read_cells n base m = Some l →
    (i < n)%nat → (j < n)%nat →
    mem_read_cells n base
      (mach_write (base + Z.of_nat j) (CellVal vi)
         (mach_write (base + Z.of_nat i) (CellVal vj) m))
      = Some (upd_nth j vi (upd_nth i vj l)).
Proof.
  intros n i j base m l vi vj Hview Hi Hj.
  apply mem_read_cells_insert_inside; [exact Hj|].
  apply mem_read_cells_insert_inside; [exact Hi | exact Hview].
Qed.


Theorem swap_cells_correct :
  ∀ (n i j : nat) (base : Z) (m : Machine Z Z) (l : list Z) (vi vj : Z),
    mem_read_cells n base m = Some l →
    nth_error l i = Some vi →
    nth_error l j = Some vj →
    EvalZ (swap_cells base (ret_t (Z.of_nat i)) (ret_t (Z.of_nat j))) m (Some tt)
          (mach_write (base + Z.of_nat j) (CellVal vi)
             (mach_write (base + Z.of_nat i) (CellVal vj) m)) ∧
    mem_read_cells n base
      (mach_write (base + Z.of_nat j) (CellVal vi)
         (mach_write (base + Z.of_nat i) (CellVal vj) m))
      = Some (upd_nth j vi (upd_nth i vj l)).
Proof.
  intros n i j base m l vi vj Hview Hni Hnj.
  assert (Hlen : length l = n) by exact (mem_read_cells_length n base m l Hview).
  assert (Hi : (i < n)%nat)
    by (rewrite <- Hlen; apply nth_error_Some; rewrite Hni; discriminate).
  assert (Hj : (j < n)%nat)
    by (rewrite <- Hlen; apply nth_error_Some; rewrite Hnj; discriminate).
  split; [|exact (swap_cells_view n i j base m l vi vj Hview Hi Hj)].
  destruct (mem_read_cells_nth_error n i base m l Hview Hi) as (wi & Hwi & Hlki).
  destruct (mem_read_cells_nth_error n j base m l Hview Hj) as (wj & Hwj & Hlkj).
  rewrite Hni in Hwi; injection Hwi as <-.
  rewrite Hnj in Hwj; injection Hwj as <-.
  eapply swap_cells_eval;
    [ apply eval_ret_t | apply eval_ret_t | exact Hlki | exact Hlkj ].
Qed.



Theorem swap_l1_correct :
  ∀ (n : nat) (base i j : Z) (m : Machine Z Z) (l : list Z) (vi vj : Z),
    mem_read_cells n base m = Some l →
    0 ≤ i → 0 ≤ j →
    nth_error l (Z.to_nat i) = Some vi →
    nth_error l (Z.to_nat j) = Some vj →
    EvalZ (swap_l1 base i j) m (Some tt)
          (mach_write (base + j) (CellVal vi)
             (mach_write (base + i) (CellVal vj) m)) ∧
    mem_read_cells n base
      (mach_write (base + j) (CellVal vi) (mach_write (base + i) (CellVal vj) m))
      = Some (upd_nth (Z.to_nat j) vi (upd_nth (Z.to_nat i) vj l)).
Proof.
  intros n base i j m l vi vj Hview Hi Hj Hni Hnj.
  destruct (swap_cells_correct n (Z.to_nat i) (Z.to_nat j) base m l vi vj
              Hview Hni Hnj) as [Hev Hout].
  rewrite (Z2Nat.id i ltac:(lia)), (Z2Nat.id j ltac:(lia)) in Hev.
  rewrite (Z2Nat.id i ltac:(lia)), (Z2Nat.id j ltac:(lia)) in Hout.
  exact (conj Hev Hout).
Qed.


Lemma interp_swap_cells_ret :
  ∀ (fuel : nat) (base i j vi vj : Z) (m : Machine Z Z),
    (7 ≤ fuel)%nat →
    store_lookup Z.eqb (base + i) (mach_store m) = Some (CellVal vi) →
    store_lookup Z.eqb (base + j) (mach_store m) = Some (CellVal vj) →
    interpZ fuel (swap_cells base (ret_t i) (ret_t j)) m
      = (Some (Some tt),
         mach_write (base + j) (CellVal vi) (mach_write (base + i) (CellVal vj) m)).
Proof.
  intros fuel base i j vi vj m Hf Hli Hlj.
  apply (interp_monoZ 7 _ _ _ _ _ fuel Hf).
  rewrite swap_cells_ret.
  cbn [interp].
  rewrite Hli, Hlj.
  cbn [interp].
  rewrite (store_lookup_member Z.eqb _ _ _ Hli).
  cbn [interp].
  cbn [mach_store mach_sp mach_hp].
  rewrite (store_member_insert_monotone Z.eqb (base + i) (base + j) (CellVal vj)
             (mach_store m) (store_lookup_member Z.eqb _ _ _ Hlj)).
  cbn [interp mach_store mach_sp mach_hp].
  reflexivity.
Qed.



Lemma run_z_setup_then_swap_l1 :
  ∀ (setup : MemT Z Z Z) (memsize fuel1 fuel n : nat) (i j : Z)
         (l : list Z) (vi vj : Z),
    (let '(r, m) := run_z fuel1 setup memsize in mem_to_list m r n) = Some l →
    0 ≤ i → 0 ≤ j →
    nth_error l (Z.to_nat i) = Some vi →
    nth_error l (Z.to_nat j) = Some vj →
    (fuel1 + 8 ≤ fuel)%nat →
    (let '(r, m) := run_z fuel (do! base <- setup;
                                swap_l1 base i j >> ret_t base) memsize in
     mem_to_list m r n)
      = Some (upd_nth (Z.to_nat j) vi (upd_nth (Z.to_nat i) vj l)).
Proof.
  intros setup memsize fuel1 fuel n i j l vi vj Hsetup Hi Hj Hni Hnj Hfuel.
  unfold run_z, run_t in Hsetup |- ×.
  destruct (interpZ fuel1 setup (z_machine memsize)) as [r0 m0] eqn:E0.
  destruct r0 as [[base|]|]; try discriminate Hsetup.
  rewrite mem_to_list_some in Hsetup.
  destruct (swap_l1_correct n base i j m0 l vi vj Hsetup Hi Hj Hni Hnj)
    as [_ Hout].
  assert (Hlen : length l = n)
    by exact (mem_read_cells_length n base m0 l Hsetup).
  destruct (mem_read_cells_nth_error n (Z.to_nat i) base m0 l Hsetup
              ltac:(rewrite <- Hlen; apply nth_error_Some;
                    rewrite Hni; discriminate))
    as (wi & Hwi & Hlki).
  destruct (mem_read_cells_nth_error n (Z.to_nat j) base m0 l Hsetup
              ltac:(rewrite <- Hlen; apply nth_error_Some;
                    rewrite Hnj; discriminate))
    as (wj & Hwj & Hlkj).
  rewrite Hni in Hwi; injection Hwi as <-.
  rewrite Hnj in Hwj; injection Hwj as <-.
  rewrite (Z2Nat.id i ltac:(lia)) in Hlki.
  rewrite (Z2Nat.id j ltac:(lia)) in Hlkj.
  assert (Hrun : interpZ fuel
                   (bind_t setup (fun b ⇒ then_t (swap_l1 b i j) (ret_t b)))
                   (z_machine memsize)
                 = (Some (Some base),
                    mach_write (base + j) (CellVal vi)
                      (mach_write (base + i) (CellVal vj) m0))).
  { apply (interp_monoZ (fuel1 + (7 + 1)) _ _ _ _ _ fuel ltac:(lia)).
    unfold bind_t at 1.
    apply (interp_bindZ fuel1 _ _ setup _ _ (Some base) m0 (7 + 1) _ _ E0).
    cbn beta iota. unfold then_t, bind_t, swap_l1.
    apply (interp_bindZ 7 _ _ (swap_cells base (ret_t i) (ret_t j)) _ _
             (Some tt) _ 1 _ _
             (interp_swap_cells_ret 7 base i j vi vj m0 (le_n 7) Hlki Hlkj)).
    reflexivity. }
  rewrite Hrun.
  exact Hout.
Qed.


Corollary swap_l1_cells :
  (let '(r, m) := run_z 1000 swap_l1_demo 64 in mem_to_list m r 5)
  = Some [10; 40; 30; 20; 50].
Proof.
  unfold swap_l1_demo.
  rewrite (run_z_setup_then_swap_l1 (list_to_mem Z.of_nat [10; 20; 30; 40; 50])
             64 20 1000 5 1 3 [10; 20; 30; 40; 50] 20 40
             ltac:(vm_compute; reflexivity) ltac:(lia) ltac:(lia)
             ltac:(vm_compute; reflexivity) ltac:(vm_compute; reflexivity)
             ltac:(lia)).
  vm_compute. reflexivity.
Qed.



Definition quicksort_sorts_in_place : Prop :=
  ∀ (n : nat) (base : Z) (m : Machine Z Z) (input : list Z),
    mem_read_cells n base m = Some input →
    ∃ (m' : Machine Z Z) (output : list Z),
      EvalZ (call_qsort_l1 base (Z.of_nat n)) m (Some tt) m' ∧
      mem_read_cells n base m' = Some output ∧
      sorted_output input output.
