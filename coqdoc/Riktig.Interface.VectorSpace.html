From Coq Require Export Morphisms.
From Riktig Require Export Ops Interface.AbGroup Interface.Field.

Class ScaleLaws (F V : Type) `{Field.Field F} `{AbGroup.AbGroup V} `{HasScale F V} := {
  #[global] scale_proper :: Proper (equ ==> equ ==> equ) (@scale F V _);
  scale_distributes_over_vector_add : ∀ (a : F) (u v : V), a *: (u + v) == a *: u + a *: v;
  scale_distributes_over_scalar_add : ∀ (a b : F) (u : V), (a + b) *: u == a *: u + b *: u;
  scale_associative : ∀ (a b : F) (u : V), (a × b) *: u == a *: (b *: u);
  scale_identity : ∀ u : V, 1 *: u == u
}.

Module VectorSpace.
  Class VectorSpace (F V : Type) `{Field.Field F}
      {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V} {Vs : HasScale F V} := {
    equivalence : Equivalence (@equ V _);
    associative : ∀ x y w : V, (x + y) + w == x + (y + w);
    left_identity : ∀ x : V, 0 + x == x;
    right_identity : ∀ x : V, x + 0 == x;
    left_inverse : ∀ x : V, (- x) + x == 0;
    right_inverse : ∀ x : V, x + (- x) == 0;
    commutative : ∀ x y : V, x + y == y + x;
    add_proper : ∀ x y : V, x == y → ∀ u v : V, u == v → (x + u) == (y + v);
    neg_proper : ∀ x y : V, x == y → (- x) == (- y);
    scale_proper : ∀ a b : F, a == b → ∀ u v : V, u == v → a *: u == b *: v;
    scale_vector_add : ∀ (a : F) (u v : V), a *: (u + v) == a *: u + a *: v;
    scale_scalar_add : ∀ (a b : F) (u : V), (a + b) *: u == a *: u + b *: u;
    scale_assoc : ∀ (a b : F) (u : V), (a × b) *: u == a *: (b *: u);
    scale_identity : ∀ u : V, 1 *: u == u
  }.
End VectorSpace.
