From Riktig Require Export Structure.Order Theory.Field.

Section OrderedFieldTheory.
  Context {F : Type} `{OrderedField F}.

  Theorem le_of_eq : ∀ a b : F, a == b → a ≤ b.
  Proof. intros a b Hab. setoid_rewrite Hab. reflexivity. Qed.

  Theorem lt_irrefl : ∀ a : F, ¬ a < a.
  Proof. intros a [_ Hne]. exact (Hne (reflexivity a)). Qed.

  Theorem le_of_lt : ∀ a b : F, a < b → a ≤ b.
  Proof. intros a b [Hle _]. exact Hle. Qed.

  Theorem lt_of_le_of_lt : ∀ a b c : F, a ≤ b → b < c → a < c.
  Proof.
    intros a b c Hab [Hbc Hne]. split.
    - exact (transitivity Hab Hbc).
    - intros Hac. apply Hne. apply antisymmetry; [ exact Hbc | ].
      setoid_rewrite <- Hac. exact Hab.
  Qed.

  Theorem lt_of_lt_of_le : ∀ a b c : F, a < b → b ≤ c → a < c.
  Proof.
    intros a b c [Hab Hne] Hbc. split.
    - exact (transitivity Hab Hbc).
    - intros Hac. apply Hne. apply antisymmetry; [ exact Hab | ].
      setoid_rewrite Hac. exact Hbc.
  Qed.

  Theorem lt_trans : ∀ a b c : F, a < b → b < c → a < c.
  Proof. intros a b c Hab Hbc. exact (lt_of_lt_of_le a b c Hab (le_of_lt b c Hbc)). Qed.

  Theorem lt_of_not_le : ∀ a b : F, ¬ a ≤ b → b < a.
  Proof.
    intros a b Hn.
    destruct (le_total a b) as [Hab | Hba]; [ destruct (Hn Hab) | ].
    split; [ exact Hba | ].
    intros Hba'. apply Hn. setoid_rewrite Hba'. reflexivity.
  Qed.

  Theorem le_of_not_lt : ∀ a b : F, ¬ b < a → a ≤ b.
  Proof.
    intros a b Hn.
    apply le_stable. intros Hnle.
    exact (Hn (lt_of_not_le a b Hnle)).
  Qed.

  Theorem add_le_mono_r : ∀ a b c : F, a ≤ b → a + c ≤ b + c.
  Proof.
    intros a b c Hab.
    setoid_rewrite (commutative a c); setoid_rewrite (commutative b c).
    exact (add_le_mono_l a b c Hab).
  Qed.

  Theorem add_le_mono : ∀ a b c d : F, a ≤ b → c ≤ d → a + c ≤ b + d.
  Proof.
    intros a b c d Hab Hcd.
    exact (transitivity (add_le_mono_r a b c Hab) (add_le_mono_l c d b Hcd)).
  Qed.

  Theorem neg_le_mono : ∀ a b : F, a ≤ b → - b ≤ - a.
  Proof.
    intros a b Hab.
    assert (H1 := add_le_mono_l a b (- a) Hab).
    setoid_rewrite (left_inverse (op := add) a) in H1.
    assert (H2 := add_le_mono_r 0 (- a + b) (- b) H1).
    setoid_rewrite (left_identity (action := add) (- b)) in H2; setoid_rewrite (add_neg_cancel_right (- a) b) in H2.
    exact H2.
  Qed.

  Theorem sub_nonneg_of_le : ∀ a b : F, a ≤ b → 0 ≤ b + - a.
  Proof.
    intros a b Hab.
    assert (Hsum := add_le_mono_r a b (- a) Hab).
    setoid_rewrite (right_inverse (op := add) a) in Hsum.
    exact Hsum.
  Qed.

  Theorem le_of_sub_nonneg : ∀ a b : F, 0 ≤ b + - a → a ≤ b.
  Proof.
    intros a b Hab.
    assert (Hsum := add_le_mono_r 0 (b + - a) a Hab).
    setoid_rewrite (left_identity (action := add) a) in Hsum; setoid_rewrite (neg_add_cancel_right b a) in Hsum.
    exact Hsum.
  Qed.

  Theorem zero_le_one : (0 : F) ≤ 1.
  Proof.
    destruct (le_total (0 : F) 1) as [H01 | H10]; [ exact H01 | ].
    assert (Hn := neg_le_mono 1 0 H10).
    setoid_rewrite neg_zero in Hn.
    assert (Hsq := mul_nonneg (- (1)) (- (1)) Hn Hn).
    setoid_rewrite neg_mul_neg in Hsq; setoid_rewrite (left_identity (action := mult) (1 : F)) in Hsq.
    exact Hsq.
  Qed.

  Theorem zero_lt_one : (0 : F) < 1.
  Proof. split; [ exact zero_le_one | exact zero_ne_one ]. Qed.

  Theorem neg_nonneg_of_nonpos : ∀ a : F, a ≤ 0 → 0 ≤ - a.
  Proof.
    intros a Ha.
    assert (Hn := neg_le_mono a 0 Ha).
    setoid_rewrite neg_zero in Hn. exact Hn.
  Qed.

  Theorem abs_of_nonpos : ∀ a : F, a ≤ 0 → abs a == - a.
  Proof.
    intros a Ha.
    calc (abs a)
      == (abs (- a)) { abs_neg }
      == (- a) { (abs_of_nonneg (- a) (neg_nonneg_of_nonpos a Ha)) }.
  Qed.

  Theorem abs_nonneg : ∀ a : F, 0 ≤ abs a.
  Proof.
    intros a.
    destruct (le_total 0 a) as [Ha | Ha].
    - setoid_rewrite (abs_of_nonneg a Ha). exact Ha.
    - setoid_rewrite (abs_of_nonpos a Ha). exact (neg_nonneg_of_nonpos a Ha).
  Qed.

  Theorem abs_zero : abs (0 : F) == 0.
  Proof. exact (abs_of_nonneg 0 (reflexivity 0)). Qed.

  Theorem eq_zero_of_abs_eq_zero : ∀ a : F, abs a == 0 → a == 0.
  Proof.
    intros a Ha.
    destruct (le_total 0 a) as [H0a | Ha0].
    - setoid_rewrite (abs_of_nonneg a H0a) in Ha. exact Ha.
    - setoid_rewrite (abs_of_nonpos a Ha0) in Ha.
      calc a
        == (- - a) { neg_neg }
        == (- (0 : F)) { Ha }
        == (0 : F) { neg_zero }.
  Qed.

  Theorem le_abs : ∀ a : F, a ≤ abs a.
  Proof.
    intros a.
    destruct (le_total 0 a) as [Ha | Ha].
    - setoid_rewrite (abs_of_nonneg a Ha). reflexivity.
    - setoid_rewrite (abs_of_nonpos a Ha).
      exact (transitivity Ha (neg_nonneg_of_nonpos a Ha)).
  Qed.

  Theorem neg_le_abs : ∀ a : F, - a ≤ abs a.
  Proof.
    intros a.
    setoid_rewrite <- (abs_neg a). exact (le_abs (- a)).
  Qed.

  Theorem abs_triangle : ∀ a b : F, abs (a + b) ≤ abs a + abs b.
  Proof.
    intros a b.
    destruct (le_total 0 (a + b)) as [Hab | Hab].
    - setoid_rewrite (abs_of_nonneg (a + b) Hab).
      exact (add_le_mono a (abs a) b (abs b) (le_abs a) (le_abs b)).
    - setoid_rewrite (abs_of_nonpos (a + b) Hab); setoid_rewrite neg_add.
      exact (add_le_mono (- a) (abs a) (- b) (abs b) (neg_le_abs a) (neg_le_abs b)).
  Qed.

  Theorem nonpos_of_neg_nonneg : ∀ a : F, 0 ≤ - a → a ≤ 0.
  Proof.
    intros a Ha.
    pose proof (neg_le_mono _ _ Ha) as H'.
    setoid_rewrite neg_neg in H'. setoid_rewrite neg_zero in H'. exact H'.
  Qed.

  Theorem mul_nonpos_of_nonneg_of_nonpos : ∀ a b : F, 0 ≤ a → b ≤ 0 → a × b ≤ 0.
  Proof.
    intros a b Ha Hb. apply nonpos_of_neg_nonneg.
    setoid_rewrite <- mul_neg. exact (mul_nonneg a (- b) Ha (neg_nonneg_of_nonpos b Hb)).
  Qed.

  Theorem mul_nonpos_of_nonpos_of_nonneg : ∀ a b : F, a ≤ 0 → 0 ≤ b → a × b ≤ 0.
  Proof.
    intros a b Ha Hb. setoid_rewrite (commutative a b).
    exact (mul_nonpos_of_nonneg_of_nonpos b a Hb Ha).
  Qed.

  Theorem mul_nonneg_of_nonpos_of_nonpos : ∀ a b : F, a ≤ 0 → b ≤ 0 → 0 ≤ a × b.
  Proof.
    intros a b Ha Hb. setoid_rewrite <- (neg_mul_neg a b).
    exact (mul_nonneg (- a) (- b) (neg_nonneg_of_nonpos a Ha) (neg_nonneg_of_nonpos b Hb)).
  Qed.

  Theorem abs_mul : ∀ a b : F, abs (a × b) == abs a × abs b.
  Proof.
    intros a b.
    destruct (le_total 0 a) as [Ha | Ha]; destruct (le_total 0 b) as [Hb | Hb].
    - calc (abs (a × b))
        == (a × b) { (abs_of_nonneg (a × b) (mul_nonneg a b Ha Hb)) }
        == (abs a × b) { (abs_of_nonneg a Ha) }
        == (abs a × abs b) { (abs_of_nonneg b Hb) }.
    - calc (abs (a × b))
        == (- (a × b)) { (abs_of_nonpos (a × b) (mul_nonpos_of_nonneg_of_nonpos a b Ha Hb)) }
        == (a × - b) { mul_neg }
        == (abs a × - b) { (abs_of_nonneg a Ha) }
        == (abs a × abs b) { (abs_of_nonpos b Hb) }.
    - calc (abs (a × b))
        == (- (a × b)) { (abs_of_nonpos (a × b) (mul_nonpos_of_nonpos_of_nonneg a b Ha Hb)) }
        == (- a × b) { neg_mul }
        == (abs a × b) { (abs_of_nonpos a Ha) }
        == (abs a × abs b) { (abs_of_nonneg b Hb) }.
    - calc (abs (a × b))
        == (a × b) { (abs_of_nonneg (a × b) (mul_nonneg_of_nonpos_of_nonpos a b Ha Hb)) }
        == (- a × - b) { neg_mul_neg }
        == (abs a × - b) { (abs_of_nonpos a Ha) }
        == (abs a × abs b) { (abs_of_nonpos b Hb) }.
  Qed.

  Theorem mul_le_mono_l : ∀ a b c : F, 0 ≤ c → a ≤ b → c × a ≤ c × b.
  Proof.
    intros a b c Hc Hab.
    apply le_of_sub_nonneg.
    setoid_rewrite <- mul_neg; setoid_rewrite <- (left_distributive (op := mult) c b (- a)).
    exact (mul_nonneg c (b + - a) Hc (sub_nonneg_of_le a b Hab)).
  Qed.

  Theorem mul_le_mono_r : ∀ a b c : F, 0 ≤ c → a ≤ b → a × c ≤ b × c.
  Proof.
    intros a b c Hc Hab.
    setoid_rewrite (commutative a c); setoid_rewrite (commutative b c).
    exact (mul_le_mono_l a b c Hc Hab).
  Qed.

  Theorem inv_pos : ∀ a : F, 0 < a → 0 < inv a.
  Proof.
    intros a Ha.
    apply lt_of_not_le. intros Hle.
    pose proof (mul_nonpos_of_nonneg_of_nonpos a (inv a) (le_of_lt 0 a Ha) Hle) as Hprod.
    setoid_rewrite (mul_inv_cancel a (fun H0 ⇒ proj2 Ha (symmetry H0))) in Hprod.
    assert (Heq : (1 : F) == 0) by (apply antisymmetry; [ exact Hprod | exact zero_le_one ]).
    exact (proj2 zero_lt_one (symmetry Heq)).
  Qed.

  Theorem two_pos : (0 : F) < 1 + 1.
  Proof.
    apply (lt_of_lt_of_le 0 1 (1 + 1) zero_lt_one).
    setoid_rewrite <- (right_identity (action := add) (1 : F)) at 1.
    exact (add_le_mono_l 0 1 1 zero_le_one).
  Qed.

  Theorem mul_pos : ∀ a b : F, 0 < a → 0 < b → 0 < a × b.
  Proof.
    intros a b Ha Hb. split.
    - exact (mul_nonneg a b (le_of_lt 0 a Ha) (le_of_lt 0 b Hb)).
    - intros Heq.
      exact (mul_ne_zero a b (fun H0 ⇒ proj2 Ha (symmetry H0)) (fun H0 ⇒ proj2 Hb (symmetry H0)) (symmetry Heq)).
  Qed.

  Theorem lt_add_one : ∀ a : F, a < a + 1.
  Proof.
    intros a. split.
    - apply (proj1 (le_proper (a + 0) a (right_identity (action := add) a) (a + 1) (a + 1) (reflexivity _))).
      exact (add_le_mono_l 0 1 a zero_le_one).
    - intros Heq.
      exact (zero_ne_one (add_left_cancel a 0 1 (transitivity (right_identity (action := add) a) Heq))).
  Qed.

  Theorem half_pos : ∀ a : F, 0 < a → 0 < a × inv (1 + 1).
  Proof.
    intros a Ha. split.
    - exact (mul_nonneg a (inv (1 + 1)) (le_of_lt 0 a Ha) (le_of_lt _ _ (inv_pos (1 + 1) two_pos))).
    - intros Heq.
      exact (mul_ne_zero a (inv (1 + 1)) (fun H0 ⇒ proj2 Ha (symmetry H0))
               (inv_ne_zero (1 + 1) (fun H0 ⇒ proj2 two_pos (symmetry H0))) (symmetry Heq)).
  Qed.

  Theorem half_add_half : ∀ a : F, a × inv (1 + 1) + a × inv (1 + 1) == a.
  Proof.
    intros a.
    calc (a × inv (1 + 1) + a × inv (1 + 1))
      == ((1 + 1) × (a × inv (1 + 1))) { two_mul }
      == ((1 + 1) × (inv (1 + 1) × a)) { (commutative (op := mult) a (inv (1 + 1))) }
      == (((1 + 1) × inv (1 + 1)) × a) { associative }
      == ((1 : F) × a) { (mul_inv_cancel (1 + 1) (fun H0 ⇒ proj2 two_pos (symmetry H0))) }
      == a { (left_identity (action := mult) a) }.
  Qed.

  Theorem lt_proper : Proper (equ ==> equ ==> iff) (fun a b : F ⇒ a < b).
  Proof.
    intros a a' Ha b b' Hb. split; intros [Hle Hne]; split.
    - exact (proj1 (le_proper a a' Ha b b' Hb) Hle).
    - intros Heq. apply Hne. exact (transitivity Ha (transitivity Heq (symmetry Hb))).
    - exact (proj2 (le_proper a a' Ha b b' Hb) Hle).
    - intros Heq. apply Hne. exact (transitivity (symmetry Ha) (transitivity Heq Hb)).
  Qed.

  Theorem mul_lt_mono_l : ∀ a b c : F, 0 < c → a < b → c × a < c × b.
  Proof.
    intros a b c Hc [Hab Hne]. split.
    - exact (mul_le_mono_l a b c (le_of_lt 0 c Hc) Hab).
    - intros Heq. apply Hne. exact (mul_left_cancel c a b (fun H0 ⇒ proj2 Hc (symmetry H0)) Heq).
  Qed.

  Theorem lt_of_mul_lt_mul_l : ∀ a b c : F, 0 < c → c × a < c × b → a < b.
  Proof.
    intros a b c Hc Hlt. apply lt_of_not_le. intros Hba.
    pose proof (mul_le_mono_l b a c (le_of_lt 0 c Hc) Hba) as Hle.
    apply (proj2 Hlt). apply antisymmetry; [ exact (proj1 Hlt) | exact Hle ].
  Qed.

  Theorem inv_lt_inv : ∀ a b : F, 0 < a → a < b → inv b < inv a.
  Proof.
    intros a b Ha Hab.
    assert (Hb : 0 < b) by exact (lt_trans 0 a b Ha Hab).
    assert (Hbne : b =/= 0) by exact (fun H0 ⇒ proj2 Hb (symmetry H0)).
    assert (Hane : a =/= 0) by exact (fun H0 ⇒ proj2 Ha (symmetry H0)).
    apply (lt_of_mul_lt_mul_l (inv b) (inv a) a Ha).
    assert (Heq : inv b × b == a × inv a).
    { calc (inv b × b)
        == (1 : F) { (left_inverse_on (defined_on := nonzero F) b Hbne) }
        == (a × inv a) { (mul_inv_cancel a Hane) }. }
    apply (proj1 (lt_proper (inv b × a) (a × inv b) (commutative (op := mult) (inv b) a)
                    (inv b × b) (a × inv a) Heq)).
    exact (mul_lt_mono_l a b (inv b) (inv_pos b Hb) Hab).
  Qed.

End OrderedFieldTheory.
