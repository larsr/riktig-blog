Determinants in the plane over the rationals, through the plane algebra's frame: of a pair of vectors, and of a 2 x 2 matrix given by its columns.
Instance Plane_Det : HasDet Q two V2 :=
  Frame_Det (A := A2) (N := nat) (pseudoscalar_coefficient := Plane_Top).

Instance Plane_Determinant : Determinant Q two V2 plane_basis Plane_Det :=
  Frame_Determinant (A := A2) (N := nat) (pseudoscalar_coefficient := Plane_Top).

Instance Plane_Columns_Det : HasDet Q two (two → Q) :=
  Columns_Det (V := V2) (N := nat) (basis := plane_basis) (DT := Plane_Det).

Instance Plane_Columns_Determinant :
  Determinant Q two (two → Q) (fun j ⇒ tabulate (fun i ⇒ if equ_dec i j then (1 : Q) else 0)) Plane_Columns_Det :=
  Columns_Determinant (V := V2) (N := nat) (basis := plane_basis) (DT := Plane_Det).
