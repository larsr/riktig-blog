Fixpoint nth0 {F : Type} {ZF : Zero F} (xs : list F) (i : nat) : F :=
  match xs with
  | nil ⇒ 0
  | cons x t ⇒
      match i with
      | O ⇒ x
      | S j ⇒ nth0 t j
      end
  end.

Fixpoint ladd {F : Type} {AF : HasAdd F} (xs ys : list F) : list F :=
  match xs with
  | nil ⇒ ys
  | cons x xt ⇒
      match ys with
      | nil ⇒ cons x xt
      | cons y yt ⇒ cons (x + y) (ladd xt yt)
      end
  end.

Fixpoint lneg {F : Type} {NF : HasNeg F} (xs : list F) : list F :=
  match xs with
  | nil ⇒ nil
  | cons x t ⇒ cons (- x) (lneg t)
  end.

Fixpoint lscale {F : Type} {MF : HasMult F} (a : F) (xs : list F) : list F :=
  match xs with
  | nil ⇒ nil
  | cons x t ⇒ cons (a × x) (lscale a t)
  end.

#[global] Instance LV_EqRel {F : Type} `{EqRel F} `{Zero F} :
  EqRel (list F) :=
  fun xs ys ⇒ ∀ i : nat, nth0 xs i == nth0 ys i.

#[global] Instance LV_Zero {F : Type} : Zero (list F) := nil.

#[global] Instance LV_Add {F : Type} `{HasAdd F} : HasAdd (list F) := ladd.

#[global] Instance LV_Neg {F : Type} `{HasNeg F} : HasNeg (list F) := lneg.

#[global] Instance LV_Scale {F : Type} `{HasMult F} : HasScale F (list F) :=
  lscale.

#[global] Instance LV_Equivalence {F : Type} {e : EqRel F} {z : Zero F}
    {E : Equivalence e} : Equivalence (@LV_EqRel F e z).
Proof.
  apply Build_Equivalence.
  - intros xs i. reflexivity.
  - intros xs ys Hxy i. symmetry. apply Hxy.
  - intros xs ys zs Hxy Hyz i. rewrite (Hxy i). apply Hyz.
Qed.

Section ListVSMonoid.
  Context {F : Type} `{Monoid F}.

  Lemma nth0_add : ∀ (xs ys : list F) (i : nat),
    nth0 (xs + ys) i == nth0 xs i + nth0 ys i.
  Proof.
    intros xs.
    induction xs as [| x xt IH]; intros ys i.
    - symmetry. exact (left_identity _).
    - destruct ys as [| y yt].
      + symmetry. exact (right_identity _).
      + destruct i as [| j].
        × reflexivity.
        × exact (IH yt j).
  Qed.

  #[global] Instance lv_associative : Associative (list F) (list F) add add.
  Proof.
    intros xs ys zs i.
    rewrite nth0_add, nth0_add, nth0_add, nth0_add.
    exact (associative _ _ _).
  Qed.

  #[global] Instance lv_left_identity : LeftIdentity (list F) (list F) add 0.
  Proof.
    intros xs i.
    rewrite nth0_add.
    exact (left_identity _).
  Qed.

  #[global] Instance lv_right_identity : RightIdentity (list F) (list F) add 0.
  Proof.
    intros xs i.
    rewrite nth0_add.
    exact (right_identity _).
  Qed.

  #[global] Instance lv_add_proper : Proper (equ ==> equ ==> equ) (@add (list F) _).
  Proof.
    intros xs xs' Hx ys ys' Hy i.
    rewrite nth0_add, nth0_add.
    exact (Magma.op_proper _ _ (Hx i) _ _ (Hy i)).
  Qed.

  #[global] Instance LV_Magma : Magma (list F) add :=
    { Magma.equivalence := LV_Equivalence }.
  #[global] Instance LV_Semigroup : Semigroup (list F) add := {}.
  #[global] Instance LV_Monoid : Monoid (list F) := {}.

End ListVSMonoid.

Section ListVSGroup.
  Context {F : Type} `{Group F}.

  Lemma nth0_neg : ∀ (xs : list F) (i : nat),
    nth0 (- xs) i == - nth0 xs i.
  Proof.
    intros xs.
    induction xs as [| x xt IH]; intros i.
    - symmetry. apply neg_zero.
    - destruct i as [| j].
      + reflexivity.
      + exact (IH j).
  Qed.

  #[global] Instance lv_left_inverse : LeftInverse (list F) add neg 0.
  Proof.
    intros xs i.
    rewrite nth0_add, nth0_neg.
    exact (left_inverse _).
  Qed.

  #[global] Instance lv_right_inverse : RightInverse (list F) add neg 0.
  Proof.
    intros xs i.
    rewrite nth0_add, nth0_neg.
    exact (right_inverse _).
  Qed.

  #[global] Instance lv_neg_proper : Proper (equ ==> equ) (@neg (list F) _).
  Proof.
    intros xs xs' Hx i.
    rewrite nth0_neg, nth0_neg.
    apply Group.neg_proper, Hx.
  Qed.

  #[global] Instance LV_Group : Group (list F) := {}.

End ListVSGroup.

Section ListVSVectorSpace.
  Context {F : Type} `{Field F}.

  Lemma nth0_scale : ∀ (a : F) (xs : list F) (i : nat),
    nth0 (a *: xs) i == a × nth0 xs i.
  Proof.
    intros a xs.
    induction xs as [| x xt IH]; intros i.
    - symmetry. apply mul_0_r.
    - destruct i as [| j].
      + reflexivity.
      + exact (IH j).
  Qed.

  #[global] Instance lv_sc_proper : Proper (equ ==> equ ==> equ) (@scale F (list F) _).
  Proof.
    intros a a' Ha xs xs' Hx i.
    rewrite nth0_scale, nth0_scale.
    apply Magma.op_proper; [exact Ha | apply Hx].
  Qed.

  #[global] Instance lv_scale_left_distributive : LeftDistributive F (list F) (list F) scale.
  Proof.
    intros a u v i.
    rewrite nth0_scale, nth0_add, nth0_add, nth0_scale, nth0_scale.
    exact (left_distributive _ _ _).
  Qed.

  #[global] Instance lv_scale_right_distributive : RightDistributive F (list F) (list F) scale.
  Proof.
    intros a b u i.
    rewrite nth0_scale, nth0_add, nth0_scale, nth0_scale.
    exact (right_distributive _ _ _).
  Qed.

  #[global] Instance lv_scale_associative : Associative F (list F) scale mult.
  Proof.
    intros a b u i.
    rewrite nth0_scale, nth0_scale, nth0_scale.
    exact (associative _ _ _).
  Qed.

  #[global] Instance lv_scale_left_identity : LeftIdentity F (list F) scale 1.
  Proof.
    intros u i.
    rewrite nth0_scale.
    exact (left_identity _).
  Qed.

  #[global] Instance LV_ScaleLaws : ScaleLaws F (list F) := {}.

  #[global] Instance LV_VectorSpace : VectorSpace F (list F) := {}.

End ListVSVectorSpace.
