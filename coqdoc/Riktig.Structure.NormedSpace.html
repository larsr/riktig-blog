A normed space is a vector space over an ordered field with a norm that respects equality, is nonnegative and vanishes only at zero, scales by the absolute value of the scalar, and satisfies the triangle inequality.
Module NormedSpace.
  Class NormedSpace (F V : Type) `{VectorSpace F V}
      {le : HasLe F} {absF : HasAbs F} {OF : OrderedField F le absF} (normV : HasNorm F V) := {
    norm_proper :: Proper (equ ==> equ) (@norm F V normV);
    norm_nonneg : ∀ v : V, 0 ≤ ‖ v ‖;
    eq_zero_of_norm_eq_zero : ∀ v : V, ‖ v ‖ == 0 → v == 0;
    norm_smul : ∀ (a : F) (v : V), ‖ a *: v ‖ == abs a × ‖ v ‖;
    norm_add_le : ∀ u v : V, ‖ u + v ‖ ≤ ‖ u ‖ + ‖ v ‖;
  }.
  Hint Mode NormedSpace - ! - - - - - - - - - - - - - - - - - - : typeclass_instances.
End NormedSpace.
