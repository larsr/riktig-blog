A field is an algebra over itself: scaling is multiplication, and the scalars commute with the product by associativity and commutativity.
Section FieldAlgebra.
  Context {F : Type} `{Field F}.

  #[export] Instance Field_Algebra : Algebra F F 0 add neg 1 mult scale :=
    {| Algebra.vectors := Field_VectorSpace; Algebra.ring := ring;
       Algebra.smul_mul_assoc := fun a x y ⇒ associative a x y;
       Algebra.mul_smul_comm := fun a x y ⇒ mul_left_comm x a y |}.
End FieldAlgebra.
