From Riktig.Structure Require Export GradeInvolution Reversal.

A rotor of an algebra with a grade involution and a reversal: an even element whose reversal is its inverse on both sides.
Module Rotor.
  Class Rotor (F V A : Type) `{GradeInvolution F V A} {reversal : HasReverse A} (R : A) := {
    rotor_even : involute R == R;
    rotor_unit_r : R × reverse R == 1;
    rotor_unit_l : reverse R × R == 1;
  }.
  Hint Mode Rotor - - ! - - - - - - - - - - - - - - - - - - - - - - - - - - - - - ! : typeclass_instances.
End Rotor.
Export Rotor.
