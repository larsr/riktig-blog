Coq's sty is the simple-type tree over Coq's types: build is the two constructors and cata is the structural fold.
Section S.
  Context (B : Type).

  #[export] Instance Sty_HasBuild : HasBuild Type (StlcType.shape (fun T0 : Type ⇒ T0) B) (sty B) :=
    fun s ⇒ match s with
             | inl b ⇒ SBase b
             | inr (a, c) ⇒ SArrow a c
             end.

  #[export] Instance Sty_HasCata : HasCata Type (StlcType.shape (fun T0 : Type ⇒ T0) B) (sty B) :=
    fun X h ⇒ fix go (t : sty B) : X :=
      match t with
      | SBase b ⇒ h (inl b)
      | SArrow a c ⇒ h (inr (go a, go c))
      end.

  #[export] Instance Sty_StlcType :
    StlcType Type (fun T0 : Type ⇒ T0) B (sty B) Sty_HasBuild Sty_HasCata.
  Proof.
    split. split.
    - exact (layer_functor (fun T0 : Type ⇒ T0) (styS (S := sig Type) B)).
    - intros X h h' Hh t. induction t as [b | a IHa c IHc].
      + exact (Hh (inl b)).
      + exact (eq_trans (f_equal2 (fun x y ⇒ h (inr (x, y))) IHa IHc) (Hh (inr (_, _)))).
    - intros X h [b | [a c]]; reflexivity.
    - intros X h u Hu t. induction t as [b | a IHa c IHc].
      + exact (Hu (inl b)).
      + exact (eq_trans (Hu (inr (a, c))) (f_equal2 (fun x y ⇒ h (inr (x, y))) IHa IHc)).
  Qed.
End S.
