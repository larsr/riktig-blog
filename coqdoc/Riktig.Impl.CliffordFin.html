Section FieldScaleBridge.
  Context {F : Type} `{FF : Field F}.

  Lemma field_scale_eq : ∀ a u : F, a *: u == a × u.
  Proof.
    intros a u.
    reflexivity.
  Qed.

End FieldScaleBridge.

Section TrivialGA.
  Context {F : Type} `{FF : Field F}.

  Add Ring trivial_ga_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition self_inner (u v : F) : F := u × v.

  #[export] Instance self_inner_proper : Proper (equ ==> equ ==> equ) self_inner.
  Proof.
    intros u u' Hu v v' Hv.
    exact (op_proper u u' Hu v v' Hv).
  Qed.

  Lemma self_inner_sym : ∀ u v : F, self_inner u v == self_inner v u.
  Proof.
    intros u v.
    exact (mul_comm u v).
  Qed.

  Lemma self_inner_add_l : ∀ u v w : F,
    self_inner (u + v) w == self_inner u w + self_inner v w.
  Proof.
    intros u v w.
    exact (right_distributive u v w).
  Qed.

  Lemma self_inner_scale_l : ∀ a u v : F,
    self_inner (a *: u) v == a × self_inner u v.
  Proof.
    intros a u v.
    rewrite field_scale_eq.
    exact (associative a u v).
  Qed.

  Definition SelfDot_Inner : HasInner F F := self_inner.

  Definition SelfDot_InnerProduct
    : InnerProductSpace F F (Vi := SelfDot_Inner) :=
    {| InnerProductSpace.inner_linear := fun v ⇒
         {| Linear.proper := fun x y Hxy ⇒ self_inner_proper x y Hxy v v (reflexivity v);
            Linear.is_additive := fun x y ⇒ self_inner_add_l x y v;
            Linear.is_homogeneous := fun a x ⇒ self_inner_scale_l a x v |};
       InnerProductSpace.is_commutative := self_inner_sym |}.

  Local Existing Instance SelfDot_InnerProduct.

  Lemma self_inner_norm2 : ∀ u : F,
    InnerProductSpace.norm2 (V:=F) u == self_inner u u.
  Proof.
    intros u.
    reflexivity.
  Qed.

  Definition trivial_vec (u : F) : F := u.

  #[export] Instance trivial_vec_proper : Proper (equ ==> equ) trivial_vec.
  Proof.
    intros u v Huv.
    exact Huv.
  Qed.

  Lemma trivial_vec_add : ∀ u v : F,
    trivial_vec (u + v) == trivial_vec u + trivial_vec v.
  Proof.
    intros u v.
    reflexivity.
  Qed.

  Lemma trivial_vec_scale : ∀ a u : F,
    trivial_vec (a *: u) == a *: trivial_vec u.
  Proof.
    intros a u.
    reflexivity.
  Qed.

  Lemma trivial_scale_mul_l : ∀ a x y : F,
    (a *: x) × y == a *: (x × y).
  Proof.
    intros a x y. rewrite !field_scale_eq. ring.
  Qed.

  Lemma trivial_scale_mul_r : ∀ a x y : F,
    x × (a *: y) == a *: (x × y).
  Proof.
    intros a x y. rewrite !field_scale_eq. ring.
  Qed.

  Lemma trivial_contract : ∀ u : F,
    trivial_vec u × trivial_vec u
    == InnerProductSpace.norm2 (V:=F) u *: 1.
  Proof.
    intros u.
    unfold trivial_vec. rewrite self_inner_norm2, field_scale_eq.
    unfold self_inner. ring.
  Qed.

  Definition GA_trivial : GeometricAlgebra F F F (Q := square_form) :=
    {| GeometricAlgebra.ring := _;
       GeometricAlgebra.scaling := Field_ScaleLaws;
       GeometricAlgebra.mul_linear_l := fun y ⇒
         {| Linear.proper := fun x x' Hx ⇒ op_proper x x' Hx y y (reflexivity y);
            Linear.is_additive := fun x x' ⇒ right_distributive x x' y;
            Linear.is_homogeneous := fun a x ⇒ trivial_scale_mul_l a x y |};
       GeometricAlgebra.mul_linear_r := fun x ⇒
         {| Linear.proper := fun y y' Hy ⇒ op_proper x x (reflexivity x) y y' Hy;
            Linear.is_additive := fun y y' ⇒ left_distributive x y y';
            Linear.is_homogeneous := fun a y ⇒ trivial_scale_mul_r a x y |};
       GeometricAlgebra.ga_vec := trivial_vec;
       GeometricAlgebra.vec_linear :=
         {| Linear.proper := trivial_vec_proper;
            Linear.is_additive := trivial_vec_add;
            Linear.is_homogeneous := trivial_vec_scale |};
       GeometricAlgebra.contraction := trivial_contract |}.

End TrivialGA.

Example ga_trivial_check (F : Type) `{FF : Field F} :
  GeometricAlgebra F F F (Q := square_form) :=
  GA_trivial.

Section Cliff1GA.
  Context {F : Type} `{FF : Field F}.

  Add Ring cliff1_ga_ring : (ring_theory_of_Ring (R := F) commutative).

  Definition cliff1_neg_inner (u v : F) : F := - (u × v).

  #[export] Instance cliff1_ip_proper : Proper (equ ==> equ ==> equ) cliff1_neg_inner.
  Proof.
    intros u u' Hu v v' Hv.
    exact (neg_proper (u × v) (u' × v') (op_proper u u' Hu v v' Hv)).
  Qed.

  Lemma cliff1_ip_sym : ∀ u v : F,
    cliff1_neg_inner u v == cliff1_neg_inner v u.
  Proof.
    intros u v.
    exact (neg_proper (u × v) (v × u) (mul_comm u v)).
  Qed.

  Lemma cliff1_ip_add_l : ∀ u v w : F,
    cliff1_neg_inner (u + v) w
    == cliff1_neg_inner u w + cliff1_neg_inner v w.
  Proof.
    intros u v w. unfold cliff1_neg_inner. ring.
  Qed.

  Lemma cliff1_ip_scale_l : ∀ a u v : F,
    cliff1_neg_inner (a *: u) v == a × cliff1_neg_inner u v.
  Proof.
    intros a u v. unfold cliff1_neg_inner. rewrite field_scale_eq. ring.
  Qed.

  Definition Cliff1_Inner : HasInner F F := cliff1_neg_inner.

  Definition Cliff1_NegInner
    : InnerProductSpace F F (Vi := Cliff1_Inner) :=
    {| InnerProductSpace.inner_linear := fun v ⇒
         {| Linear.proper := fun x y Hxy ⇒ cliff1_ip_proper x y Hxy v v (reflexivity v);
            Linear.is_additive := fun x y ⇒ cliff1_ip_add_l x y v;
            Linear.is_homogeneous := fun a x ⇒ cliff1_ip_scale_l a x v |};
       InnerProductSpace.is_commutative := cliff1_ip_sym |}.

  #[export] Instance Cliff1_Scale : HasScale F (C F) :=
    fun (a : F) (z : C F) ⇒ mkC (a × re z) (a × im z).

  #[export] Instance cliff1_sc_proper : Proper (equ ==> equ ==> equ) Cliff1_Scale.
  Proof.
    intros a a' Ha z z' [Hr Hi].
    split.
    - exact (op_proper a a' Ha (re z) (re z') Hr).
    - exact (op_proper a a' Ha (im z) (im z') Hi).
  Qed.

  Lemma cliff1_scale_add : ∀ (a : F) (z w : C F),
    a *: (z + w) == a *: z + a *: w.
  Proof.
    intros a z w.
    split.
    - exact (left_distributive a (re z) (re w)).
    - exact (left_distributive a (im z) (im w)).
  Qed.

  Lemma cliff1_add_scale : ∀ (a b : F) (u : C F),
    (a + b) *: u == a *: u + b *: u.
  Proof.
    intros a b u.
    split.
    - exact (right_distributive a b (re u)).
    - exact (right_distributive a b (im u)).
  Qed.

  Lemma cliff1_scale_scale : ∀ (a b : F) (u : C F),
    (a × b) *: u == a *: (b *: u).
  Proof.
    intros a b u.
    split.
    - exact (associative a b (re u)).
    - exact (associative a b (im u)).
  Qed.

  Lemma cliff1_scale_1 : ∀ u : C F, 1 *: u == u.
  Proof.
    intros u.
    split.
    - exact (left_identity (re u)).
    - exact (left_identity (im u)).
  Qed.

  Definition Cliff1_ScaleLaws : ScaleLaws F (C F) :=
    {| ScaleLaws.scale_proper := cliff1_sc_proper;
       ScaleLaws.is_left_distributive := cliff1_scale_add;
       ScaleLaws.is_right_distributive := cliff1_add_scale;
       ScaleLaws.is_associative := cliff1_scale_scale;
       ScaleLaws.is_left_identity := cliff1_scale_1 |}.

  Definition cliff1_ga_vec (u : F) : C F := mkC 0 u.

  #[export] Instance cliff1_ga_vec_proper : Proper (equ ==> equ) cliff1_ga_vec.
  Proof.
    intros u v Huv.
    split.
    - reflexivity.
    - exact Huv.
  Qed.

  Lemma cliff1_ga_vec_add : ∀ u v : F,
    cliff1_ga_vec (u + v) == cliff1_ga_vec u + cliff1_ga_vec v.
  Proof.
    intros u v.
    split.
    - symmetry. exact (left_identity (0 : F)).
    - reflexivity.
  Qed.

  Lemma cliff1_ga_vec_scale : ∀ a u : F,
    cliff1_ga_vec (a *: u) == a *: cliff1_ga_vec u.
  Proof.
    intros a u.
    split.
    - symmetry. exact (mul_0_r a).
    - exact (field_scale_eq a u).
  Qed.

  Lemma cliff1_contract_re : ∀ u : F,
    0 × 0 + - (u × u) == (- (u × u)) × 1.
  Proof. intros u. ring. Qed.

  Lemma cliff1_contract_im : ∀ u : F,
    0 × u + u × 0 == (- (u × u)) × 0.
  Proof. intros u. ring. Qed.

  Lemma cliff1_ga_contract : ∀ u : F,
    cliff1_ga_vec u × cliff1_ga_vec u
    == InnerProductSpace.norm2 (F:=F) (V:=F) (H1:=Cliff1_NegInner) u *: 1.
  Proof.
    intros u.
    split.
    - exact (cliff1_contract_re u).
    - exact (cliff1_contract_im u).
  Qed.

  Lemma cliff1_ga_scale_mul_l : ∀ (a : F) (x y : C F),
    (a *: x) × y == a *: (x × y).
  Proof. intros a x y. split; simpl; ring. Qed.

  Lemma cliff1_ga_scale_mul_r : ∀ (a : F) (x y : C F),
    x × (a *: y) == a *: (x × y).
  Proof. intros a x y. split; simpl; ring. Qed.

  Definition GA_Cliff1 : GeometricAlgebra F F (C F) (Q := neg_form square_form) :=
    {| GeometricAlgebra.ring := C_Ring;
       GeometricAlgebra.scaling := Cliff1_ScaleLaws;
       GeometricAlgebra.mul_linear_l := fun y ⇒
         {| Linear.proper := fun x x' Hx ⇒ op_proper x x' Hx y y (reflexivity y);
            Linear.is_additive := fun x x' ⇒ right_distributive x x' y;
            Linear.is_homogeneous := fun a x ⇒ cliff1_ga_scale_mul_l a x y |};
       GeometricAlgebra.mul_linear_r := fun x ⇒
         {| Linear.proper := fun y y' Hy ⇒ op_proper x x (reflexivity x) y y' Hy;
            Linear.is_additive := fun y y' ⇒ left_distributive x y y';
            Linear.is_homogeneous := fun a y ⇒ cliff1_ga_scale_mul_r a x y |};
       GeometricAlgebra.ga_vec := cliff1_ga_vec;
       GeometricAlgebra.vec_linear :=
         {| Linear.proper := cliff1_ga_vec_proper;
            Linear.is_additive := cliff1_ga_vec_add;
            Linear.is_homogeneous := cliff1_ga_vec_scale |};
       GeometricAlgebra.contraction := cliff1_ga_contract |}.

End Cliff1GA.

Example ga_cliff1_check (F : Type) `{FF : Field F} :
  GeometricAlgebra F F (C F) (Q := neg_form square_form) := GA_Cliff1.
