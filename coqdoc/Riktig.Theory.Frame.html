Families of vectors over a frame: the wedge of a family in the order of the index is linear in each slot, vanishes when a vector repeats, and is the pseudoscalar on the basis.

The wedge of a family of vectors, in the order of the index.
  Local Notation "'wedge_fam' vs" :=
    (natrec A 1 (fun k acc ⇒ acc ⋏ vs (elem k)) card) (at level 10, vs at level 9).

  Lemma wedge_vec_ext : ∀ (x x' : A) (v v' : V), x == x' → v == v' → x ⋏ v == x' ⋏ v'.
  Proof.
    intros x x' v v' Hx Hv.
    exact (ScaleLaws.scale_proper _ _ (reflexivity _) _ _
             (Magma.op_proper _ _ (Magma.op_proper _ _ Hx _ _ (Linear.proper (f := vec) _ _ Hv))
                              _ _ (Magma.op_proper _ _ (Linear.proper (f := vec) _ _ Hv) _ _ (GradeInvolution.involute_proper _ _ Hx)))).
  Qed.

  Lemma wstep_proper : ∀ d : N → V, Proper (equ ==> equ) d →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc ⋏ d k).
  Proof. intros d Hd k k' Hk acc acc' Hacc. apply (wedge_vec_ext acc acc' (d k) (d k') Hacc). apply Hd. exact Hk. Qed.

  Lemma fam_seq_proper : ∀ vs : I → V, Proper (equ ==> equ) vs → Proper (equ ==> equ) (fun k : N ⇒ vs (elem k)).
  Proof. intros vs Hvs k k' Hk. apply Hvs. apply Finite.elem_proper. exact Hk. Qed.

  Theorem wedge_fam_ext : ∀ vs vs' : I → V, Proper (equ ==> equ) vs → Proper (equ ==> equ) vs' →
    (∀ i : I, vs i == vs' i) → wedge_fam vs == wedge_fam vs'.
  Proof.
    intros vs vs' Hvs Hvs' Hvv.
    apply (natrec_ext A _ _ 1 _ _ (wstep_proper _ (fam_seq_proper vs Hvs)) (wstep_proper _ (fam_seq_proper vs' Hvs'))).
    intros k acc. apply (wedge_vec_ext acc acc _ _ (reflexivity acc)). exact (Hvv (elem k)).
  Qed.

Splitting a wedge fold at one position

  Section FoldSplit.
    Context (c c' c'' : N → V)
      (Hc : Proper (equ ==> equ) c) (Hc' : Proper (equ ==> equ) c') (Hc'' : Proper (equ ==> equ) c'')
      (p : N) (Hpc : p < card).

    Local Notation "'W' d n" := (natrec A 1 (fun k acc ⇒ acc ⋏ d k) n) (at level 10, d at level 9, n at level 9).

    Lemma W_proper : ∀ d : N → V, Proper (equ ==> equ) d →
      ∀ n n' : N, n == n' → W d n == W d n'.
    Proof. intros d Hd. exact (NatRec.natrec_proper A _ _ 1 _ (wstep_proper d Hd)). Qed.

    Lemma W_zero : ∀ d : N → V, W d 0 == 1.
    Proof. intros d. exact (NatRec.natrec_zero A _ _ 1 _). Qed.

    Lemma W_succ : ∀ (d : N → V) (n : N), W d (succ n) == W d n ⋏ d n.
    Proof. intros d n. exact (NatRec.natrec_succ A _ _ 1 _ n). Qed.

Columns agreeing except at p, where the first is the sum of the others: the folds agree up to p, and past it the first fold is the sum of the others.
    Section Add.
      Hypothesis Hother : ∀ k : N, k < card → k =/= p → c k == c' k ∧ c k == c'' k.
      Hypothesis Hp : c p == c' p + c'' p.

      Lemma fold_add_below' : ∀ n : N, n ≤ p → n ≤ card → W c n == W c' n.
      Proof.
        apply (Theory.NatOrder.nat_induction_le2 p card (fun n ⇒ W c n == W c' n)).
        - intros n n' Hn. rewrite (W_proper c Hc n n' Hn), (W_proper c' Hc' n n' Hn). reflexivity.
        - calc (W c 0) == (1 : A) { (W_zero c) } == (W c' 0) { (W_zero c') }.
        - intros n Hnp Hnc IH.
          calc (W c (succ n))
            == (W c n ⋏ c n) { (W_succ c n) }
            == (W c' n ⋏ c' n) { (wedge_vec_ext (W c n) (W c' n) (c n) (c' n) IH (proj1 (Hother n Hnc (proj2 Hnp)))) }
            == (W c' (succ n)) { (W_succ c' n) }.
      Qed.

      Lemma fold_add_below'' : ∀ n : N, n ≤ p → n ≤ card → W c n == W c'' n.
      Proof.
        apply (Theory.NatOrder.nat_induction_le2 p card (fun n ⇒ W c n == W c'' n)).
        - intros n n' Hn. rewrite (W_proper c Hc n n' Hn), (W_proper c'' Hc'' n n' Hn). reflexivity.
        - calc (W c 0) == (1 : A) { (W_zero c) } == (W c'' 0) { (W_zero c'') }.
        - intros n Hnp Hnc IH.
          calc (W c (succ n))
            == (W c n ⋏ c n) { (W_succ c n) }
            == (W c'' n ⋏ c'' n) { (wedge_vec_ext (W c n) (W c'' n) (c n) (c'' n) IH (proj2 (Hother n Hnc (proj2 Hnp)))) }
            == (W c'' (succ n)) { (W_succ c'' n) }.
      Qed.

      Theorem fold_split_add : ∀ n : N, p < n → n ≤ card → W c n == W c' n + W c'' n.
      Proof.
        apply (Theory.NatOrder.nat_induction_between p card (fun n ⇒ W c n == W c' n + W c'' n)).
        - intros n n' Hn.
          rewrite (W_proper c Hc n n' Hn), (W_proper c' Hc' n n' Hn), (W_proper c'' Hc'' n n' Hn). reflexivity.
        - calc (W c (succ p))
            == (W c p ⋏ c p) { (W_succ c p) }
            == (W c p ⋏ (c' p + c'' p)) { (wedge_vec_ext (W c p) (W c p) (c p) (c' p + c'' p) (reflexivity _) Hp) }
            == (W c p ⋏ c' p + W c p ⋏ c'' p) { (wedge_vec_add_r (W c p) (c' p) (c'' p)) }
            == (W c' p ⋏ c' p + W c p ⋏ c'' p)
                 { (wedge_vec_ext (W c p) (W c' p) (c' p) (c' p)
                      (fold_add_below' p (Theory.NatOrder.le_refl p) (Theory.NatOrder.le_of_lt p card Hpc)) (reflexivity _)) }
            == (W c' p ⋏ c' p + W c'' p ⋏ c'' p)
                 { (wedge_vec_ext (W c p) (W c'' p) (c'' p) (c'' p)
                      (fold_add_below'' p (Theory.NatOrder.le_refl p) (Theory.NatOrder.le_of_lt p card Hpc)) (reflexivity _)) }
            == (W c' (succ p) + W c'' p ⋏ c'' p) { (W_succ c' p) }
            == (W c' (succ p) + W c'' (succ p)) { (W_succ c'' p) }.
        - intros n Hpn Hnc IH.
          assert (Hnp : n =/= p) by exact (fun Hn ⇒ proj2 Hpn (symmetry Hn)).
          calc (W c (succ n))
            == (W c n ⋏ c n) { (W_succ c n) }
            == ((W c' n + W c'' n) ⋏ c n) { (wedge_vec_ext (W c n) (W c' n + W c'' n) (c n) (c n) IH (reflexivity _)) }
            == (W c' n ⋏ c n + W c'' n ⋏ c n) { (wedge_vec_add_l (W c' n) (W c'' n) (c n)) }
            == (W c' n ⋏ c' n + W c'' n ⋏ c n) { (wedge_vec_ext (W c' n) (W c' n) (c n) (c' n) (reflexivity _) (proj1 (Hother n Hnc Hnp))) }
            == (W c' n ⋏ c' n + W c'' n ⋏ c'' n) { (wedge_vec_ext (W c'' n) (W c'' n) (c n) (c'' n) (reflexivity _) (proj2 (Hother n Hnc Hnp))) }
            == (W c' (succ n) + W c'' n ⋏ c'' n) { (W_succ c' n) }
            == (W c' (succ n) + W c'' (succ n)) { (W_succ c'' n) }.
      Qed.
    End Add.

Columns agreeing except at p, where the first is a multiple of the second: the folds agree up to p, and past it the first fold is that multiple of the second.
    Section Smul.
      Variable a : F.
      Hypothesis Hother : ∀ k : N, k < card → k =/= p → c k == c' k.
      Hypothesis Hp : c p == a *: c' p.

      Lemma fold_smul_below : ∀ n : N, n ≤ p → n ≤ card → W c n == W c' n.
      Proof.
        apply (Theory.NatOrder.nat_induction_le2 p card (fun n ⇒ W c n == W c' n)).
        - intros n n' Hn. rewrite (W_proper c Hc n n' Hn), (W_proper c' Hc' n n' Hn). reflexivity.
        - calc (W c 0) == (1 : A) { (W_zero c) } == (W c' 0) { (W_zero c') }.
        - intros n Hnp Hnc IH.
          calc (W c (succ n))
            == (W c n ⋏ c n) { (W_succ c n) }
            == (W c' n ⋏ c' n) { (wedge_vec_ext (W c n) (W c' n) (c n) (c' n) IH (Hother n Hnc (proj2 Hnp))) }
            == (W c' (succ n)) { (W_succ c' n) }.
      Qed.

      Theorem fold_split_smul : ∀ n : N, p < n → n ≤ card → W c n == a *: W c' n.
      Proof.
        apply (Theory.NatOrder.nat_induction_between p card (fun n ⇒ W c n == a *: W c' n)).
        - intros n n' Hn. rewrite (W_proper c Hc n n' Hn), (W_proper c' Hc' n n' Hn). reflexivity.
        - calc (W c (succ p))
            == (W c p ⋏ c p) { (W_succ c p) }
            == (W c p ⋏ (a *: c' p)) { (wedge_vec_ext (W c p) (W c p) (c p) (a *: c' p) (reflexivity _) Hp) }
            == (a *: (W c p ⋏ c' p)) { (wedge_vec_smul_r a (W c p) (c' p)) }
            == (a *: (W c' p ⋏ c' p))
                 { (wedge_vec_ext (W c p) (W c' p) (c' p) (c' p)
                      (fold_smul_below p (Theory.NatOrder.le_refl p) (Theory.NatOrder.le_of_lt p card Hpc)) (reflexivity _)) }
            == (a *: W c' (succ p)) { (W_succ c' p) }.
        - intros n Hpn Hnc IH.
          calc (W c (succ n))
            == (W c n ⋏ c n) { (W_succ c n) }
            == ((a *: W c' n) ⋏ c n) { (wedge_vec_ext (W c n) (a *: W c' n) (c n) (c n) IH (reflexivity _)) }
            == (a *: (W c' n ⋏ c n)) { (wedge_vec_smul_l a (W c' n) (c n)) }
            == (a *: (W c' n ⋏ c' n))
                 { (wedge_vec_ext (W c' n) (W c' n) (c n) (c' n) (reflexivity _)
                      (Hother n Hnc (fun Hn ⇒ proj2 Hpn (symmetry Hn)))) }
            == (a *: W c' (succ n)) { (W_succ c' n) }.
      Qed.
    End Smul.
  End FoldSplit.

    Lemma sgn_proper : Proper (equ ==> equ) (fun m : N ⇒ sgn m).
    Proof. apply (NatRec.natrec_proper F _ _ 1 _). intros k k' _ s s' Hs. apply Group.neg_proper. exact Hs. Qed.

    Lemma from_step_proper : ∀ i : N, Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc ⋏ c (i + k)).
    Proof.
      intros i k k' Hk acc acc' Hacc. apply (wedge_vec_ext acc acc' (c (i + k)) (c (i + k')) Hacc).
      apply Hc. apply Magma.op_proper; [ reflexivity | exact Hk ].
    Qed.

A vector wedged on before the fold comes out at the end, with the sign of the number of columns it passed.
    Lemma from_wedge : ∀ (y : A) (v : V) (i m : N),
      from (y ⋏ v) i m == sgn m *: (from y i m ⋏ v).
    Proof.
      intros y v i.
      apply (Peano.nat_induction (fun m ⇒ from (y ⋏ v) i m == sgn m *: (from y i m ⋏ v))).
      - intros m m' Hm.
        setoid_rewrite (NatRec.natrec_proper A _ _ (y ⋏ v) _ (from_step_proper i) m m' Hm).
        setoid_rewrite (NatRec.natrec_proper A _ _ y _ (from_step_proper i) m m' Hm).
        setoid_rewrite (sgn_proper m m' Hm). reflexivity.
      - calc (from (y ⋏ v) i 0)
          == (y ⋏ v) { (NatRec.natrec_zero A _ _ (y ⋏ v) _) }
          == ((1 : F) *: (y ⋏ v)) { (left_identity (action := scale) (y ⋏ v)) }
          == ((1 : F) *: (from y i 0 ⋏ v)) { (NatRec.natrec_zero A _ _ y (fun k acc ⇒ acc ⋏ c (i + k))) }
          == (sgn 0 *: (from y i 0 ⋏ v)) { (NatRec.natrec_zero F _ _ 1 (fun (_ : N) (s : F) ⇒ - s)) }.
      - intros m IH.
        transitivity ((- sgn m) *: ((from y i m ⋏ c (i + m)) ⋏ v)).
        2: { calc ((- sgn m) *: ((from y i m ⋏ c (i + m)) ⋏ v))
               == (sgn (succ m) *: ((from y i m ⋏ c (i + m)) ⋏ v)) { (NatRec.natrec_succ F _ _ 1 (fun (_ : N) (s : F) ⇒ - s) m) }
               == (sgn (succ m) *: (from y i (succ m) ⋏ v)) { (NatRec.natrec_succ A _ _ y (fun k acc ⇒ acc ⋏ c (i + k)) m) }. }
        calc (from (y ⋏ v) i (succ m))
          == (from (y ⋏ v) i m ⋏ c (i + m)) { (NatRec.natrec_succ A _ _ (y ⋏ v) _ m) }
          == ((sgn m *: (from y i m ⋏ v)) ⋏ c (i + m)) { (wedge_vec_ext _ _ (c (i + m)) (c (i + m)) IH (reflexivity _)) }
          == (sgn m *: ((from y i m ⋏ v) ⋏ c (i + m))) { (wedge_vec_smul_l (sgn m) (from y i m ⋏ v) (c (i + m))) }
          == (sgn m *: - ((from y i m ⋏ c (i + m)) ⋏ v))
               { (ScaleLaws.scale_proper (sgn m) (sgn m) (reflexivity _) _ _ (wedge_vec_anticomm (from y i m) v (c (i + m)))) }
          == (- (sgn m *: ((from y i m ⋏ c (i + m)) ⋏ v))) { (smul_neg (sgn m) ((from y i m ⋏ c (i + m)) ⋏ v)) }
          == ((- sgn m) *: ((from y i m ⋏ c (i + m)) ⋏ v)) { (neg_smul (sgn m) ((from y i m ⋏ c (i + m)) ⋏ v)) }.
    Qed.

    Lemma from_zero : ∀ i m : N, from 0 i m == 0.
    Proof.
      intros i.
      apply (Peano.nat_induction (fun m ⇒ from 0 i m == 0)).
      - intros m m' Hm. setoid_rewrite (NatRec.natrec_proper A _ _ 0 _ (from_step_proper i) m m' Hm). reflexivity.
      - exact (NatRec.natrec_zero A _ _ 0 _).
      - intros m IH.
        calc (from 0 i (succ m))
          == (from 0 i m ⋏ c (i + m)) { (NatRec.natrec_succ A _ _ 0 _ m) }
          == ((0 : A) ⋏ c (i + m)) { (wedge_vec_ext _ _ (c (i + m)) (c (i + m)) IH (reflexivity _)) }
          == (0 : A) { (wedge_vec_zero_l (c (i + m))) }.
    Qed.

With two equal columns the whole fold vanishes.
    Theorem fold_repeat : ∀ p q : N, p < q → q < card → c p == c q → W card == 0.
    Proof.
      intros p q Hpq Hq Hcpq.
      destruct (proj1 (NatOrder.le_add (succ p) q) (Theory.NatOrder.succ_le_of_lt p q Hpq)) as [m Hm].
      destruct (proj1 (NatOrder.le_add (succ q) card) (Theory.NatOrder.succ_le_of_lt q card Hq)) as [r Hr].
      assert (Hsq : W (succ q) == 0).
      { assert (Hq' : W q == sgn m *: (from (W p) (succ p) m ⋏ c p)).
        { calc (W q)
            == (W (succ p + m)) { (NatRec.natrec_proper A _ _ 1 _ (wstep_proper c Hc) q (succ p + m) (symmetry Hm)) }
            == (from (W (succ p)) (succ p) m) { (natrec_shift A _ _ 1 _ (wstep_proper c Hc) (succ p) m) }
            == (from (W p ⋏ c p) (succ p) m) { (natrec_init A _ _ _ _ _ (from_step_proper (succ p)) (NatRec.natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c k) p) m) }
            == (sgn m *: (from (W p) (succ p) m ⋏ c p)) { (from_wedge (W p) (c p) (succ p) m) }. }
        calc (W (succ q))
          == (W q ⋏ c q) { (NatRec.natrec_succ A _ _ 1 _ q) }
          == ((sgn m *: (from (W p) (succ p) m ⋏ c p)) ⋏ c q) { (wedge_vec_ext _ _ (c q) (c q) Hq' (reflexivity _)) }
          == (sgn m *: ((from (W p) (succ p) m ⋏ c p) ⋏ c q)) { (wedge_vec_smul_l (sgn m) _ (c q)) }
          == (sgn m *: ((from (W p) (succ p) m ⋏ c p) ⋏ c p)) { (wedge_vec_ext _ _ (c q) (c p) (reflexivity _) (symmetry Hcpq)) }
          == (sgn m *: (0 : A)) { (wedge_vec_alternating (from (W p) (succ p) m) (c p)) }
          == (0 : A) { (smul_zero (V := A) (sgn m)) }. }
      calc (W card)
        == (W (succ q + r)) { (NatRec.natrec_proper A _ _ 1 _ (wstep_proper c Hc) card (succ q + r) (symmetry Hr)) }
        == (from (W (succ q)) (succ q) r) { (natrec_shift A _ _ 1 _ (wstep_proper c Hc) (succ q) r) }
        == (from 0 (succ q) r) { (natrec_init A _ _ _ _ _ (from_step_proper (succ q)) Hsq r) }
        == (0 : A) { (from_zero (succ q) r) }.
    Qed.
  End FoldRepeat.

The wedge of a family is linear in each slot, vanishes on a

repeated vector, and is the pseudoscalar on the basis.
  Theorem wedge_fam_upd_add : ∀ (vs : I → V) (j : I) (u v : V), Proper (equ ==> equ) vs →
    wedge_fam (upd vs j (u + v)) == wedge_fam (upd vs j u) + wedge_fam (upd vs j v).
  Proof.
    intros vs j u v Hvs.
    refine (fold_split_add
      (fun k ⇒ upd vs j (u + v) (elem k)) (fun k ⇒ upd vs j u (elem k)) (fun k ⇒ upd vs j v (elem k))
      (fam_seq_proper _ (Upd.upd_proper vs j (u + v) Hvs)) (fam_seq_proper _ (Upd.upd_proper vs j u Hvs))
      (fam_seq_proper _ (Upd.upd_proper vs j v Hvs)) (index j) (Finite.index_lt j) _ _
      card (Finite.index_lt j) (Theory.NatOrder.le_refl card)).
    - intros k Hk Hkj. split.
      + calc (upd vs j (u + v) (elem k)) == (vs (elem k)) { (upd_off_elem vs j (u + v) k Hk Hkj) }
          == (upd vs j u (elem k)) { (upd_off_elem vs j u k Hk Hkj) }.
      + calc (upd vs j (u + v) (elem k)) == (vs (elem k)) { (upd_off_elem vs j (u + v) k Hk Hkj) }
          == (upd vs j v (elem k)) { (upd_off_elem vs j v k Hk Hkj) }.
    - calc (upd vs j (u + v) (elem (index j))) == (u + v) { (upd_at_elem vs j (u + v)) }
        == (upd vs j u (elem (index j)) + v) { (upd_at_elem vs j u) }
        == (upd vs j u (elem (index j)) + upd vs j v (elem (index j))) { (upd_at_elem vs j v) }.
  Qed.

  Theorem wedge_fam_upd_smul : ∀ (vs : I → V) (j : I) (a : F) (u : V), Proper (equ ==> equ) vs →
    wedge_fam (upd vs j (a *: u)) == a *: wedge_fam (upd vs j u).
  Proof.
    intros vs j a u Hvs.
    refine (fold_split_smul
      (fun k ⇒ upd vs j (a *: u) (elem k)) (fun k ⇒ upd vs j u (elem k))
      (fam_seq_proper _ (Upd.upd_proper vs j (a *: u) Hvs)) (fam_seq_proper _ (Upd.upd_proper vs j u Hvs))
      (index j) (Finite.index_lt j) a _ _ card (Finite.index_lt j) (Theory.NatOrder.le_refl card)).
    - intros k Hk Hkj.
      calc (upd vs j (a *: u) (elem k)) == (vs (elem k)) { (upd_off_elem vs j (a *: u) k Hk Hkj) }
        == (upd vs j u (elem k)) { (upd_off_elem vs j u k Hk Hkj) }.
    - calc (upd vs j (a *: u) (elem (index j))) == (a *: u) { (upd_at_elem vs j (a *: u)) }
        == (a *: upd vs j u (elem (index j))) { (upd_at_elem vs j u) }.
  Qed.

  Theorem wedge_fam_repeat : ∀ (vs : I → V) (j j' : I), Proper (equ ==> equ) vs →
    j =/= j' → vs j == vs j' → wedge_fam vs == 0.
  Proof.
    intros vs j j' Hvs Hjj' Hv.
    assert (Hne : index j =/= index j') by (intros Hi; apply Hjj'; exact (index_injective j j' Hi)).
    assert (Hcc : vs (elem (index j)) == vs (elem (index j'))).
    { calc (vs (elem (index j))) == (vs j) { (Hvs _ _ (Finite.elem_index j)) } == (vs j') { Hv }
        == (vs (elem (index j'))) { (Hvs _ _ (symmetry (Finite.elem_index j'))) }. }
    destruct (Theory.NatOrder.le_total (index j) (index j')) as [Hle | Hle].
    - exact (fold_repeat (fun k ⇒ vs (elem k)) (fam_seq_proper vs Hvs) (index j) (index j')
               (Theory.NatOrder.lt_of_le_of_ne _ _ Hle Hne) (Finite.index_lt j') Hcc).
    - exact (fold_repeat (fun k ⇒ vs (elem k)) (fam_seq_proper vs Hvs) (index j') (index j)
               (Theory.NatOrder.lt_of_le_of_ne _ _ Hle (fun H0 ⇒ Hne (symmetry H0))) (Finite.index_lt j) (symmetry Hcc)).
  Qed.
