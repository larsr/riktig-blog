Families of vectors over a frame: the wedge of a family in the order of the index is linear in each slot, vanishes when a vector repeats, and is the pseudoscalar on the basis.

The wedge of a family of vectors, in the order of the index.
  Local Notation "'wedge_fam' vs" :=

The family with slot j replaced by u.
  Local Notation "'upd' vs j u" := (fun i : I ⇒ if equ_dec i j then u else vs i)

  Lemma wedge_vec_ext : ∀ (x x' : A) (v v' : V), x == x' → v == v' → x ⋏ v == x' ⋏ v'.
  Proof.
    intros x x' v v' Hx Hv.
    apply scale_proper; [ reflexivity | ].
    apply op_proper.
    - apply op_proper; [ exact Hx | apply (proper (f := vec)); exact Hv ].
    - apply op_proper; [ apply (proper (f := vec)); exact Hv | apply involute_proper; exact Hx ].
  Qed.

  Lemma wstep_proper : ∀ d : N → V, Proper (equ ==> equ) d →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc ⋏ d k).
  Proof. intros d Hd k k' Hk acc acc' Hacc. apply (wedge_vec_ext acc acc' (d k) (d k') Hacc). apply Hd. exact Hk. Qed.

  Lemma fam_seq_proper : ∀ vs : I → V, Proper (equ ==> equ) vs → Proper (equ ==> equ) (fun k : N ⇒ vs (elem k)).
  Proof. intros vs Hvs k k' Hk. apply Hvs. apply elem_proper. exact Hk. Qed.

  Theorem wedge_fam_ext : ∀ vs vs' : I → V, Proper (equ ==> equ) vs → Proper (equ ==> equ) vs' →
    (∀ i : I, vs i == vs' i) → wedge_fam vs == wedge_fam vs'.
  Proof.
    intros vs vs' Hvs Hvs' Hvv.
    apply (natrec_ext A _ _ 1 _ _ (wstep_proper _ (fam_seq_proper vs Hvs)) (wstep_proper _ (fam_seq_proper vs' Hvs'))).
    intros k acc. apply (wedge_vec_ext acc acc _ _ (reflexivity acc)). exact (Hvv (elem k)).
  Qed.

Columns agreeing except at p, where the first is the sum of the others: the fold is the sum past p.
    Theorem fold_split_add :
      (∀ k : N, k < card → k =/= p → c k == c' k ∧ c k == c'' k) →
      c p == c' p + c'' p →
      ∀ n : N, n ≤ card →
        (n ≤ p → W c n == W c' n ∧ W c n == W c'' n) ∧ (p < n → W c n == W c' n + W c'' n).
    Proof.
      intros Hother Hp.
      apply (nat_induction (fun n ⇒ n ≤ card →
        (n ≤ p → W c n == W c' n ∧ W c n == W c'' n) ∧ (p < n → W c n == W c' n + W c'' n))).
      - intros n n' Hn.
        assert (E : W c n == W c n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c Hc) n n' Hn).
        assert (E' : W c' n == W c' n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c' Hc') n n' Hn).
        assert (E'' : W c'' n == W c'' n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c'' Hc'') n n' Hn).
        split; intros Hall Hcd.
        + assert (Hcd' : n ≤ card) by (setoid_rewrite Hn; exact Hcd).
          destruct (Hall Hcd') as [HA HB]. split.
          × intros Hle. assert (Hle' : n ≤ p) by (setoid_rewrite Hn; exact Hle).
            destruct (HA Hle') as [HA1 HA2]. split.
            { setoid_rewrite <- E. setoid_rewrite <- E'. exact HA1. }
            { setoid_rewrite <- E. setoid_rewrite <- E''. exact HA2. }
          × intros Hlt. setoid_rewrite <- E. setoid_rewrite <- E'. setoid_rewrite <- E''.
            apply HB. exact (proj2 (Theory.NatOrder.lt_proper p p (reflexivity p) n n' Hn) Hlt).
        + assert (Hcd' : n' ≤ card) by (setoid_rewrite <- Hn; exact Hcd).
          destruct (Hall Hcd') as [HA HB]. split.
          × intros Hle. assert (Hle' : n' ≤ p) by (setoid_rewrite <- Hn; exact Hle).
            destruct (HA Hle') as [HA1 HA2]. split.
            { setoid_rewrite E. setoid_rewrite E'. exact HA1. }
            { setoid_rewrite E. setoid_rewrite E''. exact HA2. }
          × intros Hlt. setoid_rewrite E. setoid_rewrite E'. setoid_rewrite E''.
            apply HB. exact (proj1 (Theory.NatOrder.lt_proper p p (reflexivity p) n n' Hn) Hlt).
      - intros _. split.
        + intros _. split.
          × calc (W c 0) == (1 : A) { (natrec_zero A _ _ 1 _) } == (W c' 0) { (natrec_zero A _ _ 1 (fun k acc ⇒ acc ⋏ c' k)) }.
          × calc (W c 0) == (1 : A) { (natrec_zero A _ _ 1 _) } == (W c'' 0) { (natrec_zero A _ _ 1 (fun k acc ⇒ acc ⋏ c'' k)) }.
        + intros Hlt. destruct (Theory.NatOrder.not_lt_zero p Hlt).
      - intros n IH Hsn.
        assert (Hn : n ≤ card) by exact (Theory.NatOrder.le_trans n (succ n) card (Theory.NatOrder.le_succ_diag n) Hsn).
        assert (Hlt : n < card) by exact (Theory.NatOrder.lt_of_lt_of_le n (succ n) card (Theory.NatOrder.lt_succ_diag n) Hsn).
        destruct (IH Hn) as [HA HB].
        pose proof (natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c k) n) as S.
        pose proof (natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c' k) n) as S'.
        pose proof (natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c'' k) n) as S''.
        simpl in S, S', S''.
        destruct (Theory.Naturals.eq_or_ne n p) as [Hnp | Hnp].
        + assert (Hle : n ≤ p) by (setoid_rewrite Hnp; exact (Theory.NatOrder.le_refl p)).
          destruct (HA Hle) as [HA1 HA2].
          split.
          × intros Hle'. exfalso. apply (Theory.NatOrder.lt_irrefl n).
            apply (Theory.NatOrder.lt_of_lt_of_le n (succ n) n (Theory.NatOrder.lt_succ_diag n)).
            assert (Hback : p ≤ n) by (setoid_rewrite Hnp; exact (Theory.NatOrder.le_refl p)).
            exact (Theory.NatOrder.le_trans (succ n) p n Hle' Hback).
          × intros _.
            assert (Hcn : c n == c' n + c'' n).
            { calc (c n) == (c p) { (Hc n p Hnp) } == (c' p + c'' p) { Hp }
                == (c' n + c'' p) { (Hc' p n (symmetry Hnp)) } == (c' n + c'' n) { (Hc'' p n (symmetry Hnp)) }. }
            calc (W c (succ n))
              == (W c n ⋏ c n) { S }
              == (W c n ⋏ (c' n + c'' n)) { (wedge_vec_ext (W c n) (W c n) (c n) (c' n + c'' n) (reflexivity _) Hcn) }
              == (W c n ⋏ c' n + W c n ⋏ c'' n) { (wedge_vec_add_r (W c n) (c' n) (c'' n)) }
              == (W c' n ⋏ c' n + W c n ⋏ c'' n) { (wedge_vec_ext (W c n) (W c' n) (c' n) (c' n) HA1 (reflexivity _)) }
              == (W c' n ⋏ c' n + W c'' n ⋏ c'' n) { (wedge_vec_ext (W c n) (W c'' n) (c'' n) (c'' n) HA2 (reflexivity _)) }
              == (W c' (succ n) + W c'' n ⋏ c'' n) { S' }
              == (W c' (succ n) + W c'' (succ n)) { S'' }.
        + destruct (Hother n Hlt Hnp) as [Hn' Hn''].
          split.
          × intros Hle'.
            assert (Hle : n ≤ p) by exact (Theory.NatOrder.le_trans n (succ n) p (Theory.NatOrder.le_succ_diag n) Hle').
            destruct (HA Hle) as [HA1 HA2]. split.
            { calc (W c (succ n)) == (W c n ⋏ c n) { S }
                == (W c' n ⋏ c' n) { (wedge_vec_ext (W c n) (W c' n) (c n) (c' n) HA1 Hn') }
                == (W c' (succ n)) { S' }. }
            { calc (W c (succ n)) == (W c n ⋏ c n) { S }
                == (W c'' n ⋏ c'' n) { (wedge_vec_ext (W c n) (W c'' n) (c n) (c'' n) HA2 Hn'') }
                == (W c'' (succ n)) { S'' }. }
          × intros Hlt'.
            assert (Hpn : p < n).
            { destruct (proj1 (Theory.NatOrder.lt_succ_iff p n) Hlt') as [Hpn | Hpn]; [ exact Hpn | ].
              destruct (Hnp (symmetry Hpn)). }
            calc (W c (succ n))
              == (W c n ⋏ c n) { S }
              == ((W c' n + W c'' n) ⋏ c n) { (wedge_vec_ext (W c n) (W c' n + W c'' n) (c n) (c n) (HB Hpn) (reflexivity _)) }
              == (W c' n ⋏ c n + W c'' n ⋏ c n) { (wedge_vec_add_l (W c' n) (W c'' n) (c n)) }
              == (W c' n ⋏ c' n + W c'' n ⋏ c n) { (wedge_vec_ext (W c' n) (W c' n) (c n) (c' n) (reflexivity _) Hn') }
              == (W c' n ⋏ c' n + W c'' n ⋏ c'' n) { (wedge_vec_ext (W c'' n) (W c'' n) (c n) (c'' n) (reflexivity _) Hn'') }
              == (W c' (succ n) + W c'' n ⋏ c'' n) { S' }
              == (W c' (succ n) + W c'' (succ n)) { S'' }.
    Qed.

Columns agreeing except at p, where the first is a multiple of the second: the fold is that multiple past p.
    Theorem fold_split_smul : ∀ a : F,
      (∀ k : N, k < card → k =/= p → c k == c' k) →
      c p == a *: c' p →
      ∀ n : N, n ≤ card →
        (n ≤ p → W c n == W c' n) ∧ (p < n → W c n == a *: W c' n).
    Proof.
      intros a Hother Hp.
      apply (nat_induction (fun n ⇒ n ≤ card →
        (n ≤ p → W c n == W c' n) ∧ (p < n → W c n == a *: W c' n))).
      - intros n n' Hn.
        assert (E : W c n == W c n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c Hc) n n' Hn).
        assert (E' : W c' n == W c' n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c' Hc') n n' Hn).

        + assert (Hcd' : n ≤ card) by (setoid_rewrite Hn; exact Hcd).

          × intros Hle. assert (Hle' : n ≤ p) by (setoid_rewrite Hn; exact Hle).

          × intros Hlt. setoid_rewrite <- E. setoid_rewrite <- E'.
            apply HB. exact (proj2 (Theory.NatOrder.lt_proper p p (reflexivity p) n n' Hn) Hlt).
        + assert (Hcd' : n' ≤ card) by (setoid_rewrite <- Hn; exact Hcd).

          × intros Hle. assert (Hle' : n' ≤ p) by (setoid_rewrite <- Hn; exact Hle).

          × intros Hlt. setoid_rewrite E. setoid_rewrite E'.

        + intros Hlt. destruct (Theory.NatOrder.not_lt_zero p Hlt).
      - intros n IH Hsn.
        assert (Hn : n ≤ card) by exact (Theory.NatOrder.le_trans n (succ n) card (Theory.NatOrder.le_succ_diag n) Hsn).
        assert (Hlt : n < card) by exact (Theory.NatOrder.lt_of_lt_of_le n (succ n) card (Theory.NatOrder.lt_succ_diag n) Hsn).
        destruct (IH Hn) as [HA HB].
        pose proof (natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c k) n) as S.
        pose proof (natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c' k) n) as S'.
        simpl in S, S'.
        destruct (Theory.Naturals.eq_or_ne n p) as [Hnp | Hnp].
        + assert (Hle : n ≤ p) by (setoid_rewrite Hnp; exact (Theory.NatOrder.le_refl p)).
          split.
          × intros Hle'. exfalso. apply (Theory.NatOrder.lt_irrefl n).
            apply (Theory.NatOrder.lt_of_lt_of_le n (succ n) n (Theory.NatOrder.lt_succ_diag n)).
            assert (Hback : p ≤ n) by (setoid_rewrite Hnp; exact (Theory.NatOrder.le_refl p)).
            exact (Theory.NatOrder.le_trans (succ n) p n Hle' Hback).
          × intros _.
            assert (Hcn : c n == a *: c' n).
            { calc (c n) == (c p) { (Hc n p Hnp) } == (a *: c' p) { Hp } == (a *: c' n) { (Hc' p n (symmetry Hnp)) }. }
            calc (W c (succ n))
              == (W c n ⋏ c n) { S }
              == (W c n ⋏ (a *: c' n)) { (wedge_vec_ext (W c n) (W c n) (c n) (a *: c' n) (reflexivity _) Hcn) }
              == (a *: (W c n ⋏ c' n)) { (wedge_vec_smul_r a (W c n) (c' n)) }
              == (a *: (W c' n ⋏ c' n)) { (wedge_vec_ext (W c n) (W c' n) (c' n) (c' n) (HA Hle) (reflexivity _)) }
              == (a *: W c' (succ n)) { S' }.

          split.
          × intros Hle'.
            assert (Hle : n ≤ p) by exact (Theory.NatOrder.le_trans n (succ n) p (Theory.NatOrder.le_succ_diag n) Hle').
            calc (W c (succ n)) == (W c n ⋏ c n) { S }
              == (W c' n ⋏ c' n) { (wedge_vec_ext (W c n) (W c' n) (c n) (c' n) (HA Hle) Hn') }
              == (W c' (succ n)) { S' }.

              == (W c n ⋏ c n) { S }

              == (a *: W c' (succ n)) { S' }.
    Qed.

    Lemma sgn_proper : Proper (equ ==> equ) (fun m : N ⇒ sgn m).

A vector wedged on before the fold comes out at the end, with the sign of the number of columns it passed.
    Lemma from_wedge : ∀ (y : A) (v : V) (i m : N),
      from (y ⋏ v) i m == sgn m *: (from y i m ⋏ v).
    Proof.
      intros y v i.
      apply (nat_induction (fun m ⇒ from (y ⋏ v) i m == sgn m *: (from y i m ⋏ v))).
      - intros m m' Hm.
        setoid_rewrite (natrec_proper A _ _ (y ⋏ v) _ (from_step_proper i) m m' Hm).
        setoid_rewrite (natrec_proper A _ _ y _ (from_step_proper i) m m' Hm).
        setoid_rewrite (sgn_proper m m' Hm). reflexivity.
      - calc (from (y ⋏ v) i 0)
          == (y ⋏ v) { (natrec_zero A _ _ (y ⋏ v) _) }
          == ((1 : F) *: (y ⋏ v)) { (left_identity (action := scale) (y ⋏ v)) }
          == ((1 : F) *: (from y i 0 ⋏ v)) { (natrec_zero A _ _ y (fun k acc ⇒ acc ⋏ c (i + k))) }
          == (sgn 0 *: (from y i 0 ⋏ v)) { (natrec_zero F _ _ 1 (fun (_ : N) (s : F) ⇒ - s)) }.
      - intros m IH.
        transitivity ((- sgn m) *: ((from y i m ⋏ c (i + m)) ⋏ v)).
        2: { calc ((- sgn m) *: ((from y i m ⋏ c (i + m)) ⋏ v))
               == (sgn (succ m) *: ((from y i m ⋏ c (i + m)) ⋏ v)) { (natrec_succ F _ _ 1 (fun (_ : N) (s : F) ⇒ - s) m) }
               == (sgn (succ m) *: (from y i (succ m) ⋏ v)) { (natrec_succ A _ _ y (fun k acc ⇒ acc ⋏ c (i + k)) m) }. }
        calc (from (y ⋏ v) i (succ m))
          == (from (y ⋏ v) i m ⋏ c (i + m)) { (natrec_succ A _ _ (y ⋏ v) _ m) }
          == ((sgn m *: (from y i m ⋏ v)) ⋏ c (i + m)) { (wedge_vec_ext _ _ (c (i + m)) (c (i + m)) IH (reflexivity _)) }
          == (sgn m *: ((from y i m ⋏ v) ⋏ c (i + m))) { (wedge_vec_smul_l (sgn m) (from y i m ⋏ v) (c (i + m))) }
          == (sgn m *: - ((from y i m ⋏ c (i + m)) ⋏ v))
               { (scale_proper (sgn m) (sgn m) (reflexivity _) _ _ (wedge_vec_anticomm (from y i m) v (c (i + m)))) }
          == (- (sgn m *: ((from y i m ⋏ c (i + m)) ⋏ v))) { (smul_neg (sgn m) ((from y i m ⋏ c (i + m)) ⋏ v)) }
          == ((- sgn m) *: ((from y i m ⋏ c (i + m)) ⋏ v)) { (neg_smul (sgn m) ((from y i m ⋏ c (i + m)) ⋏ v)) }.
    Qed.

    Lemma from_zero : ∀ i m : N, from 0 i m == 0.
    Proof.
      intros i.
      apply (nat_induction (fun m ⇒ from 0 i m == 0)).
      - intros m m' Hm. setoid_rewrite (natrec_proper A _ _ 0 _ (from_step_proper i) m m' Hm). reflexivity.
      - exact (natrec_zero A _ _ 0 _).
      - intros m IH.
        calc (from 0 i (succ m))
          == (from 0 i m ⋏ c (i + m)) { (natrec_succ A _ _ 0 _ m) }
          == ((0 : A) ⋏ c (i + m)) { (wedge_vec_ext _ _ (c (i + m)) (c (i + m)) IH (reflexivity _)) }
          == (0 : A) { (wedge_vec_zero_l (c (i + m))) }.
    Qed.

With two equal columns the whole fold vanishes.
    Theorem fold_repeat : ∀ p q : N, p < q → q < card → c p == c q → W card == 0.
    Proof.
      intros p q Hpq Hq Hcpq.
      destruct (proj1 (le_add (succ p) q) (Theory.NatOrder.succ_le_of_lt p q Hpq)) as [m Hm].
      destruct (proj1 (le_add (succ q) card) (Theory.NatOrder.succ_le_of_lt q card Hq)) as [r Hr].
      assert (Hsq : W (succ q) == 0).
      { assert (Hq' : W q == sgn m *: (from (W p) (succ p) m ⋏ c p)).
        { calc (W q)
            == (W (succ p + m)) { (natrec_proper A _ _ 1 _ (wstep_proper c Hc) q (succ p + m) (symmetry Hm)) }
            == (from (W (succ p)) (succ p) m) { (natrec_shift A _ _ 1 _ (wstep_proper c Hc) (succ p) m) }
            == (from (W p ⋏ c p) (succ p) m) { (natrec_init A _ _ _ _ _ (from_step_proper (succ p)) (natrec_succ A _ _ 1 (fun k acc ⇒ acc ⋏ c k) p) m) }
            == (sgn m *: (from (W p) (succ p) m ⋏ c p)) { (from_wedge (W p) (c p) (succ p) m) }. }
        calc (W (succ q))
          == (W q ⋏ c q) { (natrec_succ A _ _ 1 _ q) }
          == ((sgn m *: (from (W p) (succ p) m ⋏ c p)) ⋏ c q) { (wedge_vec_ext _ _ (c q) (c q) Hq' (reflexivity _)) }
          == (sgn m *: ((from (W p) (succ p) m ⋏ c p) ⋏ c q)) { (wedge_vec_smul_l (sgn m) _ (c q)) }

repeated vector, and is the pseudoscalar on the basis.
  Theorem wedge_fam_upd_add : ∀ (vs : I → V) (j : I) (u v : V), Proper (equ ==> equ) vs →
    wedge_fam (upd vs j (u + v)) == wedge_fam (upd vs j u) + wedge_fam (upd vs j v).
  Proof.
    intros vs j u v Hvs.
    refine (proj2 (fold_split_add
      (fun k ⇒ upd vs j (u + v) (elem k)) (fun k ⇒ upd vs j u (elem k)) (fun k ⇒ upd vs j v (elem k))
      (fam_seq_proper _ (upd_proper vs j (u + v) Hvs)) (fam_seq_proper _ (upd_proper vs j u Hvs))
      (fam_seq_proper _ (upd_proper vs j v Hvs)) (index j) _ _ card (Theory.NatOrder.le_refl card)) (index_lt j)).
    - intros k Hk Hkj. split.
      + calc (upd vs j (u + v) (elem k)) == (vs (elem k)) { (upd_off vs j (u + v) k Hk Hkj) }
          == (upd vs j u (elem k)) { (upd_off vs j u k Hk Hkj) }.
      + calc (upd vs j (u + v) (elem k)) == (vs (elem k)) { (upd_off vs j (u + v) k Hk Hkj) }
          == (upd vs j v (elem k)) { (upd_off vs j v k Hk Hkj) }.
    - calc (upd vs j (u + v) (elem (index j))) == (u + v) { (upd_at vs j (u + v)) }
        == (upd vs j u (elem (index j)) + v) { (upd_at vs j u) }
        == (upd vs j u (elem (index j)) + upd vs j v (elem (index j))) { (upd_at vs j v) }.
  Qed.

  Theorem wedge_fam_upd_smul : ∀ (vs : I → V) (j : I) (a : F) (u : V), Proper (equ ==> equ) vs →
    wedge_fam (upd vs j (a *: u)) == a *: wedge_fam (upd vs j u).
  Proof.
    intros vs j a u Hvs.
    refine (proj2 (fold_split_smul
      (fun k ⇒ upd vs j (a *: u) (elem k)) (fun k ⇒ upd vs j u (elem k))
      (fam_seq_proper _ (upd_proper vs j (a *: u) Hvs)) (fam_seq_proper _ (upd_proper vs j u Hvs))
      (index j) a _ _ card (Theory.NatOrder.le_refl card)) (index_lt j)).

      calc (upd vs j (a *: u) (elem k)) == (vs (elem k)) { (upd_off vs j (a *: u) k Hk Hkj) }
        == (upd vs j u (elem k)) { (upd_off vs j u k Hk Hkj) }.
    - calc (upd vs j (a *: u) (elem (index j))) == (a *: u) { (upd_at vs j (a *: u)) }
        == (a *: upd vs j u (elem (index j))) { (upd_at vs j u) }.
  Qed.

  Theorem wedge_fam_repeat : ∀ (vs : I → V) (j j' : I), Proper (equ ==> equ) vs →
    j =/= j' → vs j == vs j' → wedge_fam vs == 0.
  Proof.
    intros vs j j' Hvs Hjj' Hv.
    assert (Hne : index j =/= index j') by (intros Hi; apply Hjj'; exact (index_injective j j' Hi)).
    assert (Hcc : vs (elem (index j)) == vs (elem (index j'))).
    { calc (vs (elem (index j))) == (vs j) { (Hvs _ _ (elem_index j)) } == (vs j') { Hv }
        == (vs (elem (index j'))) { (Hvs _ _ (symmetry (elem_index j'))) }. }
    destruct (Theory.NatOrder.le_total (index j) (index j')) as [Hle | Hle].
    - exact (fold_repeat (fun k ⇒ vs (elem k)) (fam_seq_proper vs Hvs) (index j) (index j')
               (Theory.NatOrder.lt_of_le_of_ne _ _ Hle Hne) (index_lt j') Hcc).
    - exact (fold_repeat (fun k ⇒ vs (elem k)) (fam_seq_proper vs Hvs) (index j') (index j)
               (Theory.NatOrder.lt_of_le_of_ne _ _ Hle (fun H0 ⇒ Hne (symmetry H0))) (index_lt j) (symmetry Hcc)).
  Qed.
