Matrices over a frame: the column vectors of a square matrix, the wedge of all the columns in the order of the index, and the identity matrix, whose columns are the basis.
Section FrameTheory.
  Context {F V A I N : Type} `{Frame F V A I N} {EI : @EquDec I _}.

  Local Notation "'col' M j" :=
    (natrec V 0 (fun k acc ⇒ acc + M (elem k) j *: basis (elem k)) card) (at level 10, M at level 9, j at level 9).
  Local Notation "'wedge_cols' M" :=
    (natrec A 1 (fun k acc ⇒ acc ⋏ col M (elem k)) card) (at level 10, M at level 9).
  Local Notation "'proper_matrix' M" := (Proper (equ ==> equ ==> equ) M) (at level 10, M at level 9).

  Lemma col_step_proper : ∀ (M : I → I → F) (j : I), proper_matrix M →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : V) ⇒ acc + M (elem k) j *: basis (elem k)).
  Proof.
    intros M j HM k k' Hk acc acc' Hacc.
    apply op_proper; [ exact Hacc | ].
    apply scale_proper; [ apply HM; [ apply elem_proper; exact Hk | reflexivity ] | ].
    apply basis_proper. apply elem_proper. exact Hk.
  Qed.

  Lemma col_ext : ∀ (M M' : I → I → F) (j : I), proper_matrix M → proper_matrix M' →
    (∀ i k : I, M i k == M' i k) → col M j == col M' j.
  Proof.
    intros M M' j HM HM' HMM'.
    apply (natrec_ext V _ _ 0 _ _ (col_step_proper M j HM) (col_step_proper M' j HM')).
    intros k acc. apply op_proper; [ reflexivity | ].
    apply scale_proper; [ exact (HMM' (elem k) j) | reflexivity ].
  Qed.

  Lemma col_proper : ∀ M : I → I → F, proper_matrix M → Proper (equ ==> equ) (fun j : I ⇒ col M j).
  Proof.
    intros M HM j j' Hj.
    apply (natrec_ext V _ _ 0 _ _ (col_step_proper M j HM) (col_step_proper M j' HM)).
    intros k acc. apply op_proper; [ reflexivity | ].
    apply scale_proper; [ apply HM; [ reflexivity | exact Hj ] | reflexivity ].
  Qed.

  Lemma wedge_step_proper : ∀ M : I → I → F, proper_matrix M →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : A) ⇒ acc ⋏ col M (elem k)).
  Proof.
    intros M HM k k' Hk acc acc' Hacc.
    assert (Hc : col M (elem k) == col M (elem k')) by (apply (col_proper M HM); apply elem_proper; exact Hk).
    apply scale_proper; [ reflexivity | ].
    apply op_proper.
    - apply op_proper; [ exact Hacc | apply (proper (f := vec)); exact Hc ].
    - apply op_proper; [ apply (proper (f := vec)); exact Hc | apply involute_proper; exact Hacc ].
  Qed.

  Lemma wedge_cols_ext : ∀ M M' : I → I → F, proper_matrix M → proper_matrix M' →
    (∀ i k : I, M i k == M' i k) → wedge_cols M == wedge_cols M'.
  Proof.
    intros M M' HM HM' HMM'.
    apply (natrec_ext A _ _ 1 _ _ (wedge_step_proper M HM) (wedge_step_proper M' HM')).
    intros k acc.
    assert (Hc : col M (elem k) == col M' (elem k)) by exact (col_ext M M' (elem k) HM HM' HMM').
    apply scale_proper; [ reflexivity | ].
    apply op_proper.
    - apply op_proper; [ reflexivity | apply (proper (f := vec)); exact Hc ].
    - apply op_proper; [ apply (proper (f := vec)); exact Hc | reflexivity ].
  Qed.

The identity matrix.
  Local Notation Id := (fun i j : I ⇒ if equ_dec i j then (1 : F) else 0).

  Lemma id_proper : proper_matrix Id.
  Proof.
    intros i i' Hi j j' Hj.
    destruct (equ_dec i j) as [Hij | Hij]; destruct (equ_dec i' j') as [Hij' | Hij']; try reflexivity.
    - destruct Hij'. exact (transitivity (symmetry Hi) (transitivity Hij Hj)).
    - destruct Hij. exact (transitivity Hi (transitivity Hij' (symmetry Hj))).
  Qed.

The partial column sums of the identity: zero before the index of the column, the basis vector from there on.
  Lemma col_id_partial : ∀ (j : I) (n : N), n ≤ card →
    (n ≤ index j → natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n == 0)
    ∧ (index j < n → natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n == basis j).
  Proof.
    intros j.
    apply (nat_induction (fun n ⇒ n ≤ card →
      (n ≤ index j → natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n == 0)
      ∧ (index j < n → natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n == basis j))).
    - intros n n' Hn.
      assert (He : natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n
                   == natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n')
        by exact (natrec_proper V _ _ 0 _ (col_step_proper Id j id_proper) n n' Hn).
      split; intros Hall Hc.
      + assert (Hc' : n ≤ card) by (setoid_rewrite Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite <- He. apply HZ. setoid_rewrite Hn. exact Hle.
        × intros Hlt. setoid_rewrite <- He. apply HB.
          exact (proj2 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
      + assert (Hc' : n' ≤ card) by (setoid_rewrite <- Hn; exact Hc).
        destruct (Hall Hc') as [HZ HB]. split.
        × intros Hle. setoid_rewrite He. apply HZ. setoid_rewrite <- Hn. exact Hle.
        × intros Hlt. setoid_rewrite He. apply HB.
          exact (proj1 (Theory.NatOrder.lt_proper _ _ (reflexivity _) n n' Hn) Hlt).
    - intros _. split.
      + intros _. exact (natrec_zero V _ _ 0 _).
      + intros Hlt. destruct (Theory.NatOrder.not_lt_zero (index j) Hlt).
    - intros n IH Hsn.
      assert (Hn : n ≤ card) by exact (Theory.NatOrder.le_trans n (succ n) card (Theory.NatOrder.le_succ_diag n) Hsn).
      destruct (IH Hn) as [HZ HB].
      assert (Hlt : n < card) by exact (Theory.NatOrder.lt_of_lt_of_le n (succ n) card (Theory.NatOrder.lt_succ_diag n) Hsn).
      pose proof (natrec_succ V _ _ 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n) as Hs.
      simpl in Hs.
      destruct (equ_dec (elem n) j) as [Hej | Hej].
      + assert (Hnj : n == index j).
        { calc n == (index (I := I) (N := N) (elem (I := I) (N := N) n)) { (index_elem (I := I) (N := N) n Hlt) } == (index j) { (index_proper (elem n) j Hej) }. }
        split.
        × intros Hle. exfalso. apply (Theory.NatOrder.lt_irrefl n).
          apply (Theory.NatOrder.lt_of_lt_of_le n (succ n) n (Theory.NatOrder.lt_succ_diag n)).
          assert (Hback : index j ≤ n) by (setoid_rewrite Hnj; exact (Theory.NatOrder.le_refl (index j))).
          exact (Theory.NatOrder.le_trans (succ n) (index j) n Hle Hback).
        × intros _.
          assert (Hle0 : n ≤ index j) by (setoid_rewrite <- Hnj; exact (Theory.NatOrder.le_refl n)).
          calc (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) (succ n))
            == (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n + (1 : F) *: basis (elem n)) { Hs }
            == (0 + (1 : F) *: basis (elem n)) { (HZ Hle0) }
            == ((1 : F) *: basis (elem n)) { (left_identity (action := add) ((1 : F) *: basis (elem n))) }
            == (basis (elem n)) { (left_identity (action := scale) (basis (elem n))) }
            == (basis j) { (basis_proper (elem n) j Hej) }.
      + assert (Heq : natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) (succ n)
                      == natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n).
        { calc (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) (succ n))
            == (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n + (0 : F) *: basis (elem n)) { Hs }
            == (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n + 0) { (zero_smul (basis (elem n))) }
            == (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n)
                 { (right_identity (action := add) (natrec V 0 (fun k acc ⇒ acc + Id (elem k) j *: basis (elem k)) n)) }. }
        split.
        × intros Hle. setoid_rewrite Heq. apply HZ.
          exact (Theory.NatOrder.le_trans n (succ n) (index j) (Theory.NatOrder.le_succ_diag n) Hle).
        × intros Hlt'. setoid_rewrite Heq. apply HB.
          destruct (proj1 (Theory.NatOrder.lt_succ_iff (index j) n) Hlt') as [Hlt'' | Heq'']; [ exact Hlt'' | ].
          exfalso. apply Hej.
          calc (elem (I := I) (N := N) n)
            == (elem (I := I) (N := N) (index j)) { (elem_proper n (index j) (symmetry Heq'')) }
            == j { (elem_index j) }.
  Qed.

  Theorem col_id : ∀ j : I, col Id j == basis j.
  Proof.

    exact (proj2 (col_id_partial j card (Theory.NatOrder.le_refl card)) (index_lt j)).
  Qed.

  Theorem wedge_cols_id : top (wedge_cols Id) == 1.
  Proof.
    apply (transitivity (y := top (natrec A 1 (fun k acc ⇒
      inv (1 + 1) *: (acc × vec (basis (elem k)) + vec (basis (elem k)) × involute acc)) card)));
      [ | exact top_pseudoscalar ].
    apply top_proper.

      + apply op_proper; [ exact Hacc | apply (proper (f := vec)); apply basis_proper; apply elem_proper; exact Hk ].

      + apply op_proper; [ reflexivity | apply (proper (f := vec)); exact (col_id (elem k)) ].
      + apply op_proper; [ apply (proper (f := vec)); exact (col_id (elem k)) | reflexivity ].
  Qed.

The matrix with column j replaced by u.

The vector with the coordinates u.
  Local Notation "'vecof' u" := (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) card)
    (at level 10, u at level 9).

  Lemma upd_proper : ∀ (M : I → I → F) (j : I) (u : I → F),
    proper_matrix M → Proper (equ ==> equ) u → proper_matrix (upd M j u).
  Proof.
    intros M j u HM Hu i i' Hi k k' Hk.
    destruct (equ_dec k j) as [Hkj | Hkj]; destruct (equ_dec k' j) as [Hkj' | Hkj'].
    - apply Hu. exact Hi.
    - destruct Hkj'. exact (transitivity (symmetry Hk) Hkj).
    - destruct Hkj. exact (transitivity Hk Hkj').
    - apply HM; assumption.
  Qed.

  Lemma col_upd_same : ∀ (M : I → I → F) (j : I) (u : I → F),
    proper_matrix M → Proper (equ ==> equ) u → col (upd M j u) j == vecof u.
  Proof.
    intros M j u HM Hu.
    apply (natrec_ext V _ _ 0 _ _ (col_step_proper (upd M j u) j (upd_proper M j u HM Hu))).
    - intros k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
      apply scale_proper; [ apply Hu; apply elem_proper; exact Hk | apply basis_proper; apply elem_proper; exact Hk ].
    - intros k acc. apply op_proper; [ reflexivity | ]. apply scale_proper; [ | reflexivity ].
      destruct (equ_dec j j) as [_ | Hjj]; [ reflexivity | destruct (Hjj (reflexivity j)) ].
  Qed.

  Lemma col_upd_other : ∀ (M : I → I → F) (j k : I) (u : I → F),
    proper_matrix M → Proper (equ ==> equ) u → k =/= j → col (upd M j u) k == col M k.
  Proof.
    intros M j k u HM Hu Hkj.
    apply (natrec_ext V _ _ 0 _ _ (col_step_proper (upd M j u) k (upd_proper M j u HM Hu)) (col_step_proper M k HM)).
    intros l acc. apply op_proper; [ reflexivity | ]. apply scale_proper; [ | reflexivity ].
    destruct (equ_dec k j) as [Hkj' | _]; [ destruct (Hkj Hkj') | reflexivity ].
  Qed.

  Lemma vecof_step_proper : ∀ u : I → F, Proper (equ ==> equ) u →
    Proper (equ ==> equ ==> equ) (fun (k : N) (acc : V) ⇒ acc + u (elem k) *: basis (elem k)).
  Proof.
    intros u Hu k k' Hk acc acc' Hacc. apply op_proper; [ exact Hacc | ].
    apply scale_proper; [ apply Hu; apply elem_proper; exact Hk | apply basis_proper; apply elem_proper; exact Hk ].
  Qed.

  Lemma vecof_add : ∀ u v : I → F, Proper (equ ==> equ) u → Proper (equ ==> equ) v →
    vecof (fun i ⇒ u i + v i) == vecof u + vecof v.
  Proof.
    intros u v Hu Hv.
    assert (Huv : Proper (equ ==> equ) (fun i ⇒ u i + v i))
      by (intros i i' Hi; apply op_proper; [ apply Hu | apply Hv ]; exact Hi).
    apply (nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) n
      == natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n
         + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper V _ _ 0 _ (vecof_step_proper _ Huv) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (vecof_step_proper u Hu) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (vecof_step_proper v Hv) n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) 0)
        == (0 : V) { (natrec_zero V _ _ 0 _) }
        == ((0 : V) + 0) { (right_identity (action := add) (0 : V)) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) 0 + 0) { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k))) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) 0
            + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + (u (elem k) + v (elem k)) *: basis (elem k)) n
            + (u (elem n) + v (elem n)) *: basis (elem n))
             { (natrec_succ V _ _ 0 _ n) }
        == ((natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n)
            + (u (elem n) + v (elem n)) *: basis (elem n))
             { IH }
        == ((natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n
             + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n)
            + (u (elem n) *: basis (elem n) + v (elem n) *: basis (elem n)))
             { (right_distributive (op := scale) (u (elem n)) (v (elem n)) (basis (elem n))) }
        == ((natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + u (elem n) *: basis (elem n))
            + (natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n + v (elem n) *: basis (elem n)))
             { (add_add_add_comm _ _ _ _) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) (succ n)
            + (natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n + v (elem n) *: basis (elem n)))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n) }
        == (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) (succ n)
            + natrec V 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) (succ n))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + v (elem k) *: basis (elem k)) n) }.
  Qed.

  Lemma vecof_smul : ∀ (a : F) (u : I → F), Proper (equ ==> equ) u →
    vecof (fun i ⇒ a × u i) == a *: vecof u.
  Proof.
    intros a u Hu.
    assert (Hau : Proper (equ ==> equ) (fun i ⇒ a × u i))
      by (intros i i' Hi; apply op_proper; [ reflexivity | apply Hu; exact Hi ]).
    apply (nat_induction (fun n ⇒
      natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) n
      == a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n)).
    - intros n n' Hn.
      setoid_rewrite (natrec_proper V _ _ 0 _ (vecof_step_proper _ Hau) n n' Hn).
      setoid_rewrite (natrec_proper V _ _ 0 _ (vecof_step_proper u Hu) n n' Hn). reflexivity.
    - calc (natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) 0)
        == (0 : V) { (natrec_zero V _ _ 0 _) }
        == (a *: (0 : V)) { (smul_zero (V := V) a) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) 0)
             { (natrec_zero V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k))) }.
    - intros n IH.
      calc (natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) (succ n))
        == (natrec V 0 (fun k acc ⇒ acc + (a × u (elem k)) *: basis (elem k)) n + (a × u (elem n)) *: basis (elem n))
             { (natrec_succ V _ _ 0 _ n) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + (a × u (elem n)) *: basis (elem n))
             { IH }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + a *: (u (elem n) *: basis (elem n)))
             { (associative (op := mult) (action := scale) a (u (elem n)) (basis (elem n))) }
        == (a *: (natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n + u (elem n) *: basis (elem n)))
             { (left_distributive (op := scale) a _ _) }
        == (a *: natrec V 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) (succ n))
             { (natrec_succ V _ _ 0 (fun k acc ⇒ acc + u (elem k) *: basis (elem k)) n) }.
  Qed.

    Lemma wstep_proper : ∀ d : N → V, Proper (equ ==> equ) d →

      apply scale_proper; [ reflexivity | ].
      apply op_proper.
      - apply op_proper; [ exact Hacc | apply (proper (f := vec)); apply Hd; exact Hk ].
      - apply op_proper; [ apply (proper (f := vec)); apply Hd; exact Hk | apply involute_proper; exact Hacc ].
    Qed.

    Proof.
      intros x x' v v' Hx Hv.
      apply scale_proper; [ reflexivity | ].
      apply op_proper.
      - apply op_proper; [ exact Hx | apply (proper (f := vec)); exact Hv ].
      - apply op_proper; [ apply (proper (f := vec)); exact Hv | apply involute_proper; exact Hx ].

Columns agreeing except at p, where the first is the sum of the others: the fold is the sum past p.
    Theorem fold_split_add :
      (∀ k : N, k < card → k =/= p → c k == c' k ∧ c k == c'' k) →
      c p == c' p + c'' p →
      ∀ n : N, n ≤ card →
        (n ≤ p → W c n == W c' n ∧ W c n == W c'' n) ∧ (p < n → W c n == W c' n + W c'' n).
    Proof.
      intros Hother Hp.
      apply (nat_induction (fun n ⇒ n ≤ card →
        (n ≤ p → W c n == W c' n ∧ W c n == W c'' n) ∧ (p < n → W c n == W c' n + W c'' n))).
      - intros n n' Hn.
        assert (E : W c n == W c n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c Hc) n n' Hn).
        assert (E' : W c' n == W c' n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c' Hc') n n' Hn).
        assert (E'' : W c'' n == W c'' n') by exact (natrec_proper A _ _ 1 _ (wstep_proper c'' Hc'') n n' Hn).
        split; intros Hall Hcd.
        + assert (Hcd' : n ≤ card) by (setoid_rewrite Hn; exact Hcd).
          destruct (Hall Hcd') as [HA HB]. split.

  Lemma col_eq_of_columns : ∀ (M : I → I → F) (j j' : I), proper_matrix M →
    (∀ i : I, M i j == M i j') → col M j == col M j'.
  Proof.
    intros M j j' HM Hjj'.
    apply (natrec_ext V _ _ 0 _ _ (col_step_proper M j HM) (col_step_proper M j' HM)).
    intros k acc. apply op_proper; [ reflexivity | ]. apply scale_proper; [ exact (Hjj' (elem k)) | reflexivity ].
  Qed.

  Theorem wedge_cols_repeat : ∀ (M : I → I → F) (j j' : I), proper_matrix M →
    j =/= j' → (∀ i : I, M i j == M i j') → wedge_cols M == 0.
  Proof.
    intros M j j' HM Hjj' Hcols.
    assert (Hne : index j =/= index j') by (intros Hi; apply Hjj'; exact (index_injective j j' Hi)).
    assert (Hcc : col M (elem (index j)) == col M (elem (index j'))).
    { calc (col M (elem (index j)))
        == (col M j) { (col_proper M HM (elem (index j)) j (elem_index j)) }
        == (col M j') { (col_eq_of_columns M j j' HM Hcols) }
        == (col M (elem (index j'))) { (col_proper M HM j' (elem (index j')) (symmetry (elem_index j'))) }. }
    destruct (Theory.NatOrder.le_total (index j) (index j')) as [Hle | Hle].
    - exact (fold_repeat (fun k ⇒ col M (elem k)) (col_seq_proper M HM) (index j) (index j')
               (Theory.NatOrder.lt_of_le_of_ne _ _ Hle Hne) (index_lt j') Hcc).
    - exact (fold_repeat (fun k ⇒ col M (elem k)) (col_seq_proper M HM) (index j') (index j)
               (Theory.NatOrder.lt_of_le_of_ne _ _ Hle (fun H0 ⇒ Hne (symmetry H0))) (index_lt j) (symmetry Hcc)).
  Qed.
