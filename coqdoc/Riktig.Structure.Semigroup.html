Library Riktig.Structure.Semigroup

From Riktig.Structure Require Export Magma.

A semigroup is a magma whose operation is associative.
Module Semigroup.
  Class Semigroup (S : Type) {eq : EqRel S} (op : BinOp S) := {
    magma :: Magma S op;
    is_associative :: Associative S S op op;
  }.
  Hint Mode Semigroup ! - - : typeclass_instances.
  Hint Rewrite @associative : algebra_db.
End Semigroup.
Export Semigroup.
