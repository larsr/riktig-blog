From Riktig Require Export Props.

The Peano numbers: zero is no successor, the successor is injective, and a property that holds at zero and passes from each number to its successor holds of every number.
Module Peano.
  Class Peano (N : Type) {eq : EqRel N} (zero : Zero N) (succ : HasSucc N) := {
    equivalence :: Equivalence eq;
    succ_proper :: Proper (equ ==> equ) succ;
    succ_injective :: Injective N N succ;
    succ_not_zero : ∀ n : N, succ n =/= 0;
    nat_induction : ∀ P : N → Prop, Proper (equ ==> iff) P →
      P 0 → (∀ n : N, P n → P (succ n)) → ∀ n : N, P n;
  }.
  Hint Mode Peano ! - - - : typeclass_instances.
End Peano.
Export Peano.
