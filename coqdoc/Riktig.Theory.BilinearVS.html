From Coq Require Export Morphisms.
From Riktig Require Export Theory.Hom.

Section BilinearVS.
  Context {F U V W : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.

  Local Notation Bilin := (BilinearMap F U V W).


  #[global] Instance Bilin_Zero : Zero Bilin :=
    bilinear_zero_map.

  Section BilinPlus.
    Context (b b' : Bilin).

    #[global] Instance bilin_plus_proper :
      Proper (equ ==> equ ==> equ) (fun (u : U) (v : V) ⇒ b u v + b' u v).
    Proof.
      intros u u' Hu v v' Hv.
      rewrite (bilinear_map_proper b u u' Hu v v' Hv), (bilinear_map_proper b' u u' Hu v v' Hv).
      reflexivity.
    Qed.

    Lemma bilin_plus_add_l : ∀ (u u' : U) (v : V),
      b (u + u') v + b' (u + u') v == (b u v + b' u v) + (b u' v + b' u' v).
    Proof.
      intros u u' v.
      rewrite (bilinear_left_additive b u u' v), (bilinear_left_additive b' u u' v).
      apply add_interchange.
    Qed.

    Lemma bilin_plus_scale_l : ∀ (a : F) (u : U) (v : V),
      b (a *: u) v + b' (a *: u) v == a *: (b u v + b' u v).
    Proof.
      intros a u v.
      rewrite (bilinear_left_scale b a u v), (bilinear_left_scale b' a u v).
      symmetry.
      exact (left_distributive _ _ _).
    Qed.

    Lemma bilin_plus_add_r : ∀ (u : U) (v v' : V),
      b u (v + v') + b' u (v + v') == (b u v + b' u v) + (b u v' + b' u v').
    Proof.
      intros u v v'.
      rewrite (bilinear_right_additive b u v v'), (bilinear_right_additive b' u v v').
      apply add_interchange.
    Qed.

    Lemma bilin_plus_scale_r : ∀ (a : F) (u : U) (v : V),
      b u (a *: v) + b' u (a *: v) == a *: (b u v + b' u v).
    Proof.
      intros a u v.
      rewrite (bilinear_right_scale b a u v), (bilinear_right_scale b' a u v).
      symmetry.
      exact (left_distributive _ _ _).
    Qed.

    Definition bilin_plus : Bilin :=
      bilinear_map_from_laws
        (fun (u : U) (v : V) ⇒ b u v + b' u v)
        bilin_plus_proper
        bilin_plus_add_l
        bilin_plus_scale_l
        bilin_plus_add_r
        bilin_plus_scale_r.

  End BilinPlus.

  #[global] Instance Bilin_Add : HasAdd Bilin := bilin_plus.

  Section BilinOpp.
    Context (b : Bilin).

    #[global] Instance bilin_opp_proper :
      Proper (equ ==> equ ==> equ) (fun (u : U) (v : V) ⇒ - b u v).
    Proof.
      intros u u' Hu v v' Hv.
      rewrite (bilinear_map_proper b u u' Hu v v' Hv).
      reflexivity.
    Qed.

    Lemma bilin_opp_add_l : ∀ (u u' : U) (v : V),
      - b (u + u') v == - b u v + - b u' v.
    Proof.
      intros u u' v.
      rewrite (bilinear_left_additive b u u' v).
      apply neg_add.
    Qed.

    Lemma bilin_opp_scale_l : ∀ (a : F) (u : U) (v : V),
      - b (a *: u) v == a *: - b u v.
    Proof.
      intros a u v.
      rewrite (bilinear_left_scale b a u v).
      symmetry.
      apply scale_neg.
    Qed.

    Lemma bilin_opp_add_r : ∀ (u : U) (v v' : V),
      - b u (v + v') == - b u v + - b u v'.
    Proof.
      intros u v v'.
      rewrite (bilinear_right_additive b u v v').
      apply neg_add.
    Qed.

    Lemma bilin_opp_scale_r : ∀ (a : F) (u : U) (v : V),
      - b u (a *: v) == a *: - b u v.
    Proof.
      intros a u v.
      rewrite (bilinear_right_scale b a u v).
      symmetry.
      apply scale_neg.
    Qed.

    Definition bilin_opp : Bilin :=
      bilinear_map_from_laws
        (fun (u : U) (v : V) ⇒ - b u v)
        bilin_opp_proper
        bilin_opp_add_l
        bilin_opp_scale_l
        bilin_opp_add_r
        bilin_opp_scale_r.

  End BilinOpp.

  #[global] Instance Bilin_Neg : HasNeg Bilin := bilin_opp.

  Section BilinSmul.
    Context (a : F) (b : Bilin).

    #[global] Instance bilin_smul_proper :
      Proper (equ ==> equ ==> equ) (fun (u : U) (v : V) ⇒ a *: b u v).
    Proof.
      intros u u' Hu v v' Hv.
      rewrite (bilinear_map_proper b u u' Hu v v' Hv).
      reflexivity.
    Qed.

    Lemma bilin_smul_add_l : ∀ (u u' : U) (v : V),
      a *: b (u + u') v == a *: b u v + a *: b u' v.
    Proof.
      intros u u' v.
      rewrite (bilinear_left_additive b u u' v).
      exact (left_distributive _ _ _).
    Qed.

    Lemma bilin_smul_scale_l : ∀ (c : F) (u : U) (v : V),
      a *: b (c *: u) v == c *: (a *: b u v).
    Proof.
      intros c u v.
      rewrite (bilinear_left_scale b c u v).
      apply scale_swap.
    Qed.

    Lemma bilin_smul_add_r : ∀ (u : U) (v v' : V),
      a *: b u (v + v') == a *: b u v + a *: b u v'.
    Proof.
      intros u v v'.
      rewrite (bilinear_right_additive b u v v').
      exact (left_distributive _ _ _).
    Qed.

    Lemma bilin_smul_scale_r : ∀ (c : F) (u : U) (v : V),
      a *: b u (c *: v) == c *: (a *: b u v).
    Proof.
      intros c u v.
      rewrite (bilinear_right_scale b c u v).
      apply scale_swap.
    Qed.

    Definition bilin_smul : Bilin :=
      bilinear_map_from_laws
        (fun (u : U) (v : V) ⇒ a *: b u v)
        bilin_smul_proper
        bilin_smul_add_l
        bilin_smul_scale_l
        bilin_smul_add_r
        bilin_smul_scale_r.

  End BilinSmul.

  #[global] Instance Bilin_Scale : HasScale F Bilin := bilin_smul.

  Lemma bilin_assoc : ∀ b c d : Bilin, (b + c) + d == b + (c + d).
  Proof.
    intros b c d u v.
    exact (associative (b u v) (c u v) (d u v)).
  Qed.

  Lemma bilin_idl : ∀ b : Bilin, 0 + b == b.
  Proof.
    intros b u v.
    exact (left_identity (b u v)).
  Qed.

  Lemma bilin_idr : ∀ b : Bilin, b + 0 == b.
  Proof.
    intros b u v.
    exact (right_identity (b u v)).
  Qed.

  #[global] Instance bilin_add_proper : Proper (equ ==> equ ==> equ) (@add Bilin Bilin_Add).
  Proof.
    intros b b' Hb c c' Hc u v.
    exact (Magma.op_proper (b u v) (b' u v) (Hb u v)
             (c u v) (c' u v) (Hc u v)).
  Qed.

  Lemma bilin_invl : ∀ b : Bilin, (- b) + b == 0.
  Proof.
    intros b u v.
    exact (left_inverse (b u v)).
  Qed.

  Lemma bilin_invr : ∀ b : Bilin, b + (- b) == 0.
  Proof.
    intros b u v.
    exact (right_inverse (b u v)).
  Qed.

  #[global] Instance bilin_neg_proper : Proper (equ ==> equ) (@neg Bilin Bilin_Neg).
  Proof.
    intros b b' Hb u v.
    exact (Group.neg_proper (b u v) (b' u v) (Hb u v)).
  Qed.

  #[global] Instance Bilin_Group : Group Bilin :=
    {| Group.monoid :=
         {| Monoid.semigroup :=
              {| Semigroup.magma :=
                   {| Magma.equivalence := Bilin_Equivalence;
                      Magma.op_proper := bilin_add_proper |};
                 Semigroup.is_associative := bilin_assoc |};
            Monoid.is_left_identity := bilin_idl;
            Monoid.is_right_identity := bilin_idr |};
       Group.neg_proper := bilin_neg_proper;
       Group.is_left_inverse := bilin_invl;
       Group.is_right_inverse := bilin_invr |}.

  Lemma bilin_comm : ∀ b c : Bilin, b + c == c + b.
  Proof.
    intros b c u v.
    exact (commutative (b u v) (c u v)).
  Qed.

  #[global] Instance Bilin_AbGroup : AbGroup Bilin :=
    {| AbGroup.group :=
         {| Group.monoid :=
              {| Monoid.semigroup :=
                   {| Semigroup.magma :=
                        {| Magma.equivalence := Bilin_Equivalence;
                           Magma.op_proper := bilin_add_proper |};
                      Semigroup.is_associative := bilin_assoc |};
                 Monoid.is_left_identity := bilin_idl;
                 Monoid.is_right_identity := bilin_idr |};
            Group.neg_proper := bilin_neg_proper;
            Group.is_left_inverse := bilin_invl;
            Group.is_right_inverse := bilin_invr |};
       AbGroup.is_commutative := bilin_comm |}.

  #[global] Instance bilin_sc_proper : Proper (equ ==> equ ==> equ) (@scale F Bilin Bilin_Scale).
  Proof.
    intros a a' Ha b b' Hb u v.
    exact (ScaleLaws.scale_proper a a' Ha (b u v) (b' u v) (Hb u v)).
  Qed.

  Lemma bilin_scale_add : ∀ (a : F) (b c : Bilin),
    a *: (b + c) == a *: b + a *: c.
  Proof.
    intros a b c u v.
    exact (left_distributive a (b u v) (c u v)).
  Qed.

  Lemma bilin_add_scale : ∀ (a c : F) (b : Bilin),
    (a + c) *: b == a *: b + c *: b.
  Proof.
    intros a c b u v.
    exact (right_distributive a c (b u v)).
  Qed.

  Lemma bilin_scale_scale : ∀ (a c : F) (b : Bilin),
    (a × c) *: b == a *: (c *: b).
  Proof.
    intros a c b u v.
    exact (associative a c (b u v)).
  Qed.

  Lemma bilin_scale_1 : ∀ b : Bilin, 1 *: b == b.
  Proof.
    intros b u v.
    exact (left_identity (b u v)).
  Qed.

  #[global] Instance Bilin_ScaleLaws : ScaleLaws F Bilin :=
    {| ScaleLaws.scale_proper := bilin_sc_proper;
       ScaleLaws.is_left_distributive := bilin_scale_add;
       ScaleLaws.is_right_distributive := bilin_add_scale;
       ScaleLaws.is_associative := bilin_scale_scale;
       ScaleLaws.is_left_identity := bilin_scale_1 |}.

  #[global] Instance Bilin_VectorSpace : VectorSpace F Bilin :=
    {| VectorSpace.vectors := Bilin_AbGroup; VectorSpace.scaling := Bilin_ScaleLaws |}.

End BilinearVS.
