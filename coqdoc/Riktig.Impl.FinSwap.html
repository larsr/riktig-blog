Library Riktig.Impl.FinSwap

From Coq Require Export Arith Lia List.
From Coq.Vectors Require Fin.
From Riktig Require Export Impl.Fin Impl.Swap.
From Riktig Require Import Ops Lib.Calc.

Definition swap_fin {n : nat} (i j r : Fin.t n) : Fin.t n := swap i j r.

Lemma swap_fin_at_i : ∀ (n : nat) (i j : Fin.t n), swap_fin i j i = j.
Proof.
  intros n i j.
  exact (swap_at_i swap i j i eq_refl).
Qed.

Lemma swap_fin_at_j : ∀ (n : nat) (i j : Fin.t n), i ≠ j → swap_fin i j j = i.
Proof.
  intros n i j Hij.
  exact (swap_at_j swap i j j Hij eq_refl).
Qed.

Lemma swap_fin_ne : ∀ (n : nat) (i j r : Fin.t n),
  r ≠ i → r ≠ j → swap_fin i j r = r.
Proof.
  intros n i j r Hri Hrj.
  exact (swap_ne swap i j r Hri Hrj).
Qed.

Lemma swap_fin_F1_fixed : ∀ (n : nat) (a b : Fin.t n),
  swap_fin (Fin.FS a) (Fin.FS b) Fin.F1 = Fin.F1.
Proof.
  intros n a b.
  exact (swap_fin_ne (S n) (Fin.FS a) (Fin.FS b) Fin.F1
           (fin_F1_neq_FS n a) (fin_F1_neq_FS n b)).
Qed.

Lemma swap_fin_FS : ∀ (n : nat) (a b y : Fin.t n),
  swap_fin (Fin.FS a) (Fin.FS b) (Fin.FS y) = Fin.FS (swap_fin a b y).
Proof.
  intros n a b y.
  destruct (Fin.eq_dec y a) as [Heq | Hne].
  - rewrite Heq.
    calc (swap_fin (Fin.FS a) (Fin.FS b) (Fin.FS a))
      == (Fin.FS b) { (swap_fin_at_i (S n) (Fin.FS a) (Fin.FS b)) }
      == (Fin.FS (swap_fin a b a)) { (swap_fin_at_i n a b) }.
  - destruct (Fin.eq_dec y b) as [Heq2 | Hne2].
    + rewrite Heq2.
      destruct (Fin.eq_dec a b) as [Heqab | Hneab].
      × rewrite Heqab.
        calc (swap_fin (Fin.FS b) (Fin.FS b) (Fin.FS b))
          == (Fin.FS b) { (swap_fin_at_i (S n) (Fin.FS b) (Fin.FS b)) }
          == (Fin.FS (swap_fin b b b)) { (swap_fin_at_i n b b) }.
      × rewrite (swap_fin_at_j n a b Hneab),
          (swap_fin_at_j (S n) (Fin.FS a) (Fin.FS b)
             (fun Heq ⇒ Hneab (Fin.FS_inj a b Heq))).
        reflexivity.
    + rewrite (swap_fin_ne n a b y Hne Hne2),
        (swap_fin_ne (S n) (Fin.FS a) (Fin.FS b) (Fin.FS y)
           (fun Heq ⇒ Hne (Fin.FS_inj y a Heq))
           (fun Heq ⇒ Hne2 (Fin.FS_inj y b Heq))).
      reflexivity.
Qed.

Lemma swap_fin_invol : ∀ (n : nat) (i j x : Fin.t n), swap_fin i j (swap_fin i j x) = x.
Proof.
  intros n i j x.
  exact (swap_invol swap i j x).
Qed.

Lemma swap_fin_inj : ∀ (n : nat) (i j x y : Fin.t n),
  swap_fin i j x = swap_fin i j y → x = y.
Proof.
  intros n i j x y.
  exact (swap_inj swap i j x y).
Qed.

Lemma swap_fin_symmetric : ∀ (n : nat) (i j x : Fin.t n), swap_fin i j x = swap_fin j i x.
Proof.
  intros n i j x.
  exact (swap_symmetric swap i j x).
Qed.

Lemma swap_fin_conj : ∀ (n : nat) (i j k : Fin.t n),
  i ≠ j → i ≠ k → j ≠ k →
  ∀ x : Fin.t n,
  swap_fin i j x = swap_fin i k (swap_fin k j (swap_fin i k x)).
Proof.
  intros n i j k Hij Hik Hjk x.
  exact (swap_conj swap i j k Hij Hik Hjk x).
Qed.
