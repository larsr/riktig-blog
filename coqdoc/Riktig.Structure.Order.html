Library Riktig.Structure.Order

From Coq Require Export Morphisms.
From Riktig Require Export Structure.Field.

Class HasLe T := le : Rel T.
Infix "≤" := le (at level 70, no associativity).

Class HasAbs T := abs : T → T.

Hint Mode HasLe ! : typeclass_instances.
Hint Mode HasAbs ! : typeclass_instances.

Typeclasses Transparent HasLe HasAbs.

Definition lt {T : Type} `{EqRel T} `{HasLe T} (a b : T) : Prop :=
  a ≤ b ∧ a =/= b.
Infix "<" := lt (at level 70, no associativity).

An ordered field is a field with a total order compatible with addition and multiplication, together with the absolute value the order determines.
Module OrderedField.
  Class OrderedField (F : Type) `{Field F}
                     {le : HasLe F} {abs : HasAbs F} := {
    le_proper :: Proper (equ ==> equ ==> iff) le;
    abs_proper :: Proper (equ ==> equ) abs;

    le_reflexive :: Reflexive le;
    le_transitive :: Transitive le;
    le_antisymmetric :: Antisymmetric F equ le;
    le_total : ∀ a b : F, a ≤ b ∨ b ≤ a;
    le_stable : ∀ a b : F, ¬ ¬ a ≤ b → a ≤ b;
    add_le_monotone_left : ∀ a b c : F, a ≤ b → c + a ≤ c + b;
    mul_nonneg_closed : ∀ a b : F, 0 ≤ a → 0 ≤ b → 0 ≤ a × b;
    abs_nonneg : ∀ a : F, 0 ≤ abs a;
    abs_zero_iff : ∀ a : F, abs a == 0 ↔ a == 0;
    abs_mul : ∀ a b : F, abs (a × b) == abs a × abs b;
    abs_neg : ∀ a : F, abs (- a) == abs a;
    abs_triangle : ∀ a b : F, abs (a + b) ≤ abs a + abs b;
    abs_nonneg_identity : ∀ a : F, 0 ≤ a → abs a == a;
  }.
  Hint Mode OrderedField ! - - - - - - - - - - : typeclass_instances.
End OrderedField.
Export OrderedField.
