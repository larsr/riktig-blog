From Riktig.Structure Require Export Field.

An ordered field is a field with a total order that respects equality, is stable under double negation, and is kept by adding on the left and by multiplying nonnegatives; its absolute value fixes the nonnegatives and forgets the sign.
Module OrderedField.
  Class OrderedField (F : Type) `{Field F} (le : HasLe F) (abs : HasAbs F) := {
    le_proper :: Proper (equ ==> equ ==> iff) le;
    abs_proper :: Proper (equ ==> equ) abs;
    le_reflexive :: Reflexive le;
    le_transitive :: Transitive le;
    le_antisymmetric :: Antisymmetric F equ le;
    le_total : ∀ a b : F, a ≤ b ∨ b ≤ a;
    le_stable : ∀ a b : F, ¬ ¬ a ≤ b → a ≤ b;
    add_le_mono_l : ∀ a b c : F, a ≤ b → c + a ≤ c + b;
    mul_nonneg : ∀ a b : F, 0 ≤ a → 0 ≤ b → 0 ≤ a × b;
    abs_of_nonneg : ∀ a : F, 0 ≤ a → abs a == a;
    abs_neg : ∀ a : F, abs (- a) == abs a;
  }.
  Hint Mode OrderedField ! - - - - - - - - - - : typeclass_instances.
End OrderedField.
Export OrderedField.
