Definition perm_head {n : nat} (sigma : Perm (S n)) : Fin.t (S n) :=
  match sigma in Perm n0
    return match n0 with O ⇒ unit | S m ⇒ Fin.t (S m) end
  with
  | Perm_O ⇒ tt
  | Perm_S c _ ⇒ c
  end.

Definition perm_tail {n : nat} (sigma : Perm (S n)) : Perm n :=
  match sigma in Perm n0
    return match n0 with O ⇒ unit | S m ⇒ Perm m end
  with
  | Perm_O ⇒ tt
  | Perm_S _ p ⇒ p
  end.

Lemma perm_head_S : ∀ (n : nat) (c : Fin.t (S n)) (p : Perm n),
  perm_head (Perm_S c p) = c.
Proof.
  intros n c p.
  reflexivity.
Qed.

Lemma perm_tail_S : ∀ (n : nat) (c : Fin.t (S n)) (p : Perm n),
  perm_tail (Perm_S c p) = p.
Proof.
  intros n c p.
  reflexivity.
Qed.

Lemma Perm_O_eta : ∀ sigma : Perm O, sigma = Perm_O.
Proof.
  intros sigma.
  exact (match sigma as s in Perm n0
           return match n0 return Perm n0 → Prop with
                  | O ⇒ fun s0 ⇒ s0 = Perm_O
                  | S _ ⇒ fun _ ⇒ True
                  end s
         with
         | Perm_O ⇒ eq_refl
         | Perm_S _ _ ⇒ I
         end).
Qed.

Lemma Perm_S_eta : ∀ (n : nat) (sigma : Perm (S n)),
  sigma = Perm_S (perm_head sigma) (perm_tail sigma).
Proof.
  intros n sigma.
  exact (match sigma as s in Perm n0
           return match n0 return Perm n0 → Prop with
                  | O ⇒ fun _ ⇒ True
                  | S m ⇒ fun s0 ⇒ s0 = Perm_S (perm_head s0) (perm_tail s0)
                  end s
         with
         | Perm_O ⇒ I
         | Perm_S _ _ ⇒ eq_refl
         end).
Qed.

Fixpoint perm_eq_dec {n : nat} : ∀ sigma tau : Perm n,
    {sigma = tau} + {sigma ≠ tau} :=
  match n as n0 return ∀ sigma tau : Perm n0, {sigma = tau} + {sigma ≠ tau} with
  | O ⇒ fun sigma tau ⇒ left (eq_trans (Perm_O_eta sigma) (eq_sym (Perm_O_eta tau)))
  | S m ⇒ fun sigma tau ⇒
      match Fin.eq_dec (perm_head sigma) (perm_head tau) with
      | left Hhead ⇒
          match perm_eq_dec (perm_tail sigma) (perm_tail tau) with
          | left Htail ⇒
              left (eq_trans (Perm_S_eta m sigma)
                     (eq_trans (f_equal2 (@Perm_S m) Hhead Htail)
                        (eq_sym (Perm_S_eta m tau))))
          | right Htail ⇒ right (fun Heq ⇒ Htail (f_equal perm_tail Heq))
          end
      | right Hhead ⇒ right (fun Heq ⇒ Hhead (f_equal perm_head Heq))
      end
  end.

Instance EqDec_Perm {n : nat} : EqDec (Perm n) := perm_eq_dec.


Fixpoint perm_enum (n : nat) : list (Perm n) :=
  match n with
  | O ⇒ cons Perm_O nil
  | S m ⇒ flat_map (fun c : Fin.t (S m) ⇒
             map (fun p : Perm m ⇒ Perm_S c p) (perm_enum m)) (fin_all (S m))
  end.

Lemma perm_enum_complete : ∀ (n : nat) (sigma : Perm n),
  In sigma (perm_enum n).
Proof.
  intros n sigma; induction sigma as [| m c p IH].
  - left. reflexivity.
  - apply in_flat_map.
    ∃ c.
    split.
    + exact (fin_all_complete (S m) c).
    + exact (in_map (fun p0 ⇒ Perm_S c p0) (perm_enum m) p IH).
Qed.

Lemma perm_enum_NoDup : ∀ n : nat, NoDup (perm_enum n).
Proof.
  intros n; induction n as [| m IH].
  - constructor.
    + intros Hin. destruct Hin.
    + constructor.
  - apply (NoDup_flat_map
      (fun c : Fin.t (S m) ⇒ map (fun p : Perm m ⇒ Perm_S c p) (perm_enum m))
      (fin_all (S m))).
    + intros c _.
      apply (NoDup_map_injective (fun p : Perm m ⇒ Perm_S c p) (perm_enum m)).
      × intros p1 p2 Heq. exact (f_equal perm_tail Heq).
      × exact IH.
    + intros c1 c2 x _ _ Hx1 Hx2.
      apply in_map_iff in Hx1. destruct Hx1 as [p1 [Hp1 _]].
      apply in_map_iff in Hx2. destruct Hx2 as [p2 [Hp2 _]].
      exact (f_equal perm_head (eq_trans Hp1 (eq_sym Hp2))).
    + exact (fin_all_NoDup (S m)).
Qed.

Lemma perm_enum_length : ∀ n : nat, length (perm_enum n) = fact n.
Proof.
  intros n; induction n as [| m IH].
  - reflexivity.
  - assert (Hblock : ∀ c : Fin.t (S m), In c (fin_all (S m)) →
      length (map (fun p : Perm m ⇒ Perm_S c p) (perm_enum m)) = fact m).
    { intros c _.
      rewrite (length_map (fun p : Perm m ⇒ Perm_S c p) (perm_enum m)).
      exact IH. }
    calc (length (perm_enum (S m)))
      == ((length (fin_all (S m)) × fact m)%nat)
           { (length_flat_map_constant _ (fact m) (fin_all (S m)) Hblock) }
      == (fact (S m))
           { (fin_all_length (S m)) }.
Qed.

Instance FinIndex_Perm (n : nat) : FinIndex (fact n) (Perm n) :=
  {| FinIndex.enum := perm_enum n;
     FinIndex.enum_NoDup := perm_enum_NoDup n;
     FinIndex.enum_complete := perm_enum_complete n;
     FinIndex.enum_length := perm_enum_length n |}.


Fixpoint perm_of {n : nat} : (Fin.t n → Fin.t n) → Perm n :=
  match n as n0 return (Fin.t n0 → Fin.t n0) → Perm n0 with
  | O ⇒ fun _ ⇒ Perm_O
  | S m ⇒ fun f ⇒ Perm_S (f Fin.F1) (perm_of (fin_restrict f))
  end.

Lemma perm_of_O : ∀ f : Fin.t O → Fin.t O, perm_of f = Perm_O.
Proof.
  intros f.
  reflexivity.
Qed.

Lemma perm_of_S : ∀ (m : nat) (f : Fin.t (S m) → Fin.t (S m)),
  perm_of f = Perm_S (f Fin.F1) (perm_of (fin_restrict f)).
Proof.
  intros m f.
  reflexivity.
Qed.

Lemma perm_of_ext : ∀ (n : nat) (f g : Fin.t n → Fin.t n),
  (∀ x : Fin.t n, f x = g x) → perm_of f = perm_of g.
Proof.
  intros n; induction n as [| m IH]; intros f g Hfg.
  - reflexivity.
  - rewrite (perm_of_S m f), (perm_of_S m g), (Hfg Fin.F1).
    apply f_equal.
    exact (IH (fin_restrict f) (fin_restrict g) (fin_restrict_ext m f g Hfg)).
Qed.

Lemma perm_apply_inj : ∀ (n : nat) (sigma : Perm n) (x y : Fin.t n),
  perm_apply sigma x = perm_apply sigma y → x = y.
Proof.
  intros n sigma; induction sigma as [| m c p IH]; intros x y.
  - exact (Fin.case0 (fun x0 ⇒ perm_apply Perm_O x0 = perm_apply Perm_O y → x0 = y) x).
  - apply (Fin.caseS' x (fun x0 ⇒
      perm_apply (Perm_S c p) x0 = perm_apply (Perm_S c p) y → x0 = y)).
    + apply (Fin.caseS' y (fun y0 ⇒
        perm_apply (Perm_S c p) Fin.F1 = perm_apply (Perm_S c p) y0 → Fin.F1 = y0)).
      × intros _. reflexivity.
      × intros y' Heq.
        rewrite (perm_apply_F1 m c p), (perm_apply_FS m c p y') in Heq.
        destruct (skip_ne m c (perm_apply p y') (eq_sym Heq)).
    + intros x'.
      apply (Fin.caseS' y (fun y0 ⇒
        perm_apply (Perm_S c p) (Fin.FS x') = perm_apply (Perm_S c p) y0 → Fin.FS x' = y0)).
      × intros Heq.
        rewrite (perm_apply_FS m c p x'), (perm_apply_F1 m c p) in Heq.
        destruct (skip_ne m c (perm_apply p x') Heq).
      × intros y' Heq.
        rewrite (perm_apply_FS m c p x'), (perm_apply_FS m c p y') in Heq.
        exact (f_equal Fin.FS
          (IH x' y' (skip_inj m c (perm_apply p x') (perm_apply p y') Heq))).
Qed.

Lemma perm_apply_perm_of : ∀ (n : nat) (f : Fin.t n → Fin.t n),
  (∀ x y : Fin.t n, f x = f y → x = y) →
  ∀ x : Fin.t n, perm_apply (perm_of f) x = f x.
Proof.
  intros n; induction n as [| m IH]; intros f Hinj x.
  - exact (Fin.case0 (fun x0 ⇒ perm_apply (perm_of f) x0 = f x0) x).
  - rewrite (perm_of_S m f).
    apply (Fin.caseS' x (fun x0 ⇒
      perm_apply (Perm_S (f Fin.F1) (perm_of (fin_restrict f))) x0 = f x0)).
    + exact (perm_apply_F1 m (f Fin.F1) (perm_of (fin_restrict f))).
    + intros x'.
      calc (perm_apply (Perm_S (f Fin.F1) (perm_of (fin_restrict f))) (Fin.FS x'))
        == (skip (f Fin.F1) (perm_apply (perm_of (fin_restrict f)) x'))
             { (perm_apply_FS m (f Fin.F1) (perm_of (fin_restrict f)) x') }
        == (skip (f Fin.F1) (fin_restrict f x'))
             { (IH (fin_restrict f) (fin_restrict_inj m f Hinj) x') }
        == (f (Fin.FS x'))
             { (fin_restrict_skip m f Hinj x') }.
Qed.

Lemma perm_of_perm_apply : ∀ (n : nat) (sigma : Perm n),
  perm_of (perm_apply sigma) = sigma.
Proof.
  intros n sigma; induction sigma as [| m c p IH].
  - reflexivity.
  - assert (Hres : ∀ y : Fin.t m,
      fin_restrict (perm_apply (Perm_S c p)) y = perm_apply p y).
    { intros y.
      unfold fin_restrict.
      rewrite (perm_apply_F1 m c p), (perm_apply_FS m c p y),
        (unskip_skip m c (perm_apply p y)).
      reflexivity. }
    calc (perm_of (perm_apply (Perm_S c p)))
      == (Perm_S c (perm_of (fin_restrict (perm_apply (Perm_S c p)))))
           { (perm_of_S m (perm_apply (Perm_S c p))) }
      == (Perm_S c (perm_of (perm_apply p)))
           { (perm_of_ext m (fin_restrict (perm_apply (Perm_S c p))) (perm_apply p) Hres) }
      == (Perm_S c p)
           { IH }.
Qed.


Definition perm_postcompose {n : nat} (sigma : Perm n)
    (tau : Fin.t n → Fin.t n) : Perm n :=
  perm_of (fun x ⇒ tau (perm_apply sigma x)).

Lemma perm_postcompose_ext : ∀ (n : nat) (sigma : Perm n)
    (tau1 tau2 : Fin.t n → Fin.t n),
  (∀ x : Fin.t n, tau1 x = tau2 x) →
  perm_postcompose sigma tau1 = perm_postcompose sigma tau2.
Proof.
  intros n sigma tau1 tau2 Heq.
  apply (perm_of_ext n).
  intros x.
  exact (Heq (perm_apply sigma x)).
Qed.

Lemma perm_postcompose_id : ∀ (n : nat) (sigma : Perm n),
  perm_postcompose sigma (fun x : Fin.t n ⇒ x) = sigma.
Proof.
  intros n sigma.
  exact (perm_of_perm_apply n sigma).
Qed.

Theorem perm_apply_postcompose : ∀ (n : nat) (sigma : Perm n)
    (tau : Fin.t n → Fin.t n),
  (∀ x y : Fin.t n, tau x = tau y → x = y) →
  ∀ x : Fin.t n,
  perm_apply (perm_postcompose sigma tau) x = tau (perm_apply sigma x).
Proof.
  intros n sigma tau Htau x.
  apply (perm_apply_perm_of n (fun x0 ⇒ tau (perm_apply sigma x0))).
  intros x1 x2 Heq.
  exact (perm_apply_inj n sigma x1 x2 (Htau (perm_apply sigma x1) (perm_apply sigma x2) Heq)).
Qed.

Theorem perm_postcompose_compose : ∀ (n : nat) (sigma : Perm n)
    (tau1 tau2 : Fin.t n → Fin.t n),
  (∀ x y : Fin.t n, tau1 x = tau1 y → x = y) →
  perm_postcompose (perm_postcompose sigma tau1) tau2
    = perm_postcompose sigma (fun x ⇒ tau2 (tau1 x)).
Proof.
  intros n sigma tau1 tau2 Hinj1.
  apply (perm_of_ext n).
  intros x.
  rewrite (perm_apply_postcompose n sigma tau1 Hinj1 x).
  reflexivity.
Qed.

Theorem perm_postcompose_inverse : ∀ (n : nat) (sigma : Perm n)
    (tau tau' : Fin.t n → Fin.t n),
  (∀ x : Fin.t n, tau' (tau x) = x) →
  perm_postcompose (perm_postcompose sigma tau) tau' = sigma.
Proof.
  intros n sigma tau tau' H1.
  assert (Hinj1 : ∀ x y : Fin.t n, tau x = tau y → x = y).
  { intros x y Heq. rewrite <- (H1 x), <- (H1 y), Heq. reflexivity. }
  calc (perm_postcompose (perm_postcompose sigma tau) tau')
    == (perm_postcompose sigma (fun x ⇒ tau' (tau x)))
         { (perm_postcompose_compose n sigma tau tau' Hinj1) }
    == (perm_postcompose sigma (fun x : Fin.t n ⇒ x))
         { (perm_postcompose_ext n sigma _ _ H1) }
    == sigma
         { (perm_postcompose_id n sigma) }.
Qed.

Section PermSum.
  Context {F : Type} `{FF : Field F}.

  Lemma psum_enum : ∀ (n : nat) (h : Perm n → F),
    psum h == bigop_list h (perm_enum n).
  Proof.
    intros n; induction n as [| m IH]; intros h.
    - calc (psum h)
        == (h Perm_O + 0) { right_identity }
        == (bigop_list h (cons Perm_O nil)) { (bigop_list_cons h Perm_O nil) }.
    - calc (psum h)
        == (bigopF (fun i : Fin.t (S m) ⇒ psum (fun p : Perm m ⇒ h (Perm_S i p))))
             { (psum_S m h) }
        == (bigop_list (fun i : Fin.t (S m) ⇒ psum (fun p : Perm m ⇒ h (Perm_S i p)))
              (fin_all (S m)))
             { (bigopF_enum (S m) (fun i ⇒ psum (fun p ⇒ h (Perm_S i p)))) }
        == [blocks] (bigop_list (fun i : Fin.t (S m) ⇒
              bigop_list h (map (fun p : Perm m ⇒ Perm_S i p) (perm_enum m)))
              (fin_all (S m)))
        == (bigop_list h (perm_enum (S m)))
             { (bigop_list_flat_map
                 (fun i : Fin.t (S m) ⇒ map (fun p : Perm m ⇒ Perm_S i p) (perm_enum m))
                 h (fin_all (S m))) }.
    [blocks]: apply bigop_list_ext. intros i _.
      calc (psum (fun p : Perm m ⇒ h (Perm_S i p)))
        == (bigop_list (fun p : Perm m ⇒ h (Perm_S i p)) (perm_enum m))
             { (IH (fun p ⇒ h (Perm_S i p))) }
        == (bigop_list h (map (fun p : Perm m ⇒ Perm_S i p) (perm_enum m)))
             { (bigop_list_map (fun p : Perm m ⇒ Perm_S i p) h (perm_enum m)) }.
  Qed.

  Lemma psum_bigopI : ∀ (n : nat) (h : Perm n → F),
    psum h == bigopI h.
  Proof.
    intros n h.
    exact (psum_enum n h).
  Qed.

  Theorem psum_reindex : ∀ (n : nat) (h : Perm n → F)
      (tau tau' : Fin.t n → Fin.t n),
    (∀ x : Fin.t n, tau' (tau x) = x) →
    (∀ y : Fin.t n, tau (tau' y) = y) →
    psum h == psum (fun sigma : Perm n ⇒ h (perm_postcompose sigma tau)).
  Proof.
    intros n h tau tau' H1 H2.
    calc (psum h)
      == (bigopI h)
           { (psum_bigopI n h) }
      == (bigopI (fun sigma : Perm n ⇒ h (perm_postcompose sigma tau)))
           { (bigopI_reindex h
               (fun sigma ⇒ perm_postcompose sigma tau)
               (fun sigma ⇒ perm_postcompose sigma tau')
               (fun sigma ⇒ perm_postcompose_inverse n sigma tau tau' H1)
               (fun sigma ⇒ perm_postcompose_inverse n sigma tau' tau H2)) }
      == (psum (fun sigma : Perm n ⇒ h (perm_postcompose sigma tau)))
           { (psum_bigopI n (fun sigma ⇒ h (perm_postcompose sigma tau))) }.
  Qed.

End PermSum.

Section PermSignSwap.
  Context {F : Type} `{FF : Field F}.
  Add Ring F_ring_sign : (ring_theory_of_Ring (R := F) commutative).

  Theorem sign_at_fweaken_FS : ∀ (n : nat) (k : Fin.t n),
    (sign_at (Fin.FS k) : F) == - sign_at (fin_weaken k).
  Proof.
    intros n k; induction k as [n0 | n0 k' IH].
    - rewrite fin_weaken_F1.
      reflexivity.
    - rewrite fin_weaken_FS, (sign_at_FS (S n0) (Fin.FS k')), IH.
      reflexivity.
  Qed.

  Theorem perm_sign_swap_adjacent : ∀ (n : nat) (k : Fin.t n)
      (g : Fin.t (S n) → Fin.t (S n)),
    (∀ x y : Fin.t (S n), g x = g y → x = y) →
    (perm_sign (perm_of (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x))) : F)
      == - perm_sign (perm_of g).
  Proof.
    intros n; induction n as [| n0 IHn]; intros k g Hinj.
    - exact (Fin.case0 (fun k0 : Fin.t 0 ⇒
        (perm_sign (perm_of (fun x ⇒ swap_fin (fin_weaken k0) (Fin.FS k0) (g x))) : F)
          == - perm_sign (perm_of g)) k).
    - rewrite (perm_of_S (S n0) (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x))),
        (perm_of_S (S n0) g),
        (perm_sign_S (S n0) (swap_fin (fin_weaken k) (Fin.FS k) (g Fin.F1))
           (perm_of (fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x))))),
        (perm_sign_S (S n0) (g Fin.F1) (perm_of (fin_restrict g))).
      destruct (Fin.eq_dec (g Fin.F1) (fin_weaken k)) as [Hc1 | Hc1].
      + rewrite Hc1, (swap_fin_at_i (S (S n0)) (fin_weaken k) (Fin.FS k)),
          (perm_of_ext (S n0)
            (fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x)))
            (fin_restrict g)
            (fin_restrict_swap_adjacent_at_lower (S n0) k g Hinj Hc1)),
          (sign_at_fweaken_FS (S n0) k).
        exact (mul_neg_l (sign_at (fin_weaken k)) (perm_sign (perm_of (fin_restrict g)))).
      + destruct (Fin.eq_dec (g Fin.F1) (Fin.FS k)) as [Hc2 | Hc2].
        × rewrite Hc2,
            (swap_fin_at_j (S (S n0)) (fin_weaken k) (Fin.FS k) (fin_weaken_neq_FS (S n0) k)),
            (perm_of_ext (S n0)
              (fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x)))
              (fin_restrict g)
              (fin_restrict_swap_adjacent_at_upper (S n0) k g Hinj Hc2)),
            (sign_at_fweaken_FS (S n0) k).
          ring.
        × destruct (unskip_adjacent_shape n0 k (g Fin.F1) Hc1 Hc2) as [k' [Hu1 Hu2]].
          rewrite (swap_fin_ne (S (S n0)) (fin_weaken k) (Fin.FS k) (g Fin.F1) Hc1 Hc2),
            (perm_of_ext (S n0)
              (fin_restrict (fun x ⇒ swap_fin (fin_weaken k) (Fin.FS k) (g x)))
              (fun y ⇒ swap_fin (fin_weaken k') (Fin.FS k') (fin_restrict g y))
              (fin_restrict_swap_adjacent_missed n0 k k' g Hinj Hc1 Hc2 Hu1 Hu2)),
            (IHn k' (fin_restrict g) (fin_restrict_inj (S n0) g Hinj)).
          exact (mul_neg_r (sign_at (g Fin.F1)) (perm_sign (perm_of (fin_restrict g)))).
  Qed.

  Theorem perm_sign_postcompose_adj : ∀ (n : nat) (k : Fin.t n)
      (sigma : Perm (S n)),
    (perm_sign (perm_postcompose sigma (swap_fin (fin_weaken k) (Fin.FS k))) : F)
      == - perm_sign sigma.
  Proof.
    intros n k sigma.
    calc (perm_sign (perm_postcompose sigma (swap_fin (fin_weaken k) (Fin.FS k))) : F)
      == (- perm_sign (perm_of (perm_apply sigma)) : F)
           { (perm_sign_swap_adjacent n k (perm_apply sigma) (perm_apply_inj (S n) sigma)) }
      == (- perm_sign sigma : F)
           { (perm_of_perm_apply (S n) sigma) }.
  Qed.

End PermSignSwap.

Section PsumNeg.
  Context {F : Type} `{FF : Field F}.
  Add Ring F_ring_psum_neg : (ring_theory_of_Ring (R := F) commutative).

  Theorem psum_neg : ∀ (n : nat) (f : Perm n → F), psum (fun p ⇒ - f p) == - psum f.
  Proof.
    intros n f.
    calc (psum (fun p ⇒ - f p))
      == (psum (fun p ⇒ (- (1)) × f p))
      == (- psum f).
    - apply psum_ext. intros p. ring.
    - rewrite (psum_scale n (- (1)) f). ring.
  Qed.

End PsumNeg.

Section DetSwapAdjacent.
  Context {F : Type} `{FF : Field F}.

  Theorem detM_leibniz_swap_adjacent : ∀ (n : nat) (k : Fin.t n) (A : Matrix (S n) (S n) F),
    detM_leibniz (row_swap A (fin_weaken k) (Fin.FS k)) == - detM_leibniz A.
  Proof.
    intros n k A.
    calc (detM_leibniz (row_swap A (fin_weaken k) (Fin.FS k)))
      == (psum (fun sigma : Perm (S n) ⇒ perm_sign sigma ×
            prodF (fun c ⇒ A (swap_fin (fin_weaken k) (Fin.FS k) (perm_apply sigma c)) c)))
      == (psum (fun sigma : Perm (S n) ⇒ - (perm_sign sigma × perm_prod A sigma)))
      == (- detM_leibniz A).
    - unfold detM_leibniz at 1.
      apply psum_ext. intros sigma.
      apply op_proper.
      + reflexivity.
      + exact (perm_prod_row_swap (S n) A (fin_weaken k) (Fin.FS k) sigma).
    - rewrite (psum_reindex (S n)
        (fun sigma : Perm (S n) ⇒ perm_sign sigma ×
          prodF (fun c ⇒ A (swap_fin (fin_weaken k) (Fin.FS k) (perm_apply sigma c)) c))
        (swap_fin (fin_weaken k) (Fin.FS k)) (swap_fin (fin_weaken k) (Fin.FS k))
        (swap_fin_invol (S n) (fin_weaken k) (Fin.FS k)) (swap_fin_invol (S n) (fin_weaken k) (Fin.FS k))).
      apply psum_ext. intros sigma.
      transitivity ((- perm_sign sigma) ×
        prodF (fun c ⇒ A (swap_fin (fin_weaken k) (Fin.FS k)
          (perm_apply (perm_postcompose sigma (swap_fin (fin_weaken k) (Fin.FS k))) c)) c)).
      + apply op_proper.
        × exact (perm_sign_postcompose_adj n k sigma).
        × reflexivity.
      + rewrite (mul_neg_l (perm_sign sigma)
          (prodF (fun c ⇒ A (swap_fin (fin_weaken k) (Fin.FS k)
            (perm_apply (perm_postcompose sigma (swap_fin (fin_weaken k) (Fin.FS k))) c)) c))).
        apply neg_proper.
        apply op_proper.
        × reflexivity.
        × unfold perm_prod.
          apply prodF_ext. intros c.
          rewrite (perm_apply_postcompose (S n) sigma (swap_fin (fin_weaken k) (Fin.FS k))
                     (swap_fin_inj (S n) (fin_weaken k) (Fin.FS k)) c),
             (swap_fin_invol (S n) (fin_weaken k) (Fin.FS k) (perm_apply sigma c)).
          reflexivity.
    - exact (psum_neg (S n) (fun sigma ⇒ perm_sign sigma × perm_prod A sigma)).
  Qed.

  Theorem detM_swap_adjacent : ∀ (n : nat) (k : Fin.t n) (A : Matrix (S n) (S n) F),
    detM (row_swap A (fin_weaken k) (Fin.FS k)) == - detM A.
  Proof.
    intros n k A.
    calc (detM (row_swap A (fin_weaken k) (Fin.FS k)))
      == (detM_leibniz (row_swap A (fin_weaken k) (Fin.FS k)))
           { (detM_leibniz_eq_detM (S n) (row_swap A (fin_weaken k) (Fin.FS k))) }
      == (- detM_leibniz A) { (detM_leibniz_swap_adjacent n k A) }
      == (- detM A) { (detM_leibniz_eq_detM (S n) A) }.
  Qed.

End DetSwapAdjacent.

Definition row3 {A : Type} (a b c : A) : Fin.t 3 → A :=
  fun j ⇒ match j with
           | Fin.F1 ⇒ a
           | Fin.FS Fin.F1 ⇒ b
           | Fin.FS (Fin.FS _) ⇒ c
           end.

Definition mat3 {A : Type} (a11 a12 a13 a21 a22 a23 a31 a32 a33 : A) : Matrix 3 3 A :=
  fun i ⇒ match i with
           | Fin.F1 ⇒ row3 a11 a12 a13
           | Fin.FS Fin.F1 ⇒ row3 a21 a22 a23
           | Fin.FS (Fin.FS _) ⇒ row3 a31 a32 a33
           end.
