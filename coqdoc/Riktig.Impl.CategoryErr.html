Library Riktig.Impl.CategoryErr

From Riktig.Structure Require Export Bicartesian Functor.
From Riktig.Theory Require Import Coproducts.

Arrows that may fail: over a bicartesian category and an object e of errors, an arrow from a to b is a base arrow a ~> b + e; the identity never fails and composition propagates the first error.
Definition Err (O : Type) (e : O) : Type := O.
#[global] Typeclasses Opaque Err.

Section ErrCategory.
  Context {O : Type} `{Bicartesian O} (e : O).

  Let err_equ (a b : O) : Rel (a ~> b ⨿ e) := fun f g ⇒ f == g.
  Let err_id (a : O) : a ~> a ⨿ e := inj1.
  Let err_comp (a b c : O) (g : b ~> c ⨿ e) (f : a ~> b ⨿ e) : a ~> c ⨿ e :=
    desc g inj2 ∘ f.
  Let err_fmap (a b : O) (f : a ~> b) : a ~> b ⨿ e := inj1 ∘ f.

  Let err_equivalence (a b : O) : Equivalence (err_equ a b).
  Proof.
    unfold err_equ. split.
    - intros f. reflexivity.
    - intros f g Hfg. symmetry. exact Hfg.
    - intros f g h Hfg Hgh. etransitivity; eassumption.
  Qed.

  Let err_comp_proper (a b c : O) (g g' : b ~> c ⨿ e) (f f' : a ~> b ⨿ e) :
    g == g' → f == f' → err_comp a b c g f == err_comp a b c g' f'.
  Proof. intros Hg Hf. unfold err_comp. rewrite Hf, Hg. reflexivity. Qed.

  Let err_left_id (a b : O) (f : a ~> b ⨿ e) : err_comp a b b (err_id b) f == f.
  Proof.
    unfold err_comp, err_id. rewrite (desc_inj (O := O)). apply left_identity.
  Qed.

  Let err_right_id (a b : O) (f : a ~> b ⨿ e) : err_comp a a b f (err_id a) == f.
  Proof. unfold err_comp, err_id. apply Coproducts.desc_inj1. Qed.

  Let err_assoc (w x y z : O) (f : y ~> z ⨿ e) (g : x ~> y ⨿ e) (h : w ~> x ⨿ e) :
    err_comp w y z f (err_comp w x y g h) == err_comp w x z (err_comp x y z f g) h.
  Proof.
    unfold err_comp.
    rewrite (Category.comp_assoc (O := O)). rewrite (comp_desc (O := O)).
    rewrite (Coproducts.desc_inj2 (O := O)). reflexivity.
  Qed.

  Let err_fmap_proper (a b : O) (f g : a ~> b) :
    f == g → err_fmap a b f == err_fmap a b g.
  Proof. intros Hfg. unfold err_fmap. rewrite Hfg. reflexivity. Qed.

  Let err_fmap_id (a : O) : err_fmap a a cat_id == err_id a.
  Proof. unfold err_fmap, err_id. apply right_identity. Qed.

  Let err_fmap_comp (a b c : O) (f : b ~> c) (g : a ~> b) :
    err_fmap a c (f ∘ g) == err_comp a b c (err_fmap b c f) (err_fmap a b g).
  Proof.
    unfold err_fmap, err_comp. symmetry.
    rewrite (Category.comp_assoc (O := O)). rewrite (Coproducts.desc_inj1 (O := O)).
    rewrite <- (Category.comp_assoc (O := O)). reflexivity.
  Qed.

  #[export] Instance Err_HasHom : HasHom (Err O e) :=
    fun a b ⇒ (a : O) ~> (b : O) ⨿ e.

  #[export] Instance Err_EqRel : ∀ a b : Err O e, EqRel (a ~> b) := err_equ.

  #[export] Instance Err_HasId : HasId (Err O e) := err_id.

  #[export] Instance Err_HasComp : HasComp (Err O e) := err_comp.

  #[export] Instance Err_Category : Category (Err O e) Err_HasId Err_HasComp.
  Proof.
    split.
    - intros a b. exact (err_equivalence a b).
    - intros a b c g g' Hg f f' Hf. exact (err_comp_proper a b c g g' f f' Hg Hf).
    - intros a b f. exact (err_left_id a b f).
    - intros a b f. exact (err_right_id a b f).
    - intros w x y z f g h. exact (err_assoc w x y z f g h).
  Qed.

Programs that cannot fail run over the wire unchanged.
  #[export] Instance Err_HasFmap : HasFmap O (Err O e) (fun a : O ⇒ a) := err_fmap.

  #[export] Instance Err_Functor : Functor O (Err O e) (fun a : O ⇒ a) Err_HasFmap.
  Proof.
    split.
    - intros a b f g Hfg. exact (err_fmap_proper a b f g Hfg).
    - intros a. exact (err_fmap_id a).
    - intros a b c f g. exact (err_fmap_comp a b c f g).
  Qed.
End ErrCategory.
