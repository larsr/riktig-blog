The commutative semiring laws give the recursion equations: each is the successor written as adding one, then a semiring law.
Section NaturalsTheory.
  Context {N : Type} `{Naturals N}.

  Theorem one_succ : (1 : N) == succ (0 : N).
  Proof.
    calc (1 : N)
      == ((0 : N) + 1) { left_identity }
      == (succ (0 : N)) { Naturals.succ_add }.
  Qed.

  Theorem add_zero : ∀ a : N, a + 0 == a.
  Proof. intros a. exact (right_identity a). Qed.

  Theorem add_succ : ∀ a b : N, a + succ b == succ (a + b).
  Proof.
    intros a b.
    calc (a + succ b)
      == (a + (b + 1)) { Naturals.succ_add }
      == ((a + b) + 1) { associative }
      == (succ (a + b)) { Naturals.succ_add }.
  Qed.

  Theorem mul_zero : ∀ a : N, a × 0 == 0.
  Proof. intros a. exact (right_absorbing a). Qed.

  Theorem mul_succ : ∀ a b : N, a × succ b == a × b + a.
  Proof.
    intros a b.
    calc (a × succ b)
      == (a × (b + 1)) { Naturals.succ_add }
      == (a × b + a × 1) { left_distributive }
      == (a × b + a) { (right_identity (action := mult) a) }.
  Qed.

Induction: the predicate is given to Peano.nat_induction, which asks that it respect ==, then holds at zero, then passes to the successor.
  Theorem zero_or_succ : ∀ n : N, n == 0 ∨ ∃ m : N, n == succ m.
  Proof.
    apply (Peano.nat_induction (fun n ⇒ n == 0 ∨ ∃ m : N, n == succ m)).
    - intros x y Hxy.
      exact (or_iff_morphism _ _ (equiv_iff Hxy (reflexivity 0))
                             _ _ (ex_iff_morphism _ _ (fun m ⇒ equiv_iff Hxy (reflexivity (succ m))))).
    - left. reflexivity.
    - intros n _. right. ∃ n. reflexivity.
  Qed.

  Theorem succ_ne_self : ∀ n : N, succ n =/= n.
  Proof.
    apply (Peano.nat_induction (fun n ⇒ succ n =/= n)).
    - intros x y Hxy. exact (not_iff_morphism _ _ (equiv_iff (Peano.succ_proper x y Hxy) Hxy)).
    - exact (Peano.succ_not_zero 0).
    - intros n IH Heq. exact (IH (injective (f := succ) (succ n) n Heq)).
  Qed.

  Theorem eq_zero_of_add_eq_zero : ∀ a b : N, a + b == 0 → a == 0.
  Proof.
    intros a.
    apply (Peano.nat_induction (fun b ⇒ a + b == 0 → a == 0)).
    - intros x y Hxy. split; intros Hx Hy; apply Hx; [ rewrite Hxy | rewrite <- Hxy ]; exact Hy.
    - intros Ha.
      calc a
        == (a + 0) { right_identity }
        == (0 : N) { Ha }.
    - intros b _ Hab. exfalso. apply (Peano.succ_not_zero (a + b)).
      calc (succ (a + b))
        == (a + succ b) { add_succ }
        == (0 : N) { Hab }.
  Qed.

  Theorem add_succ_l : ∀ a b : N, succ a + b == succ (a + b).
  Proof.
    intros a b.
    calc (succ a + b)
      == (b + succ a) { commutative }
      == (succ (b + a)) { add_succ }
      == (succ (a + b)) { (commutative b a) }.
  Qed.

  Theorem add_one_l : ∀ n : N, 1 + n == succ n.
  Proof.
    intros n.
    calc ((1 : N) + n)
      == (succ 0 + n) { (Magma.op_proper _ _ one_succ n n (reflexivity n)) }
      == (succ (0 + n)) { (add_succ_l 0 n) }
      == (succ n) { (Peano.succ_proper _ _ (left_identity (action := add) n)) }.
  Qed.

  Theorem add_left_cancel : ∀ a b c : N, a + b == a + c → b == c.
  Proof.
    apply (Peano.nat_induction (fun a ⇒ ∀ b c : N, a + b == a + c → b == c)).
    - intros a a' Ha. split; intros Hall b c Heq; apply (Hall b c).
      + setoid_rewrite Ha. exact Heq.
      + setoid_rewrite <- Ha. exact Heq.
    - intros b c Heq.
      calc b == (0 + b) { (left_identity (action := add) b) } == (0 + c) { Heq } == c { (left_identity (action := add) c) }.
    - intros a IH b c Heq. apply IH. apply (injective (f := succ)).
      calc (succ (a + b)) == (succ a + b) { (add_succ_l a b) } == (succ a + c) { Heq } == (succ (a + c)) { (add_succ_l a c) }.
  Qed.

  Theorem eq_zero_of_add_eq_self : ∀ a z : N, a + z == a → z == 0.
  Proof.
    intros a z. revert a.
    apply (Peano.nat_induction (fun a ⇒ a + z == a → z == 0)).
    - intros x y Hxy. split; intros Hx Hy; apply Hx; [ rewrite Hxy | rewrite <- Hxy ]; exact Hy.
    - intros Hz.
      calc z
        == (0 + z) { left_identity }
        == (0 : N) { Hz }.
    - intros a IH Haz. apply IH. apply (injective (f := succ)).
      calc (succ (a + z))
        == (succ a + z) { add_succ_l }
        == (succ a) { Haz }.
  Qed.

Equality of naturals is decided, by a double induction.
  Theorem eq_or_ne : ∀ n m : N, n == m ∨ n =/= m.
  Proof.
    apply (Peano.nat_induction (fun n ⇒ ∀ m : N, n == m ∨ n =/= m)).
    - intros x y Hxy.
      exact (all_iff_morphism _ _ (fun m ⇒ or_iff_morphism _ _ (equiv_iff Hxy (reflexivity m))
                                                            _ _ (not_iff_morphism _ _ (equiv_iff Hxy (reflexivity m))))).
    - intros m. destruct (zero_or_succ m) as [Hm | [k Hk]].
      + left. exact (symmetry Hm).
      + right. intros H0. exact (Peano.succ_not_zero k (transitivity (symmetry Hk) (symmetry H0))).
    - intros n IH m. destruct (zero_or_succ m) as [Hm | [k Hk]].
      + right. intros Hs. exact (Peano.succ_not_zero n (transitivity Hs Hm)).
      + destruct (IH k) as [Hnk | Hnk].
        × left. calc (succ n) == (succ k) { Hnk } == m { Hk }.
        × right. intros Hs. exact (Hnk (injective (f := succ) n k (transitivity Hs Hk))).
  Qed.

"Something added gives the other" respects equality, so it can be an order (Impl/NatOrderNaturals.v).
  Theorem le_by_add_proper :
    Proper (equ ==> equ ==> iff) (fun a b : N ⇒ ∃ c : N, a + c == b).
  Proof.
    intros a a' Ha b b' Hb. split; intros [c Hc]; ∃ c.
    - rewrite <- Ha, <- Hb. exact Hc.
    - rewrite Ha, Hb. exact Hc.
  Qed.

No natural is a successor of itself plus something.
  Theorem ne_succ_add : ∀ m j : N, m =/= succ (m + j).
  Proof.
    apply (Peano.nat_induction (fun m ⇒ ∀ j : N, m =/= succ (m + j))).
    - intros m m' Hm.
      exact (all_iff_morphism _ _ (fun j ⇒ not_iff_morphism _ _
               (equiv_iff Hm (Peano.succ_proper _ _ (Magma.op_proper m m' Hm j j (reflexivity j)))))).
    - intros j Hj. exact (Peano.succ_not_zero (0 + j) (symmetry Hj)).
    - intros m IH j Hj. apply (IH j). apply (injective (f := succ)).
      calc (succ m) == (succ (succ m + j)) { Hj } == (succ (succ (m + j))) { (add_succ_l m j) }.
  Qed.

  Theorem ne_succ_succ : ∀ n : N, n =/= succ (succ n).
  Proof.
    intros n Hn. apply (ne_succ_add n (succ 0)).
    assert (Ha : n + succ 0 == succ n).
    { calc (n + succ 0) == (succ (n + 0)) { (add_succ n 0) } == (succ n) { (add_zero n) }. }
    apply (transitivity Hn). apply Peano.succ_proper. exact (symmetry Ha).
  Qed.

  Theorem ne_succ_succ' : ∀ n : N, succ (succ n) =/= n.
  Proof. intros n Hn. exact (ne_succ_succ n (symmetry Hn)). Qed.

End NaturalsTheory.
