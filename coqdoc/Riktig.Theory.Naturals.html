The commutative semiring laws give the recursion equations: each is the successor written as adding one, then a semiring law.
Section NaturalsTheory.
  Context {N : Type} `{Naturals N}.

  Theorem one_succ : (1 : N) == succ (0 : N).
  Proof.
    calc (1 : N)
      == ((0 : N) + 1) { left_identity }
      == (succ (0 : N)) { succ_add }.
  Qed.

  Theorem add_zero : ∀ a : N, a + 0 == a.
  Proof. intros a. exact (right_identity a). Qed.

  Theorem add_succ : ∀ a b : N, a + succ b == succ (a + b).
  Proof.
    intros a b.
    calc (a + succ b)
      == (a + (b + 1)) { succ_add }
      == ((a + b) + 1) { associative }
      == (succ (a + b)) { succ_add }.
  Qed.

  Theorem mul_zero : ∀ a : N, a × 0 == 0.
  Proof. intros a. exact (right_absorbing a). Qed.

  Theorem mul_succ : ∀ a b : N, a × succ b == a × b + a.
  Proof.
    intros a b.
    calc (a × succ b)
      == (a × (b + 1)) { succ_add }
      == (a × b + a × 1) { left_distributive }
      == (a × b + a) { (right_identity (action := mult) a) }.
  Qed.

Induction: the predicate is given to nat_induction, which asks that it respect ==, then holds at zero, then passes to the successor.
  Theorem zero_or_succ : ∀ n : N, n == 0 ∨ ∃ m : N, n == succ m.
  Proof.
    apply (nat_induction (fun n ⇒ n == 0 ∨ ∃ m : N, n == succ m)).
    - intros x y Hxy. split.
      + intros [Hz | [m Hm]]; [ left | right; ∃ m ]; rewrite <- Hxy; assumption.
      + intros [Hz | [m Hm]]; [ left | right; ∃ m ]; rewrite Hxy; assumption.
    - left. reflexivity.
    - intros n _. right. ∃ n. reflexivity.
  Qed.

  Theorem succ_ne_self : ∀ n : N, succ n =/= n.
  Proof.
    apply (nat_induction (fun n ⇒ succ n =/= n)).
    - intros x y Hxy. split; intros Hne Heq; apply Hne; [ rewrite Hxy | rewrite <- Hxy ]; exact Heq.
    - exact (succ_not_zero 0).
    - intros n IH Heq. exact (IH (injective (f := succ) (succ n) n Heq)).
  Qed.

  Theorem eq_zero_of_add_eq_zero : ∀ a b : N, a + b == 0 → a == 0.
  Proof.
    intros a.
    apply (nat_induction (fun b ⇒ a + b == 0 → a == 0)).
    - intros x y Hxy. split; intros Hx Hy; apply Hx; [ rewrite Hxy | rewrite <- Hxy ]; exact Hy.
    - intros Ha.
      calc a
        == (a + 0) { right_identity }
        == (0 : N) { Ha }.
    - intros b _ Hab. exfalso. apply (succ_not_zero (a + b)).
      calc (succ (a + b))
        == (a + succ b) { add_succ }
        == (0 : N) { Hab }.
  Qed.

  Theorem add_succ_l : ∀ a b : N, succ a + b == succ (a + b).
  Proof.
    intros a b.
    calc (succ a + b)
      == (b + succ a) { commutative }
      == (succ (b + a)) { add_succ }
      == (succ (a + b)) { (commutative b a) }.
  Qed.

  Theorem add_left_cancel : ∀ a b c : N, a + b == a + c → b == c.
  Proof.
    apply (nat_induction (fun a ⇒ ∀ b c : N, a + b == a + c → b == c)).
    - intros a a' Ha. split; intros Hall b c Heq; apply (Hall b c).
      + setoid_rewrite Ha. exact Heq.
      + setoid_rewrite <- Ha. exact Heq.
    - intros b c Heq.
      calc b == (0 + b) { (left_identity (action := add) b) } == (0 + c) { Heq } == c { (left_identity (action := add) c) }.
    - intros a IH b c Heq. apply IH. apply (injective (f := succ)).
      calc (succ (a + b)) == (succ a + b) { (add_succ_l a b) } == (succ a + c) { Heq } == (succ (a + c)) { (add_succ_l a c) }.
  Qed.

  Theorem eq_zero_of_add_eq_self : ∀ a z : N, a + z == a → z == 0.
  Proof.
    intros a z. revert a.
    apply (nat_induction (fun a ⇒ a + z == a → z == 0)).
    - intros x y Hxy. split; intros Hx Hy; apply Hx; [ rewrite Hxy | rewrite <- Hxy ]; exact Hy.
    - intros Hz.
      calc z
        == (0 + z) { left_identity }
        == (0 : N) { Hz }.
    - intros a IH Haz. apply IH. apply (injective (f := succ)).
      calc (succ (a + z))
        == (succ a + z) { add_succ_l }
        == (succ a) { Haz }.
  Qed.

Equality of naturals is decided, by a double induction.
  Theorem eq_or_ne : ∀ n m : N, n == m ∨ n =/= m.
  Proof.
    apply (nat_induction (fun n ⇒ ∀ m : N, n == m ∨ n =/= m)).
    - intros x y Hxy. split; intros Hx m; destruct (Hx m) as [Hm | Hm].
      + left. transitivity x; [ symmetry; exact Hxy | exact Hm ].
      + right. intros Hym. apply Hm. transitivity y; [ exact Hxy | exact Hym ].
      + left. transitivity y; [ exact Hxy | exact Hm ].
      + right. intros Hxm. apply Hm. transitivity x; [ symmetry; exact Hxy | exact Hxm ].
    - intros m. destruct (zero_or_succ m) as [Hm | [k Hk]].
      + left. symmetry. exact Hm.
      + right. intros H0. apply (succ_not_zero k).
        transitivity m; [ symmetry; exact Hk | symmetry; exact H0 ].
    - intros n IH m. destruct (zero_or_succ m) as [Hm | [k Hk]].
      + right. intros Hs. apply (succ_not_zero n). transitivity m; [ exact Hs | exact Hm ].
      + destruct (IH k) as [Hnk | Hnk].
        × left. rewrite Hk, Hnk. reflexivity.
        × right. intros Hs. apply Hnk. apply (injective (f := succ)).
          rewrite <- Hk. exact Hs.
  Qed.

"Something added gives the other" respects equality, so it can be an order (Impl/NatOrderNaturals.v).
  Theorem le_by_add_proper :
    Proper (equ ==> equ ==> iff) (fun a b : N ⇒ ∃ c : N, a + c == b).
  Proof.
    intros a a' Ha b b' Hb. split; intros [c Hc]; ∃ c.
    - rewrite <- Ha, <- Hb. exact Hc.
    - rewrite Ha, Hb. exact Hc.
  Qed.

No natural is a successor of itself plus something.
  Theorem ne_succ_add : ∀ m j : N, m =/= succ (m + j).
  Proof.
    apply (nat_induction (fun m ⇒ ∀ j : N, m =/= succ (m + j))).
    - intros m m' Hm. split; intros Hall j Hj; apply (Hall j).
      + setoid_rewrite Hm. exact Hj.
      + setoid_rewrite <- Hm. exact Hj.
    - intros j Hj. exact (succ_not_zero (0 + j) (symmetry Hj)).
    - intros m IH j Hj. apply (IH j). apply (injective (f := succ)).
      transitivity (succ (succ m + j)); [ exact Hj | apply succ_proper; exact (add_succ_l m j) ].
  Qed.

End NaturalsTheory.
