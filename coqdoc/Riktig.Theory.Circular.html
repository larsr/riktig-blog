  Lemma scale_mul_l : ∀ (a : F) (x y : A), (a *: x) × y == a *: (x × y).
  Proof. intros a x y. exact (homogeneous[fun z ⇒ z × y] a x). Qed.

  Lemma scale_mul_r : ∀ (a : F) (x y : A), x × (a *: y) == a *: (x × y).
  Proof. intros a x y. exact (homogeneous[fun z ⇒ x × z] a y). Qed.

  Lemma scale_mul_scale : ∀ (a b : F) (x y : A),
    (a *: x) × (b *: y) == (a × b) *: (x × y).
  Proof.
    intros a b x y.
    calc (((a *: x) × (b *: y)) : A)
      == ((a *: (x × (b *: y))) : A) { scale_mul_l }
      == ((a *: (b *: (x × y))) : A) { scale_mul_r }
      == (((a × b) *: (x × y)) : A) { associative }.
  Qed.

Two scalings of one element commute, which discharges the guard on cexp_add for arguments on a common line.
  Lemma collinear_commute : ∀ (t u : F) (x : A),
    (t *: x) × (u *: x) == (u *: x) × (t *: x).
  Proof.
    intros t u x.
    calc (((t *: x) × (u *: x)) : A)
      == (((t × u) *: (x × x)) : A) { scale_mul_scale }
      == (((u × t) *: (x × x)) : A) { (mul_commutative t u) }
      == (((u *: x) × (t *: x)) : A) { scale_mul_scale }.
  Qed.

Along a line the exponential carries addition to multiplication unconditionally.
  Theorem cexp_collinear_add : ∀ (t u : F) (x : A),
    cexp ((t + u) *: x) == cexp (t *: x) × cexp (u *: x).
  Proof.
    intros t u x.
    calc (cexp ((t + u) *: x))
      == (cexp (t *: x + u *: x))
         { right_distributive }
      == (cexp (t *: x) × cexp (u *: x))
         { (cexp_add (t *: x) (u *: x) (collinear_commute t u x)) }.
  Qed.


  Lemma scale_one_mul_scale : ∀ (a c : F) (x : A),
    (a *: (1 : A)) × (c *: x) == (a × c) *: x.
  Proof.
    intros a c x.
    calc (((a *: (1 : A)) × (c *: x)) : A)
      == (((a × c) *: ((1 : A) × x)) : A) { scale_mul_scale }
      == (((a × c) *: x) : A) { left_identity }.
  Qed.

  Lemma scale_mul_scale_one : ∀ (a c : F) (x : A),
    (a *: x) × (c *: (1 : A)) == (a × c) *: x.
  Proof.
    intros a c x.
    calc (((a *: x) × (c *: (1 : A))) : A)
      == (((a × c) *: (x × (1 : A))) : A) { scale_mul_scale }
      == (((a × c) *: x) : A) { right_identity }.
  Qed.

  Lemma root_mul_root : ∀ (b d : F) (e : A),
    e × e == - (1) → (b *: e) × (d *: e) == (- (b × d)) *: (1 : A).
  Proof.
    intros b d e Hroot.
    calc (((b *: e) × (d *: e)) : A)
      == (((b × d) *: (e × e)) : A) { scale_mul_scale }
      == (((b × d) *: (- (1 : A))) : A) { Hroot }
      == ((- ((b × d) *: (1 : A))) : A) { scale_neg }
      == (((- (b × d)) *: (1 : A)) : A) { neg_scale }.
  Qed.

  Lemma scale_one_mul_two_term : ∀ (a c d : F) (x : A),
    (a *: (1 : A)) × (c *: 1 + d *: x) == (a × c) *: 1 + (a × d) *: x.
  Proof.
    intros a c d x.
    calc (((a *: (1 : A)) × (c *: 1 + d *: x)) : A)
      == (((a *: (1 : A)) × (c *: 1) + (a *: (1 : A)) × (d *: x)) : A)
         { left_distributive }
      == (((a × c) *: (1 : A) + (a *: (1 : A)) × (d *: x)) : A)
         { scale_one_mul_scale }
      == (((a × c) *: (1 : A) + (a × d) *: x) : A)
         { scale_one_mul_scale }.
  Qed.

  Lemma root_mul_two_term : ∀ (b c d : F) (e : A),
    e × e == - (1) →
    (b *: e) × (c *: 1 + d *: e) == (b × c) *: e + (- (b × d)) *: 1.
  Proof.
    intros b c d e Hroot.
    calc (((b *: e) × (c *: 1 + d *: e)) : A)
      == (((b *: e) × (c *: (1 : A)) + (b *: e) × (d *: e)) : A)
         { left_distributive }
      == (((b × c) *: e + (b *: e) × (d *: e)) : A)
         { scale_mul_scale_one }
      == (((b × c) *: e + (- (b × d)) *: (1 : A)) : A)
         { (root_mul_root b d e Hroot) }.
  Qed.

  Lemma two_term_collect : ∀ (p q r s : F) (x : A),
    (p *: 1 + q *: x) + (r *: x + s *: 1) == (p + s) *: 1 + (q + r) *: x.
  Proof.
    intros p q r s x.
    calc (((p *: (1 : A) + q *: x) + (r *: x + s *: 1)) : A)
      == (((p *: (1 : A) + s *: 1) + (q *: x + r *: x)) : A)
         { add_rotate }
      == (((p + s) *: (1 : A) + (q *: x + r *: x)) : A)
         { right_distributive }
      == (((p + s) *: (1 : A) + (q + r) *: x) : A)
         { right_distributive }.
  Qed.

The complex multiplication table, over any square root of - (1).
  Theorem two_term_product : ∀ (a b c d : F) (e : A),
    e × e == - (1) →
    (a *: 1 + b *: e) × (c *: 1 + d *: e)
    == (a × c + - (b × d)) *: 1 + (a × d + b × c) *: e.
  Proof.
    intros a b c d e Hroot.
    calc (((a *: 1 + b *: e) × (c *: 1 + d *: e)) : A)
      == (((a *: (1 : A)) × (c *: 1 + d *: e)
             + (b *: e) × (c *: 1 + d *: e)) : A)
         { right_distributive }
      == ((((a × c) *: (1 : A) + (a × d) *: e)
             + (b *: e) × (c *: 1 + d *: e)) : A)
         { scale_one_mul_two_term }
      == ((((a × c) *: (1 : A) + (a × d) *: e)
             + ((b × c) *: e + (- (b × d)) *: 1)) : A)
         { (root_mul_two_term b c d e Hroot) }
      == (((a × c + - (b × d)) *: (1 : A) + (a × d + b × c) *: e) : A)
         { two_term_collect }.
  Qed.


Two exponentials on a common line, multiplied out by euler on each factor.
  Lemma euler_product : ∀ (t u : F) (e : A),
    e × e == - (1) →
    cexp (t *: e) × cexp (u *: e)
    == (cos t × cos u + - (sin t × sin u)) *: 1
       + (cos t × sin u + sin t × cos u) *: e.
  Proof.
    intros t u e Hroot.
    calc ((cexp (t *: e) × cexp (u *: e)) : A)
      == (((cos t *: 1 + sin t *: e) × cexp (u *: e)) : A)
         { (euler t e Hroot) }
      == (((cos t *: 1 + sin t *: e) × (cos u *: 1 + sin u *: e)) : A)
         { (euler u e Hroot) }
      == (((cos t × cos u + - (sin t × sin u)) *: 1
             + (cos t × sin u + sin t × cos u) *: e) : A)
         { (two_term_product (cos t) (sin t) (cos u) (sin u) e Hroot) }.
  Qed.

Expanding cexp ((t + u) *: e) through cexp_add and then through euler on each factor.
  Theorem cexp_angle_addition : ∀ (t u : F) (e : A),
    e × e == - (1) →
    cexp ((t + u) *: e)
    == (cos t × cos u + - (sin t × sin u)) *: 1
       + (cos t × sin u + sin t × cos u) *: e.
  Proof.
    intros t u e Hroot.
    calc (cexp ((t + u) *: e))
      == ((cexp (t *: e) × cexp (u *: e)) : A)
         { cexp_collinear_add }
      == (((cos t × cos u + - (sin t × sin u)) *: 1
             + (cos t × sin u + sin t × cos u) *: e) : A)
         { (euler_product t u e Hroot) }.
  Qed.

The same fact with euler applied to the left-hand side too: the two coefficient pairs of one element of the plane.
  Theorem euler_angle_addition : ∀ (t u : F) (e : A),
    e × e == - (1) →
    cos (t + u) *: 1 + sin (t + u) *: e
    == (cos t × cos u + - (sin t × sin u)) *: 1
       + (cos t × sin u + sin t × cos u) *: e.
  Proof.
    intros t u e Hroot.
    calc ((cos (t + u) *: 1 + sin (t + u) *: e) : A)
      == (cexp ((t + u) *: e))
         { (euler (t + u) e Hroot) }
      == (((cos t × cos u + - (sin t × sin u)) *: 1
             + (cos t × sin u + sin t × cos u) *: e) : A)
         { (cexp_angle_addition t u e Hroot) }.
  Qed.


The two coefficients of cexp (t *: e) × cexp ((- t) *: e): cos even and sin odd turn the first into the Pythagorean sum and cancel the second.
  Lemma pythagoras_reflected : ∀ t : F,
    cos t × cos (- t) + - (sin t × sin (- t)) == 1.
  Proof.
    intros t.
    calc (cos t × cos (- t) + - (sin t × sin (- t)))
      == (cos t × cos t + - (sin t × sin (- t))) { (cos_even (A := A) t) }
      == (cos t × cos t + - (sin t × (- sin t))) { (sin_odd (A := A) t) }
      == [regroup] (cos t × cos t + sin t × sin t)
      == (1 : F) { (pythagoras (A := A) t) }.
    [regroup]: ring.
  Qed.

  Lemma sine_reflected_cancels : ∀ t : F,
    cos t × sin (- t) + sin t × cos (- t) == 0.
  Proof.
    intros t.
    calc (cos t × sin (- t) + sin t × cos (- t))
      == (cos t × (- sin t) + sin t × cos (- t)) { (sin_odd (A := A) t) }
      == (cos t × (- sin t) + sin t × cos t) { (cos_even (A := A) t) }
      == [cancel] (0 : F).
    [cancel]: ring.
  Qed.

  Lemma scale_one_scale_zero : ∀ x : A,
    (1 : F) *: (1 : A) + (0 : F) *: x == 1.
  Proof.
    intros x.
    calc (((1 : F) *: (1 : A) + (0 : F) *: x) : A)
      == (((1 : A) + (0 : F) *: x) : A) { left_identity }
      == (((1 : A) + (0 : A)) : A) { zero_scale }
      == ((1 : A)) { right_identity }.
  Qed.

  Theorem cexp_scale_right_inverse : ∀ (t : F) (e : A),
    e × e == - (1) → cexp (t *: e) × cexp ((- t) *: e) == 1.
  Proof.
    intros t e Hroot.
    calc ((cexp (t *: e) × cexp ((- t) *: e)) : A)
      == (((cos t × cos (- t) + - (sin t × sin (- t))) *: 1
             + (cos t × sin (- t) + sin t × cos (- t)) *: e) : A)
         { (euler_product t (- t) e Hroot) }
      == (((1 : F) *: (1 : A)
             + (cos t × sin (- t) + sin t × cos (- t)) *: e) : A)
         { (pythagoras_reflected t) }
      == (((1 : F) *: (1 : A) + (0 : F) *: e) : A)
         { (sine_reflected_cancels t) }
      == ((1 : A))
         { scale_one_scale_zero }.
  Qed.

  Theorem cexp_scale_left_inverse : ∀ (t : F) (e : A),
    e × e == - (1) → cexp ((- t) *: e) × cexp (t *: e) == 1.
  Proof.
    intros t e Hroot.
    calc ((cexp ((- t) *: e) × cexp (t *: e)) : A)
      == ((cexp ((- t) *: e) × cexp ((- (- t)) *: e)) : A)
         { neg_involutive }
      == ((1 : A))
         { (cexp_scale_right_inverse (- t) e Hroot) }.
  Qed.

invertible is Props.v's predicate: a two-sided inverse exists.
  Corollary cexp_scale_invertible : ∀ (t : F) (e : A),
    e × e == - (1) → invertible A mult 1 (cexp (t *: e)).
  Proof.
    intros t e Hroot.
    ∃ (cexp ((- t) *: e)).
    split; [ exact (cexp_scale_right_inverse t e Hroot)
           | exact (cexp_scale_left_inverse t e Hroot) ].
  Qed.

The exponential of the zero angle is the unit: it is its own square and it is invertible.
  Theorem cexp_zero_scale : ∀ e : A,
    e × e == - (1) → cexp ((0 : F) *: e) == 1.
  Proof.
    intros e Hroot.
    assert (Hneg : cexp ((0 : F) *: e) == cexp ((- (0 : F)) *: e)).
    { calc (cexp ((0 : F) *: e))
        == (cexp ((- (0 : F)) *: e)) { neg_zero }. }
    calc (cexp ((0 : F) *: e))
      == (cexp (((0 : F) + 0) *: e))
         { (add_left_identity (0 : F)) }
      == ((cexp ((0 : F) *: e) × cexp ((0 : F) *: e)) : A)
         { cexp_collinear_add }
      == ((cexp ((0 : F) *: e) × cexp ((- (0 : F)) *: e)) : A)
         { Hneg }
      == ((1 : A))
         { (cexp_scale_right_inverse 0 e Hroot) }.
  Qed.

End CircularAlgebra.


1 and e are independent when no combination of them vanishes without both coefficients vanishing.
Definition independent_from_one (F : Type) {A : Type}
    {Fe : EqRel F} {Fz : Zero F}
    {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A} {Ao : One A}
    {As : HasScale F A} (e : A) : Prop :=
  ∀ a b : F, a *: 1 + b *: e == 0 → a == 0 ∧ b == 0.

Section AngleSeparation.
  Context {F A : Type} `{CA : Circular F A}.

Independent coordinates are unique coordinates.
  Lemma independent_separates : ∀ e : A,
    independent_from_one F e →
    ∀ a b c d : F,
      a *: 1 + b *: e == c *: 1 + d *: e → a == c ∧ b == d.
  Proof.
    intros e Hindependent a b c d Heq.
    assert (Hzero : (a + - c) *: (1 : A) + (b + - d) *: e == 0).
    { calc (((a + - c) *: (1 : A) + (b + - d) *: e) : A)
        == (((a *: (1 : A) + - (c *: (1 : A))) + (b + - d) *: e) : A)
           { sub_scale }
        == (((a *: (1 : A) + - (c *: (1 : A)))
               + (b *: e + - (d *: e))) : A)
           { sub_scale }
        == (((a *: (1 : A) + b *: e)
               + - (c *: (1 : A) + d *: e)) : A)
           { sub_pair }
        == (((c *: (1 : A) + d *: e)
               + - (c *: (1 : A) + d *: e)) : A)
           { Heq }
        == ((0 : A))
           { add_right_inverse }. }
    destruct (Hindependent (a + - c) (b + - d) Hzero) as [Hac Hbd].
    split; [ exact (sub_eq_zero a c Hac) | exact (sub_eq_zero b d Hbd) ].
  Qed.

  Context (e : A) (root : e × e == - (1))
          (independence : independent_from_one F e).

Angle addition, in the form the trigonometry tables print.
  Theorem cos_angle_addition : ∀ t u : F,
    cos (t + u) == cos t × cos u + - (sin t × sin u).
  Proof.
    intros t u.
    exact (proj1 (independent_separates e independence _ _ _ _
                    (euler_angle_addition t u e root))).
  Qed.

  Theorem sin_angle_addition : ∀ t u : F,
    sin (t + u) == cos t × sin u + sin t × cos u.
  Proof.
    intros t u.
    exact (proj2 (independent_separates e independence _ _ _ _
                    (euler_angle_addition t u e root))).
  Qed.

  Lemma euler_at_zero :
    cos (0 : F) *: (1 : A) + sin (0 : F) *: e == (1 : F) *: 1 + (0 : F) *: e.
  Proof.
    calc ((cos (0 : F) *: (1 : A) + sin (0 : F) *: e) : A)
      == (cexp ((0 : F) *: e))
         { (euler 0 e root) }
      == ((1 : A))
         { (cexp_zero_scale e root) }
      == (((1 : F) *: (1 : A) + (0 : F) *: e) : A)
         { scale_one_scale_zero }.
  Qed.

The values at the zero angle, which the parity laws alone leave open in characteristic two.
  Theorem cos_zero : cos (0 : F) == 1.
  Proof.
    exact (proj1 (independent_separates e independence _ _ _ _ euler_at_zero)).
  Qed.

  Theorem sin_zero : sin (0 : F) == 0.
  Proof.
    exact (proj2 (independent_separates e independence _ _ _ _ euler_at_zero)).
  Qed.

End AngleSeparation.

Section PlaneBivector.
  Context {F : Type} `{FF : Field F}.
  Add Ring plane_bivector_scalars : (ring_theory_of_Ring (R := F) commutative).

  Lemma rotor_neg_one : rotor (- (1 : F)) 0 == - (1 : Cl F 2).
  Proof.
    assert (Hscalar : (- (1 : F)) × 1 == - (1)) by ring.
    assert (Hbivector : (- (1 : F)) × 0 == 0) by ring.
    calc ((rotor (- (1 : F)) 0) : Cl F 2)
      == ((rotor ((- (1 : F)) × 1) 0) : Cl F 2)
         { Hscalar }
      == ((rotor ((- (1 : F)) × 1) ((- (1 : F)) × 0)) : Cl F 2)
         { Hbivector }
      == (((- (1 : F)) *: rotor 1 0) : Cl F 2)
         { scale_rotor }
      == (((- (1 : F)) *: (1 : Cl F 2)) : Cl F 2)
         { rotor_one }
      == ((- (1 : Cl F 2)) : Cl F 2)
         { neg1_scale }.
  Qed.

The plane algebra's bivector e12 is rotor 0 1, and it is a square root of - (1): `euler` applies to it.
  Theorem plane_bivector_square :
    (rotor (0 : F) 1) × rotor 0 1 == - (1 : Cl F 2).
  Proof.
    assert (Hscalar : (0 : F) × 0 + - (1 × 1) == - (1)) by ring.
    assert (Hbivector : (0 : F) × 1 + 1 × 0 == 0) by ring.
    calc (((rotor (0 : F) 1) × rotor 0 1) : Cl F 2)
      == ((rotor ((0 : F) × 0 + - (1 × 1)) ((0 : F) × 1 + 1 × 0)) : Cl F 2)
         { rotor_mul }
      == ((rotor (- (1 : F)) ((0 : F) × 1 + 1 × 0)) : Cl F 2)
         { Hscalar }
      == ((rotor (- (1 : F)) 0) : Cl F 2)
         { Hbivector }
      == ((- (1 : Cl F 2)) : Cl F 2)
         { rotor_neg_one }.
  Qed.

... and its plane has independent coordinates, the two of them being two different blades.
  Theorem plane_independent_from_one :
    independent_from_one F (rotor (0 : F) 1).
  Proof.
    intros a b Hab.
    split.
    - calc a
        == [scalar_blade] (a × 1 + b × 0)
        == (0 : F) { (Hab (pb false false)) }.
      [scalar_blade]: ring.
    - calc b
        == [bivector_blade] (a × 0 + b × 1)
        == (0 : F) { (Hab (pb true true)) }.
      [bivector_blade]: ring.
  Qed.

End PlaneBivector.

Section RotorAngles.
  Context {F A : Type} `{CA : Circular F A}.

A rotor at an angle is a unit rotor: that is Pythagoras.
  Theorem rotor_angle_unit : ∀ t : F,
    rotor (cos t) (sin t) × cl_reverse (rotor (cos t) (sin t)) == 1.
  Proof.
    intros t.
    exact (rotor_unit (cos t) (sin t) (pythagoras (A := A) t)).
  Qed.

Reversal negates the angle: that is cos even and sin odd.
  Theorem rotor_angle_reverse : ∀ t : F,
    cl_reverse (rotor (cos t) (sin t)) == rotor (cos (- t)) (sin (- t)).
  Proof.
    intros t.
    calc ((cl_reverse (rotor (cos t) (sin t))) : Cl F 2)
      == ((rotor (cos t) (- sin t)) : Cl F 2)
         { rotor_reverse }
      == ((rotor (cos (- t)) (- sin t)) : Cl F 2)
         { (cos_even (A := A) t) }
      == ((rotor (cos (- t)) (sin (- t))) : Cl F 2)
         { (sin_odd (A := A) t) }.
  Qed.

  Context (e : A) (root : e × e == - (1))
          (independence : independent_from_one F e).

  Theorem rotor_angle_addition : ∀ t u : F,
    rotor (cos t) (sin t) × rotor (cos u) (sin u)
    == rotor (cos (t + u)) (sin (t + u)).
  Proof.
    intros t u.
    calc ((rotor (cos t) (sin t) × rotor (cos u) (sin u)) : Cl F 2)
      == ((rotor (cos t × cos u + - (sin t × sin u))
                 (cos t × sin u + sin t × cos u)) : Cl F 2)
         { rotor_mul }
      == ((rotor (cos (t + u)) (cos t × sin u + sin t × cos u)) : Cl F 2)
         { (cos_angle_addition e root independence t u) }
      == ((rotor (cos (t + u)) (sin (t + u))) : Cl F 2)
         { (sin_angle_addition e root independence t u) }.
  Qed.

  Theorem rotor_angle_zero : rotor (cos (0 : F)) (sin (0 : F)) == 1.
  Proof.
    calc ((rotor (cos (0 : F)) (sin (0 : F))) : Cl F 2)
      == ((rotor (1 : F) (sin (0 : F))) : Cl F 2)
         { (cos_zero e root independence) }
      == ((rotor (1 : F) (0 : F)) : Cl F 2)
         { (sin_zero e root independence) }
      == ((1 : Cl F 2))
         { rotor_one }.
  Qed.

End RotorAngles.

  Lemma scale_two_term : ∀ (n a b : F) (x : A),
    (n × a) *: 1 + (n × b) *: x == n *: (a *: 1 + b *: x).
  Proof.
    intros n a b x.
    calc (((n × a) *: (1 : A) + (n × b) *: x) : A)
      == ((n *: (a *: (1 : A)) + (n × b) *: x) : A)
         { associative }
      == ((n *: (a *: (1 : A)) + n *: (b *: x)) : A)
         { associative }
      == ((n *: (a *: (1 : A) + b *: x)) : A)
         { left_distributive }.
  Qed.

  Lemma scale_inv_cancel : ∀ (n : F) (x : A),
    n =/= 0 → n *: (inv n *: x) == x.
  Proof.
    intros n x Hn.
    calc ((n *: (inv n *: x)) : A)
      == (((n × inv n) *: x) : A) { associative }
      == (((1 : F) *: x) : A) { (field_right_inverse n Hn) }
      == (x : A) { left_identity }.
  Qed.


A product of two vectors of the plane whose grade-0 and grade-2 parts are cos t and sin t scaled by ‖ u ‖ × ‖ v ‖ is that scaling of the exponential of t.
  Theorem polar_product : ∀ (u v e : A) (t : F),
    e × e == - (1) →
    ⟨ u × v ⟩_ 1 == 0 →
    ⟨ u × v ⟩_ 0 == (‖ u ‖ × ‖ v ‖ × cos t) *: 1 →
    ⟨ u × v ⟩_ 2 == (‖ u ‖ × ‖ v ‖ × sin t) *: e →
    u × v == (‖ u ‖ × ‖ v ‖) *: cexp (t *: e).
  Proof.
    intros u v e t Hroot Hvector Hscalar Hbivector.
    calc ((u × v) : A)
      == ((⟨ u × v ⟩_ 0 + ⟨ u × v ⟩_ 2) : A)
         { (grade_decomposition_two_even (u × v) Hvector) }
      == (((‖ u ‖ × ‖ v ‖ × cos t) *: 1 + ⟨ u × v ⟩_ 2) : A)
         { Hscalar }
      == (((‖ u ‖ × ‖ v ‖ × cos t) *: 1
             + (‖ u ‖ × ‖ v ‖ × sin t) *: e) : A)
         { Hbivector }
      == (((‖ u ‖ × ‖ v ‖) *: (cos t *: 1 + sin t *: e)) : A)
         { scale_two_term }
      == (((‖ u ‖ × ‖ v ‖) *: cexp (t *: e)) : A)
         { (euler t e Hroot) }.
  Qed.

The same statement with the identification written as it is meant: cos t and sin t ARE the grade-0 and grade-2 parts of the product, divided by ‖ u ‖ × ‖ v ‖.
  Theorem polar_product_normalized : ∀ (u v e : A) (t : F),
    e × e == - (1) →
    ‖ u ‖ × ‖ v ‖ =/= 0 →
    ⟨ u × v ⟩_ 1 == 0 →
    inv (‖ u ‖ × ‖ v ‖) *: ⟨ u × v ⟩_ 0 == cos t *: 1 →
    inv (‖ u ‖ × ‖ v ‖) *: ⟨ u × v ⟩_ 2 == sin t *: e →
    u × v == (‖ u ‖ × ‖ v ‖) *: cexp (t *: e).
  Proof.
    intros u v e t Hroot Hnonzero Hvector Hcos Hsin.
    apply (polar_product u v e t Hroot Hvector).
    - calc ((⟨ u × v ⟩_ 0) : A)
        == (((‖ u ‖ × ‖ v ‖) *: (inv (‖ u ‖ × ‖ v ‖) *: ⟨ u × v ⟩_ 0)) : A)
           { (scale_inv_cancel (‖ u ‖ × ‖ v ‖) (⟨ u × v ⟩_ 0) Hnonzero) }
        == (((‖ u ‖ × ‖ v ‖) *: (cos t *: (1 : A))) : A)
           { Hcos }
        == (((‖ u ‖ × ‖ v ‖ × cos t) *: (1 : A)) : A)
           { associative }.
    - calc ((⟨ u × v ⟩_ 2) : A)
        == (((‖ u ‖ × ‖ v ‖) *: (inv (‖ u ‖ × ‖ v ‖) *: ⟨ u × v ⟩_ 2)) : A)
           { (scale_inv_cancel (‖ u ‖ × ‖ v ‖) (⟨ u × v ⟩_ 2) Hnonzero) }
        == (((‖ u ‖ × ‖ v ‖) *: (sin t *: e)) : A)
           { Hsin }
        == (((‖ u ‖ × ‖ v ‖ × sin t) *: e) : A)
           { associative }.
  Qed.

The identification is consistent with Pythagoras: the two coefficients of the polar form have square sum (‖u‖ ‖v‖)².
  Theorem polar_modulus : ∀ n t : F,
    (n × cos t) × (n × cos t) + (n × sin t) × (n × sin t) == n × n.
  Proof.
    intros n t.
    calc ((n × cos t) × (n × cos t) + (n × sin t) × (n × sin t))
      == [factor] ((n × n) × (cos t × cos t + sin t × sin t))
      == ((n × n) × 1) { (pythagoras (A := A) t) }
      == (n × n) { mul_right_identity }.
    [factor]: ring.
  Qed.
