Section EvalSums.
  Context {F V : Type} `{VV : VectorSpace F V}.

  Lemma hom_bigop_list_at : ∀ {J : Type} (g : J → LinearMap F V F)
      (l : list J) (v : V),
    bigop_list g l v == bigop_list (fun j ⇒ g j v) l.
  Proof.
    intros J g l v; induction l as [| a l IH].
    - reflexivity.
    - calc (bigop_list g (a :: l) v)
        == (g a v + bigop_list g l v)
             { bigop_list_cons }
        == (g a v + bigop_list (fun j ⇒ g j v) l)
             { op_proper }
        == (bigop_list (fun j ⇒ g j v) (a :: l))
             { bigop_list_cons }.
  Qed.

  Lemma hom_bigopI_at : ∀ {k : nat} {I : Type} {eI : EqRel I}
      {Ieq : EquDec I} {IFin : FinIndex k I}
      (g : I → LinearMap F V F) (v : V),
    bigopI g v == bigopI (fun i ⇒ g i v).
  Proof.
    intros k I eI Ieq IFin g v.
    exact (hom_bigop_list_at g enum v).
  Qed.

End EvalSums.

Section DualBasis.
  Context {F V : Type} `{VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context {delta : I → I → F} {FdL : Delta I F delta}.
  Context (B : Basis n I F V).

  Definition basis_coord (i : I) : LinearMap F V F :=
    linear_map_from_laws (fun v ⇒ Basis.coords B v i)
      (fun u v Huv ⇒ coords_proper B u v Huv i)
      (fun u v ⇒ coords_add B u v i)
      (fun a v ⇒ coords_scale B a v i).

  Definition dual_coords : LinearMap F (LinearMap F V F) (I → F) :=
    linear_map_from_laws
      (fun (f : LinearMap F V F) (i : I) ⇒ f (Basis.vec B i))
      (fun f g Hfg i ⇒ Hfg (Basis.vec B i))
      (fun f g i ⇒ reflexivity (R := equ) (f (Basis.vec B i) + g (Basis.vec B i)))
      (fun a f i ⇒ reflexivity (R := equ) (a × f (Basis.vec B i))).

  Lemma dual_reconstruction : ∀ f : LinearMap F V F,
    lincomb basis_coord (dual_coords f) == f.
  Proof.
    intros f v.
    calc (lincomb basis_coord (dual_coords f) v)
      == (bigopI (fun i ⇒ f (Basis.vec B i) × Basis.coords B v i))
      == (bigopI (fun i ⇒ Basis.coords B v i × f (Basis.vec B i)))
      == (f (lincomb (Basis.vec B) (Basis.coords B v)))
      == (f v).
    - exact (hom_bigopI_at (fun i ⇒ f (Basis.vec B i) *: basis_coord i) v).
    - apply bigopI_ext. intros i.
      exact (commutative (f (Basis.vec B i)) (Basis.coords B v i)).
    - symmetry.
      exact (lm_lincomb f (Basis.vec B) (Basis.coords B v)).
    - apply (Linear.proper (f := LinearMap.map f)).
      exact (Basis.reconstruction B v).
  Qed.

  Lemma dual_coords_lincomb : ∀ c : I → F,
    Proper (equ ==> equ) c →
    dual_coords (lincomb basis_coord c) == c.
  Proof.
    intros c Pc j.
    calc (lincomb basis_coord c (Basis.vec B j))
      == (bigopI (fun i ⇒ c i × Basis.coords B (Basis.vec B j) i))
      == (bigopI (fun i ⇒ delta j i *: c i))
      == (c j).
    - exact (hom_bigopI_at (fun i ⇒ c i *: basis_coord i) (Basis.vec B j)).
    - apply bigopI_ext. intros i.
      calc (c i × Basis.coords B (Basis.vec B j) i)
        == (c i × delta j i) { (coords_vec B j i) }
        == (delta j i *: c i) { commutative }.
    -
      exact (lincomb_delta (F := F) (V := F)
               (sV := Field_Scale) (VV := Field_VectorSpace) c j Pc).
  Qed.

  Lemma basis_coord_proper : Proper (equ ==> equ) basis_coord.
  Proof.
    intros i i' Hii' v.
    exact (Basis.coords_family_proper B v i i' Hii').
  Qed.

  Lemma dual_coords_family_proper : ∀ f : LinearMap F V F,
    Proper (equ ==> equ) (dual_coords f).
  Proof.
    intros f i i' Hii'.
    exact (linear_map_proper f (Basis.vec B i) (Basis.vec B i')
             (Basis.vec_proper B i i' Hii')).
  Qed.

  Definition dual_basis : Basis n I F (LinearMap F V F) :=
    {| Basis.vec := basis_coord;
       Basis.vec_proper := basis_coord_proper;
       Basis.coords := dual_coords;
       Basis.coords_family_proper := dual_coords_family_proper;
       Basis.reconstruction := dual_reconstruction;
       Basis.coords_linear_combination := dual_coords_lincomb |}.


  Local Notation DDualV := (LinearMap F (LinearMap F V F) F).

  Lemma from_dd_proper : Proper (equ ==> equ)
      (fun p : DDualV ⇒ lincomb (Basis.vec B) (fun i ⇒ p (basis_coord i))).
  Proof.
    intros p q Hpq.
    exact (lincomb_proper (Basis.vec B)
             (fun i ⇒ p (basis_coord i)) (fun i ⇒ q (basis_coord i))
             (fun i ⇒ Hpq (basis_coord i))).
  Qed.

  Lemma from_dd_add : ∀ p q : DDualV,
    lincomb (Basis.vec B) (fun i ⇒ (p + q) (basis_coord i))
      == lincomb (Basis.vec B) (fun i ⇒ p (basis_coord i))
         + lincomb (Basis.vec B) (fun i ⇒ q (basis_coord i)).
  Proof.
    intros p q.
    exact (lincomb_add (Basis.vec B)
             (fun i ⇒ p (basis_coord i)) (fun i ⇒ q (basis_coord i))).
  Qed.

  Lemma from_dd_scale : ∀ (a : F) (p : DDualV),
    lincomb (Basis.vec B) (fun i ⇒ (a *: p) (basis_coord i))
      == a *: lincomb (Basis.vec B) (fun i ⇒ p (basis_coord i)).
  Proof.
    intros a p.
    exact (lincomb_scale a (Basis.vec B) (fun i ⇒ p (basis_coord i))).
  Qed.

  Definition from_double_dual : LinearMap F DDualV V :=
    linear_map_from_laws
      (fun p : DDualV ⇒ lincomb (Basis.vec B) (fun i ⇒ p (basis_coord i)))
      from_dd_proper from_dd_add from_dd_scale.

  Lemma double_dual_fromto : ∀ v : V,
    from_double_dual (into_double_dual v) == v.
  Proof.
    intros v.
    exact (Basis.reconstruction B v).
  Qed.

  Lemma double_dual_tofrom : ∀ p : DDualV,
    into_double_dual (from_double_dual p) == p.
  Proof.
    intros p f.
    calc (into_double_dual (from_double_dual p) f)
      == (bigopI (fun i ⇒ p (basis_coord i) × f (Basis.vec B i)))
      == (bigopI (fun i ⇒ f (Basis.vec B i) × p (basis_coord i)))
      == (p (lincomb basis_coord (dual_coords f)))
      == (p f).
    - exact (lm_lincomb f (Basis.vec B) (fun i ⇒ p (basis_coord i))).
    - apply bigopI_ext. intros i.
      exact (commutative (p (basis_coord i)) (f (Basis.vec B i))).
    - symmetry.
      exact (lm_lincomb p basis_coord (dual_coords f)).
    - apply (Linear.proper (f := LinearMap.map p)).
      exact (dual_reconstruction f).
  Qed.

  Definition double_dual_iso : VSIso F V DDualV :=
    vsiso_from_laws into_double_dual from_double_dual
      double_dual_fromto double_dual_tofrom.

End DualBasis.

Example double_dual_round (F V : Type) `{FF : Field F}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V}
    {sV : HasScale F V} {VV : VectorSpace F V}
    {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}
    (B : Basis n I F V) (v : V) :
  VSIso.from (double_dual_iso B) (VSIso.to (double_dual_iso B) v) == v :=
  vsiso_from_to (double_dual_iso B) v.

Section HomPrecompose.
  Context {F U W X : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context {eX : EqRel X} {zX : Zero X} {aX : HasAdd X} {nX : HasNeg X} {sX : HasScale F X}.
  Context {VX : VectorSpace F X}.

  Context (h : LinearMap F U W).

  Lemma precompose_proper : Proper (equ ==> equ)
      (fun g : LinearMap F W X ⇒ linear_map_comp g h).
  Proof.
    intros g g' Hg u. exact (Hg (h u)).
  Qed.

  Lemma precompose_add : ∀ g g' : LinearMap F W X,
    linear_map_comp (g + g') h == linear_map_comp g h + linear_map_comp g' h.
  Proof.
    intros g g' u. reflexivity.
  Qed.

  Lemma precompose_scale : ∀ (a : F) (g : LinearMap F W X),
    linear_map_comp (a *: g) h == a *: linear_map_comp g h.
  Proof.
    intros a g u. reflexivity.
  Qed.

  Definition precompose : LinearMap F (LinearMap F W X) (LinearMap F U X) :=
    linear_map_from_laws (fun g ⇒ linear_map_comp g h)
      precompose_proper precompose_add precompose_scale.

End HomPrecompose.

Section HomIso.
  Context {F U W X : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context {eX : EqRel X} {zX : Zero X} {aX : HasAdd X} {nX : HasNeg X} {sX : HasScale F X}.
  Context {VX : VectorSpace F X}.
  Context (i : VSIso F U W).

  Lemma precompose_fromto : ∀ g : LinearMap F W X,
    precompose (X := X) (VSIso.from i) (precompose (X := X) (VSIso.to i) g) == g.
  Proof.
    intros g w.
    apply (Linear.proper (f := LinearMap.map g)).
    exact (vsiso_to_from i w).
  Qed.

  Lemma precompose_tofrom : ∀ g : LinearMap F U X,
    precompose (X := X) (VSIso.to i) (precompose (X := X) (VSIso.from i) g) == g.
  Proof.
    intros g u.
    apply (Linear.proper (f := LinearMap.map g)).
    exact (vsiso_from_to i u).
  Qed.

  Definition vsiso_hom_dom : VSIso F (LinearMap F W X) (LinearMap F U X) :=
    vsiso_from_laws (precompose (VSIso.to i)) (precompose (VSIso.from i))
      precompose_fromto precompose_tofrom.

End HomIso.

Section DualSpaceBasis.
  Context {F V : Type} `{FF : Field F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context (B : Basis n I F V).
  Context {D : Type} {eD : EqRel D} {zD : Zero D} {aD : HasAdd D}
          {nD : HasNeg D} {sD : HasScale F D} {VD : VectorSpace F D}.
  Context (DS : DualSpace F V D).

  Definition dual_space_basis : Basis n I F D :=
    basis_transport (VSIso.from DualSpace.represent) (VSIso.to DualSpace.represent)
      (vsiso_to_from DualSpace.represent) (vsiso_from_to DualSpace.represent)
      (dual_basis B).

End DualSpaceBasis.

Section AbstractDoubleDual.
  Context {F V : Type} `{FF : Field F}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context (B : Basis n I F V).
  Context {D : Type} {eD : EqRel D} {zD : Zero D} {aD : HasAdd D}
          {nD : HasNeg D} {sD : HasScale F D} {VD : VectorSpace F D}.
  Context {DD : Type} {eDD : EqRel DD} {zDD : Zero DD} {aDD : HasAdd DD}
          {nDD : HasNeg DD} {sDD : HasScale F DD} {VDD : VectorSpace F DD}.
  Context (DS : DualSpace F V D) (DDS : DualSpace F D DD).

  Definition double_dual : DualSpace F D V := {|
    DualSpace.represent :=
      vsiso_trans (double_dual_iso B)
        (vsiso_hom_dom (DualSpace.represent (DualSpace := DS)));
  |}.

  Definition double_dual_abstract : VSIso F V DD :=
    dual_unique double_dual DDS.

End AbstractDoubleDual.

Section DoubleDualBasis.
  Context {F V : Type} `{VV : VectorSpace F V}.
  Context {n : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex n I}.
  Context (B : Basis n I F V).

  Theorem double_dual_basis_vec : ∀ j : I,
    Basis.vec (dual_basis (dual_basis B)) j
      == into_double_dual (Basis.vec B j).
  Proof. intros j f. reflexivity. Qed.

  Theorem double_dual_basis_coords : ∀ v : V,
    Basis.coords (dual_basis (dual_basis B)) (into_double_dual v)
      == Basis.coords B v.
  Proof. intros v j. reflexivity. Qed.

End DoubleDualBasis.
