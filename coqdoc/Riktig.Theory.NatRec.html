Section NatRecTheory.
  Context {N : Type} `{PeanoArith N} {natrec : HasNatRec N} `{!NatRec N natrec}.

Recursion at one is one step from the value at zero.
  Theorem natrec_one : ∀ (A : Type) (eA : EqRel A) (EA : Equivalence eA)
      (a : A) (f : N → A → A), Proper (equ ==> equ ==> equ) f →
    natrec A a f 1 == f 0 a.
  Proof.
    intros A eA EA a f Hf.
    calc (natrec A a f 1)
      == (natrec A a f (succ 0)) { (natrec_proper A eA EA a f Hf 1 (succ 0) one_succ) }
      == (f 0 (natrec A a f 0)) { (natrec_succ A eA EA a f 0) }
      == (f 0 a) { (Hf 0 0 (reflexivity 0) _ _ (natrec_zero A eA EA a f)) }.
  Qed.

End NatRecTheory.
