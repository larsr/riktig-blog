From Coq Require Import Morphisms.
From Riktig Require Import Ops Props Interface.Order Interface.VectorSpace
  Interface.Norm.

Module Integral.
  Class Integral (F V : Type) `{OrderedField.OrderedField F}
      {intgE : EqRel V} {intgZ : Zero V} {intgA : HasAdd V} {intgN : HasNeg V}
      {intgS : HasScale F V} {VS : VectorSpace F V}
      {intgNorm : HasNorm F V} {NS : NormedSpace F V} := {
    
    int : (F → V) → F → F → V;
    proper : ∀ f g : F → V, (∀ t : F, f t == g t) →
                      ∀ a b : F, a == b →
                      ∀ c d : F, c == d → int f a c == int g b d;
    linear_additive : ∀ (f g : F → V) (a b : F),
      int (fun t : F ⇒ f t + g t) a b == int f a b + int g a b;
    linear_scale : ∀ (c : F) (f : F → V) (a b : F),
      int (fun t : F ⇒ c *: f t) a b == c *: int f a b;
    additivity : ∀ (f : F → V) (a b c : F),
      int f a b + int f b c == int f a c;
    constant : ∀ (v : V) (a b : F),
      int (fun _ : F ⇒ v) a b == (b + - a) *: v;
    bound : ∀ (f : F → V) (a b M : F),
      (∀ t : F, a ≤ t → t ≤ b → ‖ f t ‖ ≤ M) →
      a ≤ b →
      ‖ int f a b ‖ ≤ M × (b + - a)
  }.
End Integral.
Notation Integral := Integral.Integral.
Notation int := Integral.int.

#[global] Instance Integral_int_Proper (F V : Type)
  `{OrderedField.OrderedField F}
  {intgE : EqRel V} {intgZ : Zero V} {intgA : HasAdd V} {intgN : HasNeg V}
  {intgS : HasScale F V} {VS : VectorSpace F V}
  {intgNorm : HasNorm F V} {NS : NormedSpace F V}
  (Int : Integral.Integral F V)
  : Proper (pointwise_relation F equ ==> equ ==> equ ==> equ) int
  := Integral.proper.
