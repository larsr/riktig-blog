Library Riktig.Interface.Integral

From Coq Require Import Morphisms.
From Riktig Require Import Interface.Norm.

Module Integral.
  Class Integral (F V : Type) `{OrderedField F}
      {Ve : EqRel V} {Vz : Zero V} {Va : HasAdd V} {Vn : HasNeg V}
      {Vs : HasScale F V} {VS : VectorSpace F V}
      {Vnorm : HasNorm F V} {NS : NormedSpace F V} := {
    int : (F → V) → F → F → V;
    proper :: Proper (pointwise_relation F equ ==> equ ==> equ ==> equ) int;

    is_additive :: ∀ a b : F,
      Additive (F → V) V (fun f g t ⇒ f t + g t) add (fun f ⇒ int f a b);
    
    is_homogeneous :: ∀ a b : F,
      @Homogeneous F (F → V) V Ve (fun c f t ⇒ c *: f t) (@Ops.scale F V Vs)
                   (fun c ⇒ c) (fun f ⇒ int f a b);
    additivity : ∀ (f : F → V) (a b c : F),
      int f a b + int f b c == int f a c;
    constant : ∀ (v : V) (a b : F),
      int (fun _ : F ⇒ v) a b == (b + - a) *: v;
    bound : ∀ (f : F → V) (a b M : F),
      (∀ t : F, a ≤ t → t ≤ b → ‖ f t ‖ ≤ M) →
      a ≤ b →
      ‖ int f a b ‖ ≤ M × (b + - a);
  }.
  Hint Mode Integral - ! - - - - - - - - - - - - - - - - - - - : typeclass_instances.

  Definition linear_additive {F V : Type} `{Integral F V}
    : ∀ (f g : F → V) (a b : F),
      int (fun t : F ⇒ f t + g t) a b == int f a b + int g a b :=
    fun f g a b ⇒ additive (f := fun h ⇒ int h a b) f g.

  Definition linear_scale {F V : Type} `{Integral F V}
    : ∀ (c : F) (f : F → V) (a b : F),
      int (fun t : F ⇒ c *: f t) a b == c *: int f a b :=
    fun c f a b ⇒ homogeneous (f := fun h ⇒ int h a b) c f.
End Integral.
Export Integral.
Notation int := Integral.int.
