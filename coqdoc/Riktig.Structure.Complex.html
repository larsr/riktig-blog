The complex numbers over a field: a commutative algebra with an element i squaring to - 1, every element being its real part plus its imaginary part times i, the two parts respecting equality, addition and scaling, with 1 and i the two units.
Module Complex.
  Class Complex (F C : Type) `{Algebra F C} `{!Commutative C C mult}
      (i : HasI C) (real : HasRe F C) (imag : HasIm F C) := {
    i_sq : (imaginary : C) × imaginary == - (1 : C);
    re_proper :: Proper (equ ==> equ) (@re F C real);
    im_proper :: Proper (equ ==> equ) (@im F C imag);
    re_additive :: Additive C F add add re;
    im_additive :: Additive C F add add im;
    re_homogeneous :: Homogeneous F C F mult (fun a ⇒ a) re;
    im_homogeneous :: Homogeneous F C F mult (fun a ⇒ a) im;
    re_one : re (1 : C) == (1 : F);
    im_one : im (1 : C) == (0 : F);
    re_i : re (imaginary : C) == (0 : F);
    im_i : im (imaginary : C) == (1 : F);
    re_im : ∀ z : C, z == re z *: 1 + im z *: imaginary;
  }.
  Hint Mode Complex - ! - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Complex.
Export Complex.
