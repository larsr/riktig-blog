Section GroupTheory.
  Context {G : Type} `{Group G}.

  Theorem neg_unique_l : ∀ g h : G, h + g == 0 → h == - g.
  Proof.
    intros g h Heq.
    calc h
      == (h + 0) { right_identity }
      == (h + (g + - g)) { right_inverse }
      == ((h + g) + - g) { associative }
      == (0 + - g) { Heq }
      == (- g) { left_identity }.
  Qed.

  Theorem neg_unique_r : ∀ g h : G, g + h == 0 → h == - g.
  Proof.
    intros g h Heq.
    calc h
      == (0 + h) { left_identity }
      == ((- g + g) + h) { left_inverse }
      == (- g + (g + h)) { associative }
      == (- g + 0) { Heq }
      == (- g) { right_identity }.
  Qed.

  Theorem cancel_l : ∀ a b c : G, a + b == a + c → b == c.
  Proof.
    intros a b c Heq.
    calc b
      == (0 + b) { left_identity }
      == ((- a + a) + b) { left_inverse }
      == (- a + (a + b)) { associative }
      == (- a + (a + c)) { Heq }
      == ((- a + a) + c) { associative }
      == (0 + c) { left_inverse }
      == c { left_identity }.
  Qed.

  Theorem cancel_r : ∀ a b c : G, b + a == c + a → b == c.
  Proof.
    intros a b c Heq.
    calc b
      == (b + 0) { right_identity }
      == (b + (a + - a)) { right_inverse }
      == ((b + a) + - a) { associative }
      == ((c + a) + - a) { Heq }
      == (c + (a + - a)) { associative }
      == (c + 0) { right_inverse }
      == c { right_identity }.
  Qed.

  Theorem neg_involutive : ∀ g : G, - (- g) == g.
  Proof.
    intros g.
    apply (cancel_l (- g)).
    calc ((- g) + - - g)
      == (0 : G) { right_inverse }
      == ((- g) + g) { left_inverse }.
  Qed.

  Theorem neg_zero : - (0 : G) == 0.
  Proof.
    apply (cancel_l 0).
    calc ((0 : G) + - 0)
      == (0 : G) { right_inverse }
      == ((0 : G) + 0) { right_identity }.
  Qed.

  Lemma add_sub_cancel_r : ∀ a b : G, (a + b) + - b == a.
  Proof.
    intros a b.
    calc ((a + b) + - b)
      == (a + (b + - b)) { associative }
      == a { (add_zero_r (right_inverse b)) }.
  Qed.

  Theorem neg_add_flip : ∀ a b : G, - (a + b) == (- b) + (- a).
  Proof.
    intros a b. symmetry. apply neg_unique_r.
    calc ((a + b) + (- b + - a))
      == (((a + b) + - b) + - a) { associative }
      == (a + - a) { add_sub_cancel_r }
      == (0 : G) { right_inverse }.
  Qed.

  Theorem neg_inj : ∀ a b : G, - a == - b → a == b.
  Proof.
    intros a b Heq.
    calc a
      == (- - a) { neg_involutive }
      == (- - b) { Heq }
      == b { neg_involutive }.
  Qed.

  Theorem add_neg_telescope : ∀ a b c : G,
    (a + - b) + (b + - c) == a + - c.
  Proof.
    intros a b c.
    calc ((a + - b) + (b + - c))
      == (a + (- b + (b + - c))) { associative }
      == (a + ((- b + b) + - c)) { associative }
      == (a + (0 + - c)) { left_inverse }
      == (a + - c) { left_identity }.
  Qed.

  Theorem sub_eq_zero : ∀ a b : G, a + - b == 0 → a == b.
  Proof.
    intros a b Heq.
    apply (cancel_r (- b)).
    calc (a + - b)
      == (0 : G) { Heq }
      == (b + - b) { right_inverse }.
  Qed.

  Lemma sub_add_cancel : ∀ a b : G, (a + - b) + b == a.
  Proof.
    intros a b.
    calc ((a + - b) + b)
      == (a + (- b + b)) { associative }
      == a { (add_zero_r (left_inverse b)) }.
  Qed.


  Lemma eq_sub_of_add_eq : ∀ a b c : G, a + c == b → a == b + - c.
  Proof.
    intros a b c Heq.
    calc (a : G)
      == ((a + c) + - c) { add_sub_cancel_r }
      == (b + - c) { Heq }.
  Qed.

  Lemma sub_add_mid_cancel : ∀ a x b : G, (a + - x) + (x + b) == a + b.
  Proof.
    intros a x b.
    calc ((a + - x) + (x + b))
      == (a + (- x + (x + b))) { associative }
      == (a + ((- x + x) + b)) { associative }
      == (a + b) { (add_zero_l (left_inverse x)) }.
  Qed.

End GroupTheory.

Instance Group_NegInjective G `{H : Group G} :
  Injective G G neg := neg_inj.

Section AdditiveTheory.
  Context {G K : Type} `{Group G}.
  Context `{Group K}.
  Context (f : G → K).
  Context `{fp : Proper _ (equ ==> equ) f}.
  Context {fa : Additive G K add add f}.

  Theorem additive_zero : f 0 == 0.
  Proof.
    apply (cancel_l (f 0)).
    calc (f 0 + f 0)
      == (f (0 + 0)) { additive }
      == (f 0) { left_identity }
      == (f 0 + 0) { right_identity }.
  Qed.

  Theorem additive_negation : ∀ a : G, f (- a) == - f a.
  Proof.
    intros a.
    apply neg_unique_l.
    calc (f (- a) + f a)
      == (f (- a + a)) { additive }
      == (f 0) { left_inverse }
      == (0 : K) { additive_zero }.
  Qed.

End AdditiveTheory.

Section AdditiveNormalForm.
  Context {G : Type} {eq : EqRel G} {zero : Zero G} {add : HasAdd G} {neg : HasNeg G}.
  Context `{!Group G}.

  Definition add_left_identity : ∀ x : G, 0 + x == x := left_identity.
  Definition add_right_identity : ∀ x : G, x + 0 == x := right_identity.
  Definition add_left_inverse : ∀ x : G, (- x) + x == 0 := left_inverse.
  Definition add_right_inverse : ∀ x : G, x + (- x) == 0 := right_inverse.
End AdditiveNormalForm.

                       @neg_involutive @neg_zero using typeclasses eauto : grp.

Section GroupRewriteExamples.
  Context {G : Type} `{Group G}.

  Example autorewrite_double_neg_idr : ∀ g : G, - (- g) + 0 == g.
  Proof. intros g. autorewrite with grp. reflexivity. Qed.

  Example autorewrite_multi : ∀ g : G, (0 + - (- g)) + 0 == g.
  Proof. intros g. autorewrite with grp. reflexivity. Qed.

  Example autorewrite_invl_invr : ∀ g : G,
    ((- g) + g) + (g + - g) == 0.
  Proof. intros g. autorewrite with grp. reflexivity. Qed.

End GroupRewriteExamples.
