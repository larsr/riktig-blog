Section GroupTheory.
  Context {G : Type} `{Group G}.

  Theorem add_left_cancel : ∀ a b c : G, a + b == a + c → b == c.
  Proof.
    intros a b c Heq.
    calc b
      == (0 + b) { left_identity }
      == ((- a + a) + b) { left_inverse }
      == (- a + (a + b)) { associative }
      == (- a + (a + c)) { Heq }
      == ((- a + a) + c) { associative }
      == (0 + c) { left_inverse }
      == c { left_identity }.
  Qed.

  Theorem add_right_cancel : ∀ a b c : G, b + a == c + a → b == c.
  Proof.
    intros a b c Heq.
    calc b
      == (b + 0) { right_identity }
      == (b + (a + - a)) { right_inverse }
      == ((b + a) + - a) { associative }
      == ((c + a) + - a) { Heq }
      == (c + (a + - a)) { associative }
      == (c + 0) { right_inverse }
      == c { right_identity }.
  Qed.

Inverses are unique: whatever cancels an element is its negation.
  Theorem eq_neg_of_add_eq_zero_left : ∀ a b : G, a + b == 0 → a == - b.
  Proof.
    intros a b Heq.
    apply (add_right_cancel b).
    calc (a + b)
      == (0 : G) { Heq }
      == (- b + b) { left_inverse }.
  Qed.

  Theorem eq_neg_of_add_eq_zero_right : ∀ a b : G, a + b == 0 → b == - a.
  Proof.
    intros a b Heq.
    apply (add_left_cancel a).
    calc (a + b)
      == (0 : G) { Heq }
      == (a + - a) { right_inverse }.
  Qed.

  Theorem neg_neg : ∀ a : G, - - a == a.
  Proof.
    intros a.
    symmetry.
    apply eq_neg_of_add_eq_zero_right.
    exact (left_inverse a).
  Qed.

  Theorem neg_zero : - (0 : G) == 0.
  Proof.
    symmetry.
    apply eq_neg_of_add_eq_zero_left.
    exact (right_identity 0).
  Qed.

  Theorem neg_inj : ∀ a b : G, - a == - b → a == b.
  Proof.
    intros a b Heq.
    calc a
      == (- - a) { neg_neg }
      == (- - b) { Heq }
      == b { neg_neg }.
  Qed.

  Theorem neg_add_rev : ∀ a b : G, - (a + b) == - b + - a.
  Proof.
    intros a b.
    symmetry.
    apply eq_neg_of_add_eq_zero_right.
    calc ((a + b) + (- b + - a))
      == (a + (b + (- b + - a))) { associative }
      == (a + ((b + - b) + - a)) { associative }
      == (a + (0 + - a)) { right_inverse }
      == (a + - a) { left_identity }
      == (0 : G) { right_inverse }.
  Qed.

Adding and then subtracting the same element, in the four positions rearrangements reach for.
  Theorem add_neg_cancel_right : ∀ a b : G, (a + b) + - b == a.
  Proof.
    intros a b.
    calc ((a + b) + - b)
      == (a + (b + - b)) { associative }
      == (a + 0) { right_inverse }
      == a { right_identity }.
  Qed.

  Theorem neg_add_cancel_right : ∀ a b : G, (a + - b) + b == a.
  Proof.
    intros a b.
    calc ((a + - b) + b)
      == (a + (- b + b)) { associative }
      == (a + 0) { left_inverse }
      == a { right_identity }.
  Qed.

  Theorem add_neg_cancel_left : ∀ a b : G, a + (- a + b) == b.
  Proof.
    intros a b.
    calc (a + (- a + b))
      == ((a + - a) + b) { associative }
      == (0 + b) { right_inverse }
      == b { left_identity }.
  Qed.

  Theorem neg_add_cancel_left : ∀ a b : G, - a + (a + b) == b.
  Proof.
    intros a b.
    calc (- a + (a + b))
      == ((- a + a) + b) { associative }
      == (0 + b) { left_inverse }
      == b { left_identity }.
  Qed.

  Theorem eq_of_sub_eq_zero : ∀ a b : G, a + - b == 0 → a == b.
  Proof.
    intros a b Heq.
    apply (add_right_cancel (- b)).
    calc (a + - b)
      == (0 : G) { Heq }
      == (b + - b) { right_inverse }.
  Qed.

  Theorem eq_sub_of_add_eq : ∀ a b c : G, a + c == b → a == b + - c.
  Proof.
    intros a b c Heq.
    calc a
      == ((a + c) + - c) { add_neg_cancel_right }
      == (b + - c) { Heq }.
  Qed.

The middle terms of a telescoping difference cancel.
  Theorem sub_add_sub_cancel : ∀ a b c : G,
    (a + - b) + (b + - c) == a + - c.
  Proof.
    intros a b c.
    calc ((a + - b) + (b + - c))
      == (a + (- b + (b + - c))) { associative }
      == (a + - c) { neg_add_cancel_left }.
  Qed.

End GroupTheory.

A map that respects addition sends zero to zero and negation to negation; the properness is what the rewrites inside need.
Section AdditiveMapTheory.
  Context {G K : Type} `{Group G} `{Group K} (f : G → K)
    `{!Proper (equ ==> equ) f} `{!Additive G K add add f}.

  Theorem additive_zero : f 0 == 0.
  Proof.
    apply (add_left_cancel (f 0)).
    calc (f 0 + f 0)
      == (f (0 + 0)) { additive[f] }
      == (f 0) { (left_identity (action := add) (0 : G)) }
      == (f 0 + 0) { right_identity }.
  Qed.

  Theorem additive_neg : ∀ a : G, f (- a) == - f a.
  Proof.
    intros a.
    apply eq_neg_of_add_eq_zero_left.
    calc (f (- a) + f a)
      == (f (- a + a)) { additive[f] }
      == (f 0) { (left_inverse (op := add) a) }
      == (0 : K) { additive_zero }.
  Qed.

End AdditiveMapTheory.
