Library Riktig.Interface.Monoid

From Coq Require Export Morphisms.
From Riktig Require Export Ops Props.

Module Monoid.
  Class Monoid (T : Type) {Te : EqRel T} {Tz : Zero T} {Ta : HasAdd T} := {
    equivalence : Equivalence (@equ T _);
    associative : ∀ x y w : T, (x + y) + w == x + (y + w);
    left_identity : ∀ x : T, 0 + x == x;
    right_identity : ∀ x : T, x + 0 == x;
    add_proper : ∀ x y : T, x == y → ∀ u v : T, u == v → (x + u) == (y + v)
  }.
End Monoid.
Notation Monoid := Monoid.Monoid.
