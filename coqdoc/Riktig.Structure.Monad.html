Generalizable All Variables.

A monad on setoids, as a Kleisli triple: returning and binding, each respecting equality, with return neutral for bind on either side and bind associative. The functions bound respect equality throughout.
Module Monad.
  Class Monad (P : Type → Type) {eqP : ∀ (A : Type) (eqA : EqRel A), EqRel (P A)}
      (return_ : HasReturn P) (bind_ : HasBind P) := {
    equivalence :: ∀ `{Setoid A}, Equivalence (@equ (P A) (eqP A eqA));
    ret_proper :: ∀ `{Setoid A}, Proper (equ ==> equ) (ret : A → P A);
    bind_proper : ∀ `{Setoid A} `{Setoid B} (p p' : P A) (f f' : A → P B),
      Proper (equ ==> equ) f → Proper (equ ==> equ) f' →
      p == p' → (∀ a : A, f a == f' a) → p >>= f == p' >>= f';
    bind_ret_l : ∀ `{Setoid A} `{Setoid B} (a : A) (f : A → P B),
      Proper (equ ==> equ) f → (ret a : P A) >>= f == f a;
    bind_ret_r : ∀ `{Setoid A} (p : P A), p >>= ret == p;
    bind_assoc : ∀ `{Setoid A} `{Setoid B} `{Setoid D} (p : P A) (f : A → P B) (g : B → P D),
      Proper (equ ==> equ) f → Proper (equ ==> equ) g →
      (p >>= f) >>= g == p >>= (fun a ⇒ f a >>= g);
  }.
End Monad.
