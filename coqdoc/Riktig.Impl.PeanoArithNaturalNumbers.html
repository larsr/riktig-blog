The natural numbers of Coq's types are Peano numbers with their arithmetic: zero and successor are the two halves of build, addition and multiplication are folds, and the Peano laws are initiality -- the successor is cancellable because build is an isomorphism, and induction is the uniqueness of the fold into the numbers that satisfy the property.
Section N.
  Context {N : Type} {build : HasBuild Type (NaturalNumbers.shape Type) N}
    {fold : HasCata Type (NaturalNumbers.shape Type) N} `{!NaturalNumbers Type N build fold}.

  #[export] Instance NN_EqRel : EqRel N := eq.
  #[export] Instance NN_Zero : Zero N := build (inl tt).
  #[export] Instance NN_HasSucc : HasSucc N := fun n ⇒ build (inr n).
  #[export] Instance NN_One : One N := succ 0.
  #[export] Instance NN_HasAdd : HasAdd N :=
    fun m ⇒ fold N (fun s ⇒ match s with inl _ ⇒ m | inr p ⇒ succ p end).
  #[export] Instance NN_HasMult : HasMult N :=
    fun m ⇒ fold N (fun s ⇒ match s with inl _ ⇒ 0 | inr p ⇒ p + m end).

  Notation unbuild := (fold (NaturalNumbers.shape Type N) (NaturalNumbers.map build)).

The recursion law, read at one number.
  Let fold_build (B : Type) (h : NaturalNumbers.shape Type B → B) (s : NaturalNumbers.shape Type N) :
    fold B h (build s) = h (NaturalNumbers.map (fold B h) s).
  Proof. pose proof (InitialAlgebra.cata_build (t := N) B h) as E. exact (E s). Qed.

  #[export] Instance NN_Peano : Peano N NN_Zero NN_HasSucc.
  Proof.
    split.
    - exact eq_equivalence.
    - intros n m Hnm. cbv [equ NN_EqRel] in ×. subst. reflexivity.
    - intros n m Hnm. cbv [equ NN_EqRel] in Hnm.
      transitivity (desc (build ∘ inj1) cat_id (unbuild (succ n))).
      { symmetry. exact (pred_succ n). }
      transitivity (desc (build ∘ inj1) cat_id (unbuild (succ m))).
      { exact (f_equal (fun k ⇒ desc (build ∘ inj1) cat_id (unbuild k)) Hnm). }
      exact (pred_succ m).
    - intros n Hn. cbv [equ NN_EqRel] in Hn.
      assert (E : (inr n : unit + N) = inl tt).
      { transitivity (unbuild (succ n)). { symmetry. exact (unbuild_succ n). }
        transitivity (unbuild 0). { exact (f_equal (fun k ⇒ unbuild k) Hn). }
        exact (unbuild_zero tt). }
      discriminate E.
    - intros P HP H0 HS n. exact (naturals_induction P H0 HS n).
  Qed.

  #[export] Instance NN_PeanoArith : PeanoArith N NN_Zero NN_HasSucc NN_One NN_HasAdd NN_HasMult.
  Proof.
    split.
    - exact NN_Peano.
    - intros x y Hx u v Hu. cbv [equ NN_EqRel] in ×. subst. reflexivity.
    - intros x y Hx u v Hu. cbv [equ NN_EqRel] in ×. subst. reflexivity.
    - reflexivity.
    - intros a. exact (fold_build N _ (inl tt)).
    - intros a b. exact (fold_build N _ (inr b)).
    - intros a. exact (fold_build N _ (inl tt)).
    - intros a b. exact (fold_build N _ (inr b)).
  Qed.
End N.
