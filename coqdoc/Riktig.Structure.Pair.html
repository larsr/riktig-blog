From Riktig.Structure Require Export Adt.

A pair of two objects named by a and b: the datatype of the constant signature const a × const b, with no recursion.
Module Pair.
  Notation shape den a b := (shape_of den (pairS (S := sig _) a b)).

  Class Pair (O : Type) `{Bicartesian O} {K : Type}
      (den : K → O) (a b : K) (P : O)
      (build : HasBuild O (shape den a b) P)
      (fold : HasCata O (shape den a b) P) := {
    initial :: InitialAlgebra O (shape den a b) P build fold;
  }.

  Notation mk := (build (F := shape _ _ _)).
  Notation first := (cata (F := shape _ _ _) _ pi1).
  Notation second := (cata (F := shape _ _ _) _ pi2).
End Pair.
