Recursion on lambda terms: from one answer per constructor, a function on terms, given by its three equations, which respects equality when the answers do.
Module TermRec.
  Class TermRec (N T : Type) {eqN : EqRel N} {eqT : EqRel T}
      (varT : HasVar N T) (lamT : HasLam T) (appT : HasApp T)
      (rec : HasTermRec N T) := {
    termrec_proper : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : B → B) (a : B → B → B),
      Proper (equ ==> equ) v → Proper (equ ==> equ) l →
      Proper (equ ==> equ ==> equ) a →
      Proper (equ ==> equ) (termrec B v l a : T → B);
    termrec_var : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : B → B) (a : B → B → B) (n : N),
      termrec B v l a (var n : T) == v n;
    termrec_lam : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : B → B) (a : B → B → B) (b : T),
      termrec B v l a (lam b) == l (termrec B v l a b);
    termrec_app : ∀ (B : Type) (eB : EqRel B) (EB : Equivalence eB)
        (v : N → B) (l : B → B) (a : B → B → B) (f u : T),
      termrec B v l a (app f u) == a (termrec B v l a f) (termrec B v l a u);
  }.
End TermRec.
