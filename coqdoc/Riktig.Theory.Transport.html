Library Riktig.Theory.Transport

From Riktig Require Export Theory.MatRep.


Section BasisTransport.
  Context {F W W' : Type} `{FF : Field F}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {eW' : EqRel W'} {zW' : Zero W'} {aW' : HasAdd W'} {nW' : HasNeg W'} {sW' : HasScale F W'}.
  Context {VW : VectorSpace F W} {VW' : VectorSpace F W'}.
  Context (T : LinearMap F W W') (Tinv : LinearMap F W' W).
  Context (Hlr : ∀ w : W, Tinv (T w) == w)
          (Hrl : ∀ w' : W', T (Tinv w') == w').
  Context {k : nat} {I : Type} {eI : EqRel I} {Ieq : EquDec I}
    {IFin : FinIndex k I}.
  Context (Bk : Basis k I F W).

  Definition bt_vec (i : I) : W' := T (Basis.vec Bk i).

  Lemma bt_vec_proper : Proper (equ ==> equ) bt_vec.
  Proof.
    intros i i' Hii'.
    exact (linear_map_proper T (Basis.vec Bk i) (Basis.vec Bk i')
             (Basis.vec_proper Bk i i' Hii')).
  Qed.

  Definition bt_coords : LinearMap F W' (I → F) :=
    linear_map_comp (Basis.coords Bk) Tinv.

  Lemma bt_recon : ∀ w' : W', lincomb bt_vec (bt_coords w') == w'.
  Proof.
    intros w'.
    unfold bt_coords, linear_map_comp; simpl.
    calc (lincomb bt_vec (Basis.coords Bk (Tinv w')))
      == (T (lincomb (Basis.vec Bk) (Basis.coords Bk (Tinv w'))))
      == (T (Tinv w'))
      == w'.
    - symmetry.
      exact (lm_lincomb T (Basis.vec Bk) (Basis.coords Bk (Tinv w'))).
    - apply (linear_map_proper T).
      exact (Basis.reconstruction Bk (Tinv w')).
    - exact (Hrl w').
  Qed.

  Lemma bt_coords_lincomb : ∀ c : I → F,
    Proper (equ ==> equ) c →
    bt_coords (lincomb bt_vec c) == c.
  Proof.
    intros c Pc.
    unfold bt_coords, linear_map_comp; simpl.
    calc (Basis.coords Bk (Tinv (lincomb bt_vec c)))
      == (Basis.coords Bk (Tinv (T (lincomb (Basis.vec Bk) c))))
      == (Basis.coords Bk (lincomb (Basis.vec Bk) c))
      == c.
    - apply (linear_map_proper (Basis.coords Bk)), (linear_map_proper Tinv).
      symmetry.
      exact (lm_lincomb T (Basis.vec Bk) c).
    - apply (linear_map_proper (Basis.coords Bk)).
      exact (Hlr (lincomb (Basis.vec Bk) c)).
    - exact (Basis.coords_linear_combination Bk c Pc).
  Qed.

  Definition basis_transport : Basis k I F W' :=
    {| Basis.vec := bt_vec; Basis.vec_proper := bt_vec_proper;
       Basis.coords := bt_coords;
       Basis.coords_family_proper := fun w' ⇒
         Basis.coords_family_proper Bk (Tinv w');
       Basis.reconstruction := bt_recon;
       Basis.coords_linear_combination := bt_coords_lincomb |}.

End BasisTransport.
