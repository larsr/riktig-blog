Library Riktig.Theory.Transport

From Coq Require Import Morphisms Setoid.
From Riktig Require Export Interface.VectorSpace.


Class ring_iso (A B : Type)
    {eqA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A}
    {oA : One A} {mA : HasMult A}
    {eqB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B}
    {oB : One B} {mB : HasMult B} := {
  iso_to : A → B;
  iso_of : B → A;
  iso_to_proper : Proper (equ ==> equ) iso_to;
  iso_of_proper : Proper (equ ==> equ) iso_of;
  iso_of_to : ∀ a : A, iso_of (iso_to a) == a;
  iso_to_zero : iso_to 0 == 0;
  iso_to_one : iso_to 1 == 1;
  iso_to_add : ∀ x y : A, iso_to (x + y) == iso_to x + iso_to y;
  iso_to_neg : ∀ x : A, iso_to (- x) == - iso_to x;
  iso_to_mul : ∀ x y : A, iso_to (x × y) == iso_to x × iso_to y;
}.

Arguments iso_to {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_of {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_to_proper {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_of_proper {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_of_to {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_to_zero {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_to_one {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_to_add {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_to_neg {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.
Arguments iso_to_mul {A B eqA zA aA nA oA mA eqB zB aB nB oB mB}.

Section RingTransport.
  Context {A B : Type}.
  Context {eqA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A}
          {oA : One A} {mA : HasMult A}.
  Context {eqB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B}
          {oB : One B} {mB : HasMult B}.
  Context {EqvA : Equivalence (@equ A eqA)}.
  Context {RB : Ring B}.
  Context (h : ring_iso A B).


  Let pullback : ∀ x y : A, iso_to h x == iso_to h y → x == y.
  Proof.
    intros x y Ht.
    rewrite <- (iso_of_to h x), <- (iso_of_to h y).
    apply (iso_of_proper h). exact Ht.
  Qed.

  Lemma iso_add_proper : Proper (equ ==> equ ==> equ) (@add A aA).
  Proof.
    intros x x' Hx y y' Hy. apply pullback.
    rewrite !(iso_to_add h).
    rewrite (iso_to_proper h x x' Hx), (iso_to_proper h y y' Hy).
    reflexivity.
  Qed.

  Lemma iso_neg_proper : Proper (equ ==> equ) (@neg A nA).
  Proof.
    intros x x' Hx. apply pullback.
    rewrite !(iso_to_neg h).
    rewrite (iso_to_proper h x x' Hx). reflexivity.
  Qed.

  Lemma iso_mul_proper : Proper (equ ==> equ ==> equ) (@mult A mA).
  Proof.
    intros x x' Hx y y' Hy. apply pullback.
    rewrite !(iso_to_mul h).
    rewrite (iso_to_proper h x x' Hx), (iso_to_proper h y y' Hy).
    reflexivity.
  Qed.

  Lemma iso_add_associative : ∀ x y z : A, (x + y) + z == x + (y + z).
  Proof.
    intros x y z. apply pullback.
    rewrite !(iso_to_add h). apply associative.
  Qed.

  Lemma iso_add_left_identity : ∀ x : A, 0 + x == x.
  Proof.
    intros x. apply pullback.
    rewrite (iso_to_add h), (iso_to_zero h). apply left_identity.
  Qed.

  Lemma iso_add_right_identity : ∀ x : A, x + 0 == x.
  Proof.
    intros x. apply pullback.
    rewrite (iso_to_add h), (iso_to_zero h). apply right_identity.
  Qed.

  Lemma iso_add_left_inverse : ∀ x : A, (- x) + x == 0.
  Proof.
    intros x. apply pullback.
    rewrite (iso_to_add h), (iso_to_neg h), (iso_to_zero h).
    apply left_inverse.
  Qed.

  Lemma iso_add_right_inverse : ∀ x : A, x + (- x) == 0.
  Proof.
    intros x. apply pullback.
    rewrite (iso_to_add h), (iso_to_neg h), (iso_to_zero h).
    apply right_inverse.
  Qed.

  Lemma iso_add_commutative : ∀ x y : A, x + y == y + x.
  Proof.
    intros x y. apply pullback.
    rewrite !(iso_to_add h). apply commutative.
  Qed.

  Lemma iso_mul_associative : ∀ x y z : A, (x × y) × z == x × (y × z).
  Proof.
    intros x y z. apply pullback.
    rewrite !(iso_to_mul h). apply associative.
  Qed.

  Lemma iso_mul_left_identity : ∀ x : A, 1 × x == x.
  Proof.
    intros x. apply pullback.
    rewrite (iso_to_mul h), (iso_to_one h). apply left_identity.
  Qed.

  Lemma iso_mul_right_identity : ∀ x : A, x × 1 == x.
  Proof.
    intros x. apply pullback.
    rewrite (iso_to_mul h), (iso_to_one h). apply right_identity.
  Qed.

  Lemma iso_left_distributive : ∀ x y z : A,
    x × (y + z) == x × y + x × z.
  Proof.
    intros x y z. apply pullback.
    rewrite (iso_to_mul h), !(iso_to_add h), !(iso_to_mul h).
    apply left_distributive.
  Qed.

  Lemma iso_right_distributive : ∀ x y z : A,
    (x + y) × z == x × z + y × z.
  Proof.
    intros x y z. apply pullback.
    rewrite (iso_to_mul h), !(iso_to_add h), !(iso_to_mul h).
    apply right_distributive.
  Qed.

End RingTransport.

Section ScaleTransport.
  Context {F A B : Type}.
  Context {eqF : EqRel F} {zF : Zero F} {aF : HasAdd F} {nF : HasNeg F}
          {oF : One F} {mF : HasMult F} {iF : HasInv F} {FF : Field F}.
  Context {eqA : EqRel A} {zA : Zero A} {aA : HasAdd A} {nA : HasNeg A}
          {oA : One A} {mA : HasMult A} {sA : HasScale F A}.
  Context {eqB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B}
          {oB : One B} {mB : HasMult B} {sB : HasScale F B}.
  Context {EqvA : Equivalence (@equ A eqA)}.
  Context {RB : Ring B} {LB : ScaleLaws F B}.
  Context (h : ring_iso A B).
  Context (to_scale : ∀ (a : F) (x : A),
              iso_to h (a *: x) == a *: iso_to h x).

  Let pullback : ∀ x y : A, iso_to h x == iso_to h y → x == y.
  Proof.
    intros x y Ht.
    rewrite <- (iso_of_to h x), <- (iso_of_to h y).
    apply (iso_of_proper h). exact Ht.
  Qed.

  Lemma iso_scale_proper : Proper (equ ==> equ ==> equ) (@scale F A sA).
  Proof.
    intros a a' Ha x x' Hx. apply pullback.
    rewrite !to_scale.
    rewrite Ha, (iso_to_proper h x x' Hx). reflexivity.
  Qed.

  Lemma iso_scale_add : ∀ (a : F) (x y : A),
    a *: (x + y) == a *: x + a *: y.
  Proof.
    intros a x y. apply pullback.
    rewrite to_scale, !(iso_to_add h), !to_scale.
    apply left_distributive.
  Qed.

  Lemma iso_add_scale : ∀ (a b : F) (x : A),
    (a + b) *: x == a *: x + b *: x.
  Proof.
    intros a b x. apply pullback.
    rewrite to_scale, (iso_to_add h), !to_scale.
    apply right_distributive.
  Qed.

  Lemma iso_scale_assoc : ∀ (a b : F) (x : A),
    (a × b) *: x == a *: (b *: x).
  Proof.
    intros a b x. apply pullback.
    rewrite !to_scale.
    apply associative.
  Qed.

  Lemma iso_scale_one : ∀ x : A, 1 *: x == x.
  Proof.
    intros x. apply pullback.
    rewrite to_scale. apply left_identity.
  Qed.

  Lemma iso_scale_mul_l
      (HB : ∀ (a : F) (x y : B), (a *: x) × y == a *: (x × y)) :
    ∀ (a : F) (x y : A), (a *: x) × y == a *: (x × y).
  Proof.
    intros a x y. apply pullback.
    rewrite (iso_to_mul h), to_scale, to_scale, (iso_to_mul h).
    apply HB.
  Qed.

  Lemma iso_scale_mul_r
      (HB : ∀ (a : F) (x y : B), x × (a *: y) == a *: (x × y)) :
    ∀ (a : F) (x y : A), x × (a *: y) == a *: (x × y).
  Proof.
    intros a x y. apply pullback.
    rewrite (iso_to_mul h), to_scale, to_scale, (iso_to_mul h).
    apply HB.
  Qed.

End ScaleTransport.
