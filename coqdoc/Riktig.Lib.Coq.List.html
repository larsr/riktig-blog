From Coq Require Import List Arith Lia.
Import ListNotations.

A list read as a table over an index I: the index only says how the list is read, so it is a phantom parameter of the type.
Definition ListTable (I A : Type) : Type := list A.

Definition tabulate_list {A : Type} (n : nat) (f : nat → A) : list A := map f (seq 0 n).

Lemma nth_tabulate_list : ∀ (A : Type) (f : nat → A) (n k : nat) (d : A),
  k < n → nth k (tabulate_list n f) d = f k.
Proof.
  intros A f n k d Hk. unfold tabulate_list.
  rewrite (nth_indep _ d (f 0)) by (rewrite length_map, length_seq; exact Hk).
  rewrite map_nth, seq_nth by exact Hk. reflexivity.
Qed.
