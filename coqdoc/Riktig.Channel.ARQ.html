Library Riktig.Channel.ARQ


From Coq Require Export List.
From Riktig.Theory Require Export Subseq.
From Riktig.Channel Require Export Trace.
Import ListNotations.


Record Protocol (Tag M Packet : Type) := {
  
  tag_eqb : Tag → Tag → bool;
  tag_next : Tag → Tag;
  
  mk : Tag → M → Packet;
  tag_of : Packet → Tag;
  msg_of : Packet → M;
  
  tag_eqb_true : ∀ t u : Tag, t = u → tag_eqb t u = true;
  tag_eqb_false : ∀ t u : Tag, t ≠ u → tag_eqb t u = false;
  tag_dec : ∀ t u : Tag, {t = u} + {t ≠ u};
  tag_next_neq : ∀ t : Tag, t ≠ tag_next t;
  tag_of_mk : ∀ (t : Tag) (m : M), tag_of (mk t m) = t;
  msg_of_mk : ∀ (t : Tag) (m : M), msg_of (mk t m) = m;
}.

Arguments tag_eqb {Tag M Packet}.
Arguments tag_next {Tag M Packet}.
Arguments mk {Tag M Packet}.
Arguments tag_of {Tag M Packet}.
Arguments msg_of {Tag M Packet}.
Arguments tag_eqb_true {Tag M Packet}.
Arguments tag_eqb_false {Tag M Packet}.
Arguments tag_dec {Tag M Packet}.
Arguments tag_next_neq {Tag M Packet}.
Arguments tag_of_mk {Tag M Packet}.
Arguments msg_of_mk {Tag M Packet}.

Section ARQ.
  Context {Tag M Packet : Type}.
  Context (pro : Protocol Tag M Packet).


  Fixpoint burst (t : Tag) (m : M) (k : nat) : list Packet :=
    match k with
    | O ⇒ []
    | S k' ⇒ mk pro t m :: burst t m k'
    end.

  Fixpoint stream (t : Tag) (ms : list M) (k : nat) : list Packet :=
    match ms with
    | [] ⇒ []
    | m :: ms' ⇒ burst t m k ++ stream (tag_next pro t) ms' k
    end.


  Fixpoint deliver (expect : Tag) (ps : list Packet) : list M :=
    match ps with
    | [] ⇒ []
    | p :: ps' ⇒
        if tag_eqb pro (tag_of pro p) expect
        then msg_of pro p :: deliver (tag_next pro expect) ps'
        else deliver expect ps'
    end.

  Fixpoint next (expect : Tag) (ps : list Packet) : Tag :=
    match ps with
    | [] ⇒ expect
    | p :: ps' ⇒
        if tag_eqb pro (tag_of pro p) expect
        then next (tag_next pro expect) ps'
        else next expect ps'
    end.


  Lemma deliver_app : ∀ (w1 w2 : list Packet) (expect : Tag),
    deliver expect (w1 ++ w2) = deliver expect w1 ++ deliver (next expect w1) w2.
  Proof.
    induction w1 as [| p w1' IH]; intros w2 expect.
    - simpl. reflexivity.
    - simpl. destruct (tag_eqb pro (tag_of pro p) expect) eqn:Hcmp.
      + simpl. rewrite (IH w2 (tag_next pro expect)). reflexivity.
      + apply IH.
  Qed.


  Lemma burst_all_eq : ∀ (t : Tag) (m : M) (k : nat) (p : Packet),
    In p (burst t m k) → p = mk pro t m.
  Proof.
    intros t m k.
    induction k as [| k' IH]; intros p Hp.
    - simpl in Hp. destruct Hp.
    - simpl in Hp. destruct Hp as [Hp | Hp].
      + symmetry. exact Hp.
      + apply IH. exact Hp.
  Qed.


  Lemma deliver_homog_neq : ∀ (x0 : Packet) (w : list Packet) (expect : Tag),
    (∀ p, In p w → p = x0) → tag_of pro x0 ≠ expect →
    deliver expect w = [] ∧ next expect w = expect.
  Proof.
    intros x0 w.
    induction w as [| p w' IH]; intros expect Hall Hneq.
    - split; reflexivity.
    - assert (Hp : p = x0).
      { apply Hall. simpl. left. reflexivity. }
      rewrite Hp. simpl.
      rewrite (tag_eqb_false pro (tag_of pro x0) expect Hneq).
      apply IH.
      + intros q Hq. apply Hall. simpl. right. exact Hq.
      + exact Hneq.
  Qed.

  Lemma deliver_homog_eq : ∀ (x0 : Packet) (w : list Packet) (expect : Tag),
    (∀ p, In p w → p = x0) → tag_of pro x0 = expect →
    (deliver expect w = [] ∧ next expect w = expect) ∨
    (deliver expect w = [msg_of pro x0] ∧ next expect w = tag_next pro expect).
  Proof.
    intros x0 w expect Hall Heq.
    destruct w as [| p w'].
    - left. split; reflexivity.
    - assert (Hp : p = x0).
      { apply Hall. simpl. left. reflexivity. }
      right.
      rewrite Hp. simpl.
      rewrite (tag_eqb_true pro (tag_of pro x0) expect Heq). simpl.
      assert (Hnw : ∀ q, In q w' → q = x0).
      { intros q Hq. apply Hall. simpl. right. exact Hq. }
      assert (Hneq2 : tag_of pro x0 ≠ tag_next pro expect).
      { rewrite Heq. exact (tag_next_neq pro expect). }
      destruct (deliver_homog_neq x0 w' (tag_next pro expect) Hnw Hneq2) as [Hd Hn].
      rewrite Hd. rewrite Hn.
      split; reflexivity.
  Qed.


  Lemma burst_delivery : ∀ (t : Tag) (m : M) (k : nat) (w : list Packet)
      (expect : Tag),
    lossy_dup (burst t m k) w →
    (deliver expect w = [] ∧ next expect w = expect) ∨
    (deliver expect w = [m] ∧ next expect w = tag_next pro expect).
  Proof.
    intros t m k w expect H.
    assert (Hall : ∀ p, In p w → p = mk pro t m).
    { apply (lossy_dup_preserves_all_eq (mk pro t m) (burst t m k) w H).
      intros p Hp. apply (burst_all_eq t m k p Hp). }
    destruct (tag_dec pro (tag_of pro (mk pro t m)) expect) as [Heq | Hneq].
    - destruct (deliver_homog_eq (mk pro t m) w expect Hall Heq)
        as [[Hd Hn] | [Hd Hn]].
      + left. split; [exact Hd | exact Hn].
      + right. rewrite (msg_of_mk pro t m) in Hd. split; [exact Hd | exact Hn].
    - left. apply (deliver_homog_neq (mk pro t m) w expect Hall Hneq).
  Qed.

  Lemma burst_delivery_hit : ∀ (t : Tag) (m : M) (k : nat) (w : list Packet),
    lossy_dup (burst t m k) w → w ≠ [] →
    deliver t w = [m] ∧ next t w = tag_next pro t.
  Proof.
    intros t m k w H Hne.
    assert (Hall : ∀ p, In p w → p = mk pro t m).
    { apply (lossy_dup_preserves_all_eq (mk pro t m) (burst t m k) w H).
      intros p Hp. apply (burst_all_eq t m k p Hp). }
    destruct w as [| p w'].
    - destruct (Hne eq_refl).
    - assert (Hp : p = mk pro t m).
      { apply Hall. simpl. left. reflexivity. }
      rewrite Hp. simpl.
      rewrite (tag_eqb_true pro (tag_of pro (mk pro t m)) t (tag_of_mk pro t m)).
      simpl.
      assert (Hall' : ∀ q, In q w' → q = mk pro t m).
      { intros q Hq. apply Hall. simpl. right. exact Hq. }
      assert (Hneq : tag_of pro (mk pro t m) ≠ tag_next pro t).
      { rewrite (tag_of_mk pro t m). exact (tag_next_neq pro t). }
      destruct (deliver_homog_neq (mk pro t m) w' (tag_next pro t) Hall' Hneq)
        as [Hd Hn].
      rewrite Hd. rewrite Hn. rewrite (msg_of_mk pro t m).
      split; reflexivity.
  Qed.

  Lemma burst_miss : ∀ (t : Tag) (m : M) (k : nat) (w : list Packet)
      (expect : Tag),
    lossy_dup (burst t m k) w → t ≠ expect →
    deliver expect w = [] ∧ next expect w = expect.
  Proof.
    intros t m k w expect H Hneq.
    assert (Hall : ∀ p, In p w → p = mk pro t m).
    { apply (lossy_dup_preserves_all_eq (mk pro t m) (burst t m k) w H).
      intros p Hp. apply (burst_all_eq t m k p Hp). }
    apply (deliver_homog_neq (mk pro t m) w expect Hall).
    rewrite (tag_of_mk pro t m). exact Hneq.
  Qed.


  Lemma deliver_subseq : ∀ (k : nat) (ms : list M) (t0 expect : Tag)
      (wire : list Packet),
    lossy_dup (stream t0 ms k) wire → subseq (deliver expect wire) ms.
  Proof.
    intros k ms.
    induction ms as [| m ms' IH]; intros t0 expect wire H.
    - simpl in H.
      assert (Hnil : wire = []).
      { apply (lossy_dup_nil_l [] wire H). reflexivity. }
      rewrite Hnil. simpl. apply subseq_nil.
    - simpl in H.
      destruct (lossy_dup_app_split _ _ H
                  (burst t0 m k) (stream (tag_next pro t0) ms' k)
                  eq_refl) as [w1 [w2 [Hwire [H1 H2]]]].
      rewrite Hwire.
      rewrite (deliver_app w1 w2 expect).
      destruct (burst_delivery t0 m k w1 expect H1) as [[Hd1 Hn1] | [Hd1 Hn1]].
      + rewrite Hd1. rewrite Hn1. simpl.
        apply subseq_drop.
        apply (IH (tag_next pro t0) expect w2 H2).
      + rewrite Hd1. rewrite Hn1. simpl.
        apply subseq_keep.
        apply (IH (tag_next pro t0) (tag_next pro expect) w2 H2).
  Qed.

  Theorem safety : ∀ (k : nat) (t0 : Tag) (ms : list M) (wire : list Packet),
    lossy_dup (stream t0 ms k) wire → subseq (deliver t0 wire) ms.
  Proof.
    intros k t0 ms wire H.
    apply (deliver_subseq k ms t0 t0 wire H).
  Qed.

End ARQ.
