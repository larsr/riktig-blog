From Riktig Require Export Structure.Reals Theory.Order.

Section RealsTheory.
  Context {R : Type} `{Reals R}.

A least upper bound is unique.
  Theorem lub_unique : ∀ (P : Pred R) (s t : R),
    (∀ x : R, P x → x ≤ s) →
    (∀ b : R, (∀ x : R, P x → x ≤ b) → s ≤ b) →
    (∀ x : R, P x → x ≤ t) →
    (∀ b : R, (∀ x : R, P x → x ≤ b) → t ≤ b) →
    s == t.
  Proof.
    intros P s t Hs Hs_least Ht Ht_least.
    apply antisymmetry.
    - exact (Hs_least t Ht).
    - exact (Ht_least s Hs).
  Qed.

End RealsTheory.
