Library Riktig.Theory.Reals

From Riktig.Structure Require Export Reals.
From Riktig.Theory Require Export Order.

Section RealsTheory.
  Context {R : Type} `{Reals R}.

A least upper bound is unique.
  Theorem lub_unique : ∀ (P : Pred R) (s t : R),
    (∀ x : R, P x → x ≤ s) →
    (∀ b : R, (∀ x : R, P x → x ≤ b) → s ≤ b) →
    (∀ x : R, P x → x ≤ t) →
    (∀ b : R, (∀ x : R, P x → x ≤ b) → t ≤ b) →
    s == t.
  Proof.
    intros P s t Hs Hs_least Ht Ht_least.
    apply antisymmetry.
    - exact (Hs_least t Ht).
    - exact (Ht_least s Hs).
  Qed.

A greatest element of a set is its least upper bound.
  Theorem lub_of_greatest : ∀ (P : Pred R) (b : R),
    P b → (∀ x : R, P x → x ≤ b) →
    ∃ s : R,
      (∀ x : R, P x → x ≤ s) ∧
      (∀ c : R, (∀ x : R, P x → x ≤ c) → s ≤ c) ∧
      s == b.
  Proof.
    intros P b HPb Hub.
    destruct (Reals.complete P (ex_intro _ b HPb)
                (ex_intro (fun c : R ⇒ ∀ x : R, P x → x ≤ c) b Hub))
      as [s [Hub' Hleast]].
    ∃ s. split; [ exact Hub' | split; [ exact Hleast | ] ].
    exact (lub_unique P s b Hub' Hleast Hub (fun c Hc ⇒ Hc b HPb)).
  Qed.

The numbers at most r have r as their least upper bound.
  Theorem lub_of_le_self : ∀ r : R,
    ∃ s : R,
      (∀ x : R, x ≤ r → x ≤ s) ∧
      (∀ c : R, (∀ x : R, x ≤ r → x ≤ c) → s ≤ c) ∧
      s == r.
  Proof.
    intros r. apply (lub_of_greatest (fun x : R ⇒ x ≤ r) r).
    - reflexivity.
    - intros x Hx. exact Hx.
  Qed.

End RealsTheory.
