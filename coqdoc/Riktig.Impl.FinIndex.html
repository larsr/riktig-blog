Fixpoint list_lincomb_fix {F V : Type} {Vz : Zero V} {Va : HasAdd V}
    {Vs : HasScale F V} (coeffs : list F) (vecs : list V) : V :=
  match coeffs with
  | nil ⇒ 0
  | c :: coeffs' ⇒
    match vecs with
    | nil ⇒ 0
    | v :: vecs' ⇒ c *: v + list_lincomb_fix coeffs' vecs'
    end
  end.

Instance HasListLincomb_fix {F V : Type} {Vz : Zero V} {Va : HasAdd V}
    {Vs : HasScale F V} : HasListLincomb F V := list_lincomb_fix.

Instance ListLincomb_fix {F V : Type} {Ve : EqRel V} {Vz : Zero V}
    {Va : HasAdd V} {Vs : HasScale F V} {VR : Reflexive (@equ V Ve)} :
  ListLincomb F V.
Proof.
  split; intros; reflexivity.
Qed.

Fixpoint bigopF {M : Type} `{Monoid M} {n : nat} (f : Fin.t n → M)
    {struct n} : M :=
  match n as n0 return ((Fin.t n0 → M) → M) with
  | O ⇒ fun _ ⇒ 0
  | S m ⇒ fun g ⇒ g Fin.F1 + bigopF (fun i ⇒ g (Fin.FS i))
  end f.

Section BigOpFinEquations.
  Context {M : Type} `{Monoid M}.

  Theorem bigopF_0 : ∀ f : Fin.t O → M, bigopF f == 0.
  Proof.
    intros f. reflexivity.
  Qed.

  Theorem bigopF_S : ∀ n (f : Fin.t (S n) → M),
    bigopF f == f Fin.F1 + bigopF (fun i ⇒ f (Fin.FS i)).
  Proof.
    intros n f. reflexivity.
  Qed.

  Theorem bigop_bigopF : ∀ n (f : nat → M),
    bigopF (fun i : Fin.t n ⇒ f (fin_to_nat i)) == bigop n f.
  Proof.
    intros n; induction n as [| n IH]; intros f.
    - reflexivity.
    - calc (bigopF (fun i : Fin.t (S n) ⇒ f (fin_to_nat i)))
        == (f O + bigopF (fun i : Fin.t n ⇒ f (S (fin_to_nat i))))
          { bigopF_S }
        == (f O + bigop n (fun j ⇒ f (S j))) { IH }
        == (bigop (S n) f) { bigop_S }.
  Qed.

  Lemma bigopF_enum : ∀ (n : nat) (f : Fin.t n → M),
    bigopF f == bigop_list f (fin_all n).
  Proof.
    intros n; induction n as [| m IH]; intros f.
    - reflexivity.
    - calc (bigopF f)
        == (f Fin.F1 + bigopF (fun i ⇒ f (Fin.FS i)))
             { bigopF_S }
        == (f Fin.F1 + bigop_list (fun i ⇒ f (Fin.FS i)) (fin_all m))
             { IH }
        == (f Fin.F1 + bigop_list f (map Fin.FS (fin_all m)))
             { bigop_list_map }
        == (bigop_list f (fin_all (S m)))
             { bigop_list_cons }.
  Qed.

End BigOpFinEquations.

Definition finindex_from_leibniz {n : nat} {I : Type} {eqdec : EqDec I}
    (l : list I) (nd : NoDup l) (cp : ∀ i : I, In i l)
    (len : length l = n) : FinIndex n I (eI := eq) :=
  {| FinIndex.enum_equivalence := eq_equivalence;
     FinIndex.enum := l;
     FinIndex.enum_NoDup := NoDupA_eq_of_NoDup I l nd;
     FinIndex.enum_complete := fun i ⇒ InA_eq_of_In I i l (cp i);
     FinIndex.enum_length := len |}.


Instance FinIndex_Fin (n : nat) : FinIndex n (Fin.t n) :=
  finindex_from_leibniz (fin_all n) (fin_all_NoDup n)
    (fin_all_complete n) (fin_all_length n).

Instance Unit_EqDec : EqDec unit.
Proof. intros [] []. left. reflexivity. Defined.

Lemma unit_enum_NoDup : NoDup (tt :: nil).
Proof. apply NoDup_cons; [ intros [] | apply NoDup_nil ]. Qed.

Instance Unit_FinIndex : FinIndex 1 unit :=
  finindex_from_leibniz (tt :: nil) unit_enum_NoDup
    (fun i ⇒ match i with tt ⇒ or_introl eq_refl end)
    eq_refl.

Instance Sum_EqDec {J1 J2 : Type} {d1 : EqDec J1} {d2 : EqDec J2} :
  EqDec (J1 + J2).
Proof.
  intros x y.
  destruct x as [a | a]; destruct y as [b | b].
  - destruct (dec (a = b)) as [Heq | Hne].
    + left. exact (f_equal inl Heq).
    + right. intros Hc. injection Hc. exact Hne.
  - right. discriminate.
  - right. discriminate.
  - destruct (dec (a = b)) as [Heq | Hne].
    + left. exact (f_equal inr Heq).
    + right. intros Hc. injection Hc. exact Hne.
Defined.

Section SumIndex.
  Context {J1 J2 : Type} {d1 : EqDec J1} {d2 : EqDec J2}.
  Context {n1 n2 : nat} (FI1 : FinIndex n1 J1 (eI := eq))
          (FI2 : FinIndex n2 J2 (eI := eq)).

  Definition sum_enum : list (J1 + J2) :=
    map inl (@FinIndex.enum n1 J1 eq _ FI1)
      ++ map inr (@FinIndex.enum n2 J2 eq _ FI2).

  Lemma sum_enum_NoDup : NoDup sum_enum.
  Proof.
    apply NoDup_app.
    - apply NoDup_map_injective.
      + intros x y Hxy. injection Hxy. exact (fun h ⇒ h).
      + exact (NoDup_of_NoDupA _ eq _ (@eq_Reflexive _) FinIndex.enum_NoDup).
    - apply NoDup_map_injective.
      + intros x y Hxy. injection Hxy. exact (fun h ⇒ h).
      + exact (NoDup_of_NoDupA _ eq _ (@eq_Reflexive _) FinIndex.enum_NoDup).
    - intros x Hx1 Hx2.
      apply in_map_iff in Hx1. destruct Hx1 as [a [Ha _]].
      apply in_map_iff in Hx2. destruct Hx2 as [b [Hb _]].
      subst. discriminate.
  Qed.

  Lemma sum_enum_complete : ∀ x : J1 + J2, In x sum_enum.
  Proof.
    intros [a | a]; unfold sum_enum; apply in_or_app.
    - left. apply in_map.
      exact (In_of_InA_discrete _ eq a _ (fun x y h ⇒ h)
               (@FinIndex.enum_complete n1 J1 eq _ FI1 a)).
    - right. apply in_map.
      exact (In_of_InA_discrete _ eq a _ (fun x y h ⇒ h)
               (@FinIndex.enum_complete n2 J2 eq _ FI2 a)).
  Qed.

  Lemma sum_enum_length : length sum_enum = (n1 + n2)%nat.
  Proof.
    unfold sum_enum.
    rewrite length_app, length_map, length_map.
    rewrite (@FinIndex.enum_length n1 J1 eq _ FI1),
      (@FinIndex.enum_length n2 J2 eq _ FI2).
    reflexivity.
  Qed.

End SumIndex.

Instance Sum_FinIndex {J1 J2 : Type} {d1 : EqDec J1} {d2 : EqDec J2}
    {n1 n2 : nat} {FI1 : FinIndex n1 J1 (eI := eq)}
    {FI2 : FinIndex n2 J2 (eI := eq)} :
  FinIndex (n1 + n2)%nat (J1 + J2) (eI := eq) :=
  finindex_from_leibniz (sum_enum FI1 FI2) (sum_enum_NoDup FI1 FI2)
    (sum_enum_complete FI1 FI2) (sum_enum_length FI1 FI2).

Section SumSum.
  Context {M : Type} `{MM : Monoid M}.
  Context {J1 J2 : Type} {d1 : EqDec J1} {d2 : EqDec J2}.
  Context {n1 n2 : nat} {FI1 : FinIndex n1 J1 (eI := eq)}
          {FI2 : FinIndex n2 J2 (eI := eq)}.

  #[export] Instance BigopSum_enum :
    BigopSum n1 n2 (n1 + n2) J1 J2 M (@bigop_list M _ _).
  Proof.
    intros f.
    calc (bigop_list f (@FinIndex.enum (n1 + n2) (J1 + J2) eq _ _))
      == (bigop_list f (map inl (@FinIndex.enum n1 J1 eq _ FI1))
          + bigop_list f (map inr (@FinIndex.enum n2 J2 eq _ FI2)))
           { (bigop_list_append f (map inl (@FinIndex.enum n1 J1 eq _ FI1))
                (map inr (@FinIndex.enum n2 J2 eq _ FI2))) }
      == (bigop_list (fun j ⇒ f (inl j)) (@FinIndex.enum n1 J1 eq _ FI1)
          + bigop_list f (map inr (@FinIndex.enum n2 J2 eq _ FI2)))
           { (bigop_list_map inl f (@FinIndex.enum n1 J1 eq _ FI1)) }
      == (bigop_list (fun j ⇒ f (inl j)) (@FinIndex.enum n1 J1 eq _ FI1)
          + bigop_list (fun j ⇒ f (inr j)) (@FinIndex.enum n2 J2 eq _ FI2))
           { (bigop_list_map inr f (@FinIndex.enum n2 J2 eq _ FI2)) }.
  Qed.

End SumSum.

Instance HasLincomb_enum {n : nat} {I F V : Type} {eI : EqRel I}
    {Ieq : EquDec I} {IFin : FinIndex n I} {Ve : EqRel V} {Vz : Zero V}
    {Va : HasAdd V} {Vs : HasScale F V} :
  HasLincomb I F V :=
  fun b c ⇒ bigopI (fun i ⇒ c i *: b i).

Instance Lincomb_enum {n : nat} {I F V : Type} {eI : EqRel I}
    {Ieq : EquDec I} {IFin : FinIndex n I} {Ve : EqRel V} {Vz : Zero V}
    {Va : HasAdd V} {VM : Monoid V} {Vs : HasScale F V} :
  Lincomb n I F V (@bigop_list V _ _) :=
  fun b c ⇒ reflexivity (bigopI (fun i ⇒ c i *: b i)).

Section BigOpFinMonoid.
  Context {M : Type} `{Monoid M}.

  Theorem bigopF_ext : ∀ n (f g : Fin.t n → M),
    (∀ i, f i == g i) → bigopF f == bigopF g.
  Proof.
    intros n f g Hfg.
    calc (bigopF f)
      == (bigopI f) { bigopF_bigopI }
      == (bigopI g) { bigopI_ext }
      == (bigopF g) { bigopF_bigopI }.
  Qed.

  Theorem bigopF_zero : ∀ n : nat,
    bigopF (fun _ : Fin.t n ⇒ (0 : M)) == 0.
  Proof.
    intros n.
    calc (bigopF (fun _ : Fin.t n ⇒ (0 : M)))
      == (bigopI (fun _ : Fin.t n ⇒ (0 : M)))
        { bigopF_bigopI }
      == (0 : M) { bigopI_zero }.
  Qed.

End BigOpFinMonoid.

Instance bigopF_proper {M : Type} `{Monoid M} (n : nat) :
  Proper (pointwise_relation (Fin.t n) equ ==> equ) (@bigopF M _ _ _ _ n).
Proof.
  intros f g Hfg. apply bigopF_ext. exact Hfg.
Qed.


Section BigOpFinCommMonoid.
  Context {M : Type} `{MM : Monoid M} {MC : Commutative M M add}.

  Theorem bigopF_extract : ∀ (m : nat) (p : Fin.t (S m)) (g : Fin.t (S m) → M),
    bigopF g == g p + bigopF (fun s : Fin.t m ⇒ g (skip p s)).
  Proof.
    intros m; induction m as [| m IH]; intros p g.
    - revert g.
      apply (Fin.caseS' p (fun p0 ⇒ ∀ g : Fin.t 1 → M,
        bigopF g == g p0 + bigopF (fun s : Fin.t 0 ⇒ g (skip p0 s)))).
      + intros g.
        rewrite (bigopF_S 0 g), (bigopF_0 (fun i : Fin.t 0 ⇒ g (Fin.FS i))),
          (bigopF_0 (fun s : Fin.t 0 ⇒ g (skip Fin.F1 s))).
        reflexivity.
      + intros p'.
        exact (Fin.case0 (fun p0 ⇒ ∀ g : Fin.t 1 → M,
          bigopF g == g (Fin.FS p0) + bigopF (fun s : Fin.t 0 ⇒ g (skip (Fin.FS p0) s)))
          p').
    - revert g.
      apply (Fin.caseS' p (fun p0 ⇒ ∀ g : Fin.t (S (S m)) → M,
        bigopF g == g p0 + bigopF (fun s : Fin.t (S m) ⇒ g (skip p0 s)))).
      + intros g.
        rewrite (bigopF_S (S m) g).
        apply op_proper.
        × reflexivity.
        × apply bigopF_ext. intros s.
          rewrite (skip_F1 (S m) s). reflexivity.
      + intros p' g.
        calc (bigopF g)
          == (g Fin.F1 + bigopF (fun s : Fin.t (S m) ⇒ g (Fin.FS s)))
               { bigopF_S }
          == (g Fin.F1 + (g (Fin.FS p')
                          + bigopF (fun s' : Fin.t m ⇒ g (Fin.FS (skip p' s')))))
               { IH }
          == (g (Fin.FS p') + (g Fin.F1
                               + bigopF (fun s' : Fin.t m ⇒ g (Fin.FS (skip p' s')))))
               { add_swap_front }
          == (g (Fin.FS p') + bigopF (fun s : Fin.t (S m) ⇒ g (skip (Fin.FS p') s)))
               { bigopF_S }.
  Qed.

  Theorem bigopF_extract2 : ∀ (n0 : nat) (p1 p2 : Fin.t (S (S n0)))
      (g : Fin.t (S (S n0)) → M),
    p1 ≠ p2 →
    (∀ c : Fin.t (S (S n0)), c ≠ p1 → c ≠ p2 → g c == 0) →
    bigopF g == g p1 + g p2.
  Proof.
    intros n0 p1 p2 g Hne Hzero.
    calc (bigopF g)
      == (bigopI g) { bigopF_bigopI }
      == (g p1 + g p2)
           { (bigopI_support_pair p1 p2 g (leibniz_proper g) Hne Hzero) }.
  Qed.

  Theorem bigopF_reindex : ∀ (n : nat) (g : Fin.t n → M)
      (tau tau' : Fin.t n → Fin.t n),
    (∀ x, tau' (tau x) = x) → (∀ y, tau (tau' y) = y) →
    bigopF g == bigopF (fun i ⇒ g (tau i)).
  Proof.
    intros n g tau tau' H1 H2.
    calc (bigopF g)
      == (bigopI g)
           { bigopF_bigopI }
      == (bigopI (fun i ⇒ g (tau i)))
           { (symmetry (bigopI_reindex g tau tau' (leibniz_proper g)
                (leibniz_proper tau) (leibniz_proper tau') H1 H2)) }
      == (bigopF (fun i ⇒ g (tau i))) { bigopF_bigopI }.
  Qed.

End BigOpFinCommMonoid.

Section BigOpFinPointwise.
  Context {X M : Type} `{Monoid M}.

  Theorem bigopF_app : ∀ (n : nat) (f : Fin.t n → X → M) (x : X),
    bigopF f x == bigopF (fun i ⇒ f i x).
  Proof.
    intros n f x.
    calc (bigopF f x)
      == (bigopI f x) { (bigopF_bigopI n f x) }
      == (bigopI (fun i ⇒ f i x)) { bigopI_app }
      == (bigopF (fun i ⇒ f i x)) { bigopF_bigopI }.
  Qed.

End BigOpFinPointwise.

Section BigOpFinAbGroup.
  Context {G : Type} `{AbGroup G}.

  Theorem bigopF_add : ∀ n (f g : Fin.t n → G),
    bigopF (fun i ⇒ f i + g i) == bigopF f + bigopF g.
  Proof.
    intros n f g.
    calc (bigopF (fun i ⇒ f i + g i))
      == (bigopI (fun i ⇒ f i + g i)) { bigopF_bigopI }
      == (bigopI f + bigopI g) { bigopI_add }
      == (bigopF f + bigopI g) { bigopF_bigopI }
      == (bigopF f + bigopF g) { bigopF_bigopI }.
  Qed.

  Theorem bigopF_neg : ∀ (n : nat) (f : Fin.t n → G),
    bigopF (fun i ⇒ - f i) == - bigopF f.
  Proof.
    intros n f.
    calc (bigopF (fun i ⇒ - f i))
      == (bigopI (fun i ⇒ - f i)) { bigopF_bigopI }
      == (- bigopI f) { bigopI_neg }
      == (- bigopF f) { bigopF_bigopI }.
  Qed.

  Theorem bigopF_swap : ∀ (m n : nat) (f : Fin.t m → Fin.t n → G),
    bigopF (fun i ⇒ bigopF (fun j ⇒ f i j))
      == bigopF (fun j ⇒ bigopF (fun i ⇒ f i j)).
  Proof.
    intros m n f.
    calc (bigopF (fun i ⇒ bigopF (fun j ⇒ f i j)))
      == (bigopI (fun i ⇒ bigopF (fun j ⇒ f i j)))
        { bigopF_bigopI }
      == (bigopI (fun i ⇒ bigopI (fun j ⇒ f i j)))
        { (bigopI_ext (fun i ⇒ bigopF (fun j ⇒ f i j))
            (fun i ⇒ bigopI (fun j ⇒ f i j))
            (fun i ⇒ bigopF_bigopI n (fun j ⇒ f i j))) }
      == (bigopI (fun j ⇒ bigopI (fun i ⇒ f i j)))
        { bigopI_swap }
      == (bigopI (fun j ⇒ bigopF (fun i ⇒ f i j)))
        { (bigopI_ext (fun j ⇒ bigopI (fun i ⇒ f i j))
            (fun j ⇒ bigopF (fun i ⇒ f i j))
            (fun j ⇒ symmetry (bigopF_bigopI m (fun i ⇒ f i j)))) }
      == (bigopF (fun j ⇒ bigopF (fun i ⇒ f i j)))
        { bigopF_bigopI }.
  Qed.

End BigOpFinAbGroup.

Section BigOpFinRing.
  Context {R : Type} {eR : EqRel R} {zR : Zero R} {aR : HasAdd R}
    {nR : HasNeg R} {oR : One R} {mR : HasMult R} {RR : Ring R}.

  Theorem bigopF_mul_l : ∀ n (a : R) (f : Fin.t n → R),
    bigopF (fun i ⇒ a × f i) == a × bigopF f.
  Proof.
    intros n a f.
    calc (bigopF (fun i ⇒ a × f i))
      == (bigopI (fun i ⇒ a × f i)) { bigopF_bigopI }
      == (a × bigopI f) { bigopI_mul_l }
      == (a × bigopF f) { bigopF_bigopI }.
  Qed.

  Theorem bigopF_mul_r : ∀ n (a : R) (f : Fin.t n → R),
    bigopF (fun i ⇒ f i × a) == bigopF f × a.
  Proof.
    intros n a f.
    calc (bigopF (fun i ⇒ f i × a))
      == (bigopI (fun i ⇒ f i × a)) { bigopF_bigopI }
      == (bigopI f × a) { bigopI_mul_r }
      == (bigopF f × a) { bigopF_bigopI }.
  Qed.

The product over a finite index: the same fold as bigopF, at the ring's multiplicative monoid instead of its additive one.
  Definition prodF {n : nat} (f : Fin.t n → R) : R :=
    @bigopF R eR oR mR mul_monoid n f.

  Lemma prodF_0 : ∀ f : Fin.t O → R, prodF f == 1.
  Proof. intros f. exact (@bigopF_0 R eR oR mR mul_monoid f). Qed.

  Lemma prodF_S : ∀ (m : nat) (f : Fin.t (S m) → R),
    prodF f == f Fin.F1 × prodF (fun i ⇒ f (Fin.FS i)).
  Proof. intros m f. exact (@bigopF_S R eR oR mR mul_monoid m f). Qed.

  Lemma prodF_ext : ∀ (n : nat) (f g : Fin.t n → R),
    (∀ i, f i == g i) → prodF f == prodF g.
  Proof.
    intros n f g Hfg. exact (@bigopF_ext R eR oR mR mul_monoid n f g Hfg).
  Qed.

End BigOpFinRing.

Section BigOpFinVectorSpace.
  Context {F V : Type} `{VectorSpace F V}.

  Theorem bigopF_scale : ∀ n (a : F) (f : Fin.t n → V),
    bigopF (fun i ⇒ a *: f i) == a *: bigopF f.
  Proof.
    intros n a f.
    calc (bigopF (fun i ⇒ a *: f i))
      == (bigopI (fun i ⇒ a *: f i)) { bigopF_bigopI }
      == (a *: bigopI f) { bigopI_scale }
      == (a *: bigopF f) { bigopF_bigopI }.
  Qed.

  Theorem bigopF_scale_r : ∀ (n : nat) (f : Fin.t n → F) (v : V),
    bigopF (fun i ⇒ f i *: v) == bigopF f *: v.
  Proof.
    intros n f v.
    calc (bigopF (fun i ⇒ f i *: v))
      == (bigopI (fun i ⇒ f i *: v)) { bigopF_bigopI }
      == (bigopI f *: v) { bigopI_scale_r }
      == (bigopF f *: v) { bigopF_bigopI }.
  Qed.

  Lemma lincomb_bigopF : ∀ {n : nat} (b : Fin.t n → V) (c : Fin.t n → F),
    lincomb b c == bigopF (fun i ⇒ c i *: b i).
  Proof.
    intros n b c.
    calc (lincomb b c)
      == (bigopI (fun i ⇒ c i *: b i)) { lincomb_def }
      == (bigopF (fun i ⇒ c i *: b i)) { bigopF_bigopI }.
  Qed.

End BigOpFinVectorSpace.

Section BigOpFinOrder.
  Context {F : Type} `{OrderedField F}.

  Lemma bigopF_nonneg : ∀ (n : nat) (f : Fin.t n → F),
    (∀ i, 0 ≤ f i) → 0 ≤ bigopF f.
  Proof.
    intros n f Hf.
    calc (0 : F)
      == (bigopI f) { bigopI_nonneg }
      == (bigopF f) { bigopF_bigopI }.
  Qed.

  Lemma bigopF_le : ∀ (n : nat) (f g : Fin.t n → F),
    (∀ i, f i ≤ g i) → bigopF f ≤ bigopF g.
  Proof.
    intros n f g Hfg.
    calc (bigopF f)
      == (bigopI f) { bigopF_bigopI }
      == (bigopI g) { bigopI_le }
      == (bigopF g) { bigopF_bigopI }.
  Qed.

  Lemma bigopF_nonneg_zero : ∀ (n : nat) (f : Fin.t n → F),
    (∀ i, 0 ≤ f i) → bigopF f == 0 → ∀ i, f i == 0.
  Proof.
    intros n f Hf Hsum i.
    exact (bigopI_nonneg_zero f (leibniz_proper f) Hf
             (transitivity (symmetry (bigopF_bigopI n f)) Hsum) i).
  Qed.

  Lemma bigopF_le_one_term : ∀ (n : nat) (f : Fin.t n → F)
      (j : Fin.t n), (∀ i, 0 ≤ f i) → f j ≤ bigopF f.
  Proof.
    intros n f j Hf.
    calc (f j)
      == (bigopI f) { (bigopI_le_one_term f j (leibniz_proper f) Hf) }
      == (bigopF f) { bigopF_bigopI }.
  Qed.

  Lemma abs_bigopF_le : ∀ (n : nat) (f : Fin.t n → F),
    abs (bigopF f) ≤ bigopF (fun i ⇒ abs (f i)).
  Proof.
    intros n f.
    calc (abs (bigopF f))
      == (abs (bigopI f)) { bigopF_bigopI }
      == (bigopI (fun i ⇒ abs (f i))) { abs_bigopI_le }
      == (bigopF (fun i ⇒ abs (f i))) { bigopF_bigopI }.
  Qed.

End BigOpFinOrder.

Definition delta {I R : Type} {eI : EqRel I} {Ieq : EquDec I}
    {Rz : Zero R} {Ro : One R} : I → I → R :=
  fun i j ⇒ if dec (i == j) then 1 else 0.

Instance Delta_dec {I R : Type} {eI : EqRel I} {Ieq : EquDec I}
    {Re : EqRel R} {Rz : Zero R} {Ro : One R}
    {RE : Reflexive (@equ R Re)} : Delta I R delta.
Proof.
  split; intros i j Hij; unfold delta;
    destruct (dec (i == j)) as [Heq | Hne].
  - reflexivity.
  - destruct (Hne Hij).
  - destruct (Hij Heq).
  - reflexivity.
Qed.

Example finindex_delta_check (F : Type) `{FF : Field F} (n : nat)
    (c : Fin.t n → F) (j : Fin.t n) :
  bigopI (fun i ⇒ c i × delta i j) == c j :=
  bigopI_delta c j (leibniz_proper c).

Definition upd_fun {I V : Type} {Ieq : EqDec I}
    (vs : I → V) (i : I) (x : V) : I → V :=
  fun j ⇒ if dec (i = j) then x else vs j.

Instance HasUpd_fun {I V : Type} {Ieq : EqDec I} : HasUpd I V := upd_fun.

Instance Upd_fun {I V : Type} {Ieq : EqDec I} {eV : EqRel V}
    {VR : Reflexive (@equ V eV)} : Upd I V (eI := eq).
Proof.
  split.
  - intros vs i x j Hji.
    unfold upd, HasUpd_fun, upd_fun.
    destruct (dec (i = j)) as [_ | Hne].
    + reflexivity.
    + destruct (Hne (eq_sym Hji)).
  - intros vs i j x Hij.
    unfold upd, HasUpd_fun, upd_fun.
    destruct (dec (i = j)) as [Heq | _].
    + destruct (Hij Heq).
    + reflexivity.
Qed.
