Section BoundedLinear.
  Context {F U V : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.

  Lemma bounded_const0 : linear_map_bounded (linear_map_const0 : LinearMap F U V).
  Proof.
    ∃ 0.
    split.
    - reflexivity.
    - intros u.
      apply (le_rewrite_l (‖ (0 : V) ‖)).
      + reflexivity.
      + rewrite norm_zero, (mul_0_l (‖ u ‖)).
        reflexivity.
  Qed.

  Lemma bounded_id : linear_map_bounded (linear_map_id (F := F) (U := U)).
  Proof.
    ∃ 1.
    split.
    - apply lt_le. exact lt_0_1.
    - intros u. apply eq_le. symmetry. exact (left_identity (‖ u ‖)).
  Qed.

  Lemma bounded_add : ∀ f g : LinearMap F U V,
    linear_map_bounded f → linear_map_bounded g → linear_map_bounded (f + g).
  Proof.
    intros f g [C1 [HC1 Hb1]] [C2 [HC2 Hb2]].
    ∃ (C1 + C2).
    split.
    - apply (le_rewrite_l ((0 : F) + 0)).
      + exact (left_identity _).
      + apply add_le_mono.
        × exact HC1.
        × exact HC2.
    - intros u.
      transitivity (‖ f u ‖ + ‖ g u ‖).
      + apply triangle.
      + transitivity (C1 × ‖ u ‖ + C2 × ‖ u ‖).
        × apply add_le_mono.
          -- exact (Hb1 u).
          -- exact (Hb2 u).
        × apply eq_le. symmetry. exact (right_distributive _ _ _).
  Qed.

  Lemma bounded_scale : ∀ (c : F) (f : LinearMap F U V),
    linear_map_bounded f → linear_map_bounded (c *: f).
  Proof.
    intros c f [C [HC Hb]].
    ∃ (abs c × C).
    split.
    - apply mul_nonneg_closed.
      + apply abs_nonneg.
      + exact HC.
    - intros u.
      apply (le_rewrite_l (abs c × ‖ f u ‖)).
      + symmetry. apply homogeneous[vnorm].
      + transitivity (abs c × (C × ‖ u ‖)).
        × apply mul_le_mono_l.
          -- apply abs_nonneg.
          -- exact (Hb u).
        × apply eq_le. symmetry. exact (associative _ _ _).
  Qed.

  Lemma bounded_neg : ∀ f : LinearMap F U V,
    linear_map_bounded f → linear_map_bounded (- f).
  Proof.
    intros f [C [HC Hb]].
    ∃ C.
    split.
    - exact HC.
    - intros u.
      apply (le_rewrite_l (‖ f u ‖)).
      + symmetry. exact (norm_neg (f u)).
      + exact (Hb u).
  Qed.

  Lemma lm_remainder_zero : ∀ (g : LinearMap F U V) (a h : U),
    ‖ g (a + h) + - (g a + g h) ‖ == 0.
  Proof.
    intros g a h.
    calc (‖ g (a + h) + - (g a + g h) ‖)
      == (‖ (g a + g h) + - (g a + g h) ‖) { linear_map_additive }
      == (0 : F) { norm_sub_self }.
  Qed.

End BoundedLinear.

Section BoundedLinearComp.
  Context {F U V W : Type} `{NormedSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VSV : VectorSpace F V} {nmV : HasNorm F V} {NSV : NormedSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VSW : VectorSpace F W} {nmW : HasNorm F W} {NSW : NormedSpace F W}.

  Lemma bounded_comp : ∀ (g : LinearMap F V W) (f : LinearMap F U V),
    linear_map_bounded g → linear_map_bounded f → linear_map_bounded (linear_map_comp g f).
  Proof.
    intros g f [Cg [HCg Hbg]] [Cf [HCf Hbf]].
    ∃ (Cg × Cf).
    split.
    - apply mul_nonneg_closed.
      + exact HCg.
      + exact HCf.
    - intros u.
      transitivity (Cg × ‖ f u ‖).
      + exact (Hbg (f u)).
      + transitivity (Cg × (Cf × ‖ u ‖)).
        × apply mul_le_mono_l.
          -- exact HCg.
          -- exact (Hbf u).
        × apply eq_le. symmetry. exact (associative _ _ _).
  Qed.

  Lemma comp_remainder_split : ∀ (g : V → W) (Lg : LinearMap F V W),
    Proper (equ ==> equ) g →
    ∀ b y x : V,
    ‖ g y + - (g b + Lg x) ‖ ≤
    ‖ g (b + (y + - b)) + - (g b + Lg (y + - b)) ‖ + ‖ Lg (y + - (b + x)) ‖.
  Proof.
    intros g Lg Pg b y x.
    assert (Hsplit :
      (g (b + (y + - b)) + - (g b + Lg (y + - b))) + Lg (y + - (b + x))
      == g y + - (g b + Lg x)).
    { calc ((g (b + (y + - b)) + - (g b + Lg (y + - b))) + Lg (y + - (b + x)))
        == ((g y + - (g b + Lg (y + - b))) + Lg (y + - (b + x)))
           { add_sub_cancel }
        == ((g y + - (g b + Lg (y + - b))) + Lg ((y + - b) + - x))
           { (linear_map_proper Lg (y + - (b + x)) ((y + - b) + - x)
                (neg_add_associative y b x)) }
        == ((g y + - (g b + Lg (y + - b))) + (Lg (y + - b) + Lg (- x)))
           { linear_map_additive }
        == ((g y + - (g b + Lg (y + - b))) + (Lg (y + - b) + - (Lg x)))
           { linear_map_neg }
        == (g y + - (g b + Lg x))
           { sub_mid_cancel }. }
    step == (‖ (g (b + (y + - b)) + - (g b + Lg (y + - b)))
               + Lg (y + - (b + x)) ‖)
      { Hsplit }.
    apply triangle.
  Qed.

End BoundedLinearComp.
