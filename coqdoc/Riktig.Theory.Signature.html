Induction on signatures, derived from the recursor: fold into the elements satisfying the property, and uniqueness -- applied to the projection and to the identity -- says the projection is the identity.
Section SignatureInduction.
  Context {K S : Type} `{Signature K S}.

  Theorem sig_induction : ∀ P : S → Prop,
    P 0 → P 1 → P self → (∀ k : K, P (const k)) →
    (∀ a b : S, P a → P b → P (a + b)) →
    (∀ a b : S, P a → P b → P (a × b)) →
    ∀ q : S, P q.
  Proof.
    intros P Hz Ho Hs Hc Hp Ht q.
    pose (r := poly_rec (S := S) { x : S | P x }
      (exist P 0 Hz) (exist P 1 Ho) (exist P self Hs)
      (fun k ⇒ exist P (const k) (Hc k))
      (fun a b ⇒ exist P (proj1_sig a + proj1_sig b)
                    (Hp (proj1_sig a) (proj1_sig b) (proj2_sig a) (proj2_sig b)))
      (fun a b ⇒ exist P (proj1_sig a × proj1_sig b)
                    (Ht (proj1_sig a) (proj1_sig b) (proj2_sig a) (proj2_sig b)))).
    assert (E1 : ∀ x : S,
        proj1_sig (r x) = poly_rec S 0 1 self const add mult x).
    { apply (Signature.rec_unique S 0 1 self const add mult
        (fun y ⇒ proj1_sig (r y))); cbv [r].
      - rewrite Signature.rec_zero. reflexivity.
      - rewrite Signature.rec_one. reflexivity.
      - rewrite Signature.rec_self. reflexivity.
      - intros k. rewrite Signature.rec_const. reflexivity.
      - intros a b. rewrite Signature.rec_plus. reflexivity.
      - intros a b. rewrite Signature.rec_times. reflexivity. }
    assert (E2 : ∀ x : S, x = poly_rec S 0 1 self const add mult x).
    { apply (Signature.rec_unique S 0 1 self const add mult (fun y ⇒ y));
        intros; reflexivity. }
    assert (E : proj1_sig (r q) = q). { rewrite (E1 q). exact (eq_sym (E2 q)). }
    rewrite <- E. exact (proj2_sig (r q)).
  Qed.
End SignatureInduction.

The layer of a signature is a functor, by induction on the signature: equals to equals, identities to identities, composites to composites.
Section LayerFunctor.
  Context {K S : Type} `{Signature K S}.
  Context {O : Type} `{Bicartesian O} (den : K → O).

  Lemma layer_map_proper : ∀ (q : S) (x y : O) (f g : x ~> y),
    f == g → lmap (layer den q) f == lmap (layer den q) g.
  Proof.
    apply (sig_induction (fun q ⇒ ∀ (x y : O) (f g : x ~> y),
      f == g → lmap (layer den q) f == lmap (layer den q) g));
      cbv [layer].
    - intros x y f g Hfg. rewrite Signature.rec_zero. reflexivity.
    - intros x y f g Hfg. rewrite Signature.rec_one. reflexivity.
    - intros x y f g Hfg. rewrite Signature.rec_self. assumption.
    - intros k x y f g Hfg. rewrite Signature.rec_const. reflexivity.
    - intros a b IHa IHb x y f g Hfg. rewrite Signature.rec_plus. cbn [lmap].
      rewrite (IHa _ _ _ _ Hfg), (IHb _ _ _ _ Hfg). reflexivity.
    - intros a b IHa IHb x y f g Hfg. rewrite Signature.rec_times. cbn [lmap].
      rewrite (IHa _ _ _ _ Hfg), (IHb _ _ _ _ Hfg). reflexivity.
  Qed.

  Lemma layer_map_id : ∀ (q : S) (x : O),
    lmap (layer den q) (cat_id (x := x)) == cat_id.
  Proof.
    apply (sig_induction (fun q ⇒ ∀ x : O,
      lmap (layer den q) (cat_id (x := x)) == cat_id)); cbv [layer].
    - intros x. rewrite Signature.rec_zero. reflexivity.
    - intros x. rewrite Signature.rec_one. reflexivity.
    - intros x. rewrite Signature.rec_self. reflexivity.
    - intros k x. rewrite Signature.rec_const. reflexivity.
    - intros a b IHa IHb x. rewrite Signature.rec_plus. cbn [lobj lmap].
      rewrite IHa, IHb. apply sum_id.
    - intros a b IHa IHb x. rewrite Signature.rec_times. cbn [lobj lmap].
      rewrite IHa, IHb. apply par_id.
  Qed.

  Lemma layer_map_comp : ∀ (q : S) (x y z : O) (f : y ~> z) (g : x ~> y),
    lmap (layer den q) (f ∘ g) == lmap (layer den q) f ∘ lmap (layer den q) g.
  Proof.
    apply (sig_induction (fun q ⇒ ∀ (x y z : O) (f : y ~> z) (g : x ~> y),
      lmap (layer den q) (f ∘ g) == lmap (layer den q) f ∘ lmap (layer den q) g));
      cbv [layer].
    - intros x y z f g. rewrite Signature.rec_zero. cbn [lmap].
      symmetry. apply left_identity.
    - intros x y z f g. rewrite Signature.rec_one. cbn [lmap].
      symmetry. apply left_identity.
    - intros x y z f g. rewrite Signature.rec_self. reflexivity.
    - intros k x y z f g. rewrite Signature.rec_const. cbn [lmap].
      symmetry. apply left_identity.
    - intros a b IHa IHb x y z f g. rewrite Signature.rec_plus. cbn [lobj lmap].
      rewrite IHa, IHb. rewrite (sum_comp (O := O)). reflexivity.
    - intros a b IHa IHb x y z f g. rewrite Signature.rec_times. cbn [lobj lmap].
      rewrite IHa, IHb. rewrite (par_comp (O := O)). reflexivity.
  Qed.

  #[export] Instance layer_functor (q : S) :
      Functor O O (lobj (layer den q)) (layer_fmap den q) :=
    {| Functor.fmap_proper := fun x y f g Hfg ⇒ layer_map_proper q x y f g Hfg;
       Functor.fmap_id := fun x ⇒ layer_map_id q x;
       Functor.fmap_comp := fun x y z f g ⇒ layer_map_comp q x y z f g |}.
End LayerFunctor.

Initiality yields induction for every datatype of every signature at once. The first form takes the algebra on the elements satisfying the property; the second asks only that the property propagate up one level, where holding below s means s lifts through the elements that satisfy it.
Section DatatypeInduction.
  Context {K S : Type} `{Signature K S} (den : K → Type).
  Context {T : Type} (q : S)
    {tb : HasBuild Type (lobj (layer den q)) T}
    {tf : HasCata Type (lobj (layer den q)) T}
    {IA : InitialAlgebra Type (lobj (layer den q)) T tb tf}.

  Theorem datatype_induction_alg : ∀ (P : T → Prop)
      (h : lobj (layer den q) {u : T | P u} → {u : T | P u}),
    (∀ s, proj1_sig (h s) = tb (lmap (layer den q) (@proj1_sig T P) s)) →
    ∀ t : T, P t.
  Proof.
    intros P h Hh t.
    pose (u := fun t : T ⇒ proj1_sig (tf {u : T | P u} h t)).
    assert (Hu : u ∘ tb == tb ∘ lmap (layer den q) u).
    { intros s. cbv [u].
      transitivity (proj1_sig (h (lmap (layer den q) (tf {u : T | P u} h) s))).
      { exact (f_equal (@proj1_sig T P)
          (InitialAlgebra.cata_build (t := T) {u : T | P u} h s)). }
      rewrite Hh.
      exact (f_equal tb (eq_sym (Functor.fmap_comp (F := lobj (layer den q))
        _ _ _ (@proj1_sig T P) (tf {u : T | P u} h) s))). }
    assert (E : u t = t). { exact (alg_endo_id (t := T) u Hu t). }
    rewrite <- E. exact (proj2_sig (tf {u : T | P u} h t)).
  Qed.

  Definition holds_below (P : T → Prop) (s : lobj (layer den q) T) : Prop :=
    ∃ s' : lobj (layer den q) {u : T | P u},
      lmap (layer den q) (@proj1_sig T P) s' = s.

  Theorem datatype_induction : ∀ P : T → Prop,
    (∀ s, holds_below P s → P (tb s)) → ∀ t : T, P t.
  Proof.
    intros P Hstep.
    apply (datatype_induction_alg P
      (fun s' : lobj (layer den q) {u : T | P u} ⇒
         exist P (tb (lmap (hom := Fun_HasHom) (layer den q) (@proj1_sig T P) s'))
           (Hstep _ (ex_intro _ s' eq_refl)))).
    intros s'. reflexivity.
  Qed.
End DatatypeInduction.
