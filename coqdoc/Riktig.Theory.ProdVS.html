Instance Prod_EqRel {A B : Type} `{EqRel A} `{EqRel B} :
  EqRel (A × B) :=
  fun p q ⇒ fst p == fst q ∧ snd p == snd q.

Instance Prod_Zero {A B : Type} `{Zero A} `{Zero B} :
  Zero (A × B) := (0, 0).

Instance Prod_Add {A B : Type} `{HasAdd A} `{HasAdd B} :
  HasAdd (A × B) :=
  fun p q ⇒ (fst p + fst q, snd p + snd q).

Instance Prod_Neg {A B : Type} `{HasNeg A} `{HasNeg B} :
  HasNeg (A × B) :=
  fun p ⇒ (- fst p, - snd p).

Instance Prod_Scale {F A B : Type} `{HasScale F A} `{HasScale F B} :
  HasScale F (A × B) :=
  fun a p ⇒ (a *: fst p, a *: snd p).

Instance Prod_Equivalence {A B : Type} {eA : EqRel A} {eB : EqRel B}
    {EA : Equivalence eA} {EB : Equivalence eB} :
  Equivalence (@Prod_EqRel A B eA eB).
Proof.
  apply Build_Equivalence.
  - intros p. split; reflexivity.
  - intros p q [H1 H2]. split.
    + symmetry. exact H1.
    + symmetry. exact H2.
  - intros p q r [H1 H2] [H3 H4]. split.
    + rewrite H1. exact H3.
    + rewrite H2. exact H4.
Qed.

Instance Prod_EquDec {A B : Type} {eA : EqRel A} {eB : EqRel B}
    {dA : EquDec A (e := eA)} {dB : EquDec B (e := eB)} :
  EquDec (A × B) :=
  fun p q ⇒ and_dec (fst p == fst q) (snd p == snd q).

Section ProdMonoid.
  Context {A B : Type} `{Monoid A}.
  Context `{Monoid B}.

  Lemma prod_associative : ∀ p q r : A × B, (p + q) + r == p + (q + r).
  Proof. intros [a1 b1] [a2 b2] [a3 b3]. split; [ exact (associative a1 a2 a3) | exact (associative b1 b2 b3) ]. Qed.

  Lemma prod_left_identity : ∀ p : A × B, 0 + p == p.
  Proof. intros [a b]. split; [ exact (left_identity a) | exact (left_identity b) ]. Qed.

  Lemma prod_right_identity : ∀ p : A × B, p + 0 == p.
  Proof. intros [a b]. split; [ exact (right_identity a) | exact (right_identity b) ]. Qed.

  #[export] Instance prod_add_proper : Proper (equ ==> equ ==> equ) (@add (A × B) _).
  Proof.
    intros p p' [Hp1 Hp2] q q' [Hq1 Hq2].
    split.
    - exact (op_proper _ _ Hp1 _ _ Hq1).
    - exact (op_proper _ _ Hp2 _ _ Hq2).
  Qed.

  #[export] Instance Prod_Monoid : Monoid (A × B) :=
    {| Monoid.semigroup :=
         {| Semigroup.magma :=
              {| Magma.equivalence := Prod_Equivalence;
                 Magma.op_proper := prod_add_proper |};
            Semigroup.is_associative := prod_associative |};
       Monoid.is_left_identity := prod_left_identity;
       Monoid.is_right_identity := prod_right_identity |}.

End ProdMonoid.

Section ProdVectorSpace.
  Context {F A B : Type} `{VectorSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B}
    {sB : HasScale F B} {VS2 : VectorSpace F B}.

  #[export] Instance prod_scale_proper : Proper (equ ==> equ ==> equ) (@scale F (A × B) _).
  Proof.
    intros a a' Ha p p' [H1 H2].
    split.
    - exact (scale_proper a a' Ha (fst p) (fst p') H1).
    - exact (scale_proper a a' Ha (snd p) (snd p') H2).
  Qed.

  Lemma prod_scale_add : ∀ (a : F) (u v : A × B),
    a *: (u + v) == a *: u + a *: v.
  Proof. intros a [u1 u2] [v1 v2]. split; exact (left_distributive _ _ _). Qed.

  Lemma prod_add_scale : ∀ (a b : F) (u : A × B),
    (a + b) *: u == a *: u + b *: u.
  Proof. intros a b [u1 u2]. split; exact (right_distributive _ _ _). Qed.

  Lemma prod_scale_scale : ∀ (a b : F) (u : A × B),
    (a × b) *: u == a *: (b *: u).
  Proof. intros a b [u1 u2]. split; exact (associative _ _ _). Qed.

  Lemma prod_scale_1 : ∀ u : A × B, 1 *: u == u.
  Proof. intros [u1 u2]. split; exact (left_identity _). Qed.

  #[export] Instance Prod_ScaleLaws : ScaleLaws F (A × B) :=
    {| ScaleLaws.scale_proper := prod_scale_proper;
       ScaleLaws.is_left_distributive := prod_scale_add;
       ScaleLaws.is_right_distributive := prod_add_scale;
       ScaleLaws.is_associative := prod_scale_scale;
       ScaleLaws.is_left_identity := prod_scale_1 |}.

  #[export] Instance Prod_VectorSpace : VectorSpace F (A × B) := {}.

End ProdVectorSpace.

Section ProdNormedSpace.
  Context {F A B : Type} `{NormedSpace F A}.
  Context {eB : EqRel B} {zB : Zero B} {aB : HasAdd B} {nB : HasNeg B} {sB : HasScale F B}.
  Context {VB : VectorSpace F B} {nmB : HasNorm F B} {NB : NormedSpace F B}.

  Definition prod_norm (p : A × B) : F := ‖ fst p ‖ + ‖ snd p ‖.

  #[export] Instance Prod_HasNorm : HasNorm F (A × B) := prod_norm.

  #[export] Instance prod_norm_proper : Proper (equ ==> equ) (@vnorm F (A × B) Prod_HasNorm).
  Proof.
    intros [u1 v1] [u2 v2] [Hu Hv].
    unfold vnorm, Prod_HasNorm, prod_norm; simpl.
    rewrite Hu, Hv.
    reflexivity.
  Qed.

  Lemma prod_norm_nonneg : ∀ p : A × B, 0 ≤ ‖ p ‖.
  Proof.
    intros [u v].
    unfold vnorm, Prod_HasNorm, prod_norm; simpl.
    apply (le_rewrite_l ((0 : F) + 0)).
    - exact (left_identity _).
    - apply add_le_mono; apply nonneg.
  Qed.

  Lemma prod_norm_def : ∀ p : A × B, ‖ p ‖ == 0 → p == 0.
  Proof.
    intros [u v] Heq.
    unfold vnorm, Prod_HasNorm, prod_norm in Heq; simpl in Heq.
    destruct (nonneg_add_eq_zero (‖ u ‖) (‖ v ‖) (nonneg u)
      (nonneg v) Heq) as [Hu Hv].
    split; simpl.
    - apply definite. exact Hu.
    - apply definite. exact Hv.
  Qed.

  Lemma prod_norm_scale : ∀ (a : F) (p : A × B), ‖ a *: p ‖ == abs a × ‖ p ‖.
  Proof.
    intros a [u v].
    unfold vnorm, Prod_HasNorm, prod_norm; simpl.
    calc (‖ a *: u ‖ + ‖ a *: v ‖)
      == (abs a × ‖ u ‖ + ‖ a *: v ‖) { homogeneous[vnorm] }
      == (abs a × ‖ u ‖ + abs a × ‖ v ‖) { homogeneous[vnorm] }
      == (abs a × (‖ u ‖ + ‖ v ‖)) { left_distributive }.
  Qed.

  Lemma prod_norm_triangle : ∀ p q : A × B, ‖ p + q ‖ ≤ ‖ p ‖ + ‖ q ‖.
  Proof.
    intros [u1 v1] [u2 v2].
    unfold vnorm, Prod_HasNorm, prod_norm; simpl.
    transitivity ((‖ u1 ‖ + ‖ u2 ‖) + (‖ v1 ‖ + ‖ v2 ‖)).
    - apply add_le_mono; apply triangle.
    - apply eq_le. apply add_interchange.
  Qed.

  #[export] Instance Prod_NormedSpace : NormedSpace F (A × B) := {|
    NormedSpace.vnorm_proper := prod_norm_proper;
    NormedSpace.nonneg := prod_norm_nonneg;
    NormedSpace.definite := prod_norm_def;
    NormedSpace.is_homogeneous := prod_norm_scale;
    NormedSpace.triangle := prod_norm_triangle
  |}.

End ProdNormedSpace.
