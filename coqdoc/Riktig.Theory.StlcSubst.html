From Riktig.Programs Require Export StlcSubst.
From Riktig.Structure Require Import NatRec.

What the typed-term programs compute, one equation per constructor: each is the recursion's own law read at Coq's equality, so the equations hold at any implementation.
Section Equations.
  Context {N Ty T : Type} `{TypedTermRec N Ty T} `{NatRec N}.

  Theorem stermrec_var : ∀ (B : Type) (v : N → B) (l : Ty → B → B) (a : B → B → B) (n : N),
    stermrec B v l a (var n : T) = v n.
  Proof. intros B v l a n. exact (TypedTermRec.stermrec_var B eq eq_equivalence v l a n). Qed.

  Theorem stermrec_lam : ∀ (B : Type) (v : N → B) (l : Ty → B → B) (a : B → B → B) (A : Ty) (b : T),
    stermrec B v l a (tylam A b) = l A (stermrec B v l a b).
  Proof. intros B v l a A b. exact (TypedTermRec.stermrec_tylam B eq eq_equivalence v l a A b). Qed.

  Theorem stermrec_app : ∀ (B : Type) (v : N → B) (l : Ty → B → B) (a : B → B → B) (f u : T),
    stermrec B v l a (app f u) = a (stermrec B v l a f) (stermrec B v l a u).
  Proof. intros B v l a f u. exact (TypedTermRec.stermrec_app B eq eq_equivalence v l a f u). Qed.

The equations of renaming and of substitution.
  Theorem srename_var : ∀ (n : N) (r : N → N),
    srename (var n : T) r = var (r n).
  Proof. intros n r. exact (f_equal (fun g ⇒ g r) (stermrec_var _ _ _ _ n)). Qed.

  Theorem srename_lam : ∀ (A : Ty) (b : T) (r : N → N),
    srename (tylam A b) r = tylam A (srename b (sup_ren r)).
  Proof. intros A b r. exact (f_equal (fun g ⇒ g r) (stermrec_lam _ _ _ _ A b)). Qed.

  Theorem srename_app : ∀ (f u : T) (r : N → N),
    srename (app f u) r = app (srename f r) (srename u r).
  Proof. intros f u r. exact (f_equal (fun g ⇒ g r) (stermrec_app _ _ _ _ f u)). Qed.

  Theorem ssubst_var : ∀ (n : N) (s : N → T),
    ssubst (var n : T) s = s n.
  Proof. intros n s. exact (f_equal (fun g ⇒ g s) (stermrec_var _ _ _ _ n)). Qed.

  Theorem ssubst_lam : ∀ (A : Ty) (b : T) (s : N → T),
    ssubst (tylam A b) s = tylam A (ssubst b (sup_subst s)).
  Proof. intros A b s. exact (f_equal (fun g ⇒ g s) (stermrec_lam _ _ _ _ A b)). Qed.

  Theorem ssubst_app : ∀ (f u : T) (s : N → T),
    ssubst (app f u) s = app (ssubst f s) (ssubst u s).
  Proof. intros f u s. exact (f_equal (fun g ⇒ g s) (stermrec_app _ _ _ _ f u)). Qed.
End Equations.
