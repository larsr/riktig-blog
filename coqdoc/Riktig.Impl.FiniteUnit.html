From Riktig Require Export Structure.Finite Theory.NatOrder.

Coq's unit is a finite index of count one, at any naturals.
Instance Unit_EqRel : EqRel unit := eq.

Section FiniteUnit.
  Context {N : Type} `{NatOrder N}.

  #[export] Instance Unit_Finite : Finite unit N 1 (fun _ ⇒ 0) (fun _ ⇒ tt).
  Proof.
    split.
    - exact eq_equivalence.
    - intros _ _ _. reflexivity.
    - intros _ _ _. reflexivity.
    - intros _. exact zero_lt_one.
    - intros []. reflexivity.
    - intros k Hk. symmetry. exact (eq_zero_of_lt_one k Hk).
  Qed.
End FiniteUnit.
