From Riktig.Structure Require Export GeometricAlgebra.

A grade involution on a geometric algebra is an automorphism of the algebra that is its own inverse and negates every vector.
Module GradeInvolution.
  Class GradeInvolution (F V A : Type) `{GeometricAlgebra F V A} (involution : HasInvolute A) := {
    involute_proper :: Proper (equ ==> equ) (@involute A involution);
    involute_additive :: Additive A A add add involute;
    involute_multiplicative :: Additive A A mult mult involute;
    involute_homogeneous :: Homogeneous F A A scale (fun a ⇒ a) involute;
    involute_one : involute (1 : A) == 1;
    involute_involute : ∀ x : A, involute (involute x) == x;
    involute_vec : ∀ u : V, involute (vec u) == - vec u;
  }.
  Hint Mode GradeInvolution - - ! - - - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End GradeInvolution.
Export GradeInvolution.
