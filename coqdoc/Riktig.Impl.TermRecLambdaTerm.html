From Riktig.Structure Require Export TermRec.
From Riktig.Theory Require Import InitialAlgebra LambdaTerm.
From Riktig.Impl Require Export CategoryFun SignaturePoly LambdaTermOps.

Any lambda-term datatype at Coq's types recurses: the recursion is the fold, and the equations are the fold's own, read at one level.
Section S.
  Context {N T : Type} {eqN : EqRel N}
    {tb : HasBuild Type (LambdaTerm.shape (fun T0 : Type ⇒ T0) N) T}
    {tf : HasCata Type (LambdaTerm.shape (fun T0 : Type ⇒ T0) N) T}
    `{!LambdaTerm Type (fun T0 : Type ⇒ T0) N T tb tf}.

  #[export] Instance LambdaTerm_HasTermRec : HasTermRec N T :=
    fun B v l a ⇒ tf B (fun s ⇒ match s with
      | inl (inl n) ⇒ v n
      | inl (inr b) ⇒ l b
      | inr (f, x) ⇒ a f x
      end).

  #[export] Instance LambdaTerm_TermRec :
      TermRec N T (eqT := eq)
        LambdaTerm_HasVar LambdaTerm_HasLam LambdaTerm_HasApp LambdaTerm_HasTermRec.
  Proof.
    split.
    - intros B eB EB v l a _ _ _ t t' Ht. rewrite Ht. reflexivity.
    - intros B eB EB v l a n.
      assert (E : termrec B v l a (var n : T) = v n).
      { exact (InitialAlgebra.cata_build (t := T) B _ (inl (inl n))). }
      rewrite E. reflexivity.
    - intros B eB EB v l a b.
      assert (E : termrec B v l a (lam b : T) = l (termrec B v l a b)).
      { exact (InitialAlgebra.cata_build (t := T) B _ (inl (inr b))). }
      rewrite E. reflexivity.
    - intros B eB EB v l a f u.
      assert (E : termrec B v l a (app f u : T) = a (termrec B v l a f) (termrec B v l a u)).
      { exact (InitialAlgebra.cata_build (t := T) B _ (inr (f, u))). }
      rewrite E. reflexivity.
  Qed.
End S.
