From Riktig.Structure Require Export Category.

Binary products: an object with two projections through which every pair of arrows from a common source factors uniquely.
Module Products.
  Class Products (O : Type) `{Category O} (product : HasProd O)
      (first : HasPi1 O) (second : HasPi2 O) (lifting : HasLift O) := {
    lift_proper :: ∀ x a b : O, Proper (equ ==> equ ==> equ) (@lift O _ _ lifting x a b);
    pi1_lift : ∀ (x a b : O) (f : x ~> a) (g : x ~> b), pi1 ∘ lift f g == f;
    pi2_lift : ∀ (x a b : O) (f : x ~> a) (g : x ~> b), pi2 ∘ lift f g == g;
    lift_unique : ∀ (x a b : O) (f : x ~> a) (g : x ~> b) (h : x ~> a ⨯ b),
      pi1 ∘ h == f → pi2 ∘ h == g → h == lift f g;
  }.
  Hint Mode Products ! - - - - - - - - - : typeclass_instances.
End Products.
