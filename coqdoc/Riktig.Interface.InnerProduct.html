From Coq Require Export Morphisms.
From Riktig Require Export Ops Interface.VectorSpace.

Module InnerProduct.
  Class InnerProduct (F V : Type) `{VectorSpace.VectorSpace F V}
      {Vi : HasInner F V} := {
    proper : ∀ u v : V, u == v →
                    ∀ x y : V, x == y → ⟨ u , x ⟩ == ⟨ v , y ⟩;
    symmetric : ∀ u v : V, ⟨ u , v ⟩ == ⟨ v , u ⟩;
    left_additive : ∀ u v w : V, ⟨ u + v , w ⟩ == ⟨ u , w ⟩ + ⟨ v , w ⟩;
    left_scale : ∀ (a : F) (u v : V), ⟨ a *: u , v ⟩ == a × ⟨ u , v ⟩
  }.
End InnerProduct.

Definition norm2 {F V : Type} `{InnerProduct.InnerProduct F V} (u : V) : F :=
  ⟨ u , u ⟩.

Notation InnerProduct := InnerProduct.InnerProduct.

#[global] Instance InnerProduct_inner_Proper (F V : Type)
  `{VectorSpace.VectorSpace F V} {Vi : HasInner F V}
  (IP : InnerProduct.InnerProduct F V)
  : Proper (equ ==> equ ==> equ) (@inner F V _) := InnerProduct.proper.
