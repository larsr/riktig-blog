  Theorem involute_zero : involute (0 : A) == 0.
  Proof. exact (additive_zero involute). Qed.

  Theorem involute_neg : ∀ x : A, involute (- x) == - involute x.
  Proof. intros x. exact (additive_neg involute x). Qed.

  Theorem involute_smul_one : ∀ t : F, involute (t *: (1 : A)) == t *: 1.
  Proof.
    intros t.
    calc (involute (t *: (1 : A)))
      == (t *: involute (1 : A)) { (homogeneous (f := involute) t (1 : A)) }
      == (t *: (1 : A)) { involute_one }.
  Qed.

The two ways of associating a product of three elements a + b e, c + d e, e' + f e, where e squares to the scalar s, agree in each component: both are the same four products, summed.
  Lemma double_assoc_l_expand : ∀ (s : F) (a b c d e f : A),
    (a × c + s *: (b × involute d)) × e + s *: ((a × d + b × involute c) × involute f)
      == ((a × c) × e + (s *: (b × involute d)) × e) + (s *: ((a × d) × involute f) + s *: ((b × involute c) × involute f)).
  Proof.
    intros s a b c d e f.
    calc ((a × c + s *: (b × involute d)) × e + s *: ((a × d + b × involute c) × involute f))
      == (((a × c) × e + (s *: (b × involute d)) × e) + s *: ((a × d + b × involute c) × involute f))
           { (right_distributive (op := mult) (a × c) (s *: (b × involute d)) e) }
      == (((a × c) × e + (s *: (b × involute d)) × e) + s *: ((a × d) × involute f + (b × involute c) × involute f))
           { (right_distributive (op := mult) (a × d) (b × involute c) (involute f)) }
      == (((a × c) × e + (s *: (b × involute d)) × e) + (s *: ((a × d) × involute f) + s *: ((b × involute c) × involute f)))
           { (left_distributive (op := scale) s ((a × d) × involute f) ((b × involute c) × involute f)) }.
  Qed.

  Lemma double_assoc_l : ∀ (s : F) (a b c d e f : A),
    (a × c + s *: (b × involute d)) × e + s *: ((a × d + b × involute c) × involute f)
      == (a × (c × e) + s *: (b × (involute d × e))) + (s *: (a × (d × involute f)) + s *: (b × (involute c × involute f))).
  Proof.
    intros s a b c d e f.
    calc ((a × c + s *: (b × involute d)) × e + s *: ((a × d + b × involute c) × involute f))
      == (((a × c) × e + (s *: (b × involute d)) × e) + (s *: ((a × d) × involute f) + s *: ((b × involute c) × involute f)))
           { double_assoc_l_expand }
      == ((a × (c × e) + (s *: (b × involute d)) × e) + (s *: ((a × d) × involute f) + s *: ((b × involute c) × involute f)))
           { (associative (op := mult) (action := mult) a c e) }
      == ((a × (c × e) + s *: ((b × involute d) × e)) + (s *: ((a × d) × involute f) + s *: ((b × involute c) × involute f)))
           { (smul_mul_assoc s (b × involute d) e) }
      == ((a × (c × e) + s *: (b × (involute d × e))) + (s *: ((a × d) × involute f) + s *: ((b × involute c) × involute f)))
           { (associative (op := mult) (action := mult) b (involute d) e) }
      == ((a × (c × e) + s *: (b × (involute d × e))) + (s *: (a × (d × involute f)) + s *: ((b × involute c) × involute f)))
           { (associative (op := mult) (action := mult) a d (involute f)) }
      == ((a × (c × e) + s *: (b × (involute d × e))) + (s *: (a × (d × involute f)) + s *: (b × (involute c × involute f))))
           { (associative (op := mult) (action := mult) b (involute c) (involute f)) }.
  Qed.

  Lemma double_assoc_r_involute : ∀ (s : F) (b c d e f : A),
    s *: (b × involute (c × f + d × involute e))
      == s *: (b × (involute d × e)) + s *: (b × (involute c × involute f)).
  Proof.
    intros s b c d e f.
    calc (s *: (b × involute (c × f + d × involute e)))
      == (s *: (b × (involute (c × f) + involute (d × involute e))))
           { (additive (f := involute) (c × f) (d × involute e)) }
      == (s *: (b × (involute c × involute f + involute (d × involute e))))
           { (additive (f := involute) (opA := mult) c f) }
      == (s *: (b × (involute c × involute f + involute d × involute (involute e))))
           { (additive (f := involute) (opA := mult) d (involute e)) }
      == (s *: (b × (involute c × involute f + involute d × e)))
           { involute_involute }
      == (s *: (b × (involute c × involute f) + b × (involute d × e)))
           { (left_distributive (op := mult) b (involute c × involute f) (involute d × e)) }
      == (s *: (b × (involute c × involute f)) + s *: (b × (involute d × e)))
           { (left_distributive (op := scale) s (b × (involute c × involute f)) (b × (involute d × e))) }
      == (s *: (b × (involute d × e)) + s *: (b × (involute c × involute f)))
           { (commutative (op := add) (s *: (b × (involute c × involute f))) (s *: (b × (involute d × e)))) }.
  Qed.

  Lemma double_assoc_r : ∀ (s : F) (a b c d e f : A),
    a × (c × e + s *: (d × involute f)) + s *: (b × involute (c × f + d × involute e))
      == (a × (c × e) + s *: (b × (involute d × e))) + (s *: (a × (d × involute f)) + s *: (b × (involute c × involute f))).
  Proof.
    intros s a b c d e f.
    calc (a × (c × e + s *: (d × involute f)) + s *: (b × involute (c × f + d × involute e)))
      == ((a × (c × e) + a × (s *: (d × involute f))) + s *: (b × involute (c × f + d × involute e)))
           { (left_distributive (op := mult) a (c × e) (s *: (d × involute f))) }
      == ((a × (c × e) + s *: (a × (d × involute f))) + s *: (b × involute (c × f + d × involute e)))
           { (mul_smul_comm s a (d × involute f)) }
      == ((a × (c × e) + s *: (a × (d × involute f))) + (s *: (b × (involute d × e)) + s *: (b × (involute c × involute f))))
           { double_assoc_r_involute }
      == ((a × (c × e) + s *: (b × (involute d × e))) + (s *: (a × (d × involute f)) + s *: (b × (involute c × involute f))))
           { (add_add_add_comm (a × (c × e)) (s *: (a × (d × involute f))) (s *: (b × (involute d × e))) (s *: (b × (involute c × involute f)))) }.
  Qed.

  Lemma double_assoc_snd_l : ∀ (s : F) (a b c d e f : A),
    (a × c + s *: (b × involute d)) × f + (a × d + b × involute c) × involute e
      == (a × (c × f) + s *: (b × (involute d × f))) + (a × (d × involute e) + b × (involute c × involute e)).

    calc ((a × c + s *: (b × involute d)) × f + (a × d + b × involute c) × involute e)

           { (right_distributive (op := mult) (a × c) (s *: (b × involute d)) f) }
      == (((a × c) × f + (s *: (b × involute d)) × f) + ((a × d) × involute e + (b × involute c) × involute e))
           { (right_distributive (op := mult) (a × d) (b × involute c) (involute e)) }
      == ((a × (c × f) + (s *: (b × involute d)) × f) + ((a × d) × involute e + (b × involute c) × involute e))

           { (smul_mul_assoc s (b × involute d) f) }

           { (associative (op := mult) (action := mult) b (involute d) f) }
      == ((a × (c × f) + s *: (b × (involute d × f))) + (a × (d × involute e) + (b × involute c) × involute e))
           { (associative (op := mult) (action := mult) a d (involute e)) }
      == ((a × (c × f) + s *: (b × (involute d × f))) + (a × (d × involute e) + b × (involute c × involute e)))
           { (associative (op := mult) (action := mult) b (involute c) (involute e)) }.
  Qed.

  Lemma double_assoc_snd_involute : ∀ (s : F) (b c d e f : A),
    b × involute (c × e + s *: (d × involute f))
      == b × (involute c × involute e) + s *: (b × (involute d × f)).
  Proof.
    intros s b c d e f.
    calc (b × involute (c × e + s *: (d × involute f)))
      == (b × (involute (c × e) + involute (s *: (d × involute f))))
           { (additive (f := involute) (c × e) (s *: (d × involute f))) }
      == (b × (involute c × involute e + involute (s *: (d × involute f))))
           { (additive (f := involute) (opA := mult) c e) }
      == (b × (involute c × involute e + s *: involute (d × involute f)))
           { (homogeneous (f := involute) s (d × involute f)) }
      == (b × (involute c × involute e + s *: (involute d × involute (involute f))))
           { (additive (f := involute) (opA := mult) d (involute f)) }
      == (b × (involute c × involute e + s *: (involute d × f)))
           { involute_involute }
      == (b × (involute c × involute e) + b × (s *: (involute d × f)))
           { (left_distributive (op := mult) b (involute c × involute e) (s *: (involute d × f))) }
      == (b × (involute c × involute e) + s *: (b × (involute d × f)))
           { (mul_smul_comm s b (involute d × f)) }.
  Qed.

  Theorem double_assoc_snd : ∀ (s : F) (a b c d e f : A),
    (a × c + s *: (b × involute d)) × f + (a × d + b × involute c) × involute e
      == a × (c × f + d × involute e) + b × involute (c × e + s *: (d × involute f)).
  Proof.
    intros s a b c d e f.
    calc ((a × c + s *: (b × involute d)) × f + (a × d + b × involute c) × involute e)
      == ((a × (c × f) + s *: (b × (involute d × f))) + (a × (d × involute e) + b × (involute c × involute e)))
           { double_assoc_snd_l }
      == ((a × (c × f) + a × (d × involute e)) + (s *: (b × (involute d × f)) + b × (involute c × involute e)))
           { (add_add_add_comm (a × (c × f)) (s *: (b × (involute d × f))) (a × (d × involute e)) (b × (involute c × involute e))) }
      == (a × (c × f + d × involute e) + (s *: (b × (involute d × f)) + b × (involute c × involute e)))
           { (left_distributive (op := mult) a (c × f) (d × involute e)) }
      == (a × (c × f + d × involute e) + (b × (involute c × involute e) + s *: (b × (involute d × f))))
           { (commutative (op := add) (s *: (b × (involute d × f))) (b × (involute c × involute e))) }
      == (a × (c × f + d × involute e) + b × involute (c × e + s *: (d × involute f)))
           { double_assoc_snd_involute }.
  Qed.

  Theorem double_assoc : ∀ (s : F) (a b c d e f : A),
    (a × c + s *: (b × involute d)) × e + s *: ((a × d + b × involute c) × involute f)
      == a × (c × e + s *: (d × involute f)) + s *: (b × involute (c × f + d × involute e)).
  Proof.
    intros s a b c d e f.
    calc ((a × c + s *: (b × involute d)) × e + s *: ((a × d + b × involute c) × involute f))
      == ((a × (c × e) + s *: (b × (involute d × e))) + (s *: (a × (d × involute f)) + s *: (b × (involute c × involute f))))
           { double_assoc_l }
      == (a × (c × e + s *: (d × involute f)) + s *: (b × involute (c × f + d × involute e)))
           { double_assoc_r }.
  Qed.

The wedge of a multivector with a vector: half the product plus the product the other way round through the involution.
Notation "x ⋏ v" := (inv (1 + 1) *: (x × vec v + vec v × involute x)) (at level 40, left associativity).

  Context (two_ne_zero : (1 + 1 : F) =/= 0).

  Local Notation half := (inv ((1 : F) + 1)).

  Lemma half_double_A : ∀ x : A, half *: (x + x) == x.
  Proof.
    intros x.
    calc (half *: (x + x))
      == (half *: (((1 : F) + 1) *: x)) { (two_smul x) }
      == ((half × (1 + 1)) *: x) { (associative (op := mult) (action := scale) half (1 + 1) x) }
      == ((1 : F) *: x) { (left_inverse_on (defined_on := nonzero F) (1 + 1) two_ne_zero) }
      == x { (left_identity (action := scale) x) }.
  Qed.

  Theorem wedge_vec_one : ∀ v : V, (1 : A) ⋏ v == vec v.
  Proof.
    intros v.
    calc ((1 : A) ⋏ v)
      == (half *: (vec v + vec v × involute (1 : A))) { (left_identity (action := mult) (vec v)) }
      == (half *: (vec v + vec v × 1)) { involute_one }
      == (half *: (vec v + vec v)) { (right_identity (action := mult) (vec v)) }
      == (vec v) { (half_double_A (vec v)) }.
  Qed.

  Theorem wedge_vec_add_l : ∀ (x y : A) (v : V), (x + y) ⋏ v == x ⋏ v + y ⋏ v.
  Proof.
    intros x y v.
    calc ((x + y) ⋏ v)
      == (half *: ((x × vec v + y × vec v) + vec v × involute (x + y)))
           { (right_distributive (op := mult) x y (vec v)) }
      == (half *: ((x × vec v + y × vec v) + vec v × (involute x + involute y)))
           { (additive (f := involute) x y) }
      == (half *: ((x × vec v + y × vec v) + (vec v × involute x + vec v × involute y)))
           { (left_distributive (op := mult) (vec v) (involute x) (involute y)) }
      == (half *: ((x × vec v + vec v × involute x) + (y × vec v + vec v × involute y)))
           { (add_add_add_comm (x × vec v) (y × vec v) (vec v × involute x) (vec v × involute y)) }
      == (x ⋏ v + y ⋏ v)
           { (left_distributive (op := scale) half (x × vec v + vec v × involute x) (y × vec v + vec v × involute y)) }.
  Qed.

  Theorem wedge_vec_add_r : ∀ (x : A) (v w : V), x ⋏ (v + w) == x ⋏ v + x ⋏ w.

    calc (x ⋏ (v + w))
      == (half *: (x × (vec v + vec w) + vec (v + w) × involute x))

      == (half *: (x × (vec v + vec w) + (vec v + vec w) × involute x))

      == (half *: ((x × vec v + x × vec w) + (vec v × involute x + vec w × involute x)))
           { (right_distributive (op := mult) (vec v) (vec w) (involute x)) }
      == (half *: ((x × vec v + vec v × involute x) + (x × vec w + vec w × involute x)))

      == (x ⋏ v + x ⋏ w)
           { (left_distributive (op := scale) half (x × vec v + vec v × involute x) (x × vec w + vec w × involute x)) }.

      == (a *: (x ⋏ v)) { (smul_comm half a (x × vec v + vec v × involute x)) }.
  Qed.

  Theorem involute_wedge_vec : ∀ (x : A) (v : V), involute (x ⋏ v) == - (involute x ⋏ v).
  Proof.
    intros x v.
    transitivity (half *: (involute x × - vec v + - vec v × x)).
    calc (involute (x ⋏ v))
      == (half *: involute (x × vec v + vec v × involute x))
           { (homogeneous (f := involute) half (x × vec v + vec v × involute x)) }
      == (half *: (involute (x × vec v) + involute (vec v × involute x)))
           { (additive (f := involute) (x × vec v) (vec v × involute x)) }
      == (half *: (involute x × involute (vec v) + involute (vec v × involute x)))
           { (additive (f := involute) (opA := mult) x (vec v)) }
      == (half *: (involute x × involute (vec v) + involute (vec v) × involute (involute x)))
           { (additive (f := involute) (opA := mult) (vec v) (involute x)) }
      == (half *: (involute x × - vec v + involute (vec v) × involute (involute x)))
           { (involute_vec v) }
      == (half *: (involute x × - vec v + - vec v × involute (involute x)))
           { (involute_vec v) }
      == (half *: (involute x × - vec v + - vec v × x))
           { (involute_involute x) }.
    calc (half *: (involute x × - vec v + - vec v × x))
      == (half *: (- (involute x × vec v) + - vec v × x)) { (mul_neg (involute x) (vec v)) }
      == (half *: (- (involute x × vec v) + - (vec v × x))) { (neg_mul (vec v) x) }
      == (half *: (- (involute x × vec v + vec v × x))) { (neg_add (involute x × vec v) (vec v × x)) }
      == (- (half *: (involute x × vec v + vec v × x))) { (smul_neg half (involute x × vec v + vec v × x)) }
      == (- (half *: (involute x × vec v + vec v × involute (involute x)))) { (involute_involute x) }.
  Qed.

  Theorem wedge_vec_zero_l : ∀ v : V, (0 : A) ⋏ v == 0.

      == (half *: (0 + vec v × involute (0 : A))) { (zero_mul (vec v)) }

      == (half *: ((0 : A) + 0)) { (mul_zero (vec v)) }

  Lemma wedge_vec_mul_vec : ∀ (x : A) (v : V),
    (x ⋏ v) × vec v == half *: (x × (vec v × vec v) + vec v × (involute x × vec v)).
  Proof.
    intros x v.
    calc ((x ⋏ v) × vec v)
      == (half *: ((x × vec v + vec v × involute x) × vec v))
           { (smul_mul_assoc half (x × vec v + vec v × involute x) (vec v)) }
      == (half *: ((x × vec v) × vec v + (vec v × involute x) × vec v))
           { (right_distributive (op := mult) (x × vec v) (vec v × involute x) (vec v)) }
      == (half *: (x × (vec v × vec v) + (vec v × involute x) × vec v))
           { (associative (op := mult) (action := mult) x (vec v) (vec v)) }
      == (half *: (x × (vec v × vec v) + vec v × (involute x × vec v)))
           { (associative (op := mult) (action := mult) (vec v) (involute x) (vec v)) }.
  Qed.

  Lemma vec_mul_involute_wedge : ∀ (x : A) (v : V),
    vec v × (involute x ⋏ v) == half *: (vec v × (involute x × vec v) + (vec v × vec v) × x).
  Proof.
    intros x v.
    calc (vec v × (involute x ⋏ v))
      == (half *: (vec v × (involute x × vec v + vec v × involute (involute x))))
           { (mul_smul_comm half (vec v) (involute x × vec v + vec v × involute (involute x))) }
      == (half *: (vec v × (involute x × vec v + vec v × x)))
           { (involute_involute x) }
      == (half *: (vec v × (involute x × vec v) + vec v × (vec v × x)))
           { (left_distributive (op := mult) (vec v) (involute x × vec v) (vec v × x)) }
      == (half *: (vec v × (involute x × vec v) + (vec v × vec v) × x))
           { (associative (op := mult) (action := mult) (vec v) (vec v) x) }.
  Qed.

  Lemma scalar_commutes : ∀ (c : F) (x : A), x × (c *: 1) + - ((c *: 1) × x) == 0.
  Proof.
    intros c x.
    calc (x × (c *: 1) + - ((c *: 1) × x))
      == (c *: x + - ((c *: (1 : A)) × x)) { (mul_smul_one c x) }
      == (c *: x + - (c *: x)) { (smul_one_mul c x) }
      == (0 : A) { (right_inverse (op := add) (c *: x)) }.
  Qed.

A wedge with the same vector twice vanishes.
  Theorem wedge_vec_alternating : ∀ (x : A) (v : V), (x ⋏ v) ⋏ v == 0.
  Proof.
    intros x v.
    transitivity (half *: (half *: ((x × (vec v × vec v) + vec v × (involute x × vec v))
                                    + - (vec v × (involute x × vec v) + (vec v × vec v) × x)))).
    - calc ((x ⋏ v) ⋏ v)
        == (half *: ((x ⋏ v) × vec v + vec v × - (involute x ⋏ v)))
             { (involute_wedge_vec x v) }
        == (half *: ((x ⋏ v) × vec v + - (vec v × (involute x ⋏ v))))
             { (mul_neg (vec v) (involute x ⋏ v)) }
        == (half *: (half *: (x × (vec v × vec v) + vec v × (involute x × vec v))
                     + - (vec v × (involute x ⋏ v))))
             { (wedge_vec_mul_vec x v) }
        == (half *: (half *: (x × (vec v × vec v) + vec v × (involute x × vec v))
                     + - (half *: (vec v × (involute x × vec v) + (vec v × vec v) × x))))
             { (vec_mul_involute_wedge x v) }
        == (half *: (half *: (x × (vec v × vec v) + vec v × (involute x × vec v))
                     + half *: (- (vec v × (involute x × vec v) + (vec v × vec v) × x))))
             { (smul_neg half (vec v × (involute x × vec v) + (vec v × vec v) × x)) }
        == (half *: (half *: ((x × (vec v × vec v) + vec v × (involute x × vec v))
                              + - (vec v × (involute x × vec v) + (vec v × vec v) × x))))
             { (left_distributive (op := scale) half (x × (vec v × vec v) + vec v × (involute x × vec v))
                  (- (vec v × (involute x × vec v) + (vec v × vec v) × x))) }.
    - calc (half *: (half *: ((x × (vec v × vec v) + vec v × (involute x × vec v))
                              + - (vec v × (involute x × vec v) + (vec v × vec v) × x))))
        == (half *: (half *: ((x × (vec v × vec v) + - (vec v × (involute x × vec v)))
                              + (vec v × (involute x × vec v) + - ((vec v × vec v) × x)))))
             { (add_sub_add_comm (x × (vec v × vec v)) (vec v × (involute x × vec v))
                  (vec v × (involute x × vec v)) ((vec v × vec v) × x)) }
        == (half *: (half *: (x × (vec v × vec v) + - ((vec v × vec v) × x))))
             { (sub_add_sub_cancel (x × (vec v × vec v)) (vec v × (involute x × vec v)) ((vec v × vec v) × x)) }
        == (half *: (half *: (x × (⟪ v , v ⟫ *: 1) + - ((⟪ v , v ⟫ *: 1) × x))))
             { (vec_sq v) }
        == (half *: (half *: (0 : A)))
             { (scalar_commutes ⟪ v , v ⟫ x) }
        == (half *: (0 : A))
             { (scale_proper half half (reflexivity half) _ _ (smul_zero (V := A) half)) }
        == (0 : A)
             { (smul_zero (V := A) half) }.
  Qed.

  Lemma wedge_vec_polar : ∀ (x : A) (v w : V),
    (x ⋏ (v + w)) ⋏ (v + w) == ((x ⋏ v) ⋏ v + (x ⋏ v) ⋏ w) + ((x ⋏ w) ⋏ v + (x ⋏ w) ⋏ w).
  Proof.
    intros x v w.
    calc ((x ⋏ (v + w)) ⋏ (v + w))
      == ((x ⋏ v + x ⋏ w) ⋏ (v + w)) { (wedge_vec_add_r x v w) }
      == ((x ⋏ v) ⋏ (v + w) + (x ⋏ w) ⋏ (v + w)) { (wedge_vec_add_l (x ⋏ v) (x ⋏ w) (v + w)) }
      == (((x ⋏ v) ⋏ v + (x ⋏ v) ⋏ w) + (x ⋏ w) ⋏ (v + w)) { (wedge_vec_add_r (x ⋏ v) v w) }
      == (((x ⋏ v) ⋏ v + (x ⋏ v) ⋏ w) + ((x ⋏ w) ⋏ v + (x ⋏ w) ⋏ w)) { (wedge_vec_add_r (x ⋏ w) v w) }.
  Qed.

Two vectors wedged on in either order give opposite results.
  Theorem wedge_vec_anticomm : ∀ (x : A) (v w : V), (x ⋏ v) ⋏ w == - ((x ⋏ w) ⋏ v).
  Proof.
    intros x v w.
    apply eq_neg_of_add_eq_zero_left.
    calc ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v)
      == (0 + ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v))
           { (left_identity (action := add) ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v)) }
      == ((x ⋏ v) ⋏ v + ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v))
           { (wedge_vec_alternating x v) }
      == (((x ⋏ v) ⋏ v + ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v)) + 0)
           { (right_identity (action := add) ((x ⋏ v) ⋏ v + ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v))) }
      == (((x ⋏ v) ⋏ v + ((x ⋏ v) ⋏ w + (x ⋏ w) ⋏ v)) + (x ⋏ w) ⋏ w)
           { (wedge_vec_alternating x w) }
      == (((x ⋏ v) ⋏ v + (x ⋏ v) ⋏ w) + ((x ⋏ w) ⋏ v + (x ⋏ w) ⋏ w))
           { (add_add_assoc ((x ⋏ v) ⋏ v) ((x ⋏ v) ⋏ w) ((x ⋏ w) ⋏ v) ((x ⋏ w) ⋏ w)) }
      == ((x ⋏ (v + w)) ⋏ (v + w))
           { (wedge_vec_polar x v w) }
      == (0 : A)
           { (wedge_vec_alternating x (v + w)) }.
  Qed.
