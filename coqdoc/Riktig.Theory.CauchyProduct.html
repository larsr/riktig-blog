The Cauchy product: summing a j × b (k - j) over the triangle j ≤ k < n in either order. Exchanging the order turns the partial sums of the product series into sum_j a j × B (n - j), the form Mertens' theorem estimates.

  Local Notation "'pred' n" := (natrec N 0 (fun k _ ⇒ k) n) (at level 10, n at level 9).

  Theorem succ_monus : ∀ n j : N, j ≤ n → succ n ∸ j == succ (n ∸ j).
  Proof.
    intros n j Hjn.
    destruct (proj1 (NatOrder.le_add j n) Hjn) as [d Hd].
    assert (Hnj : n ∸ j == d).
    { calc (n ∸ j)
        == ((j + d) ∸ j) { (monus_proper_l n (j + d) (symmetry Hd) j) }
        == ((d + j) ∸ j) { (monus_proper_l (j + d) (d + j) (commutative (op := add) j d) j) }
        == (d) { (add_monus d j) }. }
    calc (succ n ∸ j)
      == (succ (j + d) ∸ j) { (monus_proper_l (succ n) (succ (j + d)) (Peano.succ_proper n (j + d) (symmetry Hd)) j) }
      == ((j + succ d) ∸ j) { (monus_proper_l (succ (j + d)) (j + succ d) (symmetry (Theory.Naturals.add_succ j d)) j) }
      == ((succ d + j) ∸ j) { (monus_proper_l (j + succ d) (succ d + j) (commutative (op := add) j (succ d)) j) }
      == (succ d) { (add_monus (succ d) j) }
      == (succ (n ∸ j)) { (Peano.succ_proper d (n ∸ j) (symmetry Hnj)) }.
  Qed.

Partial sums only read the summand below the bound.
  Theorem psum_ext_below : ∀ f g : N → V,
    Proper (equ ==> equ) f → Proper (equ ==> equ) g →
    ∀ n : N, (∀ k : N, succ k ≤ n → f k == g k) → psum f n == psum g n.
  Proof.
    intros f g Hf Hg.
    apply (Peano.nat_induction (fun n ⇒ (∀ k : N, succ k ≤ n → f k == g k) → psum f n == psum g n)).
    - intros n n' Hn. split; intros HP Hb.
      + apply (transitivity (symmetry (psum_proper f Hf n n' Hn))).
        apply (transitivity (y := psum g n)); [ | exact (psum_proper g Hg n n' Hn) ].
        apply HP. intros k Hk. apply Hb.
        exact (proj1 (NatOrder.le_proper (succ k) (succ k) (reflexivity (succ k)) n n' Hn) Hk).
      + apply (transitivity (psum_proper f Hf n n' Hn)).
        apply (transitivity (y := psum g n')); [ | exact (symmetry (psum_proper g Hg n n' Hn)) ].
        apply HP. intros k Hk. apply Hb.
        exact (proj1 (NatOrder.le_proper (succ k) (succ k) (reflexivity (succ k)) n' n (symmetry Hn)) Hk).
    - intros _.
      calc (psum f (0 : N)) == (0 : V) { (psum_zero f) } == (psum g (0 : N)) { (psum_zero g) }.
    - intros n IH Hb.
      calc (psum f (succ n))
        == (psum f n + f n) { (psum_succ f n) }
        == (psum g n + f n) { (IH (fun k Hk ⇒ Hb k (le_trans (succ k) n (succ n) Hk (le_succ_diag n)))) }
        == (psum g n + g n) { (Hb n (le_refl (succ n))) }
        == (psum g (succ n)) { (psum_succ g n) }.
  Qed.

The exchange: summing the product triangle row by row equals summing it column by column.
  Theorem cauchy_product_exchange : ∀ a b : N → V,
    Proper (equ ==> equ) a → Proper (equ ==> equ) b → ∀ n : N,
    psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) n
      == psum (fun j ⇒ a j × psum b (n ∸ j)) n.
  Proof.
    intros a b Ha Hb.
    assert (Hrow : ∀ k : N, Proper (equ ==> equ) (fun j ⇒ a j × b (k ∸ j))).
    { intros k j j' Hj. apply Magma.op_proper; [ apply Ha; exact Hj | apply Hb; exact (monus_index_proper k j j' Hj) ]. }
    assert (Hc : Proper (equ ==> equ) (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k))).
    { intros k k' Hk.
      calc (psum (fun j ⇒ a j × b (k ∸ j)) (succ k))
        == (psum (fun j ⇒ a j × b (k' ∸ j)) (succ k))
             { (psum_ext _ _ (Hrow k) (Hrow k')
                  (fun j ⇒ Magma.op_proper _ _ (reflexivity (a j)) _ _ (Hb _ _ (monus_proper_l k k' Hk j))) (succ k)) }
        == (psum (fun j ⇒ a j × b (k' ∸ j)) (succ k'))
             { (psum_proper _ (Hrow k') (succ k) (succ k') (Peano.succ_proper k k' Hk)) }. }
    assert (Hcol : ∀ n : N, Proper (equ ==> equ) (fun j ⇒ a j × psum b (n ∸ j))).
    { intros n j j' Hj. apply Magma.op_proper; [ apply Ha; exact Hj | ].
      exact (psum_proper b Hb (n ∸ j) (n ∸ j') (monus_index_proper n j j' Hj)). }
    assert (Hcols : ∀ n n' : N, n == n' →
      psum (fun j ⇒ a j × psum b (n ∸ j)) n == psum (fun j ⇒ a j × psum b (n' ∸ j)) n').
    { intros n n' Hn.
      calc (psum (fun j ⇒ a j × psum b (n ∸ j)) n)
        == (psum (fun j ⇒ a j × psum b (n' ∸ j)) n)
             { (psum_ext _ _ (Hcol n) (Hcol n')
                  (fun j ⇒ Magma.op_proper _ _ (reflexivity (a j)) _ _ (psum_proper b Hb _ _ (monus_proper_l n n' Hn j))) n) }
        == (psum (fun j ⇒ a j × psum b (n' ∸ j)) n') { (psum_proper _ (Hcol n') n n' Hn) }. }
    apply (Peano.nat_induction (fun n ⇒
      psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) n
        == psum (fun j ⇒ a j × psum b (n ∸ j)) n)).
    - intros n n' Hn. rewrite (psum_proper _ Hc n n' Hn), (Hcols n n' Hn). reflexivity.
    - calc (psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) (0 : N))
        == (0 : V) { (psum_zero _) }
        == (psum (fun j ⇒ a j × psum b ((0 : N) ∸ j)) (0 : N)) { (psum_zero _) }.
    - intros n IH.
      assert (Hboth : Proper (equ ==> equ) (fun j ⇒ a j × psum b (n ∸ j) + a j × b (n ∸ j))).
      { intros j j' Hj. apply Magma.op_proper; [ exact (Hcol n j j' Hj) | exact (Hrow n j j' Hj) ]. }
      assert (Hterm : ∀ k : N, succ k ≤ n →
        a k × psum b (succ n ∸ k) == a k × psum b (n ∸ k) + a k × b (n ∸ k)).
      { intros k Hk.
        calc (a k × psum b (succ n ∸ k))
          == (a k × psum b (succ (n ∸ k))) { (psum_proper b Hb _ _ (succ_monus n k (le_trans k (succ k) n (le_succ_diag k) Hk))) }
          == (a k × (psum b (n ∸ k) + b (n ∸ k))) { (psum_succ b (n ∸ k)) }
          == (a k × psum b (n ∸ k) + a k × b (n ∸ k)) { (left_distributive (a k) (psum b (n ∸ k)) (b (n ∸ k))) }. }
      assert (Hmid : psum (fun j ⇒ a j × psum b (succ n ∸ j)) n
        == psum (fun j ⇒ a j × psum b (n ∸ j)) n + psum (fun j ⇒ a j × b (n ∸ j)) n).
      { calc (psum (fun j ⇒ a j × psum b (succ n ∸ j)) n)
          == (psum (fun j ⇒ a j × psum b (n ∸ j) + a j × b (n ∸ j)) n) { (psum_ext_below _ _ (Hcol (succ n)) Hboth n Hterm) }
          == (psum (fun j ⇒ a j × psum b (n ∸ j)) n + psum (fun j ⇒ a j × b (n ∸ j)) n) { (psum_add _ _ (Hcol n) (Hrow n) n) }. }
      assert (Hlast : a n × psum b (succ n ∸ n) == a n × b (n ∸ n)).
      { calc (a n × psum b (succ n ∸ n))
          == (a n × psum b (succ (n ∸ n))) { (psum_proper b Hb _ _ (succ_monus n n (le_refl n))) }
          == (a n × (psum b (n ∸ n) + b (n ∸ n))) { (psum_succ b (n ∸ n)) }
          == (a n × (psum b 0 + b (n ∸ n))) { (psum_proper b Hb (n ∸ n) 0 (monus_self n)) }
          == (a n × (0 + b (n ∸ n))) { (psum_zero b) }
          == (a n × b (n ∸ n)) { (left_identity (action := add) (b (n ∸ n))) }. }
      calc (psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) (succ n))
        == (psum (fun k ⇒ psum (fun j ⇒ a j × b (k ∸ j)) (succ k)) n
              + psum (fun j ⇒ a j × b (n ∸ j)) (succ n)) { (psum_succ _ n) }
        == (psum (fun j ⇒ a j × psum b (n ∸ j)) n
              + psum (fun j ⇒ a j × b (n ∸ j)) (succ n)) { IH }
        == (psum (fun j ⇒ a j × psum b (n ∸ j)) n
              + (psum (fun j ⇒ a j × b (n ∸ j)) n + a n × b (n ∸ n))) { (psum_succ _ n) }
        == ((psum (fun j ⇒ a j × psum b (n ∸ j)) n + psum (fun j ⇒ a j × b (n ∸ j)) n)
              + a n × b (n ∸ n)) { (associative (op := add) (action := add) _ _ _) }
        == (psum (fun j ⇒ a j × psum b (succ n ∸ j)) n + a n × b (n ∸ n)) { Hmid }
        == (psum (fun j ⇒ a j × psum b (succ n ∸ j)) n + a n × psum b (succ n ∸ n)) { Hlast }
        == (psum (fun j ⇒ a j × psum b (succ n ∸ j)) (succ n)) { (psum_succ _ n) }.
  Qed.

A common right factor comes out of a partial sum.
  Theorem psum_mul_r : ∀ (f : N → V) (x : V), Proper (equ ==> equ) f →
    ∀ n : N, psum (fun k ⇒ f k × x) n == psum f n × x.
  Proof.
    intros f x Hf.
    apply (Peano.nat_induction (fun n ⇒ psum (fun k ⇒ f k × x) n == psum f n × x)).
    - intros n n' Hn.
      assert (Hfx : Proper (equ ==> equ) (fun k ⇒ f k × x))
        by (intros k k' Hk; apply Magma.op_proper; [ apply Hf; exact Hk | reflexivity ]).
      split; intros Heq.
      + apply (transitivity (symmetry (psum_proper _ Hfx n n' Hn))).
        apply (transitivity Heq).
        apply Magma.op_proper; [ exact (psum_proper f Hf n n' Hn) | reflexivity ].
      + apply (transitivity (psum_proper _ Hfx n n' Hn)).
        apply (transitivity Heq).
        apply Magma.op_proper; [ exact (psum_proper f Hf n' n (symmetry Hn)) | reflexivity ].
    - calc (psum (fun k ⇒ f k × x) (0 : N))
        == (0 : V) { (psum_zero _) }
        == ((0 : V) × x) { (symmetry (zero_mul x)) }
        == (psum f (0 : N) × x) { (psum_zero f) }.
    - intros n IH.
      calc (psum (fun k ⇒ f k × x) (succ n))
        == (psum (fun k ⇒ f k × x) n + f n × x) { (psum_succ _ n) }
        == (psum f n × x + f n × x) { IH }
        == ((psum f n + f n) × x) { (symmetry (right_distributive (psum f n) (f n) x)) }
        == (psum f (succ n) × x) { (Magma.op_proper _ _ (symmetry (psum_succ f n)) x x (reflexivity x)) }.
  Qed.

  Theorem monus_add_cancel : ∀ n K : N, K ≤ n → (n ∸ K) + K == n.
  Proof.
    intros n K HK.
    destruct (proj1 (NatOrder.le_add K n) HK) as [d Hd].
    assert (Hnd : n ∸ K == d).
    { calc (n ∸ K)
        == ((K + d) ∸ K) { (monus_proper_l n (K + d) (symmetry Hd) K) }
        == ((d + K) ∸ K) { (monus_proper_l (K + d) (d + K) (commutative (op := add) K d) K) }
        == (d) { (add_monus d K) }. }
    calc ((n ∸ K) + K)
      == (d + K) { (Magma.op_proper _ _ Hnd K K (reflexivity K)) }
      == (K + d) { (commutative (op := add) d K) }
      == (n) { Hd }.
  Qed.

  Theorem le_monus : ∀ n j K : N, j + K ≤ n → K ≤ n ∸ j.
  Proof.
    intros n j K HK.
    destruct (proj1 (NatOrder.le_add (j + K) n) HK) as [c Hc].
    assert (Hn2 : n == j + (K + c)).
    { calc (n)
        == ((j + K) + c) { Hc }
        == (j + (K + c)) { (associative (op := add) (action := add) j K c) }. }
    assert (Hnj : n ∸ j == K + c).
    { calc (n ∸ j)
        == ((j + (K + c)) ∸ j) { (monus_proper_l n (j + (K + c)) Hn2 j) }
        == (((K + c) + j) ∸ j) { (monus_proper_l (j + (K + c)) ((K + c) + j) (commutative (op := add) j (K + c)) j) }
        == (K + c) { (add_monus (K + c) j) }. }
    apply (proj1 (NatOrder.le_proper K K (reflexivity K) (K + c) (n ∸ j) (symmetry Hnj))).
    exact (le_add_r K c).
  Qed.
