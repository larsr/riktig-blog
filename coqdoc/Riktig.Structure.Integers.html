From Riktig.Structure Require Export Ring.

The integers: a commutative ring in which every element is reached from zero by adding and subtracting one.
Module Integers.
  Class Integers (Z : Type) {eq : EqRel Z} (zero : Zero Z) (add : HasAdd Z)
      (neg : HasNeg Z) (one : One Z) (mult : HasMult Z) := {
    ring :: Ring Z 0 add neg 1 mult;
    mul_commutative :: Commutative Z Z mult;
    int_induction : ∀ P : Z → Prop, Proper (equ ==> iff) P →
      P 0 →
      (∀ z : Z, P z → P (z + 1)) →
      (∀ z : Z, P z → P (z + - (1 : Z))) →
      ∀ z : Z, P z;
  }.
  Hint Mode Integers ! - - - - - - : typeclass_instances.
End Integers.
Export Integers.
