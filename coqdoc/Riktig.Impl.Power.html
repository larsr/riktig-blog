Section MonoidPower.
  Context {M : Type} {eM : EqRel M} {oM : One M} {mM : HasMult M}
    {MM : @Monoid M eM oM mM}.

The specification read as a program: one multiplication per unit of the exponent.
  Fixpoint pow_nat (x : M) (n : nat) : M :=
    match n with
    | O ⇒ 1
    | S n' ⇒ x × pow_nat x n'
    end.

  #[export] Instance pow_nat_Power : Power M pow_nat := {
    pow_zero := fun x ⇒ reflexivity (1 : M);
    pow_succ := fun x n ⇒ reflexivity (x × pow_nat x n);
  }.

Repeated squaring: one multiplication per bit of the exponent, and one more for each bit that is set.
  Fixpoint pow_positive (x : M) (p : positive) : M :=
    match p with
    | xH ⇒ x
    | xO p' ⇒ pow_positive x p' × pow_positive x p'
    | xI p' ⇒ x × (pow_positive x p' × pow_positive x p')
    end.

Repeated squaring computes the naive power. positive is a binary numeral for a nat, so the two operations have different exponent types: this is a refinement along Pos.to_nat, not two instances of one operation. Every step is pow_add at n + n, which is where associativity is spent.
  Theorem pow_positive_is_pow_nat : ∀ (x : M) (p : positive),
    pow_positive x p == pow_nat x (Pos.to_nat p).
  Proof.
    intros x p.
    induction p as [p' IH | p' IH |].
    - calc (x × (pow_positive x p' × pow_positive x p'))
        == (x × (pow_nat x (Pos.to_nat p') × pow_nat x (Pos.to_nat p')))
             { (op_proper x x (reflexivity x) _ _ (op_proper _ _ IH _ _ IH)) }
        == (x × pow_nat x (Pos.to_nat p' + Pos.to_nat p')%nat)
             { (op_proper x x (reflexivity x) _ _
                  (symmetry (pow_add pow_nat x
                               (Pos.to_nat p') (Pos.to_nat p')))) }
        == (pow_nat x (Pos.to_nat p'~1))
             { (pow_exponent pow_nat x _ _
                  (eq_sym (eq_trans (Pos2Nat.inj_xI p')
                             (succ_double_add (Pos.to_nat p'))))) }.
    - calc (pow_positive x p' × pow_positive x p')
        == (pow_nat x (Pos.to_nat p') × pow_nat x (Pos.to_nat p'))
             { (op_proper _ _ IH _ _ IH) }
        == (pow_nat x (Pos.to_nat p' + Pos.to_nat p')%nat)
             { (symmetry (pow_add pow_nat x (Pos.to_nat p') (Pos.to_nat p'))) }
        == (pow_nat x (Pos.to_nat p'~0))
             { (pow_exponent pow_nat x _ _
                  (eq_sym (eq_trans (Pos2Nat.inj_xO p')
                             (double_add (Pos.to_nat p'))))) }.
    - exact (symmetry (right_identity x : x × (1 : M) == x)).
  Qed.

Repeated squaring at a nat exponent, which pow_positive cannot reach: positive has no zero. Pos.of_succ_nat is the refinement map back, and it is a fixpoint on S, so the successor clause below holds by conversion.
  Definition pow_by_squaring (x : M) (n : nat) : M :=
    match n with
    | O ⇒ 1
    | S m ⇒ pow_positive x (Pos.of_succ_nat m)
    end.

  Lemma pow_by_squaring_is_pow_nat : ∀ (x : M) (n : nat),
    pow_by_squaring x n == pow_nat x n.
  Proof.
    intros x [| m].
    - reflexivity.
    - calc (pow_positive x (Pos.of_succ_nat m))
        == (pow_nat x (Pos.to_nat (Pos.of_succ_nat m)))
             { (pow_positive_is_pow_nat x (Pos.of_succ_nat m)) }
        == (pow_nat x (S m))
             { (pow_exponent pow_nat x _ _ (SuccNat2Pos.id_succ m)) }.
  Qed.

  #[export] Instance pow_by_squaring_Power : Power M pow_by_squaring.
  Proof.
    split.
    - intros x. reflexivity.
    - intros x n.
      calc (pow_by_squaring x (S n))
        == (x × pow_nat x n) { (pow_by_squaring_is_pow_nat x (S n)) }
        == (x × pow_by_squaring x n)
             { (op_proper x x (reflexivity x) _ _
                  (symmetry (pow_by_squaring_is_pow_nat x n))) }.
  Qed.

End MonoidPower.
