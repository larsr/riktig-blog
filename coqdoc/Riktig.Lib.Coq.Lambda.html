Lambda terms with de Bruijn variables and named constants, a context of named definitions, and a call-by-value evaluator with closures. A constant is looked up in the context -- the delta rule -- and one that is not defined is inert: applying it just collects the arguments. Evaluation takes fuel, since a term need not terminate.
Inductive term : Type :=
  | Var (n : nat)
  | Lam (body : term)
  | App (f a : term)
  | Const (x : string).

Inductive value : Type :=
  | Clo (env : list value) (body : term)
  | Neu (x : string) (args : list value).

Definition ctx : Type := list (string × term).

Fixpoint lookup (x : string) (G : ctx) : option term :=
  match G with
  | [] ⇒ None
  | (y, t) :: G' ⇒ if String.eqb x y then Some t else lookup x G'
  end.

Substitution

Renamings and parallel substitutions on de Bruijn terms, with the laws that make them a substitution calculus.
Definition sub : Type := nat → term.

Definition scons {X : Type} (x : X) (f : nat → X) : nat → X :=
  fun n ⇒ match n with 0 ⇒ x | S n ⇒ f n end.

Fixpoint ren (r : nat → nat) (t : term) : term :=
  match t with
  | Var n ⇒ Var (r n)
  | Lam b ⇒ Lam (ren (scons 0 (fun n ⇒ S (r n))) b)
  | App f a ⇒ App (ren r f) (ren r a)
  | Const x ⇒ Const x
  end.

Definition up (s : sub) : sub := scons (Var 0) (fun n ⇒ ren S (s n)).

Fixpoint psub (s : sub) (t : term) : term :=
  match t with
  | Var n ⇒ s n
  | Lam b ⇒ Lam (psub (up s) b)
  | App f a ⇒ App (psub s f) (psub s a)
  | Const x ⇒ Const x
  end.

Definition shift : sub := fun n ⇒ Var (S n).

Lemma scons_ext : ∀ (X : Type) (x : X) (f g : nat → X),
  (∀ n, f n = g n) → ∀ n, scons x f n = scons x g n.
Proof. intros X x f g H [| n]; [ reflexivity | apply H ]. Qed.

Lemma ren_ext : ∀ (t : term) (r r' : nat → nat), (∀ n, r n = r' n) → ren r t = ren r' t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros r r' H; simpl.
  - rewrite H. reflexivity.
  - f_equal. apply IH. intros [| n]; [ reflexivity | simpl; rewrite H; reflexivity ].
  - f_equal; [ apply IHf | apply IHa ]; exact H.
  - reflexivity.
Qed.

Lemma psub_ext : ∀ (t : term) (s s' : sub), (∀ n, s n = s' n) → psub s t = psub s' t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros s s' H; simpl.
  - apply H.
  - f_equal. apply IH. intros [| n]; [ reflexivity | simpl; rewrite H; reflexivity ].
  - f_equal; [ apply IHf | apply IHa ]; exact H.
  - reflexivity.
Qed.

Lemma ren_ren : ∀ (t : term) (r r' : nat → nat), ren r' (ren r t) = ren (fun n ⇒ r' (r n)) t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros r r'; simpl.
  - reflexivity.
  - f_equal. rewrite IH. apply ren_ext. intros [| n]; reflexivity.
  - f_equal; [ apply IHf | apply IHa ].
  - reflexivity.
Qed.

Lemma psub_ren : ∀ (t : term) (r : nat → nat) (s : sub), psub s (ren r t) = psub (fun n ⇒ s (r n)) t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros r s; simpl.
  - reflexivity.
  - f_equal. rewrite IH. apply psub_ext. intros [| n]; reflexivity.
  - f_equal; [ apply IHf | apply IHa ].
  - reflexivity.
Qed.

Lemma ren_psub : ∀ (t : term) (s : sub) (r : nat → nat), ren r (psub s t) = psub (fun n ⇒ ren r (s n)) t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros s r; simpl.
  - reflexivity.
  - f_equal. rewrite IH. apply psub_ext. intros [| n]; [ reflexivity | ].
    simpl. rewrite !ren_ren. apply ren_ext. intros m. reflexivity.
  - f_equal; [ apply IHf | apply IHa ].
  - reflexivity.
Qed.

Lemma psub_psub : ∀ (t : term) (s s' : sub), psub s' (psub s t) = psub (fun n ⇒ psub s' (s n)) t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros s s'; simpl.
  - reflexivity.
  - f_equal. rewrite IH. apply psub_ext. intros [| n]; [ reflexivity | ].
    simpl. rewrite psub_ren, ren_psub. apply psub_ext. intros m. reflexivity.
  - f_equal; [ apply IHf | apply IHa ].
  - reflexivity.
Qed.

Lemma ren_as_psub : ∀ (t : term) (r : nat → nat), ren r t = psub (fun n ⇒ Var (r n)) t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; intros r; simpl.
  - reflexivity.
  - f_equal. rewrite IH. apply psub_ext. intros [| n]; reflexivity.
  - f_equal; [ apply IHf | apply IHa ].
  - reflexivity.
Qed.

Lemma psub_var : ∀ t : term, psub Var t = t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x]; simpl.
  - reflexivity.
  - f_equal. rewrite <- IH at 2. apply psub_ext. intros [| n]; reflexivity.
  - f_equal; assumption.
  - reflexivity.
Qed.

Substitution of a substituted body: the key commutation.

Lemma psub_beta : ∀ (b u : term) (s : sub),
  psub (scons (psub s u) Var) (psub (up s) b) = psub s (psub (scons u Var) b).
Proof.
  intros b u s. rewrite !psub_psub. apply psub_ext. intros [| n]; simpl.
  - reflexivity.
  - rewrite psub_ren. simpl. apply psub_var.
Qed.

Reduction

One step of beta anywhere in a term, its reflexive-transitive closure, and Takahashi's complete development, which proves the closure confluent. Conversion is joinability: two terms are convertible when they reduce to a common term.
Section Star.
  Context {X : Type} (R : X → X → Prop).

  Inductive star : X → X → Prop :=
    | star_refl : ∀ t : X, star t t
    | star_step : ∀ t u v : X, R t u → star u v → star t v.

  Lemma star_one : ∀ t u : X, R t u → star t u.
  Proof. intros t u H. exact (star_step t u u H (star_refl u)). Qed.

  Lemma star_trans : ∀ t u v : X, star t u → star u v → star t v.
  Proof.
    intros t u v H1. revert v. induction H1 as [t | t w u H Hs IH]; intros v H2.
    - exact H2.
    - exact (star_step t w v H (IH v H2)).
  Qed.

  Lemma star_snoc : ∀ t u v : X, star t u → R u v → star t v.
  Proof. intros t u v H1 H2. exact (star_trans t u v H1 (star_one u v H2)). Qed.
End Star.

Inductive bstep : term → term → Prop :=
  | bstep_beta : ∀ b u : term, bstep (App (Lam b) u) (psub (scons u Var) b)
  | bstep_lam : ∀ b b' : term, bstep b b' → bstep (Lam b) (Lam b')
  | bstep_app_l : ∀ f f' a : term, bstep f f' → bstep (App f a) (App f' a)
  | bstep_app_r : ∀ f a a' : term, bstep a a' → bstep (App f a) (App f a').

Definition bsteps : term → term → Prop := star bstep.

Lemma bsteps_lam : ∀ b b' : term, bsteps b b' → bsteps (Lam b) (Lam b').
Proof.
  intros b b' H. induction H as [t | t u v H Hs IH].
  - apply star_refl.
  - exact (star_step bstep _ _ _ (bstep_lam _ _ H) IH).
Qed.

Lemma bsteps_app_l : ∀ f f' a : term, bsteps f f' → bsteps (App f a) (App f' a).
Proof.
  intros f f' a H. induction H as [t | t u v H Hs IH].
  - apply star_refl.
  - exact (star_step bstep _ _ _ (bstep_app_l _ _ a H) IH).
Qed.

Lemma bsteps_app_r : ∀ f a a' : term, bsteps a a' → bsteps (App f a) (App f a').
Proof.
  intros f a a' H. induction H as [t | t u v H Hs IH].
  - apply star_refl.
  - exact (star_step bstep _ _ _ (bstep_app_r f _ _ H) IH).
Qed.

Lemma bsteps_app : ∀ f f' a a' : term, bsteps f f' → bsteps a a' → bsteps (App f a) (App f' a').
Proof.
  intros f f' a a' Hf Ha.
  exact (star_trans bstep _ _ _ (bsteps_app_l f f' a Hf) (bsteps_app_r f' a a' Ha)).
Qed.

Lemma bstep_psub : ∀ (t t' : term) (s : sub), bstep t t' → bstep (psub s t) (psub s t').
Proof.
  intros t t' s H. revert s. induction H as [b u | b b' H IH | f f' a H IH | f a a' H IH]; intros s; simpl.
  - rewrite <- (psub_beta b u s). exact (bstep_beta (psub (up s) b) (psub s u)).
  - exact (bstep_lam _ _ (IH (up s))).
  - exact (bstep_app_l _ _ _ (IH s)).
  - exact (bstep_app_r _ _ _ (IH s)).
Qed.

Lemma bsteps_psub : ∀ (s : sub) (t t' : term), bsteps t t' → bsteps (psub s t) (psub s t').
Proof.
  intros s t t' H. induction H as [t | t u v H Hs IH].
  - apply star_refl.
  - exact (star_step bstep _ _ _ (bstep_psub _ _ s H) IH).
Qed.

Lemma bsteps_var_inv : ∀ (n : nat) (t : term), bsteps (Var n) t → t = Var n.
Proof.
  intros n t H. remember (Var n) as v eqn:Hv. induction H as [t | t u w H Hs IH]; subst.
  - reflexivity.
  - inversion H.
Qed.

Parallel reduction and the complete development

Inductive par : term → term → Prop :=
  | par_var : ∀ n : nat, par (Var n) (Var n)
  | par_const : ∀ x : string, par (Const x) (Const x)
  | par_lam : ∀ b b' : term, par b b' → par (Lam b) (Lam b')
  | par_app : ∀ f f' a a' : term, par f f' → par a a' → par (App f a) (App f' a')
  | par_beta : ∀ b b' u u' : term, par b b' → par u u' → par (App (Lam b) u) (psub (scons u' Var) b').

Lemma par_refl : ∀ t : term, par t t.
Proof.
  induction t as [n | b IH | f IHf a IHa | x].
  - apply par_var.
  - apply par_lam; exact IH.
  - apply par_app; assumption.
  - apply par_const.
Qed.

Lemma bstep_par : ∀ t u : term, bstep t u → par t u.
Proof.
  intros t u H. induction H as [b u | b b' H IH | f f' a H IH | f a a' H IH].
  - exact (par_beta b b u u (par_refl b) (par_refl u)).
  - apply par_lam; exact IH.
  - apply par_app; [ exact IH | apply par_refl ].
  - apply par_app; [ apply par_refl | exact IH ].
Qed.

Lemma par_bsteps : ∀ t u : term, par t u → bsteps t u.
Proof.
  intros t u H. induction H as [n | x | b b' H IH | f f' a a' Hf IHf Ha IHa | b b' u u' Hb IHb Hu IHu].
  - apply star_refl.
  - apply star_refl.
  - apply bsteps_lam; exact IH.
  - apply bsteps_app; assumption.
  - apply (star_snoc bstep _ (App (Lam b') u')).
    + exact (bsteps_app _ _ _ _ (bsteps_lam _ _ IHb) IHu).
    + exact (bstep_beta b' u').
Qed.

Lemma ren_beta : ∀ (b u : term) (r : nat → nat),
  ren r (psub (scons u Var) b) = psub (scons (ren r u) Var) (ren (scons 0 (fun n ⇒ S (r n))) b).
Proof.
  intros b u r. rewrite ren_psub, psub_ren. apply psub_ext. intros [| n]; reflexivity.
Qed.

Lemma par_ren : ∀ t t' : term, par t t' → ∀ r : nat → nat, par (ren r t) (ren r t').
Proof.
  intros t t' H. induction H as [n | x | b b' H IH | f f' a a' Hf IHf Ha IHa | b b' u u' Hb IHb Hu IHu]; intros r; simpl.
  - apply par_var.
  - apply par_const.
  - apply par_lam; exact (IH _).
  - apply par_app; [ exact (IHf r) | exact (IHa r) ].
  - rewrite ren_beta. exact (par_beta _ _ _ _ (IHb _) (IHu r)).
Qed.

Lemma up_par : ∀ s s' : sub, (∀ n : nat, par (s n) (s' n)) → ∀ n : nat, par (up s n) (up s' n).
Proof. intros s s' Hs [| n]; simpl; [ apply par_var | apply par_ren; apply Hs ]. Qed.

Lemma par_psub : ∀ t t' : term, par t t' →
  ∀ s s' : sub, (∀ n : nat, par (s n) (s' n)) → par (psub s t) (psub s' t').
Proof.
  intros t t' H. induction H as [n | x | b b' H IH | f f' a a' Hf IHf Ha IHa | b b' u u' Hb IHb Hu IHu]; intros s s' Hs; simpl.
  - apply Hs.
  - apply par_const.
  - apply par_lam. exact (IH _ _ (up_par s s' Hs)).
  - apply par_app; [ exact (IHf s s' Hs) | exact (IHa s s' Hs) ].
  - rewrite <- (psub_beta b' u' s').
    exact (par_beta _ _ _ _ (IHb _ _ (up_par s s' Hs)) (IHu s s' Hs)).
Qed.

Lemma par_psub_single : ∀ b b' u u' : term,
  par b b' → par u u' → par (psub (scons u Var) b) (psub (scons u' Var) b').
Proof.
  intros b b' u u' Hb Hu. apply (par_psub b b' Hb).
  intros [| n]; simpl; [ exact Hu | apply par_var ].
Qed.

The complete development: contract every visible redex at once.
Fixpoint develop (t : term) : term :=
  match t with
  | Var n ⇒ Var n
  | Const x ⇒ Const x
  | Lam b ⇒ Lam (develop b)
  | App f a ⇒
    match f with
    | Lam b ⇒ psub (scons (develop a) Var) (develop b)
    | _ ⇒ App (develop f) (develop a)
    end
  end.

Lemma par_lam_inv : ∀ b t : term, par (Lam b) t → ∃ b' : term, t = Lam b' ∧ par b b'.
Proof. intros b t H. inversion H; subst. ∃ b'. split; [ reflexivity | assumption ]. Qed.

Every parallel reduct of t parallel-reduces to the complete development of t: the triangle property, and so the diamond.
Theorem par_triangle : ∀ t u : term, par t u → par u (develop t).
Proof.
  intros t u H. induction H as [n | x | b b' H IH | f f' a a' Hf IHf Ha IHa | b b' u u' Hb IHb Hu IHu]; simpl.
  - apply par_var.
  - apply par_const.
  - apply par_lam; exact IH.
  - destruct f as [n | b | f1 f2 | x].
    + apply par_app; assumption.
    + destruct (par_lam_inv _ _ Hf) as [b2 [Hf' Hb2]]. subst f'.
      destruct (par_lam_inv _ _ IHf) as [b3 [Hb3 Hpar3]].
      injection Hb3 as Hb3. subst b3.
      exact (par_beta _ _ _ _ Hpar3 IHa).
    + apply par_app; assumption.
    + apply par_app; assumption.
  - exact (par_psub_single _ _ _ _ IHb IHu).
Qed.

Corollary par_diamond : ∀ t u v : term, par t u → par t v → ∃ w : term, par u w ∧ par v w.
Proof. intros t u v Hu Hv. ∃ (develop t). split; apply par_triangle; assumption. Qed.

Definition pars : term → term → Prop := star par.

Lemma par_strip : ∀ t u v : term, par t u → pars t v → ∃ w : term, pars u w ∧ par v w.
Proof.
  intros t u v Hp Hs. revert u Hp. induction Hs as [t | t x v Hx Hs IH]; intros u Hp.
  - ∃ u. split; [ apply star_refl | exact Hp ].
  - destruct (par_diamond _ _ _ Hx Hp) as [y [Hxy Huy]].
    destruct (IH y Hxy) as [w [Hyw Hvw]].
    ∃ w. split; [ exact (star_step par _ _ _ Huy Hyw) | exact Hvw ].
Qed.

Lemma pars_confluent : ∀ t u v : term, pars t u → pars t v → ∃ w : term, pars u w ∧ pars v w.
Proof.
  intros t u v Hu. revert v. induction Hu as [t | t x u Hx Hu IH]; intros v Hv.
  - ∃ v. split; [ exact Hv | apply star_refl ].
  - destruct (par_strip _ _ _ Hx Hv) as [y [Hxy Hvy]].
    destruct (IH _ Hxy) as [w [Huw Hyw]].
    ∃ w. split; [ exact Huw | exact (star_step par _ _ _ Hvy Hyw) ].
Qed.

Lemma bsteps_pars : ∀ t u : term, bsteps t u → pars t u.
Proof.
  intros t u H. induction H as [t | t x u Hx Hs IH].
  - apply star_refl.
  - exact (star_step par _ _ _ (bstep_par _ _ Hx) IH).
Qed.

Lemma pars_bsteps : ∀ t u : term, pars t u → bsteps t u.
Proof.
  intros t u H. induction H as [t | t x u Hx Hs IH].
  - apply star_refl.
  - exact (star_trans bstep _ _ _ (par_bsteps _ _ Hx) IH).
Qed.

Theorem bsteps_confluent : ∀ t u v : term, bsteps t u → bsteps t v → ∃ w : term, bsteps u w ∧ bsteps v w.
Proof.
  intros t u v Hu Hv.
  destruct (pars_confluent t u v (bsteps_pars _ _ Hu) (bsteps_pars _ _ Hv)) as [w [H1 H2]].
  ∃ w. split; apply pars_bsteps; assumption.
Qed.

Conversion

Two terms are convertible when they reduce to a common term; confluence is what makes this transitive.
Definition joinable (t u : term) : Prop := ∃ w : term, bsteps t w ∧ bsteps u w.

Lemma joinable_refl : ∀ t : term, joinable t t.
Proof. intros t. ∃ t. split; apply star_refl. Qed.

Lemma eq_joinable : ∀ t u : term, t = u → joinable t u.
Proof. intros t u H. subst. apply joinable_refl. Qed.

Church numerals: n is fun f x ⇒ f (... (f x)).
Fixpoint church_body (n : nat) : term :=
  match n with 0 ⇒ Var 0 | S n ⇒ App (Var 1) (church_body n) end.
Definition church (n : nat) : term := Lam (Lam (church_body n)).

Reading a numeral back: apply it to the inert constants S and Z and count.
Fixpoint count_S (v : value) : option nat :=
  match v with
  | Neu x [] ⇒ if String.eqb x "Z" then Some 0 else None
  | Neu x [v'] ⇒ if String.eqb x "S" then option_map S (count_S v') else None
  | _ ⇒ None
  end.

Terms with names, as written, and their resolution to de Bruijn indices: a name bound by an enclosing lambda becomes its index, any other name a constant, and a number its Church numeral.
Inductive named : Type :=
  | NVar (x : string)
  | NLam (x : string) (body : named)
  | NApp (f a : named)
  | NNum (n : nat).

Fixpoint index_of (x : string) (bound : list string) : option nat :=
  match bound with
  | [] ⇒ None
  | y :: bound' ⇒ if String.eqb x y then Some 0 else option_map S (index_of x bound')
  end.

Fixpoint resolve (bound : list string) (t : named) : term :=
  match t with
  | NVar x ⇒ match index_of x bound with Some n ⇒ Var n | None ⇒ Const x end
  | NLam x b ⇒ Lam (resolve (x :: bound) b)
  | NApp f a ⇒ App (resolve bound f) (resolve bound a)
  | NNum n ⇒ church n
  end.

Writing terms, and normalizing them

The notation of the calculus itself -- λ a abstracts, # n is a variable, ⦅ f a .. b ⦆ applies -- and a cheap normalizer: iterated complete development. Two terms are convertible whenever some developments of each coincide, so conversion of closed computations is one vm_compute away.
Declare Scope term_scope.
Delimit Scope term_scope with tm.
Bind Scope term_scope with term.
Notation "'λ' a" := (Lam a) (at level 10) : term_scope.
Notation "# a" := (Var a) (at level 0) : term_scope.
Notation "⦅ a b .. c ⦆" := (App .. (App a b) .. c)
  (at level 0, a at level 0, b at level 0, c at level 0) : term_scope.

Fixpoint normalize (k : nat) (t : term) : term :=
  match k with 0 ⇒ t | S k ⇒ normalize k (develop t) end.

Lemma par_develop : ∀ t : term, par t (develop t).
Proof. intros t. exact (par_triangle t t (par_refl t)). Qed.

Lemma bsteps_normalize : ∀ (k : nat) (t : term), bsteps t (normalize k t).
Proof.
  intros k. induction k as [| k IH]; intros t; simpl.
  - apply star_refl.
  - exact (star_trans bstep _ _ _ (par_bsteps _ _ (par_develop t)) (IH (develop t))).
Qed.

Lemma joinable_of_normalize : ∀ (i j : nat) (s t : term),
  normalize i s = normalize j t → joinable s t.
Proof.
  intros i j s t H. ∃ (normalize i s).
  split; [ apply bsteps_normalize | rewrite H; apply bsteps_normalize ].
Qed.
