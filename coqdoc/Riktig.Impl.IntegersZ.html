Coq's Z.
Instance Z_EqRel : EqRel Z := eq.
Instance Z_Zero : Zero Z := 0%Z.
Instance Z_One : One Z := 1%Z.
Instance Z_HasAdd : HasAdd Z := Z.add.
Instance Z_HasMult : HasMult Z := Z.mul.
Instance Z_HasNeg : HasNeg Z := Z.opp.

Instance Z_add_Magma : Magma Z Z_HasAdd :=
  {| Magma.equivalence := eq_equivalence;
     Magma.op_proper := fun x y Hx u v Hu ⇒ f_equal2 Z.add Hx Hu |}.
Instance Z_add_Semigroup : Semigroup Z Z_HasAdd :=
  {| Semigroup.magma := Z_add_Magma;
     Semigroup.is_associative := fun a b c ⇒ eq_sym (Z.add_assoc a b c) |}.
Instance Z_add_Monoid : Monoid Z Z_Zero Z_HasAdd :=
  {| Monoid.semigroup := Z_add_Semigroup;
     Monoid.is_left_identity := Z.add_0_l; Monoid.is_right_identity := Z.add_0_r |}.
Instance Z_Group : Group Z Z_Zero Z_HasAdd Z_HasNeg :=
  {| Group.monoid := Z_add_Monoid;
     Group.neg_proper := f_equal Z.opp;
     Group.is_left_inverse := Z.add_opp_diag_l; Group.is_right_inverse := Z.add_opp_diag_r |}.
Instance Z_AbGroup : AbGroup Z Z_Zero Z_HasAdd Z_HasNeg :=
  {| AbGroup.group := Z_Group; AbGroup.is_commutative := Z.add_comm |}.

Instance Z_mul_Magma : Magma Z Z_HasMult :=
  {| Magma.equivalence := eq_equivalence;
     Magma.op_proper := fun x y Hx u v Hu ⇒ f_equal2 Z.mul Hx Hu |}.
Instance Z_mul_Semigroup : Semigroup Z Z_HasMult :=
  {| Semigroup.magma := Z_mul_Magma;
     Semigroup.is_associative := fun a b c ⇒ eq_sym (Z.mul_assoc a b c) |}.
Instance Z_mul_Monoid : Monoid Z Z_One Z_HasMult :=
  {| Monoid.semigroup := Z_mul_Semigroup;
     Monoid.is_left_identity := Z.mul_1_l; Monoid.is_right_identity := Z.mul_1_r |}.

Instance Z_Ring : Ring Z Z_Zero Z_HasAdd Z_HasNeg Z_One Z_HasMult :=
  {| Ring.abgroup := Z_AbGroup; Ring.mul_monoid := Z_mul_Monoid;
     Ring.is_left_distributive := Z.mul_add_distr_l;
     Ring.is_right_distributive := Z.mul_add_distr_r |}.

Instance Z_Integers : Integers Z Z_Zero Z_HasAdd Z_HasNeg Z_One Z_HasMult :=
  {| Integers.ring := Z_Ring;
     Integers.mul_commutative := Z.mul_comm;
     Integers.int_induction := fun P _ ⇒ Z_induction P |}.
