Predecessor and truncated subtraction on any naturals with recursion: the predecessor forgets one successor, subtraction iterates it. Both are spelled as notations on natrec, the way partial sums are, so no new operation classes are introduced.
Section Monus.
  Context {N : Type} `{Naturals N} {natrecN : HasNatRec N} {NR : NatRec N natrecN}.

  Local Notation "'pred' n" := (natrec N 0 (fun k _ ⇒ k) n) (at level 10, n at level 9).
  Local Notation "m ∸ j" := (natrec N m (fun _ acc ⇒ pred acc) j) (at level 50, left associativity).

  Lemma pred_step_proper : Proper (equ ==> equ ==> equ) (fun (k : N) (_ : N) ⇒ k).
  Proof. intros k k' Hk a a' Ha. exact Hk. Qed.

  Theorem pred_proper : Proper (equ ==> equ) (fun n : N ⇒ pred n).
  Proof. exact (natrec_proper N _ _ 0 _ pred_step_proper). Qed.

  Theorem pred_zero : pred (0 : N) == 0.
  Proof. exact (natrec_zero N _ _ 0 _). Qed.

  Theorem pred_succ : ∀ n : N, pred (succ n) == n.
  Proof. intros n. exact (natrec_succ N _ _ 0 _ n). Qed.

  Lemma monus_step_proper : Proper (equ ==> equ ==> equ) (fun (_ : N) (acc : N) ⇒ pred acc).
  Proof. intros k k' Hk a a' Ha. exact (pred_proper a a' Ha). Qed.

  Theorem monus_index_proper : ∀ m : N, Proper (equ ==> equ) (fun j : N ⇒ m ∸ j).
  Proof. intros m. exact (natrec_proper N _ _ m _ monus_step_proper). Qed.

  Theorem monus_zero_r : ∀ m : N, m ∸ (0 : N) == m.
  Proof. intros m. exact (natrec_zero N _ _ m _). Qed.

  Theorem monus_succ_r : ∀ m j : N, m ∸ succ j == pred (m ∸ j).
  Proof. intros m j. exact (natrec_succ N _ _ m _ j). Qed.

  Theorem monus_proper_l : ∀ (m m' : N), m == m' → ∀ j : N, m ∸ j == m' ∸ j.
  Proof.
    intros m m' Hm.
    apply (nat_induction (fun j ⇒ m ∸ j == m' ∸ j)).
    - intros j j' Hj. split; intros Heq.
      + apply (transitivity (symmetry (monus_index_proper m j j' Hj))).
        apply (transitivity Heq). exact (monus_index_proper m' j j' Hj).
      + apply (transitivity (monus_index_proper m j j' Hj)).
        apply (transitivity Heq). exact (symmetry (monus_index_proper m' j j' Hj)).
    - calc (m ∸ (0 : N)) == (m) { (monus_zero_r m) } == (m') { Hm } == (m' ∸ (0 : N)) { (monus_zero_r m') }.
    - intros j IH.
      calc (m ∸ succ j)
        == (pred (m ∸ j)) { (monus_succ_r m j) }
        == (pred (m' ∸ j)) { (pred_proper _ _ IH) }
        == (m' ∸ succ j) { (monus_succ_r m' j) }.
  Qed.

  Theorem zero_monus : ∀ j : N, (0 : N) ∸ j == 0.
  Proof.
    apply (nat_induction (fun j ⇒ (0 : N) ∸ j == 0)).
    - intros j j' Hj. split; intros Heq.
      + apply (transitivity (symmetry (monus_index_proper 0 j j' Hj))). exact Heq.
      + apply (transitivity (monus_index_proper 0 j j' Hj)). exact Heq.
    - exact (monus_zero_r 0).
    - intros j IH.
      calc ((0 : N) ∸ succ j)
        == (pred ((0 : N) ∸ j)) { (monus_succ_r 0 j) }
        == (pred (0 : N)) { (pred_proper _ _ IH) }
        == (0 : N) { pred_zero }.
  Qed.

  Theorem monus_succ_succ : ∀ m j : N, succ m ∸ succ j == m ∸ j.
  Proof.
    intros m.
    apply (nat_induction (fun j ⇒ succ m ∸ succ j == m ∸ j)).
    - intros j j' Hj. split; intros Heq.
      + apply (transitivity (symmetry (monus_index_proper (succ m) (succ j) (succ j') (succ_proper j j' Hj)))).
        apply (transitivity Heq). exact (monus_index_proper m j j' Hj).
      + apply (transitivity (monus_index_proper (succ m) (succ j) (succ j') (succ_proper j j' Hj))).
        apply (transitivity Heq). exact (symmetry (monus_index_proper m j j' Hj)).
    - calc (succ m ∸ succ (0 : N))
        == (pred (succ m ∸ (0 : N))) { (monus_succ_r (succ m) 0) }
        == (pred (succ m)) { (pred_proper _ _ (monus_zero_r (succ m))) }
        == (m) { (pred_succ m) }
        == (m ∸ (0 : N)) { (monus_zero_r m) }.
    - intros j IH.
      calc (succ m ∸ succ (succ j))
        == (pred (succ m ∸ succ j)) { (monus_succ_r (succ m) (succ j)) }
        == (pred (m ∸ j)) { (pred_proper _ _ IH) }
        == (m ∸ succ j) { (monus_succ_r m j) }.
  Qed.

  Theorem monus_self : ∀ m : N, m ∸ m == 0.
  Proof.
    apply (nat_induction (fun m ⇒ m ∸ m == 0)).
    - intros m m' Hm. split; intros Heq.
      + apply (transitivity (symmetry (monus_proper_l m m' Hm m'))).
        apply (transitivity (symmetry (monus_index_proper m m m' Hm))). exact Heq.
      + apply (transitivity (monus_index_proper m m m' Hm)).
        apply (transitivity (monus_proper_l m m' Hm m')). exact Heq.
    - exact (monus_zero_r 0).
    - intros m IH.
      calc (succ m ∸ succ m)
        == (m ∸ m) { (monus_succ_succ m m) }
        == (0 : N) { IH }.
  Qed.

  Theorem add_monus : ∀ m j : N, (m + j) ∸ j == m.
  Proof.
    intros m.
    apply (nat_induction (fun j ⇒ (m + j) ∸ j == m)).
    - intros j j' Hj. split; intros Heq.
      + apply (transitivity (symmetry (monus_proper_l (m + j) (m + j') (op_proper m m (reflexivity m) j j' Hj) j'))).
        apply (transitivity (symmetry (monus_index_proper (m + j) j j' Hj))). exact Heq.
      + apply (transitivity (monus_index_proper (m + j) j j' Hj)).
        apply (transitivity (monus_proper_l (m + j) (m + j') (op_proper m m (reflexivity m) j j' Hj) j')). exact Heq.
    - calc ((m + 0) ∸ (0 : N))
        == (m + 0) { (monus_zero_r (m + 0)) }
        == (m) { (Theory.Naturals.add_zero m) }.
    - intros j IH.
      calc ((m + succ j) ∸ succ j)
        == (succ (m + j) ∸ succ j) { (monus_proper_l (m + succ j) (succ (m + j)) (Theory.Naturals.add_succ m j) (succ j)) }
        == ((m + j) ∸ j) { (monus_succ_succ (m + j) j) }
        == (m) { IH }.
  Qed.
End Monus.
