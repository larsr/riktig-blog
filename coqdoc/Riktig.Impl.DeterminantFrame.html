The determinant through a frame: the pseudoscalar coefficient of the wedge of the family, in the order of the index.
Section DeterminantFrame.
  Context {F V A I N : Type} `{Frame F V A I N} {EI : @EquDec I _} {update : HasUpd I V} {U : Upd I V update}.

  #[export] Instance Frame_Det : HasDet F I V :=
    fun vs ⇒ top (natrec A 1 (fun k acc ⇒ acc ⋏ vs (elem k)) card).

  #[export] Instance Frame_Multilinear : Multilinear F I V Frame_Det.
  Proof.
    split; unfold det, Frame_Det.
    - intros vs vs' Hvs Hvs' Hvv. apply top_proper. exact (wedge_fam_ext vs vs' Hvs Hvs' Hvv).
    - intros vs j Hvs u v.
      apply (transitivity (top_proper _ _ (wedge_fam_upd_add vs j u v Hvs))).
      exact (Frame.top_additive _ _).
    - intros vs j Hvs a u.
      apply (transitivity (top_proper _ _ (wedge_fam_upd_smul vs j a u Hvs))).
      exact (Frame.top_homogeneous a _).
  Qed.

  #[export] Instance Frame_Alternating : Alternating F I V Frame_Det.
  Proof.
    intros vs i j Hvs Hij Hv. unfold det, Frame_Det.
    apply (transitivity (top_proper _ _ (wedge_fam_repeat vs i j Hvs Hij Hv))).
    apply (add_left_cancel (top (0 : A))).
    calc (top (0 : A) + top (0 : A))
      == (top ((0 : A) + 0)) { (Frame.top_additive (0 : A) 0) }
      == (top (0 : A)) { (top_proper _ _ (left_identity (action := add) (0 : A))) }
      == (top (0 : A) + 0) { (right_identity (action := add) (top (0 : A))) }.
  Qed.

  #[export] Instance Frame_Determinant : Determinant F I V basis Frame_Det.
  Proof.
    split; [ exact Frame_Multilinear | exact Frame_Alternating | ].
    exact (top_wedge_fam_basis (F := F) (V := V) (A := A) (I := I) (N := N) (basis := basis) (pseudoscalar_coefficient := pseudoscalar_coefficient)).
  Qed.
End DeterminantFrame.
