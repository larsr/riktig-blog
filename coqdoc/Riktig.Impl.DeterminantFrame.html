The determinant through a frame: the pseudoscalar coefficient of the wedge of the columns, in the order of the index.
Section DeterminantFrame.
  Context {F V A I N : Type} `{Frame F V A I N} {EI : @EquDec I _}.

  #[export] Instance Frame_Det : HasDet F I :=
    fun M ⇒ top (natrec A 1 (fun k acc ⇒
      acc ⋏ natrec V 0 (fun l acc' ⇒ acc' + M (elem l) (elem k) *: basis (elem l)) card) card).

  #[export] Instance Frame_Determinant : Determinant F I Frame_Det.
  Proof.
    split; unfold det, Frame_Det.
    - intros M M' HM HM' HMM'. apply top_proper. exact (wedge_cols_ext M M' HM HM' HMM').
    - intros M j u v HM Hu Hv.
      apply (transitivity (top_proper _ _ (wedge_cols_upd_add M j u v HM Hu Hv))).
      exact (Frame.top_additive _ _).
    - intros M j a u HM Hu.
      apply (transitivity (top_proper _ _ (wedge_cols_upd_smul M j a u HM Hu))).
      exact (Frame.top_homogeneous a _).
    - intros M j j' HM Hjj' Hcols.
      apply (transitivity (top_proper _ _ (wedge_cols_repeat M j j' HM Hjj' Hcols))).
      apply (add_left_cancel (top (0 : A))).
      calc (top (0 : A) + top (0 : A))
        == (top ((0 : A) + 0)) { (Frame.top_additive (0 : A) 0) }
        == (top (0 : A)) { (top_proper _ _ (left_identity (action := add) (0 : A))) }
        == (top (0 : A) + 0) { (right_identity (action := add) (top (0 : A))) }.
    - exact wedge_cols_id.
  Qed.
End DeterminantFrame.
