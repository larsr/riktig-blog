From Riktig.Structure Require Export Functor.

An algebra of an endofunctor F is an object t with an arrow F t ~> t that builds one level. It is initial when for every algebra F x ~> x there is exactly one arrow t ~> x carrying build to build. The datatypes are the initial algebras: build packs the constructors, cata is the recursion, and its uniqueness is induction.
Module InitialAlgebra.
  Class InitialAlgebra (O : Type) `{Category O} (F : O → O) {map : HasFmap O O F} (t : O)
      (build : HasBuild O F t) (fold : HasCata O F t) := {
    functor :: Functor O O F map;
    cata_proper :: ∀ x : O, Proper (equ ==> equ) (cata (F := F) (t := t) x);
    cata_build : ∀ (x : O) (h : F x ~> x),
      cata x h ∘ build == h ∘ fmap (cata x h);
    cata_unique : ∀ (x : O) (h : F x ~> x) (u : t ~> x),
      u ∘ build == h ∘ fmap u → u == cata x h;
  }.
End InitialAlgebra.
