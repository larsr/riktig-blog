Module Magnitude.
  Class Magnitude (A : Type) {Ae : EqRel A} {Az : Zero A} {Aa : HasAdd A}
      {An : HasNeg A} {Ao : One A} {Am : HasMult A} {AR : Ring A}
      {Ag : HasGrade A}
      (rev : A → A) {RV : Reversal A rev}
      (mag2 : A → A) :=
    mag2_def : ∀ a : A, mag2 a == ⟨ rev a × a ⟩_ 0.
  Hint Mode Magnitude ! - - - - - - - - ! - ! : typeclass_instances.
End Magnitude.
