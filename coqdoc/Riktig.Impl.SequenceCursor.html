A table read through a cursor: the sequence is the table with the position of the next element and the length; a step reads the entry at the position and advances it. The stream is compared by Coq's equality on the triple, so the table's own equality must be Coq's.
Definition Cursor (T N : Type) : Type := (T × (N × N))%type.

Section SequenceCursor.
  Context {C N T : Type} `{NatOrder N} {eqC : EqRel C} {EqvC : Equivalence (A := C) equ}
    {getT : HasGet T N C} {tabT : HasTabulate T N C} {TT : Table N C T getT tabT (eqT := eq)}
    {LD : LtDec N}.

  #[export] Instance Cursor_EqRel : EqRel (Cursor T N) := eq.
  #[export] Instance Cursor_HasStep : HasStep C (Cursor T N) :=
    fun s c s' ⇒ fst (snd s) < snd (snd s) ∧ c == get (fst s) (fst (snd s))
                  ∧ s' = (fst s, (succ (fst (snd s)), snd (snd s))).
  #[export] Instance Cursor_HasNext : HasNext C (Cursor T N) :=
    fun s ⇒ if dec (fst (snd s) < snd (snd s))
             then Some (get (fst s) (fst (snd s)), (fst s, (succ (fst (snd s)), snd (snd s))))
             else None.

  #[export] Instance Cursor_Sequence : Sequence C (Cursor T N) Cursor_HasStep.
  Proof.
    split.
    - exact EqvC.
    - exact eq_equivalence.
    - intros s s0 Hs c c0 Hc t t0 Ht. unfold equ, Cursor_EqRel in Hs, Ht. subst.
      unfold step, Cursor_HasStep. rewrite Hc. reflexivity.
    - intros s c c' s' s'' [_ [Hc Hs]] [_ [Hc' Hs']]. subst. split; [ | reflexivity ].
      transitivity (get (fst s) (fst (snd s))); [ exact Hc | symmetry; exact Hc' ].
  Qed.

  Lemma cursor_next_step : ∀ (s : Cursor T N) (c : C) (s' : Cursor T N),
    step s c s' ↔ ∃ c0 : C, next s = Some (c0, s') ∧ c0 == c.
  Proof.
    intros s c s'. unfold next, Cursor_HasNext, step, Cursor_HasStep.
    destruct (dec (P := fst (snd s) < snd (snd s))) as [Hlt | Hlt].
    - split.
      + intros [_ [Hc Hs]]. subst. ∃ (get (fst s) (fst (snd s))). split; [ reflexivity | symmetry; exact Hc ].
      + intros [c0 [Hn Hc]]. injection Hn as Hc0 Hs. subst. split; [ exact Hlt | split; [ symmetry; exact Hc | reflexivity ] ].
    - split; [ intros [Hlt' _]; destruct (Hlt Hlt') | intros [c0 [Hn _]]; discriminate Hn ].
  Qed.

How much is left to read, for the greedy repetitions.
  Definition cursor_size (of_nat : N → nat) (s : Cursor T N) : nat := of_nat (snd (snd s)) - of_nat (fst (snd s)).
End SequenceCursor.
