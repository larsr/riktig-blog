The lambda terms as a grammar writes them, as a Coq inductive: what Impl/NamedTermRecNamed needs, the recursion with its equations by computation.
Inductive named (K N : Type) : Type :=
  | NVar (x : K)
  | NNum (n : N)
  | NLam (x : K) (body : named K N)
  | NApp (f a : named K N).
Arguments NVar {K N} x.
Arguments NNum {K N} n.
Arguments NLam {K N} x body.
Arguments NApp {K N} f a.

Fixpoint named_fold {K N : Type} (B : Type) (v : K → B) (c : N → B) (l : K → B → B) (a : B → B → B)
    (t : named K N) : B :=
  match t with
  | NVar x ⇒ v x
  | NNum n ⇒ c n
  | NLam x b ⇒ l x (named_fold B v c l a b)
  | NApp f u ⇒ a (named_fold B v c l a f) (named_fold B v c l a u)
  end.
