From Coq Require Import FunctionalExtensionality.

Coq's types and functions as a category's data: pointwise equality of functions, and what the universal constructions of unit, Empty_set, pairs, sums and function types satisfy, in the shape the category structures ask for. Currying is unique only up to functional extensionality, the one axiom used here.
Definition fn_equ (X Y : Type) (f g : X → Y) : Prop := ∀ x : X, f x = g x.

Lemma fn_equ_refl : ∀ (X Y : Type) (f : X → Y), fn_equ X Y f f.
Proof. intros X Y f x. reflexivity. Qed.

Lemma fn_equ_sym : ∀ (X Y : Type) (f g : X → Y), fn_equ X Y f g → fn_equ X Y g f.
Proof. intros X Y f g Hfg x. symmetry. exact (Hfg x). Qed.

Lemma fn_equ_trans : ∀ (X Y : Type) (f g h : X → Y),
  fn_equ X Y f g → fn_equ X Y g h → fn_equ X Y f h.
Proof. intros X Y f g h Hfg Hgh x. rewrite (Hfg x). exact (Hgh x). Qed.

Lemma fn_comp_ext : ∀ (X Y Z : Type) (g g' : Y → Z) (f f' : X → Y),
  fn_equ Y Z g g' → fn_equ X Y f f' → fn_equ X Z (fun x ⇒ g (f x)) (fun x ⇒ g' (f' x)).
Proof. intros X Y Z g g' f f' Hg Hf x. rewrite (Hf x). exact (Hg (f' x)). Qed.

Lemma unit_unique : ∀ (X : Type) (f : X → unit), fn_equ X unit f (fun _ ⇒ tt).
Proof. intros X f x. destruct (f x). reflexivity. Qed.

Lemma empty_unique : ∀ (X : Type) (f : Empty_set → X),
  fn_equ Empty_set X f (fun e ⇒ match e with end).
Proof. intros X f e. destruct e. Qed.

Lemma pair_ext : ∀ (X A B : Type) (f f' : X → A) (g g' : X → B),
  fn_equ X A f f' → fn_equ X B g g' → fn_equ X (A × B) (fun x ⇒ (f x, g x)) (fun x ⇒ (f' x, g' x)).
Proof. intros X A B f f' g g' Hf Hg x. rewrite (Hf x), (Hg x). reflexivity. Qed.

Lemma pair_unique : ∀ (X A B : Type) (f : X → A) (g : X → B) (h : X → A × B),
  fn_equ X A (fun x ⇒ fst (h x)) f → fn_equ X B (fun x ⇒ snd (h x)) g →
  fn_equ X (A × B) h (fun x ⇒ (f x, g x)).
Proof.
  intros X A B f g h Hf Hg x. rewrite <- (Hf x), <- (Hg x). apply surjective_pairing.
Qed.

Definition case_fn (X A B : Type) (f : A → X) (g : B → X) (s : A + B) : X :=
  match s with inl a ⇒ f a | inr b ⇒ g b end.

Lemma case_ext : ∀ (X A B : Type) (f f' : A → X) (g g' : B → X),
  fn_equ A X f f' → fn_equ B X g g' → fn_equ (A + B) X (case_fn X A B f g) (case_fn X A B f' g').
Proof. intros X A B f f' g g' Hf Hg [a | b]; [ exact (Hf a) | exact (Hg b) ]. Qed.

Lemma case_unique : ∀ (X A B : Type) (f : A → X) (g : B → X) (h : A + B → X),
  fn_equ A X (fun a ⇒ h (inl a)) f → fn_equ B X (fun b ⇒ h (inr b)) g →
  fn_equ (A + B) X h (case_fn X A B f g).
Proof. intros X A B f g h Hf Hg [a | b]; [ exact (Hf a) | exact (Hg b) ]. Qed.

Lemma curry_ext : ∀ (X A B : Type) (h h' : X × A → B),
  fn_equ (X × A) B h h' → fn_equ X (A → B) (fun x a ⇒ h (x, a)) (fun x a ⇒ h' (x, a)).
Proof. intros X A B h h' Hh x. apply functional_extensionality. intros a. exact (Hh (x, a)). Qed.

Lemma eval_curry_fn : ∀ (X A B : Type) (h : X × A → B),
  fn_equ (X × A) B (fun p ⇒ h (fst p, snd p)) h.
Proof. intros X A B h p. rewrite <- surjective_pairing. reflexivity. Qed.

Lemma curry_unique_fn : ∀ (X A B : Type) (h : X × A → B) (k : X → A → B),
  fn_equ (X × A) B (fun p ⇒ k (fst p) (snd p)) h → fn_equ X (A → B) k (fun x a ⇒ h (x, a)).
Proof. intros X A B h k Hk x. apply functional_extensionality. intros a. exact (Hk (x, a)). Qed.
