A pair compared componentwise: what a fold that carries something along answers with.
#[export] Instance EqPair (A B : Type) {eA : EqRel A} {eB : EqRel B}
  : EqRel (A × B) := fun p q ⇒ fst p == fst q ∧ snd p == snd q.

#[export] Instance EqPair_Equivalence (A B : Type) {eA : EqRel A} {eB : EqRel B}
  {EA : Equivalence (@equ A eA)} {EB : Equivalence (@equ B eB)}
  : Equivalence (@equ (A × B) (EqPair A B)).
Proof.
  split.
  - intro p. split; reflexivity.
  - intros p q [H1 H2]. split; symmetry; assumption.
  - intros p q r [H1 H2] [H3 H4]. split.
    + calc (fst p) == (fst q) { H1 } == (fst r) { H3 }.
    + calc (snd p) == (snd q) { H2 } == (snd r) { H4 }.
Qed.
