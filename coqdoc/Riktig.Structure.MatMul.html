From Coq Require Export Morphisms.
From Riktig Require Export Structure.BigOp Theory.Pointwise.
From Riktig Require Import Lib.Calc.


Module MatMul.
  Class MatMul (n : nat) (J F : Type) {eJ : EqRel J} {dJ : EquDec J}
      {FJ : FinIndex n J} {eF : EqRel F} {zF : Zero F} {aF : HasAdd F}
      {mF : HasMult F}
      (bigop_list : ∀ {J' : Type}, (J' → F) → list J' → F)
      {BL : BigOp F (@bigop_list)}
      (mat_mul : ∀ {I K : Type},
         (I → J → F) → (J → K → F) → I → K → F) :=
    mat_mul_def : ∀ {I K : Type} (A : I → J → F) (B : J → K → F)
                        (i : I) (k : K),
      mat_mul A B i k == bigop_list (fun j ⇒ A i j × B j k) enum.
  Hint Mode MatMul - ! ! - - - - - - - - - - : typeclass_instances.
End MatMul.
Export MatMul.
