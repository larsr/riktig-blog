Module KleisliInverse.
  Class KleisliInverse {M : Type → Type} `{Monad M} {A B : Type}
      (f : A → M B) (g : B → M A) := {
    from_to : ∀ a : A, bind (f a) g == ret a;
    to_from : ∀ b : B, bind (g b) f == ret b;
  }.
End KleisliInverse.
Notation KleisliInverse := KleisliInverse.KleisliInverse.

Hint Mode KleisliInverse.KleisliInverse - - - - - - - ! ! : typeclass_instances.

Class KleisliIso (M : Type → Type) `{Monad M} (A B : Type) := {
  ki_to : A → M B;
  ki_from : B → M A;
  ki_inverse :: KleisliInverse ki_to ki_from;
}.

Arguments ki_to {M Me Mb Mr H A B}.
Arguments ki_from {M Me Mb Mr H A B}.
Arguments ki_inverse {M Me Mb Mr H A B}.

Section KleisliIsoLaws.
  Context {M : Type → Type} `{Monad M} {A B : Type}.

  Definition ki_fromto (k : KleisliIso M A B)
    : ∀ a : A, bind (ki_to k a) (ki_from k) == ret a :=
    KleisliInverse.from_to (f := ki_to k) (g := ki_from k).

  Definition ki_tofrom (k : KleisliIso M A B)
    : ∀ b : B, bind (ki_from k b) (ki_to k) == ret b :=
    KleisliInverse.to_from (f := ki_to k) (g := ki_from k).

  Definition kleisli_iso_from_laws (to : A → M B) (from : B → M A)
      (fromto : ∀ a : A, bind (to a) from == ret a)
      (tofrom : ∀ b : B, bind (from b) to == ret b)
    : KleisliIso M A B :=
    {| ki_to := to;
       ki_from := from;
       ki_inverse :=
         {| KleisliInverse.from_to := fromto;
            KleisliInverse.to_from := tofrom |} |}.
End KleisliIsoLaws.

Section ChannelInsertion.
  Context {M : Type → Type} `{Monad M} {A B B' C : Type}.
  Context (prog1 : A → M B) (prog2 : B → M C) (codec : KleisliIso M B B').

  Lemma ki_insert_pointwise : ∀ b : B,
    bind (ki_to codec b) (fun b' ⇒ bind (ki_from codec b') prog2) == prog2 b.
  Proof.
    intro b.
    calc (bind (ki_to codec b) (fun b' ⇒ bind (ki_from codec b') prog2))
      == (bind (bind (ki_to codec b) (ki_from codec)) prog2)
      == (bind (ret b) prog2)
      == (prog2 b).
    - symmetry. apply bind_associative.
    - apply bind_proper.
      + exact (ki_fromto codec b).
      + intro x. reflexivity.
    - apply bind_left_identity.
  Qed.

  Theorem channel_insertion : ∀ a : A,
    (prog1 >=> prog2) a == (prog1 >=> ki_to codec >=> ki_from codec >=> prog2) a.
  Proof.
    intro a.
    apply bind_proper.
    - reflexivity.
    - intro b. symmetry. exact (ki_insert_pointwise b).
  Qed.

End ChannelInsertion.

From Coq Require Import Cantor.
From Riktig Require Import Impl.OptionMonad.

Lemma nat_pair_fromto : ∀ p : nat × nat,
  bind (Some (to_nat p)) (fun n ⇒ Some (of_nat n)) == Some p.
Proof.
  intro p.
  unfold bind, Option_HasBind, equ, Option_EqRel.
  rewrite cancel_of_to.
  reflexivity.
Qed.

Lemma nat_pair_tofrom : ∀ n : nat,
  bind (Some (of_nat n)) (fun p : nat × nat ⇒ Some (to_nat p)) == Some n.
Proof.
  intro n.
  unfold bind, Option_HasBind, equ, Option_EqRel.
  rewrite cancel_to_of.
  reflexivity.
Qed.

Definition nat_pair_codec : KleisliIso option (nat × nat) nat :=
  kleisli_iso_from_laws
    (fun p : nat × nat ⇒ Some (to_nat p))
    (fun n : nat ⇒ Some (of_nat n))
    nat_pair_fromto
    nat_pair_tofrom.

Definition split_n (n : nat) : option (nat × nat) := Some (n, n).
Definition sum_pair (p : nat × nat) : option nat := Some (Nat.add (fst p) (snd p)).

Example channel_insertion_nat_pair_check :=
  channel_insertion split_n sum_pair nat_pair_codec.
