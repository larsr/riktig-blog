From Coq.Vectors Require Import Fin Vector.
Export Vector.VectorNotations.
From Riktig.Structure Require Export Table.
From Riktig.Lib Require Import Coq.Vector.

A vector of length n is a table over Fin.t n: its length is its index's count, so the index is read off the vector and needs no annotation. Two vectors are equal when they agree at every index.
Section TableVector.
  Context {A : Type} {eqA : EqRel A} {EA : Equivalence eqA} {n : nat}.

  #[export] Instance VecTable_EqRel : EqRel (Vector.t A n) :=
    fun s t ⇒ ∀ i : Fin.t n, Vector.nth s i == Vector.nth t i.
  #[export] Instance VecTable_HasGet : HasGet (Vector.t A n) (Fin.t n) A := Vector.nth.
  #[export] Instance VecTable_HasTabulate : HasTabulate (Vector.t A n) (Fin.t n) A :=
    vec_tabulate n.

  #[export] Instance VecTable_Table :
    Table (Fin.t n) A (Vector.t A n) VecTable_HasGet VecTable_HasTabulate (eqI := eq).
  Proof.
    split.
    - split.
      + intros s i. reflexivity.
      + intros s t Hst i. symmetry. exact (Hst i).
      + intros s t u Hst Htu i. transitivity (Vector.nth t i); [ exact (Hst i) | exact (Htu i) ].
    - intros s t Hst i j Hij. rewrite Hij. exact (Hst j).
    - intros f g Hfg i. unfold tabulate, VecTable_HasTabulate.
      rewrite !nth_vec_tabulate. exact (Hfg i).
    - intros f i. unfold get, VecTable_HasGet, tabulate, VecTable_HasTabulate.
      rewrite nth_vec_tabulate. reflexivity.
    - intros t i. unfold get, VecTable_HasGet, tabulate, VecTable_HasTabulate.
      rewrite nth_vec_tabulate. reflexivity.
  Qed.
End TableVector.
