Library Riktig.Theory.Stlc

From Riktig.Structure Require Export Stlc.
From Riktig.Theory Require Export Signature.
From Riktig.Impl Require Import CategoryFun SignaturePoly.

Induction, from the one theorem for every signature: nothing term-specific remains to prove.
Section StlcInduction.
  Context {N Ty T : Type}
    {tb : HasBuild Type (Stlc.shape (fun T0 : Type ⇒ T0) N Ty) T}
    {tf : HasCata Type (Stlc.shape (fun T0 : Type ⇒ T0) N Ty) T}
    `{!Stlc Type (fun T0 : Type ⇒ T0) N Ty T tb tf}.

  Theorem stlc_induction : ∀ P : T → Prop,
    (∀ n : N, P (tb (inl (inl n)))) →
    (∀ (A : Ty) (b : T), P b → P (tb (inl (inr (A, b))))) →
    (∀ f a : T, P f → P a → P (tb (inr (f, a)))) →
    ∀ t : T, P t.
  Proof.
    intros P Hvar Hlam Happ.
    apply (datatype_induction (fun T0 : Type ⇒ T0) (stlcS (S := sig Type) N Ty) (T := T)
      (tb := tb) (tf := tf) (IA := Stlc.initial)).
    intros s [s' Es]. destruct s' as [[n | [A [b pb]]] | [[f pf] [a pa]]]; cbn in Es; subst; cbn.
    - exact (Hvar n).
    - exact (Hlam A b pb).
    - exact (Happ f a pf pa).
  Qed.
End StlcInduction.
