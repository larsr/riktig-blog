From Riktig.Theory Require Export Stlc StlcType.
From Riktig.Impl Require Export CategoryFun SignaturePoly.

The constructors of any simple-type algebra and any typed-term algebra over Coq's types, as the operations the notation classes name.
Section TypeOps.
  Context {B Ty : Type}
    {yb : HasBuild Type (StlcType.shape (fun T0 : Type ⇒ T0) B) Ty}
    {yf : HasCata Type (StlcType.shape (fun T0 : Type ⇒ T0) B) Ty}
    `{!StlcType Type (fun T0 : Type ⇒ T0) B Ty yb yf}.

  #[export] Instance StlcType_HasBase : HasBase B Ty := fun b ⇒ yb (inl b).
  #[export] Instance StlcType_HasArrow : HasArrow Ty := fun a c ⇒ yb (inr (a, c)).
End TypeOps.

Section TermOps.
  Context {N Ty T : Type}
    {tb : HasBuild Type (Stlc.shape (fun T0 : Type ⇒ T0) N Ty) T}
    {tf : HasCata Type (Stlc.shape (fun T0 : Type ⇒ T0) N Ty) T}
    `{!Stlc Type (fun T0 : Type ⇒ T0) N Ty T tb tf}.

  #[export] Instance Stlc_HasVar : HasVar N T := fun n ⇒ tb (inl (inl n)).
  #[export] Instance Stlc_HasTyLam : HasTyLam Ty T := fun A b ⇒ tb (inl (inr (A, b))).
  #[export] Instance Stlc_HasApp : HasApp T := fun f a ⇒ tb (inr (f, a)).
End TermOps.
