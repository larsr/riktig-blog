The zero space has the zero inner product.
Section UnitInner.
  Context {F : Type} `{Field F}.

  #[export] Instance Unit_HasInner : HasInner F unit := fun _ _ ⇒ 0.

  #[export] Instance Unit_InnerProductSpace : InnerProductSpace F unit :=
    {| InnerProductSpace.inner_proper := fun _ _ _ _ _ _ ⇒ reflexivity 0;
       InnerProductSpace.inner_additive := fun _ _ _ ⇒ symmetry (left_identity (action := add) (0 : F));
       InnerProductSpace.inner_homogeneous := fun _ a _ ⇒ symmetry (mul_zero a);
       InnerProductSpace.is_commutative := fun _ _ ⇒ reflexivity 0 |}.
End UnitInner.
