From Riktig.Structure Require Export Order.

The reals: an ordered field in which every inhabited set with an upper bound has a least upper bound.
Module Reals.
  Class Reals (R : Type) `{OrderedField R} := {
    complete : ∀ P : Pred R,
      (∃ x : R, P x) →
      (∃ b : R, ∀ x : R, P x → x ≤ b) →
      ∃ s : R,
        (∀ x : R, P x → x ≤ s) ∧
        (∀ b : R, (∀ x : R, P x → x ≤ b) → s ≤ b);
  }.
  Hint Mode Reals ! - - - - - - - - - - - : typeclass_instances.
End Reals.
Export Reals.
