The convolution identity behind exp (x + y) == exp x × exp y: the Cauchy-product term of the two exponential series at k is the k-th term of the series at x + y. No binomial coefficients: weight the sum by k + 1 = j + (k + 1 - j), shift each half once, and both recurrences close.

  Lemma cast_succ_cancel : ∀ (x : R) (k : N),
    cast (of_nat (succ k)) × (cast (inv (of_nat (succ k))) × x) == x.
  Proof.
    intros x k.
    apply (transitivity (y := (cast (of_nat (succ k)) × cast (inv (of_nat (succ k)))) × x)).
    { exact (symmetry (associative (op := mult) (action := mult) (cast (of_nat (succ k))) (cast (inv (of_nat (succ k)))) x)). }
    apply (transitivity (y := cast (of_nat (succ k) × inv (of_nat (succ k))) × x)).
    { apply op_proper; [ symmetry; exact (additive (f := cast) (opA := mult) (of_nat (succ k)) (inv (of_nat (succ k)))) | reflexivity ]. }
    apply (transitivity (y := cast 1 × x)).
    { apply op_proper; [ apply hom_proper; exact (mul_inv_cancel (of_nat (succ k)) (of_nat_succ_ne k)) | reflexivity ]. }
    apply (transitivity (y := (1 : R) × x)).
    { apply op_proper; [ exact hom_one | reflexivity ]. }
    exact (left_identity (action := mult) x).
  Qed.

The weighted recurrence: (k+1) t_{k+1} = t_k x.
  Lemma exp_term_weighted : ∀ (x : R) (k : N),
    cast (of_nat (succ k)) × exp_term x (succ k) == exp_term x k × x.
  Proof.
    intros x k.
    calc (cast (of_nat (succ k)) × exp_term x (succ k))
      == (cast (of_nat (succ k)) × (exp_term x k × (cast (inv (of_nat (succ k))) × x)))
           { (op_proper _ _ (reflexivity _) _ _ (exp_term_succ x k)) }
      == ((cast (of_nat (succ k)) × exp_term x k) × (cast (inv (of_nat (succ k))) × x))
           { (associative (op := mult) (action := mult) _ _ _) }
      == ((exp_term x k × cast (of_nat (succ k))) × (cast (inv (of_nat (succ k))) × x))
           { (op_proper _ _ (commutative (op := mult) _ _) _ _ (reflexivity _)) }
      == (exp_term x k × (cast (of_nat (succ k)) × (cast (inv (of_nat (succ k))) × x)))
           { (symmetry (associative (op := mult) (action := mult) _ _ _)) }
      == (exp_term x k × x)
           { (op_proper _ _ (reflexivity _) _ _ (cast_succ_cancel x k)) }.
  Qed.

  Lemma exp_term_prod_proper : ∀ (x y : R) (k : N),
    Proper (equ ==> equ) (fun j : N ⇒ exp_term x j × exp_term y (k ∸ j)).
  Proof.
    intros x y k j j' Hj. apply op_proper.
    - exact (exp_term_proper x j j' Hj).
    - exact (exp_term_proper y _ _ (monus_index_proper k j j' Hj)).
  Qed.

  Lemma conv_proper : ∀ x y : R, Proper (equ ==> equ) (fun k : N ⇒ conv x y k).
  Proof.
    intros x y k k' Hk.
    apply (transitivity (y := psum (fun j ⇒ exp_term x j × exp_term y (k' ∸ j)) (succ k))).
    - apply psum_ext; [ exact (exp_term_prod_proper x y k) | exact (exp_term_prod_proper x y k') | ].
      intros j. apply op_proper; [ reflexivity | ].
      exact (exp_term_proper y _ _ (monus_proper_l k k' Hk j)).
    - exact (psum_proper _ (exp_term_prod_proper x y k') (succ k) (succ k') (succ_proper k k' Hk)).
  Qed.

The base case: the convolution at zero is one.
  Lemma conv_zero : ∀ x y : R, conv x y (0 : N) == exp_term (x + y) (0 : N).
  Proof.
    intros x y.
    calc (conv x y (0 : N))
      == (psum (fun j ⇒ exp_term x j × exp_term y ((0 : N) ∸ j)) 0
            + exp_term x (0 : N) × exp_term y ((0 : N) ∸ (0 : N))) { (psum_succ _ (0 : N)) }
      == ((0 : R) + exp_term x (0 : N) × exp_term y ((0 : N) ∸ (0 : N))) { (psum_zero _) }
      == (exp_term x (0 : N) × exp_term y ((0 : N) ∸ (0 : N))) { (left_identity (action := add) _) }
      == (exp_term x (0 : N) × exp_term y (0 : N))
           { (op_proper _ _ (reflexivity _) _ _ (exp_term_proper y _ _ (monus_zero_r (0 : N)))) }
      == ((1 : R) × exp_term y (0 : N)) { (op_proper _ _ (exp_term_zero x) _ _ (reflexivity _)) }
      == (exp_term y (0 : N)) { (left_identity (action := mult) _) }
      == (exp_term (x + y) (0 : N)) { (transitivity (exp_term_zero y) (symmetry (exp_term_zero (x + y)))) }.
  Qed.
