A frame on a geometric algebra over a field in which 2 is invertible: an ordered basis of the vectors over a finite index, and a linear functional reading the coefficient of the pseudoscalar, the wedge of the basis in order, which it sends to 1.
Module Frame.
  Class Frame (F V A I N : Type) `{GradeInvolution F V A} `{Finite I N}
      {natrecN : HasNatRec N} {NR : NatRec N natrecN}
      (basis : I → V) (pseudoscalar_coefficient : HasTop F A) := {
    char_not_two : (1 + 1 : F) =/= 0;
    basis_proper :: Proper (equ ==> equ) basis;
    top_proper :: Proper (equ ==> equ) (@top F A pseudoscalar_coefficient);
    top_additive :: Additive A F add add top;
    top_homogeneous :: Homogeneous F A F mult (fun a ⇒ a) top;
    top_pseudoscalar :
      top (natrec A 1 (fun k acc ⇒
             inv (1 + 1) *: (acc × vec (basis (elem k)) + vec (basis (elem k)) × involute acc)) card) == 1;
  }.
  Hint Mode Frame ! - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - : typeclass_instances.
End Frame.
