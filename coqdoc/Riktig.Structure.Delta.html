delta i j is the Kronecker delta: one where the indices agree, zero where they differ.

Module Delta.
  Class Delta (I R : Type) {eI : EqRel I}
      {Re : EqRel R} {Rz : Zero R} {Ro : One R} (delta : I → I → R) := {
    is_delta_at_eq : ∀ i j : I, i == j → delta i j == (1 : R);
    is_delta_at_neq : ∀ i j : I, ¬ (i == j) → delta i j == (0 : R);
  }.
  Hint Mode Delta ! ! - - - - - : typeclass_instances.
End Delta.
