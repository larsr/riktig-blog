Library Riktig.Theory.Coproducts

From Riktig.Structure Require Export Coproducts.
From Riktig.Theory Require Export Category.

Section CoproductsTheory.
  Context {O : Type} `{Coproducts O}.

  Theorem comp_desc : ∀ (x y a b : O) (f : a ~> x) (g : b ~> x) (h : x ~> y),
    h ∘ desc f g == desc (h ∘ f) (h ∘ g).
  Proof.
    intros x y a b f g h. apply Coproducts.desc_unique.
    - calc (h ∘ desc f g ∘ inj1)
        == (h ∘ (desc f g ∘ inj1)) { Category.comp_assoc }
        == (h ∘ f) { Coproducts.desc_inj1 }.
    - calc (h ∘ desc f g ∘ inj2)
        == (h ∘ (desc f g ∘ inj2)) { Category.comp_assoc }
        == (h ∘ g) { Coproducts.desc_inj2 }.
  Qed.

  Theorem desc_inj : ∀ a b : O, desc inj1 inj2 == cat_id (x := a ⨿ b).
  Proof.
    intros a b. symmetry. apply Coproducts.desc_unique; apply left_identity.
  Qed.

  Theorem sum_id : ∀ a b : O, (cat_id ⨿⨿ cat_id) == cat_id (x := a ⨿ b).
  Proof.
    intros a b. symmetry.
    apply Coproducts.desc_unique; etransitivity; [ apply left_identity | symmetry; apply right_identity
                                      | apply left_identity | symmetry; apply right_identity ].
  Qed.

  Theorem sum_comp : ∀ (a b c d e e' : O) (f : c ~> e) (g : d ~> e') (h : a ~> c) (k : b ~> d),
    (f ⨿⨿ g) ∘ (h ⨿⨿ k) == (f ∘ h) ⨿⨿ (g ∘ k).
  Proof.
    intros a b c d e e' f g h k. apply Coproducts.desc_unique.
    - rewrite <- (Category.comp_assoc _ _ _ _ _ _ inj1), (Coproducts.desc_inj1 (O := O)), Category.comp_assoc, (Coproducts.desc_inj1 (O := O)).
      symmetry. apply Category.comp_assoc.
    - rewrite <- (Category.comp_assoc _ _ _ _ _ _ inj2), (Coproducts.desc_inj2 (O := O)), Category.comp_assoc, (Coproducts.desc_inj2 (O := O)).
      symmetry. apply Category.comp_assoc.
  Qed.
End CoproductsTheory.
