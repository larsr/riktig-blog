Library Riktig.Interface.LinearMap

From Coq Require Import Morphisms.
From Riktig.Interface Require Export VectorSpace Linear.

Class LinearMap (F U V : Type)
    {eU : EqRel U} {aU : HasAdd U} {sU : HasScale F U}
    {eV : EqRel V} {aV : HasAdd V} {sV : HasScale F V} := {
  lm_fun : U → V;
  lm_linear :: Linear F U V lm_fun;
}.

Arguments lm_fun {F U V eU aU sU eV aV sV}.
Coercion lm_fun : LinearMap >-> Funclass.
Arguments lm_linear {F U V eU aU sU eV aV sV}.

Section LinearMapLaws.
  Context {F U V : Type}
    {eU : EqRel U} {aU : HasAdd U} {sU : HasScale F U}
    {eV : EqRel V} {aV : HasAdd V} {sV : HasScale F V}.

  Definition linear_map_proper (f : LinearMap F U V)
    : Proper (equ ==> equ) (lm_fun f) := proper (f := lm_fun f).

  Definition linear_map_additive (f : LinearMap F U V)
    : ∀ u v : U, f (u + v) == f u + f v :=
    additive (f := lm_fun f).

  Definition linear_map_scale (f : LinearMap F U V)
    : ∀ (a : F) (u : U), f (a *: u) == a *: f u :=
    homogeneous (f := lm_fun f).

  Definition linear_map_from_laws (f : U → V)
      (f_proper : Proper (equ ==> equ) f)
      (f_additive : ∀ u v : U, f (u + v) == f u + f v)
      (f_scale : ∀ (a : F) (u : U), f (a *: u) == a *: f u)
    : LinearMap F U V :=
    {| lm_fun := f;
       lm_linear :=
         {| Linear.proper := f_proper;
            Linear.is_additive := f_additive;
            Linear.is_homogeneous := f_scale |} |}.
End LinearMapLaws.

Section LinearMapEqu.
  Context {F U V : Type} `{FF : Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  #[export] Instance linear_map_equ : EqRel (LinearMap F U V) :=
    fun f g ⇒ ∀ u : U, f u == g u.

  #[export] Instance linear_map_equ_Equivalence : Equivalence (@equ (LinearMap F U V) linear_map_equ).
  Proof.
    apply Build_Equivalence.
    - intros f u. reflexivity.
    - intros f g Hfg u. symmetry. apply Hfg.
    - intros f g h Hfg Hgh u. rewrite (Hfg u). apply Hgh.
  Qed.

End LinearMapEqu.
