Record LinearMap (F U V : Type) `{FF : Field.Field F}
    {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}
    {VU : VectorSpace F U}
    {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}
    {VV : VectorSpace F V} := {
  lm_fun :> U → V;
  #[global] linear_map_proper :: Proper (equ ==> equ) lm_fun;
  linear_map_additive : ∀ u v : U, lm_fun (u + v) == lm_fun u + lm_fun v;
  linear_map_scale : ∀ (a : F) (u : U), lm_fun (a *: u) == a *: lm_fun u
}.

Arguments lm_fun {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.
Arguments linear_map_proper {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.
Arguments linear_map_additive {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.
Arguments linear_map_scale {F U V Fe Fz Fa Fn Fo Fm Fi FF eU zU aU nU sU VU eV zV aV nV sV VV}.


Section LinearMapEqu.
  Context {F U V : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  #[global] Instance linear_map_equ : EqRel (LinearMap F U V) :=
    fun f g ⇒ ∀ u : U, f u == g u.

  #[global] Instance linear_map_equ_Equivalence : Equivalence (@equ (LinearMap F U V) linear_map_equ).
  Proof.
    apply Build_Equivalence.
    - intros f u. reflexivity.
    - intros f g Hfg u. symmetry. apply Hfg.
    - intros f g h Hfg Hgh u. rewrite (Hfg u). apply Hgh.
  Qed.

End LinearMapEqu.


Section LinearMapTheory.
  Context {F U V : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context (f : LinearMap F U V).

  Theorem linear_map_zero : f 0 == 0.
  Proof.
    exact (hom_zero (lm_fun f) (linear_map_additive f)).
  Qed.

  Theorem linear_map_neg : ∀ u : U, f (- u) == - f u.
  Proof.
    intros u.
    exact (hom_neg (lm_fun f) (linear_map_additive f) u).
  Qed.

End LinearMapTheory.


Section LinearMapId.
  Context {F U : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.

  Lemma linear_map_id_proper : Proper (equ ==> equ) (fun u : U ⇒ u).
  Proof.
    intros u v Huv.
    exact Huv.
  Qed.

  Lemma linear_map_id_add : ∀ u v : U, u + v == u + v.
  Proof.
    intros u v.
    reflexivity.
  Qed.

  Lemma linear_map_id_scale : ∀ (a : F) (u : U), a *: u == a *: u.
  Proof.
    intros a u.
    reflexivity.
  Qed.

  Definition linear_map_id : LinearMap F U U :=
    {| lm_fun := (fun u : U ⇒ u);
       linear_map_proper := linear_map_id_proper;
       linear_map_additive := linear_map_id_add;
       linear_map_scale := linear_map_id_scale |}.

End LinearMapId.


Section LinearMapComp.
  Context {F U V W : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.
  Context {eW : EqRel W} {zW : Zero W} {aW : HasAdd W} {nW : HasNeg W} {sW : HasScale F W}.
  Context {VW : VectorSpace F W}.
  Context (g : LinearMap F V W) (f : LinearMap F U V).

  Lemma linear_map_comp_proper : Proper (equ ==> equ) (fun u : U ⇒ g (f u)).
  Proof.
    intros u v Huv.
    apply (linear_map_proper g), (linear_map_proper f).
    exact Huv.
  Qed.

  Lemma linear_map_comp_add : ∀ u v : U, g (f (u + v)) == g (f u) + g (f v).
  Proof.
    intros u v.
    rewrite (linear_map_proper g _ _ (linear_map_additive f u v)).
    exact (linear_map_additive g (f u) (f v)).
  Qed.

  Lemma linear_map_comp_scale : ∀ (a : F) (u : U), g (f (a *: u)) == a *: g (f u).
  Proof.
    intros a u.
    rewrite (linear_map_proper g _ _ (linear_map_scale f a u)).
    exact (linear_map_scale g a (f u)).
  Qed.

  Definition linear_map_comp : LinearMap F U W :=
    {| lm_fun := (fun u : U ⇒ g (f u));
       linear_map_proper := linear_map_comp_proper;
       linear_map_additive := linear_map_comp_add;
       linear_map_scale := linear_map_comp_scale |}.

End LinearMapComp.


Section LinearMapConst0.
  Context {F U V : Type} `{FF : Field.Field F}.
  Context {eU : EqRel U} {zU : Zero U} {aU : HasAdd U} {nU : HasNeg U} {sU : HasScale F U}.
  Context {VU : VectorSpace F U}.
  Context {eV : EqRel V} {zV : Zero V} {aV : HasAdd V} {nV : HasNeg V} {sV : HasScale F V}.
  Context {VV : VectorSpace F V}.

  Lemma linear_map_const0_proper : Proper (equ ==> equ) (fun _ : U ⇒ (0 : V)).
  Proof.
    intros u v Huv.
    reflexivity.
  Qed.

  Lemma linear_map_const0_add : ∀ u v : U, (0 : V) == 0 + 0.
  Proof.
    intros u v.
    symmetry.
    exact (Monoid.left_identity 0).
  Qed.

  Lemma linear_map_const0_scale : ∀ (a : F) (u : U), (0 : V) == a *: 0.
  Proof.
    intros a u.
    symmetry.
    exact (scale_zero a).
  Qed.

  Definition linear_map_const0 : LinearMap F U V :=
    {| lm_fun := (fun _ : U ⇒ 0);
       linear_map_proper := linear_map_const0_proper;
       linear_map_additive := linear_map_const0_add;
       linear_map_scale := linear_map_const0_scale |}.

End LinearMapConst0.

Example linear_map_check (F U : Type) `{FF : Field.Field F} `{VU : VectorSpace.VectorSpace F U} :
  LinearMap F U U := linear_map_comp linear_map_id linear_map_id.
