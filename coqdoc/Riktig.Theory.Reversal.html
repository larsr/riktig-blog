A reversal fixes zero and one and commutes with negation.
Section ReversalTheory.
  Context {A : Type} `{Reversal A}.

  Theorem reverse_zero : reverse (0 : A) == 0.
  Proof. exact (additive_zero reverse). Qed.

  Theorem reverse_neg : ∀ x : A, reverse (- x) == - reverse x.
  Proof. intros x. exact (additive_neg reverse x). Qed.

  Theorem reverse_one : reverse (1 : A) == 1.
  Proof.
    symmetry.
    calc (1 : A)
      == (reverse (reverse (1 : A))) { (symmetry (reverse_involutive (1 : A))) }
      == (reverse (reverse (1 : A) × (1 : A))) { (reverse_proper _ _ (symmetry (right_identity (action := mult) (reverse (1 : A))))) }
      == (reverse (1 : A) × reverse (reverse (1 : A))) { (reverse_anti (reverse (1 : A)) (1 : A)) }
      == (reverse (1 : A) × (1 : A)) { (op_proper _ _ (reflexivity _) _ _ (reverse_involutive (1 : A))) }
      == (reverse (1 : A)) { (right_identity (action := mult) (reverse (1 : A))) }.
  Qed.
End ReversalTheory.
