Coq's constructive Cauchy reals, in the shape the constructive reals structure asks for: the rationals' strict order as "at most and not equal", approximation by the sequence at the precision's denominator, and completeness over a rational precision rather than a positive one.
A strict inequality x < y between Cauchy reals is an index n at which the sequences are further apart than twice 2^n. Coq proves all such inequalities opaquely, so the index is not computable and no comparison built on it evaluates. The inequalities below compute their index: from an index below which the gap exceeds a rational margin, the witness is where twice the power of two drops below the margin.
Lemma Q_unit_pos : ∀ p : positive, (0 ≤ 1 # p)%Q ∧ ¬ (0 == 1 # p)%Q.
Proof. intros p. apply Qle_ne_of_lt. unfold Qlt. simpl. lia. Qed.

Lemma inject_Q_proper : ∀ q r : Q, (q == r)%Q → inject_Q q == inject_Q r.
Proof.
  intros q r Hqr. split.
  - apply inject_Q_le. rewrite Hqr. apply Qle_refl.
  - apply inject_Q_le. rewrite Hqr. apply Qle_refl.
Qed.

Lemma inject_Q_one : inject_Q 1 == inject_Q 1.
Proof. reflexivity. Qed.

Lemma CReal_lt_cast : ∀ q r : Q, inject_Q q < inject_Q r → (q ≤ r)%Q ∧ ¬ (q == r)%Q.
Proof. intros q r H. exact (Qle_ne_of_lt q r (lt_inject_Q q r H)). Qed.

Lemma CReal_equ_of_le : ∀ x y : CReal, x ≤ y → y ≤ x → x == y.
Proof. intros x y Hxy Hyx. exact (conj Hyx Hxy). Qed.

Powers of two and the Cauchy bound

Lemma Qpow2_pos : ∀ k : Z, (0 < 2 ^ k)%Q.
Proof. intros k. apply Qpower_0_lt. lra. Qed.

Lemma Qpow2_le : ∀ k m : Z, (k ≤ m)%Z → (2 ^ k ≤ 2 ^ m)%Q.
Proof. intros k m H. apply Qpower_le_compat_l; [ exact H | lra ]. Qed.

Lemma Qpow2_lt : ∀ k m : Z, (k < m)%Z → (2 ^ k < 2 ^ m)%Q.
Proof. intros k m H. apply Qpower_lt_compat_l; [ exact H | lra ]. Qed.

Below index n the sequence stays within 2^n of its value at n.
Lemma seq_close : ∀ (x : CReal) (n k : Z), (k ≤ n)%Z →
  (seq x n - 2 ^ n < seq x k)%Q ∧ (seq x k < seq x n + 2 ^ n)%Q.
Proof.
  intros x n k Hk.
  pose proof (cauchy x n k n Hk (Z.le_refl n)) as Hc.
  apply Qabs_Qlt_condition in Hc. lra.
Qed.

What is not strictly below stays within twice the power of two.
Lemma seq_gap_le : ∀ (x y : CReal) (k : Z), (x < y → False) →
  (seq y k - seq x k ≤ 2 × 2 ^ k)%Q.
Proof.
  intros x y k Hnlt.
  destruct (Qlt_le_dec (2 × 2 ^ k) (seq y k - seq x k)) as [H | H].
  - destruct (Hnlt (exist _ k H)).
  - exact H.
Qed.

Evidence that evaluates

The index at which twice the power of two is below a positive margin.
Definition margin_index (delta : Q) : Z := Qlowbound_ltabs_ZExp2 (delta × (1 # 2)).

Lemma margin_index_spec : ∀ delta : Q, (0 < delta)%Q →
  (2 × 2 ^ margin_index delta < delta)%Q.
Proof.
  intros delta Hd. unfold margin_index.
  pose proof (Qlowbound_lt_ZExp2_spec (delta × (1 # 2)) ltac:(lra)). lra.
Qed.

A gap above a positive margin at every index below n0 is a strict inequality, witnessed below n0 where the margin is large enough.
Definition CRealLt_of_margin (x y : CReal) (n0 : Z) (delta : Q) (Hd : (0 < delta)%Q)
  (H : ∀ k : Z, (k ≤ n0)%Z → (delta < seq y k - seq x k)%Q) : x < y.
Proof.
  ∃ (Z.min n0 (margin_index delta)).
  abstract (pose proof (margin_index_spec delta Hd);
    pose proof (Qpow2_le (Z.min n0 (margin_index delta)) (margin_index delta) (Z.le_min_r _ _));
    pose proof (H (Z.min n0 (margin_index delta)) (Z.le_min_l _ _)); lra).
Defined.

Definition CRealLt_cast (q r : Q) (H : (q ≤ r)%Q ∧ ¬ (q == r)%Q) : inject_Q q < inject_Q r.
Proof.
  pose proof (Qlt_of_le_ne q r H) as Hqr.
  refine (CRealLt_of_margin (inject_Q q) (inject_Q r) 0 ((r - q) × (1 # 2)) _ _).
  - abstract lra.
  - abstract (intros k _; simpl; lra).
Defined.

Definition CRealLt_zero_one : inject_Q 0 < inject_Q 1 := CRealLt_cast 0 1 (Q_unit_pos 1).

Definition CRealLt_trans (x y z : CReal) (Hxy : x < y) (Hyz : y < z) : x < z.
Proof.
  destruct Hxy as [n Hn], Hyz as [m Hm].
  refine (CRealLt_of_margin x z (Z.min n m) (2 ^ Z.min n m) (Qpow2_pos _) _).
  abstract (intros k Hk;
    pose proof (seq_close z m k ltac:(lia)); pose proof (seq_close x n k ltac:(lia));
    destruct (Z.le_ge_cases n m) as [Hnm | Hnm];
    [ rewrite (Z.min_l n m Hnm); pose proof (seq_close y m n Hnm)
    | rewrite (Z.min_r n m Hnm); pose proof (seq_close y n m Hnm) ]; lra).
Defined.

Definition CRealLt_equ_l (x x' y : CReal) (Heq : x == x') (H : x < y) : x' < y.
Proof.
  destruct H as [n Hn].
  refine (CRealLt_of_margin x' y
    (Z.min n (margin_index ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2))))
    ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2)) _ _).
  - abstract lra.
  - abstract (intros k Hk;
      pose proof (seq_close y n k ltac:(lia)); pose proof (seq_close x n k ltac:(lia));
      pose proof (seq_gap_le x x' k (proj1 Heq));
      pose proof (margin_index_spec ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2)) ltac:(lra));
      pose proof (Qpow2_le k (margin_index ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2))) ltac:(lia));
      lra).
Defined.

Definition CRealLt_equ_r (x y y' : CReal) (Heq : y == y') (H : x < y) : x < y'.
Proof.
  destruct H as [n Hn].
  refine (CRealLt_of_margin x y'
    (Z.min n (margin_index ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2))))
    ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2)) _ _).
  - abstract lra.
  - abstract (intros k Hk;
      pose proof (seq_close y n k ltac:(lia)); pose proof (seq_close x n k ltac:(lia));
      pose proof (seq_gap_le y' y k (proj2 Heq));
      pose proof (margin_index_spec ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2)) ltac:(lra));
      pose proof (Qpow2_le k (margin_index ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2))) ltac:(lia));
      lra).
Defined.

Cotransitivity is the decision: at a fine enough index, the third real is compared with the midpoint of the two.
Definition CRealLt_cotrans (x y z : CReal) (H : x < z) : ((x < y) + (y < z))%type.
Proof.
  destruct H as [n Hn].
  set (k := Z.min n (margin_index ((seq z n - seq x n - 2 × 2 ^ n) × (1 # 2)))).
  destruct (Qlt_le_dec ((seq x n + seq z n) × (1 # 2)) (seq y k)) as [Hmid | Hmid].
  - left. ∃ k.
    abstract (pose proof (seq_close x n k (Z.le_min_l _ _));
      pose proof (margin_index_spec ((seq z n - seq x n - 2 × 2 ^ n) × (1 # 2)) ltac:(lra));
      pose proof (Qpow2_le k (margin_index ((seq z n - seq x n - 2 × 2 ^ n) × (1 # 2))) (Z.le_min_r _ _));
      lra).
  - right. ∃ k.
    abstract (pose proof (seq_close z n k (Z.le_min_l _ _));
      pose proof (margin_index_spec ((seq z n - seq x n - 2 × 2 ^ n) × (1 # 2)) ltac:(lra));
      pose proof (Qpow2_le k (margin_index ((seq z n - seq x n - 2 × 2 ^ n) × (1 # 2))) (Z.le_min_r _ _));
      lra).
Defined.

Definition CRealLt_add_mono_l (x y z : CReal) (H : y < z) : x + y < x + z.
Proof.
  destruct H as [n Hn].
  refine (CRealLt_of_margin (x + y) (x + z) (n + 1) (seq z n - seq y n - 2 × 2 ^ n) _ _).
  - abstract lra.
  - abstract (intros k Hk;
      change (seq (x + y) k) with (Qred (seq x (k - 1) + seq y (k - 1)));
      change (seq (x + z) k) with (Qred (seq x (k - 1) + seq z (k - 1)));
      rewrite !Qred_correct;
      pose proof (seq_close z n (k - 1) ltac:(lia)); pose proof (seq_close y n (k - 1) ltac:(lia));
      lra).
Defined.

Definition CRealLt_add_reg_l (x y z : CReal) (H : x + y < x + z) : y < z.
Proof.
  destruct H as [n Hn]. ∃ (n - 1)%Z.
  abstract (change (seq (x + y) n) with (Qred (seq x (n - 1) + seq y (n - 1))) in Hn;
    change (seq (x + z) n) with (Qred (seq x (n - 1) + seq z (n - 1))) in Hn;
    rewrite !Qred_correct in Hn; pose proof (Qpow2_lt (n - 1) n ltac:(lia)); lra).
Defined.

Definition CRealLt_mul_pos (x y : CReal) (Hx : inject_Q 0 < x) (Hy : inject_Q 0 < y) :
  inject_Q 0 < x × y.
Proof.
  destruct Hx as [n Hn], Hy as [m Hm].
  refine (CRealLt_of_margin (inject_Q 0) (x × y)
    (Z.min (n + scale y + 1) (m + scale x + 1)) (2 ^ n × 2 ^ m) _ _).
  - abstract (apply Qmult_lt_0_compat; apply Qpow2_pos).
  - abstract (intros k Hk; simpl in Hn, Hm; simpl;
      unfold CReal_mult_seq;
      pose proof (seq_close x n (k - scale y - 1) ltac:(lia));
      pose proof (seq_close y m (k - scale x - 1) ltac:(lia));
      pose proof (Qpow2_pos n); pose proof (Qpow2_pos m);
      pose proof (Qmult_lt_compat_nonneg (2 ^ n) (seq x (k - scale y - 1))
        (2 ^ m) (seq y (k - scale x - 1)) ltac:(lra) ltac:(lra));
      lra).
Defined.

Definition CRealLt_inv_pos (x : CReal) (Hx : ((x < inject_Q 0) + (inject_Q 0 < x))%type)
  (Hpos : inject_Q 0 < x) : inject_Q 0 < CReal_inv x Hx.
Proof.
  destruct Hx as [Hneg | Hpos'].
  - destruct (CRealLt_asym _ _ Hneg Hpos).
  - refine (CRealLt_of_margin (inject_Q 0) (CReal_inv_pos x Hpos') 0 (2 ^ (- scale x)) (Qpow2_pos _) _).
    abstract (intros k _; simpl; unfold CReal_inv_pos_seq;
      pose proof (CRealLowerBoundSpec x Hpos' (CReal_inv_pos_cm x Hpos' k) (Z.le_min_l _ _));
      pose proof (bound x (CReal_inv_pos_cm x Hpos' k)) as Hb;
      apply Qabs_Qlt_condition in Hb;
      pose proof (Qpow2_pos (CRealLowerBound x Hpos')); pose proof (Qpow2_pos (scale x));
      rewrite Qpower_opp;
      pose proof (proj1 (Qinv_lt_contravar (seq x (CReal_inv_pos_cm x Hpos' k)) (2 ^ scale x)
        ltac:(lra) ltac:(lra)) ltac:(lra));
      lra).
Defined.

Definition CRealLt_dense (x y : CReal) (H : x < y) :
  {q : Q & ((x < inject_Q q) × (inject_Q q < y))%type}.
Proof.
  destruct H as [n Hn].
  ∃ ((seq x n + seq y n) × (1 # 2))%Q. split.
  - refine (CRealLt_of_margin x (inject_Q ((seq x n + seq y n) × (1 # 2))) n
      ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2)) _ _).
    + abstract lra.
    + abstract (intros k Hk; simpl; pose proof (seq_close x n k Hk); lra).
  - refine (CRealLt_of_margin (inject_Q ((seq x n + seq y n) × (1 # 2))) y n
      ((seq y n - seq x n - 2 × 2 ^ n) × (1 # 2)) _ _).
    + abstract lra.
    + abstract (intros k Hk; simpl; pose proof (seq_close y n k Hk); lra).
Defined.

Definition CRealLt_archimedean (x : CReal) : {q : Q & x < inject_Q q}.
Proof.
  ∃ (seq x 0 + 3)%Q. ∃ 0%Z.
  abstract (simpl; lra).
Defined.

Approximation and completeness

Approximation at a rational precision: the sequence at the binary exponent whose power of two is below the precision, the negated bit length of the denominator.
Definition CReal_approx (x : CReal) (eps : Q) : Q := seq x (- (Z.log2 (Z.pos (Qden eps)) + 1)).

Lemma Qpow2_neg_le : ∀ (d : positive) (a : Z), (0 < a)%Z →
  (2 ^ (- (Z.log2 (Z.pos d) + 1)) ≤ a # d)%Q.
Proof.
  intros d a Ha.
  pose proof (Z.log2_spec (Z.pos d) eq_refl) as [_ Hd].
  pose proof (Z.log2_nonneg (Z.pos d)) as Hl.
  unfold Z.succ in Hd.
  remember (Z.log2 (Z.pos d) + 1)%Z as k eqn:Hk.
  destruct k as [| p | p]; [ lia | | lia ].
  change (- Z.pos p)%Z with (Z.neg p).
  rewrite (Qpower_decomp_neg_pos p 2 1). rewrite Z.pow_1_l by lia.
  rewrite <- Pos2Z.inj_pow in Hd.
  unfold Qle. simpl. nia.
Qed.

Lemma CReal_approx_close : ∀ (x : CReal) (eps : Q),
  (0 ≤ eps)%Q ∧ ¬ (0 == eps)%Q →
  CReal_abs (x + - inject_Q (CReal_approx x eps)) ≤ inject_Q eps.
Proof.
  intros x eps Heps.
  apply (CReal_le_trans _ _ _ (CReal_cv_self' x (- (Z.log2 (Z.pos (Qden eps)) + 1)))).
  apply inject_Q_le.
  destruct eps as [a d]. simpl.
  apply Qpow2_neg_le.
  pose proof (Qlt_of_le_ne _ _ Heps) as Hlt. unfold Qlt in Hlt. simpl in Hlt. lia.
Qed.

The limit of a Cauchy sequence is Coq's CReal_from_cauchy, a transparent construction; what Coq proves about it is hidden behind the witness of Rcauchy_complete, so its proof is restated here with the witness exposed.
Lemma CReal_from_cauchy_cv : ∀ (xn : nat → CReal) (cau : Un_cauchy_mod xn),
  seq_cv xn (CReal_from_cauchy xn cau).
Proof.
  intros xn cau.

  intro p.
  pose proof (CReal_cv_self' (CReal_from_cauchy xn cau) (Z.neg p - 1)%Z) as H.

  pose proof (cau (2×p)%positive) as [k cv].

  rewrite CReal_abs_minus_sym in H.
  unfold CReal_from_cauchy at 1 in H.
  rewrite CReal_red_seq in H.
  unfold CReal_from_cauchy_seq in H.
  remember (CReal_from_cauchy_cm (Z.neg p - 1))%positive as i'.
  destruct (cau (4 × 2 ^ i')%positive) as [i imaj].
  ∃ (max k i).

  intros j H0.
  setoid_replace (xn j - CReal_from_cauchy xn cau)
  with (xn j - inject_Q (seq (xn i) (Z.neg i' - 2)%Z)
     + (inject_Q (seq (xn i) (Z.neg i' - 2)%Z) - CReal_from_cauchy xn cau)).
  2: ring.
  apply (CReal_le_trans _ _ _ (CReal_abs_triang _ _)).
  apply (CReal_le_trans _ (inject_Q (1#2×p) + inject_Q (1#2×p))).
  - apply CReal_plus_le_compat.
    2: { apply (CReal_le_trans _ _ _ H). apply inject_Q_le.
         rewrite Qpower_minus_pos.
         assert(∀ (n:Z) (p q : positive), n#(p×q) == (n#p) × (1#q))%Q as Aux
             by ( intros; unfold Qeq, Qmult, Qnum, Qden; ring ); rewrite Aux; clear Aux.
         rewrite Qmult_comm; apply Qmult_le_l; [lra|].
         pose proof Qpower_2powneg_le_inv p.
         pose proof Qpower_0_lt 2 (Z.neg p)%Z; lra. }

    
    specialize (imaj j i (Nat.le_trans _ _ _ (Nat.le_max_r _ _) H0) (Nat.le_refl _)).
    apply (CReal_le_trans _ (inject_Q (1 # 4 × p) + inject_Q (1 # 4 × p))).
    + setoid_replace (xn j - inject_Q (seq (xn i) (Z.neg i' - 2)))
        with (xn j - xn i + (xn i - inject_Q (seq (xn i) (Z.neg i' - 2)))).
      2: ring.
      apply (CReal_le_trans _ _ _ (CReal_abs_triang _ _)).
      apply CReal_plus_le_compat.
      × apply (CReal_le_trans _ _ _ imaj).
        rewrite Heqi'. change (Z.neg p - 1)%Z with (Z.neg (p + 1))%Z.
        unfold CReal_from_cauchy_cm.
        apply inject_Q_le.
        unfold Qle, Qnum, Qden.
        rewrite Z.mul_1_l, Z.mul_1_l.
        apply Pos2Z.pos_le_pos, Pos.mul_le_mono_l.
        pose proof Pospow_lin_le_2pow p.
        rewrite Pos.add_1_r, Pos.pow_succ_r.
        lia.
      × clear imaj.

        pose proof CReal_cv_self' (xn i) (Z.neg i' - 2).
        apply (CReal_le_trans _ _ _ H1).
        apply inject_Q_le.
        rewrite Heqi'. change (Z.neg p - 1)%Z with (Z.neg (p + 1))%Z.
        unfold CReal_from_cauchy_cm.
        change (Z.neg (p + 1))%Z with (Z.neg p - 1)%Z.
        ring_simplify (Z.neg p - 1 - 2)%Z.
        rewrite Qpower_minus_pos.
        assert(∀ (n:Z) (p q : positive), n#(p×q) == (n#p) × (1#q))%Q as Aux
            by ( intros; unfold Qeq, Qmult, Qnum, Qden; ring ); rewrite Aux; clear Aux.
        pose proof Qpower_2powneg_le_inv p.
        pose proof Qpower_0_lt 2 (Z.neg p)%Z; lra.

    +
      rewrite <- inject_Q_plus. rewrite (inject_Q_morph _ (1#2×p)).
      × apply CRealLe_refl.
      × rewrite Qinv_plus_distr; reflexivity.
  - rewrite <- inject_Q_plus. rewrite (inject_Q_morph _ (1#p)).
    + apply CRealLe_refl.
    + rewrite Qinv_plus_distr; reflexivity.

Qed.

Definition CReal_complete : ∀ u : nat → CReal,
  (∀ eps : Q, (0 ≤ eps)%Q ∧ ¬ (0 == eps)%Q →
    {n : nat & ∀ i j : nat, (n ≤ i)%nat → (n ≤ j)%nat →
      CReal_abs (u i + - u j) ≤ inject_Q eps}) →
  {l : CReal & ∀ eps : Q, (0 ≤ eps)%Q ∧ ¬ (0 == eps)%Q →
    {n : nat & ∀ i : nat, (n ≤ i)%nat → CReal_abs (u i + - l) ≤ inject_Q eps}}.
Proof.
  intros u Hu.
  pose (cau := (fun p : positive ⇒
    match Hu (1 # p)%Q (Q_unit_pos p) with existT _ n Hn ⇒ exist _ n Hn end) : Un_cauchy_mod u).
  ∃ (CReal_from_cauchy u cau).
  intros eps Heps.
  destruct (CReal_from_cauchy_cv u cau (Qden eps)) as [n Hn].
  ∃ n. intros i Hi.
  apply (CReal_le_trans _ _ _ (Hn i Hi)).
  apply inject_Q_le. exact (Q_unit_den_le eps (Qlt_of_le_ne 0 eps Heps)).
Defined.
