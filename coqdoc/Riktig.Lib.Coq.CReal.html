Local Open Scope CReal_scope.

Coq's constructive Cauchy reals, in the shape the constructive reals structure asks for: the rationals' strict order as "at most and not equal", cotransitivity from Coq's own packaging, approximation by the sequence at the precision's denominator, and completeness over a rational precision rather than a positive one.
Lemma Qlt_of_le_ne : ∀ q r : Q, (q ≤ r)%Q ∧ ¬ (q == r)%Q → (q < r)%Q.
Proof.
  intros q r [Hle Hne].
  destruct (proj1 (Qle_lteq q r) Hle) as [H | H]; [ exact H | destruct (Hne H) ].
Qed.

Lemma Qle_ne_of_lt : ∀ q r : Q, (q < r)%Q → (q ≤ r)%Q ∧ ¬ (q == r)%Q.
Proof. intros q r H. split; [ exact (Qlt_le_weak q r H) | exact (Qlt_not_eq q r H) ]. Qed.

Lemma inject_Q_proper : ∀ q r : Q, (q == r)%Q → inject_Q q == inject_Q r.
Proof.
  intros q r Hqr. split.
  - apply inject_Q_le. rewrite Hqr. apply Qle_refl.
  - apply inject_Q_le. rewrite Hqr. apply Qle_refl.
Qed.

Lemma inject_Q_one : inject_Q 1 == inject_Q 1.
Proof. reflexivity. Qed.

Lemma CReal_cast_lt : ∀ q r : Q, (q ≤ r)%Q ∧ ¬ (q == r)%Q → inject_Q q < inject_Q r.
Proof. intros q r H. exact (inject_Q_lt q r (Qlt_of_le_ne q r H)). Qed.

Lemma CReal_lt_cast : ∀ q r : Q, inject_Q q < inject_Q r → (q ≤ r)%Q ∧ ¬ (q == r)%Q.
Proof. intros q r H. exact (Qle_ne_of_lt q r (lt_inject_Q q r H)). Qed.

Lemma CReal_cotrans : ∀ x y z : CReal, x < z → ((x < y) + (y < z))%type.
Proof. intros x y z. exact (snd (CRltLinear CRealConstructive) x y z). Qed.

Lemma CReal_equ_of_le : ∀ x y : CReal, x ≤ y → y ≤ x → x == y.
Proof. intros x y Hxy Hyx. exact (conj Hyx Hxy). Qed.

Lemma CReal_zero_lt_one : inject_Q 0 < inject_Q 1.
Proof. apply CReal_injectQPos. reflexivity. Qed.

Definition CReal_approx (x : CReal) (eps : Q) : Q := seq x (Z.neg (Qden eps)).

Lemma CReal_approx_close : ∀ (x : CReal) (eps : Q),
  (0 ≤ eps)%Q ∧ ¬ (0 == eps)%Q →
  CReal_abs (x + - inject_Q (CReal_approx x eps)) ≤ inject_Q eps.
Proof.
  intros x eps Heps.
  apply (CReal_le_trans _ _ _ (CReal_cv_self x (Qden eps))).
  apply inject_Q_le. exact (Q_unit_den_le eps (Qlt_of_le_ne 0 eps Heps)).
Qed.

Lemma CReal_archimedean : ∀ x : CReal, {q : Q & x < inject_Q q}.
Proof. intros x. destruct (Rup_pos x) as [n Hn]. exact (existT _ (Z.pos n # 1)%Q Hn). Qed.

Lemma Q_unit_pos : ∀ p : positive, (0 ≤ 1 # p)%Q ∧ ¬ (0 == 1 # p)%Q.
Proof. intros p. apply Qle_ne_of_lt. unfold Qlt. simpl. lia. Qed.

Lemma CReal_complete : ∀ u : nat → CReal,
  (∀ eps : Q, (0 ≤ eps)%Q ∧ ¬ (0 == eps)%Q →
    {n : nat & ∀ i j : nat, (n ≤ i)%nat → (n ≤ j)%nat →
      CReal_abs (u i + - u j) ≤ inject_Q eps}) →
  {l : CReal & ∀ eps : Q, (0 ≤ eps)%Q ∧ ¬ (0 == eps)%Q →
    {n : nat & ∀ i : nat, (n ≤ i)%nat → CReal_abs (u i + - l) ≤ inject_Q eps}}.
Proof.
  intros u Hu.
  assert (Hu' : ∀ p : positive,
    {n : nat | ∀ i j : nat, (n ≤ i)%nat → (n ≤ j)%nat →
      CReal_abs (u i + - u j) ≤ inject_Q (1 # p)%Q}).
  { intros p. destruct (Hu (1 # p)%Q (Q_unit_pos p)) as [n Hn]. exact (exist _ n Hn). }
  destruct (CRealComplete u Hu') as [l Hl].
  ∃ l. intros eps Heps.
  destruct (Hl (Qden eps)) as [n Hn].
  ∃ n. intros i Hi.
  apply (CReal_le_trans _ _ _ (Hn i Hi)).
  apply inject_Q_le. exact (Q_unit_den_le eps (Qlt_of_le_ne 0 eps Heps)).
Qed.
