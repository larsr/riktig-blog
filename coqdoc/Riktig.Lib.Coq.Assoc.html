From Coq Require Import List RelationClasses.

Association lists over a key type with a decided equivalence: the lookup, equality up to lookup, and the facts a dictionary needs. Nothing here knows a structure of Riktig: the equivalence and its decision are plain arguments.
Section Assoc.
  Context {K V : Type} (eqK : K → K → Prop) (eqV : V → V → Prop)
    (dec : ∀ a b : K, {eqK a b} + {¬ eqK a b})
    {EK : Equivalence eqK} {EV : Equivalence eqV}.

  Fixpoint assoc_find (d : list (K × V)) (k : K) : option V :=
    match d with
    | nil ⇒ None
    | cons (k', v) d' ⇒ if dec k' k then Some v else assoc_find d' k
    end.

  Definition assoc_eq (d e : list (K × V)) : Prop :=
    ∀ k : K, match assoc_find d k, assoc_find e k with
                  | Some v, Some w ⇒ eqV v w
                  | None, None ⇒ True
                  | _, _ ⇒ False
                  end.

  Lemma assoc_eq_refl : ∀ d : list (K × V), assoc_eq d d.
  Proof. intros d k. destruct (assoc_find d k); [ reflexivity | exact I ]. Qed.

  Lemma assoc_eq_sym : ∀ d e : list (K × V), assoc_eq d e → assoc_eq e d.
  Proof.
    intros d e H k. specialize (H k).
    destruct (assoc_find d k), (assoc_find e k); try contradiction; [ symmetry; exact H | exact I ].
  Qed.

  Lemma assoc_eq_trans : ∀ d e f : list (K × V), assoc_eq d e → assoc_eq e f → assoc_eq d f.
  Proof.
    intros d e f H1 H2 k. specialize (H1 k). specialize (H2 k).
    destruct (assoc_find d k), (assoc_find e k), (assoc_find f k);
      try contradiction; [ exact (transitivity H1 H2) | exact I ].
  Qed.

  Instance assoc_eq_equivalence : Equivalence assoc_eq :=
    { Equivalence_Reflexive := assoc_eq_refl; Equivalence_Symmetric := assoc_eq_sym;
      Equivalence_Transitive := assoc_eq_trans }.

  Lemma assoc_cons_proper : ∀ (k k' : K) (v v' : V) (d e : list (K × V)),
    eqK k k' → eqV v v' → assoc_eq d e → assoc_eq (cons (k, v) d) (cons (k', v') e).
  Proof.
    intros k k' v v' d e Hk Hv Hd j. simpl.
    destruct (dec k j) as [ekj | nkj], (dec k' j) as [ekj' | nkj'].
    - exact Hv.
    - destruct nkj'. exact (transitivity (symmetry Hk) ekj).
    - destruct nkj. exact (transitivity Hk ekj').
    - exact (Hd j).
  Qed.

  Lemma assoc_find_congruent : ∀ (d : list (K × V)) (k k' : K),
    eqK k k' → assoc_find d k = assoc_find d k'.
  Proof.
    intros d k k' Hk. induction d as [| [k0 v0] d IH]; simpl.
    - reflexivity.
    - destruct (dec k0 k) as [e1 | n1], (dec k0 k') as [e2 | n2].
      + reflexivity.
      + destruct n2. exact (transitivity e1 Hk).
      + destruct n1. exact (transitivity e2 (symmetry Hk)).
      + exact IH.
  Qed.

  Lemma assoc_find_cons_here : ∀ (k k' : K) (v : V) (d : list (K × V)),
    eqK k k' → ∃ v' : V, assoc_find (cons (k, v) d) k' = Some v' ∧ eqV v' v.
  Proof.
    intros k k' v d Hk. simpl. destruct (dec k k') as [| n].
    - ∃ v. split; reflexivity.
    - destruct (n Hk).
  Qed.

  Lemma assoc_find_cons_there : ∀ (k k' : K) (v : V) (d : list (K × V)),
    ¬ eqK k k' → assoc_find (cons (k, v) d) k' = assoc_find d k'.
  Proof.
    intros k k' v d Hk. simpl. destruct (dec k k') as [e |].
    - destruct (Hk e).
    - reflexivity.
  Qed.
End Assoc.
