From Coq Require Import Arith Lia RelationClasses Morphisms.

Recursion on nat by nat_rect, with its equations at any setoid on the result, and the order as "something added gives the other".
Definition natrec_nat (A : Type) (a : A) (f : nat → A → A) (n : nat) : A :=
  nat_rect (fun _ ⇒ A) a f n.

Lemma natrec_nat_proper : ∀ (A : Type) (eA : A → A → Prop) (EA : Equivalence eA)
    (a : A) (f : nat → A → A), Proper (eq ==> eA ==> eA) f →
  Proper (eq ==> eA) (natrec_nat A a f).
Proof. intros A eA EA a f _ x y Hxy. rewrite Hxy. reflexivity. Qed.

Lemma natrec_nat_zero : ∀ (A : Type) (eA : A → A → Prop) (EA : Equivalence eA)
    (a : A) (f : nat → A → A), eA (natrec_nat A a f 0) a.
Proof. intros. reflexivity. Qed.

Lemma natrec_nat_succ : ∀ (A : Type) (eA : A → A → Prop) (EA : Equivalence eA)
    (a : A) (f : nat → A → A) (n : nat),
  eA (natrec_nat A a f (S n)) (f n (natrec_nat A a f n)).
Proof. intros. reflexivity. Qed.

Lemma nat_le_proper : Proper (eq ==> eq ==> iff) le.
Proof. intros a a' Ha b b' Hb. rewrite Ha, Hb. reflexivity. Qed.

Lemma nat_le_add : ∀ a b : nat, a ≤ b ↔ ∃ c : nat, a + c = b.
Proof.
  intros a b. split.
  - intros Hab. ∃ (b - a). rewrite Nat.add_comm. exact (Nat.sub_add a b Hab).
  - intros [c Hc]. rewrite <- Hc. exact (Nat.le_add_r a c).
Qed.

Lemma lt_of_le_neq : ∀ a b : nat, a ≤ b ∧ a ≠ b → a < b.
Proof. intros a b [H1 H2]. lia. Qed.
