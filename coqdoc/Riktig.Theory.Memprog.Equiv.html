Local Notation interpZ :=
  (interp Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation EvalZ :=
  (Eval Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation interp_soundZ :=
  (interp_sound Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation interp_adequateZ :=
  (interp_adequate Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation interp_monoZ :=
  (interp_mono Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation eval_interpZ :=
  (eval_interp Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).
Local Notation bind_evalZ :=
  (bind_eval Z.eqb Z.add Z.sub Z.leb Z.geb Z.of_nat 1 (fun v : Z ⇒ v) 0).


Definition pequiv {a : Type} (p q : MemT Z Z a) : Prop :=
  ∀ (m : Machine Z Z) (r : option a) (m' : Machine Z Z),
    EvalZ p m r m' ↔ EvalZ q m r m'.

Lemma pequiv_refl (a : Type) : ∀ p : MemT Z Z a, pequiv p p.
Proof. intros p m r m'. reflexivity. Qed.

Lemma pequiv_sym (a : Type) : ∀ p q : MemT Z Z a, pequiv p q → pequiv q p.
Proof. intros p q H m r m'. symmetry. apply H. Qed.

Lemma pequiv_trans (a : Type) :
  ∀ p q s : MemT Z Z a, pequiv p q → pequiv q s → pequiv p s.
Proof.
  intros p q s Hpq Hqs m r m'.
  rewrite (Hpq m r m'). apply Hqs.
Qed.

#[global] Instance MemT_EqRel (a : Type) : EqRel (MemT Z Z a) := @pequiv a.

#[global] Instance MemT_equ_Equivalence (a : Type) :
  Equivalence (@equ (MemT Z Z a) (MemT_EqRel a)).
Proof.
  apply Build_Equivalence.
  - exact (pequiv_refl a).
  - exact (pequiv_sym a).
  - exact (pequiv_trans a).
Qed.


#[global] Instance bind_t_proper (a b : Type) :
  Proper (equ ==> pointwise_relation a equ ==> equ) (@bind_t Z Z a b).
Proof.
  intros p p' Hp f g Hfg m r m2.
  unfold bind_t. rewrite !bind_evalZ.
  split.
  - intros (o1 & m1 & H1 & H2).
    ∃ o1, m1. split.
    + exact (proj1 (Hp m o1 m1) H1).
    + destruct o1 as [x|].
      × exact (proj1 (Hfg x m1 r m2) H2).
      × exact H2.
  - intros (o1 & m1 & H1 & H2).
    ∃ o1, m1. split.
    + exact (proj2 (Hp m o1 m1) H1).
    + destruct o1 as [x|].
      × exact (proj2 (Hfg x m1 r m2) H2).
      × exact H2.
Qed.

#[global] Instance then_t_proper (a b : Type) :
  Proper (equ ==> equ ==> equ) (@then_t Z Z a b).
Proof.
  intros p p' Hp k k' Hk.
  unfold then_t.
  apply bind_t_proper; [exact Hp | intros _; exact Hk].
Qed.


Lemma interp_while_cong :
  ∀ (b : Type)
         (c c' : MemT Z Z bool) (body body' : MemT Z Z unit)
         (k k' : MemT Z Z b),
    pequiv c c' → pequiv body body' → pequiv k k' →
    ∀ (fuel : nat) (m : Machine Z Z) (r : option b) (m' : Machine Z Z),
      interpZ fuel (While c body k) m = (Some r, m') →
      ∃ fuel', interpZ fuel' (While c' body' k') m = (Some r, m').
Proof.
  intros b c c' body body' k k' Hc Hbody Hk.
  induction fuel as [|f IH]; intros m r m' Hrun.
  - cbn [interpZ] in Hrun. discriminate.
  - cbn [interpZ] in Hrun.
    destruct (interpZ f c m) as [rc mach1] eqn:Ec.
    destruct rc as [[[|]|]|] eqn:Erc; try discriminate.
    +
      destruct (interpZ f body mach1) as [rb mach2] eqn:Eb.
      destruct rb as [[u|]|] eqn:Erb; try discriminate.
      destruct (IH _ _ _ Hrun) as (f3 & Hf3).
      assert (Efc : ∃ fc, interpZ fc c' m = (Some (Some true), mach1)).
      { apply interp_adequateZ, Hc. eapply interp_soundZ. exact Ec. }
      assert (Efb : ∃ fb, interpZ fb body' mach1 = (Some (Some u), mach2)).
      { apply interp_adequateZ, Hbody. eapply interp_soundZ. exact Eb. }
      destruct Efc as (fc & Hfc). destruct Efb as (fb & Hfb).
      ∃ (S (Nat.max fc (Nat.max fb f3))).
      cbn [interpZ].
      rewrite (interp_monoZ fc _ c' m _ mach1 _ (Nat.le_max_l _ _) Hfc). cbn.
      rewrite (interp_monoZ fb _ body' mach1 _ mach2 _
                 (Nat.le_trans _ _ _ (Nat.le_max_l _ _) (Nat.le_max_r _ _)) Hfb). cbn.
      exact (interp_monoZ f3 _ (While c' body' k') mach2 _ m' _
               (Nat.le_trans _ _ _ (Nat.le_max_r _ _) (Nat.le_max_r _ _)) Hf3).
    +
      assert (Efc : ∃ fc, interpZ fc c' m = (Some (Some false), mach1)).
      { apply interp_adequateZ, Hc. eapply interp_soundZ. exact Ec. }
      assert (Efk : ∃ fk, interpZ fk k' mach1 = (Some r, m')).
      { apply interp_adequateZ, Hk. eapply interp_soundZ. exact Hrun. }
      destruct Efc as (fc & Hfc). destruct Efk as (fk & Hfk).
      ∃ (S (Nat.max fc fk)).
      cbn [interpZ].
      rewrite (interp_monoZ fc _ c' m _ mach1 _ (Nat.le_max_l _ _) Hfc). cbn.
      exact (interp_monoZ fk _ k' mach1 _ m' _ (Nat.le_max_r _ _) Hfk).
Qed.

Theorem while_cong :
  ∀ (b : Type)
         (c c' : MemT Z Z bool) (body body' : MemT Z Z unit)
         (k k' : MemT Z Z b),
    c == c' → body == body' → k == k' →
    While c body k == While c' body' k'.
Proof.
  intros b c c' body body' k k' Hc Hbody Hk m r m'.
  split.
  - intros Hev.
    apply eval_interpZ in Hev as (fuel & Hfuel).
    apply eval_interpZ.
    exact (interp_while_cong b c c' body body' k k' Hc Hbody Hk fuel m r m' Hfuel).
  - intros Hev.
    apply eval_interpZ in Hev as (fuel & Hfuel).
    apply eval_interpZ.
    symmetry in Hc. symmetry in Hbody. symmetry in Hk.
    exact (interp_while_cong b c' c body' body k' k Hc Hbody Hk fuel m r m' Hfuel).
Qed.

#[global] Instance While_proper (b : Type) :
  Proper (equ ==> equ ==> equ ==> equ) (@While Z Z (option b)).
Proof.
  intros c c' Hc body body' Hbody k k' Hk.
  exact (while_cong b c c' body body' k k' Hc Hbody Hk).
Qed.


Example bind_t_proper_rewrites :
  ∀ (a b : Type) (p p' : MemT Z Z a) (f : a → MemT Z Z b),
    p == p' → p >>= f == p' >>= f.
Proof.
  intros a b p p' f Hp.
  rewrite Hp.
  reflexivity.
Qed.
