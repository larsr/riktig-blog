The complex numbers are the field with one generator squaring to - 1: the Clifford algebra Cl(0,1,0), pairs (a, b) standing for a + b i, the real and imaginary parts being the two coordinates.

  Context {F : Type} `{Field F}.

  #[export] Instance Complex_I : HasI (Double (- (1 : F)) F) := (0, 1).
  #[export] Instance Complex_Re : HasRe F (Double (- (1 : F)) F) := fst.

  #[export] Instance Complex_mul_Commutative : Commutative (Double (- (1 : F)) F) (Double (- (1 : F)) F) mult.
  Proof.
    intros [a b] [c d]. split; simpl.
    - calc (a × c + - (1 : F) × (b × d))
        == (c × a + - (1 : F) × (b × d)) { (commutative (op := mult) a c) }
        == (c × a + - (1 : F) × (d × b)) { (commutative (op := mult) b d) }.
    - calc (a × d + b × c)
        == (d × a + b × c) { (commutative (op := mult) a d) }
        == (d × a + c × b) { (commutative (op := mult) b c) }
        == (c × b + d × a) { (commutative (op := add) (d × a) (c × b)) }.
  Qed.

  #[export] Instance Complex_Double : Complex F (Double (- (1 : F)) F) Complex_I Complex_Re Complex_Im.
  Proof.
    split.
    - split; simpl.
      + calc ((0 : F) × 0 + - (1 : F) × (1 × 1))
          == (0 + - (1 : F) × (1 × 1)) { (zero_mul (0 : F)) }
          == (- (1 : F) × (1 × 1)) { (left_identity (action := add) (- (1 : F) × (1 × 1))) }
          == (- (1 : F) × 1) { (left_identity (action := mult) (1 : F)) }
          == (- (1 : F)) { (right_identity (action := mult) (- (1 : F))) }.
      + calc ((0 : F) × 1 + 1 × 0)
          == (0 + (1 : F) × 0) { (zero_mul (1 : F)) }
          == ((0 : F) + 0) { (mul_zero (1 : F)) }
          == (0 : F) { (left_identity (action := add) (0 : F)) }
          == (- (0 : F)) { neg_zero }.
    - intros [a b] [a' b'] [Ha Hb]. exact Ha.
    - intros [a b] [a' b'] [Ha Hb]. exact Hb.
    - intros [a b] [c d]. reflexivity.
    - intros [a b] [c d]. reflexivity.
    - intros r [a b]. reflexivity.
    - intros r [a b]. reflexivity.
    - reflexivity.
    - reflexivity.
    - reflexivity.
    - reflexivity.
    - intros [a b]. split; simpl.
      + calc a
          == (a + 0) { (right_identity (action := add) a) }
          == (a + b × 0) { (mul_zero b) }
          == (a × 1 + b × 0) { (right_identity (action := mult) a) }.
      + calc b
          == (0 + b) { (left_identity (action := add) b) }
          == (a × 0 + b) { (mul_zero a) }
          == (a × 0 + b × 1) { (right_identity (action := mult) b) }.
  Qed.
End ComplexDouble.
