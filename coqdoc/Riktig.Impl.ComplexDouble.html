The complex numbers are the field with one generator squaring to - 1: the Clifford algebra Cl(0,1,0), pairs (a, b) standing for a + b i, the real and imaginary parts being the two coordinates.
Section ComplexDouble.
  Context {F : Type} `{Field F}.

  #[export] Instance Complex_I : HasI (Double (- (1 : F)) F) := (0, 1).
  #[export] Instance Complex_Re : HasRe F (Double (- (1 : F)) F) := fst.
  #[export] Instance Complex_Im : HasIm F (Double (- (1 : F)) F) := snd.

  #[export] Instance Complex_mul_Commutative : Commutative (Double (- (1 : F)) F) (Double (- (1 : F)) F) mult.
  Proof.
    intros [a b] [c d]. split; simpl.
    - exact (double_mul_comm (- (1 : F)) a b c d).
    - exact (double_mul_comm_snd a b c d).
  Qed.

  #[export] Instance Complex_Double : Complex F (Double (- (1 : F)) F) Complex_I Complex_Re Complex_Im.
  Proof.
    split.
    - split; simpl.
      + exact (double_i_sq (- (1 : F))).
      + exact double_i_sq_snd.
    - intros [a b] [a' b'] [Ha Hb]. exact Ha.
    - intros [a b] [a' b'] [Ha Hb]. exact Hb.
    - intros [a b] [c d]. reflexivity.
    - intros [a b] [c d]. reflexivity.
    - intros r [a b]. reflexivity.
    - intros r [a b]. reflexivity.
    - reflexivity.
    - reflexivity.
    - reflexivity.
    - reflexivity.
    - intros [a b]. split; simpl.
      + exact (double_re_decompose a b).
      + exact (double_im_decompose a b).
  Qed.
End ComplexDouble.
