Coq's list is the initial algebra of the list shape over Coq's types: build is nil and cons, and cata is the fold.
Section L.
  Context (X : Type).

  #[export] Instance CoqList_HasBuild : HasBuild Type (List.shape (fun T0 : Type ⇒ T0) X) (list X) :=
    fun s ⇒ match s with
             | inl _ ⇒ nil
             | inr (x, xs) ⇒ x :: xs
             end.

  #[export] Instance CoqList_HasCata : HasCata Type (List.shape (fun T0 : Type ⇒ T0) X) (list X) :=
    fun B h ⇒ fix go (l : list X) : B :=
      match l with
      | nil ⇒ h (inl tt)
      | x :: xs ⇒ h (inr (x, go xs))
      end.

  #[export] Instance CoqList_List :
    List Type (fun T0 : Type ⇒ T0) X (list X) CoqList_HasBuild CoqList_HasCata.
  Proof.
    split. split.
    - exact (layer_functor (fun T0 : Type ⇒ T0) (listS (S := sig Type) X)).
    - intros B h h' Hh l. induction l as [| x xs IH].
      + exact (Hh (inl tt)).
      + exact (eq_trans (f_equal (fun b ⇒ h (inr (x, b))) IH) (Hh (inr (x, _)))).
    - intros B h [[] | [x xs]]; reflexivity.
    - intros B h u Hu l. induction l as [| x xs IH].
      + exact (Hu (inl tt)).
      + exact (eq_trans (Hu (inr (x, xs))) (f_equal (fun b ⇒ h (inr (x, b))) IH)).
  Qed.
End L.
