A field is a geometric algebra over itself: a scalar is its own vector, and its square is its inner product with itself.
Section FieldGeometricAlgebra.
  Context {F : Type} `{Field F}.

  #[export] Instance Field_Vec : HasVec F F := fun u ⇒ u.

  #[export] Instance Field_Vec_Linear : Linear F F F Field_Vec :=
    {| Linear.proper := fun u u' Hu ⇒ Hu;
       Linear.is_additive := fun u v ⇒ reflexivity (u + v);
       Linear.is_homogeneous := fun a u ⇒ reflexivity (a *: u) |}.

  #[export] Instance Field_GeometricAlgebra : GeometricAlgebra F F F Field_Vec :=
    {| GeometricAlgebra.vec_linear := Field_Vec_Linear;
       GeometricAlgebra.vec_sq := fun u ⇒ symmetry (right_identity (action := mult) (u × u)) |}.
End FieldGeometricAlgebra.
