From Coq Require Import QArith.
From Riktig Require Export Structure.Archimedean Impl.OrderedFieldQ Impl.NatOrderNat Lib.Coq.Q.

The rationals are Archimedean over Coq's naturals: a natural lies above any rational, the floor plus one.
Instance Q_Archimedean : Archimedean Q nat Q_of_nat :=
  {| Archimedean.of_nat_proper := Q_of_nat_proper;
     Archimedean.of_nat_zero := Q_of_nat_zero;
     Archimedean.of_nat_succ := Q_of_nat_succ;
     Archimedean.archimedean := Q_archimedean |}.
