Coq's prod is the datatype A × B over Coq's types: the shape at Type is prod itself, so mk is the identity and the fold applies the algebra.
Section P.
  Context (A B : Type).

  #[export] Instance CoqProd_HasBuild : HasBuild Type (Pair.shape (fun T0 : Type ⇒ T0) A B) (A × B)%type :=
    fun s ⇒ s.

  #[export] Instance CoqProd_HasCata : HasCata Type (Pair.shape (fun T0 : Type ⇒ T0) A B) (A × B)%type :=
    fun X h ⇒ h.

  #[export] Instance CoqProd_Pair :
    Pair Type (fun T0 : Type ⇒ T0) A B (A × B)%type CoqProd_HasBuild CoqProd_HasCata.
  Proof.
    split. split.
    - exact (layer_functor (fun T0 : Type ⇒ T0) (pairS (S := sig Type) A B)).
    - intros X h h' Hh s. exact (Hh s).
    - intros X h [x y]; reflexivity.
    - intros X h u Hu [x y]; exact (Hu (x, y)).
  Qed.
End P.
