The three equations of the translation, from the recursion's laws.
Section Equations.
  Context {N T : Type} `{TermRec N T} `{NatRec N}
    {ET : Equivalence (@equ T _)}.

  Theorem cps_var : ∀ n : N, cps (#n : T) == λ ⦅#0 #(succ n)⦆.
  Proof. intro n. exact (TermRec.termrec_var T _ ET _ _ _ n). Qed.

  Theorem cps_lam : ∀ b : T, cps (λ b) == λ ⦅#0 (λ (shift_under (cps b)))⦆.
  Proof. intro b. exact (TermRec.termrec_lam T _ ET _ _ _ b). Qed.

  Theorem cps_app : ∀ f a : T,
    cps ⦅f a⦆ == λ ⦅(shift (cps f))
                  (λ ⦅(shift (shift (cps a)))
                      (λ ⦅#(succ 0) #0 #(succ (succ 0))⦆)⦆)⦆.
  Proof. intros f a. exact (TermRec.termrec_app T _ ET _ _ _ f a). Qed.
End Equations.
