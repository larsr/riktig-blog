Record Cl2 (F : Type) := mkCl2 {
  scalar_part : F;
  e1_part : F;
  e2_part : F;
  e12_part : F
}.

Arguments mkCl2 {F} scalar_part e1_part e2_part e12_part.
Arguments scalar_part {F} x : rename.
Arguments e1_part {F} x : rename.
Arguments e2_part {F} x : rename.
Arguments e12_part {F} x : rename.

Section Operations.
  Context {F : Type} `{FF : Field F}.

  #[export] Instance Cl2_EqRel : EqRel (Cl2 F) :=
    fun x y ⇒ scalar_part x == scalar_part y ∧ e1_part x == e1_part y
               ∧ e2_part x == e2_part y ∧ e12_part x == e12_part y.

  #[export] Instance Cl2_Zero : Zero (Cl2 F) := mkCl2 0 0 0 0.

  #[export] Instance Cl2_One : One (Cl2 F) := mkCl2 1 0 0 0.

  #[export] Instance Cl2_Add : HasAdd (Cl2 F) :=
    fun x y ⇒ mkCl2 (scalar_part x + scalar_part y) (e1_part x + e1_part y)
                     (e2_part x + e2_part y) (e12_part x + e12_part y).

  #[export] Instance Cl2_Neg : HasNeg (Cl2 F) :=
    fun x ⇒ mkCl2 (- scalar_part x) (- e1_part x)
                   (- e2_part x) (- e12_part x).

  #[export] Instance Cl2_Mult : HasMult (Cl2 F) :=
    fun x y ⇒ mkCl2
      (scalar_part x × scalar_part y + (e1_part x × e1_part y
         + (e2_part x × e2_part y + - (e12_part x × e12_part y))))
      (scalar_part x × e1_part y + (e1_part x × scalar_part y
         + (e12_part x × e2_part y + - (e2_part x × e12_part y))))
      (scalar_part x × e2_part y + (e2_part x × scalar_part y
         + (e1_part x × e12_part y + - (e12_part x × e1_part y))))
      (scalar_part x × e12_part y + (e12_part x × scalar_part y
         + (e1_part x × e2_part y + - (e2_part x × e1_part y)))).

  #[export] Instance Cl2_Scale : HasScale F (Cl2 F) :=
    fun (a : F) (x : Cl2 F) ⇒
      mkCl2 (a × scalar_part x) (a × e1_part x)
            (a × e2_part x) (a × e12_part x).

  #[export] Instance Cl2_Equivalence : Equivalence Cl2_EqRel.
  Proof.
    apply Build_Equivalence.
    - intros x. repeat split; reflexivity.
    - intros x y [Hs [H1 [H2 H12]]]. repeat split.
      + symmetry. exact Hs.
      + symmetry. exact H1.
      + symmetry. exact H2.
      + symmetry. exact H12.
    - intros x y z [Hs1 [H11 [H21 H121]]] [Hs2 [H12' [H22 H122]]].
      repeat split.
      + rewrite Hs1. exact Hs2.
      + rewrite H11. exact H12'.
      + rewrite H21. exact H22.
      + rewrite H121. exact H122.
  Qed.

End Operations.

Section PlaneVector.
  Context {F : Type} `{FF : Field F}.

  Lemma dot_in_coordinates : ∀ v : Fin.t 2 → F,
    dot v v == v Fin.F1 × v Fin.F1 + v (Fin.FS Fin.F1) × v (Fin.FS Fin.F1).
  Proof.
    intros v.
    calc (dot v v)
      == (bigopF (fun i : Fin.t 2 ⇒ v i × v i))
         { (dot_def 2 v v) }
      == (v Fin.F1 × v Fin.F1
          + bigopF (fun i : Fin.t 1 ⇒ v (Fin.FS i) × v (Fin.FS i)))
         { (bigopF_S 1 (fun i ⇒ v i × v i)) }
      == (v Fin.F1 × v Fin.F1
          + (v (Fin.FS Fin.F1) × v (Fin.FS Fin.F1)
             + bigopF (fun i : Fin.t 0 ⇒
                         v (Fin.FS (Fin.FS i)) × v (Fin.FS (Fin.FS i)))))
         { (bigopF_S 0 (fun i ⇒ v (Fin.FS i) × v (Fin.FS i))) }
      == (v Fin.F1 × v Fin.F1 + (v (Fin.FS Fin.F1) × v (Fin.FS Fin.F1) + 0))
         { (bigopF_0 (fun i ⇒ v (Fin.FS (Fin.FS i)) × v (Fin.FS (Fin.FS i)))) }
      == (v Fin.F1 × v Fin.F1 + v (Fin.FS Fin.F1) × v (Fin.FS Fin.F1))
         { right_identity }.
  Qed.

End PlaneVector.

Section GeneralAgreement.
  Context {F : Type} `{FF : Field F}.
  Add Ring cl2_general_ring : (ring_theory_of_Ring (R := F) commutative).


  Definition cl2_sig : Signature F 2 :=
    Vector.cons F 1 1 (Vector.cons F 1 0 (Vector.nil F)).

  Definition cl2_blade (b1 b2 : bool) : Blade 2 := bcons b1 (bcons b2 bnil).

  Lemma blade2_case : ∀ D : Blade 2,
    D = cl2_blade (bhead D) (bhead (btail D)).
  Proof.
    intros D.
    apply blade_eq; [ reflexivity | ].
    apply blade_eq; [ reflexivity | ].
    exact (blade_0 (btail (btail D))).
  Qed.

  Definition cl2_to_cl (x : Cl2 F) : Cl F 2 :=
    fun D ⇒
      if bhead D
      then if bhead (btail D) then e12_part x else e1_part x
      else if bhead (btail D) then e2_part x else scalar_part x.

  Definition cl2_of_cl (f : Cl F 2) : Cl2 F :=
    mkCl2 (f (cl2_blade false false)) (f (cl2_blade true false))
          (f (cl2_blade false true)) (f (cl2_blade true true)).

  Lemma cl2_of_to : ∀ x : Cl2 F, cl2_of_cl (cl2_to_cl x) == x.
  Proof. intros x. repeat split; reflexivity. Qed.

  Lemma cl2_to_of : ∀ f : Cl F 2, cl2_to_cl (cl2_of_cl f) == f.
  Proof.
    intros f D. rewrite (blade2_case D).
    destruct (bhead D), (bhead (btail D)); reflexivity.
  Qed.

  Lemma cl2_add_agrees : ∀ x y : Cl2 F,
    cl2_of_cl (cl2_to_cl x + cl2_to_cl y) == x + y.
  Proof. intros x y. repeat split; reflexivity. Qed.

  Lemma cl2_one_agrees : cl2_of_cl cl_one == (1 : Cl2 F).
  Proof. repeat split; reflexivity. Qed.

  Lemma cl2_scale_agrees : ∀ (a : F) (x : Cl2 F),
    cl2_of_cl (a *: cl2_to_cl x) == a *: x.
  Proof. intros a x. repeat split; reflexivity. Qed.

  Lemma cl2_mul_agrees : ∀ x y : Cl2 F,
    cl2_of_cl (cl_mul cl2_sig (cl2_to_cl x) (cl2_to_cl y)) == x × y.
  Proof.
    intros x y.
    repeat split;
      unfold cl_mul, cl2_sig, cl2_blade, cl2_to_cl, bsign, parity_sign;
      simpl; ring.
  Qed.

  #[local] Instance Cl2B_One : One (Cl F 2) := cl_one.
  #[local] Instance Cl2B_Mult : HasMult (Cl F 2) := cl_mul cl2_sig.

  Definition cl2_iso : ring_iso (Cl2 F) (Cl F 2).
  Proof.
    refine {| iso_to := cl2_to_cl; iso_of := cl2_of_cl |}.
    - intros x y (Hs & H1 & H2 & H12) D.
      rewrite (blade2_case D).
      destruct (bhead D), (bhead (btail D)); simpl; assumption.
    - intros f g Hfg. repeat split; simpl; apply Hfg.
    - exact cl2_of_to.
    - intros D. rewrite (blade2_case D).
      destruct (bhead D), (bhead (btail D)); reflexivity.
    - intros D. rewrite (blade2_case D).
      destruct (bhead D), (bhead (btail D)); reflexivity.
    - intros x y D. rewrite (blade2_case D).
      destruct (bhead D), (bhead (btail D)); reflexivity.
    - intros x D. rewrite (blade2_case D).
      destruct (bhead D), (bhead (btail D)); reflexivity.
    - intros x y D.
      destruct (cl2_mul_agrees x y) as (Hs & H1 & H2 & H12).
      rewrite (blade2_case D).
      destruct (bhead D), (bhead (btail D)); simpl; symmetry; assumption.
  Defined.

End GeneralAgreement.

  Section ScalarMultiplication.
    Context {F : Type} `{FF : Field F}.

    Add Ring cl2_scale_ring :
      (ring_theory_of_Ring (R := F) commutative).

    #[export] Instance scale_proper :
      Proper (equ ==> equ ==> equ) (@scale F (Cl2 F) _).
    Proof.
      intros a a' Ha [xs x1 x2 x12] [ys y1 y2 y12] (Hs & H1 & H2 & H12).
      simpl in Hs, H1, H2, H12.
      repeat split; simpl; rewrite Ha, ?Hs, ?H1, ?H2, ?H12; reflexivity.
    Qed.

    Lemma scale_vector_add : ∀ (a : F) (x y : Cl2 F),
      a *: (x + y) == a *: x + a *: y.
    Proof.
      intros a [xs x1 x2 x12] [ys y1 y2 y12]; repeat split; simpl; ring.
    Qed.

    Lemma scale_scalar_add : ∀ (a b : F) (x : Cl2 F),
      (a + b) *: x == a *: x + b *: x.
    Proof. intros a b [xs x1 x2 x12]; repeat split; simpl; ring. Qed.

    Lemma scale_associative : ∀ (a b : F) (x : Cl2 F),
      (a × b) *: x == a *: (b *: x).
    Proof. intros a b [xs x1 x2 x12]; repeat split; simpl; ring. Qed.

    Lemma scale_identity : ∀ x : Cl2 F, 1 *: x == x.
    Proof. intros [xs x1 x2 x12]; repeat split; simpl; ring. Qed.

    Lemma scale_mul_left : ∀ (a : F) (x y : Cl2 F),
      (a *: x) × y == a *: (x × y).
    Proof.
      intros a [xs x1 x2 x12] [ys y1 y2 y12]; repeat split; simpl; ring.
    Qed.

    Lemma scale_mul_right : ∀ (a : F) (x y : Cl2 F),
      x × (a *: y) == a *: (x × y).
    Proof.
      intros a [xs x1 x2 x12] [ys y1 y2 y12]; repeat split; simpl; ring.
    Qed.

  End ScalarMultiplication.

  Section VectorEmbedding.
    Context {F : Type} `{FF : Field F}.

    Add Ring cl2_vector_ring :
      (ring_theory_of_Ring (R := F) commutative).

    Definition vec (v : Fin.t 2 → F) : Cl2 F :=
      mkCl2 0 (v Fin.F1) (v (Fin.FS Fin.F1)) 0.

    #[export] Instance vec_proper : Proper (equ ==> equ) vec.
    Proof.
      intros u v Huv. repeat split.
      - reflexivity.
      - exact (Huv Fin.F1).
      - exact (Huv (Fin.FS Fin.F1)).
      - reflexivity.
    Qed.

    Lemma vec_additive : ∀ u v : Fin.t 2 → F,
      vec (u + v) == vec u + vec v.
    Proof.
      intros u v. unfold vec.
      repeat split; simpl; first [ reflexivity | ring ].
    Qed.

    Lemma vec_scale : ∀ (a : F) (v : Fin.t 2 → F),
      vec (a *: v) == a *: vec v.
    Proof.
      intros a v. unfold vec.
      repeat split; simpl; first [ reflexivity | ring ].
    Qed.

    Lemma vec_square : ∀ v : Fin.t 2 → F,
      vec v × vec v
      == (v Fin.F1 × v Fin.F1 + v (Fin.FS Fin.F1) × v (Fin.FS Fin.F1)) *: 1.
    Proof. intros v. unfold vec. repeat split; simpl; ring. Qed.

    Lemma contraction : ∀ v : Fin.t 2 → F,
      vec v × vec v == InnerProductSpace.norm2 (F:=F) (V:=Fin.t 2 → F)
                         (H1:=FinVec_InnerProduct 2) v *: 1.
    Proof.
      intros v.
      calc (vec v × vec v)
        == ((v Fin.F1 × v Fin.F1
             + v (Fin.FS Fin.F1) × v (Fin.FS Fin.F1)) *: (1 : Cl2 F))
           { vec_square }
        == (dot v v *: (1 : Cl2 F))
           { dot_in_coordinates }.
    Qed.

  End VectorEmbedding.

  Section EvenSubalgebra.
    Context {F : Type} `{FF : Field F}.

    Add Ring cl2_even_ring :

    Definition is_even (x : Cl2 F) : Prop := e1_part x == 0 ∧ e2_part x == 0.

    Lemma even_product : ∀ a b c d : F,
      mkCl2 a 0 0 b × mkCl2 c 0 0 d
      == mkCl2 (a × c + - (b × d)) 0 0 (a × d + b × c).
    Proof. intros a b c d; repeat split; simpl; ring. Qed.

    Lemma even_closed_under_mult : ∀ x y : Cl2 F,
      is_even x → is_even y → is_even (x × y).
    Proof.
      intros [xs x1 x2 x12] [ys y1 y2 y12] (Hx1 & Hx2) (Hy1 & Hy2).
      simpl in Hx1, Hx2, Hy1, Hy2.
      split; simpl; rewrite Hx1, Hx2, Hy1, Hy2; ring.
    Qed.

  End EvenSubalgebra.

Section BasisElements.
  Context {F : Type} `{FF : Field F}.

  Add Ring cl2_basis_ring :
    (ring_theory_of_Ring (R := F) commutative).

  Definition cl2_e1 : Cl2 F := mkCl2 0 1 0 0.
  Definition cl2_e2 : Cl2 F := mkCl2 0 0 1 0.
  Definition cl2_e12 : Cl2 F := mkCl2 0 0 0 1.

  Lemma cl2_e1_squared : cl2_e1 × cl2_e1 == 1.
  Proof. unfold cl2_e1; repeat split; simpl; ring. Qed.

  Lemma cl2_e2_squared : cl2_e2 × cl2_e2 == 1.
  Proof. unfold cl2_e2; repeat split; simpl; ring. Qed.

  Lemma cl2_e1_times_e2 : cl2_e1 × cl2_e2 == cl2_e12.
  Proof. unfold cl2_e1, cl2_e2, cl2_e12; repeat split; simpl; ring. Qed.

  Lemma cl2_e2_times_e1 : cl2_e2 × cl2_e1 == - cl2_e12.
  Proof. unfold cl2_e1, cl2_e2, cl2_e12; repeat split; simpl; ring. Qed.

  Lemma cl2_e2_times_e12 : cl2_e2 × cl2_e12 == - cl2_e1.
  Proof. unfold cl2_e1, cl2_e2, cl2_e12; repeat split; simpl; ring. Qed.

  Lemma cl2_anticommutation : cl2_e1 × cl2_e2 == - (cl2_e2 × cl2_e1).
  Proof.
    calc (cl2_e1 × cl2_e2)
      == cl2_e12 { cl2_e1_times_e2 }
      == (- (- cl2_e12)) { neg_involutive }
      == (- (cl2_e2 × cl2_e1)) { cl2_e2_times_e1 }.
  Qed.

  Lemma cl2_e12_squared : cl2_e12 × cl2_e12 == - (1 : Cl2 F).
  Proof.
    calc (cl2_e12 × cl2_e12)
      == ((cl2_e1 × cl2_e2) × cl2_e12) { cl2_e1_times_e2 }
      == (cl2_e1 × (cl2_e2 × cl2_e12)) { Cl2.mult_associative }
      == (cl2_e1 × (- cl2_e1)) { cl2_e2_times_e12 }
      == (- (cl2_e1 × cl2_e1)) { mul_neg_r }
      == (- (1 : Cl2 F)) { cl2_e1_squared }.
  Qed.
