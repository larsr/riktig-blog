Library Riktig.Impl.SequenceList

From Coq Require Import List.
From Riktig Require Export Structure.Sequence Lib.Coq.Parser.

Coq lists as sequences: a list yields its head and leaves its tail. Elements and lists are compared by Coq's equality.
Section SequenceList.
  Context {C : Type}.

  #[export] Instance List_EqRel : EqRel (list C) := eq.
  #[export] Instance List_HasStep : HasStep C (list C) := fun s c s' ⇒ s = c :: s'.
  #[export] Instance List_HasNext : HasNext C (list C) :=
    fun s ⇒ match s with nil ⇒ None | c :: s' ⇒ Some (c, s') end.

  #[export] Instance List_Sequence : Sequence C (list C) List_HasStep (eqC := eq).
  Proof.
    split.
    - exact eq_equivalence.
    - exact eq_equivalence.
    - intros s s0 Hs c c0 Hc t t0 Ht. unfold equ, List_EqRel in ×. subst. reflexivity.
    - intros s c c' s' s'' H1 H2. unfold step, List_HasStep in ×. subst. injection H2 as Hc Hs. split; assumption.
  Qed.

  Lemma list_next_step : ∀ (s : list C) (c : C) (s' : list C),
    step s c s' ↔ ∃ c0 : C, next s = Some (c0, s') ∧ c0 = c.
  Proof.
    intros s c s'. unfold next, List_HasNext, step, List_HasStep. destruct s as [| c0 s0].
    - split; [ discriminate | intros [c1 [H _]]; discriminate H ].
    - split.
      + intros H. injection H as Hc Hs. subst. ∃ c. split; reflexivity.
      + intros [c1 [H Hc]]. injection H as Hc1 Hs. subst. reflexivity.
  Qed.
End SequenceList.
