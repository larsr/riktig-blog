Coq's Q, the rationals over Coq's Z.
Instance Q_EqRel : EqRel Q := Qeq.
Instance Q_Zero : Zero Q := 0%Q.
Instance Q_One : One Q := 1%Q.
Instance Q_HasAdd : HasAdd Q := Qplus.
Instance Q_HasMult : HasMult Q := Qmult.
Instance Q_HasNeg : HasNeg Q := Qopp.
Instance Q_HasInv : HasInv Q := Qinv.

  {| Magma.equivalence := Q_Setoid; Magma.op_proper := Qplus_comp |}.
Instance Q_add_Semigroup : Semigroup Q Q_HasAdd :=
  {| Semigroup.magma := Q_add_Magma;
     Semigroup.is_associative := fun a b c ⇒ Qeq_sym _ _ (Qplus_assoc a b c) |}.
Instance Q_add_Monoid : Monoid Q Q_Zero Q_HasAdd :=
  {| Monoid.semigroup := Q_add_Semigroup;
     Monoid.is_left_identity := Qplus_0_l; Monoid.is_right_identity := Qplus_0_r |}.
Instance Q_Group : Group Q Q_Zero Q_HasAdd Q_HasNeg :=
  {| Group.monoid := Q_add_Monoid; Group.neg_proper := Qopp_comp;
     Group.is_left_inverse := Qplus_opp_l; Group.is_right_inverse := Qplus_opp_r |}.
Instance Q_AbGroup : AbGroup Q Q_Zero Q_HasAdd Q_HasNeg :=
  {| AbGroup.group := Q_Group; AbGroup.is_commutative := Qplus_comm |}.

Instance Q_mul_Magma : Magma Q Q_HasMult :=
  {| Magma.equivalence := Q_Setoid; Magma.op_proper := Qmult_comp |}.
Instance Q_mul_Semigroup : Semigroup Q Q_HasMult :=
  {| Semigroup.magma := Q_mul_Magma;
     Semigroup.is_associative := fun a b c ⇒ Qeq_sym _ _ (Qmult_assoc a b c) |}.
Instance Q_mul_Monoid : Monoid Q Q_One Q_HasMult :=
  {| Monoid.semigroup := Q_mul_Semigroup;
     Monoid.is_left_identity := Qmult_1_l; Monoid.is_right_identity := Qmult_1_r |}.

Instance Q_Ring : Ring Q Q_Zero Q_HasAdd Q_HasNeg Q_One Q_HasMult :=
  {| Ring.abgroup := Q_AbGroup; Ring.mul_monoid := Q_mul_Monoid;
     Ring.is_left_distributive := Qmult_plus_distr_r;
     Ring.is_right_distributive := Qmult_plus_distr_l |}.
Instance Q_Field : Field Q Q_Zero Q_HasAdd Q_HasNeg Q_One Q_HasMult Q_HasInv :=
  {| Field.ring := Q_Ring; Field.inv_proper := Qinv_comp;

     Field.is_mul_left_inverse := Qmult_inv_l;
