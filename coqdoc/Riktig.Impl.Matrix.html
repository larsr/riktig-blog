Definition mat_mul {n : nat} {J F : Type} {eJ : EqRel J} {dJ : EquDec J}
    {FJ : FinIndex n J} {zF : Zero F} {aF : HasAdd F} {mF : HasMult F}
    {I K : Type} (A : I → J → F) (B : J → K → F) (i : I) (k : K) : F :=
  bigopI (fun j ⇒ A i j × B j k).

Instance MatMul_bigop {n : nat} {J F : Type} {eJ : EqRel J} {dJ : EquDec J}
    {FJ : FinIndex n J} {eF : EqRel F} {RF : Reflexive (@equ F eF)}
    {zF : Zero F} {aF : HasAdd F} {mF : HasMult F} :
  MatMul n J F (@bigop_list F _ _) (@mat_mul n J F eJ dJ FJ zF aF mF) :=
  fun I K A B i k ⇒ reflexivity (bigopI (fun j ⇒ A i j × B j k)).

    @mat_mul n (Fin.t n) F _ _ _ _ _ _ (Fin.t n) (Fin.t n).

    PW_AbGroup.

  #[export] Instance matrix_mul_proper (n : nat) :
    Proper (equ ==> equ ==> equ) (@mult (Matrix n n F) (Matrix_Mult n)) :=
    mat_mul_proper.

  #[export] Instance matrix_mul_associative (n : nat) :
    Associative (Matrix n n F) (Matrix n n F) mult mult :=
    fun A B C ⇒ mat_mul_assoc A B C.

  #[export] Instance matrix_mul_left_identity (n : nat) :
    LeftIdentity (Matrix n n F) (Matrix n n F) mult 1 :=
    fun A ⇒ mat_mul_left_identity A (leibniz_proper_first A).

    RightIdentity (Matrix n n F) (Matrix n n F) mult 1 :=
    fun A ⇒ mat_mul_right_identity A (leibniz_proper_second A).

Section MatMulRec.
  Context {F : Type} `{FF : Field F}.

  Fixpoint mat_mul_rec (k : nat) {I K : Type} {struct k}
      : (I → Ix k → F) → (Ix k → K → F) → I → K → F :=
    match k as k0 return
      (I → Ix k0 → F) → (Ix k0 → K → F) → I → K → F
    with
    | O ⇒ fun A B i c ⇒ A i tt × B tt c
    | S k' ⇒ fun A B ⇒
        mat_mul_rec k' (fun i j ⇒ A i (inl j)) (fun j c ⇒ B (inl j) c)
        + mat_mul_rec k' (fun i j ⇒ A i (inr j)) (fun j c ⇒ B (inr j) c)
    end.

  Theorem mat_mul_rec_correct : ∀ (k : nat) (I K : Type)
      (A : I → Ix k → F) (B : Ix k → K → F),
    mat_mul_rec k A B == mat_mul A B.
  Proof.
    intros k; induction k as [| k' IH]; intros I K A B i c.
    - calc (mat_mul_rec O A B i c)
        == (A i tt × B tt c + (0 : F))
             { right_identity }
        == (mat_mul A B i c)
             { (mat_mul_def A B i c) }.
    - calc (mat_mul_rec (S k') A B i c)
        == ((mat_mul (fun i' j ⇒ A i' (inl j)) (fun j c' ⇒ B (inl j) c')
             + mat_mul (fun i' j ⇒ A i' (inr j)) (fun j c' ⇒ B (inr j) c')) i c)
             { (op_proper _ _ (IH I K (fun i' j ⇒ A i' (inl j))
                                 (fun j c' ⇒ B (inl j) c') i c)
                          _ _ (IH I K (fun i' j ⇒ A i' (inr j))
                                 (fun j c' ⇒ B (inr j) c') i c)) }
        == (mat_mul A B i c)
             { (mat_mul_split A B i c) }.
  Qed.

End MatMulRec.

Definition row2 {A : Type} (a b : A) : Fin.t 2 → A :=
  fun j ⇒ match j with
           | Fin.F1 ⇒ a
           | Fin.FS _ ⇒ b
           end.

Definition mat2 {A : Type} (a b c d : A) : Matrix 2 2 A :=
  fun i ⇒ match i with
           | Fin.F1 ⇒ row2 a b
           | Fin.FS _ ⇒ row2 c d
           end.

Definition MA : Matrix 2 2 Q := mat2 (1#1)%Q (2#1)%Q (3#1)%Q (4#1)%Q.
Definition MB : Matrix 2 2 Q := mat2 (5#1)%Q (6#1)%Q (7#1)%Q (8#1)%Q.

Example mat_mul_2x2_entries :
     mat_mul MA MB Fin.F1 Fin.F1 == (19#1)%Q
  ∧ mat_mul MA MB Fin.F1 (Fin.FS Fin.F1) == (22#1)%Q
  ∧ mat_mul MA MB (Fin.FS Fin.F1) Fin.F1 == (43#1)%Q
  ∧ mat_mul MA MB (Fin.FS Fin.F1) (Fin.FS Fin.F1) == (50#1)%Q.
Proof. repeat split; vm_compute; reflexivity. Qed.

Example mat_mul_2x2 :
  mat_mul MA MB == mat2 (19#1)%Q (22#1)%Q (43#1)%Q (50#1)%Q.
Proof.
  intros i k; revert k; revert i.
  refine (fin2_case _ _ _); [ | intros i1 ]; refine (fin2_case _ _ _);
    intros; vm_compute; reflexivity.
Qed.


Definition ix2 {A : Type} (a b c d : A) : Ix 1 → Ix 1 → A :=
  fun i ⇒ match i with
           | inl _ ⇒ fun j ⇒ match j with inl _ ⇒ a | inr _ ⇒ b end
           | inr _ ⇒ fun j ⇒ match j with inl _ ⇒ c | inr _ ⇒ d end
           end.

Example mat_mul_rec_2x2 :
  mat_mul_rec 1 (ix2 (1#1)%Q (2#1)%Q (3#1)%Q (4#1)%Q)
                (ix2 (5#1)%Q (6#1)%Q (7#1)%Q (8#1)%Q)
    == ix2 (19#1)%Q (22#1)%Q (43#1)%Q (50#1)%Q.
Proof.
  intros i j.
  destruct i as [[] | []]; destruct j as [[] | []]; vm_compute; reflexivity.
Qed.

Example mat_mul_ix2 :
  mat_mul (ix2 (1#1)%Q (2#1)%Q (3#1)%Q (4#1)%Q)
          (ix2 (5#1)%Q (6#1)%Q (7#1)%Q (8#1)%Q)
    == ix2 (19#1)%Q (22#1)%Q (43#1)%Q (50#1)%Q.
Proof.
  intros i j.
  calc (mat_mul (ix2 (1#1)%Q (2#1)%Q (3#1)%Q (4#1)%Q)
                (ix2 (5#1)%Q (6#1)%Q (7#1)%Q (8#1)%Q) i j)
    == (mat_mul_rec 1 (ix2 (1#1)%Q (2#1)%Q (3#1)%Q (4#1)%Q)
                      (ix2 (5#1)%Q (6#1)%Q (7#1)%Q (8#1)%Q) i j)
         { (mat_mul_rec_correct 1 (Ix 1) (Ix 1)
              (ix2 (1#1)%Q (2#1)%Q (3#1)%Q (4#1)%Q)
              (ix2 (5#1)%Q (6#1)%Q (7#1)%Q (8#1)%Q) i j) }
    == (ix2 (19#1)%Q (22#1)%Q (43#1)%Q (50#1)%Q i j)
         { (mat_mul_rec_2x2 i j) }.
Qed.
